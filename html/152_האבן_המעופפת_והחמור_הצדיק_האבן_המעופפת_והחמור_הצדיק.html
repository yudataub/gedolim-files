🔹 שָׁלָב 1: בֵּרוּר מַקְדִּים וְנִתּוּחַ (לַמּוֹרֶה/לַמְסַפֵּר)

🔹 שָׁלָב 2: הַסִּפּוּר הַמָּלֵא – "הָאֶבֶן הַמְּעוֹפֶפֶת וְהַחֲמוֹר הַצַּדִּיק"

פְּתִיחָה: הַקִּנְאָה הַיְּחִידָה שֶׁמֻּתֶּרֶת יְלָדִים, אֲנַחְנוּ מַכִּירִים אֶת הַצַּדִּיק הֶעָנִי וְהַקָּדוֹשׁ מִכֻּלָּם – רַבִּי חֲנִינָא בֶּן דּוֹסָא. אָדָם שֶׁ"מִּסְתַּפֵּק בְּקַב שֶׁל חֲרוּבִים". הוּא מֵעוֹלָם, אֲבָל מֵעוֹלָם, לֹא קִנֵּא בַּשְּׁכֵנִים שֶׁלּוֹ. כְּשֶׁהוּא רָאָה מִישֶׁהוּ עִם כִּרְכָּרָה חֲדָשָׁה, אוֹ בְּגָדִים יְקָרִים וַאֲרוּחוֹת שֶׁל בָּשָׂר – הוּא חִיֵּךְ וְשָׂמַח בְּחֶלְקוֹ. אֲבָל... הָיָה דָּבָר אֶחָד, רַק דָּבָר אֶחָד בָּעוֹלָם, שֶׁכְּשֶׁרַבִּי חֲנִינָא רָאָה אוֹתוֹ – הַלֵּב שֶׁלּוֹ בָּעַר מִקִּנְאָה טְהוֹרָה!

זֶה קָרָה לִקְרַאת הַחַגִּים, כַּאֲשֶׁר עַם יִשְׂרָאֵל עָלָה לִירוּשָׁלַיִם. רַבִּי חֲנִינָא הָלַךְ בָּרְחוֹב, וְרָאָה אֶת הַשְּׁכֵנִים שֶׁלּוֹ מַעֲמִיסִים עֲגָלוֹת: זֶה מֵבִיא עֲגָלָה מְלֵאָה תְּאֵנִים וְרִמּוֹנִים כְּבִכּוּרִים, זֶה מֵבִיא גְּבִיעֵי כֶּסֶף לְבֵית הַמִּקְדָּשׁ, וְזֶה לוֹקֵחַ פָּרוֹת שְׁמֵנוֹת לְקָרְבָּנוֹת. רַבִּי חֲנִינָא הִבִּיט לַשָּׁמַיִם, וּדְמָעוֹת קְדוֹשׁוֹת עָמְדוּ בְּעֵינָיו. הוּא לָחַשׁ: "רִבּוֹנוֹ שֶׁל עוֹלָם! כֻּלָּם אוֹהֲבִים אוֹתְךָ וּמְבִיאִים מַתָּנוֹת יָפוֹת לְבֵית הַמִּקְדָּשׁ... וְרַק אֲנִי – אֵין לִי כְּלוּם לָתֵת לְךָ!". זֹאת נִקְרֵאת "קִנְאַת סוֹפְרִים". אֲנִי לֹא מְקַנֵּא בַּכֶּסֶף שֶׁלָּהֶם, אֲנִי מְקַנֵּא בַּמִּצְווֹת שֶׁלָּהֶם!

הָאֶבֶן הַיָּפָה בַּמִּדְבָּר הֶחְלִיט רַבִּי חֲנִינָא שֶׁהוּא יִמְצָא מַשֶּׁהוּ! הוּא יָצָא אֶל מִחוּץ לָעִיר, שָׁם הָיָה שְׂדֵה טְרָשִׁים מָלֵא אֲבָנִים וְחוֹל. וְהִנֵּה, תַּחַת קַרְנֵי הַשֶּׁמֶשׁ, הוּא רָאָה אֶבֶן עֲנָקִית, יָפָה מְאֹד. אֶבֶן חֲלָקָה וּמְיֻחֶדֶת שֶׁהִבְרִיקָה כְּמוֹ שַׁיִשׁ! "הִנֵּה הַמַּתָּנָה שֶׁלִּי לְבֵית הַמִּקְדָּשׁ!" חִיֵּךְ רַבִּי חֲנִינָא בְּשִׂמְחָה שֶׁל יֶלֶד קָטָן שֶׁמָּצָא אוֹצָר. הוּא לָקַח מַכּוֹשׁ וְאִזְמֵל (כְּלֵי עֲבוֹדָה שֶׁל אַבְנַיִם), וְהִתְחִיל לְסַתֵּת אֶת הָאֶבֶן. דְּפִיקָה אַחַר דְּפִיקָה: טַק, טַק, טַק. הוּא שִׁיֵּף וְהֶחְלִיק אוֹתָהּ, עַד שֶׁהִיא נִרְאֲתָה רְאוּיָה לְכַסֵּא הַמֶּלֶךְ! אֲבָל... עַכְשָׁו, אֵיךְ נִקַּח אוֹתָהּ לִירוּשָׁלַיִם? הָאֶבֶן שָׁקְלָה מֵאוֹת קִילוֹגְרַמִּים! רַבִּי חֲנִינָא נִסָּה לְהָרִים... אוֹף, אִי אֶפְשָׁר בִּכְלָל לְהָזִיז אוֹתָהּ. וְאֵין לוֹ כֶּסֶף לִשְׂכֹּר קַבְּלָנִים עִם עֲגָלוֹת. מָה עוֹשִׂים?

הַסַּבָּלִים הַמִּסְתּוֹרִיִּים לְפֶתַע, הוֹפִיעוּ בַּשָּׂדֶה חֲמִשָּׁה אֲנָשִׁים שְׁרִירִיִּים וְחֲזָקִים שֶׁלֹּא רָאָה מֵעוֹלָם. "הָרַב חֲנִינָא!" הֵם קָרְאוּ, "תִּרְצֶה שֶׁנָּרִים לְךָ אֶת הָאֶבֶן לִירוּשָׁלַיִם?" "בְּוַדַּאי!" קָפַץ רַבִּי חֲנִינָא, "אֲבָל... כַּמָּה תִּרְצוּ עַל הָעֲבוֹדָה הַקָּשָׁה הַזּוֹ?" "חֲמִשָּׁה סְלָעִים!" עָנוּ הַסַּבָּלִים. רַבִּי חֲנִינָא חִשֵּׁב בְּרֹאשׁוֹ – כֵּן, זֶה סְכוּם שֶׁהוּא יָכוֹל לְהַשִּׂיג לְאַט לְאַט. הוּא הִסְכִּים בְּשִׂמְחָה. "אֲבָל... יֵשׁ לָנוּ תְּנַאי אֶחָד," הִבִּיטוּ בּוֹ הַפּוֹעֲלִים בְּמַבָּט עָמֹק. "אֲנַחְנוּ נִסְחַב אֶת הָאֶבֶן, אַךְ אַתָּה חַיָּב לְהַנִּיחַ אֶת הָאֶצְבַּע שֶׁלְּךָ עַל הָאֶבֶן תּוֹךְ כְּדֵי לִכְתֵּנוּ!". מַדּוּעַ? כִּי מִצְוָה, יְלָדִים, אָסוּר שֶׁאֲחֵרִים יַעֲשׂוּ רַק בִּשְׁבִילְךָ – אַתָּה חַיָּב לְהִשְׁתַּתֵּף וּלְהַשְׁקִיעַ מַאֲמָץ!

רַבִּי חֲנִינָא הִנִּיחַ אֶת אֶצְבָּעוֹ הַדַּקָּה עַל הָאֶבֶן. הַסַּבָּלִים אוֹחֲזִים מִלְּמַטָּה, וְאָז – שְׁששׁשווּק! – רוּחַ גְּדוֹלָה מֵרִימָה אֶת הָאֶבֶן! בִּשְׁנִיָּה אַחַת שֶׁל מַעֲשֵׂה קְסָמִים... רַגְלָיו שֶׁל רַבִּי חֲנִינָא נָחֲתוּ עַל רִצְפַּת אֶבֶן חֲלָקָה וּקְדוֹשָׁה. הוּא הִבִּיט מִסָּבִיב – הוּא עוֹמֵד יָשָׁר בְּתוֹךְ בֵּית הַמִּקְדָּשׁ שֶׁבִּירוּשָׁלַיִם! בִּקְפִיצַת הַדֶּרֶךְ! הוּא הִסְתּוֹבֵב כְּדֵי לְשַׁלֵּם לַסַּבָּלִים, אֲבָל... הֵם נֶעֶלְמוּ. כְּלֹא הָיוּ. נִגַּשׁ רַבִּי חֲנִינָא לְבֵית הַדִּין שֶׁל הַכֹּהֲנִים וְסִפֵּר שֶׁסַּבָּלִים נֶעֶלְמוּ וְהוּא חַיָּב לָהֶם כֶּסֶף. אָמְרוּ לוֹ הַחֲכָמִים: "רַבִּי חֲנִינָא, לֹא יָדַעְתָּ? מַלְאֲכֵי הַשָּׁרֵת מִן הַשָּׁמַיִם הֵם אֵלּוּ שֶׁנָּשְׂאוּ אֶת הָאֶבֶן שֶׁלְּךָ! הַקָּדוֹשׁ בָּרוּךְ הוּא רָאָה אֶת הַלֵּב הַטָּהוֹר שֶׁלְּךָ שֶׁבּוֹעֵר מִשִּׂמְחַת מִצְוָה, וְשָׁלַח אֶת מַלְאָכָיו לַעֲזֹר לְךָ!".

הַחֲמוֹר שֶׁהֶחְלִיט לָצוּם קְדֻשָּׁתוֹ שֶׁל רַבִּי חֲנִינָא בֶּן דּוֹסָא הָיְתָה כָּל כָּךְ עֲצוּמָה, יְלָדִים, שֶׁהִיא נִדְבְּקָה לֹא רַק בַּמַּלְאָכִים – אֶלָּא אֲפִלּוּ בְּבַעֲלֵי הַחַיִּים שֶׁהָיוּ אֶצְלוֹ בֶּחָצֵר! לְרַבִּי חֲנִינָא הָיָה חֲמוֹר אֶחָד. חֲמוֹר רָגִיל מַמָּשׁ. אוֹ שֶׁלֹּא... לַיְלָה אֶחָד, הִתְגַּנְּבוּ הַבַּיְתָה לִסְטִים (גַּנָּבִים) רְשָׁעִים, שָׁבְרוּ אֶת מַנְעוּל דֶּלֶת הַחָצֵר, וְגָנְבוּ אֶת הַחֲמוֹר! הַגַּנָּבִים הוֹבִילוּ אֶת הַחֲמוֹר הַמִּסְכֵּן הַרְחֵק לְהָרִים, כָּלְאוּ אוֹתוֹ בֶּחָצֵר סְגוּרָה שֶׁלָּהֶם, וְשָׂמוּ מִמּוּלוֹ שַׂק עֲנָק מָלֵא בִּתְבוּאָה עֲסִיסִית (שְׂעוֹרִים טְעִימוֹת) וְדוּד שֶׁל מַיִם.

אַךְ תָּאֲרוּ לָכֶם מָה קָרָה: הַחֲמוֹר... עָצַם אֶת הַפֶּה וְסֵרֵב לֶאֱכֹל! הַגַּנָּבִים מִשְּׁשׁוּ לוֹ בַּבֶּטֶן, דָּחֲפוּ לוֹ אֶת הָאֹכֶל לַפֶּה, אֲבָל כְּלוּם. יוֹם עָבַר – הוּא לֹא אוֹכֵל. יוֹמַיִם. שְׁלוֹשָׁה יָמִים לְלֹא מַיִם וּלְלֹא מָזוֹן! אָמְרוּ הַגַּנָּבִים אֶחָד לַשֵּׁנִי: "חֲבֵרִים, הַחֲמוֹר הַזֶּה הוֹלֵךְ לָמוּת לָנוּ כָּאן. וְהוּא סְתָם יְלַכְלֵךְ אֶת הֶחָצֵר שֶׁלָּנוּ וְיַסְרִיחַ מְאֹד. בּוֹאוּ נִפְתַּח אֶת הַדֶּלֶת וְנָעִיף אוֹתוֹ הַחוּצָה לָמוּת בַּיַּעַר".

פָּתְחוּ הַגַּנָּבִים אֶת דַּלְתוֹת הַחָצֵר. אֲבָל הַחֲמוֹר... קָם עַל רַגְלָיו הַחַלָּשׁוֹת. הוּא הִתְחִיל לִצְעֹד לְאַט-לְאַט, "מוֹשֵׁכֶת וְהוֹלֶכֶת", עַד שֶׁ... הִגִּיעַ לַשַּׁעַר שֶׁל רַבִּי חֲנִינָא בֶּן דּוֹסָא! הַחֲמוֹר נָעַר קְצָת בְּקוֹל עָיֵף: "אִי-אָה... אִי-אָה...".

סוֹדוֹ שֶׁל הַחֲמוֹר וְהַקְּדֻשָּׁה בְּתוֹךְ הַבַּיִת, בְּנוֹ שֶׁל רַבִּי חֲנִינָא שָׁמַע אֶת הַקּוֹל. "אַבָּא!" הוּא צָעַק, "הַקּוֹל הַזֶּה מַמָּשׁ נִשְׁמַע כְּמוֹ קוֹלָהּ שֶׁל הַבְּהֵמָה הָאֲהוּבָה שֶׁלָּנוּ שֶׁנִּגְנְבָה!". שָׂמַח רַבִּי חֲנִינָא בְּהִתְרַגְּשׁוּת וְאָמַר: "רוּץ לִפְתֹּחַ אֶת הַדֶּלֶת! הִיא בֶּטַח כִּמְעַט מֵתָה מֵרָעָב!". רָץ הַבֵּן וּפָתַח אֶת הַשַּׁעַר, הַחֲמוֹר דִּדָּה פְּנִימָה. הַבֵּן רָץ מַהֵר לְמַחְסַן הַתְּבוּאָה שֶׁל הַמִּשְׁפָּחָה, שָׁפַךְ מַיִם קָרִים לַקְּעָרָה וְהִנִּיחַ שַׂק עֲנָק שֶׁל שְׂעוֹרִים טְעִימוֹת מוּל הַחֲמוֹר. וְלְפֶלֶא... הַחֲמוֹר שׁוּב עָצַם שְׂפָתַיִם וְסֵרֵב לֶאֱכֹל!

ג. אילו האבן המעופפת והחמור הצדיק לא היה פועל כפי שמתואר ב־“שָׂמַח רַבִּי חֲנִינָא בְּהִתְרַגְּשׁוּת וְאָמַר: "רוּץ לִפְתֹּחַ אֶת הַדֶּלֶת!” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “הַבֵּן מִהֵר, הוֹצִיא אֶת הַמַּעֲשֵׂר, וּמִיָּד – הַחֲמוֹר פָּקַח אֶת עֵינָיו, נִגַּשׁ לַקְּעָרָה וְאָכַל מָנָה עֲנָקִית…” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. 🔹 שָׁלָב 2: הַסִּפּוּר הַמָּלֵא – "הָאֶבֶן הַמְּעוֹפֶפֶת וְהַחֲמוֹר…
2. הוּא יָצָא אֶל מִחוּץ לָעִיר, שָׁם הָיָה שְׂדֵה טְרָשִׁים מָלֵא…
3. רַבִּי חֲנִינָא הִנִּיחַ אֶת אֶצְבָּעוֹ הַדַּקָּה עַל הָאֶבֶן.
4. הַגַּנָּבִים מִשְּׁשׁוּ לוֹ בַּבֶּטֶן, דָּחֲפוּ לוֹ אֶת הָאֹכֶל לַפֶּה,…
5. נִגַּשׁ רַבִּי חֲנִינָא אֶל קְעָרַת הָאֹכֶל, הִבִּיט בַּשְּׂעוֹרִים…
6. בּוֹאוּ נִרְאֶה וְנִלְמַד עַל נֵס הָאֶבֶן וְהַחֲמוֹר הַמֻּפְלָא!

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “אֲנַחְנוּ נִסְחַב אֶת הָאֶבֶן, אַךְ אַתָּה חַיָּב לְהַנִּיחַ אֶת הָאֶצְבַּע שֶׁלְּךָ עַל הָאֶבֶן תּוֹךְ כְּדֵי לִכְתֵּנוּ!".”.
מכתב: כתבו להאבן המעופפת והחמור הצדיק בעקבות “וְהוּא סְתָם יְלַכְלֵךְ אֶת הֶחָצֵר שֶׁלָּנוּ וְיַסְרִיחַ מְאֹד.”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “לִפְנֵי הַמַּעֲשֵׂר, הָאֹכֶל נִקְרָא "טֶבֶל" וְאָסוּר לֶאֱכֹל אוֹתוֹ).”.
דיווח: כתבו ידיעה קצרה שכותרתה “האבן המעופפת והחמור הצדיק”.

המחזה

רגע מן הסיפורהדמויותמה יאמרו או יעשו?
הוּא לָקַח מַכּוֹשׁ וְאִזְמֵל (כְּלֵי עֲבוֹדָה שֶׁל אַבְנַיִם), וְהִתְחִיל לְסַתֵּת אֶת הָאֶבֶן.
שָׂמַח רַבִּי חֲנִינָא בְּהִתְרַגְּשׁוּת וְאָמַר: "רוּץ לִפְתֹּחַ אֶת הַדֶּלֶת!

7. חיבור לחיים מתוך המעשה

א. מה אפשר ללמוד מן הדרך שבה האבן המעופפת והחמור הצדיק פעל באירוע “לַיְלָה אֶחָד, הִתְגַּנְּבוּ הַבַּיְתָה לִסְטִים (גַּנָּבִים) רְשָׁעִים, שָׁבְרוּ אֶת מַנְעוּל דֶּלֶת הַחָצֵר, וְגָנְבוּ אֶת…”?
