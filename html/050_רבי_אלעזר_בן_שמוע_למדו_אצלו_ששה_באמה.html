סיפור 50 • רבי אלעזר בן שמוע • למדו אצלו ששה באמה

למדו אצלו ששה באמה

דף עבודה בעקבות רבי אלעזר בן שמוע
שם: כיתה:

המעשה — הסיפור המלא

פְּתִיחָה: הַשֶּׁקֶט הֶעָצוּב שֶׁל אַחֲרֵי הַסְּעָרָה

יְלָדִים, אֲנַחְנוּ חוֹזְרִים בַּזְּמַן לְתְקוּפָה קָשָׁה מְאֹד בְּאֶרֶץ יִשְׂרָאֵל. תַּעֲצְמוּ רֶגַע עֵינַיִם וְתַקְשִׁיבוּ. מָה אַתֶּם שׁוֹמְעִים?

שֶׁקֶט. שֶׁקֶט מַפְחִיד.

לִפְנֵי כַּמָּה שָׁנִים קָרָה אָסוֹן נוֹרָא. עֶשְׂרִים וְאַרְבָּעָה אֶלֶף תַּלְמִידִים שֶׁל הַתַּנָּא הַקָּדוֹשׁ רַבִּי עֲקִיבָא נִפְטְרוּ מִן הָעוֹלָם. בָּתֵּי הַמִּדְרָשׁ הָעֲנָקִיִּים שֶׁהָיוּ מְלֵאִים בְּקוֹלוֹת שֶׁל שִׁירַת הַתּוֹרָה, עָמְדוּ רֵיקִים. הַשֶּׁמֶשׁ הַחַמָּה קָפְחָה עַל הָאֲדָמָה הַיְבֵשָׁה, וְרֵיחַ שֶׁל אָבָק עָלָה מִן הַדְּרָכִים הַשּׁוֹמְמוֹת.

אֲנָשִׁים הָלְכוּ בָּרְחוֹבוֹת וּבָכוּ: "מָה יִהְיֶה עַל תּוֹרַת יִשְׂרָאֵל? הַאִם חָלִילָה נִשְׁכַּח אֶת דְּבַר ה'? הַאִם הָרוֹמָאִים הָרְשָׁעִים שֶׁהֶחְרִיבוּ אֶת בֵּית הַמִּקְדָּשׁ, יַצְלִיחוּ לְכַבּוֹת גַּם אֶת הָאוֹר שֶׁל הַתּוֹרָה?"

אֲבָל בְּתוֹךְ כָּל הַחֹשֶׁךְ הַזֶּה, הַקָּדוֹשׁ בָּרוּךְ הוּא, בְּהַשְׁגָּחָתוֹ הַפְּרָטִית וְהָאוֹהֶבֶת, דָּאַג שֶׁהָאוֹר יַמְשִׁיךְ לִדְלֹק.

רַבִּי עֲקִיבָא הַזָּקֵן הָלַךְ לַדָּרוֹם, וְלִמֵּד שָׁם חֲמִשָּׁה תַּלְמִידִים חֲדָשִׁים וּגְאוֹנִים. הַתַּלְמִידִים הָאֵלֶּה הֵם שֶׁהֵאִירוּ אֶת הָעוֹלָם מֵחָדָשׁ. אֶחָד מֵהֶם, תַּלְמִיד חָכָם וְקָדוֹשׁ בְּיוֹתֵר, הָיָה רַבִּי אֶלְעָזָר בֶּן שַׁמּוּעַ .

הַצָּמָא הַגָּדוֹל לִדְבַר ה'

רַבִּי אֶלְעָזָר בֶּן שַׁמּוּעַ לֹא פָּחַד. אַף עַל פִּי שֶׁהָרוֹמָאִים אָסְרוּ לִלְמֹד תּוֹרָה, הוּא אָסַף אֵלָיו תַּלְמִידִים. הוּא פָּתַח יְשִׁיבָה. אַךְ זֶה לֹא הָיָה בִּנְיָן מְפֹאָר מִשַּׁיִשׁ כְּמוֹ שֶׁאֲנַחְנוּ רוֹאִים הַיּוֹם. זֶה הָיָה חֶדֶר צַר. אוּלַי מְעָרָה גְּדוֹלָה אוֹ מִבְנֶה אֶבֶן יָשָׁן. קִירוֹת הָאֶבֶן הָיוּ קָרִים, אֲבָל בִּפְנִים – בָּעֲרָה אֵשׁ חַמָּה! אֵשׁ הַתּוֹרָה!

וְהִנֵּה, מַשֶּׁהוּ מַדְהִים קָרָה.

שְׁמוּעָה עָבְרָה מִכְּפָר לִכְפָר, מֵעִיר לְעִיר: "רַבִּי אֶלְעָזָר בֶּן שַׁמּוּעַ מְלַמֵּד אֶת הַתּוֹרָה שֶׁל רַבִּי עֲקִיבָא!"

בַּחוּרִים צְעִירִים, שֶׁהָיוּ צְמֵאִים לְדִבְרֵי תּוֹרָה כְּמוֹ אֲדָמָה שֶׁצְּמֵאָה לְגֶשֶׁם אַחֲרֵי קַיִץ אָרֹךְ, עָזְבוּ אֶת הַבָּתִּים שֶׁלָּהֶם. הֵם אָרְזוּ בְּשַׂקִּית קְטַנָּה כִּכַּר לֶחֶם יְבֵשָׁה וּמְעַט מַיִם, וְהִתְחִילוּ לָלֶכֶת בָּרֶגֶל.

עוֹד תַּלְמִיד הִגִּיעַ לַפֶּתַח... וְעוֹד אֶחָד... וְעוֹד עֲשָׂרָה.

הַחֶדֶר הַקָּטָן שֶׁל רַבִּי אֶלְעָזָר הִתְחִיל לְהִתְמַלֵּא.

הַדֹּחַק בְּבֵית הַמִּדְרָשׁ הַצַּר

אֶחָד מֵהַנְּעָרִים שֶׁהִגִּיעוּ הָיָה נַעַר צָעִיר, חָכָם וּטְהוֹר עֵינַיִם, שֶׁקָּרְאוּ לוֹ יְהוּדָה (זֶהוּ מִי שֶׁיִּגְדַּל וְיִהְיֶה 'רַבִּי יְהוּדָה הַנָּשִׂיא', כּוֹתֵב הַמִּשְׁנָה).

יְהוּדָה הַצָּעִיר נִכְנַס לְבֵית הַמִּדְרָשׁ, וְנֶעֱמַד נִדְהָם. הוּא כִּמְעַט לֹא הִצְלִיחַ לְהַכְנִיס אֶת כַּף רַגְלוֹ פְּנִימָה!

הַחֶדֶר הָיָה דָּחוּס וְצָפוּף. אִי אֶפְשָׁר הָיָה לְהָזִיז כִּסֵּא, אִי אֶפְשָׁר הָיָה לִפְרֹשׂ אֶת הַיָּדַיִם, וּבְקֹשִׁי אֶפְשָׁר הָיָה לִנְשֹׁם מִשֶּׁפַע הָאֲנָשִׁים. מֵרֹב תַּלְמִידִים גְּדוֹלִים שֶׁבָּאוּ לִשְׁמֹעַ הֲלָכוֹת, הַצְּעִירִים נִדְחֲקוּ.

רַבִּי אֶלְעָזָר בֶּן שַׁמּוּעַ יָשַׁב בִּקְצֵה הַחֶדֶר. עֵינָיו הֵאִירוּ כְּכוֹכָבִים מִבַּעַד לְזַקְנוֹ. כְּשֶׁהוּא פָּתַח אֶת פִּיו, הַשֶּׁקֶט שֶׁהִשְׁתָּרֵר הָיָה מֻחְלָט. קוֹלוֹ הָיָה עָמֹק וּמָתוֹק כִּדְבַשׁ. כָּל מִלָּה שֶׁאָמַר יָרְדָה יָשָׁר לַלֵּב שֶׁל הַתַּלְמִידִים.

הוּא לִמֵּד אוֹתָם סוֹדוֹת נִפְלָאִים מֵהַתּוֹרָה, הֲלָכוֹת כֵּיצַד לִשְׁמֹר שַׁבָּת, כֵּיצַד לְכַבֵּד אֶת הַהוֹרִים, וְאֵיךְ נִרְאָה בֵּית הַמִּקְדָּשׁ כְּשֶׁהָיָה קַיָּם.

שִׁשָּׁה בְּאַמָּה אַחַת

אַךְ הַצְּפִיפוּת... אוֹי, אֵיזוֹ צְפִיפוּת זוֹ הָיְתָה!

יְלָדִים, אַתֶּם תּוֹפְסִים כַּמָּה זֶה "אַמָּה"? אַמָּה בַּתּוֹרָה הִיא מִדָּה שֶׁל אֹרֶךְ יָד שֶׁל אָדָם מְבֻגָּר (מֵהַמַּרְפֵּק וְעַד לִקְצֵה הָאֶצְבַּע הָאֶמְצָעִית). בְּעֵרֶךְ חֲצִי מֶטֶר! זֶה מֶרְחָב שֶׁבְּקֹשִׁי מַסְפִּיק לְאָדָם אֶחָד לָשֶׁבֶת בְּנוֹחוּת עַל כִּסֵּא.

אֲבָל שָׁם, בְּחַדְרוֹ שֶׁל רַבִּי אֶלְעָזָר, כָּל כָּךְ הַרְבֵּה רָצוּ לִשְׁמֹעַ! לְאַף אֶחָד לֹא הָיָה אִכְפַּת שֶׁאֵין אוֹיר לִנְשֹׁם.

תּאֲרוּ לָכֶם מַצָּב, שֶׁבְּתוֹךְ מֶרְחָב קָטָן כָּזֶה שֶׁל חֲצִי מֶטֶר ("אַמָּה"), אֲנָשִׁים יָשְׁבוּ צְמוּדִים, דְּבוּקִים זֶה לָזֶה, מַמָּשׁ שׁוֹלְבִים אֶת הַזְּרוֹעוֹת אוֹ יוֹשְׁבִים בְּזָוִית כְּדֵי שֶׁכֻּלָּם יִכָּנְסוּ.

רַבִּי יְהוּדָה הַנָּשִׂיא סִפֵּר שָׁנִים רַבּוֹת לְאַחַר מִכֵּן לְתַלְמִידָיו, וּדְמָעוֹת שֶׁל גַּעֲגוּעַ בְּעֵינָיו:

"כְּשֶׁהָיִינוּ לוֹמְדִים תּוֹרָה אֵצֶל רַבִּי אֶלְעָזָר בֶּן שַׁמּוּעַ – הָיִינוּ יוֹשְׁבִים שִׁשָּׁה שִׁשָּׁה בְּאַמָּה!"

שִׁשָּׁה בַּחוּרִים! כָּתֵף צְמוּדָה לְכָתֵף, רֶגֶל נוֹגַעַת בְּרֶגֶל. אֶגְלֵי זֵעָה חַמִּים זָלְגוּ מִמִּצְחָם. הֵם לֹא הָיוּ יְכוֹלִים לָזוּז כִּמְלֹא הַנִּימָה.

אִם מִישֶׁהוּ מֵהֶם הָיָה רָעֵב אוֹ שֶׁהָיָה קָשֶׁה לוֹ וְכָאַב לוֹ הַגַּב – הוּא לֹא אָמַר מִלָּה.

אָדָם רָגִיל שֶׁעוֹמֵד כָּךְ בְּאֹטוֹבּוּס צָפוּף מִיָּד צוֹעֵק: "אִי אֶפְשָׁר כָּכָה! דּוֹחֲפִים אוֹתִי! מָה זֶה הַחֹם הַזֶּה?!"

אֲבָל בְּבֵית הַמִּדְרָשׁ הַזֶּה לֹא שָׁמְעוּ שׁוּם קְלוּנְיָא וְשׁוּם כַּעַס. אַדְרַבָּה! קוֹלוֹת שֶׁל שִׂמְחָה עָלוּ מֵהֶם. כָּל מִלָּה שֶׁהוֹצִיא רַבִּי אֶלְעָזָר בֶּן שַׁמּוּעַ, הֵם בָּלְעוּ בִּצְמָא גָּדוֹל. הַנְּשָׁמָה שֶׁלָּהֶם הִרְגִּישָׁה שֶׁהִיא נִמְצֵאת בְּגַן עֵדֶן הֲכִי פָּתוּחַ וְהֲכִי מְרֻוָּח בָּעוֹלָם!

הָאוֹר שֶׁלֹּא נִכְבָּה לְעוֹלָם

רַבִּי אֶלְעָזָר הִבִּיט עַל הַתַּלְמִידִים הַצְּפוּפִים שֶׁלּוֹ. הוּא רָאָה אֶת הַמַּאֲמָץ הֶעָצוּם שֶׁלָּהֶם. הוּא רָאָה מַהִי מְסִירוּת נֶפֶשׁ אֲמִתִּית לַתּוֹרָה. בְּלִבּוֹ הוּא יָדַע – שֶׁמִּשֵּׁשֶׁת הַנְּעָרִים הָאֵלֶּה הַמִּצְטַפְּפִים בְּאַמָּה אַחַת, תָּקוּם תּוֹרָה גְּדוֹלָה לְכָל עַם יִשְׂרָאֵל, תּוֹרָה שֶׁתַּעֲבֹר מִדּוֹר לְדוֹר, עַד לְיָמֵינוּ אָנוּ!

וְאָכֵן, תּוֹרָתוֹ שֶׁל רַבִּי אֶלְעָזָר בֶּן שַׁמּוּעַ (שֶׁהָיָה לְאַחַר מִכֵּן אֶחָד מֵעֲשֶׂרֶת הֲרוּגֵי מַלְכוּת וּמָסַר נַפְשׁוֹ בְּיָדַיִם הָרוֹמָאִים) הִמְשִׁיכָה לִחְיוֹת לָנֶצַח! רַבִּי יְהוּדָה הַנָּשִׂיא כָּתַב אֶת הַמִּשְׁנָה, וְכָל תַּלְמִיד וְתַלְמִיד מֵהֶם הִפְרִיחַ אֶת הַתּוֹרָה כָּךְ שֶׁאִישׁ כְּבָר לֹא יוּכַל לְכַבּוֹת אוֹתָהּ עוֹד.

מקורות: תלמוד בבלי, מסכת עירובין, דף נ"ג עמוד א': "אמר רבי כשהיינו לומדין תורה אצל רבי אלעזר בן שמוע היינו יושבין ששה ששה באמה". תלמוד בבלי, מסכת יבמות, דף פ"ד עמוד א': תיאור דוחק התלמידים בישיבתו של רבי אלעזר בן שמוע. מדרש האגדה / ספר האגדה, ק"פ, רע"א: על היותו של רבי אלעזר בן שמוע אחד מחמשת התלמידים שסמך רבי יהודה בן בבא לאחר מות תלמידי רבי עקיבא, ואחד מעשרת הרוגי מלכות [מובא גם בהקשר הכללי בספרות החינוכית].

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “יְלָדִים, אֲנַחְנוּ חוֹזְרִים בַּזְּמַן לְתְקוּפָה קָשָׁה מְאֹד בְּאֶרֶץ יִשְׂרָאֵל.” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “אֶחָד מֵהֶם, תַּלְמִיד חָכָם וְקָדוֹשׁ בְּיוֹתֵר, הָיָה רַבִּי אֶלְעָזָר בֶּן שַׁמּוּעַ .”?
ג. כיצד הגיב רבי אלעזר בן שמוע למצב המתואר: “הֵם אָרְזוּ בְּשַׂקִּית קְטַנָּה כִּכַּר לֶחֶם יְבֵשָׁה וּמְעַט מַיִם, וְהִתְחִילוּ לָלֶכֶת בָּרֶגֶל.”?
ד. מה השתנה בין “כָּל מִלָּה שֶׁאָמַר יָרְדָה יָשָׁר לַלֵּב שֶׁל הַתַּלְמִידִים.” לבין “רַבִּי יְהוּדָה הַנָּשִׂיא סִפֵּר שָׁנִים רַבּוֹת לְאַחַר מִכֵּן לְתַלְמִידָיו, וּדְמָעוֹת שֶׁל גַּעֲגוּעַ בְּעֵינָיו:”?
ה. כיצד הגיע הסיפור אל האירוע: “הַנְּשָׁמָה שֶׁלָּהֶם הִרְגִּישָׁה שֶׁהִיא נִמְצֵאת בְּגַן עֵדֶן הֲכִי פָּתוּחַ וְהֲכִי מְרֻוָּח בָּעוֹלָם!”?

3. מה היה ראשון?

כתבו ליד כל אירוע מספר מ־1 עד 5 לפי סדר התרחשותו בסיפור.

□ מֵרֹב תַּלְמִידִים גְּדוֹלִים שֶׁבָּאוּ לִשְׁמֹעַ הֲלָכוֹת, הַצְּעִירִים נִדְחֲקוּ.
□ יְלָדִים, אֲנַחְנוּ חוֹזְרִים בַּזְּמַן לְתְקוּפָה קָשָׁה מְאֹד בְּאֶרֶץ יִשְׂרָאֵל.
□ רַבִּי יְהוּדָה הַנָּשִׂיא כָּתַב אֶת הַמִּשְׁנָה, וְכָל תַּלְמִיד וְתַלְמִיד מֵהֶם הִפְרִיחַ אֶת הַתּוֹרָה כָּךְ שֶׁאִישׁ כְּבָר לֹא יוּכַל לְכַבּוֹת…
□ אַךְ זֶה לֹא הָיָה בִּנְיָן מְפֹאָר מִשַּׁיִשׁ כְּמוֹ שֶׁאֲנַחְנוּ רוֹאִים הַיּוֹם.
□ כְּשֶׁהָיִינוּ לוֹמְדִים תּוֹרָה אֵצֶל רַבִּי אֶלְעָזָר בֶּן שַׁמּוּעַ – הָיִינוּ יוֹשְׁבִים שִׁשָּׁה שִׁשָּׁה בְּאַמָּה!

לפני ואחרי

מה קרה לפני: “מֵרֹב תַּלְמִידִים גְּדוֹלִים שֶׁבָּאוּ לִשְׁמֹעַ הֲלָכוֹת, הַצְּעִירִים נִדְחֲקוּ.”?
מה קרה אחרי: “כְּשֶׁהָיִינוּ לוֹמְדִים תּוֹרָה אֵצֶל רַבִּי אֶלְעָזָר בֶּן שַׁמּוּעַ – הָיִינוּ יוֹשְׁבִים שִׁשָּׁה שִׁשָּׁה בְּאַמָּה!”?

סיבה ותוצאה מתוך המעשה

האירוע הנתוןמה גרם לו?מה הייתה תוצאתו?
קִירוֹת הָאֶבֶן הָיוּ קָרִים, אֲבָל בִּפְנִים – בָּעֲרָה אֵשׁ חַמָּה!
זֶה מֶרְחָב שֶׁבְּקֹשִׁי מַסְפִּיק לְאָדָם אֶחָד לָשֶׁבֶת בְּנוֹחוּת עַל כִּסֵּא.

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “למדו אצלו ששה באמה” על רבי אלעזר בן שמוע? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “אִי אֶפְשָׁר הָיָה לְהָזִיז כִּסֵּא, אִי אֶפְשָׁר הָיָה לִפְרֹשׂ אֶת הַיָּדַיִם, וּבְקֹשִׁי אֶפְשָׁר הָיָה לִנְשֹׁם מִשֶּׁפַע הָאֲנָשִׁים.”.
ג. אילו רבי אלעזר בן שמוע לא היה פועל כפי שמתואר ב־“רַבִּי יְהוּדָה הַנָּשִׂיא סִפֵּר שָׁנִים רַבּוֹת לְאַחַר מִכֵּן לְתַלְמִידָיו, וּדְמָעוֹת שֶׁל גַּעֲגוּעַ בְּעֵינָיו:” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “הַנְּשָׁמָה שֶׁלָּהֶם הִרְגִּישָׁה שֶׁהִיא נִמְצֵאת בְּגַן עֵדֶן הֲכִי פָּתוּחַ וְהֲכִי מְרֻוָּח בָּעוֹלָם!” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. יְלָדִים, אֲנַחְנוּ חוֹזְרִים בַּזְּמַן לְתְקוּפָה קָשָׁה מְאֹד…
2. אַף עַל פִּי שֶׁהָרוֹמָאִים אָסְרוּ לִלְמֹד תּוֹרָה, הוּא אָסַף אֵלָיו…
3. אֶחָד מֵהַנְּעָרִים שֶׁהִגִּיעוּ הָיָה נַעַר צָעִיר, חָכָם וּטְהוֹר…
4. אַמָּה בַּתּוֹרָה הִיא מִדָּה שֶׁל אֹרֶךְ יָד שֶׁל אָדָם מְבֻגָּר…
5. אִם מִישֶׁהוּ מֵהֶם הָיָה רָעֵב אוֹ שֶׁהָיָה קָשֶׁה לוֹ וְכָאַב לוֹ…
6. רַבִּי יְהוּדָה הַנָּשִׂיא כָּתַב אֶת הַמִּשְׁנָה, וְכָל תַּלְמִיד…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “הֵם אָרְזוּ בְּשַׂקִּית קְטַנָּה כִּכַּר לֶחֶם יְבֵשָׁה וּמְעַט מַיִם, וְהִתְחִילוּ לָלֶכֶת בָּרֶגֶל.”.
מכתב: כתבו לרבי אלעזר בן שמוע בעקבות “זֶה מֶרְחָב שֶׁבְּקֹשִׁי מַסְפִּיק לְאָדָם אֶחָד לָשֶׁבֶת בְּנוֹחוּת עַל כִּסֵּא.”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “אָדָם רָגִיל שֶׁעוֹמֵד כָּךְ בְּאֹטוֹבּוּס צָפוּף מִיָּד צוֹעֵק: "אִי אֶפְשָׁר כָּכָה!”.
דיווח: כתבו ידיעה קצרה שכותרתה “למדו אצלו ששה באמה”.
