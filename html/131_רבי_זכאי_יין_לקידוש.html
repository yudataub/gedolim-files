יין לקידוש

דף עבודה בעקבות רבי זכאי
שם: כיתה:

המעשה — הסיפור המלא

פְּתִיחָה: סוֹד אֲרִיכוּת הַיָּמִים

יְלָדִים, בּוֹאוּ נִכָּנֵס בְּדִמְיוֹנֵנוּ לְתוֹךְ בֵּית הַמִּדְרָשׁ הַקָּדוֹשׁ שֶׁל הַתַּנָּאִים. הַשֶּׁקֶט עָטַף אֶת הַקִּירוֹת, אֲבָל בַּלֵּב בָּעֲרָה אֵשׁ שֶׁל לִמּוּד. בְּמֶרְכַּז בֵּית הַמִּדְרָשׁ, יָשַׁב אֶחָד מִגְּדוֹלֵי הַחֲכָמִים, רַבִּי זַכַּאי. רַבִּי זַכַּאי הָיָה יָשִׁישׁ מְאֹד. זְקָנוֹ הָיָה לָבָן כַּשֶּׁלֶג, וּפָנָיו מֵאִירוֹת כְּמַלְאַךְ ה' צְבָאוֹת. שָׁנִים רַבּוֹת הוּא חַי, זָכָה לְהַאֲרִיךְ יָמִים בְּאֹפֶן מֻפְלָא.

בְּאוֹתוֹ יוֹם, נִגְּשׁוּ אֵלָיו תַּלְמִידָיו הַקְּדוֹשִׁים וְשָׁאֲלוּ אוֹתוֹ שְׁאֵלָה שֶׁסִּקְרְנָה אֶת כֻּלָּם: "רַבֵּנוּ, בַּמֶּה הֶאֱרַכְתָּ יָמִים? מָה הַסּוֹד שֶׁלְּךָ שֶׁבִּזְכוּתוֹ הַקָּדוֹשׁ בָּרוּךְ הוּא נָתַן לְךָ חַיִּים אֲרֻכִּים וְטוֹבִים כָּל כָּךְ?"

רַבִּי זַכַּאי שָׁתַק לְרֶגַע. הוּא עָצַם אֶת עֵינָיו, וּלְפֶתַע, דִּמְעָה חַמָּה זָלְגָה עַל לֶחְיוֹ. הוּא נִזְכַּר בְּיַלְדוּתוֹ. הוּא נִזְכַּר בְּאִמּוֹ הַצַּדֶּקֶת. הוּא פָּתַח אֶת עֵינָיו וְאָמַר לְתַלְמִידָיו: "יָמַי הָאֲרֻכִּים לֹא בָּאוּ מֵעַצְמָם... יֵשׁ לִי שְׁלוֹשָׁה דְּבָרִים שֶׁהִקְפַּדְתִּי עֲלֵיהֶם כָּל יָמַי... אַךְ הַדָּבָר שֶׁעָמַד לִי יוֹתֵר מִכֹּל הוּא: מֵעוֹלָם, אֲבָל מֵעוֹלָם – לֹא בִּטַּלְתִּי קִדּוּשׁ שֶׁל שַׁבָּת! וְאַתֶּם יוֹדְעִים לָמָּה? אֲסַפֵּר לָכֶם מָה קָרָה כְּשֶׁהָיִיתִי יֶלֶד".

וְכָךְ הִתְחִיל רַבִּי זַכַּאי לְסַפֵּר.

הַבַּיִת הַקַּר וְהָרֵיק

"כְּשֶׁהָיִיתִי רַךְ בַּשָּׁנִים, אִמִּי וַאֲנִי חַיִּינוּ בְּעֹנִי נּוֹרָא. בַּבַּיִת הַקָּטָן שֶׁלָּנוּ לֹא הָיוּ רָהִיטִים יָפִים, לֹא תַּנּוּר מְחַמֵּם, וּבְוַדַּאי שֶׁלֹּא אֲרוֹנוֹת עֲמוּסִים בְּאֹכֶל. רוּחַ קָרָה נוֹשְׁבָה מִבַּעַד לַחֲרַכִּים שֶׁל חַלּוֹן הָעֵץ הַיָּשָׁן, וְהַבֶּטֶן שֶׁלָּנוּ הָיְתָה מְקַרְקֶרֶת תְּדִיר מֵרָעָב. אַךְ מַשֶּׁהוּ אֶחָד חִמֵּם אֶת הַבַּיִת – אַהֲבַת ה'.

וְהִנֵּה, הִגִּיעַ יוֹם שִׁישִּׁי הַגָּדוֹל. הַשֶּׁמֶשׁ כְּבָר הֵחֵלָּה לָרֶדֶת מַעֲרָבָה. בְּרַחֲבֵי הָעִיר הֵרַחְנוּ אֶת רֵיחוֹת הַתַּבְשִׁילִים שֶׁל כָּל הַשְּׁכֵנִים: רֵיחַ שֶׁל תַּפּוּחֵי אֲדָמָה צְלוּיִים, חַלּוֹת קְלוּעוֹת שֶׁמּוֹצִיאִים עַכְשָׁו מֵהַתַּנּוּר, וְתַבְשִׁיל בָּשָׂר חָם וּמַהְבִּיל. אֲבָל בַּבַּיִת שֶׁלָּנוּ? שֶׁקֶט. אֵין תַּבְשִׁיל. מָצָאנוּ רַק פְּרוּסַת לֶחֶם חֲרֵבָה אַחַת שֶׁשָּׁמַרְנוּ בִּשְׁבִיל שַׁבָּת.

אִמִּי, שֶׁהָיְתָה אִשָּׁה חֲכָמָה וְצַדֶּקֶת גְּדוֹלָה, לֹא בָּכְתָה עַל חֹסֶר הַלֶּחֶם אוֹ עַל הָרָעָב. הִיא בָּכְתָה עַל דָּבָר אַחֵר לְגַמְרֵי!

"זַכַּאי בְּנִי", הִיא קָרְאָה לִי וְקוֹלָהּ רָעַד. "הַשַּׁבָּת כְּבָר נִכְנֶסֶת. הַמַּלְכָּה עוֹמֶדֶת בַּפֶּתַח. אֵיךְ נְקַבֵּל אֶת פָּנֶיהָ? הֲרֵי אֵין לָנוּ אֲפִלּוּ טִפַּת יַיִן אַחַת כְּדֵי שֶׁתַּעֲשֶׂה עָלֶיהָ קִדּוּשׁ! אֵיךְ נַגִּיד לַשַּׁבָּת 'בְּרוּכָה הַבָּאָה' בְּלִי כּוֹס יַיִן מְלֵאָה כְּפִי שֶׁכָּתוּב בַּתּוֹרָה: 'זָכוֹר אֶת יוֹם הַשַּׁבָּת לְקַדְּשׁוֹ – זָכְרֵהוּ עַל הַיַּיִן'?"

רָאִיתִי אֶת הַדְּמָעוֹת שֶׁלָּהּ נוֹפְלוֹת וּמַרְטִיבוֹת אֶת הַשֻּׁלְחָן הָרֵיק. חִפַּשְׂנוּ בְּכָל פִּנּוֹת הַבַּיִת, הָפַכְנוּ כָּל סִיר יָשָׁן – לֹא הָיְתָה שָׁם אֲפִלּוּ מַטְבֵּעַ שֶׁל נְחֹשֶׁת אַחַת.

הַמְּסִירוּת שֶׁל אִמָּא

לְפֶתַע, אִמִּי קָמָה וְהִזְדַּקְּפָה. פָּנֶיהָ שָׁבוּ לְהָאִיר. "לֹא נְבַטֵּל אֶת כְּבוֹד הַשַּׁבָּת, בְּנִי!" אָמְרָה בְּנֶחְרָצוּת. "הַמְתֵּן כָּאן. אֲנִי יוֹצֵאת לַשּׁוּק הַקָּרוֹב".

"אֲבָל אִמָּא", תָּמַהְתִּי, "אֵין לָנוּ שׁוּם דָּבָר לִמְכֹּר! מָה תַּעֲשִׂי שָׁם?"

הִיא לֹא עָנְתָה. הִיא יָצְאָה אֶל הָרְחוֹב הַקַּר. בְּאוֹתָם יָמִים, נָשִׁים צִדְקָנִיּוֹת הָיוּ שָׂמוֹת עַל רֹאשָׁן שְׁנֵי כִּסּוּיִים לִשְׁמֹר עַל הַצְּנִיעוּת: כִּסּוּי תַּחְתּוֹן פָּשׁוּט וְחָלָק, וּמֵעָלָיו, מִשּׁוּם כָּבוֹד, "כִּפַּת הָרֹאשׁ" – כִּסּוּי עֶלְיוֹן וּמְכֻבָּד שֶׁשָּׁמַר עַל הוֹד וְהָדָר, בִּמְיֻחָד לַנָּשִׁים בְּיִשְׂרָאֵל.

כְּשֶׁהִגִּיעָה אִמִּי לַשּׁוּק, הִיא הֵסִירָה מֵעַל רֹאשָׁהּ אֶת הַכִּסּוּי הֶחָשׁוּב – הַכִּפָּה שֶׁלָּהּ. הִיא נִשְׁאֲרָה רַק עִם הַמִּטְפַּחַת הַתַּחְתּוֹנָה וְהַפְּשׁוּטָה. הִיא מָכְרָה אֶת 'כִּפַּת רֹאשָׁהּ' – אֶת הַבֶּגֶד שֶׁהָיָה הַכָּבוֹד הַיָּחִיד שֶׁנִּשְׁאַר לָהּ, הַבֶּגֶד שֶׁעָטַף אוֹתָהּ בַּחֹרֶף! בַּכֶּסֶף הַמֻּעָט שֶׁקִּבְּלָה הִיא מִהֲרָה וְרָצָה יְשִׁירוֹת אֶל מוֹכֵר הַיַּיִן.

"תֵּן לִי יַיִן לְקִדּוּשׁ!" בִּקְּשָׁה בְּשִׂמְחָה שֶׁאִי אֶפְשָׁר לְתָאֵר, מַרְגִּישָׁה אֶת קַרְבּוֹתֵיהֶם הַקָּרוֹת שֶׁל הַמַּטְבְּעוֹת בְּיָדָהּ, מַעֲבִירָה אוֹתָם לַמּוֹכֵר, וּמְקַבֶּלֶת צְלוֹחִית קְטַנָּה עִם נוֹזֵל סָמִיךְ וְאָדֹם.

הַקִּדּוּשׁ הַמֵּאִיר בָּעוֹלָם

אִמִּי חָזְרָה הַבַּיְתָה בְּדִיּוּק כְּשֶׁהַשֶּׁמֶשׁ שָׁקְעָה. נֵרוֹת הַשַּׁבָּת כְּבָר דָּלְקוּ. כְּשֶׁנִּכְנְסָה, נֶעֱמַדְתִּי עַל רַגְלַי בְּהַפְתָּעָה.

"אִמָּא... אֵיפֹה הַכִּסּוּי הַיָּפֶה שֶׁלָּךְ?" שָׁאַלְתִּי.

אִמִּי הֶרְאֲתָה לִי אֶת צְלוֹחִית הַיַּיִן, מְחַיֶּכֶת מֵאֹזֶן לְאֹזֶן, אַךְ עֵינֶיהָ דָּמְעוּ מֵאֹשֶׁר. "וִתַּרְתִּי עָלָיו, זַכַּאי בְּנִי. מָה שָׁוֶה לִי כְּבוֹד מַלְבּוּשִׁי אִם אֵינֶנִּי יְכוֹלָה לְכַבֵּד אֶת הַשַּׁבָּת שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא?! קַח, בְּנִי. מְזֹג לַכּוֹס. עֲשֵׂה קִדּוּשׁ!"

יְלָדִים, אֲנִי לֹא יָכוֹל לְתָאֵר לָכֶם אֶת הַטַּעַם שֶׁל הַיַּיִן בְּאוֹתוֹ לַיְלָה.

עָמַדְתִּי מְרֻגָּשׁ בִּמְקוֹמִי. נָטַלְתִּי אֶת הַכּוֹס. אוֹר הַנֵּרוֹת הִשְׁתַּקֵּף מִתּוֹךְ הַיַּיִן הָאָדֹם. הֵרַחְתִּי אֶת רֵיחוֹ הַמָּתוֹק וְהַמְּשַׁכֵּר שֶׁל הַיַּיִן. קוֹלִי רָעַד כְּשֶׁזִּמַּרְתִּי: 'יוֹם הַשִּׁשִּׁי, וַיְכֻלּוּ הַשָּׁמַיִם וְהָאָרֶץ וְכָל צְבָאָם...'

לֹא הָיָה לָנוּ אֹכֶל כִּמְעַט כְּלָל עַל הַשֻּׁלְחָן, אֲבָל בְּאוֹתוֹ רֶגַע, הִרְגַּשְׁנוּ כְּמוֹ מְלָכִים הַיּוֹשְׁבִים בְּאַרְמוֹן! אֲנִי וְאִמִּי יָדַעְנוּ שֶׁאֶת הַשַּׁבָּת הַזּוֹ עָשִׂינוּ לִכְבוֹד בּוֹרֵא עוֹלָם מִתּוֹךְ מְסִירוּת נֶפֶשׁ אֲמִתִּית."

הַהַשְׁגָּחָה וְהַנִּצָּחוֹן

רַבִּי זַכַּאי סִיֵּם לְסַפֵּר אֶת יַלְדוּתוֹ לְתַלְמִידָיו בְּבֵית הַמִּדְרָשׁ, וְאָז נָשָׂא אֶת עֵינָיו לַשָּׁמַיִם וְאָמַר לָהֶם:

"חוֹשְׁבִים אַתֶּם שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא נִשְׁאַר 'חַיָּב' לְאִמָּא שֶׁלִּי? מַמָּשׁ לֹא! שָׁנִים רַבּוֹת חָלְפוּ. אִמִּי לֹא וִתְּרָה עָלַי, שָׁלְחָה אוֹתִי לִלְמֹד תּוֹרָה, וַאֲנִי זָכִיתִי לִהְיוֹת לְרַב. כְּשֶׁנִּפְטְרָה אִמִּי הַצַּדֶּקֶת, בֵּרַךְ ה' אֶת רְכוּשֵׁנוּ בִּבְרָכָה גְּדוֹלָה שֶׁלְּמַעְלָה מִן הַטֶּבַע. וְיוֹדְעִים אַתֶּם מָה הִיא הִשְׁאִירָה לִי בִּירוּשָּׁה?

שְׁלוֹשׁ מֵאוֹת חָבִיּוֹת שֶׁל יַיִן יָקָר וְטָעִים!"

תַּלְמִידָיו שֶׁל רַבִּי זַכַּאי שָׁמְעוּ וְהִשְׁתָּאוּ. הִיא מָכְרָה בֶּגֶד בִּשְׁבִיל כּוֹס יַיִן אַחַת , וְהַקָּדוֹשׁ בָּרוּךְ הוּא פָּתַח לָהּ אֶת אוֹצְרוֹת הַשָּׁמַיִם וְנָתַן לָהּ שְׁלוֹשׁ מֵאוֹת חָבִיּוֹת!

וְזֶה לֹא הַכֹּל, יְלָדִים, חֲזַ"ל מְגַלִּים לָנוּ שֶׁרַבִּי זַכַּאי עַצְמוֹ הִמְשִׁיךְ לְכַבֵּד אֶת הַשַּׁבָּת וּלְקַדֵּשׁ עַל הַיַּיִן, וּכְשֶׁהוּא נִפְטַר מִן הָעוֹלָם וְהָלַךְ לְגַן עֵדֶן הֵשִׁיב לוֹ ה' כִּפְלַיִם, וְהוּא הִשְׁאִיר לְיְלָדָיו שְׁלוֹשֶׁת אֲלָפִים חָבִיּוֹת יַיִן!

אֲבָל רַבִּי זַכַּאי תָּמִיד זָכַר – כָּל אֲרִיכוּת הַיָּמִים שֶׁלּוֹ, כָּל הַבְּרָכָה, וְכָל קְדֻשַּׁת הַתּוֹרָה שֶׁלּוֹ – בָּאוּ מֵאוֹתָהּ צְלוֹחִית קְטַנָּה, וּמִדִּמְעוֹת הָאַהֲבָה וְהַמְּסִירוּת שֶׁל אִמּוֹ לִכְבוֹד שַׁבָּת קֹדֶשׁ.

מקורות: תלמוד בבלי, מסכת מגילה, דף כ"ז עמוד ב' (שאלת התלמידים: "במה הארכת ימים?" תשובתו: "מימי לא ביטלתי קידוש היום... אמא זקנה היתה לי פעם אחת מכרה כיפה שבראשה והביאה לי לקידוש היום". ושכרם – שלוש מאות, ושלושת אלפים גרבי יין). 1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “יְלָדִים, בּוֹאוּ נִכָּנֵס בְּדִמְיוֹנֵנוּ לְתוֹךְ בֵּית הַמִּדְרָשׁ הַקָּדוֹשׁ שֶׁל הַתַּנָּאִים.” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “הוּא פָּתַח אֶת עֵינָיו וְאָמַר לְתַלְמִידָיו: "יָמַי הָאֲרֻכִּים לֹא בָּאוּ מֵעַצְמָם...”?
ג. כיצד הגיב רבי זכאי למצב המתואר: “בְּרַחֲבֵי הָעִיר הֵרַחְנוּ אֶת רֵיחוֹת הַתַּבְשִׁילִים שֶׁל כָּל הַשְּׁכֵנִים: רֵיחַ שֶׁל תַּפּוּחֵי אֲדָמָה צְלוּיִים, חַלּוֹת קְלוּעוֹת שֶׁמּוֹצִיאִים…”?
ד. מה השתנה בין “לֹא נְבַטֵּל אֶת כְּבוֹד הַשַּׁבָּת, בְּנִי!" אָמְרָה בְּנֶחְרָצוּת.” לבין “אִמִּי הֶרְאֲתָה לִי אֶת צְלוֹחִית הַיַּיִן, מְחַיֶּכֶת מֵאֹזֶן לְאֹזֶן, אַךְ עֵינֶיהָ דָּמְעוּ מֵאֹשֶׁר.”?
ה. כיצד הגיע הסיפור אל האירוע: “אִמִּי לֹא וִתְּרָה עָלַי, שָׁלְחָה אוֹתִי לִלְמֹד תּוֹרָה, וַאֲנִי זָכִיתִי לִהְיוֹת לְרַב.”?

3. מה היה ראשון?

כתבו ליד כל אירוע מספר מ־1 עד 5 לפי סדר התרחשותו בסיפור.

□ רָאִיתִי אֶת הַדְּמָעוֹת שֶׁלָּהּ נוֹפְלוֹת וּמַרְטִיבוֹת אֶת הַשֻּׁלְחָן הָרֵיק.
□ יְלָדִים, בּוֹאוּ נִכָּנֵס בְּדִמְיוֹנֵנוּ לְתוֹךְ בֵּית הַמִּדְרָשׁ הַקָּדוֹשׁ שֶׁל הַתַּנָּאִים.
□ אֲבָל רַבִּי זַכַּאי תָּמִיד זָכַר – כָּל אֲרִיכוּת הַיָּמִים שֶׁלּוֹ, כָּל הַבְּרָכָה, וְכָל קְדֻשַּׁת הַתּוֹרָה שֶׁלּוֹ – בָּאוּ מֵאוֹתָהּ צְלוֹחִית…
□ אַךְ הַדָּבָר שֶׁעָמַד לִי יוֹתֵר מִכֹּל הוּא: מֵעוֹלָם, אֲבָל מֵעוֹלָם – לֹא בִּטַּלְתִּי קִדּוּשׁ שֶׁל שַׁבָּת!
□ יְלָדִים, אֲנִי לֹא יָכוֹל לְתָאֵר לָכֶם אֶת הַטַּעַם שֶׁל הַיַּיִן בְּאוֹתוֹ לַיְלָה.

לפני ואחרי

מה קרה לפני: “רָאִיתִי אֶת הַדְּמָעוֹת שֶׁלָּהּ נוֹפְלוֹת וּמַרְטִיבוֹת אֶת הַשֻּׁלְחָן הָרֵיק.”?
מה קרה אחרי: “יְלָדִים, אֲנִי לֹא יָכוֹל לְתָאֵר לָכֶם אֶת הַטַּעַם שֶׁל הַיַּיִן בְּאוֹתוֹ לַיְלָה.”?

סיבה ותוצאה מתוך המעשה

האירוע הנתוןמה גרם לו?מה הייתה תוצאתו?
כְּשֶׁהָיִיתִי רַךְ בַּשָּׁנִים, אִמִּי וַאֲנִי חַיִּינוּ בְּעֹנִי נּוֹרָא.
הִיא מָכְרָה אֶת 'כִּפַּת רֹאשָׁהּ' – אֶת הַבֶּגֶד שֶׁהָיָה הַכָּבוֹד הַיָּחִיד שֶׁנִּשְׁאַר לָהּ, הַבֶּגֶד שֶׁעָטַף אוֹתָהּ בַּחֹרֶף!

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “יין לקידוש” על רבי זכאי? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “הֲרֵי אֵין לָנוּ אֲפִלּוּ טִפַּת יַיִן אַחַת כְּדֵי שֶׁתַּעֲשֶׂה עָלֶיהָ קִדּוּשׁ!”.
ג. אילו רבי זכאי לא היה פועל כפי שמתואר ב־“אִמִּי הֶרְאֲתָה לִי אֶת צְלוֹחִית הַיַּיִן, מְחַיֶּכֶת מֵאֹזֶן לְאֹזֶן, אַךְ עֵינֶיהָ דָּמְעוּ מֵאֹשֶׁר.” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “אִמִּי לֹא וִתְּרָה עָלַי, שָׁלְחָה אוֹתִי לִלְמֹד תּוֹרָה, וַאֲנִי זָכִיתִי לִהְיוֹת לְרַב.” מחזק את המסר של הסיפור?

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “בְּרַחֲבֵי הָעִיר הֵרַחְנוּ אֶת רֵיחוֹת הַתַּבְשִׁילִים שֶׁל כָּל הַשְּׁכֵנִים: רֵיחַ שֶׁל תַּפּוּחֵי אֲדָמָה צְלוּיִים, חַלּוֹת…”.
מכתב: כתבו לרבי זכאי בעקבות “הִיא מָכְרָה אֶת 'כִּפַּת רֹאשָׁהּ' – אֶת הַבֶּגֶד שֶׁהָיָה הַכָּבוֹד הַיָּחִיד שֶׁנִּשְׁאַר לָהּ, הַבֶּגֶד שֶׁעָטַף אוֹתָהּ…”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “לֹא הָיָה לָנוּ אֹכֶל כִּמְעַט כְּלָל עַל הַשֻּׁלְחָן, אֲבָל בְּאוֹתוֹ רֶגַע, הִרְגַּשְׁנוּ כְּמוֹ מְלָכִים הַיּוֹשְׁבִים…”.
דיווח: כתבו ידיעה קצרה שכותרתה “יין לקידוש”.

המחזה
