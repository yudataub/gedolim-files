מתי התייעץ לפני שפסק הלכה

דף עבודה בעקבות ר' חנינא בר חמא
שם: כיתה:

המעשה — הסיפור המלא

פְּתִיחָה: קוֹלוֹת הַלִּמּוּד בְּעִיר הַחֲכָמִים

יְלָדִים, בּוֹאוּ נַעֲצֹם עֵינַיִם וְנִסַּע זְמַן רַב אָחוֹרָה, אֶל מְרוֹמֵי הַגָּלִיל הַיְּרַקְרַק, אֶל הָעִיר צִפּוֹרִי.

הַשֶּׁמֶשׁ שֶׁל שְׁעַת צָהֳרַיִם זָרְחָה בַּחֲוָץ וְהֵאִירָה אֶת רְחוֹבוֹת הָאֶבֶן. אִם הֱיִיתֶם צוֹעֲדִים שָׁם, הֱיִיתֶם מְרִיחִים אֶת רֵיחַ עֲצֵי הַזַּיִת הַפּוֹרְחִים, וְאֶת הָרֵיחַ הַמָּתוֹק שֶׁל הַלֶּחֶם שֶׁנֶּאֱפָה לִכְבוֹד חַכְמֵי הַיְּשִׁיבָה. אֲבָל יוֹתֵר מִכָּל הָרֵיחוֹת וְהַמַּרְאוֹת, הָיָה צְלִיל אֶחָד שֶׁמִּלֵּא אֶת כָּל הָעִיר. כְּשֶׁמִּתְקָרְבִים אֶל בֵּית הַמִּדְרָשׁ הַגָּדוֹל, שׁוֹמְעִים קוֹל אַדִּיר כְּגַלֵּי הַיָּם: עַשְׂרוֹת וּמֵאוֹת יְהוּדִים לוֹמְדִים תּוֹרָה! מִתְוַכְּחִים, שׁוֹאֲלִים, עוֹנִים וְשָׁרִים אֶת מִלּוֹת הַהֲלָכָה בְּנִגּוּן שֶׁמְּחַמֵּם אֶת הַלֵּב.

בְּמֶרְכַּז בֵּית הַמִּדְרָשׁ, יָשַׁב עַל כִּסֵּא עֵץ פָּשׁוּט הָרַב הַגָּדוֹל מִכֻּלָּם. זֶהוּ רַבִּי חֲנִינָא בַּר חָמָא .

רַבִּי חֲנִינָא הָיָה כְּבָר אָדָם זָקֵן מְאֹד. זְקָנוֹ הָיָה לָבָן כַּשֶּׁלֶג וְיָרַד עַד לְחָזֵהוּ. עֵינָיו הָיוּ עֲמֻקּוֹת וַחֲכָמוֹת. כָּל תּוֹשָׁבֵי אֶרֶץ יִשְׂרָאֵל יָדְעוּ: אִם יֵשׁ לְךָ שְׁאֵלָה קָשָׁה, אִם לֹא הֵבַנְתָּ פָּסוּק בַּתּוֹרָה הַקְּדוֹשָׁה, אוֹ שֶׁנִּתְקַלְתָּ בִּבְעָיָה שֶׁלֹּא יָדַעְתָּ אֵיךְ לִפְתֹּר – סַע לְצִפּוֹרִי, לְרַבִּי חֲנִינָא! הַתּוֹרָה כֻּלָּהּ הָיְתָה מֻנַּחַת בְּרֹאשׁוֹ וּבְלִבּוֹ, מוּכָנָה וּמְסֻדֶּרֶת, מַמָּשׁ כְּמוֹ שֶׁאֲנַחְנוּ יוֹדְעִים לְדַקְלֵם אֶת הָאָלֶף-בֵּית.

הַשְּׁאֵלָה הַסְּבוּכָה בְּלֵב הַיּוֹם

וְהִנֵּה, בְּעִיצוּמוֹ שֶׁל הַלִּמּוּד, דֶּלֶת הָעֵץ הַכְּבֵדָה שֶׁל בֵּית הַמִּדְרָשׁ נִפְתְּחָה בַּחֲרִיקָה חַלָּשָׁה. בַּפֶּתַח עָמַד יְהוּדִי אֶחָד, בְּגָדָיו מְלֵאִים אָבָק מֵהַדֶּרֶךְ וּפָנָיו מֻדְאָגוֹת. תַּלְמִידֵי הַיְּשִׁיבָה הִבְחִינוּ בּוֹ וּפִנּוּ לוֹ מָקוֹם לָגֶשֶׁת הַיְשֵׁר אֶל שֻׁלְחָנוֹ שֶׁל רַבִּי חֲנִינָא.

"רַבִּי," אָמַר הָאִישׁ בְּקוֹל רוֹעֵד, וְהִנִּיחַ אֶת יָדָיו הַמְיֹבָּלוֹת שֶׁל הַמַּסָּע עַל שֻׁלְחַן הָעֵץ, "בָּאתִי מֵרָחוֹק מְאֹד. יֵשׁ לִי שְׁאֵלָה בַּהֲלָכָה שֶׁנּוֹגַעַת לְפַרְנָסָתִי וּלְטָהֳרַת בֵּיתִי. שָׁאַלְתִּי אֶת כָּל רַבָּנֵי הָעִיר שֶׁלִּי, וְאַף אֶחָד לֹא יָדַע מָה לַעֲנוֹת לִי! הֵם אָמְרוּ שֶׁהַשְּׁאֵלָה קָשָׁה מִדַּי, וְרַק גְּדוֹל הַדּוֹר יוּכַל לִפְתֹּר אוֹתָהּ".

שָׂרַר שֶׁקֶט בְּבֵית הַמִּדְרָשׁ. כָּל הַתַּלְמִידִים סָגְרוּ אֶת סִפְרֵיהֶם, רָכְנוּ קָדִימָה וְהִטּוּ אֶת אָזְנֵיהֶם. כֻּלָּם רָצוּ לִשְׁמֹעַ כֵּיצַד רַבִּי חֲנִינָא פוֹתֵר בִּשְׁנִיָּה אַחַת אֶת הַבְּעָיָה, כְּמוֹ שֶׁהוּא תָּמִיד עוֹשֶׂה. הָאִישׁ גּוֹלֵל וּפֵרֵט אֶת שְׁאֵלָתוֹ הַקָּשָׁה, עַל סְפֵקוֹת מְרֻבִּים וְדִינִים שֶׁהִסְתַּבְּכוּ זֶה בָּזֶה.

כְּשֶׁסִּיֵּם הָאִישׁ לְדַבֵּר, הוּא עָמַד וְחִכָּה לִתְשׁוּבָה.

הַתַּלְמִידִים מִסָּבִיב הִבִּיטוּ בָּרַב. אֲבָל... מַה קָּרָה לְרַבִּי חֲנִינָא?

רַבִּי חֲנִינָא לֹא עָנָה. מִצְחוֹ הִתְקַמֵּט בְּרִכּוּז. הוּא אָחַז בִּקְצוֹת זְקָנוֹ הַלָּבָן. עֵינָיו נֶעֶצְמוּ קְלוּתוֹת. שְׁנִיָּה רָדְפָה שְׁנִיָּה, אַךְ קוֹל לֹא בָּקַע מִפִּיו.

הַהִסּוּס וְהַפַּחַד

בְּבֵית הַמִּדְרָשׁ נַעֲשָׂה שֶׁקֶט דָּרוּךְ. אֶפְשָׁר הָיָה מַמָּשׁ לִשְׁמֹעַ אֶת צִיּוּץ הַצִּפּוֹרִים מִבַּחוּץ. מַדּוּעַ רַבִּי חֲנִינָא שׁוֹתֵק?

יְלָדִים, אִם זֶה הָיָה מִישֶׁהוּ מֵאִתָּנוּ שֶׁיּוֹדֵעַ תְּשׁוּבָה שֶׁאַף אֶחָד לֹא יוֹדֵעַ, מָה הָיִינוּ עוֹשִׂים? מִיָּד הָיִינוּ צוֹעֲקִים: "אֲנִי יוֹדֵעַ! אֲנִי יוֹדֵעַ! תִּרְאוּ כַּמָּה אֲנִי חָכָם!". הָיִינוּ מוֹצִיאִים מַהֵר אֶת הַתְּשׁוּבָה כְּדֵי לִזְכּוֹת בִּמְחִיאוֹת כַּפַּיִם.

אֲבָל רַבִּי חֲנִינָא הוּא אָדָם קָדוֹשׁ. תּוֹרָה אֶצְלוֹ זֶה לֹא מִשְׂחָק, וְלֹא הִזְדַּמְּנוּת לְהַרְאוֹת מִי צָדוּק וּמִי חָכָם יּוֹתֵר. תּוֹרָה הִיא תּוֹרַת ה', וְאָסוּר לְשַׁנּוֹת אוֹתָהּ שֶׁמָּא יִטְעֶה הָאָדָם וְיַכְשִׁיל אֶת עַם יִשְׂרָאֵל.

הָאִישׁ שֶׁשָּׁאַל אֶת הַשְּׁאֵלָה הִתְחִיל לִדְאֹג. "כְּבוֹד הָרַב... הַאִם תּוּכַל לַעֲזֹר לִי?" הוּא לָחַשׁ שׁוּב. "הַאִם זֶה מֻתָּר אוֹ אָסוּר?"

רַבִּי חֲנִינָא פָּקַח אֶת עֵינָיו, הִבִּיט בַּיְּהוּדִי בְּמַבָּט חַם וְאוֹהֵב, וְאָמַר לְאַט: "בְּנִי, אֲנִי מְבַקֵּשׁ סְלִיחָה. עָלֶיךָ לְהַמְתִּין עוֹד מְעַט. עֲדַיִן אֵינֶנִּי מֵשִׁיב לְךָ תְּשׁוּבָה".

תַּלְמִידֵי הַיְּשִׁיבָה הִתְפַּלְּאוּ עַד לִמְאֹד. אֶחָד הַתַּלְמִידִים הַוָּתִיקִים אָזַר אֹמֶץ, נִגַּשׁ בְּיִרְאַת כָּבוֹד לְרַבִּי חֲנִינָא וְשָׁאַל: "מוֹרִי וְרַבִּי הַקָּדוֹשׁ, הֲרֵי אֲנַחְנוּ לוֹמְדִים מִמְּךָ שָׁנִים רַבּוֹת! אֲנַחְנוּ יוֹדְעִים שֶׁיֵּשׁ לְרַבֵּנוּ תְּשׁוּבָה בְּרֹאשׁוֹ. אֲנַחְנוּ מַכִּירִים אֶת כֹּחוֹ וְחָכְמָתוֹ שֶׁל הָרַב הָעוֹקֵר הָרִים בְּתַלְמוּדוֹ. מַדּוּעַ מְעַכֵּב הָרַב אֶת הַתְּשׁוּבָה וְאֵינוֹ מוֹרֶה הֲלָכָה לָאִישׁ הַמִּסְכֵּן שֶׁטָּרַח וּבָא?"

סוֹד הַמָּסוֹרֶת וְהַזִּכָּרוֹן

רַבִּי חֲנִינָא יָשַׁר אֶת יְשִׁיבָתוֹ עַל הַכִּסֵּא. דְּמָמָה מֻחְלֶטֶת שָׂרְרָה כָּעֵת.

"תַּלְמִידַי הַיְקָרִים," פָּתַח הָרַב אֶת דְּבָרָיו כְּשֶׁקּוֹלוֹ נִרְגָּשׁ. "אַתֶּם צְרִיכִים לָדַעַת כֵּיצַד בּוֹנִים אֶת הַתּוֹרָה וְאֵיךְ שׁוֹמְרִים עָלֶיהָ מִטָּעֻיּוֹת. לְשָׁלוֹף תְּשׁוּבוֹת זֶה הַדָּבָר הַקַּל בְּיוֹתֵר. אֲנִי זָכִיתִי בְּחַיַּי לַדָּבָר שֶׁאַתֶּם כְּבָר לֹא הִסְפַּקְתֶּם – יָשַׁבְתִּי בַּשּׁוּרוֹת הָרִאשׁוֹנוֹת בְּבֵית מִדְרָשׁוֹ שֶׁל הַנָּשִׂיא הַגָּדוֹל מִכֻּלָּם: רַבִּי יְהוּדָה הַנָּשִׂיא , 'רַבִּי' שֶׁחִבֵּר אֶת הַמִּשְׁנָה!"

עֵינָיו שֶׁל רַבִּי חֲנִינָא בָּרְקוּ מֵהַגַּעְגּוּעַ לְרַבּוֹ הַגָּדוֹל, עַל אַף שֶׁעָבְרוּ שָׁנִים רַבּוֹת מֵאָז שֶׁנִּפְטַר.

"קִבַּלְתִּי עַל עַצְמִי חֹק בַּרְזֶל מֵאָז שֶׁהָפַכְתִּי לְמוֹרֵה הוֹרָאָה בְּיִשְׂרָאֵל," הִכְרִיז רַבִּי חֲנִינָא בְּנַחַת וּבְעָצְמָה. "מֵעוֹלָם, אֲבָל מֵעוֹלָם, לֹא עָנִיתִי עַל שְׁאֵלָה הִלְכָתִית סְבוּכָה עַד שֶׁבָּנִיתִי אוֹתָהּ עַל מַה שֶּׁרָאִיתִי בְּעֵינַי. הַחֹק שֶׁלִּי הוּא כָּזֶה: אֵינֶנִּי אוֹמֵר שׁוּם פְּסַק הֲלָכָה, אֶלָּא אִם כֵּן שָׁמַעְתִּי אֶת אוֹתָהּ הֲלָכָה נֶאֱמֶרֶת מִפִּיו שֶׁל 'רַבִּי' רַבּוֹת פְּעָמִים! וְלֹא זֶה בִּלְבַד – שְׁמִיעָה לֹא מַסְפִּיקָה לִי בַּדְּבָרִים הַקָּשִׁים. אֲנִי דּוֹרֵשׁ מֵעַצְמִי לְהִזָּכֵר שֶׁרָאִיתִי אֶת 'רַבִּי' מְבַצֵּעַ אֶת זֶה בְּאֹפֶן מַעֲשִׂי בְּעַצְמוֹ בְּבֵית הַדִּין... לְפָחוֹת שָׁלוֹשׁ פְּעָמִים!"

הַתַּלְמִידִים נָשְׁמוּ לִרְוָחָה וְהִתְפַּלְּאוּ מֵהַזְּהִירוּת הָעֲצוּמָה וְהָעֲנָוָה הַטְּהוֹרָה.

רַבִּי חֲנִינָא רָצָה לִפְסֹק, אַךְ כְּדֵי לִהְיוֹת בָּטוּחַ בְּמֵאָה אֲחוּזִים, הוּא חִפֵּשׂ עַכְשָׁו בְּמַאֲגָרֵי הַזִּכָּרוֹן שֶׁל מֹחוֹ. הוּא שׁוּב עָצַם אֶת עֵינָיו.

הַקָּדוֹשׁ בָּרוּךְ הוּא, שֶׁאֹהֵב אֶת יְרֵאָיו שֶׁמְּקַדְּשִׁים מִדּוֹתֵיהֶם עַל הַתּוֹרָה, פָּתַח בְּרֹאשׁוֹ שֶׁל רַבִּי חֲנִינָא מַעְיָן נָבְע. רַבִּי חֲנִינָא חָזַר בְּדִמְיוֹנוֹ שָׁנִים לְאָחוֹר, חָזַר אֶל בֵּית הַדִּין שֶׁל רַבּוֹ, 'רַבִּי'. הִנֵּה הוּא נִזְכָּר בַּפַּעַם הָרִאשׁוֹנָה שֶׁשָּׁאֲלוּ אֶת מַה שֶּׁשָּׁאַל זֶה עַתָּה הָאָדָם... הִנֵּה הַפַּעַם הַשְּׁנִיָּה... וְהִנֵּה, בְּחַסְדֵּי הַשֵּׁם, הִצְלִיחַ לְהִזָּכֵר גַּם בַּפַּעַם הַשְּׁלִישִׁית! הוּא רָאָה אֶת 'רַבִּי' מְנִיעַ אֶת רֹאשׁוֹ וּמוֹרֶה "מֻתָּר!".

הַנִּצָּחוֹן וְאוֹר הַפְּסָק

חִיּוּךְ רָחָב וּמֵאִיר כַּשֶּׁמֶשׁ הִתְפַּשֵּׁט עַל פָּנָיו הַזְּקֵנוֹת שֶׁל רַבִּי חֲנִינָא. הוּא פָּתַח אֶת עֵינָיו. צַעַר וְדְאָגַת הַשּׁוֹאֵל נָגוֹזוּ כְּלֹא הָיוּ.

"מֻתָּר זֶה, אָחִי הַיָּקָר, וְטָהוֹר לְגַמְרֵי!" פָּסַק רַבִּי חֲנִינָא בְּחִיּוּךְ קָדוֹשׁ. הַיְּהוּדִי צָהַל מִשִּׂמְחָה וְהוֹדָה לָרַב.

כָּךְ הֵבִינוּ תַּלְמִידֵי הַגָּלִיל כֻּלָּם: חָכָם אֲמִתִּי לֹא מְחַפֵּשׂ לְהַרְאוֹת כַּמָּה הוּא יוֹדֵעַ בְּמַהִירוּת. חָכָם אֲמִתִּי בּוֹדֵק, עוֹצֵר, נִזְכָּר בְּרַבּוֹתָיו, וְאוֹמֵר תְּשׁוּבָה שֶׁיֵּשׁ לָהּ אֱמֶת נִצְחִית וְשָׁרָשִׁים הַמַּגִּיעִים עַד לְהַר סִינַי.

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “כֻּלָּם רָצוּ לִשְׁמֹעַ כֵּיצַד רַבִּי חֲנִינָא פוֹתֵר בִּשְׁנִיָּה אַחַת אֶת הַבְּעָיָה, כְּמוֹ שֶׁהוּא תָּמִיד עוֹשֶׂה.”.
מכתב: כתבו לר' חנינא בר חמא בעקבות “אֲנִי זָכִיתִי בְּחַיַּי לַדָּבָר שֶׁאַתֶּם כְּבָר לֹא הִסְפַּקְתֶּם – יָשַׁבְתִּי בַּשּׁוּרוֹת הָרִאשׁוֹנוֹת בְּבֵית מִדְרָשׁוֹ…”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “הַקָּדוֹשׁ בָּרוּךְ הוּא, שֶׁאֹהֵב אֶת יְרֵאָיו שֶׁמְּקַדְּשִׁים מִדּוֹתֵיהֶם עַל הַתּוֹרָה, פָּתַח בְּרֹאשׁוֹ שֶׁל רַבִּי…”.
דיווח: כתבו ידיעה קצרה שכותרתה “מתי התייעץ לפני שפסק הלכה”.
