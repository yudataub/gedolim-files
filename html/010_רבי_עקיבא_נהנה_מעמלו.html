סיפור 10 • רבי עקיבא • נהנה מעמלו

נהנה מעמלו

דף עבודה בעקבות רבי עקיבא
שם: כיתה:

המעשה — הסיפור המלא

פְּתִיחָה: הַיַּעַר הַקַּר וְהַיָּדַיִם הַשְּׂרוּטוֹת

יְלָדִים, סִגְרוּ לְרֶגַע אֶת הָעֵינַיִם. דַּמְיְנוּ עֶרֶב חֹרֶף קַר מְאֹד. רוּחַ חֲזָקָה וְקַפְאָנִית שׁוֹרֶקֶת בֵּין עֲנָפֵי הָעֵצִים הַחֲשׂוּפִים. שששש... שששש...

בְּמַעֲבֵה הַיַּעַר, עוֹמֵד יְהוּדִי כְּבַר שָׁנִים, לָבוּשׁ בִּבְגָדִים פְּשׁוּטִים שֶׁאֵינָם מְחַמְּמִים מַסְפִּיק. זֶהוּ רַבִּי עֲקִיבָא.

יָדָיו שֶׁל רַבִּי עֲקִיבָא מְלֵאוֹת בִּשְׂרִיטוֹת, כִּי הוּא עוֹבֵד קָשֶׁה מְאֹד. מֵהַבֹּקֶר הוּא אוֹסֵף אֶת הָעֲנָפִים הַנְּפוּלִים, חוֹטֵב בַּגַּרְזֶן אֶת גִּזְעֵי הָעֵץ הַנְּמוּכִים, וְאוֹסֵף אוֹתָם לַעֲרֵמָה עֲנָקִית. רֵיחַ שֶׁל אֲדָמָה רְטֻבָּה וְעֵץ יָבֵשׁ עוֹמֵד בָּאֲוִיר. הַבֶּטֶן מְקַרְקֶרֶת מֵרָעָב, אַךְ עַל שְׂפָתָיו מִתְנוֹסֵס חִיּוּךְ מֵאִיר.

"תּוֹדָה לְךָ, הַקָּדוֹשׁ בָּרוּךְ הוּא," מְלַמְמֵל רַבִּי עֲקִיבָא וּמֵרִים מַבָּט לַשָּׁמַיִם הָאֲפֹרִים, "תּוֹדָה שֶׁנָּתַתָּ לִי כֹּחַ בַּיָּדַיִם לַעֲבֹד, כְּדֵי שֶׁלֹּא אֶצְטָרֵךְ לְבַקֵּשׁ כֶּסֶף כְּנִדְבָה מֵאַף אָדָם."

הוּא קוֹשֵׁר אֶת חֲבִילַת הָעֵצִים הַכְּבֵדָה בְּחֶבֶל עָבֶה, מַעֲמִיס אוֹתָהּ עַל גַּבּוֹ הַכּוֹאֵב, וְהוֹלֵךְ הָעִירָה, כְּשֶׁהָעֵצִים מְשַׁפְשְׁפִים אֶת כְּתֵפָיו.

הַהַחְלָטָה בַּשּׁוּק: חֵצִי לַבֶּטֶן וְחֵצִי לַנְּשָׁמָה

כַּאֲשֶׁר הוּא מַגִּיעַ לַשּׁוּק שֶׁל הָעִיר, הַשֶּׁמֶשׁ כְּבָר כִּמְעַט שׁוֹקַעַת. הַשּׁוּק הוֹמֶה אָדָם. צְעָקוֹת שֶׁל סוֹחֲרִים נִשְׁמָעוֹת מִכָּל עֵבֶר: "תְּאֵנִים עֲסִיסִיּוֹת! לֶחֶם טָרִי וְחַם!". הָרֵיחַ שֶׁל הַלֶּחֶם מְגָרֶה אֶת הָרָעָב, אֲבָל בְּכִיסוֹ שֶׁל רַבִּי עֲקִיבָא אֵין אֲפִלּוּ אֲגוֹרָה (פְּרוּטָה) אַחַת.

הוּא פּוֹרֵק אֶת חֲבִילַת הָעֵצִים הַגְּדוֹלָה מֵהַגַּב, וּמְחַלֵּק אוֹתָהּ בִּדְיּוּק לִשְׁנֵי חֲצָאִים שָׁוִים.

אֶת הַחֵצִי הָרִאשׁוֹן – הוּא מוֹכֵר בַּשּׁוּק. בַּפְּרוּטוֹת הַבּוֹדְדוֹת שֶׁהוּא מְקַבֵּל, הוּא קוֹנֶה מְעַט מְאֹד אֹכֶל, רַק מָה שֶׁצָּרִיךְ כְּדֵי לְהַחְזִיק אוֹתוֹ וְאֶת בְּנֵי בֵּיתוֹ בַּחַיִּים.

וּמָה עִם הַחֵצִי הַשֵּׁנִי שֶׁל הָעֵצִים?

הֲרֵי אִם הָיָה מוֹכֵר אֶת כָּל הָעֵצִים, יָכוֹל הָיָה לִקְנוֹת מָרָק חַם! יָכוֹל הָיָה לִקְנוֹת תַּבְשִׁיל טָעִים!

אַךְ רַבִּי עֲקִיבָא לוֹקֵחַ אֶת עֲרֵמַת הָעֵצִים שֶׁנּוֹתְרָה, מַעֲמִיס שׁוּב עַל הַגַּב, וְהוֹלֵךְ הַבַּיְתָה בְּצַעַד מָהִיר. יֵשׁ לוֹ תָּכְנִית מְיֻחֶדֶת עֲבוּר הָעֵצִים הָאֵלּוּ.

עָשָׁן, קֹר וְאוֹר הַתּוֹרָה

בַּחוּץ יָרַד הַלַּיְלָה. קְפִיאָה מַמָּשׁ.

רַבִּי עֲקִיבָא מַגִּיעַ לְבֵיתוֹ – בַּיִת קָטָן, פָּשׁוּט וְעָלוּב. קִירוֹת אֶבֶן סְדוּקִים וְרִצְפַּת אֲדָמָה. אֵין כִּסֵּא מְרֻפָּד, אֵין תַּנּוּר הַסָּקָה, אֵין אֲפִלּוּ מְנוֹרַת שֶׁמֶן כְּדֵי לְהָאִיר אֶת הַחֹשֶׁךְ.

הוּא מֵנִיחַ אֶת הָעֵצִים עַל רִצְפַּת הָאֲדָמָה בְּמֶרְכַּז הַחֶדֶר. הוּא מַצִּית אֵשׁ.

בַּתְּחִלָּה הָעֵצִים לוֹחֲשִׁים: קְרָאק... קְרָאק... , וְאָז פּוֹרֶצֶת שַׁלְהֶבֶת אֲדֻמָּה וּכְתֻמָּה. אַךְ בִּגְלַל שֶׁאֵלּוּ עֵצִים רְטֻבִּים מְעַט מֵהַיַּעַר, הֵם מְמַלְּאִים אֶת כָּל הַחֶדֶר בְּעָשָׁן אָפֹר וְסָמִיךְ מְאֹד. עָשָׁן שֶׁגּוֹרֵם לָעֵינַיִם לִדְמֹעַ וְלַגָּרוֹן לְהִשְׁתַּעֵל. אֲבָל רַבִּי עֲקִיבָא מוֹלֵק אֶת דִּמְעוֹת הָעָשָׁן מֵעֵינָיו, פּוֹתֵחַ מְגִלָּה שֶׁל תּוֹרָה, וּמִתְחִיל לִלְמֹד.

דַּמְיְנוּ אֶת הַמַּרְאֵה: רַבִּי עֲקִיבָא יוֹשֵׁב עַל הָרִצְפָּה, קָרוֹב-קָרוֹב לַמְּדוּרָה הַקְּטַנָּה שֶׁמַּעֲלָה עָשָׁן, וְקוֹרֵא בְּקוֹל עָרֵב וּמָתוֹק אֶת דִּבְרֵי הַתּוֹרָה. קוֹלוֹ מְנַסֵּר אֶת דְּמָמַת הַלַּיְלָה: "וְאָהַבְתָּ אֵת ה' אֱלֹקֶיךָ...". הַשַּׁלְהֶבֶת רוֹקֶדֶת וּמֵאִירָה אֶת פָּנָיו הַקְּדוֹשׁוֹת. כָּךְ הוּא לוֹמֵד שָׁעוֹת עַל גַּבֵּי שָׁעוֹת, עַד שֶׁהוּא מִתְעַיֵּף בְּאִישׁוֹן לַיְלָה.

וְאָז, הֲרֵי אֵין לוֹ מִטָּה, אֵין לוֹ שְׂמִיכָה וּמִזְרוֹן – הוּא נִשְׁכָּב לִישֹׁן מַמָּשׁ עַל עֲנָפֵי הָעֵצִים הַנּוֹתָרִים, מִתְכַּרְבֵּל כְּדֵי לִשְׁמֹר עַל חֹם גּוּפוֹ, וְנִרְדָּם בְּשַׁלְוָה וּבְשִׂמְחָה.

הַשְּׁכֵנִים הַכּוֹעֲסִים בַּפֶּתַח

אֲבָל לֹא כֻּלָּם הָיוּ מְרֻצִּים.

הָעָשָׁן שֶׁל רַבִּי עֲקִיבָא לֹא נִשְׁאַר רַק בְּבֵיתוֹ, הוּא חָדַר דֶּרֶךְ הַחַלּוֹנוֹת גַּם לַבָּתִּים שֶׁל הַשְּׁכֵנִים!

קְהוּ... קְהוּ... קְהוּ! עַל הַבֹּקֶר, נִשְׁמְעוּ דְּפִיקוֹת עַזּוֹת עַל דַּלְתּוֹ שֶׁל רַבִּי עֲקִיבָא.

בַּפֶּתַח עָמְדוּ הַשְּׁכֵנִים שֶׁלּוֹ. הֵם הָיוּ אֲנָשִׁים פְּשׁוּטִים, וְהֵם שִׁפְשְׁפוּ אֶת הָעֵינַיִם שֶׁלָּהֶם בְּכַעַס.

"עֲקִיבָא!" צָעַק אֶחָד הַשְּׁכֵנִים, "אַתָּה אִבַּדְתָּ אוֹתָנוּ בֶּעָשָׁן! חָנַקְתָּ אוֹתָנוּ לְגַמְרֵי! לָמָּה אַתָּה מַדְלִיק עֵצִים יְשָׁנִים וּרְטֻבִּים בְּתוֹךְ הַבַּיִת שֶׁלְּךָ כָּל לַיְלָה?!"

הַשָּׁכֵן הַשֵּׁנִי פָּנָה אֵלָיו בְּהַצָּעָה עִסְקִית: "תִּשְׁמַע, עֲקִיבָא. אַתָּה אָדָם חָכָם וְלוֹמֵד תּוֹרָה. בּוֹא נַעֲשֶׂה עֵסֶק. תִּמְכֹּר לָנוּ אֶת חֲצִי חֲבִילַת הָעֵצִים שֶׁאַתָּה מֵבִיא כָּל יוֹם! אֲנַחְנוּ נְשַׁלֵּם לְךָ עֲלֵיהֶם יָפֶה יָפֶה. בַּכֶּסֶף הַזֶּה – תּוּכַל לִקְנוֹת לְעַצְמְךָ שֶׁמֶן זַיִת זַךְ, נָקִי וְטָהוֹר, שֶׁמֵּאִיר בְּלִי טִפַּת עָשָׁן! כָּךְ תּוּכַל לִלְמֹד כָּל הַלַּיְלָה בְּאוֹר מֻשְׁלָם, מִבְּלִי לְהֵחָנֵק וּמִבְּלִי לַחְנֹק אוֹתָנוּ!".

יְלָדִים, אֵיזוֹ הַצָּעָה מְצֻיֶּנֶת, נָכוֹן? עַכְשָׁו תִּהְיֶה לְרַבִּי עֲקִיבָא מְנוֹרָה אֲמִתִּית! בְּלִי לִדְמֹעַ, בְּלִי לְהִשְׁתַּעֵל! מָה אַתֶּם חוֹשְׁבִים, הַאִם הוּא יַסְכִּים? אָדָם רָגִיל הָיָה קוֹפֵץ עַל הַמְּצִיאָה וְאוֹמֵר: 'בְּוַדַּאי! זֶה עֵסֶק מְצֻיָּן, כָּךְ אֶלְמַד יוֹתֵר בְּלִי לִסְבֹּל!'.

הַתְּשׁוּבָה הַמְנַצַּחַת מֵרֹב שִׂמְחָה

אֲבָל רַבִּי עֲקִיבָא חִיֵּךְ חִיּוּךְ גָּדוֹל. עֵינָיו נָצְצוּ מֵאַהֲבַת ה' הַבּוֹעֶרֶת בְּלִבּוֹ. הוּא הִבִּיט בַּשְּׁכֵנִים הַכּוֹעֲסִים וְאָמַר לָהֶם דְּבָרִים מַפְלִיאִים עַד שֶׁנֶּעֶתְקוּ הַמִּלִּים מִפִּיהֶם:

"שְׁכֵנַי הַיְּקָרִים, תּוֹדָה רַבָּה עַל הַהַצָּעָה, אֲבָל אֲנִי לֹא מְוַתֵּר עַל הָעֵצִים שֶׁלִּי. הַרְבֵּה סִפּוּקִים אֲנִי מִסְתַּפֵּק בָּהֶם!" (כְּלוֹמַר: הֵם נוֹתְנִים לִי הֲמוֹן הֲנָאָה וְצֹרֶךְ).

"אֵיךְ זֶה יָכוֹל לִהְיוֹת?" תָּמְהוּ הַשְּׁכֵנִים.

הִסְבִּיר לָהֶם רַבִּי עֲקִיבָא, וּמָנָה בְּאֶצְבְּעוֹתָיו שְׁלוֹשָׁה דְּבָרִים:

"אֶחָד – שֶׁאֲנִי שׁוֹנֶה בָּהֶם (אֲנִי מַדְלִיק אֶת הָעֵצִים, וְלָאוֹר שֶׁל הָאֵשׁ אֲנִי מַצְלִיחַ לִלְמֹד תּוֹרָה בַּלַּיְלָה).

וְאֶחָד – שֶׁאֲנִי מִתְחַמֵּם כְּנֶגְדָּם! (אִם קָנִיתִי שֶׁמֶן, אָכֵן יִהְיֶה לִי אוֹר, אֲבָל שֶׁמֶן לֹא מְחַמֵּם אֶת הַבַּיִת! אֵיךְ אֶלְמַד בַּקֹּר הַזֶּה? הָעֵצִים גַּם נוֹתְנִים לִי אוֹר וְגַם נוֹתְנִים לִי תַּנּוּר חִמּוּם עֲנָק).

הַשְּׁכֵנִים שָׁתְקוּ בְּתַדְהֵמָה. הֵם רָאוּ אָדָם כָּל כָּךְ עָנִי, אָדָם שֶׁיָּשֵׁן עַל בּוּלֵי עֵץ וְנוֹשֵׁם עָשָׁן, וּבְכָל זֹאת – הוּא בָּטוּחַ שֶׁיֵּשׁ לוֹ אֶת "מְנוֹרַת הַקְּרִיאָה" הֲכִי טוֹבָה, אֶת "תַּנּוּר הַחִמּוּם" הֲכִי חָזָק, וְאֶת "הַמִּטָּה" הֲכִי מְפַנֶּקֶת בָּעוֹלָם!

רַבִּי עֲקִיבָא חִיֵּךְ. הוּא בֶּאֱמֶת הָיָה שָׂמֵחַ. כִּי מִי שֶׁיֵּשׁ לוֹ אֶת הַתּוֹרָה בַּלֵּב, בְּעֶזְרַת הַקָּדוֹשׁ בָּרוּךְ הוּא, מוֹצֵא אֶת הָאוֹר גַּם בְּתוֹךְ הַחֹשֶׁךְ.

1. שאלות ידע מתוך הסיפור

ב. מדוע התרחש הדבר המתואר במשפט: “כַּאֲשֶׁר הוּא מַגִּיעַ לַשּׁוּק שֶׁל הָעִיר, הַשֶּׁמֶשׁ כְּבָר כִּמְעַט שׁוֹקַעַת.”?

ד. מה השתנה בין “וְאָז, הֲרֵי אֵין לוֹ מִטָּה, אֵין לוֹ שְׂמִיכָה וּמִזְרוֹן – הוּא נִשְׁכָּב לִישֹׁן מַמָּשׁ עַל עֲנָפֵי הָעֵצִים…” לבין “כָּךְ תּוּכַל לִלְמֹד כָּל הַלַּיְלָה בְּאוֹר מֻשְׁלָם, מִבְּלִי לְהֵחָנֵק וּמִבְּלִי לַחְנֹק אוֹתָנוּ!".”?
ה. כיצד הגיע הסיפור אל האירוע: “אֶחָד – שֶׁאֲנִי שׁוֹנֶה בָּהֶם (אֲנִי מַדְלִיק אֶת הָעֵצִים, וְלָאוֹר שֶׁל הָאֵשׁ אֲנִי מַצְלִיחַ לִלְמֹד תּוֹרָה בַּלַּיְלָה).”?

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “נהנה מעמלו” על רבי עקיבא? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “אֲבָל רַבִּי עֲקִיבָא מוֹלֵק אֶת דִּמְעוֹת הָעָשָׁן מֵעֵינָיו, פּוֹתֵחַ מְגִלָּה שֶׁל תּוֹרָה, וּמִתְחִיל לִלְמֹד.”.
ג. אילו רבי עקיבא לא היה פועל כפי שמתואר ב־“כָּךְ תּוּכַל לִלְמֹד כָּל הַלַּיְלָה בְּאוֹר מֻשְׁלָם, מִבְּלִי לְהֵחָנֵק וּמִבְּלִי לַחְנֹק אוֹתָנוּ!".” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “אֶחָד – שֶׁאֲנִי שׁוֹנֶה בָּהֶם (אֲנִי מַדְלִיק אֶת הָעֵצִים, וְלָאוֹר שֶׁל הָאֵשׁ אֲנִי מַצְלִיחַ לִלְמֹד תּוֹרָה בַּלַּיְלָה).” מחזק את המסר של הסיפור?
