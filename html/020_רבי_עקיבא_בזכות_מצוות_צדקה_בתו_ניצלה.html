בזכות מצוות צדקה — בתו ניצלה

דף עבודה בעקבות רבי עקיבא
שם: כיתה:

המעשה — הסיפור המלא

פְּתִיחָה: הַסּוֹד הַמַּפְחִיד שֶׁל גְּדוֹל הַדּוֹר

יְלָדִים, הַאִם פַּעַם יְדַעְתֶּם סוֹד – סוֹד שֶׁאָסוּר לְגַלּוֹת לְאַף אֶחָד, אֲבָל הוּא כָּל כָּךְ מַפְחִיד שֶׁהוּא גּוֹרֵם לַלֵּב שֶׁלָּכֶם לִדְפֹּק בְּחָזְקָה?

הַסִּפּוּר שֶׁלָּנוּ מַתְחִיל בְּסוֹד כָּזֶה מַמָּשׁ. סוֹד שֶׁהִסְתִּיר גְּדוֹל הַדּוֹר, רַבִּי עֲקִיבָא.

רַבִּי עֲקִיבָא אָהַב אֶת בִּתּוֹ אַהֲבָה עֲצוּמָה. הִיא הָיְתָה יַלְדָּה צַדֶּקֶת, עֲנָוָה, מְלֵאָה בְּמִדּוֹת טוֹבוֹת, שֶׁתָּמִיד חִיְּכָה וְעָזְרָה לְכֻלָּם. אֲבָל בְּלִבּוֹ שֶׁל רַבִּי עֲקִיבָא נֶחְבָּא פַּחַד גָּדוֹל.

לִפְנֵי שָׁנִים רַבּוֹת, פָּגְשׁוּ אוֹתוֹ הַ"כַּלְדִּיִּים" – אֲנָשִׁים גּוֹיִים שֶׁהָיוּ חוֹזִים בַּכּוֹכָבִים. הֵם הִבִּיטוּ לַשָּׁמַיִם, הִצְבִּיעוּ עַל הַכּוֹכָבִים שֶׁנִּצְנְצוּ בָּאוֹר קַר וְאָמְרוּ לוֹ דָּבָר נוֹרָא: "עֲקִיבָא, בַּיּוֹם שֶׁבּוֹ תִּתְחַתֵּן בִּתְּךָ – יָבוֹא נָחָשׁ מְסֻכָּן וְאַרְסִי, יַכִּישׁ אוֹתָהּ, וְהִיא תָּמוּת!".

רַבִּי עֲקִיבָא יָדַע שֶׁאֲנַחְנוּ הַיְּהוּדִים לֹא מַאֲמִינִים בַּכּוֹכָבִים. הוּא יָדַע שֶׁ"אֵין מַזָּל לְיִשְׂרָאֵל" – הַקָּדוֹשׁ בָּרוּךְ הוּא גָּדוֹל יוֹתֵר מִכָּל הַכּוֹכָבִים, וְהוּא זֶה שֶׁקּוֹבֵעַ. וּבְכָל זֹאת, לֵב שֶׁל אַבָּא דּוֹאֵג! שָׁנִים עַל גַּבֵּי שָׁנִים הוּא הִתְפַּלֵּל בְּדִמְעוֹת שָׁלִישׁ לְבּוֹרֵא עוֹלָם שֶׁיִּשְׁמֹר עַל בִּתּוֹ, וְשָׁמַר אֶת הַסּוֹד בְּלִבּוֹ.

שִׂמְחַת הַחֲתֻנָּה וְהַקּוֹל שֶׁלֹּא נִשְׁמַע

וְהִנֵּה, הִגִּיעַ הַיּוֹם הַגָּדוֹל! יוֹם הַחֲתֻנָּה שֶׁל בִּתּוֹ!

הַבַּיִת שֶׁל רַבִּי עֲקִיבָא לָבַשׁ חָג. רֵיחוֹת מְתוּקִים שֶׁל לֶחֶם טָרִי שֶׁיָּצָא מֵהַתַּנּוּר, וּבָשָׂר צָלוּי וּמְתֻבָּל, מִלְּאוּ אֶת הָאֲוִיר (רֵיחַ). קוֹלוֹת שֶׁל שִׁירָה, תֻּפִּים וְכִינוֹרוֹת הִדְהֲדוּ לְמֵרָחוֹק (שְׁמִיעָה). אוֹרְחִים רַבִּים, גְּדוֹלֵי הַתּוֹרָה וְעַנְיֵי הָעִיר, מִלְּאוּ אֶת הָאוּלָם. בִּתּוֹ הַכַּלָּה יָשְׁבָה בַּצַּד, לְבוּשָׁה בְּבִגְדֵי לָבָן צְנוּעִים וְיָפִים, וּפָנֶיהָ זָרְחוּ מֵאוֹר שֶׁל קְדֻשָּׁה וְשִׂמְחָה.

אַךְ רַבִּי עֲקִיבָא, שֶׁיָּשַׁב בְּרֹאשׁ הַשֻּׁלְחָן, לֹא יָכוֹל הָיָה לְהֵרָגַע. מַבָּטוֹ שׁוֹטֵט שׁוּב וָשׁוּב אֶל הַפְּתָחִים, אֶל חֹרֵי הַקִּירוֹת, מְחַפֵּשׂ סַכָּנָה נוֹרֵאת שֶׁזּוֹחֶלֶת עַל גְּחוֹנָהּ.

בְּשִׂיא הַשִּׂמְחָה, מַמָּשׁ כְּשֶׁכֻּלָּם הָיוּ בְּאֶמְצַע סְעוּדַת הַמִּצְוָה וְקַמּוּ לִרְקֹד – נִשְׁמְעָה בַּדֶּלֶת נְקִישָׁה.

טוּק... טוּק... טוּק...

אַף אֶחָד לֹא שָׁמַע. הַשִּׁירָה הָיְתָה חֲזָקָה מִדַּי, הַצַּלָּחוֹת נָקְשׁוּ זוֹ בָּזוֹ, וְהַשִּׂמְחָה מִלְּאָה אֶת הֶחָלָל.

בַּפֶּתַח עָמַד אִישׁ רָזֶה וַעֲלוּב מַרְאֶה. קוֹמָתוֹ שְׁפוּפָה, בְּגָדָיו קְרוּעִים, וְרֵיחַ שֶׁל אָבָק מִן הַדְּרָכִים נָדַף מִמֶּנּוּ. הוּא שָׁמַע אֶת הַקּוֹלוֹת, הֵרִיחַ אֶת מַאֲכָלֵי הַמְּלָכִים הַמַּהְבִּילִים הַמֻּנָּחִים עַל הַשֻּׁלְחָנוֹת, וּבִטְנוֹ קִרְקְרָה מֵרָעָב שֶׁל יָמִים. אֲבָל שׁוּם מֶלְצַר לֹא שָׂם לֵב אֵלָיו. הָעָנִי עָמַד שָׁם, הִרְכִּין אֶת רֹאשׁוֹ וְהִתְכּוֹנֵן לַחְזֹר אֶל הַחֹשֶׁךְ.

הַמָּנָה שֶׁל הַכַּלָּה

אֲבָל מִישֶׁהִי אַחַת כֵּן רָאֲתָה.

בִּתּוֹ שֶׁל רַבִּי עֲקִיבָא. דַּוְקָא הִיא, שֶׁזֶּהוּ הַיּוֹם הֲכִי גָּדוֹל, עָמוּס וּמְרַגֵּשׁ בְּחַיֶּיהָ. יְלָדִים, תְּתָאֲרוּ לָכֶם – מִי בִּכְלָל מִסְתַּכֵּל הַחוּצָה בְּיוֹם כָּזֶה? כֻּלָּם מִתְעַסְּקִים רַק בָּהּ, מְבָרְכִים אוֹתָהּ וְשָׂמֵחִים בְּשִׂמְחָתָהּ.

אַךְ מַבָּטָהּ צָד אֶת הָעָנִי הַמִּסְכֵּן. הִיא לֹא קָרְאָה לַמְּשָׁרְתִים, כִּי יָדְעָה שֶׁהֵם טְרוּדִים בְּעֶשְׂרוֹת אוֹרְחִים. מָה עָשְׂתָה? עַל הַשֻּׁלְחָן הָיְתָה מֻנַּחַת 'מְנַת הַכַּלָּה' הַמְּיֻחֶדֶת שֶׁלָּהּ – לֶחֶם חַם, בָּשָׂר רַךְ וּפֵרוֹת. הִיא קָמָה בְּשֶׁקֶט בְּשִׂמְלָתָהּ, נָטְלָה אֶת צַלַּחְתָּהּ שֶׁלָּהּ, יָצְאָה אֶל מִפְתַּן הַדֶּלֶת וְהִגִּישָׁה אוֹתָהּ יְשִׁירוֹת לַעֲנִי.

"קַח, אָחִי," לָחֲשָׁה בִּמְאוֹר פָּנִים, "אֱכֹל וִיהִי לְךָ לְבָרִיאוּת".

הָעָנִי הֵרִים אֵלֶיהָ עֵינַיִם רְטֻבּוֹת. הוּא לֹא הֶאֱמִין. הַכַּלָּה בְּעַצְמָהּ דּוֹאֶגֶת לוֹ? הוּא בֵּרַךְ אוֹתָהּ בִּרְכַּת שָׁמַיִם לַאֲרִיכוּת יָמִים, יָשַׁב בַּפִּנָּה וֶאֱכַל בְּשִׂמְחָה עַד שֶׁשָּׂבַע.

בְּאִישׁוֹן לַיְלָה: סַכָּנָה בַּקִּיר

הַחֲתֻנָּה הִסְתַּיְּמָה. רַבִּי עֲקִיבָא לִוָּה אֶת אִשְׁתּוֹ וְאֶת הַזּוּג הַצָּעִיר הַבַּיְתָה, לִבּוֹ דּוֹפֵק מֵרֹב צַעַר וּתְפִלָּה.

הַכַּלָּה הָעֲיֵפָה נִכְנְסָה לְחַדְרָהּ לִישֹׁן. הַקִּירוֹת שָׁם הָיוּ בְּנוּיִים מֵאַבְנֵי יְרוּשָׁלַיִם כְּבֵדוֹת שֶׁבֵּינֵיהֶן סְדָקִים. הִיא חָלְצָה מִשְּׂעָרָהּ אֶת הַ'מַּכְבֵּנָה' – סִיכַּת זָהָב גְּדוֹלָה וְחַדָּה שֶׁנָּהֲגוּ הַנָּשִׁים לֶאֱסֹף בָּהּ אֶת מַחְלְפוֹתֵיהֶן – וְלֹא הָיָה לָהּ מָקוֹם לְהַנִּיחַ אוֹתָהּ בַּחֹשֶׁךְ.

הִיא גִּשְּׁשָׁה בְּיָדָהּ, מָצְאָה חֹר קָטָן וְקָר בְּקִיר הָאֶבֶן הַחָשׂוּף (מִשּׁוּשׁ), וְתָקְעָה אֶת סִיכַּת הַזָּהָב הַחַדָּה עָמֹק לְתוֹךְ הַקִּיר... אָז שָׁכְבָה לִישֹׁן מִתּוֹךְ תְּפִלַּת קְרִיאַת שְׁמַע רְגוּעָה.

אֲבָל... הִיא לֹא יָדְעָה שֶׁבְּאוֹתוֹ סֶדֶק דַּוְקָא, הִמְתִּין הַמָּוֶת.

נָחָשׁ גָּדוֹל שְׁחֹר-מִשְׁקוֹרֶת (קוֹבְּרָה אֶרֶסְיָת), הִתְגַּנֵּב לְתוֹךְ חֹר הַקִּיר. עֵינָיו נָצְצוּ בַּחֹשֶׁךְ, לְשׁוֹנוֹ הַשְּׁסוּעָה רִפְרְפָה, וְהוּא אָרַב בְּשֶׁקֶט, מִתְכּוֹנֵן לְזַנֵּק וּלְהַכִּישׁ דֶּרֶךְ הַקִּיר בְּשָׁעָה שֶׁהַכַּלָּה תִּישַׁן.

וְאָז – קְלִיק! הַסִּיכָּה הַחַדָּה נִנְעֲצָה יְשִׁירוֹת... בְּתוֹךְ הָעַיִן שֶׁל הַנָּחָשׁ! לְלֹא כָּל כַּוָּנָה מִצִּדָּהּ, הַסִּיכָּה חָדְרָה יְשִׁירוֹת לְרֹאשׁוֹ שֶׁל נְחַשׁ הַצֶּפַע, וּבַלַּיְלָה הַהוּא הַנָּחָשׁ פָּלַט אֶת נִשְׁמָתוֹ מִבְּלִי שֶׁאַף אֶחָד יַרְגִּישׁ.

בֹּקֶר שֶׁל יְשׁוּעָה

עָלָה הַשַּׁחַר. קַרְנֵי שֶׁמֶשׁ רִאשׁוֹנוֹת שֶׁל בֹּקֶר לִטְּפוּ אֶת חַלּוֹנוֹ שֶׁל רַבִּי עֲקִיבָא. הוּא יָשַׁב בְּבֵיתוֹ, חִוֵּר וְרוֹעֵד. הוּא לֹא עָצַם עַיִן תָּמִיד הַלַּיְלָה. "הַאִם הַנְּבוּאָה שֶׁל הַכּוֹכָבִים תִּתְקַיֵּם? הַאִם בִּתִּי הַיְּקָרָה עוֹד בַּחַיִּים?!" חָשַׁב לְעַצְמוֹ מִתּוֹךְ תְּפִלָּה עֲמֻקָּה.

בְּחַדְרָהּ, קָמָה הַכַּלָּה הַטְּרִיָּה. צִפּוֹרִים צִיְּצוּ בַּחוּץ. הִיא אָמְרָה מִיָּד "מוֹדֶה אֲנִי", הוֹשִׁיטָה אֶת יָדָהּ אֶל חֹר הַקִּיר כְּדֵי לִשְׁלֹף אֶת סִיכַּת הַזָּהָב שֶׁלָּהּ.

הִיא מָשְׁכָה וּמָשְׁכָה... "זֶה מָשֶׁהוּ כָּבֵד הַבֹּקֶר..." חָשְׁבָה, וְאָז מָשְׁכָה בְּכֹחַ.

לְפֶתַע... זְעָקָה פָּלְחָה אֶת אֲוִיר הַבַּיִת!

רַבִּי עֲקִיבָא מִהֵר לָרוּץ וּלְהִכָּנֵס עִם בְּנֵי הַבַּיִת. בְּמֶרְכַּז הַחֶדֶר, עוֹמֶדֶת בִּתּוֹ, חִוֶּרֶת, וּבְיָדָהּ הַסִּיכָּה. אַךְ מַה שֶּׁהָיָה תָּקוּעַ וּנְגוֹרָר אַחֲרֵי הַסִּיכָּה הָיָה לֹא אַחֵר מֵאֲשֶׁר נָחָשׁ עָצוּם מֵמַדִּים, אָרֹךְ וּמְאַיֵּם, אַךְ לְגַמְרֵי מֵת!

רַבִּי עֲקִיבָא נָשַׁם בְּקֹשִׁי. דְּמָעוֹת שֶׁל גִּיל וְהוֹדָאָה לַה' הֵצִיפוּ אֶת עֵינָיו.

"בִּתִּי יַקִּירָתִי!" לָחַשׁ, קוֹלוֹ רוֹעֵד מֵהִתְרַגְּשׁוּת, "אֵיזֶה זְכוּת מְיֻחֶדֶת יֵשׁ לָךְ? מָה עָשִׂית אֶתְמוֹל שֶׁזָּכִית לְנֵס כָּזֶה אֲשֶׁר קָרַע אֶת גְּזֵרַת הַשָּׁמַיִם?!"

הַבַּת שָׁתְקָה לְרֶגַע, נִזְכֶּרֶת, וְהֵשִׁיבָה בַּעֲנָוָה: "אַבָּא, אֶתְמוֹל בָּעֶרֶב כְּשֶׁכֻּלָּם הָיוּ שָׂמֵחִים בַּחֲתֻנָּה, רָאִיתִי עָנִי עוֹמֵד בַּפֶּתַח. אַף אֶחָד לֹא שָׂם לֵב אֵלָיו. קַמְתִּי, לָקַחְתִּי אֶת הַמָּנָה שֶׁלִּי וְנָתַתִּי אוֹתָהּ לוֹ. זֶה הַכֹּל."

יָצָא רַבִּי עֲקִיבָא אֶל דֶּלֶת הַבַּיִת, הִבִּיט לַשָּׁמַיִם שֶׁהָיוּ עַכְשָׁו בְּהִירִים וְכְחֻלִּים, וְדָרַשׁ קֳבָל עַם וְעֵדָה:

"אֶחָי! דְּעוּ לָכֶם כִּי יֵשׁ כֹּחַ אֶחָד שֶׁחָזָק מִכָּל גְּזֵרוֹת הַכּוֹכָבִים – צְדָקָה תַּצִּיל מִמָּוֶת! אֵינָהּ מַצִּילָה רַק מִן הַסַּכָּנָה הַקְּטַנָּה, אֶלָּא מִן הַמָּוֶת מַמָּשׁ. וּמִי שֶׁרוֹאֶה אֶת הַשֵּׁנִי וְעוֹזֵר לוֹ, אֲפִלּוּ כְּשֶׁהוּא עַצְמוֹ בְּשִׂיא שִׂמְחָתוֹ – הַקָּדוֹשׁ בָּרוּךְ הוּא יִשְׁמֹר מֵעָלָיו וְיַצִּילוֹ לָנֶצַח."

1. שאלות ידע מתוך הסיפור

□ הִיא קָמָה בְּשֶׁקֶט בְּשִׂמְלָתָהּ, נָטְלָה אֶת צַלַּחְתָּהּ שֶׁלָּהּ, יָצְאָה אֶל מִפְתַּן הַדֶּלֶת וְהִגִּישָׁה אוֹתָהּ יְשִׁירוֹת לַעֲנִי.

לפני ואחרי

מה קרה לפני: “הִיא קָמָה בְּשֶׁקֶט בְּשִׂמְלָתָהּ, נָטְלָה אֶת צַלַּחְתָּהּ שֶׁלָּהּ, יָצְאָה אֶל מִפְתַּן הַדֶּלֶת וְהִגִּישָׁה אוֹתָהּ יְשִׁירוֹת לַעֲנִי.”?
מה קרה אחרי: “קַרְנֵי שֶׁמֶשׁ רִאשׁוֹנוֹת שֶׁל בֹּקֶר לִטְּפוּ אֶת חַלּוֹנוֹ שֶׁל רַבִּי עֲקִיבָא.”?

סיבה ותוצאה מתוך המעשה

האירוע הנתוןמה גרם לו?מה הייתה תוצאתו?
מַבָּטוֹ שׁוֹטֵט שׁוּב וָשׁוּב אֶל הַפְּתָחִים, אֶל חֹרֵי הַקִּירוֹת, מְחַפֵּשׂ סַכָּנָה נוֹרֵאת שֶׁזּוֹחֶלֶת עַל גְּחוֹנָהּ.
אָז שָׁכְבָה לִישֹׁן מִתּוֹךְ תְּפִלַּת קְרִיאַת שְׁמַע רְגוּעָה.

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “בזכות מצוות צדקה — בתו ניצלה” על רבי עקיבא? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “הִיא לֹא קָרְאָה לַמְּשָׁרְתִים, כִּי יָדְעָה שֶׁהֵם טְרוּדִים בְּעֶשְׂרוֹת אוֹרְחִים.”.
ג. אילו רבי עקיבא לא היה פועל כפי שמתואר ב־“לְלֹא כָּל כַּוָּנָה מִצִּדָּהּ, הַסִּיכָּה חָדְרָה יְשִׁירוֹת לְרֹאשׁוֹ שֶׁל נְחַשׁ הַצֶּפַע, וּבַלַּיְלָה הַהוּא הַנָּחָשׁ…” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “קַמְתִּי, לָקַחְתִּי אֶת הַמָּנָה שֶׁלִּי וְנָתַתִּי אוֹתָהּ לוֹ.” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. הִיא הָיְתָה יַלְדָּה צַדֶּקֶת, עֲנָוָה, מְלֵאָה בְּמִדּוֹת טוֹבוֹת,…
2. אוֹרְחִים רַבִּים, גְּדוֹלֵי הַתּוֹרָה וְעַנְיֵי הָעִיר, מִלְּאוּ אֶת…
3. דַּוְקָא הִיא, שֶׁזֶּהוּ הַיּוֹם הֲכִי גָּדוֹל, עָמוּס וּמְרַגֵּשׁ…
4. הִיא חָלְצָה מִשְּׂעָרָהּ אֶת הַ'מַּכְבֵּנָה' – סִיכַּת זָהָב גְּדוֹלָה…
5. זֶה מָשֶׁהוּ כָּבֵד הַבֹּקֶר..." חָשְׁבָה, וְאָז מָשְׁכָה בְּכֹחַ.
6. וּמִי שֶׁרוֹאֶה אֶת הַשֵּׁנִי וְעוֹזֵר לוֹ, אֲפִלּוּ כְּשֶׁהוּא עַצְמוֹ…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “הוּא שָׁמַע אֶת הַקּוֹלוֹת, הֵרִיחַ אֶת מַאֲכָלֵי הַמְּלָכִים הַמַּהְבִּילִים הַמֻּנָּחִים עַל הַשֻּׁלְחָנוֹת, וּבִטְנוֹ…”.
מכתב: כתבו לרבי עקיבא בעקבות “אָז שָׁכְבָה לִישֹׁן מִתּוֹךְ תְּפִלַּת קְרִיאַת שְׁמַע רְגוּעָה.”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “רַבִּי עֲקִיבָא מִהֵר לָרוּץ וּלְהִכָּנֵס עִם בְּנֵי הַבַּיִת.”.
דיווח: כתבו ידיעה קצרה שכותרתה “בזכות מצוות צדקה — בתו ניצלה”.

7. חיבור לחיים מתוך המעשה

א. מה אפשר ללמוד מן הדרך שבה רבי עקיבא פעל באירוע “רַבִּי עֲקִיבָא לִוָּה אֶת אִשְׁתּוֹ וְאֶת הַזּוּג הַצָּעִיר הַבַּיְתָה, לִבּוֹ דּוֹפֵק מֵרֹב צַעַר וּתְפִלָּה.”?
