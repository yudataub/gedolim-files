הדינר המזוייף — לפנים משורת הדין

דף עבודה בעקבות ר' חייא
שם: כיתה:

המעשה — הסיפור המלא

פְּתִיחָה: עֲבוֹדָה קָשָׁה תַּחַת הַשֶּׁמֶשׁ

יְלָדִים, נַסּוּ לְדַמְיֵן אֶת הַחֹם הַגָּדוֹל שֶׁל אֶרֶץ יִשְׂרָאֵל בִּימֵי הַקַּיִץ. שֶׁמֶשׁ חֲזָקָה וְקוֹפַחַת מְאִירָה מִמְּרוֹמִים, מַכָּה עַל הָרֹאשׁ. בְּתוֹךְ שְׂדֵה פִּשְׁתָּן גָּדוֹל, עָמַד יְהוּדִי פָּשׁוּט וְיָקָר. נִקְרָא לוֹ שִׁמְעוֹן.

שִׁמְעוֹן הָיָה אָדָם עָנִי מְאֹד. כַּפּוֹת הַיָּדַיִם שֶׁלּוֹ הָיוּ חֲסְפּוּסוֹת, מְלֵאוֹת בְּיַבָּלוֹת וּשְׂרִיטוֹת מֵרֹב עֲבוֹדָה קָשָׁה. חוֹדֶשׁ שָׁלֵם הוּא קָצַר אֶת הַפִּשְׁתָּן, הִשְׁרָה אוֹתוֹ בַּמַּיִם הַקָּרִים וְהַמְּרַעֲנְנִים שֶׁל מַעְיָן קָרוֹב, וְסָרַק אוֹתוֹ בְּמַסְרְקוֹת בַּרְזֶל עַד שֶׁהָפַךְ לְחוּטִים לְבָנִים וְרַכִּים.

"בָּרוּךְ ה', מָחָר יוֹם הַשּׁוּק!" חִיֵּךְ שִׁמְעוֹן, מְנַגֵּב אֶת זֵעַת אַפּוֹ בְּשַׁרְווּל חֻלְצָתוֹ הַבְּלוּיָה. "מָחָר אֶמְכֹּר אֶת הַבַּדִּים שֶׁאָרַגְתִּי, וְסוֹף סוֹף אוּכַל לִקְנוֹת לַיְּלָדִים שֶׁלִּי לֶחֶם חַם, שֶׁמֶן זַיִת צָלוּל, וְאוּלַי... אוּלַי אֲפִלּוּ מְעַט דָּגִים לִכְבוֹד שַׁבָּת!".

הַסּוֹחֵר וְהַמַּטְבֵּעַ הַמַּחְשִׁיד

לְמָחֳרָת, הַשּׁוּק הָיָה הוֹמֶה אָדָם. צַעֲקוֹת הַסּוֹחֲרִים בָּקְעוּ מִכָּל עֵבֶר: "תְּאֵנִים דְּבַשׁ! זֵיתִים שְׁחֹרִים!". רֵיחוֹת מְתוּקִים שֶׁל מַאֲפִים שֶׁזֶּה עַתָּה יָצְאוּ מִן הַתַּנּוּר הִתְעַרְבְּבוּ עִם רֵיחַ הַדָּגִים וְהַתַּבְלִינִים.

אֶל הַדּוּכָן הַקָּטָן שֶׁל שִׁמְעוֹן נִגַּשׁ סוֹחֵר עָשִׁיר עִם בְּגָדִים יָקָרִים. הוּא מִשֵּׁשׁ אֶת בַּד הַפִּשְׁתָּן שֶׁל שִׁמְעוֹן, הִנְהֵן בְּרֹאשׁוֹ בְּסִפּוּק, וְאָמַר: "אֲנִי קוֹנֶה אֶת הַכֹּל!".

הַסּוֹחֵר הִכְנִיס אֶת יָדוֹ לְכִיס טַלִּיתוֹ, הוֹצִיא מַטְבֵּעַ גָּדוֹל, וְהִנִּיחַ אוֹתוֹ עַל שֻׁלְחַן הָעֵץ.

קְלִינְג!

צְלִיל עָמוּם נִשְׁמַע. הַמַּטְבֵּעַ הָיָה 'דִּינָר'. הוּא הָיָה נוֹצֵץ מְאֹד, כִּמְעַט מְסַנְוֵר, אֲבָל כְּשֶׁשִּׁמְעוֹן לָקַח אוֹתוֹ לַיָּד, מַשֶּׁהוּ הִרְגִּישׁ לוֹ מוּזָר. הַמַּטְבֵּעַ הָיָה מְעַט קַר מִדַּי, וּכְשֶׁהִבִּיט בּוֹ מִקָּרוֹב, הַצֶּבַע שֶׁלּוֹ הָיָה טִיפָּה שׁוֹנֶה מִדִּינָר רָגִיל.

"הַאִם זֶה דִּינָר אֲמִתִּי?" שָׁאַל שִׁמְעוֹן בְּפַחַד אֶת הַסּוֹחֵר.

"בְּוַדַּאי שֶׁאֲמִתִּי! יָשָׁר מִבֵּית הַמֶּלֶךְ!" רָטַן הַסּוֹחֵר, לָקַח אֶת הַסְּחוֹרָה וְנֶעֱלַם בְּתוֹךְ הֶהָמוֹן.

שִׁמְעוֹן עָמַד שָׁם, לִבּוֹ דּוֹפֵק בְּחָזְקָה. "אִם הַדִּינָר הַזֶּה מְזֻיָּף," אָמַר לְעַצְמוֹ בַּחֲשָׁשׁ, "עָבַדְתִּי חוֹדֶשׁ שָׁלֵם לְחִנָּם! הַיְּלָדִים שֶׁלִּי יִרְעֲבוּ! אֲנִי חַיָּב שֶׁמִּישֶׁהוּ יִבְדֹּק אֶת זֶה... אֲבָל מִי? מִי מֵבִין כָּל כָּךְ טוֹב בְּמַטְבֵּעוֹת?".

פִּתְאוֹם נִזְכַּר: אֵשֶׁת חַיִל אַחַת בָּעִיר, אוֹ אָדָם חָכָם, יֵשׁ מִישֶׁהוּ אֶחָד שֶׁיּוֹדֵעַ הַכֹּל! זֶהוּ נְשִׂיא הַדּוֹר, שֶׁעָלָיו אוֹמְרִים שֶׁהוּא "גּוֹרֵם לַתּוֹרָה שֶׁלֹּא תִּשְׁתַּכַּח מִיִּשְׂרָאֵל" – רַבִּי חִיָּיא הַגָּדוֹל . רַבִּי חִיָּיא הָיָה לֹא רַק צַדִּיק וְעָנָיו, אֶלָּא גַּם "שֻׁלְחָנִי" מֻמְחֶה – כְּלוֹמַר, אָדָם שֶׁמֵּבִין מְעֻלֶּה בְּמַטְבֵּעוֹת וְכֶסֶף (כְּמוֹ בַּנְקַאי שֶׁל יָמֵינוּ).

בְּבֵית הַמִּדְרָשׁ שֶׁל רַבִּי חִיָּיא

שִׁמְעוֹן מִהֵר אֶל בֵּית הַמִּדְרָשׁ. כְּבָר מִבַּחוּץ הוּא יָכוֹל הָיָה לִשְׁמֹעַ אֶת קוֹל הַתּוֹרָה הַמָּתוֹק מִתְנַגֵּן, "אָמַר רַבִּי... הֲלָכָה הִיא...". הוּא נִכְנַס בְּהִסּוּס. עַל סַפְסָל שֶׁל עֵץ, מוּל מְגִלּוֹת קְלָף, יָשַׁב רַבִּי חִיָּיא. צַדִּיק עָצוּם, זְקָנוֹ יָורֵד עַל מִדּוֹתָיו וּפָנָיו מְאִירוֹת בְּאוֹר שֶׁל קְדֻשָּׁה. לְצִדּוֹ יָשַׁב תַּלְמִידוֹ הַמֻּבְהָק – "רַב".

שִׁמְעוֹן נִגַּשׁ בְּיִרְאַת כָּבוֹד, יָדָיו רוֹעֲדוֹת מְעַט. "כְּבוֹד הָרַב," לָחַשׁ, "הַאִם תּוּכַל לַעֲשׂוֹת עִמִּי חֶסֶד? קִבַּלְתִּי דִּינָר. כָּל פַּרְנָסַת יְלָדַי תְּלוּיָה בּוֹ, אֲבָל... אֲנִי מְפַחֵד שֶׁמָּא הוּא מְזֻיָּף".

רַבִּי חִיָּיא הֵרִים אֶת מַבָּטוֹ מִן הַסֵּפֶר, חִיֵּךְ אֶל הַיְּהוּדִי בְּאַהֲבָה גְּדוֹלָה, וְלָקַח אֶת הַמַּטְבֵּעַ.

יְלָדִים, מוּמְחֶה לְמַטְבֵּעוֹת יוֹדֵעַ בְּדִיּוּק מָה לַעֲשׂוֹת. רַבִּי חִיָּיא הֵחְזִיק אֶת הַדִּינָר קָרוֹב לָעֵינַיִם. הוּא בָּחַן אֶת הַצּוּרָה הַטְּבוּעָה עָלָיו, מִשֵּׁשׁ אֶת הַשּׁוּלַיִם שֶׁלּוֹ כְּדֵי לְהַרְגִּישׁ אֶת הַחִיסְפּוּס, וְהִקִּישׁ אוֹתוֹ בַּעֲדִינוּת עַל הַשֻּׁלְחָן הַקָּשֶׁה. צִילִילִינְג... צְלִיל דַּק וּמְעַט מוּזָר נִשְׁמַע. רַבִּי חִיָּיא קִמֵּט אֶת מִצְחוֹ. הוּא הָפַךְ שׁוּב וָשׁוּב, וּלְבַסּוֹף חִיֵּךְ.

"אַל תִּדְאַג, בְּנִי," אָמַר רַבִּי חִיָּיא בְּקוֹל מַרְגִּיעַ. "זֶה מַטְבֵּעַ קְצָת יָשָׁן אֲבָל הוּא דִּינָר מְעֻלֶּה. דִּינָר כָּשֵׁר וְטוֹב. לֵךְ לַשּׁוּק בְּשִׂמְחָה וִקְנֵה בּוֹ אוֹכֶל לִילָדֶיךָ".

שִׁמְעוֹן רָצָה כִּמְעַט לִרְקֹד מִשִּׂמְחָה. "תּוֹדָה רַבָּה לְךָ, יְבָרֶכְךָ ה'!" הוּא צָעַק, וּמִהֵר לָרוּץ חֲזָרָה אֶל הַשּׁוּק. הוּא יָכוֹל הָיָה כְּבָר לְדַמְיֵן אֶת הַחִיּוּךְ שֶׁל הַיְּלָדִים שֶׁלּוֹ כְּשֶׁיָּבִיא לָהֶם אֶת הַלֶּחֶם הֶחָם.

הַהַלֶּם בְּדוּכַן הַתַּבְלִינִים

שִׁמְעוֹן נִגַּשׁ לַדּוּכָן הֲכִי גָּדוֹל בַּשּׁוּק. הָיוּ שָׁם שַׂקֵּי קֶמַח לְבָנִים, צִנְצָנוֹת דְּבַשׁ נוֹזֵל, וּבַקְבּוּקֵי זְכוּכִית מְלֵאִים בְּשֶׁמֶן זַיִת טָהוֹר. שִׁמְעוֹן מִלֵּא אֶת הַסַּל שֶׁלּוֹ בְּכָל טוּב. שִׂמְחַת עֲנִיִּים שֶׁאֵין כְּמוֹתָהּ!

"כַּמָּה זֶה הַכֹּל עוֹלֶה?" שָׁאַל בְּגַאֲוָה, חָשׁ כְּמוֹ אִישׁ עָשִׁיר.

"בְּדִיּוּק דִּינָר אֶחָד," עָנָה הַמּוֹכֵר.

שִׁמְעוֹן שָׂם אֶת הַדִּינָר הַנּוֹצֵץ עַל שֻׁלְחַן הַמּוֹכֵר. הַמּוֹכֵר לָקַח אוֹתוֹ, הִבִּיט בּוֹ, וּפִתְאוֹם... פָּנָיו הֶחְשִׁיכוּ.

הוּא שִׁפְשֵׁף אֶת הַמַּטְבֵּעַ בְּכֹחַ עַל אֶבֶן הַשֻּׁלְחָן. צֶבַע הַזָּהָב-כֶּסֶף שֶׁל הַמַּטְבֵּעַ הִתְחִיל לְהִתְקַלֵּף, וּמִתַּחַת נִגְלְתָה מַתֶּכֶת מְכֹעֶרֶת וַאֲפוֹרָה – עוֹפֶרֶת! מַתֶּכֶת בְּזוֹל!

"מָה זֶה? מָה אַתָּה נוֹתֵן לִי כאן?!" צָעַק הַמּוֹכֵר בְּכַעַס, קוֹלוֹ נִשְׁמַע בְּכָל הַשּׁוּק. "אַתָּה רַמַּאי! זֶה זִיּוּף גָּמוּר! חֲתִיכַת בַּרְזֶל טִפְּשִׁית שֶׁצָּבְעוּ אוֹתָהּ כְּדֵי לְרַמּוֹת תְּמִימִים. קַח מִיָּד אֶת הַמַּטְבֵּעַ שֶׁלְּךָ, הַחְזֵר לִי אֶת הָאֹכֶל, וְתִסְתַּלֵּק מִכָּאן שֶׁלֹּא אֶקְרָא לַשּׁוֹמְרִים!".

טְרָאח! הַמּוֹכֵר זָרַק אֶת הַמַּטְבֵּעַ עַל הָרִצְפָּה. הַמַּטְבֵּעַ לֹא עָשָׂה צְלִיל שֶׁל כֶּסֶף, אֶלָּא קוֹל אָטוּם שֶׁל פַּח מְעוּכָךְ.

יְלָדִים, שִׁמְעוֹן הִרְגִּישׁ שֶׁהָעוֹלָם נוֹפֵל עָלָיו. הָעֵינַיִם שֶׁלּוֹ הִתְמַלְּאוּ דְּמָעוֹת. הַבֶּטֶן כָּאֲבָה מִצַּעַר. הוּא אָסַף אֶת הַמַּטְבֵּעַ הַמְּזֻיָּף וְהִתְחִיל לְהָלִיךְ אָחוֹרָה... מַה יַּגִּיד לַיְּלָדִים? שֶׁאֵין כְּלוּם לֶאֱכוֹל?

וְאָז הוּא נִזְכַּר... אֲבָל רַבִּי חִיָּיא אָמַר שֶׁזֶּה דִּינָר טוֹב! רַבִּי חִיָּיא, הַגָּאוֹן הַגָּדוֹל, טָעָה?! אֵיךְ יִתָּכֵן?

חוֹזְרִים לְבֵית הַמִּדְרָשׁ

בְּבּוּשָׁה גְּדוֹלָה, עִם עֵינַיִם אֲדוּמוֹת מִבֶּכִי, חָזַר שִׁמְעוֹן לְבֵית הַמִּדְרָשׁ.

"רַבִּי..." בָּכָה שִׁמְעוֹן. "אַתָּה אָמַרְתָּ לִי שֶׁהַדִּינָר הוּא מְעֻלֶּה. אֲבָל הַלָּכְתִּי לַשּׁוּק, וְאַף אֶחָד לֹא מַסְכִּים לָקַחַת אוֹתוֹ! הוּא מְזֻיָּף, עָשׂוּי מֵעוֹפֶרֶת! לַיְּלָדִים שֶׁלִּי אֵין מָה לֶאֱכֹל, וְהַטָּעוּת... הַטָּעוּת הָיְתָה שֶׁלְּךָ, כְּבוֹד הָרַב".

כָּל בֵּית הַמִּדְרָשׁ הִשְׁתַּתֵּק. הַתַּלְמִידִים הִסְתַּכְּלוּ עַל רַבִּי חִיָּיא.

יְלָדִים, אֲנַחְנוּ חַיָּבִים לְהָבִין כָּאן הֲלָכָה כְּלָלִית: אִם אָדָם רָגִיל טוֹעֶה כְּשֶׁהוּא בּוֹדֵק כֶּסֶף מִבְּלִי לְקַבֵּל תַּשְׁלוּם עַל הַבְּדִיקָה, הוּא חַיָּב לְשַׁלֵּם עַל הַנֶּזֶק שֶׁגָּרַם. אֲבָל בַּתּוֹרָה יֵשׁ דִּין שֶׁנִּקְרָא "שֻׁלְחָנִי מֻמְחֶה" (מוּמְחֶה עָצוּם לְמַטְבֵּעוֹת). כֵּיוָן שֶׁרַבִּי חִיָּיא הָיָה הַמּוּמְחֶה הֲכִי גָּדוֹל, וְאִם הוּא טָעָה – זֶה נֶחְשָׁב כְּ"כֹחַ עֶלְיוֹן", תְּאוּנָה נְדִירָה שֶׁקּוֹרֵית. וְלָכֵן, לְפִי דִּין הַתּוֹרָה , רַבִּי חִיָּיא הָיָה פָּטוּר לְגַמְרֵי מִלְּשַׁלֵּם לֶעָנִי! הַחֹק אָמַר: רַבִּי חִיָּיא לֹא אָשֵׁם, וְהוּא לֹא חַיָּב לָתֵת אֲפִלּוּ פְּרוּטָה שְׁחוּקָה.

אָדָם שֶׁהוּא לֹא גָּדוֹל בְּיִשְׂרָאֵל אוּלַי הָיָה אוֹמֵר לֶעָנִי: "תִּשְׁמַע, אֲנִי בֶּאֱמֶת מִצְטַעֵר. נָכוֹן, טָעִיתִי בַּבְּדִיקָה. אֲבָל תִּפְתַּח מַסֶּכֶת בָּבָא קַמָּא... תִּרְאֶה שֶׁאֲנִי מוּמְחֶה וַאֲנִי פָּטוּר. תֵּלֵךְ הַבַּיְתָה, ה' יַעֲזֹר לְךָ".

לִפְנִים מִשּׁוּרַת הַדִּין

אֲבָל רַבִּי חִיָּיא לֹא הָיָה כָּזֶה! עֵינָיו נִמְלְאוּ רַחֲמִים. הוּא לֹא הִסְתַּכֵּל עַל הַחֹק הַיָּבֵשׁ, הוּא הִסְתַּכֵּל עַל הַיְּהוּדִי שֶׁעוֹמֵד מוּלוֹ שָׁבוּר, רָעֵב, וְאֹבֵד עֵצוֹת.

רַבִּי חִיָּיא סָגַר אֶת הַסֵּפֶר. הוּא פָּנָה אֶל תַּלְמִידוֹ רַב, הֵנִיחַ אֶת יָדוֹ עַל כְּתֵפוֹ, וְאָמַר לְקוֹל דִּמְמַת הַנּוֹכְחִים:

"רַב תַּלְמִידִי, לֵךְ בְּבַקָּשָׁה לַקֻּפָּה הַפְּרָטִית שֶׁלִּי. קַח מִשָּׁם דִּינָר שֶׁל זָהָב טָהוֹר, מַטְבֵּעַ כָּשֵׁר וּמְעֻלֶּה, וְתֵן אוֹתוֹ לַיְּהוּדִי הַזֶּה שֶׁיּוּכַל לָקַחַת מָזוֹן לְבֵיתוֹ".

רַב קָם בְּשֶׁקֶט, לָקַח דִּינָר זָהָב מַבְרִיק, וְהִנִּיחַ בְּיָדָיו שֶׁל שִׁמְעוֹן. שִׁמְעוֹן הַנִּדְהָם רָאָה מַטְבֵּעַ אֲמִתִּי יָקָר כִּפְלַיִם בְּיָדוֹ, בֵּרֵךְ אֶת הָרַב מֵעוּמְקָא דְלִיבָּא וְרָץ מְאֻשָּׁר הַבַּיְתָה!

כְּשֶׁשִׁמְעוֹן יָצָא, אָמַר רַבִּי חִיָּיא לְתַלְמִידוֹ: "רַב... אַחֲרֵי שֶׁנָּתַתָּ לוֹ אֶת הַמַּטְבֵּעַ, פְּתַח בְּבַקָּשָׁה אֶת פִּנְקַס הַחֶשְׁבּוֹנוֹת שֶׁלִּי. תִּרְשֹׁם שָׁם לְיַד הַדִּינָר הַזֶּה אֶת הַמִּלִּים: 'זֶהוּ עֵסֶק רַע' (דין עסק ביש). זֶהוּ הֶפְסֵד כַּסְפִּי פְּרָטִי שֶׁלִּי".

רַב עָצַר לְרֶגַע. מִתּוֹךְ כָּבוֹד, הוּא שָׁאַל: "רַבִּי... הֲרֵי מִדִּין תּוֹרָה מֻמְחֶה גָּדוֹל שֶׁטָּעָה פָּטוּר מִלְּשַׁלֵּם! לָמָּה שִׁלַּמְתָּ לוֹ מִכַּסְפְּךָ הַפְּרָטִי עַל מַשֶּׁהוּ שֶׁאַתָּה לֹא חַיָּב בּוֹ כְּלָל?".

עָנָה לוֹ רַבִּי חִיָּיא בַּחֲמִימוּת: "נָכוֹן, מִשּׁוּרַת הַדִּין אֲנִי פָּטוּר. אֲבָל יְהוּדִי צָרִיךְ לִחְיוֹת 'לִפְנִים מִשּׁוּרַת הַדִּין' ! אִם עָנִי זֶה סוֹבֵל בִּגְלַל טָעוּת שֶׁעָשִׂיתִי, הַאִם אֶפְטוֹר אֶת עַצְמִי בְּתֵירוּצִים הִלְכָתִיִּים וְאֶתֵּן לוֹ לִבְכּוֹת? אֲנִי מְשַׁלֵּם מִכַּסְפִּי, מִשּׁוּם שֶׁהָאַהֲבָה לְכָל יְהוּדִי וְהַיֹּשֶׁר הַפְּנִימִי שֶׁלָּנוּ, דּוֹרְשִׁים מֵאִתָּנוּ לַעֲשׂוֹת וִתּוּרִים, לָתֵת אֶת הַטִּפָּה הַנּוֹסֶפֶת, וְלֹא מָה שֶׁרַק 'מֻכְרָחִים'".

יְלָדִים יְקָרִים, כָּךְ הֶרְאָה לָנוּ רַבִּי חִיָּיא שֶׁתּוֹרָה אֲמִתִּית לֹא יוֹשֶׁבֶת רַק בָּרֹאשׁ, הִיא יוֹשֶׁבֶת בַּלֵּב. שֶׁכֵּן "לִפְנִים מִשּׁוּרַת הַדִּין", זֶהוּ כְּלַל יְסוֹד עָצוּם שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא כָּל כָּךְ אוֹהֵב לִרְאוֹת בְּעַם יִשְׂרָאֵל.

מקורות: תלמוד בבלי , מסכת בבא קמא, דף צ"ט עמוד ב'. (הסוגיא של "המראה דינר לשולחני ונמצא רע... כי הא דהנהו דאחזו ליה לרבי חייא ואמר להו מעליא הוא, לבתר הכי אתו לקמיה, אמרו ליה אחזיניה ולא נפקי. אמר ליה לרב: זיל חלפיה ניהליה וכתוב על פנקסי: דין עסק ביש").

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “יְלָדִים, נַסּוּ לְדַמְיֵן אֶת הַחֹם הַגָּדוֹל שֶׁל אֶרֶץ יִשְׂרָאֵל בִּימֵי הַקַּיִץ.” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “הַסּוֹחֵר הִכְנִיס אֶת יָדוֹ לְכִיס טַלִּיתוֹ, הוֹצִיא מַטְבֵּעַ גָּדוֹל, וְהִנִּיחַ אוֹתוֹ עַל שֻׁלְחַן הָעֵץ.”?
ג. כיצד הגיב ר' חייא למצב המתואר: “צַדִּיק עָצוּם, זְקָנוֹ יָורֵד עַל מִדּוֹתָיו וּפָנָיו מְאִירוֹת בְּאוֹר שֶׁל קְדֻשָּׁה.”?
ד. מה השתנה בין “צֶבַע הַזָּהָב-כֶּסֶף שֶׁל הַמַּטְבֵּעַ הִתְחִיל לְהִתְקַלֵּף, וּמִתַּחַת נִגְלְתָה מַתֶּכֶת מְכֹעֶרֶת וַאֲפוֹרָה –…” לבין “וְלָכֵן, לְפִי דִּין הַתּוֹרָה , רַבִּי חִיָּיא הָיָה פָּטוּר לְגַמְרֵי מִלְּשַׁלֵּם לֶעָנִי!”?
ה. כיצד הגיע הסיפור אל האירוע: “הֲרֵי מִדִּין תּוֹרָה מֻמְחֶה גָּדוֹל שֶׁטָּעָה פָּטוּר מִלְּשַׁלֵּם!”?

□ הַחֹק אָמַר: רַבִּי חִיָּיא לֹא אָשֵׁם, וְהוּא לֹא חַיָּב לָתֵת אֲפִלּוּ פְּרוּטָה שְׁחוּקָה.

לפני ואחרי

מה קרה לפני: “שִׁמְעוֹן שָׂם אֶת הַדִּינָר הַנּוֹצֵץ עַל שֻׁלְחַן הַמּוֹכֵר.”?
מה קרה אחרי: “הַחֹק אָמַר: רַבִּי חִיָּיא לֹא אָשֵׁם, וְהוּא לֹא חַיָּב לָתֵת אֲפִלּוּ פְּרוּטָה שְׁחוּקָה.”?

סיבה ותוצאה מתוך המעשה

האירוע הנתוןמה גרם לו?מה הייתה תוצאתו?
פִּתְאוֹם נִזְכַּר: אֵשֶׁת חַיִל אַחַת בָּעִיר, אוֹ אָדָם חָכָם, יֵשׁ מִישֶׁהוּ אֶחָד שֶׁיּוֹדֵעַ הַכֹּל!
בְּבּוּשָׁה גְּדוֹלָה, עִם עֵינַיִם אֲדוּמוֹת מִבֶּכִי, חָזַר שִׁמְעוֹן לְבֵית הַמִּדְרָשׁ.

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “הדינר המזוייף — לפנים משורת הדין” על ר' חייא? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “תּוֹדָה רַבָּה לְךָ, יְבָרֶכְךָ ה'!" הוּא צָעַק, וּמִהֵר לָרוּץ חֲזָרָה אֶל הַשּׁוּק.”.
ג. אילו ר' חייא לא היה פועל כפי שמתואר ב־“וְלָכֵן, לְפִי דִּין הַתּוֹרָה , רַבִּי חִיָּיא הָיָה פָּטוּר לְגַמְרֵי מִלְּשַׁלֵּם לֶעָנִי!” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “הֲרֵי מִדִּין תּוֹרָה מֻמְחֶה גָּדוֹל שֶׁטָּעָה פָּטוּר מִלְּשַׁלֵּם!” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. יְלָדִים, נַסּוּ לְדַמְיֵן אֶת הַחֹם הַגָּדוֹל שֶׁל אֶרֶץ יִשְׂרָאֵל…
2. הוּא הָיָה נוֹצֵץ מְאֹד, כִּמְעַט מְסַנְוֵר, אֲבָל כְּשֶׁשִּׁמְעוֹן…
3. הוּא בָּחַן אֶת הַצּוּרָה הַטְּבוּעָה עָלָיו, מִשֵּׁשׁ אֶת הַשּׁוּלַיִם…
4. הַמַּטְבֵּעַ לֹא עָשָׂה צְלִיל שֶׁל כֶּסֶף, אֶלָּא קוֹל אָטוּם שֶׁל…
5. הוּא פָּנָה אֶל תַּלְמִידוֹ רַב, הֵנִיחַ אֶת יָדוֹ עַל כְּתֵפוֹ,…
6. שֶׁכֵּן "לִפְנִים מִשּׁוּרַת הַדִּין", זֶהוּ כְּלַל יְסוֹד עָצוּם…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “צַדִּיק עָצוּם, זְקָנוֹ יָורֵד עַל מִדּוֹתָיו וּפָנָיו מְאִירוֹת בְּאוֹר שֶׁל קְדֻשָּׁה.”.
מכתב: כתבו לר' חייא בעקבות “בְּבּוּשָׁה גְּדוֹלָה, עִם עֵינַיִם אֲדוּמוֹת מִבֶּכִי, חָזַר שִׁמְעוֹן לְבֵית הַמִּדְרָשׁ.”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “קַח מִשָּׁם דִּינָר שֶׁל זָהָב טָהוֹר, מַטְבֵּעַ כָּשֵׁר וּמְעֻלֶּה, וְתֵן אוֹתוֹ לַיְּהוּדִי הַזֶּה שֶׁיּוּכַל לָקַחַת מָזוֹן…”.
דיווח: כתבו ידיעה קצרה שכותרתה “הדינר המזוייף — לפנים משורת הדין”.

המחזה

רגע מן הסיפורהדמויותמה יאמרו או יעשו?
פִּתְאוֹם נִזְכַּר: אֵשֶׁת חַיִל אַחַת בָּעִיר, אוֹ אָדָם חָכָם, יֵשׁ מִישֶׁהוּ אֶחָד שֶׁיּוֹדֵעַ…
וְלָכֵן, לְפִי דִּין הַתּוֹרָה , רַבִּי חִיָּיא הָיָה פָּטוּר לְגַמְרֵי מִלְּשַׁלֵּם לֶעָנִי!

7. חיבור לחיים מתוך המעשה

א. מה אפשר ללמוד מן הדרך שבה ר' חייא פעל באירוע “צֶבַע הַזָּהָב-כֶּסֶף שֶׁל הַמַּטְבֵּעַ הִתְחִיל לְהִתְקַלֵּף, וּמִתַּחַת נִגְלְתָה מַתֶּכֶת מְכֹעֶרֶת וַאֲפוֹרָה – עוֹפֶרֶת!”?
ב. תארו מצב מחיי הכיתה שבו אפשר ליישם את המסר של “הדינר המזוייף — לפנים משורת הדין”.
