סיפור 68 • רבי יהודה הנשיא (רבי) • עריכת וכתיבת המשנה

עריכת וכתיבת המשנה

דף עבודה בעקבות רבי יהודה הנשיא (רבי)
שם: כיתה:

המעשה — הסיפור המלא

פְּתִיחָה: תּוֹרָה בְּלִי סְפָרִים

יְלָדִים, נַסּוּ לְדַמְיֵן אֶת עַצְמְכֶם נִכְנָסִים לְבֵית מִדְרָשׁ גָּדוֹל, עָמוּס בְּתַלְמִידִים. קוֹל שֶׁל לִמּוּד נִשְׁמַע מִכָּל עֵבֶר, קוֹלוֹת שֶׁל שִׁירָה, שֶׁל וִכּוּחִים בַּהֲלָכָה. אַתֶּם מְחַפְּשִׂים סְפָרִים עַל הַשֻּׁלְחָנוֹת... אֲבָל – אֵין אֲפִלּוּ סֵפֶר אֶחָד! אֵין סִדּוּרִים, אֵין חֻמָּשִׁים וְאֵין גְּמָרָא.

כָּךְ בְּדִיּוּק זֶה הָיָה לִפְנֵי כְּאַלְפַּיִם שָׁנָה. הָיָה אָסוּר לִכְתֹּב אֶת הַתּוֹרָה שֶׁבְּעַל-פֶּה! הַתַּלְמִידִים הָיוּ צְרִיכִים לְשַׁנֵּן הַכֹּל בָּרֹאשׁ. לַחְזֹר עַל הַהֲלָכָה מֵאָה פְּעָמִים, מָאתַיִם וַאֲפִלּוּ אַרְבַּע מֵאוֹת פְּעָמִים, רַק כְּדֵי שֶׁהַמִּלִּים יֵחָקְקוּ בְּלִבָּם כְּמוֹ בְּסֶלַע.

בְּאוֹתָם יָמִים, לְאַחַר חֻרְבַּן בֵּית הַמִּקְדָּשׁ, הִנְהִיג אֶת עַם יִשְׂרָאֵל צַדִּיק עָצוּם, נְשִׂיא הַסַּנְהֶדְרִין – רַבִּי יְהוּדָה הַנָּשִׂיא . כֻּלָּם קָרְאוּ לוֹ בְּכָבוֹד וּבְאַהֲבָה רַק בַּמִּלָּה: "רַבִּי", אוֹ "רַבֵּנוּ הַקָּדוֹשׁ". הַקָּדוֹשׁ בָּרוּךְ הוּא נָתַן לְרַבִּי חֵן מְיֻחָד. הוּא הָיָה עָשִׁיר גָּדוֹל, וַאֲפִלּוּ הַקֵּיסָר הָרוֹמָאִי הַגָּדוֹל, אַנְטוֹנִינוּס, הָיָה חֲבֵרוֹ הַטּוֹב! בִּזְכוּת הַחֲבֵרוּת הַזּוֹ, אַנְטוֹנִינוּס לֹא הֵצִיק לַיְּהוּדִים, וְלְעַם יִשְׂרָאֵל הָיְתָה סוֹף סוֹף תְּקוּפָה שֶׁל מְנוּחָה. אֲבָל הַשֶּׁקֶט הַזֶּה... הִסְתִּיר סַכָּנָה נוֹרָאָה.

הַדְּמָעוֹת שֶׁל הַזָּקֵן (הַשֵּׁפֶל וְהַמֶּתַח)

עֶרֶב אֶחָד, יָשַׁב רַבֵּנוּ הַקָּדוֹשׁ בְּבֵית הַמִּדְרָשׁ הַגָּדוֹל בָּעִיר צִפּוֹרִי. רֵיחַ שֶׁל שֶׁמֶן זַיִת טָהוֹר עָלָה מֵהַמְּנוֹרוֹת הַדּוֹלְקוֹת. לְפֶתַע, הִבְחִין רַבִּי בְּתַלְמִיד חָכָם זָקֵן שֶׁיּוֹשֵׁב בַּפִּנָּה. הַזָּקֵן הֶחְזִיק אֶת רֹאשׁוֹ בִּשְׁתֵּי יָדָיו, עָצַם אֶת עֵינָיו בְּחָזְקָה, וּבָכָה. דְּמָעוֹת חַמּוֹת זָלְגוּ עַל זְקָנוֹ הַלָּבָן.

רַבִּי קָם מִמְּקוֹמוֹ, נִגַּשׁ בַּעֲדִינוּת אֶל הַזָּקֵן, וְהִנִּיחַ יָד אוֹהֶבֶת עַל כְּתֵפוֹ. "מַדּוּעַ תִּבְכֶּה, אָחִי הַיָּקָר?" שָׁאַל בְּקוֹל שָׁקֵט.

הַזָּקֵן נָשָׂא אֶת עֵינָיו הָאֲדֻמּוֹת. "רַבִּי," קוֹלוֹ רָעַד, "אֲנִי בּוֹכֶה כִּי... אֲנִי שׁוֹכֵחַ. כְּשֶׁהָיִיתִי צָעִיר, הָיוּ הַגְּזֵרוֹת שֶׁל הָרוֹמָאִים הָרְשָׁעִים. הָיִינוּ צְרִיכִים לִבְרֹחַ, לְהִתְחַבֵּא בַּמְּעָרוֹת. מֵרֹב הַצָּרוֹת, מֵרֹב הַנְּדוּדִים מִמָּקוֹם לְמָקוֹם... הַהֲלָכוֹת שֶׁלָּמַדְתִּי מֵרַבּוֹתַי, מַתְחִילוֹת לְהִתְעַרְבֵּב בְּרֹאשִׁי. מָה אָמַר בֵּית שַׁמַּאי? מָה אָמַר בֵּית הִלֵּל? זִכְרוֹנִי כְּבָר אֵינוֹ כְּשֶׁהָיָה. אִם אֲנִי, שֶׁלָּמַדְתִּי כָּל יָמַי, מַתְחִיל לִשְׁכֹּחַ – מָה יִהְיֶה עַל הַיְלָדִים שֶׁלָּנוּ בעוֹד מֵאָה שָׁנָה?! הַתּוֹרָה, רַבִּי, הַתּוֹרָה אֲהוּבָתֵנוּ הוֹלֶכֶת לְאִבּוּד!".

הַמִּלִּים שֶׁל הַזָּקֵן דָּקְרוּ אֶת לִבּוֹ שֶׁל רַבִּי כְּמוֹ מַחַט. רַבִּי יָדַע שֶׁהַזָּקֵן צוֹדֵק. עַם יִשְׂרָאֵל מְפֻזָּר. חֵלֶק בְּאֶרֶץ יִשְׂרָאֵל, חֵלֶק בְּבָבֶל, חֵלֶק בְּאִיטַלְיָה. רַבִּי חָשַׁב עֲלֵיכֶם, יְלָדִים. מָה יִקְרֶה כְּשֶׁאַתֶּם תִּרְצוּ לִלְמֹד גְּמָרָא, וְלֹא יִשָּׁאֵר מִי שֶׁיִּזְכֹּר אוֹתָהּ?

הַהַחְלָטָה הַדְּרָמָטִית (נְקֻדַּת הַשִּׂיא)

אַךְ מָה אֶפְשָׁר לַעֲשׂוֹת? אָסוּר לִכְתֹּב! הֲלָכָה לְמֹשֶׁה מִסִּינַי קוֹבַעַת – תּוֹרָה שֶׁבְּעַל-פֶּה לֹא נוֹתְנִים בִּכְתָב.

רַבִּי יְהוּדָה הַנָּשִׂיא לֹא יָשַׁן אוֹתוֹ לַיְלָה. הוּא הִתְהַלֵּךְ בְּחַדְרוֹ אָנָה וָאָנָה. בַּחוּץ נָשְׁבָה רוּחַ קָרָה, אַךְ בְּלִבּוֹ שֶׁל רַבִּי בָּעֲרָה אֵשׁ לַהָבָה. הַהַשְׁגָּחָה הָעֶלְיוֹנָה נָתְנָה לוֹ עַכְשָׁו, בְּדִיּוּק בַּדּוֹר שֶׁלּוֹ, שֶׁקֶט וְשָׁלוֹם עִם הָרוֹמָאִים. וְהוּא, רַבִּי, נֶחְשַׁב לְתַלְמִיד הֶחָכָם הֲכִי גָּדוֹל שֶׁזָּכַר אֶת כָּל הַמָּסוֹרֶת. "הַקָּדוֹשׁ בָּרוּךְ הוּא בָּחַר בִּי לַמְּשִׂימָה הַזּוֹ", לָחַשׁ רַבִּי.

לְמָחֳרָת בַּבֹּקֶר, בְּמוֹשַׁב הַסַּנְהֶדְרִין, עָמַד רַבִּי מֵאֲחוֹרֵי שֻׁלְחָן עֵץ מָלֵא בִּקְלָפִים חֲלָקִים. מֵאוֹת חֲכָמִים יָשְׁבוּ מוּלוֹ וְחִכּוּ לִמְצָא אֶת פִּיו. שֶׁקֶט מֻחְלָט הִשְׂתָּרֵר בָּאוּלָם. רַבִּי הֵרִים נוֹצָה שֶׁל קֶסֶת דְּיוֹ. הוּא הִסְתַּכֵּל אֶל הַחֲכָמִים, עֵינָיו בָּרְקוּ, וְקוֹלוֹ הָרָם הִרְעִיד אֶת קִירוֹת הָאֶבֶן:

"רַבּוֹתַי! הַתּוֹרָה בְּסַכָּנָה! אִם לֹא נִכְתֹּב אוֹתָהּ עַכְשָׁו – בָּנֵינוּ לֹא יֵדְעוּ לִשְׁמֹר שַׁבָּת. הֵם לֹא יֵדְעוּ אֵיךְ לְהָנִיחַ תְּפִלִּין. לָכֵן, חֲכָמִים הַקְּדוֹשִׁים, מֵהַיּוֹם אָנוּ מַתְחִילִים לִכְתֹּב! כְּמוֹ שֶׁאָמַר דָּוִד הַמֶּלֶךְ בְּתְהִלִּים: 'עֵת לַעֲשׂוֹת לַה' – הֵפֵרוּ תּוֹרָתֶךָ' ! כְּשֶׁיֵּשׁ שְׁעַת סַכָּנָה שֶׁהַתּוֹרָה תִּשָּׁכַח חָלִילָה, אֲנַחְנוּ חַיָּבִים 'לְהָפֵר' לְרֶגַע אֶת הַכְּלָל, כְּדֵי לְהַצִּיל אוֹתָהּ לְעוֹלָם וָעֶד!".

הָעֲבוֹדָה הָאֲדִירָה וְהַנִּצָּחוֹן

בְּאוֹתוֹ יוֹם הֵחֵל הַמִּפְעָל הַגָּדוֹל בְּיוֹתֵר בְּתוֹלְדוֹת עַם יִשְׂרָאֵל. רַבִּי וּגְדוֹלֵי הַתַּלְמִידִים יָשְׁבוּ יוֹמָם וָלַיְלָה. שָׁעוֹת עַל גַּבֵּי שָׁעוֹת נִשְׁמַע בַּחֶדֶר רַק צְלִיל סְרִיטַת הַנּוֹצָה וְהַדְּיוֹ הַשָּׁחֹר עַל גַּבֵּי קְלָפֵי הָעוֹר הַבְּהִירִים.

הֵם אָסְפוּ אֶת הַהֲלָכוֹת. רַבִּי מִיֵּן אוֹתָן. הַלְכוֹת תְּפִלָּה וּבְרָכוֹת? "זֶה יִהְיֶה 'סֵדֶר זְרָעִים'", אָמַר רַבִּי. הַלְכוֹת שַׁבָּת וְחַגִּים? "זֶה 'סֵדֶר מוֹעֵד'". מָה עִם נְזָקִים, כְּסָפִים וְגֵזֶל? "זֶה 'סֵדֶר נְזִיקִין'". כָּךְ יָצַר רַבֵּנוּ הַקָּדוֹשׁ אֶת שִׁשָּׁה סִדְרֵי מִשְׁנָה – הַשַּׁ"ס! הַכֹּל נִכְתַּב בְּשָׂפָה עִבְרִית קְצָרָה, טְהוֹרָה וּמְדֻיֶּקֶת לְהַפְלִיא.

שָׁנִים שֶׁל עֲבוֹדָה קָשָׁה, שֶׁל אֵין-סוֹף מַאֲמָץ וּדְבֵקוּת מִצִּדּוֹ שֶׁל הַנָּשִׂיא. כְּשֶׁסִּיְּמוּ לַעֲרֹךְ וְלִכְתֹּב אֶת הַמִּשְׁנָה, רַבִּי נָשַׁם לִרְוָחָה. הַתּוֹרָה נִצְּלָה. יְלָדִים, כְּשֶׁאַתֶּם פּוֹתְחִים הַיּוֹם מִשְׁנָה בְּכִתָּה ג' אוֹ ד', כְּשֶׁאַתֶּם לוֹמְדִים "שְׁנַיִם אוֹחֲזִין בְּטַלִּית", אוֹ "מֵאֵימָתַי קוֹרִין אֶת שְׁמַע" – דְּעוּ לָכֶם, שֶׁבְּאוֹתוֹ רֶגַע אַתֶּם מַחְזִיקִים מַתָּנָה אִישִׁית שֶׁהֵכִין לָכֶם רַבֵּנוּ הַקָּדוֹשׁ. בִּזְכוּתוֹ אֲנַחְנוּ יְהוּדִים. בִּזְכוּתוֹ תּוֹרַת יִשְׂרָאֵל חַיָּה וְקַיֶּמֶת מֵאָז וְעַד עוֹלָם!

מקורות: ההיתר לכתיבת התורה שבעל פה (עת לעשות לה'): מסכת גיטין דף ס' עמוד א', תמורה י"ד עמוד ב'. "משום עת לעשות לה' הפרו תורתך - אמרי מוטב תעקר אות אחת מן התורה ואל תשתכח תורה מישראל". רקע על עריכת המשנה ע"י רבנו הקדוש: רמב"ם, הקדמה לספרו "משנה תורה" (היד החזקה), שם מתאר הרמב"ם בהרחבה את מעשיו ושיקוליו של רבי בקיבוץ כל ההלכות וסידור ששה סדרי משנה בגלל פיזור עם ישראל. עושרו וקשריו עם אנטונינוס: מסכת עבודה זרה דף י' עמוד א' (שם מסופר על קשריהם העמוקים והשקט שאפשר את המפעל התורני). קדושתו ומסירותו של רבי: בבא מציעא דף פ"ה עמוד א' (מידת ענוותנותו "משמת רבי בטלה ענווה"), תלמוד ירושלמי כלאים פרק ט' (שנשא ייסורים רבים לטובת הדור).

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “יְלָדִים, נַסּוּ לְדַמְיֵן אֶת עַצְמְכֶם נִכְנָסִים לְבֵית מִדְרָשׁ גָּדוֹל, עָמוּס בְּתַלְמִידִים.” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “הוּא הָיָה עָשִׁיר גָּדוֹל, וַאֲפִלּוּ הַקֵּיסָר הָרוֹמָאִי הַגָּדוֹל, אַנְטוֹנִינוּס, הָיָה חֲבֵרוֹ הַטּוֹב!”?
ג. כיצד הגיב רבי יהודה הנשיא (רבי) למצב המתואר: “רַבִּי קָם מִמְּקוֹמוֹ, נִגַּשׁ בַּעֲדִינוּת אֶל הַזָּקֵן, וְהִנִּיחַ יָד אוֹהֶבֶת עַל כְּתֵפוֹ.”?
ד. מה השתנה בין “וְהוּא, רַבִּי, נֶחְשַׁב לְתַלְמִיד הֶחָכָם הֲכִי גָּדוֹל שֶׁזָּכַר אֶת כָּל הַמָּסוֹרֶת.” לבין “כְּמוֹ שֶׁאָמַר דָּוִד הַמֶּלֶךְ בְּתְהִלִּים: 'עֵת לַעֲשׂוֹת לַה' – הֵפֵרוּ תּוֹרָתֶךָ' !”?
ה. כיצד הגיע הסיפור אל האירוע: “הַכֹּל נִכְתַּב בְּשָׂפָה עִבְרִית קְצָרָה, טְהוֹרָה וּמְדֻיֶּקֶת לְהַפְלִיא.”?

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “עריכת וכתיבת המשנה” על רבי יהודה הנשיא (רבי)? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “בַּחוּץ נָשְׁבָה רוּחַ קָרָה, אַךְ בְּלִבּוֹ שֶׁל רַבִּי בָּעֲרָה אֵשׁ לַהָבָה.”.
ג. אילו רבי יהודה הנשיא (רבי) לא היה פועל כפי שמתואר ב־“כְּמוֹ שֶׁאָמַר דָּוִד הַמֶּלֶךְ בְּתְהִלִּים: 'עֵת לַעֲשׂוֹת לַה' – הֵפֵרוּ תּוֹרָתֶךָ' !” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “הַכֹּל נִכְתַּב בְּשָׂפָה עִבְרִית קְצָרָה, טְהוֹרָה וּמְדֻיֶּקֶת לְהַפְלִיא.” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. יְלָדִים, נַסּוּ לְדַמְיֵן אֶת עַצְמְכֶם נִכְנָסִים לְבֵית מִדְרָשׁ…
2. בִּזְכוּת הַחֲבֵרוּת הַזּוֹ, אַנְטוֹנִינוּס לֹא הֵצִיק לַיְּהוּדִים,…
3. הַמִּלִּים שֶׁל הַזָּקֵן דָּקְרוּ אֶת לִבּוֹ שֶׁל רַבִּי כְּמוֹ מַחַט.
4. לְמָחֳרָת בַּבֹּקֶר, בְּמוֹשַׁב הַסַּנְהֶדְרִין, עָמַד רַבִּי…
5. בְּאוֹתוֹ יוֹם הֵחֵל הַמִּפְעָל הַגָּדוֹל בְּיוֹתֵר בְּתוֹלְדוֹת עַם…
6. בִּזְכוּתוֹ תּוֹרַת יִשְׂרָאֵל חַיָּה וְקַיֶּמֶת מֵאָז וְעַד עוֹלָם!

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “רַבִּי קָם מִמְּקוֹמוֹ, נִגַּשׁ בַּעֲדִינוּת אֶל הַזָּקֵן, וְהִנִּיחַ יָד אוֹהֶבֶת עַל כְּתֵפוֹ.”.
מכתב: כתבו לרבי יהודה הנשיא (רבי) בעקבות “מֵאוֹת חֲכָמִים יָשְׁבוּ מוּלוֹ וְחִכּוּ לִמְצָא אֶת פִּיו.”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “שָׁעוֹת עַל גַּבֵּי שָׁעוֹת נִשְׁמַע בַּחֶדֶר רַק צְלִיל סְרִיטַת הַנּוֹצָה וְהַדְּיוֹ הַשָּׁחֹר עַל גַּבֵּי קְלָפֵי הָעוֹר…”.
דיווח: כתבו ידיעה קצרה שכותרתה “עריכת וכתיבת המשנה”.
