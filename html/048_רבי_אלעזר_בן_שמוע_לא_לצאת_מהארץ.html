סיפור 48 • רבי אלעזר בן שמוע • לא לצאת מהארץ

לא לצאת מהארץ

דף עבודה בעקבות רבי אלעזר בן שמוע
שם: כיתה:

המעשה — הסיפור המלא

להלן עיבוד חינוכי, רגשי ומרתק לסיפור המופלא על רבי אלעזר בן שמוע ורבי יוחנן הסנדלר בגבול ארץ ישראל . הסיפור נכתב בקפידה עבור בנים בכיתות ב'-ד' (חינוך דתי-לאומי), תוך שימוש בעקרונות הסטוריטלינג, תיאורים חושיים, והדגשת קדושתה של ארץ ישראל ואהבת התורה.

הסיפור מנוקד במלואו כנדרש.

פְּתִיחָה: הַצִּמָּאוֹן הַגָּדוֹל לְתוֹרָה

יְלָדִים, עִצְמוּ רֶגַע אֶת הָעֵינַיִם. הַאִם הִרְגַּשְׁתֶּם פַּעַם צִמָּאוֹן גָּדוֹל? לֹא צִמָּאוֹן לְמַיִם מִנֶּגֶב לָהוּט בַּקַּיִץ, אֶלָּא צִמָּאוֹן בַּלֵּב. צִמָּאוֹן לִשְׁמֹעַ דְּבַר תּוֹרָה!

לִפְנֵי קָרוֹב לְאַלְפַּיִם שָׁנָה, אֶרֶץ יִשְׂרָאֵל הָיְתָה עֲצוּבָה. הָרוֹמָאִים הָרְשָׁעִים גָּזְרוּ גְּזֵרוֹת קָשׁוֹת, וְרַבִּי עֲקִיבָא, גְּדוֹל רַבָּנֵי הַדּוֹר, נֶהֱרַג עַל קִדּוּשׁ הַשֵּׁם. שְׁנַיִם מִתַּלְמִידָיו הַגְּדוֹלִים – רַבִּי אֶלְעָזָר בֶּן שַׁמּוּעַ וְרַבִּי יוֹחָנָן הַסַּנְדְּלָר, הִרְגִּישׁוּ שֶׁהַלֵּב שֶׁלָּהֶם בּוֹעֵר! הֵם חַיָּבִים לְהַמְשִׁיךְ לִלְמֹד תּוֹרָה, הֵם חַיָּבִים לְהָבִין עוֹד סוּגְיוֹת וְהֲלָכוֹת!

בְּאוֹתָם יָמִים, הָרַב הַגָּדוֹל בְּיוֹתֵר שֶׁנִּשְׁאַר בַּחַיִּים הָיָה רַבִּי יְהוּדָה בֶּן בְּתֵירָא. אַךְ הָיְתָה בְּעָיָה מְאֹד מְאֹד כְּאֵבָה: רַבִּי יְהוּדָה בֶּן בְּתֵירָא גָּר רָחוֹק. מַמָּשׁ רָחוֹק. בְּעִיר שֶׁנִּקְרֵאת 'נְצִיבִין', וְהִיא נִמְצֵאת מִחוּץ לְאֶרֶץ יִשְׂרָאֵל, בְּאֶרֶץ בָּבֶל!

"מַה נַּעֲשֶׂה?" שָׁאַל רַבִּי יוֹחָנָן הַסַּנְדְּלָר, עֵינָיו מַבִּיטוֹת עַל עֲצֵי הַזַּיִת שֶׁל הַגָּלִיל. "אֵיךְ נַעֲזֹב אֶת אֶרֶץ הַקֹּדֶשׁ?"

"אֵין בְּרֵרָה, אָחִי," עָנָה רַבִּי אֶלְעָזָר בֶּן שַׁמּוּעַ, בְּקוֹל רוֹעֵד. "הַתּוֹרָה הִיא חַיֵּינוּ. כְּדֵי שֶׁתּוֹרַת יִשְׂרָאֵל לֹא תִּשָּׁכַח, אֲנַחְנוּ חַיָּבִים לָלֶכֶת לִנְצִיבִין."

הַמַּסָּע צָפוֹנָה וְרֵיחַ הֶעָפָר

הֵם אָרְזוּ אֶת חֲפָצֵיהֶם הַמְּעַטִּים. שַׂק לֶחֶם יָבֵשׁ, כַּד מַיִם, וּמְעַט בְּגָדִים. הַקָּדוֹשׁ בָּרוּךְ הוּא שָׁמַר עַל רַגְלֵיהֶם, וְהֵם הָלְכוּ. טְרַק... טְרַק... עִם מַקְלוֹתֵיהֶם דָּפְקוּ עַל הַסְּלָעִים וְהַשְּׁבִילִים. הַשֶּׁמֶשׁ שֶׁל אַרְצֵנוּ קָפְחָה עַל רֹאשָׁם, הַפָּנִים שֶׁלָּהֶם הִתְמַלְּאוּ בְּזֵעָה, וּכְכָל שֶׁהֵם עָלוּ לְכִוּוּן צָפוֹן, רוּחַ קְרִירָה הִתְחִילָה לְנַשֵּׁב מִכִּוּוּן הַיָּם הַגָּדוֹל.

יְלָדִים, אֶרֶץ יִשְׂרָאֵל הִיא אֶרֶץ חַיָּה. רַבִּי אֶלְעָזָר וְרַבִּי יוֹחָנָן לֹא רַק צָעֲדוּ עָלֶיהָ, הֵם הִרְגִּישׁוּ אוֹתָהּ. כָּל אֶבֶן הָיְתָה קְדוֹשָׁה, כָּל פֶּרַח הֵרִיחַ מִקְּדֻשָּׁה. כְּכָל שֶׁהִתְרַחֲקוּ, לִבָּם כָּבַד עֲלֵיהֶם כְּמוֹ אֶבֶן גְּדוֹלָה.

וְאָז, אַחֲרֵי מַסָּע מְיַגֵּעַ שֶׁל יָמִים, הֵם הִגִּיעוּ לְעִיר גְּדוֹלָה עַל הַחוֹף. קָרְאוּ לָהּ צַיְדָּן (צִידוֹן).

"אָנוּ עוֹמְדִים עַל גְּבוּל הָאָרֶץ," אָמַר רַבִּי אֶלְעָזָר, וְקוֹלוֹ נֶחְנַק. זֶהוּ זֶה. עוֹד מִסְפַּר פְּסִיעוֹת, וְהֵם בְּחוּץ לָאָרֶץ. בְּאֶרֶץ שֶׁאֵין בָּהּ קְדֻשָּׁה. הֵם הֵרִיחוּ אֶת רֵיחַ הַמֶּלַח מִן הַיָּם, אֲבָל פִּתְאוֹם – הֵם זָכְרוּ.

הַמַּבָּט לְאָחוֹר וְהַבְּכִי שֶׁבָּקַע שְׁחָקִים

לִפְנֵי שֶׁחָצוּ אֶת הַקַּו, שְׁנֵי הַתַּנָּאִים עָצְרוּ בַּמָּקוֹם. הֵם הִסְתּוֹבְבוּ אָחוֹרָה, לְדָרוֹם.

מָה הֵם רָאוּ? הֵם רָאוּ אֶת הָרֵי הַגָּלִיל הַיְּרֻקִּים. הֵם דִּמְיְנוּ מֵעֵבֶר לָהֶם אֶת הָרֵי יְרוּשָׁלַיִם. אֶת מְקוֹם מִקְדָּשֵׁנוּ. אֶת הָאֲדָמָה שֶׁנָּתַן בּוֹרֵא עוֹלָם לְאַבְרָהָם, יִצְחָק וְיַעֲקֹב.

"אֶרֶץ יִשְׂרָאֵל..." אָמַר רַבִּי יוֹחָנָן הַסַּנְדְּלָר בִּלְחִישָׁה.

פִּתְאוֹם, גַּל מְטֹרָף שֶׁל אָהֲבָה וְגַעֲגוּעַ הִכָּה בָּהֶם. הַלֵּב הַקָּדוֹשׁ שֶׁלָּהֶם לֹא יָכוֹל הָיָה לָשֵׂאת אֶת הַצַּעַר שֶׁל פְּרֵדָה מֵהַקְּדֻשָּׁה שֶׁל הָאָרֶץ!

דְּמָעוֹת, גְּדוֹלוֹת כְּמוֹ טִפּוֹת שֶׁל גֶּשֶׁם הָרִאשׁוֹן, הִתְחִילוּ לִזְלֹג מֵעֵינֵיהֶם הַטְּהוֹרוֹת שֶׁל שְׁנֵי הַצַּדִּיקִים הַגְּדוֹלִים הָאֵלֶּה. הֵם בָּכוּ! "הוֹי, אַדְמַת קֹדֶשׁ, מִשְׁכַּן הַשְּׁכִינָה!" בָּכוּ הַחֲכָמִים.

הֵם זָכְרוּ שֶׁאָדָם שֶׁנֶּעֱצָב כָּל כָּךְ עַל פְּגִיעָה בַּקֹּדֶשׁ, קוֹרֵעַ אֶת בִּגְדּוֹ. וְאָז נִשְׁמַע הַקּוֹל: קְרַאע... קְרַאעעע... הֵם תָּפְסוּ בְּצַוְּארוֹן הַמְּעִיל שֶׁלָּהֶם, וְקָרְעוּ קְרִיעָה שֶׁל אֵבֶל! הֵם לֹא הָיוּ מְסֻגָּלִים לַחְשֹׁב עַל כָּךְ שֶׁכַּף רַגְלָם תִּדְרֹךְ עַל אֲדָמָה אַחֶרֶת!.

הַהַחְלָטָה הַגּוֹרָלִית

הֵם עָמְדוּ שָׁם בַּשֶּׁמֶשׁ, שְׁנֵי גְּדוֹלֵי עַם יִשְׂרָאֵל, בּוֹכִים וּבִגְדֵיהֶם קְרוּעִים. וְאָז, הֵם נָשְׂאוּ עֵינַיִם בּוֹהֲקוֹת זֶה לָזֶה וְקָרְאוּ אֶת הַפָּסוּק שֶׁבַּתּוֹרָה מִסֵּפֶר דְּבָרִים: "וִירִשְׁתֶּם אֹתָהּ – וִישַׁבְתֶּם בָּהּ!" .

אָמַר רַבִּי אֶלְעָזָר לְרַבִּי יוֹחָנָן הַסַּנְדְּלָר: "יוֹדֵעַ אַתָּה מָה זֶה אוֹמֵר? הַתּוֹרָה שׁוֹקֶלֶת כָּאן. יְשִׁיבַת אֶרֶץ יִשְׂרָאֵל – לָגוּר בְּאֶרֶץ יִשְׂרָאֵל – שְׁקוּלָה כְּנֶגֶד כָּל הַמִּצְווֹת שֶׁבַּתּוֹרָה! לִלְמֹד אֵצֶל רַבִּי יְהוּדָה בֶּן בְּתֵירָא זֶה חָשׁוּב מְאֹד, אֲבָל לַעֲזֹב אֶת אֶרֶץ הַקֹּדֶשׁ?! זֶה כְּמוֹ לַעֲזֹב אֶת אַבָּא וְאִמָּא!".

בְּאוֹתוֹ הָרֶגַע, נֶעֶלְמוּ הַסְּפֵקוֹת. הַשָּׁמַיִם נִרְאוּ פִּתְאוֹם בְּהִירִים יוֹתֵר. עִם בְּגָדִים קְרוּעִים, אֲבָל עִם לֵב כָּל כָּךְ שָׁלֵם וְשָׂמֵחַ, הֵם סוֹבְבוּ אֶת מַקְלוֹתֵיהֶם, סָבְבוּ עַל עִקְבוֹתֵיהֶם – וְחָזְרוּ חֲזָרָה לְתוֹךְ אֶרֶץ יִשְׂרָאֵל!

הֵם חָזְרוּ לִלְמֹד וּלְלַמֵּד תּוֹרָה בְּאֶרֶץ הַקֹּדֶשׁ, וְהִקִּימוּ אֶת הַדּוֹר הַבָּא שֶׁל הַתַּנָּאִים שֶׁשָּׁמַר עַל תּוֹרַת יִשְׂרָאֵל כָּאן – קָרוֹב אֶל מְקוֹם הַשְּׁכִינָה.

א. שְׁאֵלַת פְּתִיחָה (לִפְנֵי הַסִּפּוּר):

"ילדים, מי כאן היה פעם בחוץ לארץ? ולמי יצא לחזור מנסיעה ארוכה, ואיך שהמטוס נחת בנתב"ג הוא הרגיש 'איזה כיף, חזרנו הביתה!'?

היום נשמע על שני ענקי עולם, שהיו חייבים לטוס, או יותר נכון – ללכת, לחוץ לארץ למטרה הכי קדושה שיש. אבל משהו עצר אותם בגבול..."

ב. הֶעָרוֹת בִּצּוּעַ וּסְטוֹרִיטֶלִינְג (לַמּוֹרֶה):

ג. הַמְלָצוֹת לְמַצֶּגֶת (4 שְׁקוּפִיּוֹת):

(זכור: צניעות קפדנית, כיפות/כיסוי ראש גדול לגיבורים היהודים, שרוולים ארוכים, אור רוחני במקום ייצוג אלוהות).

ד. פְּעִילוּת מְסַכֶּמֶת (לַכִּתָּה):

"מפת האהבה לארץ"

חלק לכל ילד דף שעליו מצוירת מפת ארץ ישראל. בקש מהילדים לכתוב או לצייר בתוך גבולות המפה: "מה אני הכי אוהב בארץ ישראל?" (זה יכול להיות עיר קדושה, בית הכנסת שלהם, חיילי צה"ל, המשפחה, או האדמה). מי שמסיים, ינשק את הדף שאליו צייר.

מקורות: ספרי, פרשת ראה, פיסקא נ"ג ("מעשה ברבי יהודה בן בתירא ורבי מתיא בן חרש ורבי חנניה בן אחי רבי יהושע ורבי יונתן שיוצאים לחוץ לארץ... מעשה ברבי אלעזר בן שמוע ורבי יוחנן הסנדלר שהיו הולכים לנציבים..."). מוזכר גם בספר "כה עשו חכמינו", חלק ד', עמוד 35. ספר האגדה, רע"ג. (כפי שצוין במאגר המקורות שסופק בבקשה [1-3]).

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “להלן עיבוד חינוכי, רגשי ומרתק לסיפור המופלא על רבי אלעזר בן שמוע ורבי יוחנן הסנדלר בגבול ארץ ישראל .” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “הֵם חַיָּבִים לְהַמְשִׁיךְ לִלְמֹד תּוֹרָה, הֵם חַיָּבִים לְהָבִין עוֹד סוּגְיוֹת וְהֲלָכוֹת!”?
ג. כיצד הגיב רבי אלעזר בן שמוע למצב המתואר: “הַקָּדוֹשׁ בָּרוּךְ הוּא שָׁמַר עַל רַגְלֵיהֶם, וְהֵם הָלְכוּ.”?
ד. מה השתנה בין “הֵם הֵרִיחוּ אֶת רֵיחַ הַמֶּלַח מִן הַיָּם, אֲבָל פִּתְאוֹם – הֵם זָכְרוּ.” לבין “הֵם זָכְרוּ שֶׁאָדָם שֶׁנֶּעֱצָב כָּל כָּךְ עַל פְּגִיעָה בַּקֹּדֶשׁ, קוֹרֵעַ אֶת בִּגְדּוֹ.”?
ה. כיצד הגיע הסיפור אל האירוע: “עִם בְּגָדִים קְרוּעִים, אֲבָל עִם לֵב כָּל כָּךְ שָׁלֵם וְשָׂמֵחַ, הֵם סוֹבְבוּ אֶת מַקְלוֹתֵיהֶם, סָבְבוּ עַל עִקְבוֹתֵיהֶם – וְחָזְרוּ חֲזָרָה לְתוֹךְ…”?

4. הבנה והעמקה

ב. הסבירו במילים שלכם את המשפט: “כָּל אֶבֶן הָיְתָה קְדוֹשָׁה, כָּל פֶּרַח הֵרִיחַ מִקְּדֻשָּׁה.”.
ג. אילו רבי אלעזר בן שמוע לא היה פועל כפי שמתואר ב־“הֵם זָכְרוּ שֶׁאָדָם שֶׁנֶּעֱצָב כָּל כָּךְ עַל פְּגִיעָה בַּקֹּדֶשׁ, קוֹרֵעַ אֶת בִּגְדּוֹ.” — כיצד היה הסיפור עשוי להסתיים?
