כלה נאה וחסודה

דף עבודה בעקבות רבי יהודה בר אלעאי
שם: כיתה:

המעשה — הסיפור המלא

להלן עיבוד חינוכי, רגשי ומרתק לסיפורו של רבי יהודה בר אלעאי – כלה נאה וחסודה . הסיפור נבנה בקפידה עבור בנים בכיתות ב'-ד' (חינוך דתי-לאומי), תוך שימוש בעקרונות פדגוגיים של סטוריטלינג, תיאורים חושיים, הקפדה על כבוד התורה וחכמיה, והמחשת מוסר ההשכל: לשמח כל אדם במאור פנים, גם כשלנו עצמנו חסר, ולהשתמש במילים טובות שמאירות את הלב.

הסיפור מנוקד במלואו, כנדרש.

פְּתִיחָה: תּוֹרָה מִתּוֹךְ הַדֹּחַק

יְלָדִים, אִם נַעֲצֹם אֶת עֵינֵינוּ, נוּכַל אולי לְהַרְגִּישׁ אֶת הָרוּחַ הַקְּרִירָה שֶׁל הַגָּלִיל נוֹשֶׁבֶת בֵּין קִירוֹת הָאֶבֶן שֶׁל בֵּית הַמִּדְרָשׁ הַיָּשָׁן בְּעִיר אוּשָׁא. זֶה הָיָה לִפְנֵי שָׁנִים רַבּוֹת מְאֹד, בַּיָּמִים שֶׁלְּאַחַר חֻרְבַּן בֵּית הַמִּדְרָשׁ, בְּדוֹרוֹ שֶׁל הַתַּנָּא הַקָּדוֹשׁ – רַבִּי יְהוּדָה בַּר אִלְעָאי .

רַבִּי יְהוּדָה הָיָה עֲנַק הָעֲנָקִים. כְּשֶׁהָיָה פּוֹתֵחַ אֶת פִּיו כְּדֵי לְלַמֵּד הֲלָכָה, כָּל תַּלְמִידֵי הַחֲכָמִים הָיוּ מַקְשִׁיבִים בְּשֶׁקֶט, עַד שֶׁאֶפְשָׁר הָיָה לִשְׁמֹעַ אֶת זְבוּב עוֹף בָּאֲוִיר. הַקָּדוֹשׁ בָּרוּךְ הוּא נָתַן לוֹ שֵׂכֶל מָלֵא בְּאוֹר שֶׁל חָכְמָה.

אַךְ יְלָדִים, יַחַד עִם עָשְׁרוֹ הָרוּחָנִי הֶעָצוּם, בְּבֵיתוֹ שֶׁל רַבִּי יְהוּדָה שָׂרַר עֹנִי שֶׁאֲנַחְנוּ אֲפִלּוּ לֹא מְסֻגָּלִים לְדַמְיֵן. לֹא הָיוּ לוֹ רָהִיטִים, לֹא מַאֲכָלִים טְעִימִים, וַאֲפִלּוּ... לֹא הָיוּ לוֹ מַסְפִּיק בְּגָדִים לְחִמּוּם. חֲכָמֵינוּ זִכְרוֹנָם לִבְרָכָה מְסַפְּרִים שֶׁלְּרַבִּי יְהוּדָה וּלְאִשְׁתּוֹ הַצַּדֶּקֶת הָיָה מְעִיל חֹרֶף אֶחָד יָשָׁן מִצֶּמֶר מְחֻסְפָּס. כְּשֶׁהִיא יָצְאָה לַשּׁוּק – הִיא לָבְשָׁה אוֹתוֹ, וּכְשֶׁהוּא הָלַךְ לְבֵית הַמִּדְרָשׁ – לָבַשׁ הוּא אוֹתוֹ.

אֲבָל הַאִם רַבִּי יְהוּדָה הָיָה עָצוּב? מַמָּשׁ לֹא! פָּנָיו הָיוּ זוֹרְחוֹת תָּמִיד! כְּשֶׁלָּמַד תּוֹרָה, הַשִּׂמְחָה מִלְּאָה אֶת לִבּוֹ כָּל כָּךְ, שֶׁהוּא שָׁכַח לְגַמְרֵי מֵהָרָעָב וּמֵהַקֹּר.

הַתַּהֲלוּכָה הַשְּׁקֵטָה

יוֹם אֶחָד, יָשַׁב רַבִּי יְהוּדָה בְּבֵית הַמִּדְרָשׁ וְלִמֵּד מִשְׁנָה אֶת תַּלְמִידָיו. קוֹל הַתּוֹרָה הַמָּתוֹק וְהַמְּנַגֵּן שֶׁלּוֹ יָצָא דֶּרֶךְ הַחַלּוֹן הַפָּתוּחַ אֶל הָרְחוֹב הֶעָפַר. כֻּלָּם הָיוּ שְׁקוּעִים בַּלִּמּוּד הַקָּדוֹשׁ.

וְהִנֵּה, מִחוּץ לְבֵית הַמִּדְרָשׁ, נִשְׁמְעוּ קוֹלוֹת שֶׁל צְעָדִים עַל אֲדָמַת הַכֻּרְכָּר. קְרַנְץ'... קְרַנְץ'... קְרַנְץ'...

אַךְ אֵלּוּ לֹא הָיוּ צְעָדִים רְגִילִים שֶׁל סוֹחֲרִים בַּשּׁוּק. זֶה הָיָה מִצְעָד שֶׁל חֲתֻנָּה!

הָיָה מְקֻבָּל בְּאוֹתָם יָמִים, שֶׁכַּאֲשֶׁר חָתָן וְכַלָּה יוֹצְאִים לְהִתְחַתֵּן, תּוֹשָׁבֵי הָעִיר מְלַוִּים אוֹתָם בְּתוֹפִים, בַּחֲלִילִים, בְּשִׁירָה וּבְרִקּוּדִים עַד לַחֻפָּה. הַבָּנוֹת הָיוּ שָׁרוֹת וְהַבָּנִים הָיוּ מְפַזְּזִים לִפְנֵי הַכַּלָּה. אַךְ בַּתַּהֲלוּכָה הַזּוֹ... הָיָה שֶׁקֶט עָצוּב.

הֶחָתָן וְהַכַּלָּה הָיוּ יְתוֹמִים וַעֲנִיִּים מְאֹד. לֹא הָיָה לָהֶם כֶּסֶף לִשְׂכֹּר נַגָּנִים, אוֹ לְהַזְמִין אוֹרְחִים רַבִּים וַחֲשׁוּבִים לִסְעוּדָה. לַכַּלָּה לֹא הָיְתָה שִׂמְלָה רְקוּמָה בְּחוּטֵי זָהָב וְכֶסֶף כְּמוֹ לִבְנוֹת הָעֲשִׁירִים, אֶלָּא שִׂמְלָה פְּשׁוּטָה, חֲלָקָה וּצְנוּעָה עַד מְאֹד. אִישׁ בָּרְחוֹב לֹא עָצַר לְהַבִּיט בָּהֶם. אִישׁ לֹא מָחָא כַּפַּיִם.

הַכַּלָּה הָלְכָה וְעֵינֶיהָ מֻשְׁפָּלוֹת לָאֲדָמָה. בְּלִבָּהּ חָשְׁבָה: "הַיּוֹם הוּא הַיּוֹם הַגָּדוֹל וְהַשָּׂמֵחַ בְּחַיַּי, בּוֹ אֲנִי בּוֹנָה בַּיִת נֶאֱמָן בְּיִשְׂרָאֵל, וְאַף אֶחָד אֲפִלּוּ לֹא שָׂם לֵב! אֵיזוֹ חֲתֻנָּה עֲלוּבָה יֵשׁ לִי...". דִּמְעָה קְטַנָּה שֶׁל עֶצֶב כִּמְעַט וְזָלְגָה מֵעֵינָהּ.

הַהֲדַס וְהַצַּדִּיק

רַבִּי יְהוּדָה בַּר אִלְעָאי עָצַר לְרֶגַע מִלִּמּוּדוֹ. אוֹזְנוֹ הָרְגִישָׁה שָׁמְעָה אֶת הַשֶּׁקֶט שֶׁל הַתַּהֲלוּכָה. אָדָם אֲחֵר הָיָה מַמְשִׁיךְ לִלְמֹד תּוֹרָה וְאוֹמֵר: "אֲנִי גְּדוֹל הַדּוֹר! מַה לִּי וְלַתַּהֲלוּכָה פְּשׁוּטָה זוֹ בָּרְחוֹב?".

אֲבָל רַבִּי יְהוּדָה יָדַע שֶׁהַתּוֹרָה מְלַמֶּדֶת אוֹתָנוּ לְשַׁמֵּחַ יְהוּדִי בְּיוֹם שִׂמְחָתוֹ! זֶהוּ חֶסֶד עָצוּם!

הוּא קָם מִכִּסְאוֹ, סָגַר אֶת הַסֵּפֶר וְיָצָא אֶל הַחוּץ. סָמוּךְ לְדֶלֶת בֵּית הַמִּדְרָשׁ גָּדַל שִׂיחַ יְרַקְרַק וְרֵיחָנִי מְאֹד – שִׂיחַ הֲדַס. רַבִּי יְהוּדָה קָטַף בְּיָדוֹ "בַּד שֶׁל הֲדַס" – עָנָף אֶחָד יָפֶה שֶׁל הֲדַס שֶׁהֵפִיץ רֵיחַ טוֹב וּמָתוֹק. עִם הֶעָנָף הַזֶּה בְּיָדוֹ, הוּא חָצָה אֶת הָרְחוֹב וְנֶעֱמַד רָחוֹק מְעַט מֵהַכַּלָּה וּמֵהַחָתָן.

תַּלְמִידָיו בְּבֵית הַמִּדְרָשׁ הִבִּיטוּ מֵהַחַלּוֹנוֹת כְּשֶׁהֵם הֲמוּמִים: מָה הָרַב הַגָּדוֹל שֶׁלָּנוּ עוֹשֶׂה?! הַאִם זֶה מְכֻבָּד שֶׁהָרַב בִּכְבוֹדוֹ וּבְעַצְמוֹ יַעֲמֹד בָּרְחוֹב מוּל אֲנָשִׁים פְּשׁוּטִים?

רִקּוּד לִכְבוֹד שָׁמַיִם

וְאָז זֶה קָרָה. רַבִּי יְהוּדָה בַּר אִלְעָאי – גְּדוֹל הַדּוֹר הַקָּדוֹשׁ – הֵרִים אֶת עֲנַף הַהֲדַס בָּאֲוִיר, זְקָנוֹ הַלָּבָן נָע בָּרוּחַ, וְהוּא הֵחֵל לִרְקֹד! הַצַּדִּיק קָפַץ וְהִסְתּוֹבֵב וְרָקַד לִפְנֵי הַתַּהֲלוּכָה, כְּשֶׁהוּא מְנַפְנֵף בַּהֲדַס לִכְבוֹד הַחָתָן וְהַכַּלָּה!

הוּא פָּתַח אֶת גְּרוֹנוֹ, וּבְקוֹל עָמֹק וְשָׂמֵחַ הֵחֵל לָשִׁיר מִלִּים קְדוֹשׁוֹת:

"כַּלָּה נָאָה וַחֲסוּדָה! כַּלָּה נָאָה וַחֲסוּדָה!"

(כַּלָּה נָאָה = כַּלָּה עִם מִדּוֹת טוֹבוֹת וּמַעֲשִׂים יָפִים. חֲסוּדָה = מְלֵאָה בְּחֵן וּבְחֶסֶד מֵהַקָּדוֹשׁ בָּרוּךְ הוּא).

יְלָדִים, הַכַּלָּה הָעֲנִיָּה הֵרִימָה אֶת מַבָּטָהּ בְּתַדְהֵמָה. עֲנַק הַדּוֹר, שֶׁכָּל הָעָם יָרֵא מִפָּנָיו, רוֹקֵד בְּשִׂמְחָה שֶׁל מִצְוָה, רַק בִּשְׁבִילָהּ וּבִשְׁבִיל הֶחָתָן?! הָרַב קוֹרֵא לָהּ "נָאָה וַחֲסוּדָה"? הוּא רוֹאֶה אֶת הַיֹּפִי הָרוּחָנִי שֶׁל הַמִּשְׁפָּחָה הַחֲדָשָׁה שֶׁהִיא קוֹמַמְתָּה!

דִּמְעוֹת הָעֶצֶב שֶׁל הַכַּלָּה הִתְחַלְּפוּ בְּדִמְעוֹת שִׂמְחָה. חִיּוּךְ רָחָב וְזוֹרֵחַ פָּרַץ עַל פָּנֶיהָ. לְפֶתַע, אוֹתָהּ שִׂמְלָה עֲנִיָּה וּפְשׁוּטָה הִרְגִּישָׁה לָהּ כְּמוֹ שִׂמְלַת מַלְכוּת, וְהַחֻפָּה הַשְּׁקֵטָה הָפְכָה לְחֻפָּה הַמְכֻבֶּדֶת בְּיוֹתֵר בָּעוֹלָם – כִּי הַתּוֹרָה בְּעַצְמָהּ יָצְאָה לְשַׂמֵּחַ אוֹתָם!

הַתַּלְמִידִים בְּבֵית הַמִּדְרָשׁ יָצְאוּ מִיָּד אַחֲרֵי רַבָּם, וְהִצְטָרְפוּ לַשִּׁירָה וְלַמַּעְגָּל. כָּךְ הָפְכָה תַּהֲלוּכָה דַּלָּה לְשִׂמְחָה מְאִירָה שֶׁהִגִּיעָה עַד הַשָּׁמַיִם.

תּוֹרַת בֵּית הִלֵּל

כַּאֲשֶׁר חָזַר רַבִּי יְהוּדָה לְבֵית הַמִּדְרָשׁ, מְזִיעַ אַךְ מְאֻשָּׁר, עֵינָיו בָּרְקוּ.

הוּא אָמַר לְתַלְמִידָיו: "דְּעוּ לָכֶם, בֵּית שַׁמַּאי הָיוּ אוֹמְרִים שֶׁצָּרִיךְ לָשִׁיר לַכַּלָּה מַה שֶּׁהִיא 'כַּלָּה כְּמוֹת שֶׁהִיא' – רַק מַה שֶּׁרוֹאִים בָּעֵינַיִם הָרְגִילוֹת. אֲבָל בֵּית הִלֵּל לִמְּדוּ אוֹתָנוּ שֶׁצָּרִיךְ תָּמִיד לוֹמַר 'כַּלָּה נָאָה וַחֲסוּדָה!' ".

וְלָמָּה? כִּי כְּשֶׁבּוֹחֲרִים לְהַבִּיט עַל אָדָם בְּעַיִן טוֹבָה, כְּשֶׁלּוֹקְחִים מִלָּה מְעוֹדֶדֶת, שִׁיר קָטָן אוֹ חִיּוּךְ מִכָּל הַלֵּב – רוֹאִים שֶׁכָּל כַּלָּה הִיא הֲכִי צַדֶּקֶת וַחֲסוּדָה. מִצְוָה עֲצוּמָה לְשַׂמֵּחַ יְהוּדִי, וְהַשִּׂמְחָה הַזּוֹ חֲשׁוּבָה בַּשָּׁמַיִם גַּם לִפְנֵי כְּבוֹדוֹ הָרוּחָנִי שֶׁל הָרַב אֲשֶׁר מוֹחֵל עַל כְּבוֹדוֹ לְמַעַן הַחֶסֶד.

א. שְׁאֵלַת פְּתִיחָה:

"יְלָדִים, נַסּוּ לְדַמְיֵן שֶׁלְּפֶתַע אַתֶּם הוֹלְכִים בָּרְחוֹב, וְרוֹאִים אֶת רַב הָעִיר הַגָּדוֹל שֶׁלָּנוּ – זֶה שֶׁכֻּלָּם מְכַבְּדִים אוֹתוֹ עַד מְאֹד – בָּאֲמְצַע הַכְּבִישׁ, נוֹטֵל עָנָף, רוֹקֵד וְשָׁר בְּקוֹל רָם... הַאִם זֶה נִרְאֶה לָכֶם הֶגְיוֹנִי?

הַיּוֹם נִלְמַד עַל אֶחָד מִגְּדוֹלֵי יִשְׂרָאֵל שֶׁעָשָׂה בְּדִיּוּק אֶת זֶה, וְנָבִין – לָמָּה!"

ב. הֶעָרוֹת בִּצּוּעַ וּסְטוֹרִיטֶלִינְג (Storytelling):

ג. הַמְלָצוֹת לְמַצֶּגֶת (תְּמוּנוֹת צְנוּעוֹת וּקְדוֹשׁוֹת):

ד. פְּעִילוּת מְסַכֶּמֶת (הַפְנָמָה):

מקורות: מַסֶּכֶת כְּתֻבּוֹת י"ז ע"א: "אָמְרוּ עָלָיו עַל רַבִּי יְהוּדָה בַּר אִלְעָאי, שֶׁהָיָה נוֹטֵל בַּד שֶׁל הֲדַס וּמְרַקֵּד לִפְנֵי הַכַּלָּה, וְאוֹמֵר: כַּלָּה נָאָה וַחֲסוּדָה!" + מַחֲלֹקֶת בֵּית שַׁמַּאי וּבֵית הִלֵּל לְגַבֵּי "כֵּיצַד מְרַקְּדִין לִפְנֵי הַכַּלָּה". מַסֶּכֶת נְדָרִים מ"ט ע"ב - נ' ע"א: עַל עֲנִיּוּתוֹ הַמֻּפְלֶגֶת שֶׁל רַבִּי יְהוּדָה, שֶׁהָיוּ הוּא וְאִשְׁתּוֹ חוֹלְקִים מְעִיל אֶחָד (וְעוֹד מְקוֹרוֹת בַּגְּמָרָא הַמְּעִידִים עַל גַּדְלוּתוֹ הָרוּחָנִית לְעֻמַּת דַּחְקוּתוֹ הַחָמְרִית). סֵפֶר הָאַגָּדָה: סִימָן רס"ז, "מְשַׂמֵּחַ אֲחֵרִים".

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “להלן עיבוד חינוכי, רגשי ומרתק לסיפורו של רבי יהודה בר אלעאי – כלה נאה וחסודה .” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “חֲכָמֵינוּ זִכְרוֹנָם לִבְרָכָה מְסַפְּרִים שֶׁלְּרַבִּי יְהוּדָה וּלְאִשְׁתּוֹ הַצַּדֶּקֶת הָיָה מְעִיל חֹרֶף אֶחָד יָשָׁן מִצֶּמֶר מְחֻסְפָּס.”?
ג. כיצד הגיב רבי יהודה בר אלעאי למצב המתואר: “הַבָּנוֹת הָיוּ שָׁרוֹת וְהַבָּנִים הָיוּ מְפַזְּזִים לִפְנֵי הַכַּלָּה.”?
ד. מה השתנה בין “סָמוּךְ לְדֶלֶת בֵּית הַמִּדְרָשׁ גָּדַל שִׂיחַ יְרַקְרַק וְרֵיחָנִי מְאֹד – שִׂיחַ הֲדַס.” לבין “חֲסוּדָה = מְלֵאָה בְּחֵן וּבְחֶסֶד מֵהַקָּדוֹשׁ בָּרוּךְ הוּא).”?
ה. כיצד הגיע הסיפור אל האירוע: “אֲבָל בֵּית הִלֵּל לִמְּדוּ אוֹתָנוּ שֶׁצָּרִיךְ תָּמִיד לוֹמַר 'כַּלָּה נָאָה וַחֲסוּדָה!' ".”?

3. מה היה ראשון?

כתבו ליד כל אירוע מספר מ־1 עד 5 לפי סדר התרחשותו בסיפור.

□ אֲבָל רַבִּי יְהוּדָה יָדַע שֶׁהַתּוֹרָה מְלַמֶּדֶת אוֹתָנוּ לְשַׁמֵּחַ יְהוּדִי בְּיוֹם שִׂמְחָתוֹ!
□ להלן עיבוד חינוכי, רגשי ומרתק לסיפורו של רבי יהודה בר אלעאי – כלה נאה וחסודה .
□ הַיּוֹם נִלְמַד עַל אֶחָד מִגְּדוֹלֵי יִשְׂרָאֵל שֶׁעָשָׂה בְּדִיּוּק אֶת זֶה, וְנָבִין – לָמָּה!
□ יוֹם אֶחָד, יָשַׁב רַבִּי יְהוּדָה בְּבֵית הַמִּדְרָשׁ וְלִמֵּד מִשְׁנָה אֶת תַּלְמִידָיו.
□ הוּא רוֹאֶה אֶת הַיֹּפִי הָרוּחָנִי שֶׁל הַמִּשְׁפָּחָה הַחֲדָשָׁה שֶׁהִיא קוֹמַמְתָּה!

לפני ואחרי

מה קרה לפני: “אֲבָל רַבִּי יְהוּדָה יָדַע שֶׁהַתּוֹרָה מְלַמֶּדֶת אוֹתָנוּ לְשַׁמֵּחַ יְהוּדִי בְּיוֹם שִׂמְחָתוֹ!”?
מה קרה אחרי: “הוּא רוֹאֶה אֶת הַיֹּפִי הָרוּחָנִי שֶׁל הַמִּשְׁפָּחָה הַחֲדָשָׁה שֶׁהִיא קוֹמַמְתָּה!”?

סיבה ותוצאה מתוך המעשה

האירוע הנתוןמה גרם לו?מה הייתה תוצאתו?
קוֹל הַתּוֹרָה הַמָּתוֹק וְהַמְּנַגֵּן שֶׁלּוֹ יָצָא דֶּרֶךְ הַחַלּוֹן הַפָּתוּחַ אֶל הָרְחוֹב הֶעָפַר.
רַבִּי יְהוּדָה בַּר אִלְעָאי – גְּדוֹל הַדּוֹר הַקָּדוֹשׁ – הֵרִים אֶת עֲנַף הַהֲדַס בָּאֲוִיר, זְקָנוֹ הַלָּבָן נָע בָּרוּחַ, וְהוּא הֵחֵל לִרְקֹד!

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “כלה נאה וחסודה” על רבי יהודה בר אלעאי? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “בְּלִבָּהּ חָשְׁבָה: "הַיּוֹם הוּא הַיּוֹם הַגָּדוֹל וְהַשָּׂמֵחַ בְּחַיַּי, בּוֹ אֲנִי בּוֹנָה בַּיִת נֶאֱמָן בְּיִשְׂרָאֵל, וְאַף אֶחָד אֲפִלּוּ לֹא…”.
ג. אילו רבי יהודה בר אלעאי לא היה פועל כפי שמתואר ב־“חֲסוּדָה = מְלֵאָה בְּחֵן וּבְחֶסֶד מֵהַקָּדוֹשׁ בָּרוּךְ הוּא).” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “אֲבָל בֵּית הִלֵּל לִמְּדוּ אוֹתָנוּ שֶׁצָּרִיךְ תָּמִיד לוֹמַר 'כַּלָּה נָאָה וַחֲסוּדָה!' ".” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. להלן עיבוד חינוכי, רגשי ומרתק לסיפורו של רבי יהודה בר אלעאי – כלה נאה…
2. כְּשֶׁהִיא יָצְאָה לַשּׁוּק – הִיא לָבְשָׁה אוֹתוֹ, וּכְשֶׁהוּא הָלַךְ…
3. לֹא הָיָה לָהֶם כֶּסֶף לִשְׂכֹּר נַגָּנִים, אוֹ לְהַזְמִין אוֹרְחִים…
4. עִם הֶעָנָף הַזֶּה בְּיָדוֹ, הוּא חָצָה אֶת הָרְחוֹב וְנֶעֱמַד רָחוֹק…
5. הַתַּלְמִידִים בְּבֵית הַמִּדְרָשׁ יָצְאוּ מִיָּד אַחֲרֵי רַבָּם,…
6. הַיּוֹם נִלְמַד עַל אֶחָד מִגְּדוֹלֵי יִשְׂרָאֵל שֶׁעָשָׂה בְּדִיּוּק…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “הַבָּנוֹת הָיוּ שָׁרוֹת וְהַבָּנִים הָיוּ מְפַזְּזִים לִפְנֵי הַכַּלָּה.”.
מכתב: כתבו לרבי יהודה בר אלעאי בעקבות “רַבִּי יְהוּדָה בַּר אִלְעָאי – גְּדוֹל הַדּוֹר הַקָּדוֹשׁ – הֵרִים אֶת עֲנַף הַהֲדַס בָּאֲוִיר, זְקָנוֹ הַלָּבָן נָע בָּרוּחַ,…”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “כָּךְ הָפְכָה תַּהֲלוּכָה דַּלָּה לְשִׂמְחָה מְאִירָה שֶׁהִגִּיעָה עַד הַשָּׁמַיִם.”.
דיווח: כתבו ידיעה קצרה שכותרתה “כלה נאה וחסודה”.
