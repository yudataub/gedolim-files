כגל של אגוזים

דף עבודה בעקבות רבי טרפון
שם: כיתה:

המעשה — הסיפור המלא

פְּתִיחָה: הָעֲרֵמָה שֶׁבַּשּׁוּק

יְלָדִים, הַאִם פַּעַם הֲלַכְתֶּם עִם אַבָּא לַשּׁוּק לִקְרַאת שַׁבָּת?

דַּמְיְנוּ אֶת הָרַעַשׁ שֶׁל הַשּׁוּק: הַסּוֹחֲרִים צוֹעֲקִים, עֲגָלוֹת עֵץ נוֹסְעוֹת עַל אַבְנֵי הָרְחוֹב הַמְחֻסְפָּסוֹת, וְרֵיחַ מָתוֹק שֶׁל פֵּרוֹת בְּשֵׁלִים וְתַבְלִינִים חֲרִיפִים מְמַלֵּא אֶת הָאֲוִיר.

בְּפִנַּת הַשּׁוּק עוֹמֵד דּוּכָן וּבוֹ מַשֶּׁהוּ מְיֻחָד בְּמִינוֹ. אֵין שָׁם תַּפּוּחִים אוֹ תְּמָרִים, אֶלָּא... אֱגוֹזִים! וְלֹא סְתָם קֻפְסָה שֶׁל אֱגוֹזִים, אֶלָּא הַר עָצוּם, פִּירָמִידָה עֲנָקִית וְגָבוֹהָה שֶׁל אֱגוֹזֵי מֶלֶךְ צְהַבְהַבִּים. הֵם קָשִׁים וַחֲלָקִים לְמַגָּע, וּמְסֻדָּרִים זֶה עַל גַּבֵּי זֶה בְּסֵדֶר מוֹפְתִי, מַמָּשׁ עַד לַתִּקְרָה.

יֶלֶד אֶחָד שֶׁעָבַר דֶּרֶךְ הַשּׁוּק, הוֹשִׁיט אֶת יָדוֹ וְנִסָּה לְהוֹצִיא מִלְּמַטָּה רַק אֱגוֹז אֶחָד קָטָן.

אֲבָל... מָה קָרָה בָּרֶגַע שֶׁהוּא מָשַׁךְ אֶת הָאֱגוֹז הָאֶחָד הַזֶּה?

טְרַאח! טְרַרַאח! דּוּם!

כָּל הָאֱגוֹזִים שֶׁהָיוּ לְמַעְלָה אִבְּדוּ אֶת שִׁוּוּי הַמִּשְׁקָל שֶׁלָּהֶם. מֵאוֹת אֱגוֹזִים הִתְחִילוּ לִפֹּל, לְהִתְגַּלְגֵּל, לְהִתְנַגֵּשׁ זֶה בָּזֶה וּלְקַשְׁקֵשׁ בְּרַעַשׁ מְשַׂמֵּחַ, עַד שֶׁנָּפְלוּ כֻּלָּם לְתוֹךְ יָדָיו הַפְּתוּחוֹת שֶׁל הַיֶּלֶד! יַחַד עִם אֱגוֹז אֶחָד, הוּא קִבֵּל שֶׁפַע גָּדוֹל, עֲרֵמָה שְׁלֵמָה שֶׁל אֱגוֹזִים וּבְרָכָה.

בֵּית הַמִּדְרָשׁ בָּעִיר לוֹד

שָׁנִים רַבּוֹת לְאַחַר חֻרְבַּן בֵּית הַמִּקְדָּשׁ, בָּעִיר דְּרוֹמִית לִירוּשָׁלַיִם הַנִּקְרֵאת לוֹד, הָיָה בַּיִת אֶחָד שֶׁמִּמֶּנּוּ בָּקַע אוֹר עָצוּם. זֶהוּ בֵּית מִדְרָשׁוֹ שֶׁל הַתַּנָּא הַקָּדוֹשׁ – רַבִּי טַרְפוֹן .

רַבִּי טַרְפוֹן הָיָה מֵהָאֲנָשִׁים הַגְּדוֹלִים בְּיוֹתֵר בְּעַם יִשְׂרָאֵל. הוּא הָיָה כֹּהֵן, וּכְשֶׁהָיָה יֶלֶד קָטָן עוֹד זָכָה לִרְאוֹת אֶת בֵּית הַמִּקְדָּשׁ הַשֵּׁנִי בְּתִפְאַרְתּוֹ, וְאַף לִשְׁמֹעַ אֶת הַכֹּהֵן הַגָּדוֹל. עַתָּה הוּא זָקֵן, זְקָנוֹ לָבָן וְרַךְ כְּצֶמֶר צַח, עֵינָיו מְאִירוֹת כְּכוֹכָבִים, וְלִבּוֹ... לִבּוֹ הוּא יָם שֶׁל חֶסֶד וְאַהֲבַת תּוֹרָה. רַבִּי טַרְפוֹן אָהַב כָּל יְהוּדִי כְּמוֹ בֵּן, עַד שֶׁחֲכָמִים קָרְאוּ לוֹ בְּאַהֲבָה: "אֲבִיהֶם שֶׁל יִשְׂרָאֵל".

יוֹם אֶחָד, יָשַׁב בְּבֵית הַמִּדְרָשׁ תַּלְמִיד חָכָם צָעִיר. אוּלַי קָרְאוּ לוֹ יְהוּדָה, אוּלַי שִׁמְעוֹן. הַתַּלְמִיד הַזֶּה קָרָא פָּסוּק בַּתּוֹרָה, אֲבָל... מַשֶּׁהוּ שָׁם לֹא הָיָה לוֹ בָּרוּר. קַו רֶכֶס כִּוֵּץ אֶת מִצְחוֹ. הוּא גֵּרֵד בְּפַדַּחְתּוֹ וְחָשַׁב. הוּא שָׁאַל אֶת חֲבֵרוֹ לַסַּפְסָל הֶעָשׂוּי מֵעֵץ עַתִּיק, אַךְ גַּם לַחֲבֵרוֹ לֹא הָיְתָה תְּשׁוּבָה.

"אֲנִי חַיָּב לִשְׁאֹל אֶת רַבִּי טַרְפוֹן," הֶחְלִיט הַתַּלְמִיד הַצָּעִיר, אֲבָל בִּטְנוֹ קִרְקְרָה מִמְּעַט פַּחַד. "רַבִּי טַרְפוֹן הוּא מִגְּדוֹלֵי הַדּוֹר. הוּא כָּל כָּךְ חָכָם... בֶּטַח הַשְּׁאֵלָה שֶׁלִּי הִיא קְטַנָּה וּפְשׁוּטָה מִדַּי. אוּלַי הוּא יַעֲנֶה לִי רַק כֵּן אוֹ לֹא, וְאֶשָּׁאֵר נָבוֹךְ."

מֵרָחוֹק הוּא הֵרִיחַ אֶת רֵיחַ קְלַף רַבּוֹתֵינוּ שֶׁמִּלֵּא אֶת בֵּית הַמִּדְרָשׁ. הוּא לָקַח נְשִׁימָה עֲמֻקָּה וְקָרַב אֶל הָרַב.

"רַבִּי," אָמַר הַתַּלְמִיד בְּקוֹל שָׁקֵט, "שְׁנֵה לִי. אֲנִי מְבַקֵּשׁ שֶׁתְּלַמֵּד אוֹתִי שְׁאֵלָה שֶׁיֵּשׁ לִי עַל הַמִּשְׁנָה הַזּוֹ."

מַפָּל הַתּוֹרָה

יְלָדִים, מָה עָשָׂה רַבִּי טַרְפוֹן? הַאִם עָנָה לוֹ רַק תְּשׁוּבָה קְצָרָה שֶׁל שְׁתֵּי מִלִּים?

לֹא! לֹא וָלֹא!

עֵינָיו שֶׁל רַבִּי טַרְפוֹן נִדְלְקוּ. קוֹלוֹ צִלְצֵל בְּבֵית הַמִּדְרָשׁ כְּמוֹ נִגּוּן מָתוֹק וּמְלֵא שִׂמְחָה. עֲבוּר רַבִּי טַרְפוֹן, שְׁאֵלָה בַּתּוֹרָה הָיְתָה כְּמוֹ מַפְתֵּחַ לְאוֹצָר אֲדִיר.

"בְּנִי הַיָּקָר!" עָנָה רַבִּי טַרְפוֹן בְּחִיּוּךְ רָחָב, "הַמִּשְׁנָה הַזּוֹ שֶׁשָּׁאַלְתָּ? הִיא הֲרֵי קְשׁוּרָה לְפָסוּק בְּסֵפֶר וַיִּקְרָא! וּבְוַדַּאי אַתָּה זוֹכֵר שֶׁיֵּשׁ הֲלָכָה מְדֻיֶּקֶת בְּהִלְכוֹת שַׁבָּת שֶׁלּוֹמְדִים מִכָּאן... וּבוֹא וְאֶקְרָא לְךָ אֶת מַאֲמַר חֲזַ"ל מֵהַמִּדְרָשׁ, וְגַם אַגָּדָה נִפְלָאָה עַל כָּךְ שֶׁמּוֹסִיפָה קְדֻשָּׁה לַנְּשָׁמָה!"

וְכָךְ, מוּל תַּדְהַמְתּוֹ שֶׁל הַתַּלְמִיד הַצָּעִיר, שָׁפַךְ רַבִּי טַרְפוֹן אוֹצָרוֹת. פָּסוּק הִתְחַבֵּר לַהֲלָכָה, וַהֲלָכָה הִתְחַבְּרָה לְמִדְרָשׁ, וּמִדְרָשׁ לְסִפּוּר מֵרַבּוֹתֵינוּ. מִלִּים קְדוֹשׁוֹת יָצְאוּ מִפִּיו שֶׁל רַבִּי טַרְפוֹן כְּמוֹ מַיִם רַבִּים הַזּוֹרְמִים מִמַּעְיָן לֹא פוֹסֵק.

הַתַּלְמִיד יָשַׁב מְרֻתָּק. אָזְנָיו כְּרוּיוֹת, עֵינָיו פְּעוּרוֹת מִשִּׂמְחָה. הַתְּשׁוּבָה שֶׁל רַבִּי טַרְפוֹן הָיְתָה טְעִימָה יוֹתֵר מִדְּבַשׁ. בִּמְקוֹם לָצֵאת מִבֵּית הַמִּדְרָשׁ רַק עִם תְּשׁוּבָה שֶׁל "כֵּן" אוֹ "לֹא", הוּא יָצָא כְּשֶׁלִּבּוֹ וּמוֹחוֹ מְלֵאִים וּגְדוּשִׁים בִּבְרָכָה, בְּחָכְמָה עֲצוּמָה וּבְאַהֲבַת הַשֵּׁם!

הָעֵדוּת שֶׁל רַבִּי יְהוּדָה הַנָּשִׂיא

יְלָדִים, שָׁנִים רַבּוֹת אַחַר כָּךְ, כְּשֶׁרַבִּי יְהוּדָה הַנָּשִׂיא (שֶׁאֲנַחְנוּ קוֹרְאִים לוֹ פָּשׁוּט "רַבִּי") חָשַׁב עַל הַחֲכָמִים הַגְּדוֹלִים שֶׁהִכִּיר, הוּא נִזְכַּר בְּרַבִּי טַרְפוֹן הַצַּדִּיק.

"לְמָה הָיָה דּוֹמֶה רַבִּי טַרְפוֹן?" שָׁאַל רַבִּי אֶת תַּלְמִידָיו, עֵינָיו בּוֹרְקוֹת מִגַּעְגּוּעַ וְהַעֲרָצָה.

"אֲנִי אֶגְלֶה לָכֶם: רַבִּי טַרְפוֹן הָיָה דּוֹמֶה... לְגַל שֶׁל אֱגוֹזִים!" (גַּל = עֲרֵמָה).

הַתַּלְמִידִים תָּמְהוּ: "לַעֲרֵמָה שֶׁל אֱגוֹזִים? רַבִּי, מַדּוּעַ אֵינְךָ מְשַׁוֶּה אוֹתוֹ לְתֵבַת יַהֲלוֹמִים? לְנָהָר זוֹרֵם?"

וְאָז, הִסְבִּיר הַנָּשִׂיא הַקָּדוֹשׁ אֶת הַסּוֹד הַנִּפְלָא: "אִם יֵשׁ לָכֶם עֲרֵמָה גְּדוֹלָה שֶׁל אֱגוֹזִים, וְאַתֶּם רַק נוֹגְעִים בֶּאֱגוֹז אֶחָד מִלְּמַטָּה, מָה קוֹרֶה? הַאִם רַק הוּא יוֹצֵא? מַמָּשׁ לֹא. כָּל הָאֱגוֹזִים בָּעֲרֵמָה מִתְקַשְׁקְשִׁים, מִתְגַּלְגְּלִים זֶה עַל גַּבֵּי זֶה, וְנוֹפְלִים עֲלֵיכֶם מִכָּל הַכִּוּוּנִים הֲמוֹן אֱגוֹזִים! כָּךְ, בְּדִיּוּק כָּךְ, הָיָה רַבִּי טַרְפוֹן! כַּאֲשֶׁר תַּלְמִיד חָכָם צָעִיר הָיָה בָּא אֵלָיו וְשׁוֹאֵל אוֹתוֹ: 'רַבִּי, שְׁנֵה לִי', הוּא רַק נָגַע בָּאֱגוֹז הָאֶחָד. אֲבָל רַבִּי טַרְפוֹן מִיָּד הָיָה מֵבִיא לוֹ הֲלָכוֹת שֶׁל פְּסוּקִים, הֲלָכוֹת שֶׁל מִשְׁנָה, אֲגָדוֹת, וּמִדְרָשִׁים... הַתּוֹרָה כֻּלָּהּ הָיְתָה קְשׁוּרָה אֶצְלוֹ לְדָבָר אֶחָד שָׁלֵם. הַתַּלְמִיד הָיָה יוֹצֵא מִמֶּנּוּ כְּשֶׁהוּא מָלֵא בִּבְרָכָה וָטוֹב!"

יְלָדִים, רַבִּי טַרְפוֹן לֹא שָׁמַר אֶת הַיֶּדַע לְעַצְמוֹ, הוּא לֹא הִסְתֵּיר אֶת כֹּחַ הַתּוֹרָה שֶׁלּוֹ. בְּכָל פַּעַם שֶׁמִּישֶׁהוּ בִּקֵּשׁ קְצָת אוֹר, רַבִּי טַרְפוֹן פָּתַח אֶת דַּלְתוֹת הַתּוֹרָה כֻּלָּן, וְהֵפִיץ אֶת הָאוֹר לְכָל הָעוֹלָם, כְּעֲרֵמָה אֲדִירָה שֶׁל אֱגוֹזִים הַמִּתְגַּלְגְּלִים בְּשִׂמְחָה לְתוֹךְ יָדֵינוּ הַמּוּשָׁטוֹת.

מקורות: אבות דרבי נתן, פרק י"ח, משנה א': "רבי טרפון למה הוא דומה? לגל של אגוזים, כיוון שנוטל אדם אחד מהן, כולן מתקשקשין ובאין זה על זה. כך היה רבי טרפון דומה: בשעה שתלמיד חכם נכנס אצלו ואמר לו שְנֵה לי! מביא לו מקרא ומשנה מדרש הלכות והגדות, כיוון שיצא מלפניו היה יוצא מלא ברכה וטוב." גיטין ס"ז ע"א: תיאורם של חכמים ותכונותיהם מפי רבי יהודה הנשיא.

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “דַּמְיְנוּ אֶת הָרַעַשׁ שֶׁל הַשּׁוּק: הַסּוֹחֲרִים צוֹעֲקִים, עֲגָלוֹת עֵץ נוֹסְעוֹת עַל אַבְנֵי הָרְחוֹב הַמְחֻסְפָּסוֹת, וְרֵיחַ מָתוֹק שֶׁל פֵּרוֹת…” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “יַחַד עִם אֱגוֹז אֶחָד, הוּא קִבֵּל שֶׁפַע גָּדוֹל, עֲרֵמָה שְׁלֵמָה שֶׁל אֱגוֹזִים וּבְרָכָה.”?
ג. כיצד הגיב רבי טרפון למצב המתואר: “רַבִּי טַרְפוֹן אָהַב כָּל יְהוּדִי כְּמוֹ בֵּן, עַד שֶׁחֲכָמִים קָרְאוּ לוֹ בְּאַהֲבָה: "אֲבִיהֶם שֶׁל יִשְׂרָאֵל".”?
ד. מה השתנה בין “קוֹלוֹ צִלְצֵל בְּבֵית הַמִּדְרָשׁ כְּמוֹ נִגּוּן מָתוֹק וּמְלֵא שִׂמְחָה.” לבין “מִלִּים קְדוֹשׁוֹת יָצְאוּ מִפִּיו שֶׁל רַבִּי טַרְפוֹן כְּמוֹ מַיִם רַבִּים הַזּוֹרְמִים מִמַּעְיָן לֹא פוֹסֵק.”?
ה. כיצד הגיע הסיפור אל האירוע: “הַתּוֹרָה כֻּלָּהּ הָיְתָה קְשׁוּרָה אֶצְלוֹ לְדָבָר אֶחָד שָׁלֵם.”?

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “כגל של אגוזים” על רבי טרפון? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “אֲנִי חַיָּב לִשְׁאֹל אֶת רַבִּי טַרְפוֹן," הֶחְלִיט הַתַּלְמִיד הַצָּעִיר, אֲבָל בִּטְנוֹ קִרְקְרָה מִמְּעַט פַּחַד.”.
ג. אילו רבי טרפון לא היה פועל כפי שמתואר ב־“מִלִּים קְדוֹשׁוֹת יָצְאוּ מִפִּיו שֶׁל רַבִּי טַרְפוֹן כְּמוֹ מַיִם רַבִּים הַזּוֹרְמִים מִמַּעְיָן לֹא פוֹסֵק.” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “הַתּוֹרָה כֻּלָּהּ הָיְתָה קְשׁוּרָה אֶצְלוֹ לְדָבָר אֶחָד שָׁלֵם.” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. דַּמְיְנוּ אֶת הָרַעַשׁ שֶׁל הַשּׁוּק: הַסּוֹחֲרִים צוֹעֲקִים, עֲגָלוֹת…
2. יַחַד עִם אֱגוֹז אֶחָד, הוּא קִבֵּל שֶׁפַע גָּדוֹל, עֲרֵמָה שְׁלֵמָה…
3. יוֹם אֶחָד, יָשַׁב בְּבֵית הַמִּדְרָשׁ תַּלְמִיד חָכָם צָעִיר.
4. וּבְוַדַּאי אַתָּה זוֹכֵר שֶׁיֵּשׁ הֲלָכָה מְדֻיֶּקֶת בְּהִלְכוֹת…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “רַבִּי טַרְפוֹן אָהַב כָּל יְהוּדִי כְּמוֹ בֵּן, עַד שֶׁחֲכָמִים קָרְאוּ לוֹ בְּאַהֲבָה: "אֲבִיהֶם שֶׁל יִשְׂרָאֵל".”.
מכתב: כתבו לרבי טרפון בעקבות “וּבוֹא וְאֶקְרָא לְךָ אֶת מַאֲמַר חֲזַ"ל מֵהַמִּדְרָשׁ, וְגַם אַגָּדָה נִפְלָאָה עַל כָּךְ שֶׁמּוֹסִיפָה קְדֻשָּׁה לַנְּשָׁמָה!”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “יְלָדִים, שָׁנִים רַבּוֹת אַחַר כָּךְ, כְּשֶׁרַבִּי יְהוּדָה הַנָּשִׂיא (שֶׁאֲנַחְנוּ קוֹרְאִים לוֹ פָּשׁוּט "רַבִּי") חָשַׁב עַל…”.
דיווח: כתבו ידיעה קצרה שכותרתה “כגל של אגוזים”.

המחזה
