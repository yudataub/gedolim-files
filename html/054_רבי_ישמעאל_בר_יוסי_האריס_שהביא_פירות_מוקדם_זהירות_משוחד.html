האריס שהביא פירות מוקדם — זהירות משוחד

דף עבודה בעקבות רבי ישמעאל ב"ר יוסי
שם: כיתה:

המעשה — הסיפור המלא

להלן מערך שיעור וסיפור חינוכי, רגשי ומרתק על רבי ישמעאל ב"ר יוסי והאריס (החקלאי) שלו . הסיפור נבנה ועובד בקפידה עבור בנים בכיתות ב'-ד' (חינוך דתי-לאומי), תוך שימוש בעקרונות הסטוריטלינג (Show, Don't Tell), שילוב תיאורים חושיים, והמחשת מוסר ההשכל הטהור: זהירות מוחלטת משוחד וממשפט מעוות, ואפילו מהטיה דקה שבדקות שבלב.

הסיפור מנוקד במלואו כנדרש.

פְּתִיחָה: רֵיחוֹ שֶׁל יוֹם שִׁשִּׁי

יְלָדִים, אֵיזֶה רֵיחַ מַזְכִּיר לָכֶם אֶת יוֹם שִׁשִּׁי? אוּלַי רֵיחַ שֶׁל חַלּוֹת טְרִיּוֹת שֶׁיּוֹצְאוֹת מֵהַתַּנּוּר? אוּלַי רֵיחַ שֶׁל רִצְפָּה נְקִיָּה שֶׁנִּשְׁטְפָה לִכְבוֹד שַׁבָּת?

אֵצֶל רַבִּי יִשְׁמָעֵאל בְּרַבִּי יוֹסֵי, גְּדוֹל הַדַּיָּנִים שֶׁל עַם יִשְׂרָאֵל, רֵיחוֹ שֶׁל יוֹם שִׁשִּׁי הָיָה רֵיחַ מָתוֹק וְרַעֲנָן שֶׁל פֵּרוֹת הַדָּר, תְּאֵנִים וַעֲנָבִים.

מַדּוּעַ? לְרַבִּי יִשְׁמָעֵאל הָיָה פַּרְדֵּס גָּדוֹל. אַךְ מִכֵּיוָן שֶׁהָרַב הָיָה שָׁקוּעַ יוֹמָם וָלַיְלָה בְּלִמּוּד הַתּוֹרָה וּבִפְסִיקַת הֲלָכוֹת בְּבֵית הַדִּין, לֹא הָיָה לוֹ זְמַן לְטַפֵּל בָּעֵצִים. לָכֵן, הָיָה לוֹ "אָרִיס". אָרִיס הוּא אָדָם שֶׁעוֹבֵד בַּשָּׂדֶה, מַשְׁקֶה, עוֹדֵר וְקוֹטֵף, וּבִתְמוּרָה לְעֲבוֹדָתוֹ הוּא מְקַבֵּל חֵלֶק מֵהַפֵּרוֹת, וְאֶת הַשְּׁאָר הוּא מֵבִיא לְבַעַל הַשָּׂדֶה.

בְּכָל יוֹם שִׁשִּׁי בַּבֹּקֶר, הָיָה הָאָרִיס מַגִּיעַ, תּוֹפֵס בְּיָדָיו סַלְסִלַּת קַשׁ עֲנָקִית, מְחֻסְפֶּסֶת וּכְבֵדָה, עֲמוּסָה בְּפֵרוֹת מַבְרִיקִים מִטַּל הַבֹּקֶר, וּמַנִּיחַ אוֹתָהּ עַל שֻׁלְחָנוֹ שֶׁל רַבִּי יִשְׁמָעֵאל לִכְבוֹד שַׁבָּת קֹדֶשׁ.

הַפֵּרוֹת הָאֵלּוּ הָיוּ שֶׁלּוֹ. שֶׁל הָרַב. הָאָרִיס רַק עָשָׂה עֲבוּרוֹ אֶת הַשְּׁלִיחוּת. כָּךְ הָיָה בְּכָל שָׁבוּעַ. בְּקֶבַע. בְּלִי שִׁנּוּיִים.

הַהַפְתָּעָה שֶׁל יוֹם חֲמִישִּׁי

וְהִנֵּה, בֹּקֶר אֶחָד, הָיָה זֶה יוֹם חֲמִישִּׁי. בֵּית הַמִּדְרָשׁ הָיָה מָלֵא בְּקוֹל הֲמוֹן לוֹמְדִים. קוֹלוֹת "אָמַר אַבַּיֵּי" וְ"אָמַר רָבָא" מִלְּאוּ אֶת הָאֲוִיר. רַבִּי יִשְׁמָעֵאל יָשַׁב עַל כִּסֵּא הַדַּיָּן שֶׁלּוֹ, מוּכָן לְקַבֵּל אֶת בַּעֲלֵי הַדִּין שֶׁבָּאוּ לְהִשָּׁפֵט.

לְפֶתַע... טּוּק טּוּק טּוּק . קוֹל דְּפִיקָה עַל דֶּלֶת הָעֵץ הַכְּבֵדָה.

בַּפֶּתַח עָמַד... הָאָרִיס! וּבְיָדָיו, סַלְסִלַּת הַקַּשׁ הָעֲמוּסָה בְּפֵרוֹת טְרִיִּים שֶׁרֵיחָם הַמָּתוֹק כְּבָר נִשָּׂא בַּחֶדֶר. הוּא פָּסַע הַפְּנִימָה וְהִנִּיחַ אֶת הַסַּלְסִלָּה עַל הַשֻּׁלְחָן כְּשֶׁהוּא מִתְנַשֵּׁף מֵהַמַּשָּׂא.

רַבִּי יִשְׁמָעֵאל הֵרִים אֶת מַבָּטוֹ מֵהַסֵּפֶר, גַּבּוֹתָיו מְרֻמּוֹת בִּפְלִיאָה. "מַה זֶּה?" שָׁאַל בְּקוֹל עָמֹק וְשָׁקֵט, "הֲרֵי תָּמִיד אַתָּה מֵבִיא אֶת יְבוּל הַשָּׂדֶה מָחָר, בְּיוֹם שִׁשִּׁי עֶרֶב שַׁבָּת. מַדּוּעַ הִקְדַּמְתָּ לְיוֹם חֲמִישִּׁי?".

הָאָרִיס חִיֵּךְ חִיּוּךְ מִתְנַצֵּל, מָחָה אֶת אֶגְלֵי הַזֵּעָה מִמִּצְחוֹ וְאָמַר: "רַבִּי יִשְׁמָעֵאל הַקָּדוֹשׁ, בְּעֶזְרַת הַשֵּׁם יֵשׁ לִי הַיּוֹם 'דִּין תּוֹרָה'. יֵשׁ לִי מִשְׁפָּט אֵצֶל כְּבוֹדוֹ, וִכּוּחַ מָמוֹנִי עִם אָדָם אַחֵר. אָמַרְתִּי לְעַצְמִי – אִם אֲנִי כְּבָר בָּא לָעִיר בְּיוֹם חֲמִישִּׁי לַמִּשְׁפָּט, עָדִיף שֶׁאָבִיא אֶת הַפֵּרוֹת שֶׁל הָרַב "עַל הַדֶּרֶךְ", כְּדֵי שֶׁלֹּא אֶצְטָרֵךְ לַעֲשׂוֹת אֶת הַמַּסָּע זֶה שׁוּב מָחָר!".

כְּמוֹ מִפְּנֵי אֵשׁ

יְלָדִים, בּוֹאוּ נַעֲצֹר רֶגַע. הַאִם הָאָרִיס נָתַן לָרַב מַתָּנָה? לֹא! הַפֵּרוֹת שַׁיָּכִים לָרַב! הַאִם הוּא רָמַז לוֹ לְשַׁקֵּר בַּמִּשְׁפָּט? לֹא! הוּא רַק עָשָׂה טוֹבָה קְטַנָּה וְהֵבִיא אֶת זֶה יוֹם אֶחָד קֹדֶם .

אֲבָל כְּשֶׁרַבִּי יִשְׁמָעֵאל שָׁמַע אֶת זֶה, הוּא נִרְתַּע לְאָחוֹר כְּאִלּוּ סַלְסִלַּת הַפֵּרוֹת שֶׁעַל שֻׁלְחָנוֹ עֲשׂוּיָה מִגֶּחָלִים בּוֹעֲרוֹת!

"פָּסוּל אֲנִי מִלָּדוּן אוֹתְךָ!" זָעַק רַבִּי יִשְׁמָעֵאל, וְקוֹלוֹ הִרְעִיד אֶת בֵּית הַדִּין. "קַח אֶת תְּאֵנֶיךָ וַעֲנָבֶיךָ, אֲנִי שׁוֹפֵט אוֹתְךָ הַיּוֹם לֹא אֶשְׁפֹּט!".

רַבִּי יִשְׁמָעֵאל יָדַע אֶת הַסּוֹד הַגָּדוֹל שֶׁל הַתּוֹרָה: "כִּי הַשֹּׁחַד יְעַוֵּר עֵינֵי חֲכָמִים". שֹׁחַד זֶה לֹא רַק מַעֲטָפָה שֶׁל כֶּסֶף שֶׁדּוֹחֲפִים בַּסֵּתֶר. שֹׁחַד זֶה אֲפִלּוּ חִיּוּךְ בַּזְּמַן הַלֹּא נָכוֹן, אֲפִלּוּ מִלָּה טוֹבָה, אֲפִלּוּ לְהָבִיא אֶת מָה שֶׁמַּגִּיעַ לוֹ – יוֹם קֹדֶם!

"וּמָה עִם הַמִּשְׁפָּט שֶׁלִּי?" שָׁאַל הָאָרִיס בְּבֶהָלָה.

"אַל תִּדְאַג", עָנָה רַבִּי יִשְׁמָעֵאל, "אוֹשִׁיב לְךָ שְׁנֵי דַּיָּנִים אֲחֵרִים, רַבָּנִים תַּלְמִידֵי חֲכָמִים וְאַנְשֵׁי יִרְאָה מִבֵּית הַמִּדְרָשׁ, וְהֵם יִשְׁפְּטוּ בֵּינְךָ לְבֵין חֲבֵרְךָ בְּצֶדֶק וּבֶאֱמֶת".

הַקּוֹלוֹת שֶׁמִּבַּחוּץ – וְהַקּוֹלוֹת שֶׁבַּלֵּב

הַמִּשְׁפָּט הִתְחִיל. הָאָרִיס וַחֲבֵרוֹ יָשְׁבוּ מִצַּד אֶחָד מִבִּפְנִים, מֵעֲלֵיהֶם שְׁנֵי דַּיָּנִים חֲדָשִׁים הַקּוֹבְעִים אֶת הַדִּין.

וְאֵיפֹה הָיָה רַבִּי יִשְׁמָעֵאל?

רַבִּי יִשְׁמָעֵאל לֹא הָלַךְ הַבַּיְתָה. הוּא נִשְׁאַר בַּחֶדֶר הַסָּמוּךְ, וְהִקְשִׁיב בְּרִכּוּז. הוּא שָׁמַע מִבַּעַד לַקִּיר אֶת הָאָרִיס מְגַמְגֵּם, מְנַסֶּה לְהַסְבִּיר לַדַּיָּנִים מַדּוּעַ הוּא צּוֹדֵק. הָאָרִיס לֹא הָיָה רָגִיל לְדַבֵּר לִפְנֵי אֲנָשִׁים, וְהָיָה קָשֶׁה לוֹ מְאֹד.

רַבִּי יִשְׁמָעֵאל שָׁמַע אוֹתוֹ, וּפִתְאוֹם... הוּא שָׂם לֵב לְמַשֶּׁהוּ מַפְחִיד בְּתוֹךְ הַלֵּב שֶׁל עַצְמוֹ!

בְּתוֹךְ הָרֹאשׁ שֶׁל רַבִּי יִשְׁמָעֵאל הִתְחִילוּ לִקְפֹּץ רַעֲיוֹנוֹת עֲבוּר הָאָרִיס: "אוֹי, אִם הָאָרִיס שֶׁלִּי רַק הָיָה אוֹמֵר אֶת הַמִּשְׁפָּט הַזֶּה... אִם הָאָרִיס שֶׁלִּי הָיָה רַק מֵבִיא אֶת הָרְאָיָה הַהִיא... הוּא יָכוֹל לְנַצֵּחַ בַּמִּשְׁפָּט! לָמָּה הוּא לֹא אוֹמֵר אֶת זֶה?!". הַלֵּב שֶׁל רַבִּי יִשְׁמָעֵאל מַמָּשׁ רָצָה לַעֲזֹר לָאָרִיס בַּמִּשְׁפָּט, רָצָה בְּטוֹבָתוֹ!

הָרְעִידָה וְהַמַּסְקָנָה הָאַדִּירָה

כְּשֶׁרַבִּי יִשְׁמָעֵאל קָלַט אֶת זֶה, הוּא נֶחְרַד עוֹד יוֹתֵר!

הוּא קָם מִמְּקוֹמוֹ, פָּנָיו הֶאֱדִימוּ, וְהוּא זָעַק דְּבָרִים שֶׁנִּשְׁאֲרוּ חֲקוּקִים בַּגְּמָרָא לָנֶצַח!

"תִּפַּח נַפְשָׁם שֶׁל מְקַבְּלֵי שֹׁחַד!" זָעַק רַבִּי יִשְׁמָעֵאל. (פֵּרוּשׁ: שֶׁתֵּצֵא רוּחָם שֶׁל לוֹקְחֵי הַשֹּׁחַד, כַּמָּה נוֹרָא הוּא חֶטְאָם!).

הוּא מִלְמֵל לְעַצְמוֹ בִּרְעָדָה: "רִבּוֹנוֹ שֶׁל עוֹלָם... הֲרֵי אֲנִי כְּלָל לֹא לָקַחְתִּי אֶת הַפֵּרוֹת! שָׁלַחְתִּי אוֹתוֹ מֵהַחֶדֶר! וַאֲפִלּוּ אִם הָיִיתִי לוֹקֵחַ – מָה הוּא כְּבָר נָתַן לִי? פֵּרוֹת שֶׁהֵם עַל פִּי דִּין שֶׁלִּי ! הוּא רַק נָתַן לִי אוֹתָם קְצָת יוֹתֵר מֻקְדָּם מֵהָרָגִיל. רַק הַחֲשִׁיבָה הַקְּטַנָּה שֶׁהוּא 'עָשָׂה לִי טוֹבָה' וְטָרַח לְהָבִיא לִי אוֹתָם הַיּוֹם – כְּבָר סִנְוְרָה אֶת הַלֵּב הַטָּהוֹר שֶׁלִּי לִרְצוֹת לַעֲזֹר לוֹ! אִם אֲנִי הָיִיתִי דָּן אוֹתוֹ, אֲפִלּוּ בְּלִי לָשִׂים לֵב, הָיִיתִי מַטֶּה אֶת הַמִּשְׁפָּט לְטוֹבָתוֹ!

אָז אֲנָשִׁים מֻשְׁחָתִים שֶׁלּוֹקְחִים מַעֲטָפוֹת שֶׁל כֶּסֶף מַמָּשׁ שֶׁלֹּא שַׁיָּכוֹת לָהֶם – אֵיךְ הֵם יוּכְלוּ אֵי פַּעַם לִשְׁפֹּט צֶדֶק? הֲרֵי הַלֵּב שֶׁלָּהֶם מָכוּר לְגַמְרֵי!".

בְּאוֹתוֹ יוֹם לָמְדוּ תַּלְמִידָיו שֶׁל רַבִּי יִשְׁמָעֵאל כַּמָּה גָּדוֹל, קָדוֹשׁ וְטָהוֹר הוּא בֵּית דִּין שֶׁל יִשְׂרָאֵל. לָמְדוּ, שֶׁעַל הָאֱמֶת וְהַצֶּדֶק חַיָּבִים לִשְׁמֹר בְּכָל הַכֹּחַ, כִּי יֵצֶר הָאָדָם וְהַלֵּב שֶׁלּוֹ עֲדִינִים כָּל כָּךְ, שֶׁחִיּוּךְ קָטָן אֶחָד יָכוֹל לְשַׁבֵּשׁ אֶת דִּין הַשָּׁמַיִם.

ME (רַבִּי יִשְׁמָעֵאל): רַבִּי יִשְׁמָעֵאל, הָרַב הַגָּדוֹל, בָּרַח מֵהַטּוֹבָה הֲכִי קְטַנָּה. הוּא סֵרֵב אֲפִלּוּ לְקַבֵּל מַשֶּׁהוּ שֶׁמַּגִּיעַ לוֹ, כְּדֵי שֶׁהַצֶּדֶק שֶׁלּוֹ לֹא יִפָּגַע בְּמִילִימֶטֶר.

WE (אֲנַחְנוּ): לִפְעָמִים אֲנַחְנוּ רָבִים אוֹ מְשַׂחֲקִים, וְלִפְעָמִים צְרִיכִים לִהְיוֹת "הַשּׁוֹפְטִים" בֵּין שְׁנֵי חֲבֵרִים שֶׁמִּתְוַכְּחִים. קַל לָנוּ לִתְמֹךְ בֶּחָבֵר שֶׁקָּנָה לָנוּ קַרְטִיב בַּהַפְסָקָה, אוֹ בַּחֲבֵר שֶׁמְּשַׂחֵק אִתָּנוּ בִּקְבוּצָה.

YOU (אַתָּה): מָחָר בַּהַפְסָקָה, אִם יְמַנּוּ אוֹתְךָ לִהְיוֹת שׁוֹפֵט בְּמִשְׂחַק כַּדּוּרֶגֶל, וְהֶחָבֵר הֲכִי טוֹב שֶׁלְּךָ עָשָׂה עֲבֵרָה (פאוּל) – מַה תַּעֲשֶׂה? תִּזְכֹּר אֶת רַבִּי יִשְׁמָעֵאל! מִי שֶׁעוֹשֶׂה צֶדֶק – עֵינָיו נְקִיּוֹת וְאֵינוֹ עוֹשֶׂה שׁוּם "פְּרוֹטֶקְצְיוֹת". לִשְׁפֹּט בֶּאֱמֶת, גַּם אִם זֶה נֶגֶד הֶחָבֵר הֲכִי טוֹב!

א. שְׁאֵלַת פְּתִיחָה (לִזְרֹק סַקְרָנוּת לַכִּתָּה):

"יְלָדִים, מִי מִכֶּם הָיָה פַּעַם שׁוֹפֵט בְּמִשְׂחַק כַּדּוּרֶגֶל אוֹ מַחֲנַיִם בַּהַפְסָקָה? (חכו להצבעות). וּמָה קוֹרֶה אִם הֶחָבֵר הֲכִי הֲכִי טוֹב שֶׁלָּכֶם עָשָׂה פָּאוּל... הַאִם קַל לִשְׁרֹק נֶגְדּוֹ אוֹ קָשֶׁה מְאֹד?

הַיּוֹם נִשְׁמַע עַל רַבִּי יִשְׁמָעֵאל שֶׁהָיָה צָרִיךְ לִשְׁפֹּט יְהוּדִי שֶׁעָשָׂה לוֹ כְּלַפֵּי חוּץ רַק טוֹבָה מִסְכֵּנָה אַחַת. נִרְאֶה לְאֵיזוֹ רָמָה שֶׁל נִקְיוֹן וֶאֱמֶת הוּא הִגִּיעַ!"

ב. הֶעָרוֹת בִּצּוּעַ וּסְטוֹרִיטֶלִינְג (בְּמַהֲלַךְ הַסִּפּוּר):

ג. הַמְלָצוֹת לְמַצֶּגֶת (4 שְׁקוּפִיּוֹת מֻקְפָּדוֹת עַל פִּי כְּלָלֵי הַצְּנִיעוּת וְהַקְּדֻשָּׁה):

ד. פְּעִילוּת מְסַכֶּמֶת / מִשְׂחַק מִשְׁפָּט:

משחק תפקידים "דין תורה בבית הספר":

המורה מבקש משלושה תלמידים לעלות. אחד הוא "שופט" ושניים הם "מתדיינים" (למשל, מתווכחים של מי כדור הרגל שנתקע בעץ). רגע לפני שהם מתחילים לדבר, אחד התלמידים ניגש לשופט ונותן לו במתנה "סוכריה" או מצייר לו ציור ואומר "זה בשבילך אחי". השופט צריך לעצור את המשפט ולהגיד: "פסול אני מלדון אתכם! השוחד יעוור עיני חכמים!".

מדברים על ההרגשה – איך דבר קטן של חברות יכול למנוע מאיתנו לשפוט בצדק אמיתי.

מקורות: תלמוד בבלי, מסכת כתובות, דף ק"ה עמוד ב'. (המעשה המלא על רבי ישמעאל ברבי יוסי והאריס שהביא פירות ביום חמישי במקום ביום שישי). ספר "ספר האגדה", ח"א, רמ"ז (מוזכר בחומרי החמ"ד בספרות מעובדת "כע"ח ג 37"). איסור "לא תקח שוחד" מספר דברים (פרק ט"ז, י"ט).

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “להלן מערך שיעור וסיפור חינוכי, רגשי ומרתק על רבי ישמעאל ב"ר יוסי והאריס (החקלאי) שלו .” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “רַבִּי יִשְׁמָעֵאל הֵרִים אֶת מַבָּטוֹ מֵהַסֵּפֶר, גַּבּוֹתָיו מְרֻמּוֹת בִּפְלִיאָה.”?
ג. כיצד הגיב רבי ישמעאל ב"ר יוסי למצב המתואר: “שֹׁחַד זֶה לֹא רַק מַעֲטָפָה שֶׁל כֶּסֶף שֶׁדּוֹחֲפִים בַּסֵּתֶר.”?
ד. מה השתנה בין “הַלֵּב שֶׁל רַבִּי יִשְׁמָעֵאל מַמָּשׁ רָצָה לַעֲזֹר לָאָרִיס בַּמִּשְׁפָּט, רָצָה בְּטוֹבָתוֹ!” לבין “לָמְדוּ, שֶׁעַל הָאֱמֶת וְהַצֶּדֶק חַיָּבִים לִשְׁמֹר בְּכָל הַכֹּחַ, כִּי יֵצֶר הָאָדָם וְהַלֵּב שֶׁלּוֹ עֲדִינִים…”?
ה. כיצד הגיע הסיפור אל האירוע: “נִרְאֶה לְאֵיזוֹ רָמָה שֶׁל נִקְיוֹן וֶאֱמֶת הוּא הִגִּיעַ!”?

לפני ואחרי

מה קרה לפני: “בְּתוֹךְ הָרֹאשׁ שֶׁל רַבִּי יִשְׁמָעֵאל הִתְחִילוּ לִקְפֹּץ רַעֲיוֹנוֹת עֲבוּר הָאָרִיס: "אוֹי, אִם הָאָרִיס שֶׁלִּי רַק הָיָה אוֹמֵר אֶת הַמִּשְׁפָּט…”?
מה קרה אחרי: “הוּא סֵרֵב אֲפִלּוּ לְקַבֵּל מַשֶּׁהוּ שֶׁמַּגִּיעַ לוֹ, כְּדֵי שֶׁהַצֶּדֶק שֶׁלּוֹ לֹא יִפָּגַע בְּמִילִימֶטֶר.”?

סיבה ותוצאה מתוך המעשה

האירוע הנתוןמה גרם לו?מה הייתה תוצאתו?
אֲבָל כְּשֶׁרַבִּי יִשְׁמָעֵאל שָׁמַע אֶת זֶה, הוּא נִרְתַּע לְאָחוֹר כְּאִלּוּ סַלְסִלַּת הַפֵּרוֹת שֶׁעַל שֻׁלְחָנוֹ עֲשׂוּיָה מִגֶּחָלִים בּוֹעֲרוֹת!
הוּא רַק נָתַן לִי אוֹתָם קְצָת יוֹתֵר מֻקְדָּם מֵהָרָגִיל.

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “האריס שהביא פירות מוקדם — זהירות משוחד” על רבי ישמעאל ב"ר יוסי? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “הָאָרִיס לֹא הָיָה רָגִיל לְדַבֵּר לִפְנֵי אֲנָשִׁים, וְהָיָה קָשֶׁה לוֹ מְאֹד.”.
ג. אילו רבי ישמעאל ב"ר יוסי לא היה פועל כפי שמתואר ב־“לָמְדוּ, שֶׁעַל הָאֱמֶת וְהַצֶּדֶק חַיָּבִים לִשְׁמֹר בְּכָל הַכֹּחַ, כִּי יֵצֶר הָאָדָם וְהַלֵּב שֶׁלּוֹ עֲדִינִים כָּל כָּךְ,…” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “נִרְאֶה לְאֵיזוֹ רָמָה שֶׁל נִקְיוֹן וֶאֱמֶת הוּא הִגִּיעַ!” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. להלן מערך שיעור וסיפור חינוכי, רגשי ומרתק על רבי ישמעאל ב"ר יוסי והאריס…
2. הָאָרִיס חִיֵּךְ חִיּוּךְ מִתְנַצֵּל, מָחָה אֶת אֶגְלֵי הַזֵּעָה…
3. אַל תִּדְאַג", עָנָה רַבִּי יִשְׁמָעֵאל, "אוֹשִׁיב לְךָ שְׁנֵי…
4. תִּפַּח נַפְשָׁם שֶׁל מְקַבְּלֵי שֹׁחַד!" זָעַק רַבִּי יִשְׁמָעֵאל.
5. קַל לָנוּ לִתְמֹךְ בֶּחָבֵר שֶׁקָּנָה לָנוּ קַרְטִיב בַּהַפְסָקָה, אוֹ…
6. מדברים על ההרגשה – איך דבר קטן של חברות יכול למנוע מאיתנו לשפוט בצדק…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “שֹׁחַד זֶה לֹא רַק מַעֲטָפָה שֶׁל כֶּסֶף שֶׁדּוֹחֲפִים בַּסֵּתֶר.”.
מכתב: כתבו לרבי ישמעאל ב"ר יוסי בעקבות “הוּא רַק נָתַן לִי אוֹתָם קְצָת יוֹתֵר מֻקְדָּם מֵהָרָגִיל.”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “מִי שֶׁעוֹשֶׂה צֶדֶק – עֵינָיו נְקִיּוֹת וְאֵינוֹ עוֹשֶׂה שׁוּם "פְּרוֹטֶקְצְיוֹת".”.
דיווח: כתבו ידיעה קצרה שכותרתה “האריס שהביא פירות מוקדם — זהירות משוחד”.

המחזה

רגע מן הסיפורהדמויותמה יאמרו או יעשו?
אֲבָל כְּשֶׁרַבִּי יִשְׁמָעֵאל שָׁמַע אֶת זֶה, הוּא נִרְתַּע לְאָחוֹר כְּאִלּוּ סַלְסִלַּת…
לָמְדוּ, שֶׁעַל הָאֱמֶת וְהַצֶּדֶק חַיָּבִים לִשְׁמֹר בְּכָל הַכֹּחַ, כִּי יֵצֶר הָאָדָם וְהַלֵּב…

7. חיבור לחיים מתוך המעשה

א. מה אפשר ללמוד מן הדרך שבה רבי ישמעאל ב"ר יוסי פעל באירוע “הַלֵּב שֶׁל רַבִּי יִשְׁמָעֵאל מַמָּשׁ רָצָה לַעֲזֹר לָאָרִיס בַּמִּשְׁפָּט, רָצָה בְּטוֹבָתוֹ!”?
