הצלת אחות ברוריה מידי הרומאים

דף עבודה בעקבות רבי מאיר וברוריה
שם: כיתה:

המעשה — הסיפור המלא

להלן עיבוד חינוכי, רגשי ומרתק לסיפור הצלת אחות ברוריה על ידי רבי מאיר (בעל הנס), המעובד במיוחד לבנים בכיתות ב'-ד' (חינוך דתי-לאומי). הסיפור מדגיש את כוחה של תפילה, מסירות הנפש של התנאים, ונוכחותה של ההשגחה הפרטית אפילו במקומות האפלים ביותר. בהתאם להנחיות המפורשות במקורות (עבודה זרה י"ח:), הושמט חלק ה"מבחן" שעשה רבי מאיר לאחות.

הסיפור מנוקד במלואו כנדרש.

פְּתִיחָה: דְּמָעוֹת שֶׁל צַעַר בְּבֵית הַצַּדִּיקִים

יְלָדִים, יֵשׁ רְגָעִים בַּחַיִּים שֶׁבָּהֶם הַכֹּל נִרְאֶה חָשׁוּךְ. מַמָּשׁ חָשׁוּךְ.

בְּבֵיתוֹ שֶׁל הַתַּנָּא הַקָּדוֹשׁ, רַבִּי מֵאִיר, שָׂרַר שֶׁקֶט עָצוּב וְכָבֵד. הָיָה זֶה זְמַן קָשֶׁה לְעַם יִשְׂרָאֵל. מַלְכוּת רוֹמִי הָרְשָׁעָה גָּזְרָה גְּזֵרוֹת נוֹרָאוֹת. לֹא מִזְּמַן, נֶהֱרַג עַל קִדּוּשׁ הַשֵּׁם חָמִיו שֶׁל רַבִּי מֵאִיר – רַבִּי חֲנִינָא בֶּן תְּרַדְיוֹן, שֶׁנִּשְׂרַף בָּאֵשׁ כְּשֶׁסֵּפֶר תּוֹרָה כָּרוּךְ סְבִיבוֹ.

אֲבָל הַצַּעַר שֶׁל הַמִּשְׁפָּחָה לֹא הִסְתַּיֵּם. בְּרוּרְיָה, אִשְׁתּוֹ הַחֲכָמָה וְהַצַּדֶּקֶת שֶׁל רַבִּי מֵאִיר, יָשְׁבָה בְּפִנַּת הַחֶדֶר. עֵינֶיהָ הָיוּ אֲדֻמּוֹת מִבְּכִי. דְּמָעוֹת חַמּוֹת זָלְגוּ עַל לְחָיֶיהָ לְלֹא הַפְסָקָה. הָרוֹמָאִים הָאַכְזָרִים לָקְחוּ אֶת אֲחוֹתָהּ הַקְּטַנָּה, וְכָלְאוּ אוֹתָהּ בְּבֵית אֲסוּרִים (בֵּית כֶּלֶא) שָׁמוּר וּמְפֻקָּח, כְּדֵי לְבַיֵּשׁ אֶת בְּנוֹת מִשְׁפַּחַת הַצַּדִּיק.

"בַּעֲלִי הַיָּקָר," יִבְּבָה בְּרוּרְיָה מִקִּיר לִבָּהּ, "אֲנִי יוֹדַעַת כַּמָּה סַכָּנָה יֵשׁ בַּדָּבָר, אֲבָל מִשְׁפַּחְתִּי כְּבָר נֶהֶרְגָה עַל קִדּוּשׁ הַשֵּׁם. אֲנִי לֹא מְסֻגֶּלֶת לָשֵׂאת אֶת הַמַּחֲשָׁבָה שֶׁאֲחוֹתִי נִמְצֵאת שָׁם, בַּמָּקוֹם הַטָּמֵא הַזֶּה. אָנָּא, חֲרֵף אֶת נַפְשְׁךָ וְהַצֵּל אוֹתָהּ!"

רַבִּי מֵאִיר לֹא הִסֵּס שְׁנִיָּה אַחַת. הוּא הֵרִים אֶת מַבָּטוֹ, עֵינָיו בָּרְקוּ בֶּאֱמוּנָה טְהוֹרָה שֶׁאֵין כָּמוֹהָ. "אֲנִי יוֹצֵא לַדֶּרֶךְ," אָמַר בְּקוֹל יַצִּיב וּבָטוּחַ. "הַקָּדוֹשׁ בָּרוּךְ הוּא שֶׁמַּצִּיל אֶת הַבּוֹטְחִים בּוֹ, יַעֲמֹד לִימִינִי."

הַמַּסָּע בַּחֹשֶׁךְ אֶל עֵמֶק אוֹיֵב

יְלָדִים, דַּמְיְנוּ אֶת קוֹר הַלַּיְלָה הַצּוֹנֵן. הָרוּחַ שׁוֹרֶקֶת בֵּין עֲצֵי הַיַּעַר הַחֲשׁוּכִים. רַבִּי מֵאִיר עוֹטֶה עַל עַצְמוֹ גְּלִימָה עָבָה. בְּחֶבְיוֹן בְּגָדָיו הוּא מַסְתִּיר צְרוֹר כָּבֵד – שַׂק עָמוּס בְּמַטְבְּעוֹת זָהָב נוֹצְצִים (דִּינָרִים). הוּא צוֹעֵד בְּשֶׁקֶט, שְׂפָתָיו מְמַלְמְלוֹת תְּפִלָּה וְדִבְרֵי תּוֹרָה. הוּא לֹא סוֹמֵךְ עַל כּוֹחוֹ, הוּא סוֹמֵךְ רַק עַל הַשֵּׁם יִתְבָּרַךְ.

סוֹף סוֹף, הוּא מַגִּיעַ אֶל בֵּית הָאֲסוּרִים שֶׁל הָרוֹמָאִים. זֶה הָיָה מִבְנֶה אֶבֶן שָׁחוֹר וּמַפְחִיד. רֵיחַ כָּבֵד שֶׁל עֹבֶשׁ, לַחוּת וּבַרְזֶל מַחֲלִיד עָמַד בָּאֲוִיר. סְבִיב הַחוֹמוֹת שׁוֹטְטוּ שׁוֹמְרִים חֲמוּשִׁים בַּחֲרָבוֹת, אֲבָל הַדָּבָר הַמַּפְחִיד בְּיוֹתֵר הָיָה מַה שֶׁנִּמְצָא מִתַּחַת לְפָנָסֵי הָאֵשׁ שֶׁהָיוּ תְּלוּיִים עַל הַקִּיר: כְּלָבִים אֵימְתָנִיִּים.

אֵלּוּ לֹא הָיוּ כְּלָבִים רְגִילִים. הָיוּ אֵלּוּ כַּלְבֵי תְּקִיפָה שֶׁל הַקֵּיסָר הָרוֹמָאִי – עֲנָקִיִּים, עִם עֵינַיִם רָעוֹת שֶׁזָּהֲרוּ בַּחֹשֶׁךְ, וְשִׁנַּיִם חַדּוֹת כְּתַעַר שֶׁטִּפְטְפוּ רִיר. כָּל מִי שֶׁרַק נִסָּה לְהִתְקָרֵב לַשַּׁעַר שֶׁל הַכֶּלֶא, הַכְּלָבִים הָיוּ נוֹבְחִים נְבִיחוֹת מַחֲרִישׁוֹת אָזְנַיִם וְקוֹפְצִים עָלָיו.

הַהַצָּעָה לַשּׁוֹמֵר הָרוֹמָאִי

רַבִּי מֵאִיר הִתְקָרֵב אֶל הַשַּׁעַר, שָׁלֵו וְרָגוּעַ. שׁוֹמֵר רוֹמָאִי גְּדוֹל-מְמַדִּים וְשָׁרִירִי עָצַר אוֹתוֹ

"מִי אַתָּה וּמָה אַתָּה מְחַפֵּשׂ פֹּה?" צָעַק הַשּׁוֹמֵר בְּקוֹל עָבֶה.

"בָּאתִי לִפְדּוֹת מִכָּאן נַעֲרָה יְהוּדִיָּה," אָמַר רַבִּי מֵאִיר בְּרוֹגַע.

לְפֶתַע, הוֹצִיא רַבִּי מֵאִיר מִתַּחַת לִגְלִימָתוֹ אֶת צְרוֹר הַמַּטְבְּעוֹת. הַקִּרְקוּשׁ שֶׁל דִּינְרֵי הַזָּהָב הִדְהֵד בָּאֲוִיר – צְלִיל שֶׁכָּל רוֹמָאִי אָהַב לִשְׁמֹעַ. "קַח אֶת הַכֶּסֶף הַזֶּה," אָמַר רַבִּי מֵאִיר. "חֶצְיוֹ יִהְיֶה לְךָ, שֶׁתּוּכַל לִקְנוֹת בּוֹ מָזוֹן וּבְגָדִים. וְאֶת הַחֵצִי הַשֵּׁנִי תִּשְׁמֹר לְךָ, וּכְשֶׁיָּבוֹאוּ שׁוֹמְרִים אֲחֵרִים אוֹ קְצִינִים מֵהַמַּלְכוּת וְיִשְׁאֲלוּ אוֹתְךָ אֵיפֹה הַנַּעֲרָה, תִּתֵּן לָהֶם שׁוֹחַד מֵהַחֵצִי הַשֵּׁנִי כְּדֵי שֶׁיִּשְׁתְּקוּ."

הַשּׁוֹמֵר הָרוֹמָאִי עָמַד פְּעוּר פֶּה. הָיָה לוֹ בַּיָּד סְכוּם כֶּסֶף אַדִּיר! אֲבָל אָז הוּא נִזְכַּר בְּכֹחוֹ שֶׁל הַקֵּיסָר, וַחֲרָדָה מִלְּאָה אֶת לִבּוֹ.

"אֲבָל מָה אֶעֱשֶׂה כַּאֲשֶׁר אֶתֵּן אֶת כָּל הַחֵצִי הַשֵּׁנִי שֶׁל הַכֶּסֶף לַשָּׂרִים וְיִגָּמֵר לִי הַשּׁוֹחַד?" שָׁאַל הַשּׁוֹמֵר בְּפַחַד. "הֲרֵי בַּסּוֹף הַקֵּיסָר בְּעַצְמוֹ יְגַלֶּה אֶת זֶה, וְיִשְׁלַח מִיָּד שׁוֹמְרִים לַהֲרוֹג אוֹתִי וְלִזְרוֹק אוֹתִי לַכְּלָבִים הָרְעֵבִים הָאֵלֶּה!" הוּא הִצְבִּיעַ עַל הַכְּלָבִים שֶׁשָּׁאֲגוּ בָּרֶקַע.

הַנֶּשֶׁק הַסּוֹדִי ותרגיל הכלבים

כָּאן, יְלָדִים, קוֹרֶה מַשֶּׁהוּ מַדְהִים. רַבִּי מֵאִיר לֹא הוֹצִיא עוֹד כֶּסֶף. הוּא הִצִּיעַ לַשּׁוֹמֵר מַשֶּׁהוּ שֶׁשָּׁוֶה יוֹתֵר מִכָּל הַזָּהָב בָּעוֹלָם. מַשֶּׁהוּ שֶׁשַּׁיָּךְ רַק לְעַם יִשְׂרָאֵל.

"אַל תִּפְחַד," אָמַר הַצַּדִּיק שָׁלֵו כְּמַלְאָךְ. "אִם יִגָּמֵר הַכֶּסֶף וְיָבוֹאוּ לַהֲרוֹג אוֹתְךָ אוֹ יְשַׁסּוּ בְּךָ כְּלָבִים, אַתָּה תֹּאמַר רַק מִשְׁפָּט אֶחָד קָטָן, שָׁלוֹשׁ מִלִּים, וּמִיָּד תִּנָּצֵל."

"מָה לְהַגִּיד?" הִתְפַּלֵּא הַשּׁוֹמֵר.

"תֹּאמַר: אֱלָהָא דְּמֵאִיר עֲנֵנִי! " (פֵּרוּשׁ הַמִּלִּים בַּאֲרָמִית: הָאֵ-ל, קוֹנוֹ וּבוֹרְאוֹ שֶׁל מֵאִיר, עֲנֵה לִי וְהַצֵּל אוֹתִי!).

הַשּׁוֹמֵר הָרוֹמָאִי פָּרַץ בִּצְחוֹק רָם. צְחוֹק שֶׁהִדְהֵד בַּחֲשֵׁכָה.

"אַתָּה צוֹחֵק עָלַי?!" קָרָא. "אֲנִי אֲגִיד אֶת הַמִּשְׁפָּט הַזֶּה שֶׁלְּךָ, וְהַקֵּיסָר שֶׁלִּי יִבְרַח? הַכְּלָבִים הָרְעֵבִים יַעַצְרוּ? מִי יוֹכִיחַ לִי שֶׁהַפֶּלֶא הַזֶּה בִּכְלָל עוֹבֵד?!"

"בּוֹא וְתִרְאֶה בְּעַצְמְךָ," הוֹרָה רַבִּי מֵאִיר.

רַבִּי מֵאִיר הִצְבִּיעַ הַרְחֵק לְעֵבֶר כַּלְבֵי הַתְּקִיפָה הָרוֹמָאִיִּים שֶׁשָּׁמְרוּ שָׁם. הַכְּלָבִים הָיוּ רְעֵבִים מְאוֹד, וּמְאַמְּנֵיהֶם דָּאֲגוּ שֶׁיִּהְיוּ פְרָאִיִּים וְתַקְפָּנִיִּים. הַשּׁוֹמֵר, כְּדֵי לִבְדּוֹק, לָקַח רֶגֶב עָפָר גָּדוֹל, קָטָע שֶׁל בּוֹץ וַאֲבָנִים, וְזָרַק אוֹתוֹ בְּחָזְקָה לְעֵבֶר קְבוּצַת הַכְּלָבִים.

רְאוּ זֶה פֶּלֶא!

בְּאוֹתָהּ שְׁנִיָּה, הַכְּלָבִים חָשְׂפוּ אֶת הַשִּׁנַּיִם שֶׁלָּהֶם. עֵינֵיהֶם רָשְׁפוּ רֶשַׁע. הֵם נָבְחוּ נְבִיחוֹת נִחָרוֹת, קָפְצוּ בָּאֲוִיר, וְהִסְתַּעֲרוּ מַהֵר כְּחֵץ תּוֹךְ רִיצָה טֶרֶף מַמָּשׁ לְעֵבֶר הַשּׁוֹמֵר שֶׁזָּרַק אֶת הָאֶבֶן!

הַשּׁוֹמֵר קָפָא מִפַּחַד. הֵם הָיוּ בְּמֶרְחָק שֶׁל מֶטְרִים סְפוּרִים מִמֶּנּוּ. כְּבָר רָאוּ אֶת קֶצֶף רוֹקָם. הֵם הָיוּ עֲתִידִים לְזַנֵּק עָלָיו וְלִטְרוֹף אוֹתוֹ חַי.

רָעַד הַשּׁוֹמֵר כְּעָלֶה נִדָּף בָּרוּחַ, אָסַף אֶת כָּל אַוִּיר רֵיאוֹתָיו וְזָעַק מִקִּירוֹת לִבּוֹ אֶת כָּל מָה שֶׁנּוֹתַר לוֹ:

"אֱלָהָא דְּמֵאִיר עֲנֵנִי!!!"

בִּשְׁנִיָּה אַחַת – כְּבָמַטֵּה קֶסֶם! – נֶעֶצְרוּ הַכְּלָבִים. הֵם בָּלְמוּ אֶת רַגְלֵיהֶם בָּאֲדָמָה, נְבִיחָתָם נֶחְנְקָה בִּגְרוֹנָם, הֵם הִשְׁפִּילוּ אֶת עֵינֵיהֶם, סָבְבוּ אֶת זְנָבָם לְבֵין רַגְלֵיהֶם וְ... הָלְכוּ לְאָחוֹר, רְגוּעִים וּפוֹחֲדִים, הַרְחֵק לְעֵבֶר הַכְּלוּבִים שֶׁלָּהֶם.

הַנִּצָּחוֹן וְהַשֵּׁם הַקָּדוֹשׁ

הַשּׁוֹמֵר מָחָה זֵעָה קָרָה מִמִּצְחוֹ. לִבּוֹ פָּעַם בְּחָזְקָה כְּתֹף עֲנָק. הוּא נָשַׁם עָמֹק וְהִבִּיט בְּרַבִּי מֵאִיר בִּירְאַת כָּבוֹד וְתַדְהֵמָה. הוּא הֵבִין שֶׁהָאִישׁ הָעוֹמֵד מוּלוֹ אֵינוֹ עוֹד אָדָם, אֶלָּא קְדוֹשׁ עֶלְיוֹן שֶׁל בּוֹרֵא הָעוֹלָם.

לְלֹא מִלָּה נוֹסֶפֶת, הוּא חָטַף אֶת צְרוֹר הַכֶּסֶף, רָץ פְּנִימָה לְבֵית הָאֲסוּרִים, לָקַח אֶת הַמַּפְתְּחוֹת הַגְּדוֹלִים, וְשִׁחְרֵר אֶת אֲחוֹתָהּ שֶׁל בְּרוּרְיָה!

לַיְלָה הָפַךְ לְבוֹקֶר, וְהַשֶּׁמֶשׁ זָרְחָה עַל הָרֵי אֶרֶץ יִשְׂרָאֵל. כְּשֶׁחָזַר רַבִּי מֵאִיר הַבַּיְתָה עִם הָאָחוֹת בְּרִיאָה וּשְׁלֵמָה, בָּכְתָה בְּרוּרְיָה מֵרֹב שִׂמְחָה וְהוֹדְתָה לְרִבּוֹנוֹ שֶׁל עוֹלָם.

וְהַשּׁוֹמֵר? לַמְרוֹת שֶׁנִּסָּה לְשַׁחֵד אֶת הַקְּצִינִים, הַסּוֹד הִתְגַּלָּה לַקֵּיסָר, וְגָזְרוּ לִתְלוֹת אֶת הַשּׁוֹמֵר. וּכְשֶׁהֶעֱמִידוּ אוֹתוֹ בַּכִּכָּר לַהֲרוֹג אוֹתוֹ, צָעַק שׁוּב הַשּׁוֹמֵר: "אֱלָהָא דְּמֵאִיר עֲנֵנִי!". מִיָּד נִקְרַע הַחֶבֶל כְּאִלּוּ הָיָה חוּט מֶשִׁי דַּק, וְהַשּׁוֹמֵר נִצַּל וּבָרַח!

מֵאָז וְעַד עַכְשָׁו, קוֹרְאִים לְרַבִּי מֵאִיר כֻּלָּם "רַבִּי מֵאִיר בַּעַל הַנֵּס ".

ME (הַדְּמוּת): רַבִּי מֵאִיר, מִגְּדוֹלֵי תַּלְמִידֵי חֲכָמִים, הָיָה יָכוֹל לוֹמַר שֶׁזֶּה "סַכָּנָה" אוֹ שֶׁהוּא פּוֹחֵד מֵהָרוֹמָאִים. אֲבָל הוּא אָסַף אֶת כָּל אֱמוּנָתוֹ, הִסְתַּכֵּן בִּשְׁבִיל לַעֲזוֹר לְצָרָתָהּ שֶׁל אִשְׁתּוֹ, וְסָמַךְ כְּעִוֵּר עַל תְּפִלָּה וְעַל כּוֹחוֹ שֶׁל בּוֹרֵא הָעוֹלָם.

WE (אֲנַחְנוּ): לִפְעָמִים הַרְבֵּה יוֹתֵר קַל לָנוּ לִפְחָד אוֹ לִדְאֹג כְּשֶׁמַּשֶּׁהוּ מַלְחִיץ אוֹתָנוּ, וַאֲנַחְנוּ חוֹשְׁבִים שֶׁקָּשֶׁה מִדַּי לַעֲזוֹר לֶחָבֵר בְּצָרָה כִּי זֶה "מְסֻבָּךְ לָנוּ".

YOU (אַתָּה): מָחָר בַּבֹּקֶר, אִם תִּרְאֶה חָבֵר שֶׁקָּשֶׁה לוֹ, שֶׁפּוֹחֵד אוֹ שֶׁנִּמְצָא בִּבְעָיָה – אַל תַּעֲמֹד מִן הַצַּד. נַסֵּה לַעֲזוֹר לוֹ, וְתִזְכֹּר שֶׁיֵּשׁ לְךָ "נֶשֶׁק סוֹדִי" מְנַצֵּחַ: אַתָּה תָּמִיד יָכוֹל לָשֵׂאת תְּפִלָּה לַה' יִתְבָּרַךְ שֶׁיַּעֲזוֹר לָכֶם, וְהוּא שׁוֹמֵעַ וּמַצִּיל!

א. שְׁאֵלַת פְּתִיחָה:

"יְלָדִים, אִם הֱיִיתֶם רוֹאִים פִתְאוֹם אֶת הַכֶּלֶב הֲכִי גָּדוֹל וּמַפְחִיד בָּעוֹלָם מוּלְכֶם, מָה הֱיִיתֶם עוֹשִׂים? לְמִי הֱיִיתֶם קוֹרְאִים לְעֶזְרָה? הַיּוֹם נִלְמַד מַהוּ הַ'נֶּשֶׁק הַסּוֹדִי' שֶׁנָּתַן אֶחָד מִגְּדוֹלֵי יִשְׂרָאֵל לְחַיָּל רוֹמָאִי עֲנָק בְּדִיּוּק בְּמַצָּב כָּזֶה!"

ב. הֶעָרוֹת בִּצּוּעַ וּסְטוֹרִיטֶלִינְג (Storytelling):

ג. הַמְלָצוֹת לְמַצֶּגֶת (4 שְׁקוּפִיּוֹת עִם דְּגֵשֵׁי צְנִיעוּת):

ד. פְּעִילוּת מְסַכֶּמֶת לַכִּתָּה:

מקורות: תַּלְמוּד בַּבְלִי, מַסֶּכֶת עֲבוֹדָה זָרָה, דַּף י"ח עַמּוּד ב' – עֻבְדַּת לְקִיחָתָהּ שֶׁל אֲחוֹת בְּרוּרְיָה עַל יְדֵי הָרוֹמָאִים, שִׂיחַת רַבִּי מֵאִיר עִם הַשּׁוֹמֵר, מַתַּן שַׂק הַדִּינָרִים, נִסָּיוֹן כַּלְבֵי הַתְּקִיפָה הָרוֹמָאִיִּים שֶׁשֻּׁסּוּ עַל יְדֵי הָאֶבֶן, וְהַקְּרִיאָה "אֱלָהָא דְּמֵאִיר עֲנֵנִי" אֲשֶׁר עָצְרָה אֶת הַכְּלָבִים וּפָעֲלָה בְּאֹרַח נֵס, גַּם בִּשְׁעַת הַגַּרְדּוֹם [הצלת אחות ברוריה, מדוע נקרא בעל הנס; ספר 'כה עשו חכמינו'; ספר האגדה].

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “להלן עיבוד חינוכי, רגשי ומרתק לסיפור הצלת אחות ברוריה על ידי רבי מאיר (בעל הנס), המעובד במיוחד לבנים בכיתות ב'-ד' (חינוך דתי-לאומי).” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “אֲנִי לֹא מְסֻגֶּלֶת לָשֵׂאת אֶת הַמַּחֲשָׁבָה שֶׁאֲחוֹתִי נִמְצֵאת שָׁם, בַּמָּקוֹם הַטָּמֵא הַזֶּה.”?
ג. כיצד הגיב רבי מאיר וברוריה למצב המתואר: “כָּל מִי שֶׁרַק נִסָּה לְהִתְקָרֵב לַשַּׁעַר שֶׁל הַכֶּלֶא, הַכְּלָבִים הָיוּ נוֹבְחִים נְבִיחוֹת מַחֲרִישׁוֹת אָזְנַיִם וְקוֹפְצִים עָלָיו.”?
ד. מה השתנה בין “אִם יִגָּמֵר הַכֶּסֶף וְיָבוֹאוּ לַהֲרוֹג אוֹתְךָ אוֹ יְשַׁסּוּ בְּךָ כְּלָבִים, אַתָּה תֹּאמַר רַק מִשְׁפָּט אֶחָד…” לבין “הוּא נָשַׁם עָמֹק וְהִבִּיט בְּרַבִּי מֵאִיר בִּירְאַת כָּבוֹד וְתַדְהֵמָה.”?
ה. כיצד הגיע הסיפור אל האירוע: “ME (הַדְּמוּת): רַבִּי מֵאִיר, מִגְּדוֹלֵי תַּלְמִידֵי חֲכָמִים, הָיָה יָכוֹל לוֹמַר שֶׁזֶּה "סַכָּנָה" אוֹ שֶׁהוּא פּוֹחֵד מֵהָרוֹמָאִים.”?

□ הֲרֵי בַּסּוֹף הַקֵּיסָר בְּעַצְמוֹ יְגַלֶּה אֶת זֶה, וְיִשְׁלַח מִיָּד שׁוֹמְרִים לַהֲרוֹג אוֹתִי וְלִזְרוֹק אוֹתִי לַכְּלָבִים הָרְעֵבִים הָאֵלֶּה!"…
□ להלן עיבוד חינוכי, רגשי ומרתק לסיפור הצלת אחות ברוריה על ידי רבי מאיר (בעל הנס), המעובד במיוחד לבנים בכיתות ב'-ד' (חינוך דתי-לאומי).
□ הַיּוֹם נִלְמַד מַהוּ הַ'נֶּשֶׁק הַסּוֹדִי' שֶׁנָּתַן אֶחָד מִגְּדוֹלֵי יִשְׂרָאֵל לְחַיָּל רוֹמָאִי עֲנָק בְּדִיּוּק בְּמַצָּב כָּזֶה!
□ הוּא צוֹעֵד בְּשֶׁקֶט, שְׂפָתָיו מְמַלְמְלוֹת תְּפִלָּה וְדִבְרֵי תּוֹרָה.
□ הוּא הֵבִין שֶׁהָאִישׁ הָעוֹמֵד מוּלוֹ אֵינוֹ עוֹד אָדָם, אֶלָּא קְדוֹשׁ עֶלְיוֹן שֶׁל בּוֹרֵא הָעוֹלָם.

לפני ואחרי

מה קרה לפני: “הֲרֵי בַּסּוֹף הַקֵּיסָר בְּעַצְמוֹ יְגַלֶּה אֶת זֶה, וְיִשְׁלַח מִיָּד שׁוֹמְרִים לַהֲרוֹג אוֹתִי וְלִזְרוֹק אוֹתִי לַכְּלָבִים הָרְעֵבִים הָאֵלֶּה!"…”?
מה קרה אחרי: “הוּא הֵבִין שֶׁהָאִישׁ הָעוֹמֵד מוּלוֹ אֵינוֹ עוֹד אָדָם, אֶלָּא קְדוֹשׁ עֶלְיוֹן שֶׁל בּוֹרֵא הָעוֹלָם.”?

סיבה ותוצאה מתוך המעשה

האירוע הנתוןמה גרם לו?מה הייתה תוצאתו?
הוּא לֹא סוֹמֵךְ עַל כּוֹחוֹ, הוּא סוֹמֵךְ רַק עַל הַשֵּׁם יִתְבָּרַךְ.
הֵם נָבְחוּ נְבִיחוֹת נִחָרוֹת, קָפְצוּ בָּאֲוִיר, וְהִסְתַּעֲרוּ מַהֵר כְּחֵץ תּוֹךְ רִיצָה טֶרֶף מַמָּשׁ לְעֵבֶר הַשּׁוֹמֵר שֶׁזָּרַק אֶת הָאֶבֶן!

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “הצלת אחות ברוריה מידי הרומאים” על רבי מאיר וברוריה? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “וְאֶת הַחֵצִי הַשֵּׁנִי תִּשְׁמֹר לְךָ, וּכְשֶׁיָּבוֹאוּ שׁוֹמְרִים אֲחֵרִים אוֹ קְצִינִים מֵהַמַּלְכוּת וְיִשְׁאֲלוּ אוֹתְךָ אֵיפֹה הַנַּעֲרָה, תִּתֵּן…”.
ג. אילו רבי מאיר וברוריה לא היה פועל כפי שמתואר ב־“הוּא נָשַׁם עָמֹק וְהִבִּיט בְּרַבִּי מֵאִיר בִּירְאַת כָּבוֹד וְתַדְהֵמָה.” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “ME (הַדְּמוּת): רַבִּי מֵאִיר, מִגְּדוֹלֵי תַּלְמִידֵי חֲכָמִים, הָיָה יָכוֹל לוֹמַר שֶׁזֶּה "סַכָּנָה" אוֹ שֶׁהוּא פּוֹחֵד…” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. להלן עיבוד חינוכי, רגשי ומרתק לסיפור הצלת אחות ברוריה על ידי רבי מאיר…
2. הוּא הֵרִים אֶת מַבָּטוֹ, עֵינָיו בָּרְקוּ בֶּאֱמוּנָה טְהוֹרָה שֶׁאֵין…
3. לְפֶתַע, הוֹצִיא רַבִּי מֵאִיר מִתַּחַת לִגְלִימָתוֹ אֶת צְרוֹר…
4. הַכְּלָבִים הָיוּ רְעֵבִים מְאוֹד, וּמְאַמְּנֵיהֶם דָּאֲגוּ שֶׁיִּהְיוּ…
5. כְּשֶׁחָזַר רַבִּי מֵאִיר הַבַּיְתָה עִם הָאָחוֹת בְּרִיאָה וּשְׁלֵמָה,…
6. הַיּוֹם נִלְמַד מַהוּ הַ'נֶּשֶׁק הַסּוֹדִי' שֶׁנָּתַן אֶחָד מִגְּדוֹלֵי…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “כָּל מִי שֶׁרַק נִסָּה לְהִתְקָרֵב לַשַּׁעַר שֶׁל הַכֶּלֶא, הַכְּלָבִים הָיוּ נוֹבְחִים נְבִיחוֹת מַחֲרִישׁוֹת אָזְנַיִם…”.
מכתב: כתבו לרבי מאיר וברוריה בעקבות “הֵם נָבְחוּ נְבִיחוֹת נִחָרוֹת, קָפְצוּ בָּאֲוִיר, וְהִסְתַּעֲרוּ מַהֵר כְּחֵץ תּוֹךְ רִיצָה טֶרֶף מַמָּשׁ לְעֵבֶר הַשּׁוֹמֵר…”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “לַמְרוֹת שֶׁנִּסָּה לְשַׁחֵד אֶת הַקְּצִינִים, הַסּוֹד הִתְגַּלָּה לַקֵּיסָר, וְגָזְרוּ לִתְלוֹת אֶת הַשּׁוֹמֵר.”.
דיווח: כתבו ידיעה קצרה שכותרתה “הצלת אחות ברוריה מידי הרומאים”.

המחזה

רגע מן הסיפורהדמויותמה יאמרו או יעשו?
הוּא לֹא סוֹמֵךְ עַל כּוֹחוֹ, הוּא סוֹמֵךְ רַק עַל הַשֵּׁם יִתְבָּרַךְ.
הוּא נָשַׁם עָמֹק וְהִבִּיט בְּרַבִּי מֵאִיר בִּירְאַת כָּבוֹד וְתַדְהֵמָה.

7. חיבור לחיים מתוך המעשה

א. מה אפשר ללמוד מן הדרך שבה רבי מאיר וברוריה פעל באירוע “אִם יִגָּמֵר הַכֶּסֶף וְיָבוֹאוּ לַהֲרוֹג אוֹתְךָ אוֹ יְשַׁסּוּ בְּךָ כְּלָבִים, אַתָּה תֹּאמַר רַק מִשְׁפָּט אֶחָד קָטָן,…”?
