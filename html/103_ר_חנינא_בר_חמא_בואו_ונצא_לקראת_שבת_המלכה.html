סיפור 103 • ר' חנינא בר חמא • "בואו ונצא לקראת שבת המלכה"

"בואו ונצא לקראת שבת המלכה"

דף עבודה בעקבות ר' חנינא בר חמא
שם: כיתה:

המעשה — הסיפור המלא

פְּתִיחָה: רַעַשׁ, אָבָק וְזֵעָה

הַשֶּׁמֶשׁ שֶׁל יוֹם שִׁשִּׁי עָמְדָה בְּאֶמְצַע הַשָּׁמַיִם מֵעַל הָעִיר צִפּוֹרִי שֶׁבַּגָּלִיל. הָיָה חַם. חַם מְאֹד.

יְלָדִים, אִם הֱיִיתֶם עוֹצְמִים אֶת הָעֵינַיִם בְּאוֹתוֹ רֶגַע, מָה הֱיִיתֶם שׁוֹמְעִים? מִכָּל עֵבֶר נִשְׁמְעוּ דְּפִיקוֹת פַּטִּישׁ שֶׁל נַגָּרִים: טְרַאח! בּוּם! . מֵהַשָּׂדוֹת נִשְׁמְעוּ צְעָקוֹת שֶׁל חַקְלָאִים הַמְזָרְזִים אֶת הַחֲמוֹרִים שֶׁלָּהֶם. מֵהַמִּטְבָּחִים עָלָה קוֹל תִּסְסוּס הַשֶּׁמֶן בַּמַּחֲבַתּוֹת.

וְאִם הֱיִיתֶם לוֹקְחִים נְשִׁימָה עֲמֻקָּה? הֱיִיתֶם מְרִיחִים עִרְבּוּב שֶׁל רֵיחוֹת: רֵיחַ נִפְלָא שֶׁל תַּבְשִׁילֵי דָּגִים וְחַלּוֹת שֶׁנֶּאֱפוֹת בַּתַּנּוּר, אֲבָל יַחַד אִתּוֹ... רֵיחַ כָּבֵד שֶׁל זֵעָה, שֶׁל אֲנָשִׁים שֶׁעָבְדוּ קָשֶׁה מְאֹד כָּל הַשָּׁבוּעַ.

בָּרְחוֹבוֹת שֶׁל צִפּוֹרִי, מִהֲרוּ אֲנָשִׁים בְּצַעֲדֵי רִיצָה. הַבְּגָדִים שֶׁלָּהֶם הָיוּ מְלֵאִים בַּאֲדָמָה וּבְאֵפֶר, הַיָּדַיִם הָיוּ מְלֻכְלָכוֹת מִיְּגִיעַ הַמְּלָאכָה.

"מַהֵר! מַהֵר!" צָעַק נַפָּח אֶחָד לִבְנוֹ, "הַשֶּׁמֶשׁ כְּבָר נוֹטָה מַעֲרָבָה! עוֹד מְעַט שַׁבָּת וְלֹא הִסְפַּקְנוּ לַחְטֹב עֵצִים!".

"הַמַּיִם עֲדַיִן לֹא חַמִּים!" רָטְנָה אִשָּׁה אַחַת בְּלַחַץ עַל יַד הַבְּאֵר. "מָתַי כְּבָר נַסְפִּיק לִרְחֹץ אֶת הַיְּלָדִים?".

הַכֹּל הָיָה בְּלַחַץ. הַכֹּל הָיָה מָתוּחַ. הַשַּׁבָּת קְרוֹבָה, אֲבָל בִּמְקוֹם לִשְׂמֹחַ, אֲנָשִׁים הִרְגִּישׁוּ שֶׁהֵם פָּשׁוּט כּוֹרְעִים תַּחַת הָעֹמֶס. הָעֲיֵפוּת שֶׁל יוֹם שִׁשִּׁי גָּרְמָה לִמְרִיבוֹת קְטַנּוֹת. הַקָּדוֹשׁ בָּרוּךְ הוּא נָתַן אֶת הַשַּׁבָּת בְּמַתָּנָה, אֲבָל בְּאוֹתָם רְגָעִים מְלֵאֵי אָבָק שֶׁל עֶרֶב שַׁבָּת, אֲנָשִׁים רַק רָצוּ לְנַקּוֹת אֶת הַבּוֹץ וְלָנוּחַ.

הָרַב שֶׁאוֹהֵב אֶת הָאֲדָמָה

בְּתוֹךְ כָּל הָרַעַשׁ הַזֶּה, עָמַד אִישׁ זָקֵן שֶׁקָּמָתוֹ זְקוּפָה וְעֵינָיו מְאִירוֹת כְּכוֹכָבִים. זֶה הָיָה הַצַּדִּיק, סַבָּא קַדִּישָׁא, רַבִּי חֲנִינָא בַּר חָמָא .

רַבִּי חֲנִינָא הָיָה גְּדוֹל הַדּוֹר בְּאֶרֶץ יִשְׂרָאֵל. לַמְרוֹת שֶׁהָיָה כְּבָר זָקֵן מְאֹד, וַאֲפִלּוּ בֶּן שְׁמוֹנִים שָׁנָה יָכוֹל הָיָה לַעֲמֹד עַל רֶגֶל אַחַת וְלִנְעֹל אֶת נַעֲלוֹ כְּבָחוּר צָעִיר! (כָּךְ שָׁמַר לוֹ ה' עַל בְּרִיאוּתוֹ כִּי שָׁמַר עַל הַתּוֹרָה).

רַבִּי חֲנִינָא יָרַד אֶל הָרְחוֹב. גַּם הוּא לֹא נָח בְּיוֹם חֹל. חֲכָמֵינוּ מְסַפְּרִים שֶׁמֵּרֹב אַהֲבָתוֹ לְאֶרֶץ יִשְׂרָאֵל, רַבִּי חֲנִינָא הָיָה יורד לַדְּרָכִים וּמְסַקֵּל (מֵזִיז) הַחוּצָה בְּעַצְמוֹ אֲבָנִים וּסְלָעִים! כֵּן, רֹאשׁ הַיְשִׁיבָה הַגָּדוֹל, כְּדֵי שֶׁדַּרְכֵי אֶרֶץ יִשְׂרָאֵל יִהְיוּ מְיֻשָּׁרוֹת וְיָפוֹת וְאִישׁ לֹא יִמְעַד!

בְּאוֹתוֹ יוֹם שִׁשִּׁי, הוּא חָזַר מֵעֲבוֹדָתוֹ זוֹ. גַּם עַל בְּגָדָיו שֶׁל הָרַב הָיָה מְעַט אָבָק מֵאֲדְמַת הַקֹּדֶשׁ. הוּא רָאָה אֶת תּוֹשָׁבֵי צִפּוֹרִי בְּלַחַץ. הוּא רָאָה אֶת הַכַּעַס בְּעֵינֵיהֶם. הוּא יָדַע שֶׁהֵם עֲיֵפִים.

אֲבָל רַבִּי חֲנִינָא גַּם יָדַע סוֹד. הוּא יָדַע, שֶׁשַּׁבָּת הִיא לֹא עוֹד יוֹם בַּלּוּחַ הַשָּׁנָה. הוּא יָדַע שֶׁאִם רַק מְשַׁנִּים אֶת הַמַּחֲשָׁבָה, הַלַּחַץ יָכוֹל לַהֲפֹךְ לְהִתְרַגְּשׁוּת וּלְשִׂמְחָה עֲצוּמָה.

הַהִשְׁתַּנּוּת

הַשֶּׁמֶשׁ יָרְדָה עוֹד יוֹתֵר נָמוּךְ, צוֹבַעַת אֶת בָּתֵּי עִיר בְּצֶבַע שֶׁל כָּתֹם, אָדֹם וְזָהָב.

הִגִּיעָה הַשָּׁעָה שֶׁנִּקְרֵאת 'אַפַּנְיָא דְּמַעֲלֵי שַׁבְּתָא' (לִפְנֵי כְּנִיסַת שַׁבָּת).

רַבִּי חֲנִינָא בַּר חָמָא נִכְנַס לְבֵיתוֹ. הוּא שָׁטַף אֶת עַצְמוֹ בְּמַיִם, וְהֵסִיר אֶת כָּל אֲבַק הַחֹל. וְאָז – הוּא לָבַשׁ אֶת הַבְּגָדִים שֶׁל שַׁבָּת. בְּגָדִים לְבָנִים, מְצֻחְצָחִים, רַכִּים וּנְקִיִּים.

בָּרֶגַע שֶׁהוּא הִתְעַטֵּף בַּבֶּגֶד שֶׁל שַׁבָּת, מַשֶּׁהוּ מַדְהִים קָרָה לַפָּנִים שֶׁלּוֹ. קִמְטֵי הַדְּאָגָה נֶעֶלְמוּ. חִיּוּךְ רָחָב, עָמֹק וְשָׁלֵו הֵאִיר אֶת עֵינָיו. הוּא הִרְגִּישׁ שֶׁנְּשָׁמָה יְתֵרָה יורֶדֶת מִן הַשָּׁמַיִם וְנִכְנֶסֶת יְשִׁירוֹת לְתוֹךְ הַלֵּב שֶׁלּוֹ! הָעֲיֵפוּת נֶעֶלְמָה! הָעֶצֶב נֶעֱלַם! הוּא הִרְגִּישׁ... כְּמוֹ נָסִיךְ שֶׁעוֹמֵד לִפְגֹּשׁ אֶת בַּת הַמֶּלֶךְ!

רַבִּי חֲנִינָא הַגָּדוֹל, רֹאשׁ הַיְשִׁיבָה וּמַנְהִיג הַדּוֹר, יָצָא שׁוּב לָרְחוֹב.

הָרְחוֹב הָיָה עֲדַיִן סוֹעֵר. פֹּה תִּינוֹק בָּכָה, שָׁם אָדָם רָץ עִם דְּלִי מַיִם, וּבַקָּצֶה הַשֵּׁנִי שְׁנֵי אֲנָשִׁים הִתְוַכְּחוּ.

אֲבָל פִּתְאוֹם, עָמַד רַבִּי חֲנִינָא בְּאֶמְצַע הָעִיר, סִדֵּר אֶת גְּלִימָתוֹ הַלְּבָנָה סְבִיב כְּתֵפָיו, עָצַם עֵינַיִם – וְקָרָא קְרִיאָה בְּקוֹל כֹּה צָלוּל, כֹּה מָלֵא בְּאַהֲבָה וְשִׂמְחָה, עַד שֶׁכָּל תּוֹשָׁבֵי צִפּוֹרִי שָׁמְעוּ אוֹתָהּ. הוּא אָמַר:

"בּוֹאוּ! וְנֵצֵא לִקְרַאת שַׁבָּת... הַמַּלְכָּה!"

הַקֶּסֶם שֶׁל שַׁבָּת הַמַּלְכָּה

יְלָדִים, אִם רַבִּי חֲנִינָא הָיָה צוֹעֵק עֲלֵיהֶם: "לָמָּה אַתֶּם בְּלַחַץ? תַּפְסִיקוּ לִצְעֹק!", אוּלַי הֵם הָיוּ נֶעֱלָבִים וְכוֹעֲסִים.

אֲבָל הוּא לֹא צָעַק צְעָקָה שֶׁל כַּעַס. הוּא שָׁר שִׁיר שֶׁל הַזְמָנָה! הוּא פָּשׁוּט קָרָא לָהֶם לָבוֹא וּלְקַבֵּל בְּפָנִים מְאִירוֹת אֶת הַדָּבָר הֲכִי יָקָר בָּעוֹלָם.

הוּא אָמַר: שַׁבָּת הִיא לֹא מַשָּׂא שֶׁסּוֹחֲבִים עַל הַגַּב. הִיא מַלְכָּה ! וּכְשֶׁמַּלְכָּה גְּדוֹלָה מַגִּיעָה לָעִיר, קוֹפְצִים, רוֹקְדִים וְיוֹצְאִים לִקְרָאתָהּ יַחַד!

וְאָז, הַקֶּסֶם קָרָה. דִּבְרֵי הַתּוֹרָה הַקְּדוֹשִׁים, הַיּוֹצְאִים מִלֵּב טָהוֹר – נִכְנָסִים יָשָׁר אֶל הַלֵּב!

הַנַּפָּח הִנִּיחַ אֶת הַפַּטִּישׁ שֶׁלּוֹ בְּשֶׁקֶט עַל הַשֻּׁלְחָן. הוּא מָחָה אֶת הַזֵּעָה מִמִּצְחוֹ וְחִיֵּךְ אֶל בְּנוֹ. "זֶהוּ יַלְדִּי. נָנוּחַ. הַמַּלְכָּה כָּאן".

הָאִשָּׁה עַל יַד הַבְּאֵר עָצְרָה אֶת אֲנָחָתָהּ, חִבְּקָה אֶת רֹאשׁוֹ שֶׁל הַיֶּלֶד שֶׁלָּהּ, וְאָמְרָה בְּרֹךְ: "שְּׁשְּׁשּׁ... בּוֹאוּ נִלְבַּשׁ בִּגְדֵי שַׁבָּת".

אֲנָשִׁים שֶׁהִתְוַכְּחוּ עָצְרוּ, הִבִּיטוּ זֶה בָּזֶה, מָשְׁכוּ כְּתֵפַיִם, וְאָמְרוּ: "שַׁבָּת שָׁלוֹם אָחִי, מְחִילָה".

פִּתְאוֹם, בְּרֶגַע אֶחָד, הָרְחוֹבוֹת שֶׁל צִפּוֹרִי לֹא הָיוּ מְלֵאִים בְּאֲנָשִׁים עֲיֵפִים. הֵם הָיוּ מְלֵאִים בִּבְנֵי מְלָכִים. אֲנָשִׁים וִילָדִים שֶׁהִתְרַחֲצוּ וְלָבְשׁוּ לָבָן, יָצְאוּ מֵהַבָּתִּים וְהָלְכוּ אֶל הַשָּׂדֶה אוֹ לַמִּרְפֶּסֶת, נֶעֶמְדוּ סְבִיב הַצַּדִּיק שֶׁעָמַד שָׁם כְּמַלְאָךְ ה', וְקִבְּלוּ בְּשִׂמְחָה וּבְשִׁירָה אֶת שַׁבָּת קֹדֶשׁ.

הַשֶּׁמֶשׁ צָלְלָה מֵאֲחוֹרֵי הֶהָרִים, רוּחַ קְרִירָה וּנְעִימָה הֵחֵלָּה לְנַשֵּׁב בָּעִיר, וְשַׁבָּת שֶׁל מְנוּחָה, קְדֻשָּׁה וְאַהֲבַת ה' יָרְדָה עַל עַם יִשְׂרָאֵל.

מקורות: המקורות מופיעים בקובץ המקורי.

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “הַשֶּׁמֶשׁ שֶׁל יוֹם שִׁשִּׁי עָמְדָה בְּאֶמְצַע הַשָּׁמַיִם מֵעַל הָעִיר צִפּוֹרִי שֶׁבַּגָּלִיל.” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “הַמַּיִם עֲדַיִן לֹא חַמִּים!" רָטְנָה אִשָּׁה אַחַת בְּלַחַץ עַל יַד הַבְּאֵר.”?
ג. כיצד הגיב ר' חנינא בר חמא למצב המתואר: “(כָּךְ שָׁמַר לוֹ ה' עַל בְּרִיאוּתוֹ כִּי שָׁמַר עַל הַתּוֹרָה).”?
ד. מה השתנה בין “הִגִּיעָה הַשָּׁעָה שֶׁנִּקְרֵאת 'אַפַּנְיָא דְּמַעֲלֵי שַׁבְּתָא' (לִפְנֵי כְּנִיסַת שַׁבָּת).” לבין “הוּא פָּשׁוּט קָרָא לָהֶם לָבוֹא וּלְקַבֵּל בְּפָנִים מְאִירוֹת אֶת הַדָּבָר הֲכִי יָקָר בָּעוֹלָם.”?
ה. כיצד הגיע הסיפור אל האירוע: “אֲנָשִׁים שֶׁהִתְוַכְּחוּ עָצְרוּ, הִבִּיטוּ זֶה בָּזֶה, מָשְׁכוּ כְּתֵפַיִם, וְאָמְרוּ: "שַׁבָּת שָׁלוֹם אָחִי, מְחִילָה".”?

3. מה היה ראשון?

כתבו ליד כל אירוע מספר מ־1 עד 5 לפי סדר התרחשותו בסיפור.

□ הוּא יָדַע שֶׁאִם רַק מְשַׁנִּים אֶת הַמַּחֲשָׁבָה, הַלַּחַץ יָכוֹל לַהֲפֹךְ לְהִתְרַגְּשׁוּת וּלְשִׂמְחָה עֲצוּמָה.
□ הַשֶּׁמֶשׁ שֶׁל יוֹם שִׁשִּׁי עָמְדָה בְּאֶמְצַע הַשָּׁמַיִם מֵעַל הָעִיר צִפּוֹרִי שֶׁבַּגָּלִיל.
□ הַשֶּׁמֶשׁ צָלְלָה מֵאֲחוֹרֵי הֶהָרִים, רוּחַ קְרִירָה וּנְעִימָה הֵחֵלָּה לְנַשֵּׁב בָּעִיר, וְשַׁבָּת שֶׁל מְנוּחָה, קְדֻשָּׁה וְאַהֲבַת ה' יָרְדָה עַל…
□ הַקָּדוֹשׁ בָּרוּךְ הוּא נָתַן אֶת הַשַּׁבָּת בְּמַתָּנָה, אֲבָל בְּאוֹתָם רְגָעִים מְלֵאֵי אָבָק שֶׁל עֶרֶב שַׁבָּת, אֲנָשִׁים רַק רָצוּ לְנַקּוֹת אֶת…
□ הוּא אָמַר: שַׁבָּת הִיא לֹא מַשָּׂא שֶׁסּוֹחֲבִים עַל הַגַּב.

לפני ואחרי

מה קרה לפני: “הוּא יָדַע שֶׁאִם רַק מְשַׁנִּים אֶת הַמַּחֲשָׁבָה, הַלַּחַץ יָכוֹל לַהֲפֹךְ לְהִתְרַגְּשׁוּת וּלְשִׂמְחָה עֲצוּמָה.”?
מה קרה אחרי: “הוּא אָמַר: שַׁבָּת הִיא לֹא מַשָּׂא שֶׁסּוֹחֲבִים עַל הַגַּב.”?

סיבה ותוצאה מתוך המעשה

האירוע הנתוןמה גרם לו?מה הייתה תוצאתו?
בְּתוֹךְ כָּל הָרַעַשׁ הַזֶּה, עָמַד אִישׁ זָקֵן שֶׁקָּמָתוֹ זְקוּפָה וְעֵינָיו מְאִירוֹת כְּכוֹכָבִים.
הוּא הִרְגִּישׁ שֶׁנְּשָׁמָה יְתֵרָה יורֶדֶת מִן הַשָּׁמַיִם וְנִכְנֶסֶת יְשִׁירוֹת לְתוֹךְ הַלֵּב שֶׁלּוֹ!

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “"בואו ונצא לקראת שבת המלכה"” על ר' חנינא בר חמא? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “גַּם עַל בְּגָדָיו שֶׁל הָרַב הָיָה מְעַט אָבָק מֵאֲדְמַת הַקֹּדֶשׁ.”.
ג. אילו ר' חנינא בר חמא לא היה פועל כפי שמתואר ב־“הוּא פָּשׁוּט קָרָא לָהֶם לָבוֹא וּלְקַבֵּל בְּפָנִים מְאִירוֹת אֶת הַדָּבָר הֲכִי יָקָר בָּעוֹלָם.” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “אֲנָשִׁים שֶׁהִתְוַכְּחוּ עָצְרוּ, הִבִּיטוּ זֶה בָּזֶה, מָשְׁכוּ כְּתֵפַיִם, וְאָמְרוּ: "שַׁבָּת שָׁלוֹם אָחִי, מְחִילָה".” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. הַשֶּׁמֶשׁ שֶׁל יוֹם שִׁשִּׁי עָמְדָה בְּאֶמְצַע הַשָּׁמַיִם מֵעַל…
2. הַמַּיִם עֲדַיִן לֹא חַמִּים!" רָטְנָה אִשָּׁה אַחַת בְּלַחַץ עַל יַד…
3. חֲכָמֵינוּ מְסַפְּרִים שֶׁמֵּרֹב אַהֲבָתוֹ לְאֶרֶץ יִשְׂרָאֵל, רַבִּי…
4. בָּרֶגַע שֶׁהוּא הִתְעַטֵּף בַּבֶּגֶד שֶׁל שַׁבָּת, מַשֶּׁהוּ מַדְהִים…
5. דִּבְרֵי הַתּוֹרָה הַקְּדוֹשִׁים, הַיּוֹצְאִים מִלֵּב טָהוֹר –…
6. הַשֶּׁמֶשׁ צָלְלָה מֵאֲחוֹרֵי הֶהָרִים, רוּחַ קְרִירָה וּנְעִימָה…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “(כָּךְ שָׁמַר לוֹ ה' עַל בְּרִיאוּתוֹ כִּי שָׁמַר עַל הַתּוֹרָה).”.
מכתב: כתבו לר' חנינא בר חמא בעקבות “הוּא הִרְגִּישׁ שֶׁנְּשָׁמָה יְתֵרָה יורֶדֶת מִן הַשָּׁמַיִם וְנִכְנֶסֶת יְשִׁירוֹת לְתוֹךְ הַלֵּב שֶׁלּוֹ!”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “דִּבְרֵי הַתּוֹרָה הַקְּדוֹשִׁים, הַיּוֹצְאִים מִלֵּב טָהוֹר – נִכְנָסִים יָשָׁר אֶל הַלֵּב!”.
דיווח: כתבו ידיעה קצרה שכותרתה “"בואו ונצא לקראת שבת המלכה"”.

המחזה

הוּא פָּשׁוּט קָרָא לָהֶם לָבוֹא וּלְקַבֵּל בְּפָנִים מְאִירוֹת אֶת הַדָּבָר הֲכִי יָקָר בָּעוֹלָם.
