מבקר את תלמידו החולה

דף עבודה בעקבות רבי עקיבא
שם: כיתה:

המעשה — הסיפור המלא

להלן מערך חינוכי מקיף, רגשי ומרתק, המבוסס על הסיפור המפורסם של רבי עקיבא המבקר את תלמידו החולה . התוכן מותאם בדיוק לבנים בכיתות ב'-ד' בחינוך הדתי-לאומי, וערוך על פי כל כללי הסטוריטלינג שביקשת.

הסיפור מנוקד במלואו, תוך הקפדה עליונה על כבוד התורה וחכמיה.

פְּתִיחָה: קוֹלָהּ שֶׁל תּוֹרָה וְהַכִּסֵּא הָרֵיק

יְלָדִים, עִצְמוּ רֶגַע עֵינַיִם וְנַסּוּ לְדַמְיֵן אֶת בֵּית הַמִּדְרָשׁ הַגָּדוֹל שֶׁל גְּדוֹל הַתַּנָּאִים, רַבִּי עֲקִיבָא. אַלְפֵי תַּלְמִידִים יוֹשְׁבִים עַל סַפְסָלִים אֲרֻכִּים. אֵיזֶה קוֹל נִשְׁמָע שָׁם? קוֹל אַדִּיר, כְּמוֹ גַּלִּים שֶׁל יָם סוֹעֵר! קוֹלוֹת שֶׁל לִמּוּד תּוֹרָה, שֶׁל שְׁאֵלוֹת וּתְשׁוּבוֹת, מְתוּקִים מִדְּבַשׁ. כֻּלָּם מַבִּיטִים קָדִימָה, בְּיִרְאַת כָּבוֹד, אֶל רַבָּם הַגָּדוֹל שֶׁיּוֹשֵׁב בָּרֹאשׁ וְדוֹרֵשׁ לָהֶם אֶת סוֹדוֹת הַתּוֹרָה [1].

אֲבָל יוֹם אֶחָד, בְּאֶמְצַע הַשִּׁעוּר, עֵינָיו הַקְּדוֹשׁוֹת שֶׁל רַבִּי עֲקִיבָא סָרְקוּ אֶת מֵאוֹת הַסַּפְסָלִים. רַבִּי עֲקִיבָא הָיָה כְּמוֹ אַבָּא אֹוהֵב לְכָל תַּלְמִיד. הוּא שָׂם לֵב לְמַשֶּׁהוּ שֶׁאַף אֶחָד אַחֵר כִּמְעַט לֹא שָׂם לֵב אֵלָיו. בְּאַחַד הַסַּפְסָלִים הָאֲחוֹרִיִּים – הָיָה כִּסֵּא אֶחָד רֵיק.

"הֵיכָן תַּלְמִידִי הָאָהוּב?" שָׁאַל רַבִּי עֲקִיבָא אֶת הַיּוֹשְׁבִים לְיָדוֹ. הַתַּלְמִידִים הִבִּיטוּ יָמִינָה וּשְׂמֹאלָה. "כְּבוֹד הָרַב," הֵשִׁיב אֶחָד מֵהֶם, "הוּא לֹא הִגִּיעַ כְּבָר יוֹמַיִם. שָׁמַעְנוּ שֶׁהוּא חוֹלֶה מְאֹד. אֲבָל... שָׁקַעְנוּ בְּלִמּוּד הַתּוֹרָה, וְטֶרֶם הִסְפַּקְנוּ לְבַקְּרוֹ" [1].

רַבִּי עֲקִיבָא לֹא חִכָּה אֲפִלּוּ רֶגַע אֶחָד עֲבוּר תַּלְמִידוֹ. הוּא סָגַר אֶת גְּוִיל הַתּוֹרָה וְקָם מִמְּקוֹמוֹ. "אִם תַּלְמִידִי חוֹלֶה – אֲנִי הוֹלֵךְ לְבַקֵּר אוֹתוֹ עַכְשָׁו," פָּסַק. כַּמָּה מֵהַתַּלְמִידִים נִבְהֲלוּ וּמִהֲרוּ לְהִצְטָרֵף לְרַבָּם כְּדֵי שֶׁלֹּא יֵלֵךְ לְבַדּוֹ [2].

הַהֲלִיכָה אֶל הַבַּיִת הַחָשׁוּךְ וְהַמַּחֲנִיק

הֵם צָעֲדוּ בְּסִמְטְאוֹתֶיהָ שֶׁל יְרוּשָׁלַיִם. הַשֶּׁמֶשׁ קָפְחָה מִלְמַעְלָה. הֵם הִגִּיעוּ לְבַיִת קָטָן בִּקְצֵה הָעִיר, וְדָפְקוּ קַלּוֹת עַל הַדֶּלֶת הֶעָשׁוּיָה מֵעֵץ יָשָׁן. דְּמָמָה הָיְתָה בִּפְנִים. לֹא נִשְׁמְעָה כָּל תְּשׁוּבָה.

רַבִּי עֲקִיבָא דָּחַף בַּעֲדִינוּת אֶת הַדֶּלֶת וְנִכְנַס. מִיָּד כְּשֶׁנִּכְנְסוּ, הִכָּה בְּאַפָּם גַּל שֶׁל חֹם וְרֵיחַ כָּבֵד וְרָע. הַחַלּוֹנוֹת הָיוּ סְגוּרִים וּמוּגָפִים, אֲוִיר חַם וְדָחוּס עָמַד בַּחֶדֶר. מַמָּשׁ קָשֶׁה הָיָה לִנְשֹׁם שָׁם! שִׁכְבָה עָבָה שֶׁל אָבָק אֲפַרְפַּר כִּסְּתָה אֶת הַשֻּׁלְחָן הַקָּטָן, אֶת כַּד הַמַּיִם הָרֵיק, וְאֶת רִצְפַּת הָאֶבֶן [2].

וּבַפִּנָּה, עַל מִטָּה נְמוּכָה, שָׁכַב הַתַּלְמִיד. אֵיזֶה מַרְאֶה עָצוּב זֶה הָיָה. פָּנָיו שֶׁל הַבָּחוּר הָיוּ חִוְּרוֹת כְּמוֹ קִיר. שְׂפָתָיו הָיוּ סְדוּקוֹת וִיבֵשׁוֹת מֵחֹסֶר טִפַּת מַיִם. מִגְּרוֹנוֹ נִשְׁמְעוּ קוֹלוֹת צִפְצוּף וְשִׁעוּל דַּק. הוּא הָיָה חַלָּשׁ כָּל כָּךְ, שֶׁאֲפִלּוּ לֹא הָיָה לוֹ כֹּחַ לִפְקֹחַ אֶת עֵינָיו אוֹ לְבַקֵּשׁ עֶזְרָה [2]. הַתַּלְמִיד הָיָה נָתוּן בְּסַכָּנָה אֲמִתִּית!

הָרַב מַפְשִׁיל שַׁרְווּלִים!

יְלָדִים, מָה אָדָם חָשׁוּב הָיָה עוֹשֶׂה עַכְשָׁו? אוּלַי הָיָה עוֹמֵד מֵרָחוֹק, אוֹמֵר מִזְמוֹר תְּהִלִּים קָצָר לִרְפוּאָה, מוֹדִיעַ: "שֶׁתִּהְיֶה בָּרִיא!", וּמְמַהֵר לַחְזֹר לְבֵית הַמִּדְרָשׁ?

אַךְ לֹא רַבִּי עֲקִיבָא! רַבִּי עֲקִיבָא לִמֵּד אוֹתָנוּ שֶׁאָהֲבַת יִשְׂרָאֵל הִיא עֲבוֹדָה, לֹא רַק דִּבּוּרִים. עֵינָיו נִמְלְאוּ בְּרַחֲמִים וְהוּא הִסְתּוֹבֵב אֶל תַּלְמִידָיו הַבְּרִיאִים שֶׁבָּאוּ עִמּוֹ:

"לָמָּה אַתֶּם עוֹמְדִים?!" אָמַר לָהֶם בִּנְחִישׁוּת, "אֲנַחְנוּ מַתְחִילִים בַּמְּלָאכָה! מַהֲרוּ וְהָבִיאוּ מַיִם טְרִיִּים מִן הַבְּאֵר! פִּתְחוּ אֶת הַחַלּוֹנוֹת לִרְוָחָה! נַקּוּ אֶת הָאָבָק!" [3].

וְהֵם עָשׂוּ כָּךְ. מִיָּד נִפְתְּחוּ הַתְּרִיסִים הַכְּבֵדִים, וְקַרְנֵי שֶׁמֶשׁ נְעִימוֹת נִכְנְסוּ פְּנִימָה יַחַד עִם רוּחַ טוֹבָה. אֶחָד הַתַּלְמִידִים הֵבִיא כַּד מַיִם מָלֵא. רַבִּי עֲקִיבָא קָחַז בַּמַּטְאֲטֵא, וּבְעַצְמוֹ הִתְחִיל לְנַקּוֹת. אַחַר כָּךְ לָקַח קְעָרָה עִם מַיִם נְקִיִּים וְשָׁפַךְ עַל רִצְפַּת הָאֶבֶן שֶׁל הַחֶדֶר [3].

פְשְׁשְׁשׁ... קוֹל הַמַּיִם שֶׁנָּשְׁקוּ לָאֶבֶן הַחַמָּה נִשְׁמַע בַּחֶדֶר. מִיָּד הִתְפַּשֵּׁט בָּאֲוִיר רֵיחַ נִפְלָא שֶׁל נִקָּיוֹן וְטַל רַעֲנָן. הַקֹּר שֶׁל הַמַּיִם מִלֵּא אֶת הַחֶדֶר, וְהַחֹם הַמַּחֲנִיק נֶעֱלַם.

"הֶחֱיִיתַנִי, רַבִּי!"

רַבִּי עֲקִיבָא נִגַּשׁ לַתַּלְמִיד, שָׂם אֶת יָדוֹ הֶחַמָּה וְהָאוֹהֶבֶת עַל הַמֵּצַח הַלּוֹהֵט שֶׁל הַבָּחוּר, הִשְׁקָה אוֹתוֹ מְעַט מַיִם צוֹנְנִים עַל שְׂפָתָיו הַסְּדוּקוֹת, וְלִטֵּף אֶת רֹאשׁוֹ.

לְפֶתַע... רִיסָיו שֶׁל הַתַּלְמִיד רָעֲדוּ. הוּא פָּקַח אֶת עֵינָיו בְּאִטִּיּוּת. אֲוִיר נָקִי וְטָהוֹר נִכְנַס סוֹף-סוֹף לְרֵאוֹתָיו. מִצְבַּע לְחָיָיו חָזַר לִהְיוֹת וְרַרְדַם קְצָת. הוּא הִבִּיט יָמִינָה וְרָאָה עֵינַיִם אוֹהֲבוֹת מַבִּיטוֹת בּוֹ. הוּא זִהָה אֶת רַבּוֹ הַגָּדוֹל, גְּדוֹל הַדּוֹר!

חִיּוּךְ קָטָן עָלָה עַל שִׂפְתֵי הַתַּלְמִיד. הוּא לָקַח נְשִׁימָה עֲמֻקָּה, קוֹלוֹ כְּבָר לֹא צִפְצֵף, וְהוּא לָחַשׁ:

"רַבִּי... רַבִּי שֶׁלִּי... הֶחֱיִיתַנִי! אַתָּה מַמָּשׁ הִצַּלְתָּ אֶת הַחַיִּים שֶׁלִּי! אִלּוּלֵא בָּאתֶם לְהָבִיא לִי אֲוִיר וּמַיִם... וְנִקָּיוֹן... מִי יוֹדֵעַ אִם הָיִיתִי נִשְׁאָר בַּחַיִּים!" [3].

הַלֶּקַח הַגָּדוֹל בְּבֵית הַמִּדְרָשׁ

רַבִּי עֲקִיבָא דִּבֵּר עִמּוֹ דְּבָרִים נְעִימִים, חִזֵּק אֶת לִבּוֹ, בֵּרֵךְ אוֹתוֹ בִּרְפוּאָה שְׁלֵמָה מֵעֹמֶק לִבּוֹ, וְדָאַג שֶׁהַתַּלְמִידִים יַשְׁאִירוּ לוֹ אֹכֶל [4].

כַּאֲשֶׁר יָצְאוּ וְחָזְרוּ לְבֵית הַמִּדְרָשׁ, נֶעֱמַד רַבִּי עֲקִיבָא לִפְנֵי כָּל אַלְפֵי הַתַּלְמִידִים, דָּפַק עַל הַשֻּׁלְחָן וְהִכְרִיז לִפְנֵי כֻּלָּם כְּלָל שֶׁנִּזְכֹּר כָּל הַחַיִּים:

"יְלָדַי וְתַלְמִידַי הָאֲהוּבִים. דְּעוּ לָכֶם דָּבָר עָצוּם: מִי שֶׁאֵינוֹ מְבַקֵּר חֲבֵרוֹ הַחוֹלֶה, כְּאִלּוּ פּוֹגֵעַ בּוֹ וְשׁוֹפֵךְ אֶת דָּמוֹ! לֹא מַסְפִּיק רַק לַחְשֹׁב עָלָיו מֵרָחוֹק. צָרִיךְ לָלֶכֶת, לְנַקּוֹת לוֹ אֶת הַחֶדֶר, לְהָבִיא לוֹ אוֹכֶל, לִפְתֹּחַ חַלּוֹן, לְהִתְפַּלֵּל עָלָיו מִקָּרוֹב – וּלְשַׂמֵּחַ אֶת לִבּוֹ! בִּקּוּר חוֹלִים אֲמִתִּי – מֵבִיא חַיִּים!" [4].

א. שְׁאֵלַת פְּתִיחָה (לְפְנֵי הַסִּפּוּר):

"ילדים, תחשבו רגע – מי כאן היה פעם חולה ונשאר בבית כמה ימים רצופים? נכון שזה קצת משעמם? איך הרגשתם כשאף חבר לא התקשר או בא לשאול לשלומכם? (לתת לילדים לענות). ואיך הרגשתם כשפתאום דפקו בדלת, וחבר הגיע עם חיוך ומכתב החלמה? נכון שהכאב פתאום עבר קצת? היום נשמע על תלמיד אחד כזה, ועל האורח המפתיע שהגיע להציל את חייו!"

ב. הֶעָרוֹת בִּצּוּעַ וּסְטוֹרִיטֵלִינְג (איך לספר):

ג. הַמְלָצוֹת לְ-4 שְׁקוּפִיּוֹת בְּמַצֶּגֶת (צְנִיעוּת וּקְדֻשָּׁה):

ד. פְּעִילוּת מְסַכֶּמֶת (הַפְנָמָה): "עֶרְכַּת הַחַיִּים שֶׁל רַבִּי עֲקִיבָא"

פעילות יצירה: המורה מביא לכל ילד "כרטיס ביקור חולים" שבו כתוב "החיייתני!" מבחוץ. מבפנים הילדים יציירו ויכתבו ברכה יפה לחבר. המורה מבקש שכל ילד ישמור את הכרטיס בתיק, ובפעם הראשונה שחבר לכיתה חסר (בגלל מחלה), הילד ימסור לו את הברכה, כדי להיות ה"רבי עקיבא" שלו ולשמח אותו.

לחלופין (משחק תפקידים קצר): מזמינים 2 ילדים לבמה. אחד מציג חבר שיושב לבד בבית ומשעמם לו, והשני צריך "לשחק" חבר שבא לבקר – מה הוא אומר לו? איך הוא מצחיק אותו? מה הוא שואל אותו? (המורה מכוון אותם לא להגיד רק "תרגיש טוב", אלא לשאול "רוצה נשחק משחק קופסה?").

מקורות: תלמוד בבלי, מסכת נדרים, דף מ' עמוד א': "מעשה באחד מתלמידיו של ר"ע שחלה, ולא נכנסו חכמים לבקרו, ונכנס ר"ע לבקרו, ובשביל שכיבדו וריבצו לפניו (=טִאטאוּ והתיזו מים על הרצפה כדי לקרר ולנקות את האוויר) - חיה (=התרפא וניצל). אמר לו: 'רבי, החייתני!'. יצא ר"ע ודרש: כל מי שאינו מבקר חולים כאילו שופך דמים." הערה למקורות: בסיפור זה הורחב התיאור החושי של ה"ריבוץ והכיבוד" (פעולות הניקיון שהוזכרו בגמרא) לציור ברור וחי של מצוקת התלמיד וההקלה שהביאה העשייה הפיזית של רבי עקיבא, על מנת להנגיש את משמעות ה"החייתני" לגיל הצעיר. (מופיע גם באוספי סיפורים כמו "כה עשו חכמינו" וספרי הגות שעסקו בסיפור זה כגון קטע [1] עד [4] במקורות שצורפו).

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “להלן מערך חינוכי מקיף, רגשי ומרתק, המבוסס על הסיפור המפורסם של רבי עקיבא המבקר את תלמידו החולה .” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “אֲבָל יוֹם אֶחָד, בְּאֶמְצַע הַשִּׁעוּר, עֵינָיו הַקְּדוֹשׁוֹת שֶׁל רַבִּי עֲקִיבָא סָרְקוּ אֶת מֵאוֹת הַסַּפְסָלִים.”?
ג. כיצד הגיב רבי עקיבא למצב המתואר: “כַּמָּה מֵהַתַּלְמִידִים נִבְהֲלוּ וּמִהֲרוּ לְהִצְטָרֵף לְרַבָּם כְּדֵי שֶׁלֹּא יֵלֵךְ לְבַדּוֹ [2].”?
ד. מה השתנה בין “רַבִּי עֲקִיבָא לִמֵּד אוֹתָנוּ שֶׁאָהֲבַת יִשְׂרָאֵל הִיא עֲבוֹדָה, לֹא רַק דִּבּוּרִים.” לבין “רַבִּי עֲקִיבָא נִגַּשׁ לַתַּלְמִיד, שָׂם אֶת יָדוֹ הֶחַמָּה וְהָאוֹהֶבֶת עַל הַמֵּצַח הַלּוֹהֵט שֶׁל הַבָּחוּר,…”?
ה. כיצד הגיע הסיפור אל האירוע: “היום נשמע על תלמיד אחד כזה, ועל האורח המפתיע שהגיע להציל את חייו!”?

3. מה היה ראשון?

כתבו ליד כל אירוע מספר מ־1 עד 5 לפי סדר התרחשותו בסיפור.

□ הוּא הָיָה חַלָּשׁ כָּל כָּךְ, שֶׁאֲפִלּוּ לֹא הָיָה לוֹ כֹּחַ לִפְקֹחַ אֶת עֵינָיו אוֹ לְבַקֵּשׁ עֶזְרָה [2].
□ להלן מערך חינוכי מקיף, רגשי ומרתק, המבוסס על הסיפור המפורסם של רבי עקיבא המבקר את תלמידו החולה .
□ המורה מבקש שכל ילד ישמור את הכרטיס בתיק, ובפעם הראשונה שחבר לכיתה חסר (בגלל מחלה), הילד ימסור לו את הברכה, כדי להיות ה"רבי עקיבא" שלו ולשמח אותו.
□ בְּאַחַד הַסַּפְסָלִים הָאֲחוֹרִיִּים – הָיָה כִּסֵּא אֶחָד רֵיק.
□ הוּא הִבִּיט יָמִינָה וְרָאָה עֵינַיִם אוֹהֲבוֹת מַבִּיטוֹת בּוֹ.

לפני ואחרי

מה קרה לפני: “הוּא הָיָה חַלָּשׁ כָּל כָּךְ, שֶׁאֲפִלּוּ לֹא הָיָה לוֹ כֹּחַ לִפְקֹחַ אֶת עֵינָיו אוֹ לְבַקֵּשׁ עֶזְרָה [2].”?
מה קרה אחרי: “הוּא הִבִּיט יָמִינָה וְרָאָה עֵינַיִם אוֹהֲבוֹת מַבִּיטוֹת בּוֹ.”?

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “מבקר את תלמידו החולה” על רבי עקיבא? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “הַחַלּוֹנוֹת הָיוּ סְגוּרִים וּמוּגָפִים, אֲוִיר חַם וְדָחוּס עָמַד בַּחֶדֶר.”.
ג. אילו רבי עקיבא לא היה פועל כפי שמתואר ב־“רַבִּי עֲקִיבָא נִגַּשׁ לַתַּלְמִיד, שָׂם אֶת יָדוֹ הֶחַמָּה וְהָאוֹהֶבֶת עַל הַמֵּצַח הַלּוֹהֵט שֶׁל הַבָּחוּר, הִשְׁקָה אוֹתוֹ…” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “היום נשמע על תלמיד אחד כזה, ועל האורח המפתיע שהגיע להציל את חייו!” מחזק את המסר של הסיפור?
