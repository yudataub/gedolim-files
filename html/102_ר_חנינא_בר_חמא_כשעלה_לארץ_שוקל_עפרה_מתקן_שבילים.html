כשעלה לארץ — שוקל עפרה, מתקן שבילים

דף עבודה בעקבות ר' חנינא בר חמא
שם: כיתה:

המעשה — הסיפור המלא

להלן מערך חינוכי, רגשי ומרתק על רבי חנינא בר חמא ואהבתו לארץ ישראל , המעובד במיוחד לבנים בכיתות ב'-ד' (חינוך דתי-לאומי). הסיפור מדגיש את החשיבות של אהבת הארץ לא רק בדיבורים, אלא במעשים של ממש, מתוך קדושה ויראת שמים.

הסיפור מנוקד במלואו כנדרש.

פְּתִיחָה: הַגַּעְגּוּעַ בְּבָבֶל הָרְחוֹקָה

יְלָדִים, נַסּוּ לְדַמְיֵן אֶת עַצְמְכֶם נִמְצָאִים רָחוֹק-רָחוֹק מִכָּאן. לֹא בְּאֶרֶץ יִשְׂרָאֵל, אֶלָּא בְּאֶרֶץ בָּבֶל (שֶׁנִּקְרֵאת הַיּוֹם עִירָאק). בָּבֶל שֶׁל אוֹתָם יָמִים הָיְתָה מְדִינָה עֲשִׁירָה, מְלֵאָה בִּנְהָרוֹת עֲנָקִיִּים עִם מַיִם מְתוּקִים וְקָרִים, עֲצֵי דֶּקֶל וּפֵרוֹת עֲסִיסִיִּים. בְּבָבֶל יָשְׁבוּ חֲכָמִים גְּדוֹלִים וְלָמְדוּ תּוֹרָה בְּיְשִׁיבוֹת עֲנָקִיּוֹת.

אֶחָד מֵהַתַּלְמִידִים הַחֲכָמִים, הַצַּדִּיקִים וְהַמַּתְמִידִים בְּיוֹתֵר בְּבָבֶל הָיָה הָרַב הַגָּדוֹל, רַבִּי חֲנִינָא בַּר חָמָא .

כָּל יוֹם הוּא יָשַׁב בְּבֵית הַמִּדְרָשׁ, מֵהַבֹּקֶר עַד הַלַּיְלָה, וְלָמַד תּוֹרָה בִּקְדֻשָּׁה. אֲבָל... הָיָה מַשֶּׁהוּ שֶׁהָיָה חָסֵר לוֹ.

בְּכָל פַּעַם שֶׁרַבִּי חֲנִינָא הָיָה קוֹרֵא בַּתּוֹרָה עַל אֶרֶץ יִשְׂרָאֵל – עַל יְרוּשָׁלַיִם, עַל הָרֵי יְהוּדָה וְעַל הַגָּלִיל – דְּמָעוֹת הָיוּ עוֹמְדוֹת בְּעֵינָיו.

"מָה שָׁוִים כָּל הַנְּהָרוֹת וְהַדְּקָלִים שֶׁל בָּבֶל?" חָשַׁב בְּלִבּוֹ. "הַקָּדוֹשׁ בָּרוּךְ הוּא נָתַן לָנוּ אֶרֶץ קְדוֹשָׁה וּמְיֻחֶדֶת. אֶרֶץ אֲשֶׁר עֵינֵי ה' אֱלֹקֶיךָ בָּהּ! אֲנִי חַיָּב לַעֲלוֹת לְאֶרֶץ יִשְׂרָאֵל! גַּם אִם הַדֶּרֶךְ קָשָׁה, אֲנִי אוֹרֵז אֶת חֲפָצַי וְעוֹלֶה לָאָרֶץ!".

הַמַּסָּע הַגָּדוֹל בַּמִּדְבָּר

וְכָךְ, רַבִּי חֲנִינָא בַּר חָמָא יָצָא לַמַּסָּע. יְלָדִים, בְּאוֹתָם יָמִים לֹא הָיוּ מְטוֹסִים בַּשָּׁמַיִם, גַּם לֹא אוֹטוֹבּוּסִים מְמֻזָּגִים. הַמַּסָּע לְאֶרֶץ יִשְׂרָאֵל לָקַח חֳדָשִׁים אֲרֻכִּים בַּמִּדְבָּר הַגָּדוֹל.

הַשֶּׁמֶשׁ קָפְחָה עַל רֹאשׁוֹ וְהָיְתָה לוֹהֶטֶת כְּמוֹ תַּנּוּר. הָרוּחַ הֵעִיפָה גַּרְגְּרֵי חוֹל לְתוֹךְ עֵינָיו. שׁוֹדְדֵי דְּרָכִים הִסְתּוֹבְבוּ בַּאֲזוֹרִים אֵלּוּ. אֲבָל רַבִּי חֲנִינָא לֹא הִפְסִיק לִצְעֹד. בְּכָל צַעַד שֶׁהוּא עָשָׂה, לִבּוֹ דָּפַק בְּשִׂמְחָה: "הִנֵּה אֲנִי מִתְקָרֵב... הִנֵּה אֲנִי מַגִּיעַ לָאֲדָמָה שֶׁדָּרְכוּ עָלֶיהָ אַבְרָהָם, יִצְחָק וְיַעֲקֹב!".

הַהֲגָעָה לָאֲדָמָה הַקְּדוֹשָׁה

וְאָז, זֶה קָרָה. רַבִּי חֲנִינָא חָצָה אֶת הַגְּבוּל וְנִכְנַס לְתוֹךְ שִׁטְחָהּ שֶׁל אֶרֶץ יִשְׂרָאֵל.

הוּא הִשְׁתַּטֵּחַ (נָפַל) עַל הָאֲדָמָה הֶחָרֵבָה, מַמָּשׁ עַל הָרִצְפָּה מְלֵאַת הָאֲבָנִים וְהָאָבָק. הוּא הֵרִיחַ אֶת רֵיחָהּ שֶׁל הָאֲדָמָה הַיְּבֵשָׁה, רֵיחַ מָתוֹק וְטוֹב מִכָּל הַבְּשָׂמִים שֶׁל בָּבֶל, וְנָשַׁק לָאֲבָנִים שֶׁל אֶרֶץ יִשְׂרָאֵל. "בָּרוּךְ ה' אֱלֹקֵי יִשְׂרָאֵל, שֶׁהֵבִיא אוֹתִי לְאַרְצוֹ!" לָחַשׁ הָרַב.

כְּשֶׁהִגִּיעַ הָרַב לָעִיר, חֲכָמִים וְתַלְמִידִים רַבִּים קִבְּלוּ אֶת פָּנָיו בְּכָבוֹד עָצוּם. הֵם הֵכִינוּ לוֹ מָקוֹם יָפֶה בְּבֵית הַמִּדְרָשׁ, מָזְגוּ לוֹ מַיִם קָרִים מִן הַבְּאֵר כְּדֵי שֶׁיָּנוּחַ מֵהַמַּסָּע הַקָּשֶׁה. הֵם חָשְׁבוּ: הָרַב וַדַּאי שָׁבוּר מֵהַדֶּרֶךְ, וְעַכְשָׁו הוּא יֵשֵׁב תַּחַת עֵץ הַזַּיִת הַמֵּצֵל וְיִלְמַד תּוֹרָה בְּשֶׁקֶט.

אֲבָל כָּאן, יְלָדִים, קָרָה מַשֶּׁהוּ שֶׁאִישׁ מֵהַתַּלְמִידִים לֹא זָכָה לִרְאוֹת מֵעוֹלָם.

הָרַב הַגָּדוֹל הוֹפֵךְ לְפוֹעֵל בִּנְיָן

בִּמְקוֹם לָנוּחַ בַּצֵּל, רַבִּי חֲנִינָא קָם עַל רַגְלָיו. הוּא לָקַח בְּיָדָיו שַׂק עָשׂוּי מִבַּד גַּס, וְלָקַח מַעְדֵּר קָטָן שֶׁל פּוֹעֲלִים.

הוּא יָצָא אֶל דַּרְכֵי אֶרֶץ יִשְׂרָאֵל וְהִתְחִיל לַעֲבֹד.

אֵיזוֹ עֲבוֹדָה? הוּא רָאָה שֶׁרַבִּים מֵהַשְּׁבִילִים שֶׁל אֶרֶץ יִשְׂרָאֵל מְלֵאִים בַּאֲבָנִים חַדּוֹת, בְּקוֹצִים וּבְעֵשֶׂב שׁוֹטֶה שֶׁצָּמַח בְּאֶמְצַע הַדֶּרֶךְ.

"זֶה הַשְּׁבִיל שֶׁל אֶרֶץ הַקֹּדֶשׁ!" אָמַר רַבִּי חֲנִינָא. "כָּאן יְהוּדִים קְדוֹשִׁים הָלְכוּ! כָּאן יְהוּדִים יֵלְכוּ בְּעָתִיד! אֵיךְ אֶפְשָׁר שֶׁאֶרֶץ יִשְׂרָאֵל תִּהְיֶה כָּךְ עֲזוּבָה וּמְלֵאָה מִכְשׁוֹלִים?".

בְּזֵעָה רַבָּה שֶׁנָּטְפָה מִמִּצְחוֹ הַקָּדוֹשׁ, הָרַב, נְשִׂיא יִשְׂרָאֵל שֶׁהָיָה רָגִיל שֶׁמְּשָׁרְתִים אוֹתוֹ, הִתְכּוֹפֵף. יָדָיו הַלְּבָנוֹת הִתְלַכְלְכוּ בְּבֹץ. הוּא הֵרִים אֶבֶן אַחַר אֶבֶן. אֲבָנִים עֲנָקִיּוֹת, וַאֲבָנִים קְטַנּוֹת. הוּא נִיקָּה אֶת הַדֶּרֶךְ וְהֵזִיז אֶת תִּלֵּי הֶעָפָר וְהַלִּכְלוּךְ אֶל צִדֵּי הַדֶּרֶךְ ("מְתַקֵּן מַתְקָלֶיהָ שֶׁל אֶרֶץ יִשְׂרָאֵל"). הוּא סִדֵּר אֶת הַשְּׁבִיל שֶׁיִּהְיֶה יָשָׁר, נוֹחַ וְיָפֶה!

הַתַּלְמִידִים רָאוּ אֶת רַבָּם הַגָּדוֹל עוֹבֵד כְּפוֹעֵל בַּשֶּׁמֶשׁ הַחַמָּה, וְנִבְהֲלוּ. "רַבִּי! אֲנַחְנוּ נַעֲשֶׂה אֶת זֶה! עֲזֹב אֶת זֶה, זֶה לֹא לִכְבוֹדְךָ!".

אַךְ רַבִּי חֲנִינָא חִיֵּךְ חִיּוּךְ רָחָב, הִבִּיט עֲלֵיהֶם בְּעֵינָיו הַטְּהוֹרוֹת וְאָמַר: "וְכִי יֵשׁ כָּבוֹד גָּדוֹל יוֹתֵר מִלְּכַבֵּד וּלְנַקּוֹת אֶת אַדְמָתָהּ שֶׁל אֶרֶץ יִשְׂרָאֵל?".

הַמֹּאזְנַיִם וְהַחוֹל שֶׁל זָהָב

וְאָז הוּא עָשָׂה דָּבָר עוֹד יוֹתֵר מַפְלִיא וּמְרַגֵּשׁ.

מִתּוֹךְ מְעִילוֹ, הוֹצִיא רַבִּי חֲנִינָא כְּלִי נוֹצֵץ מִכֶּסֶף. אֵלּוּ הָיוּ מֹאזְנַיִם. יְלָדִים, אַתֶּם יוֹדְעִים שֶׁמֹּאזְנַיִם קְטַנִּים וּמְדֻיָּקִים כָּאֵלּוּ שַׁיָּכִים לְצוֹרְפֵי תַּכְשִׁיטִים? אֲנָשִׁים הָיוּ לוֹקְחִים מֹאזְנַיִם כָּאֵלּוּ כְּדֵי לִשְׁקֹל פֵּרוּרִים שֶׁל זָהָב יָקָר. וְאִם יֵשׁ פֵּרוּר אֶחָד יוֹתֵר מִדַּי, מִיָּד מוֹצִיאִים אוֹתוֹ שֶׁלֹּא יֵלֵךְ לְאִבּוּד.

לָקַח רַבִּי חֲנִינָא אֶת הַמֹּאזְנַיִם הַיְּקָרִים וְהֶעֱמִיד אוֹתָם. וּמָה הוּא שָׂם עֲלֵיהֶם? יַהֲלוֹמִים? זָהָב?

לֹא.

הוּא אָסַף בְּיָדָיו שְׁנֵי חוֹפְנִים שֶׁל אֲדָמָה חוּמָה וּפְשׁוּטָה שֶׁל אֶרֶץ יִשְׂרָאֵל! הוּא הִנִּיחַ קְצָת עָפָר עַל כַּף אַחַת, וּקְצָת עָפָר עַל הַכַּף הַשְּׁנִיָּה.

הַתַּלְמִידִים פָּעֲרוּ אֶת פִּיהֶם בְּהֶלֶם: "רַבִּי הַקָּדוֹשׁ... מָה אַתָּה עוֹשֶׂה? בִּשְׁבִיל מָה אַתָּה שׁוֹקֵל אֶת הֶעָפָר שֶׁל הָאֲדָמָה? חָסֵר עָפָר בָּעוֹלָם? יֵשׁ הֲמוֹן אֲדָמָה!".

הִנְמִיךְ רַבִּי חֲנִינָא אֶת קוֹלוֹ. הַמֹּאזְנַיִם עָלוּ וְיָרְדוּ עַד שֶׁהִשְׁתַּוּוּ לְגַמְרֵי. דִּמְעָה אַחַת שֶׁל הִתְרַגְּשׁוּת וְאַהֲבָה יָרְדָה עַל לֶחְיוֹ, וְאָז הוּא אָמַר אֶת הַמִּשְׁפָּט הַמַּדְהִים בְּיוֹתֵר, מִתּוֹךְ סֵפֶר תְּהִלִּים (ק"ב, ט"ו):

" כִּי רָצוּ עֲבָדֶיךָ אֶת אֲבָנֶיהָ... וְאֶת עֲפָרָהּ יְחֹנֵנוּ! " (שֶׁפֵּרוּשׁוֹ: הָעֲבָדִים שֶׁל ה', אוֹהֲבִים כָּל אֶבֶן מֵאֶרֶץ יִשְׂרָאֵל, וּמְחַבְּקִים וְרוֹצִים כָּל חֲתִיכַת עָפָר שֶׁלָּהּ!).

הוּא הִמְשִׁיךְ וְאָמַר לְתַלְמִידָיו: "אַתֶּם מְבִינִים, בָּנַי הַיְּקָרִים? בִּשְׁבִילִי – אֵין זֶה רַק אָבָק. רִגְבֵי הֶעָפָר שֶׁל אֶרֶץ יִשְׂרָאֵל הֵם יְקָרִים לִי הַרְבֵּה יוֹתֵר מִפֵּרוּרִים שֶׁל זָהָב אוֹ יַהֲלוֹמִים שֶׁל כֶּסֶף. אֲנִי בּוֹדֵק אוֹתָם, אֲנִי שׁוֹקֵל אוֹתָם בְּמֹאזְנַיִם כְּדֵי שֶׁכֻּלָּם יֵדְעוּ כַּמָּה הֶעָפָר הַזֶּה קָדוֹשׁ. הָאֲדָמָה הַזּוֹ הִיא מַתָּנָה מֵהַקָּדוֹשׁ בָּרוּךְ הוּא. וְאָנוּ צְרִיכִים לִשְׁמֹר עָלֶיהָ, לְנַקּוֹת אוֹתָהּ, וְלֶאֱהֹב כָּל אֶבֶן שֶׁלָּהּ!".

הַתַּלְמִידִים שָׁתְקוּ, וְעֵינֵיהֶם הִתְמַלְּאוּ בִּדְמָעוֹת. הֵם לָמְדוּ אֶת הַשִּׁעוּר הַגָּדוֹל בְּיוֹתֵר בְּאַהֲבַת אֶרֶץ יִשְׂרָאֵל.

א. שְׁאֵלַת פְּתִיחָה:

"יְלָדִים, מִי מִכֶּם יֵשׁ לוֹ בַּבַּיִת קֻפְסָה קְטַנָּה שֶׁל יַהֲלוֹמִים אוֹ תַּכְשִׁיטֵי זָהָב שֶׁל אִמָּא? אֵיךְ שׁוֹמְרִים עֲלֵיהֶם? נוֹעֲלִים בַּכַּסֶּפֶת, נָכוֹן? הַיּוֹם נִשְׁמַע עַל צַדִּיק גָּדוֹל, שֶׁשָּׁקַל מַשֶּׁהוּ מֻזָּר מְאֹד – כְּאִלּוּ הוּא הָיָה הַיַּהֲלוֹם הֲכִי יָקָר בָּעוֹלָם!"

ב. הֶעָרוֹת בִּצּוּעַ וְסְטוֹרִיטֶלִינְג (Show, Don't Tell):

ג. הַמְלָצוֹת לְמַצֶּגֶת (4 שְׁקוּפִיּוֹת תּוֹאֲמוֹת לַהֲלָכָה):

ד. פְּעִילוּת מְסַכֶּמֶת (הַפְנָמַת הַלֶּקַח):

מקורות: הַגְּמָרָא – מַסֶּכֶת כְּתֻבּוֹת קִי"ב ע"א: "רבי חנינא מתקן מתקליה" (רבי חנינא היה מתקן את מכשולי הדרכים בארץ ישראל מרוב חיבתה). הַתַּלְמוּד הַיְרוּשַׁלְמִי, מַסֶּכֶת שְׁבִיעִית, פֶּרֶק ד' הֲלָכָה ז': "ר' חנינא בר חמא כד סליק להכא הוה שקיל עפרה דהכא ויהיב עפרה דהכא הוה שקיל טינריה דין ויהיב טינריה דין דיהוי הדא שקול כהדא..." (כאשר עלה לכאן, לארץ, היה שוקל את עפרה... כדי לקיים את הפסוק "כי רצו עבדיך את אבניה"). סֵפֶר תְּהִלִּים, ק"ב, ט"ו: "כִּי רָצוּ עֲבָדֶיךָ אֶת אֲבָנֶיהָ, וְאֶת עֲפָרָהּ יְחֹנֵנוּ".

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “להלן מערך חינוכי, רגשי ומרתק על רבי חנינא בר חמא ואהבתו לארץ ישראל , המעובד במיוחד לבנים בכיתות ב'-ד' (חינוך דתי-לאומי).” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “בְּכָל פַּעַם שֶׁרַבִּי חֲנִינָא הָיָה קוֹרֵא בַּתּוֹרָה עַל אֶרֶץ יִשְׂרָאֵל – עַל יְרוּשָׁלַיִם, עַל הָרֵי יְהוּדָה וְעַל הַגָּלִיל – דְּמָעוֹת הָיוּ…”?
ג. כיצד הגיב ר' חנינא בר חמא למצב המתואר: “הוּא הִשְׁתַּטֵּחַ (נָפַל) עַל הָאֲדָמָה הֶחָרֵבָה, מַמָּשׁ עַל הָרִצְפָּה מְלֵאַת הָאֲבָנִים וְהָאָבָק.”?
ד. מה השתנה בין “הוּא לָקַח בְּיָדָיו שַׂק עָשׂוּי מִבַּד גַּס, וְלָקַח מַעְדֵּר קָטָן שֶׁל פּוֹעֲלִים.” לבין “וְאָז הוּא עָשָׂה דָּבָר עוֹד יוֹתֵר מַפְלִיא וּמְרַגֵּשׁ.”?
ה. כיצד הגיע הסיפור אל האירוע: “רִגְבֵי הֶעָפָר שֶׁל אֶרֶץ יִשְׂרָאֵל הֵם יְקָרִים לִי הַרְבֵּה יוֹתֵר מִפֵּרוּרִים שֶׁל זָהָב אוֹ יַהֲלוֹמִים שֶׁל כֶּסֶף.”?

לפני ואחרי

מה קרה לפני: “אֲבָל כָּאן, יְלָדִים, קָרָה מַשֶּׁהוּ שֶׁאִישׁ מֵהַתַּלְמִידִים לֹא זָכָה לִרְאוֹת מֵעוֹלָם.”?
מה קרה אחרי: “מִתּוֹךְ מְעִילוֹ, הוֹצִיא רַבִּי חֲנִינָא כְּלִי נוֹצֵץ מִכֶּסֶף.”?

סיבה ותוצאה מתוך המעשה

האירוע הנתוןמה גרם לו?מה הייתה תוצאתו?
הַשֶּׁמֶשׁ קָפְחָה עַל רֹאשׁוֹ וְהָיְתָה לוֹהֶטֶת כְּמוֹ תַּנּוּר.
בְּזֵעָה רַבָּה שֶׁנָּטְפָה מִמִּצְחוֹ הַקָּדוֹשׁ, הָרַב, נְשִׂיא יִשְׂרָאֵל שֶׁהָיָה רָגִיל שֶׁמְּשָׁרְתִים אוֹתוֹ, הִתְכּוֹפֵף.

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “כשעלה לארץ — שוקל עפרה, מתקן שבילים” על ר' חנינא בר חמא? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “הֵם הֵכִינוּ לוֹ מָקוֹם יָפֶה בְּבֵית הַמִּדְרָשׁ, מָזְגוּ לוֹ מַיִם קָרִים מִן הַבְּאֵר כְּדֵי שֶׁיָּנוּחַ מֵהַמַּסָּע הַקָּשֶׁה.”.
ג. אילו ר' חנינא בר חמא לא היה פועל כפי שמתואר ב־“וְאָז הוּא עָשָׂה דָּבָר עוֹד יוֹתֵר מַפְלִיא וּמְרַגֵּשׁ.” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “רִגְבֵי הֶעָפָר שֶׁל אֶרֶץ יִשְׂרָאֵל הֵם יְקָרִים לִי הַרְבֵּה יוֹתֵר מִפֵּרוּרִים שֶׁל זָהָב אוֹ יַהֲלוֹמִים שֶׁל כֶּסֶף.” מחזק את המסר של הסיפור?

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “הוּא הִשְׁתַּטֵּחַ (נָפַל) עַל הָאֲדָמָה הֶחָרֵבָה, מַמָּשׁ עַל הָרִצְפָּה מְלֵאַת הָאֲבָנִים וְהָאָבָק.”.
מכתב: כתבו לר' חנינא בר חמא בעקבות “בְּזֵעָה רַבָּה שֶׁנָּטְפָה מִמִּצְחוֹ הַקָּדוֹשׁ, הָרַב, נְשִׂיא יִשְׂרָאֵל שֶׁהָיָה רָגִיל שֶׁמְּשָׁרְתִים אוֹתוֹ, הִתְכּוֹפֵף.”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “הוּא אָסַף בְּיָדָיו שְׁנֵי חוֹפְנִים שֶׁל אֲדָמָה חוּמָה וּפְשׁוּטָה שֶׁל אֶרֶץ יִשְׂרָאֵל!”.
דיווח: כתבו ידיעה קצרה שכותרתה “כשעלה לארץ — שוקל עפרה, מתקן שבילים”.

המחזה

רגע מן הסיפורהדמויותמה יאמרו או יעשו?
הַשֶּׁמֶשׁ קָפְחָה עַל רֹאשׁוֹ וְהָיְתָה לוֹהֶטֶת כְּמוֹ תַּנּוּר.
וְאָז הוּא עָשָׂה דָּבָר עוֹד יוֹתֵר מַפְלִיא וּמְרַגֵּשׁ.

7. חיבור לחיים מתוך המעשה
