ביקור אצל ת"ח עוור — ענוותנותו

דף עבודה בעקבות רבי יהודה הנשיא (רבי)
שם: כיתה:

המעשה — הסיפור המלא

פְּתִיחָה: שַׁיֶּרֶת הַנָּשִׂיא בְּדַרְכֵי הָאָרֶץ

יְלָדִים, נַסּוּ לְדַמְיֵן מַסָּע שֶׁל מֶלֶךְ. כְּשֶׁמֶּלֶךְ יוֹצֵא לַדֶּרֶךְ, יֵשׁ לְפָנָיו שׁוֹמְרִים, סוּסִים אֲצִילִיִּים הָרוֹקְעִים בְּפַרְסוֹתֵיהֶם עַל הָאֲדָמָה, וּמֶרְכָּבוֹת מְפֹאָרוֹת הַמַּעֲלוֹת עֲנַן אָבָק.

כָּךְ בְּדִיּוּק נִרְאֲתָה שַׁיַּרְתּוֹ שֶׁל רַבִּי יְהוּדָה הַנָּשִׂיא, שֶׁכֻּלָּם קָרְאוּ לוֹ פָּשׁוּט: "רַבִּי" אוֹ "רַבֵּינוּ הַקָּדוֹשׁ".

רַבִּי הָיָה הָאִישׁ הֶחָשׁוּב בְּיוֹתֵר בְּעַם יִשְׂרָאֵל. מַנְהִיג הַסַּנְהֶדְרִין. הַקָּדוֹשׁ בָּרוּךְ הוּא נָתַן לוֹ עֹשֶׁר עָצוּם שֶׁלֹּא הָיָה כְּמוֹתוֹ לְאִישׁ, וְגַם כָּבוֹד אֲפִלּוּ אֵצֶל קֵיסַר רוֹמָא שֶׁהָיָה מֵבִיא לוֹ אוֹצָרוֹת. אֲבָל יוֹתֵר מֵהַכֹּל – רַבִּי הָיָה חָכָם וְקָדוֹשׁ מֵאֵין כָּמוֹהוּ. הוּא זֶה שֶׁאָסַף וְכָתַב אֶת "הַמִּשְׁנָה", כְּדֵי שֶׁהַתּוֹרָה הַקְּדוֹשָׁה שֶׁלָּנוּ לְעוֹלָם לֹא תִּשָּׁכַח!

אֲבָל בְּתוֹךְ לִבּוֹ הַפְּנִימִי, מִתַּחַת לִבְגָדָיו הַנָּאִים שֶׁהֵרִיחוּ מִבְּשָׂמִים יְקָרִים, רַבִּי הָיָה אָדָם עָנָיו מְאֹד. אָדָם שֶׁיּוֹדֵעַ שֶׁהַכֹּל – אֲבָל הַכֹּל – שַׁיָּךְ לַה' יִתְבָּרַךְ.

הַהַגָּעָה לָעֲיָרָה וְהַבְּשׂוֹרָה הַמְרַגֶּשֶׁת

יוֹם אֶחָד, צָעַד רַבִּי בְּדַרְכֵי אֶרֶץ יִשְׂרָאֵל, כְּשֶׁהַשֶּׁמֶשׁ הַחַמָּה מְחַמֶּמֶת אֶת הָהָר. לְצִדּוֹ צָעַד תַּלְמִידוֹ הָאָהוּב, הַגָּאוֹן הֶעָצוּם רַבִּי חִיָּיא. הֵם לֹא שׂוֹחֲחוּ עַל מֶזֶג הָאֲוִיר אוֹ עַל הַנּוֹף, אֶלָּא דִּבְּרוּ דִּבְרֵי תּוֹרָה עֲמֻקִּים.

כַּאֲשֶׁר הִתְקָרְבוּ לְאַחַת הָעֲיָרוֹת, שָׁמְעוּ קוֹלוֹת שֶׁל לִמּוּד מִבֵּית הַמִּדְרָשׁ. אַנְשֵׁי הָעֲיָרָה שֶׁיָּצאוּ לִקְרָאתָם סִפְּרוּ לָהֶם בְּהִתְרַגְּשׁוּת:

"רַבֵּינוּ הַנָּשִׂיא! יֵשׁ בָּעִיר שֶׁלָּנוּ צַדִּיק עָצוּם, 'צוּרְבָּא מֵרַבָּנָן' (תַּלְמִיד חָכָם חָרִיף כְּאֵשׁ), שֶׁלּוֹמֵד תּוֹרָה יוֹמָם וָלַיְלָה בִּקְדֻשָּׁה עֲצוּמָה!".

רַבִּי יְהוּדָה הַנָּשִׂיא חִיֵּךְ מִיָּד. עֵינָיו אוֹרוּ בְּשִׂמְחָה. "אִם יֵשׁ כָּאן תַּלְמִיד חָכָם שֶׁכָּזֶה," אָמַר לְרַבִּי חִיָּיא, "אֲנַחְנוּ חַיָּבִים לָלֶכֶת וּלְהַקְבִּיל אֶת פָּנָיו! נֵלֵךְ לְבַקֵּר אוֹתוֹ בְּבֵיתוֹ!".

אֲבָל אָז, שָׁפְלוּ מְעַט פְּנֵיהֶם שֶׁל אַנְשֵׁי הָעִיר, וְהֵם הוֹסִיפוּ בְּקוֹל שָׁקֵט וְעָצוּב:

"אַךְ יֵשׁ מַשֶּׁהוּ שֶׁכְּבוֹדוֹ חַיָּב לָדַעַת... הַתַּלְמִיד חָכָם הַזֶּה... הוּא עִוֵּר. מְאוֹר עֵינָיו כָּבָה. מֵרֹב עֹנִי וְצַעַר, הוּא חַי בְּבַיִת קָטָן וְדַל. בְּגָדָיו קְרוּעִים, וְהוּא אֵינוֹ יָכוֹל לִרְאוֹת דָּבָר בָּעוֹלָם הַזֶּה".

הַוִּכּוּחַ בַּדֶּרֶךְ – מָהוּ הַכָּבוֹד הָאֲמִתִּי?

רַבִּי חִיָּיא, שֶׁכָּל כָּךְ אָהַב וְכִבֵּד אֶת רַבִּי יְהוּדָה הַנָּשִׂיא, נִבְהַל מְאֹד מִן הַמַּחֲשָׁבָה הַזּוֹ.

הוּא פָּנָה לְרַבּוֹ וְאָמַר לוֹ בְּחֶרְדַת קֹדֶשׁ: "רַבִּי! אָנָּא מִמְּךָ, שֵׁב כָּאן. אַל תֵּלֵךְ! מַדּוּעַ שֶׁתְּזַלְזֵל כָּךְ בִּכְבוֹד הַנְּשִׂיאוּת שֶׁלְּךָ? אַתָּה מֶלֶךְ יִשְׂרָאֵל! לֹא יִתָּכֵן שֶׁתִּכָּנֵס לְבִקְתָּה עֲנִיָּה וַחֲשׁוּכָה, אֵצֶל אָדָם פָּשׁוּט וְעִוֵּר. הַכָּבוֹד שֶׁלְּךָ הוּא הַכָּבוֹד שֶׁל עַם יִשְׂרָאֵל כֻּלּוֹ! אֲנִי – שֶׁאֲנִי תַּלְמִידְךָ – אֵלֵךְ בִּמְקוֹמְךָ לְבַקֵּר אוֹתוֹ. אֲנִי אַעֲבִיר לוֹ אֶת הַבְּרָכוֹת שֶׁלְּךָ".

יְלָדִים, מָה אַתֶּם חוֹשְׁבִים שֶׁרַבִּי אָמַר לְרַבִּי חִיָּיא? הַאִם הוּא יָשַׁב עַל הַכִּסֵּא הַמְרֻפָּד בַּמֶּרְכָּבָה וְהִמְתִּין? מַמָּשׁ לֹא!

רַבִּי יְהוּדָה הַנָּשִׂיא הִבִּיט בְּרַבִּי חִיָּיא וְעֵינָיו בָּרְקוּ. "כָּבוֹד?!" תָּמַהּ רַבִּי, "הַכָּבוֹד הַיְּחִידִי בָּעוֹלָם שַׁיָּךְ לַתּוֹרָה! אִם יֵשׁ כָּאן אָדָם שֶׁמָּלֵא בְּתוֹרָה, הַכָּבוֹד שֶׁלִּי הוּא לְכַבֵּד אוֹתוֹ!".

רַבִּי לֹא הִמְתִּין לִתְשׁוּבָה. כְּמוֹ אָרִי צָעִיר, הוּא אָחַז בְּזְרוֹעוֹ שֶׁל רַבִּי חִיָּיא – "תַּקְפֵיהּ וַאֲזַל בַּהֲדֵיהּ" – מַמָּשׁ תָּפַס אוֹתוֹ בְּחָזְקָה, וְצָעַד יַחַד אִתּוֹ קָדִימָה בְּמַעֲלֵה הָרְחוֹב, הַיְשֵׁר אֶל בֵּיתוֹ שֶׁל הֶחָכָם הָעִוֵּר.

הַבִּקּוּר וְהַבְּרָכָה הֶעֲצוּמָה

הֵם הִגִּיעוּ לַבַּיִת. דֶּלֶת עֵץ פְּשׁוּטָה נִפְתְּחָה עִם חֲרִיקָה קַלָּה. רַבִּי יְהוּדָה הַנָּשִׂיא, שֶׁרָגִיל לְאַרְמוֹנוֹת שַׁיִשׁ, נִכְנַס בְּרֹאשׁ מֻרְכָּן לַחֶדֶר הַקָּטָן.

עַל כִּסֵּא עֵץ פָּשׁוּט יָשַׁב הַחָכָם הָעִוֵּר. הַבְּגָדִים שֶׁלּוֹ הָיוּ בָּלִים מִיֹּשֶׁן, אַךְ מֵהַפָּנִים שֶׁלּוֹ בָּקַע אוֹר נִפְלָא – אוֹר שֶׁל תּוֹרָה!

הַחָכָם לֹא יָכוֹל הָיָה לִרְאוֹת אֶת גְּלִימַת נְשִׂיא-יִשְׂרָאֵל, אַךְ הַנְּשָׁמָה שֶׁלּוֹ הִרְגִּישָׁה אֶת הַקְּדֻשָּׁה שֶׁנִּכְנְסָה לַחֶדֶר. רַבִּי וְרַבִּי חִיָּיא יָשְׁבוּ מוּלוֹ, שׂוֹחֲחוּ עִמּוֹ דִּבְרֵי תּוֹרָה, מִשְׁנָיוֹת עֲמֻקּוֹת וְסוֹדוֹת הֲלָכָה. זֶה הָיָה מַעֲמָד שֶׁל אֵשׁ רוּחָנִית. אֲהָבָה שֶׁל תּוֹרָה.

כַּאֲשֶׁר קָמוּ שְׁנֵי הַגְּדוֹלִים לָצֵאת לְדַרְכָּם, קָם גַּם הַחָכָם הָעִוֵּר עַל רַגְלָיו. הוּא הֵרִים אֶת יָדָיו, עֵינָיו הַסְּגוּרוֹת כֻּוְּנוּ לַשָּׁמַיִם, וְאָז, מִקִּירוֹת לִבּוֹ הַטָּהוֹר, נָשָׂא אֶת קוֹלוֹ בִּבְרָכָה שֶׁהִרְעִידָה אֶת הַחֶדֶר כֻּלּוֹ:

"רַבֵּינוּ הַקָּדוֹשׁ!" קָרָא הֶחָכָם, "אַתֶּם קִבַּלְתֶּם וְכִבַּדְתֶּם פָּנִים שֶׁנִּרְאִים – אַךְ אֵינָם רוֹאִים... תִּזְכּוּ לְהַקְבִּיל אֶת פָּנָיו שֶׁל זֶה שֶׁרוֹאֶה אֶת כֻּלָּם – אַךְ אֵינוֹ נִרְאֶה!" .

הַשֶּׁקֶט בַּחֶדֶר הָיָה עָמֹק וּמְתוּחַ. אֵיזוֹ בְּרָכָה פְּלָאִית וּמַדְהִימָה! הֶחָכָם הָעִוֵּר בֵּרַךְ אֶת רַבִּי בַּבְּרָכָה הַגְּדוֹלָה בְּיוֹתֵר בָּעוֹלָם: שֶׁבִּזְכוּת זֶה שֶׁכִּבְּדוּ אוֹתוֹ, אָדָם שֶׁרַק נִרְאֶה אַךְ לֹא רוֹאֶה, הֵם יִזְכּוּ בָּעוֹלָם הַבָּא לִרְאוֹת וּלְקַבֵּל אֶת פָּנָיו שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא – הַבּוֹרֵא שֶׁרוֹאֶה הַכֹּל וְאַף אֶחָד לֹא יָכוֹל לִרְאוֹתוֹ בָּעוֹלָם הַזֶּה!

סִכּוּם הַמַּסָּע

רַבִּי יְהוּדָה הַנָּשִׂיא יָצָא מִן הַבַּיִת הַקָּטָן כְּשֶׁלִּבּוֹ פּוֹעֵם בְּשִׂמְחָה עֲצוּמָה. הוּא הִרְגִּישׁ שֶׁהַבְּרָכָה הַזּוֹ יְקָרָה לוֹ יוֹתֵר מִכָּל זְהַב הַקֵּיסָרִים הָרוֹמָאִים שֶׁקִּבֵּל מִיָּמָיו.

הוּא הִסְתּוֹבֵב אֶל תַּלְמִידוֹ רַבִּי חִיָּיא, הִנִּיחַ אֶת יָדוֹ בְּחֹם עַל כְּתֵפוֹ, וְאָמַר לוֹ בַּחִיּוּךְ מָלֵא תּוֹכָחָה נְעִימָה:

"רוֹאֶה אַתָּה, רַבִּי חִיָּיא? 'אִיכּוּ הַשְׁתָּא מְנַעְתָּן מֵהַאי בִּרְכְּתָא!' – תָּאֵר לְךָ שֶׁהָיִיתִי שׁוֹמֵעַ בְּקוֹלְךָ, שׁוֹמֵר עַל 'הַכָּבוֹד' שֶׁלִּי וְנִשְׁאָר בַּחוּץ... הָיִיתִי מַפְסִיד אֶת הַבְּרָכָה הַקְּדוֹשָׁה וְהַטְּהוֹרָה הַזּוֹ מִפִּיו שֶׁל הַצַּדִּיק!".

וְכָךְ, הִמְשִׁיכוּ שְׁנֵיהֶם בְּדַרְכָּם, מְתוּקִים וַעֲנָוִים כְּתָמִיד, יוֹדְעִים שֶׁהַכָּבוֹד הַיְּחִידִי הָאֲמִתִּי בָּעוֹלָם – הוּא לְכַבֵּד אֶת לוֹמְדֵי הַתּוֹרָה בְּאֶמֶת.

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “כְּשֶׁמֶּלֶךְ יוֹצֵא לַדֶּרֶךְ, יֵשׁ לְפָנָיו שׁוֹמְרִים, סוּסִים אֲצִילִיִּים הָרוֹקְעִים בְּפַרְסוֹתֵיהֶם עַל הָאֲדָמָה, וּמֶרְכָּבוֹת מְפֹאָרוֹת…” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “יוֹם אֶחָד, צָעַד רַבִּי בְּדַרְכֵי אֶרֶץ יִשְׂרָאֵל, כְּשֶׁהַשֶּׁמֶשׁ הַחַמָּה מְחַמֶּמֶת אֶת הָהָר.”?

ד. מה השתנה בין “רַבִּי יְהוּדָה הַנָּשִׂיא הִבִּיט בְּרַבִּי חִיָּיא וְעֵינָיו בָּרְקוּ.” לבין “רַבִּי וְרַבִּי חִיָּיא יָשְׁבוּ מוּלוֹ, שׂוֹחֲחוּ עִמּוֹ דִּבְרֵי תּוֹרָה, מִשְׁנָיוֹת עֲמֻקּוֹת וְסוֹדוֹת הֲלָכָה.”?
ה. כיצד הגיע הסיפור אל האירוע: “הוּא הִסְתּוֹבֵב אֶל תַּלְמִידוֹ רַבִּי חִיָּיא, הִנִּיחַ אֶת יָדוֹ בְּחֹם עַל כְּתֵפוֹ, וְאָמַר לוֹ בַּחִיּוּךְ מָלֵא תּוֹכָחָה נְעִימָה:”?

סיבה ותוצאה מתוך המעשה

האירוע הנתוןמה גרם לו?מה הייתה תוצאתו?
כַּאֲשֶׁר הִתְקָרְבוּ לְאַחַת הָעֲיָרוֹת, שָׁמְעוּ קוֹלוֹת שֶׁל לִמּוּד מִבֵּית הַמִּדְרָשׁ.
רַבִּי יְהוּדָה הַנָּשִׂיא, שֶׁרָגִיל לְאַרְמוֹנוֹת שַׁיִשׁ, נִכְנַס בְּרֹאשׁ מֻרְכָּן לַחֶדֶר הַקָּטָן.

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “ביקור אצל ת"ח עוור — ענוותנותו” על רבי יהודה הנשיא (רבי)? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “הוּא פָּנָה לְרַבּוֹ וְאָמַר לוֹ בְּחֶרְדַת קֹדֶשׁ: "רַבִּי!”.
ג. אילו רבי יהודה הנשיא (רבי) לא היה פועל כפי שמתואר ב־“רַבִּי וְרַבִּי חִיָּיא יָשְׁבוּ מוּלוֹ, שׂוֹחֲחוּ עִמּוֹ דִּבְרֵי תּוֹרָה, מִשְׁנָיוֹת עֲמֻקּוֹת וְסוֹדוֹת הֲלָכָה.” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “הוּא הִסְתּוֹבֵב אֶל תַּלְמִידוֹ רַבִּי חִיָּיא, הִנִּיחַ אֶת יָדוֹ בְּחֹם עַל כְּתֵפוֹ, וְאָמַר לוֹ בַּחִיּוּךְ מָלֵא תּוֹכָחָה…” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

1. כְּשֶׁמֶּלֶךְ יוֹצֵא לַדֶּרֶךְ, יֵשׁ לְפָנָיו שׁוֹמְרִים, סוּסִים…
