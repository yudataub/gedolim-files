🔹 שָׁלָב 1: בֵּרוּר מַקְדִּים וְנִתּוּחַ (לַמּוֹרֶה/לַמְסַפֵּר)

🔹 שָׁלָב 2: הַסִּפּוּר הַמָּלֵא – "הַשֶּׁלֶג שֶׁהָפַךְ לְאֵשׁ"

פְּתִיחָה: הָעוֹלֶה הֶחָדָשׁ מִבָּבֶל יְלָדִים, תּאֲרוּ לָכֶם לַעֲזֹב אֶת הַבַּיִת שֶׁלָּכֶם, לֶאֱרֹז אֶת מְעַט הַחֲפָצִים, וְלָלֶכֶת בָּרֶגֶל חֳדָשִׁים אֲרֻכִּים אֶל אֶרֶץ אַחֶרֶת, רַק כְּדֵי לִלְמֹד תּוֹרָה. כָּךְ בְּדִיּוּק עָשָׂה הִלֵּל הַזָּקֵן. הוּא הָיָה אָדָם פָּשׁוּט וְעָנִי, שֶׁהֶחְלִיט לַעֲזוֹב אֶת אֶרֶץ בָּבֶל שֶׁבָּהּ נוֹלַד, וְלַעֲלוֹת לְאֶרֶץ יִשְׂרָאֵל, לִירוּשָׁלַיִם. מַדּוּעַ? כִּי בִּירוּשָׁלַיִם לִמְּדּוּ תּוֹרָה שְׁנֵי גְּדוֹלֵי הַדּוֹר – שְׁמַעְיָה וְאַבְטַלְיוֹן, וְהִלֵּל רָצָה לִשְׁמֹעַ אֶת שִׁעוּרֵי הַתּוֹרָה שֶׁלָּהֶם כָּל יוֹם וָיוֹם!

אֲבָל בִּירוּשָׁלַיִם, הַחַיִּים הָיוּ קָשִׁים מְאֹד. הִלֵּל לֹא מָצָא עֲבוֹדָה קַלָּה. בְּכָל בֹּקֶר הוּא יָצָא לַחְטֹב עֵצִים וְלִסְחֹב מַשָּׂאוֹת כְּבֵדִים. הַיָּדַיִם שֶׁלּוֹ הָיוּ שְׂרוּטוֹת וְהַגַּב דָּאַב, אַךְ עַל פָּנָיו הָיָה תָּמִיד חִיּוּךְ מֵאִיר. בְּסוֹפוֹ שֶׁל כָּל יוֹם הוּא קִבֵּל מַטְבֵּעַ אֶחָד קָטָן (מַטְבֵּעַ שֶׁנִּקְרָא 'טְרַפְּעִיק'). אֶת חֲצִי הַמַּטְבֵּעַ נָתַן לְאִשְׁתּוֹ שֶׁתִּקְנֶה מְעַט לֶחֶם לַמִּשְׁפָּחָה. וּמָה עָשָׂה בַּחֵצִי הַשֵּׁנִי? קָנָה סֻכָּרִיּוֹת? קָנָה בְּגָדִים? לֹא. בְּאוֹתָם יָמִים, הָיָה שׁוֹמֵר עֲנָק עוֹמֵד בְּפֶתַח בֵּית הַמִּדְרָשׁ שֶׁל שְׁמַעְיָה וְאַבְטַלְיוֹן, וּמִי שֶׁרָצָה לְהִכָּנֵס הָיָה צָרִיךְ לְשַׁלֵּם כֶּסֶף (כְּדֵי שֶׁיּוּכְלוּ לְשַׁלֵּם עַל הַשּׁוֹמֵרִים וְעַל עֵצִים לְחִמּוּם הַמָּקוֹם). הִלֵּל נָתַן אֶת חֲצִי הַמַּטְבֵּעַ לַשּׁוֹמֵר, וְנִכְנַס בְּשִׂמְחָה עֲצוּמָה לִלְמֹד תּוֹרָה. בִּשְׁבִיל הִלֵּל, הַשִּׁעוּר הַזֶּה הָיָה כָּל הַחַיִּים שֶׁלּוֹ!

יוֹם שִׁשִּׁי, חֹדֶשׁ טֵבֵת הִגִּיעַ חֹדֶשׁ טֵבֵת, שִׂיא הַחֹרֶף. קֹר נוֹרָא נָשַׁב בִּרְחוֹבוֹת יְרוּשָׁלַיִם. רוּחַ חֲזָקָה שָׁרְקָה, וַעֲנָנִים שְׁחֹרִים כִּסּוּ אֶת הַשָּׁמַיִם. הָיָה זֶה יוֹם שִׁשִּׁי, עֶרֶב שַׁבָּת. הִלֵּל חִפֵּשׂ עֲבוֹדָה בַּשּׁוּק, חִפֵּשׂ וְחִפֵּשׂ – אַךְ אִישׁ לֹא בִּקֵּשׁ פּוֹעֲלִים בְּקֹר שֶׁכָּזֶה. הִלֵּל נִשְׁאַר בְּלִי אֲגוֹרָה יְחִידָה. הוּא בָּא לְבֵית הַמִּדְרָשׁ וְעָמַד מוּל הַדֶּלֶת. "תֵּן לִי לְהִכָּנֵס, בְּבַקָּשָׁה," הִתְחַנֵּן הִלֵּל לַשּׁוֹמֵר, "הַיּוֹם לֹא הִרְוַחְתִּי כְּלוּם, אֲבָל אֲנִי חַיָּב לִשְׁמֹעַ אֶת הַשִּׁעוּר שֶׁל רַבּוֹתַי! אֲנִי לֹא מְסֻגָּל לְוַתֵּר אֲפִלּוּ עַל יוֹם אֶחָד!". אַךְ הַשּׁוֹמֵר סֵרֵב: "אֵין כֶּסֶף – אֵין כְּנִיסָה!".

הִלֵּל עָמַד בַּחוּץ רוֹעֵד מִקֹּר, כְּשֶׁלִּבּוֹ כִּמְעַט נִשְׁבָּר מִצַּעַר. בְּתוֹךְ בֵּית הַמִּדְרָשׁ דּוֹלֵק תַּנּוּר מְחַמֵּם, אַךְ זֶה לֹא מָה שֶׁעִנְיֵן אֶת הִלֵּל. הוּא הִתְגַּעְגֵּעַ לְאוֹר אַחֵר לְגַמְרֵי – הָאוֹר שֶׁל הַתּוֹרָה!

הַטִּפּוּס אֶל הָאֲרֻבָּה הִלֵּל נָשָׂא אֶת עֵינָיו וְרָאָה שֶׁבְּגַג בֵּית הַמִּדְרָשׁ יֵשׁ אֲרֻבָּה קְטַנָּה – חַלּוֹן עַל הַגַּג שֶׁמַּכְנִיס אוֹר וַאֲוִיר. הִלֵּל לֹא חָשַׁב פַּעֲמַיִם. אָדָם אַחֵר הָיָה חוֹזֵר הַבַּיְתָה אֶל הַפּוּךְ הֶחָם, אֲבָל בִּשְׁבִיל הִלֵּל הַתּוֹרָה הִיא תּוֹרַת חַיִּים. הוּא תִּפֵּס עַל קִירוֹת הָאֶבֶן הַחֲלָקִים, נֶאֱחַז בַּזִּיזִים, וְעָלָה עַל הַגַּג. הוּא נִשְׁכַּב לְיַד הָאֲרֻבָּה, וְהִצְמִיד אֶת הָאֹזֶן שֶׁלּוֹ לַחֹר הַקָּטָן.

וְאָז... אוֹי, אֵיזֶה תַּעֲנוּג! עָלָה בְּאָזְנָיו הַקּוֹל הַמָּתוֹק וְהָאַדִּיר שֶׁל שְׁמַעְיָה וְאַבְטַלְיוֹן הַדּוֹרְשִׁים בַּתּוֹרָה. הֵם דִּבְּרוּ עַל הֲלָכוֹת, עַל מִצְווֹת, עַל קְדֻשָּׁה. הִלֵּל הִקְשִׁיב בְּרִכּוּז כָּל כָּךְ עָמֹק, עַד שֶׁלִּבּוֹ הִתְמַלֵּא בְּשִׂמְחָה גְּדוֹלָה שֶׁאִי אֶפְשָׁר לְתָאֵר אוֹתָהּ בְּמִלִּים!

בַּבַּיִת פְּנִימָה הָיָה חַם. אֲבָל בַּחוּץ עַל הַגַּג, הֵחֵל לָרֶדֶת שֶׁלֶג. פְּתִיתִים לְבָנִים וְקָרִים נָשְׁרוּ מִן הַשָּׁמַיִם. הֵם נָפְלוּ עַל כְּתֵפָיו שֶׁל הִלֵּל, עַל רֹאשׁוֹ, עַל בְּגָדָיו הַדַּקִּים. הַשֶּׁלֶג נֶעֱרַם לְגֹבַהּ רַב, כִּמְעַט מֶטֶר וָחֵצִי (שָׁלוֹשׁ אַמּוֹת)! הַכְּפוֹר חָדַר יָשָׁר אֶל עֲצָמוֹתָיו. אֲבָל יְלָדִים, שִׁמְעוּ דָּבָר נִפְלָא: מֵרֹב הַשִּׂמְחָה וְהָהִתְרַגְּשׁוּת מִדִּבְרֵי הַתּוֹרָה שֶׁשָּׁמַע – הִלֵּל פָּשׁוּט... לֹא הִרְגִּישׁ אֶת הַשֶּׁלֶג! דִּבְרֵי הַתּוֹרָה חִמְּמוּ לוֹ אֶת הַנְּשָׁמָה כְּמוֹ אֵשׁ. הוּא הִמְשִׁיךְ לְהַקְשִׁיב וּלְהַקְשִׁיב, מְסָרֵב לְוַתֵּר אֲפִלּוּ עַל מִלָּה אַחַת שֶׁל שִׁעוּר! בְּסוֹפוֹ שֶׁל דָּבָר, מֵרֹב עֲיֵפוּת וְקֹר מַקְפִּיא, הִלֵּל אִבֵּד אֶת הַהַכָּרָה וְהִתְעַלֵּף תַּחַת הַשֶּׁלֶג.

שַׂק הַשֶּׁלֶג הַיָּקָר בְּיוֹתֵר עָבַר כָּל הַלַּיְלָה שֶׁל לֵיל שַׁבָּת, וְהִגִּיעַ הַבֹּקֶר. בְּתוֹךְ בֵּית הַמִּדְרָשׁ הִמְשִׁיכוּ שְׁמַעְיָה וְאַבְטַלְיוֹן לְלַמֵּד. פִּתְאוֹם אָמַר שְׁמַעְיָה לְאַבְטַלְיוֹן: "אָחִי אַבְטַלְיוֹן, בְּכָל יוֹם יֵשׁ לָנוּ אוֹר שֶׁנִּכְנָס מֵהַחַלּוֹן שֶׁבַּגַּג, וְהַיּוֹם – הַחֶדֶר חָשׁוּךְ! לָמָּה אָפֵל הַיּוֹם? שֶׁמָּא יֵשׁ עֲנָנִים רַבִּים מִדַּי?". הֵם הֵרִימוּ אֶת רֹאשָׁם וְהִסְתַּכְּלוּ דֶּרֶךְ חֹר הָאֲרֻבָּה, וּמָה רָאוּ? – "דְּמוּת אָדָם בָּאֲרֻבָּה!".

מִיָּד מִהֲרוּ הַתַּלְמִידִים וְטִפְּסוּ אֶל הַגַּג. הֵם חָפְרוּ וְחָפְרוּ בְּתוֹךְ הַשֶּׁלֶג הָעָמֹק וְהַקַּר. וּלְפֶתַע, תַּחַת הַשֶּׁלֶג – מָצְאוּ אֶת חֲבֵרָם, אֶת הִלֵּל! הוּא הָיָה קָפוּא כְּמוֹ בְּלוֹק קֶרַח. אֲבָל הוּא מָסַר אֶת נַפְשׁוֹ עַל הַתּוֹרָה! הֵם הוֹרִידוּ אוֹתוֹ מַהֵר לְמַטָּה, אַף שֶׁהָיְתָה זֶה יוֹם שַׁבָּת. הֵם רָחֲצוּ אוֹתוֹ, עָשׂוּ לוֹ מַסָּאז' בְּשֶׁמֶן, וְהוֹשִׁיבוּ אוֹתוֹ מוּל הָאֵשׁ הַדּוֹלֶקֶת כְּדֵי לְחַמֵּם אֶת הַגּוּף הַקָּפוּא שֶׁלּוֹ.

עָמְדוּ גְּדוֹלֵי יִשְׂרָאֵל, שְׁמַעְיָה וְאַבְטַלְיוֹן, מַבִּיטִים בַּפּוֹעֵל הֶעָנִי הַזֶּה שֶׁכָּל כָּךְ אָהַב לִלְמֹד עַד שֶׁלֹּא חָשׁ בַּשֶּׁלֶג, וְאָמְרוּ: "רָאוּי זֶה שֶׁנְּחַלֵּל עָלָיו אֶת הַשַּׁבָּת!" (כְּדֵי לְהַצִּיל מָוֶת, מֻתָּר לְחַלֵּל שַׁבָּת, אֲבָל עַל צַדִּיק שֶׁמּוֹסֵר נַפְשׁוֹ מִתּוֹךְ כָּזוֹ שִׂמְחָה עַל הַתּוֹרָה – הַזְּכוּת גְּדוֹלָה פִּי כַּמָּה וְכַמָּה!).

וּמִי צָמַח וְנִהְיָה מֵאוֹתוֹ עוֹלֶה חָדָשׁ בַּבְלִי שֶׁכָּמְהָה נַפְשׁוֹ לְשִׁעוּר? הִלֵּל הַזָּקֵן! נְשִׂיא כָּל יִשְׂרָאֵל, שֶׁעָלָיו לוֹמְדִים עַד הַיּוֹם!

ב. הֶעָרוֹת בִּצּוּעַ (לַמְסַפֵּר):

ג. הַמְלָצוֹת שִׁלּוּב בְּמַצֶּגֶת (לְפִי בַּקָּשַׁת הַמִּשְׁתַּמֵּשׁ הַקּוֹדֶמֶת לַמַּצָּגוֹת):

ד. פְּעִילוּת מְסַכֶּמֶת ("הַמַּטְבֵּעַ הַחָצוּי"): כָּל יֶלֶד מְקַבֵּל שְׁנֵי "מַטְבְּעוֹת" מִקַּרְטוֹן / נְיָר הֲדָרִים לָבָן. עַל מַטְבֵּעַ אֶחָד כּוֹתְבִים "דְּבָרִים שֶׁהַגּוּף צָרִיךְ" (לֶחֶם, צַעֲצוּעִים), וְעַל הַמַּטְבֵּעַ הַשֵּׁנִי כּוֹתְבִים "דְּבָרִים שֶׁהַנְּשָׁמָה שְׂמֵחָה בָּהֶם" (לִמּוּד תּוֹרָה, תְּפִלָּה מֵהַלֵּב). הַיְלָדִים מְקַשְּׁטִים אֶת מַטְבֵּעַ הַנְּשָׁמָה בְּצֶבַע שֶׁל מַשֶּׁהוּ שֶׁמֵּאִיר – כְּדֵי לִזְכֹּר שֶׁאֵלּוּ הֵם "מַטְבְּעוֹת הִלֵּל" שֶׁנּוֹתְנִים אֶת הַשִּׂמְחָה הָאֲמִתִּית!

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “🔹 שָׁלָב 2: הַסִּפּוּר הַמָּלֵא – "הַשֶּׁלֶג שֶׁהָפַךְ לְאֵשׁ” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “אֶת חֲצִי הַמַּטְבֵּעַ נָתַן לְאִשְׁתּוֹ שֶׁתִּקְנֶה מְעַט לֶחֶם לַמִּשְׁפָּחָה.”?
ג. כיצד הגיב השלג שהפך לאש למצב המתואר: “הִלֵּל עָמַד בַּחוּץ רוֹעֵד מִקֹּר, כְּשֶׁלִּבּוֹ כִּמְעַט נִשְׁבָּר מִצַּעַר.”?
ד. מה השתנה בין “עָלָה בְּאָזְנָיו הַקּוֹל הַמָּתוֹק וְהָאַדִּיר שֶׁל שְׁמַעְיָה וְאַבְטַלְיוֹן הַדּוֹרְשִׁים בַּתּוֹרָה.” לבין “בְּסוֹפוֹ שֶׁל דָּבָר, מֵרֹב עֲיֵפוּת וְקֹר מַקְפִּיא, הִלֵּל אִבֵּד אֶת הַהַכָּרָה וְהִתְעַלֵּף תַּחַת הַשֶּׁלֶג.”?
ה. כיצד הגיע הסיפור אל האירוע: “אֲבָל אִם כֵּן עָמַד שָׁם מִישֶׁהוּ וַאֲפִלּוּ חִיֵּךְ...”?

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “השלג שהפך לאש” על השלג שהפך לאש? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “אָדָם אַחֵר הָיָה חוֹזֵר הַבַּיְתָה אֶל הַפּוּךְ הֶחָם, אֲבָל בִּשְׁבִיל הִלֵּל הַתּוֹרָה הִיא תּוֹרַת חַיִּים.”.
ג. אילו השלג שהפך לאש לא היה פועל כפי שמתואר ב־“בְּסוֹפוֹ שֶׁל דָּבָר, מֵרֹב עֲיֵפוּת וְקֹר מַקְפִּיא, הִלֵּל אִבֵּד אֶת הַהַכָּרָה וְהִתְעַלֵּף תַּחַת הַשֶּׁלֶג.” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “אֲבָל אִם כֵּן עָמַד שָׁם מִישֶׁהוּ וַאֲפִלּוּ חִיֵּךְ...” מחזק את המסר של הסיפור?
