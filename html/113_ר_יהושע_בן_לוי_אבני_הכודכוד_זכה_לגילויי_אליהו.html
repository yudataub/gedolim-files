אבני הכודכוד — זכה לגילויי אליהו

דף עבודה בעקבות ר' יהושע בן לוי
שם: כיתה:

המעשה — הסיפור המלא

להלן עיבוד חינוכי, רגשי ומרתק על רבי יהושע בן לוי, אליהו הנביא ואבני הכדכוד , המותאם במיוחד לבנים בכיתות ב'-ד' בחינוך הדתי-לאומי. הסיפור בנוי על פי כללי הסטוריטלינג, עשיר בתיאורים חושיים, ומדגיש את ערכי האמונה, הענווה וכוחם של ילדים.

הסיפור מנוקד במלואו כנדרש.

פְּתִיחָה: בַּקָּשָׁתוֹ שֶׁל גְּדוֹל הַדּוֹר

יְלָדִים, עִצְמוּ רֶגַע אֶת הָעֵינַיִם. הַאִם פַּעַם יְשַׁבְתֶּם בְּחֶדֶר חָשׁוּךְ לְגַמְרֵי, וּפִתְאוֹם מִישֶׁהוּ הִדְלִיק פָּנָס חָזָק שֶׁסִּנְוֵר אֶתְכֶם? עַכְשָׁו, דַּמְיְנוּ שֶׁהָאוֹר הַזֶּה לֹא בּוֹקֵעַ מִפָּנָס, אֶלָּא מִתּוֹךְ מִינְרָל, מִתּוֹךְ אֶבֶן פֶּלֶא מֵהָאֲדָמָה! בּוֹאוּ נִשְׁמַע סִפּוּר אֲמִתִּי עַל אוֹר כָּזֶה מַמָּשׁ.

בְּאֶרֶץ יִשְׂרָאֵל, לִפְנֵי שָׁנִים רַבּוֹת מְאֹד, חַי צַדִּיק וְחָכָם עָצוּם. שְׁמוֹ הָיָה רַבִּי יְהוֹשֻׁעַ בֶּן לֵוִי . רַבִּי יְהוֹשֻׁעַ הָיָה לוֹמֵד תּוֹרָה כָּל הַיּוֹם. הָיָה רֵיחַ מָתוֹק וְעַתִּיק שֶׁל גְּוִילֵי סְפָרִים עוֹמֵד בְּבֵית מִדְרָשׁוֹ בָּעִיר הַגְּדוֹלָה לוֹד. קוֹל לִמּוּד הַתּוֹרָה שֶׁלּוֹ הָיָה נָעִים וּמַרְגִּיעַ, וְהַקָּדוֹשׁ בָּרוּךְ הוּא אָהַב אוֹתוֹ אַהֲבָה גְּדוֹלָה מְאֹד.

עַד כְּדֵי כָּךְ הָיָה קָדוֹשׁ רַבִּי יְהוֹשֻׁעַ, שֶׁהוּא זָכָה לְדָבָר שֶׁרַק מְעַטִּים בַּדּוֹרוֹת זָכוּ לוֹ: אֵלִיָּהוּ הַנָּבִיא זָכוּר לַטּוֹב הָיָה נִגְלֶה אֵלָיו בְּחֶזְיוֹנוֹת עִלָּאִיִּים, וְלוֹמֵד עִמּוֹ סוֹדוֹת הַתּוֹרָה!

יוֹם אֶחָד, עָמַד רַבִּי יְהוֹשֻׁעַ בֶּן לֵוִי עִם אֵלִיָּהוּ הַנָּבִיא עַל מְרוֹמֵי הַר הַכַּרְמֶל. הֵם דִּבְּרוּ עַל בֵּית הַמִּקְדָּשׁ שֶׁחָרַב, וְעַל הַגְּאֻלָּה הַשְּׁלֵמָה שֶׁעֲתִידָה לָבוֹא.

רַבִּי יְהוֹשֻׁעַ הִזְכִּיר פָּסוּק מִנְּבִיאוּת יְשַׁעְיָהוּ: "וְשַׂמְתִּי כַּדְכֹד שִׁמְשֹׁתַיִךְ". הַנָּבִיא מַבְטִיחַ שֶׁבֶּעָתִיד, הַקָּדוֹשׁ בָּרוּךְ הוּא יִבְנֶה אֶת חַלּוֹנוֹת יְרוּשָׁלַיִם מֵאֲבָנִים יְקָרוֹת וְנָדִירוֹת שֶׁנִּקְרָאוֹת 'אַבְנֵי כַּדְכֹד'.

"אֵלִיָּהוּ הַנָּבִיא הַקָּדוֹשׁ," בִּקֵּשׁ רַבִּי יְהוֹשֻׁעַ, פָנָיו מְלֵאוֹת צִפִּיָּה, "הַאִם אַתָּה יָכוֹל לְהַרְאוֹת לִי... אֵיךְ בְּדִיּוּק נִרְאוֹת אַבְנֵי הַכַּדְכֹד הָאֵלֶּה? אֵיךְ יֵרָאֶה הָאוֹר שֶׁל יְרוּשָׁלַיִם הַבְּנוּיָה?".

חִיֵּךְ אֵלִיָּהוּ הַנָּבִיא וְאָמַר: "הֵן. אֲנִי אַרְאֶה לְךָ בְּעֶזְרַת הַשֵּׁם. אַךְ זֶה יִקְרֶה... עַל יְדֵי נֵס". וּמִיָּד מִיָּד נֶעֱלָם אֵלִיָּהוּ.

הַסְּעָרָה בַּלֵּב הַיָּם

יְלָדִים, מֵהַר הַכַּרְמֶל הַשָּׁקֵט, אֲנַחְנוּ עוֹבְרִים בְּבַת אַחַת לְלֵב הַיָּם הַגָּדוֹל (הַיָּם הַתִּיכוֹן).

חֹשֶׁךְ אֵימָתִי! סוּפָה אֲדִירָה, שֶׁנִּקְרֵאת 'נַחְשׁוֹל', הִכְּתָה בְּעָצְמָה עַל פְּנֵי הַמַּיִם. הַגַּלִּים הִתְנַשְּׂאוּ לְגֹבַהּ שֶׁל עֲשָׂרוֹת מֶטְרִים, מִתְנַפְּצִים עַל סְפִינַת עֵץ רְעוּעָה. מֵי מֶלַח קָרִים שָׁטְפוּ אֶת הַסִּפּוּן. טְרַאח! טְרַאח! עֵצֵי הַסְּפִינָה חָרְקוּ וְאִיְּמוּ לְהִשָּׁבֵר לִרְסִיסִים. הַסְּפִינָה מַמָּשׁ הִטָּרְפָה בַּיָּם! הַמַּלָּחִים, שֶׁהָיוּ אַנְשֵׁי יָם חֲזָקִים, רָעֲדוּ מִפַּחַד, בּוֹכִים וְצוֹעֲקִים, בְּטוּחִים שֶׁהִגִּיעַ סוֹפָם.

אֲבָל בְּתוֹךְ כָּל הַבֶּהָלָה וְהַקֹּר הַזֶּה, בְּפִנַּת הַסְּפִינָה... יָשַׁב נַעַר אֶחָד קָטָן. הָיָה זֶה נַעַר עִבְרִי . בְּגָדָיו הָיוּ רְטוּבִים מִמֵּי הַיָּם הַמְּלוּחִים, פָנָיו קוֹפְאוֹת מֵהָרוּחַ, אַךְ שְׂפָתָיו מִלְמְלוּ פְּסוּקֵי תְּפִלָּה לְבוֹרֵא עוֹלָם. הוּא לֹא צָעַק, הוּא בָּטַח בַּה'.

לְפֶתַע, בְּתוֹךְ קוֹלוֹת הָרַעַם, בָּקַע אוֹר נָעִים וְרַךְ מִשּׁוּם מָקוֹם. מוּל הַנַּעַר הָרוֹעֵד עָמַד... אֵלִיָּהוּ הַנָּבִיא!

"נַעַר עִבְרִי צַדִּיק," אָמַר אֵלִיָּהוּ הַנָּבִיא בְּקוֹל שֶׁגָּבַר עַל רַעַשׁ הַגַּלִּים. "אִם אַתָּה מַסְכִּים לַעֲשׂוֹת שְׁלִיחוּת חֲשׁוּבָה בִּשְׁבִילִי, אֲנִי מַבְטִיחַ לְמַלֵּט אוֹתְךָ וְאֶת כָּל הָאֳנִיָּה הַזּוֹ לְחוֹף מִבְטַחִים – הַכֹּל בִּזְכוּתְךָ!".

הַנַּעַר הִנְהֵן בְּרֹאשׁוֹ בְּלִי הִסּוּס. "הֵן! אֶעֱשֶׂה כָּל מָה שֶׁתְּצַוֶּה לְמַעַן ה'!".

"שְׁמַע הֵיטֵב," הִנְחָה אוֹתוֹ אֵלִיָּהוּ, "כְּשֶׁתַּגִּיעוּ לַיַּבָּשָׁה, לֵךְ לָעִיר הַגְּדוֹלָה לוֹד. חַפֵּשׂ שָׁם אֶת צַדִּיק הַדּוֹר, רַבִּי יְהוֹשֻׁעַ בֶּן לֵוִי. אֲנִי שׁוֹלֵחַ בְּיָדְךָ אוֹצָר: קַח אֶת אַבְנֵי הַכַּדְכֹד הָאֵלֶּה. אַל תַּרְאֶה לוֹ אוֹתָן סְתָם בָּרְחוֹב, אֶלָּא קַח אוֹתוֹ אֶל מְעָרָה חֲשׁוּכָה, מְעָרָה לְלֹא שׁוּם אוֹר – וְשָׁם, וְרַק שָׁם, תַּרְאֶה לוֹ אוֹתָן".

הַנַּעַר לָקַח אֶת צְרוֹר הָאֲבָנִים הֶחָתוּם שֶׁהוֹשִׁיט לוֹ הַנָּבִיא. בְּאוֹתָהּ שְׁנִיָּה, רָגְעָה הָרוּחַ הַשּׁוֹרֶקֶת, הַגַּלִּים שָׁכְכוּ וְהָפְכוּ לְמַיִם רְגוּעִים, וְהַסְּפִינָה עָשְׂתָה אֶת דַּרְכָּהּ בְּשָׁלוֹם לְאֶרֶץ יִשְׂרָאֵל.

גְּדוֹל הַדּוֹר הוֹלֵךְ אַחֲרֵי נַעַר

כְּשֶׁהִגִּיעַ הַנַּעַר לָעִיר לוֹד, הוּא חִפֵּשׂ אֶת רַבִּי יְהוֹשֻׁעַ בֶּן לֵוִי. הוּא מָצָא אֶת הָרַב הַזָּקֵן, יוֹשֵׁב מְעֻטָּף בְּטַלִּית, רָכוּן עַל סְפָרָיו.

הַנַּעַר הַצָּעִיר נִגַּשׁ לְגְדוֹל הַדּוֹר. לִבּוֹ דָּפַק. "כְּבוֹד הָרַב," הִתְחַנֵּן בְּשֶׁקֶט, "אָבָא אֵלִיָּהוּ (אֵלִיָּהוּ הַנָּבִיא) בָּא אֵלַי... אֲנִי נוֹשֵׂא אִתִּי אֶת אַבְנֵי הַכַּדְכֹד שֶׁבִּקַּשְׁתָּ לִרְאוֹת. אֲנָא, בּוֹא אַחֲרַי!".

יְלָדִים, אָדָם שֶׁגַּאַוְתָן בְּלִבּוֹ, הָיָה אוֹמֵר לַנַּעַר: "מָה?! אֲנִי, הָרַב הַגָּדוֹל, אֵלֵךְ אַחֲרֶיךָ?! תּוֹצִיא מִיָּד אֶת הָאֲבָנִים וְתַרְאֶה לִי כָּאן!". אֲבָל רַבִּי יְהוֹשֻׁעַ בֶּן לֵוִי הָיָה מָלֵא בַּעֲנָוָה עֲצוּמָה. הוּא הֵבִין שֶׁגַּם נַעַר צָעִיר יָכוֹל לִהְיוֹת שָׁלִיחַ קָדוֹשׁ שֶׁל ה'.

בְּלִי לִשְׁאֹל שְׁאֵלוֹת מְיֻתָּרוֹת, קָם הָרַב הַקָּדוֹשׁ מִמְּקוֹמוֹ, וְהָלַךְ אַחֲרֵי הַנַּעַר הַקָּטָן דֶּרֶךְ סִמְטָאוֹת הָעִיר לוֹד, עַד שֶׁהִגִּיעוּ אֶל הָהָר בְּקָצֶה הָעִיר. שָׁם, פָּעֲרָה אֶת פִּיהָ מְעָרָה עֲמֻקָּה וַחֲשׁוּכָה מְאֹד.

אוֹר כַּדְכֹד מֵאִיר אֶת הַחֹשֶׁךְ

רַבִּי יְהוֹשֻׁעַ וְהַנַּעַר צָעֲדוּ פְּנִימָה. אֲוִיר טְחוּב, קַר וְלַח הָיָה בַּמְּעָרָה. חוֹשֶׁךְ מִצְרַיִם. מַמָּשׁ אִי אֶפְשָׁר הָיָה לִרְאוֹת כַּף יָד צְמוּדָה לַפָּנִים.

"כָּאן, כְּבוֹד הָרַב," לָחַשׁ הַנַּעַר. הוּא הֵנִיחַ אֶת יָדוֹ עַל הַצְּרוֹר, נָשַׁם נְשִׁימָה עֲמֻקָּה... וּפָתַח אוֹתוֹ!

וָואאאאאוו!

לֹא, לֹא נִשְׁמַע קוֹל פִּצּוּץ, אֲבָל עֵינָיו שֶׁל רַבִּי יְהוֹשֻׁעַ בֶּן לֵוִי נִפְעֲרוּ לִרְוָחָה! מִתּוֹךְ הָאֲבָנִים פָּרַץ אוֹר שֶׁאִי אֶפְשָׁר לְתָאֵר בְּמִלִּים! זֶה לֹא הָיָה אוֹר רָגִיל שֶׁל מְנוֹרָה, וְלֹא אוֹר קַר שֶׁל פָּנָס. זֶה הָיָה אוֹר צָהֹב-כָּתֹם, חַם, זוֹהֵר מִתּוֹךְ תּוֹכָן שֶׁל הָאֲבָנִים. הָאוֹר בָּקַע מֵהַמְּעָרָה כְּלַפֵּי חוּץ.

הַמִּדְרָשׁ מְסַפֵּר: מִיָּד כְּשֶׁהֶרְאָה הַנַּעַר אֶת הָאֲבָנִים – הִבְהִיקָה כָּל עִיר לוֹד מֵאוֹרָם!! הָאוֹר שֶׁל אֶבֶן אַחַת שֶׁל אֵלִיָּהוּ הַנָּבִיא מִלֵּא אֶת כָּל הָעִיר בְּנֹגַהּ מַדְהִים עַד שֶׁנִּרְאָה שֶׁמִּדָּת הַלַּיְלָה נֶעֶלְמָה!

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “אִם אַתָּה מַסְכִּים לַעֲשׂוֹת שְׁלִיחוּת חֲשׁוּבָה בִּשְׁבִילִי, אֲנִי מַבְטִיחַ לְמַלֵּט אוֹתְךָ וְאֶת כָּל הָאֳנִיָּה הַזּוֹ…”.
מכתב: כתבו לר' יהושע בן לוי בעקבות “הוּא הֵנִיחַ אֶת יָדוֹ עַל הַצְּרוֹר, נָשַׁם נְשִׁימָה עֲמֻקָּה...”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “ME (אני/הסיפור): רבי יהושע בן לוי לא אמר: "אני גדול הדור!”.
דיווח: כתבו ידיעה קצרה שכותרתה “אבני הכודכוד — זכה לגילויי אליהו”.

המחזה

רגע מן הסיפורהדמויותמה יאמרו או יעשו?
הַמַּלָּחִים, שֶׁהָיוּ אַנְשֵׁי יָם חֲזָקִים, רָעֲדוּ מִפַּחַד, בּוֹכִים וְצוֹעֲקִים, בְּטוּחִים…
הַמִּדְרָשׁ מְסַפֵּר: מִיָּד כְּשֶׁהֶרְאָה הַנַּעַר אֶת הָאֲבָנִים – הִבְהִיקָה כָּל עִיר לוֹד…
