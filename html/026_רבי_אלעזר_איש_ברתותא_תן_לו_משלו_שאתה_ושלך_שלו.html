סיפור 26 • רבי אלעזר איש ברתותא • "תן לו משלו שאתה ושלך שלו"

"תן לו משלו שאתה ושלך שלו"

דף עבודה בעקבות רבי אלעזר איש ברתותא
שם: כיתה:

המעשה — הסיפור המלא

להלן עיבוד חינוכי, רגשי ומרתק לסיפורו המופלא של רבי אלעזר איש ברתותא , המותאם במיוחד לבנים בכיתות ב'-ד' (חינוך דתי-לאומי). הסיפור שם דגש על ראיית העולם האמונית של הצדיק – ההבנה שכל הרכוש שלנו שייך למעשה לקב"ה, ועל כן הנתינה היא זכות עצומה.

הסיפור מנוקד במלואו כנדרש וערוך לפי כללי הסטוריטלינג.

פְּתִיחָה: הַגַּבָּאִים הַבּוֹרְחִים

יְלָדִים, עִצְמוּ רֶגַע אֶת הָעֵינַיִם. דַּמְיְנוּ שֶׁאַתֶּם הוֹלְכִים בָּרְחוֹב, וּפִתְאוֹם אַתֶּם רוֹאִים אֶת אוֹסֵף הַצְּדָקָה שֶׁל הַשְּׁכוּנָה. בְּדֶרֶךְ כְּלָל, אֲנָשִׁים שֶׁלֹּא רוֹצִים לָתֵת כֶּסֶף עוֹבְרִים לַמִּדְרָכָה הַשְּׁנִיָּה, נָכוֹן?

אֲבָל בִּימֵי הַתַּנָּאִים הַקְּדוֹשִׁים, קָרָה בַּגָּלִיל מַשֶּׁהוּ הָפוּךְ לְגַמְרֵי!

גַּבָּאֵי הַצְּדָקָה, אֵלּוּ שֶׁאָסְפוּ כֶּסֶף לַעֲנִיִּים, עָמְדוּ בְּפִנַּת הָרְחוֹב. פִּתְאוֹם, אֶחָד מֵהֶם הִצְבִּיעַ לְמֶרְחָק, עֵינָיו נִפְעֲרוּ מִבֶּהָלָה וְהוּא קָרָא בִּלְחִישָׁה דְּחוּפָה: "אוֹי וַאֲבוֹי! הִנֵּה הוּא מַגִּיעַ! רַבִּי אֶלְעָזָר עוֹלֶה בַּמַּעֲלֶה הָרְחוֹב! תִּתְחַבְּאוּ! מַהֵר!".

הַגַּבָּאִים אָסְפוּ אֶת מְעִילֵיהֶם הָאֲרֻכִּים, וְרָצוּ לְהִסְתַּתֵּר מֵאֲחוֹרֵי אֶחָד הַקִּירוֹת.

לָמָּה הֵם בָּרְחוּ? הַאִם רַבִּי אֶלְעָזָר כָּעַס עֲלֵיהֶם? חָלִילָה!

הַגַּבָּאִים בָּרְחוּ מִמֶּנּוּ כִּי רַבִּי אֶלְעָזָר אִישׁ בַּרְתּוֹתָא הָיָה צַדִּיק בְּרָמָה כָּזוֹ, שֶׁבְּכָל פַּעַם שֶׁרָאָה גַּבַּאי צְדָקָה... הוּא הָיָה נוֹתֵן אֶת כָּל מָה שֶׁהָיָה לוֹ בַּכִּיסִים. הַכֹּל! הוּא הָיָה חוֹזֵר הַבַּיְתָה בְּלִי אֲגוֹרָה, וְהַגַּבָּאִים כְּבָר יָדְעוּ שֶׁיֵּשׁ לוֹ מִשְׁפָּחָה לְפַרְנֵס וְהֵם פָּשׁוּט רִחֲמוּ עָלָיו, וְהִשְׁתַּדְּלוּ שֶׁלֹּא יִרְאֶה אוֹתָם.

הַשִּׂמְחָה שֶׁבַּדֶּרֶךְ לַשּׁוּק

בְּאוֹתוֹ בֹּקֶר, רַבִּי אֶלְעָזָר יָצָא מִבֵּיתוֹ שָׂמֵחַ וְטוֹב לֵב. הַשֶּׁמֶשׁ שֶׁל הַגָּלִיל זָרְחָה בְּחֹם, וְצִיּוּץ הַצִּפֳּרִים נִשְׁמַע סְבִיב עֲצֵי הַזַּיִת. בְּיָדוֹ הֶחְזִיק שַׂקִּית עוֹר כְּבֵדָה, עֲמוּסָה בְּמַטְבְּעוֹת שֶׁקִּרְקְשׁוּ וְצִלְצְלוּ – צְלִיל שֶׁל עֲמָל כַּפַּיִם מְבֹרָךְ: קְלִינְג... קְלִינְג...

רַבִּי אֶלְעָזָר חָסַךְ אֶת הַכֶּסֶף הַזֶּה פְּרוּטָה לִפְרוּטָה מֶשֶׁךְ חֳדָשִׁים אֲרֻכִּים. זֶה הָיָה כֶּסֶף לְמַטָּרָה קְדוֹשָׁה וּמְשַׂמַּחַת: בִּתּוֹ הָאֲהוּבָה עָמְדָה לְהִתְחַתֵּן! בַּכֶּסֶף הַזֶּה הוּא תִּכְנֵן לָלֶכֶת לַשּׁוּק וְלִקְנוֹת לָהּ "נְדוּנְיָה" – בְּגָדִים חֲדָשִׁים, כְּלֵי בַּיִת מֵחֶרֶס וּנְחֹשֶׁת, שֶׁמֶן זַיִת מְשֻׁבָּח וְכָל מָה שֶׁכַּלָּה צְרִיכָה כְּדֵי לִבְנוֹת בַּיִת נֶאֱמָן בְּיִשְׂרָאֵל.

כְּשֶׁהִתְקָרֵב לַשּׁוּק, הִכּוּ בְּאַפּוֹ הָרֵיחוֹת הַמְּתוּקִים שֶׁל תַּבְלִינִים קְלוּיִים, וְרֵיחַ שֶׁל לֶחֶם טָרִי שֶׁזֶּה עַתָּה יָצָא מֵהַתַּנּוּר. סוֹחֲרִים צָעֲקוּ בְּקוֹל גָּדוֹל, מַצִּיעִים מֵיטַב בַּדִּים וּפֵרוֹת.

רַבִּי אֶלְעָזָר הִתְקַדֵּם לְעֵבֶר סוֹחֵר הַבַּדִּים, אֲבָל אָז – זָוִית עֵינוֹ קָלְטָה תְּנוּעָה חֲשׁוּדָה.

מֵאֲחוֹרֵי דּוּכַן תְּאֵנִים יָבֵשׁ, הוּא רָאָה שְׁנֵי אֲנָשִׁים מוּכָרִים מְנַסִּים לְהִסְתַּתֵּר. אֵלּוּ הָיוּ הַגַּבָּאִים הַנֶּאֱמָנִים שֶׁל הָעִיר!

הַמִּרְדָּף בְּסִמְטְאוֹת הַשּׁוּק

רַבִּי אֶלְעָזָר לֹא הִסֵּס. כְּשֶׁצַּדִּיק מְרִיחַ מִצְוָה, לִבּוֹ בּוֹעֵר כְּלַפִּיד שֶׁל אֵשׁ.

"הֵי! הַמְתִּינוּ!" קָרָא רַבִּי אֶלְעָזָר, וּמִהֵר לְעֶבְרָם.

הַגַּבָּאִים, שֶׁשָּׁמְעוּ אֶת קוֹלוֹ, הֵחֵלּוּ לִמְלֹט עַל נַפְשָׁם. הֵם רָצוּ לְתוֹךְ סִמְטָא צָרָה, עֲנַן אָבָק עָלָה מִכַּפּוֹת רַגְלֵיהֶם. הֵם הִשְׁתַּדְּלוּ שֶׁלֹּא לִפְגֹּעַ בִּרְכוּשׁוֹ הַמֻּעָט שֶׁל הַצַּדִּיק, אֲבָל רַבִּי אֶלְעָזָר רָץ אַחֲרֵיהֶם! הַשַּׂקִּית עִם הַמַּטְבְּעוֹת טָפְחָה עַל צִדּוֹ.

בְּסוֹפָהּ שֶׁל סִמְטָא, עָצְרוּ הַגַּבָּאִים מִתְנַשְּׁמִים בִּכְבֵדוּת. רַבִּי אֶלְעָזָר עָמַד מוּלָם. מַבָּטוֹ הָיָה קָדוֹשׁ וְחָדוּר אַהֲבָה עֲצוּמָה.

"לָמָּה אַתֶּם בּוֹרְחִים מִמֶּנִּי?" שָׁאַל בְּנַחַת. "אֲנִי מַשְׁבִּיעַ אֶתְכֶם בַּשֵּׁם הַקָּדוֹשׁ – אִמְרוּ לִי: בְּאֵיזוֹ מִצְוָה אַתֶּם עוֹסְקִים הַיּוֹם? לְמִי אַתֶּם אוֹסְפִים מָמוֹן?"

הַגַּבָּאִים יָדְעוּ שֶׁמִּשְּׁבוּעָה כָּזוֹ אִי אֶפְשָׁר לִבְרֹחַ. הֵם הִשְׁפִּילוּ אֶת מַבָּטָם הָרַחֲמָנִי, וְהַגַּבַּאי הָרָאשִׁי לָחַשׁ:

"רַבֵּנוּ... יֵשׁ נַעַר וְנַעֲרָה בָּעִיר הַסְּמוּכָה. שְׁנֵיהֶם יְתוֹמִים לְלֹא אַבָּא וְלְלֹא אִמָּא. אֵין מִי שֶׁיִּדְאַג לָהֶם. הֵם עוֹמְדִים לְהִתְחַתֵּן, וְאֵין לָהֶם כְּלוּם. לֹא בֶּגֶד, לֹא לֶחֶם לִסְעוּדַת הַחֲתֻנָּה, וְאֵין לָהֶם קוֹרַת גַּג... בַּעֲבוּרָם אֲנַחְנוּ טוֹרְחִים מִפֶּתַח לְפֶתַח."

הַהַחְלָטָה בְּהֵיכַל הַלֵּב

יְלָדִים... עִצְרוּ וְתַחְשְׁבוּ. מָה אָדָם רָגִיל הָיָה עוֹשֶׂה? הֲרֵי הוּא צָרִיךְ לְחַתֵּן אֶת הַבַּת שֶׁלּוֹ! בְּוַדַּאי הוּא הָיָה נוֹתֵן מַטְבֵּעַ אֶחָד אוֹ שְׁנַיִם, אוֹמֵר 'תִּזְכּוּ לְמִצְווֹת' וְהוֹלֵךְ לַשּׁוּק כְּדֵי לִקְנוֹת שִׂמְלָה יָפָה לְבִתּוֹ שֶׁלּוֹ.

אֲבָל אֵצֶל רַבִּי אֶלְעָזָר, הַכֶּסֶף בִּכְלָל לֹא הָיָה שֶׁלּוֹ. בְּרֹאשׁוֹ הָיְתָה הֲבָנָה פְּשׁוּטָה וַאֲמִתִּית: "הַקָּדוֹשׁ בָּרוּךְ הוּא נָתַן לִי אֶת הַכֶּסֶף הַזֶּה פִּקָּדוֹן. אִם מֶלֶךְ מַלְכֵי הַמְּלָכִים מְבַקֵּשׁ עַכְשָׁו אֶת הַכֶּסֶף כְּדֵי שֶׁהַיְּתוֹמִים שֶׁלּוֹ יִתְחַתְּנוּ – אֵיךְ אוּכַל לַעֲצֹר אוֹתוֹ אֶצְלִי?!"

לְפֶתַע, אוֹר גָּדוֹל קָרַן מִפָּנָיו שֶׁל הַתַּנָּא. קוֹלוֹ רָעַד מִתּוֹךְ אַהֲבַת ה' וּמְסִירוּת נֶפֶשׁ, וְהוּא קָרָא:

"הָעֲבוֹדָה!" (שְׁבוּעָה כְּלַפֵּי תְּמִידִים וְעוֹלוֹת שֶׁל בֵּית הַמִּקְדָּשׁ), "שֶׁהֵם קוֹדְמִים לְבִתִּי!" .

"הַיְּתוֹמִים הַלָּלוּ, שֶׁאֵין לָהֶם אָב שֶׁיִּדְאַג לָהֶם – הֵם קוֹדְמִים לְבִתִּי, שֶׁלָּהּ עֲדַיִן יֵשׁ אָב הַדּוֹאֵג לָהּ!".

בְּתְנוּעָה מְלֵאַת שִׂמְחָה עֲצוּמָה, לְלֹא טִפַּת צַעַר אוֹ עַצְבוּת, הֵרִים רַבִּי אֶלְעָזָר אֶת שַׂקִּית הַכֶּסֶף הַכְּבֵדָה – תּוֹצָר שֶׁל חֳדָשִׁים שֶׁל זֵעָה וַעֲבוֹדָה קָשָׁה – וְהִנִּיחַ אוֹתָהּ לַחֲלוּטִין בְּתוֹךְ יְדֵיהֶם שֶׁל הַגַּבָּאִים הַהֲמוּמִים.

הוּא לֹא הִשְׁאִיר לְעַצְמוֹ אֲפִלּוּ אֲגוֹרָה אַחַת. הוּא נָתַן אֶת הַכֹּל! מֵרֹב הִתְרַגְּשׁוּת וּשְׂמֹחַ עַל הַמִּצְוָה, הוּא חִיֵּךְ חִיּוּךְ רָחָב כְּשֶׁפָּנָיו מְאִירוֹת בְּקְדֻשָּׁה, סוֹבֵב אֶת גַּבּוֹ מֵהַשּׁוּק וְחָזַר בְּדַרְכּוֹ לְבֵית הַמִּדְרָשׁ לְהַמְשִׁיךְ לִלְמֹד אֶת תּוֹרַת ה', הַיְּקָרָה לוֹ מִכָּל זָהָב וָכֶסֶף.

אֵלּוּ מִלִּים הוּא הָיָה אוֹמֵר תָּמִיד? אֶת סוֹד חַיָּיו:

"תֵּן לוֹ – מִשֶּׁלּוֹ. שֶׁאַתָּה, וְשֶׁלְּךָ – שֶׁלּוֹ הֵם!".

הַקָּדוֹשׁ בָּרוּךְ הוּא נָתַן לָנוּ אֶת הַכֹּל. וּכְשֶׁאֲנַחְנוּ נוֹתְנִים – אֲנַחְנוּ פָּשׁוּט מַעֲבִירִים אֶת הָאוֹצָר שֶׁל הַמֶּלֶךְ לַיְּלָדִים הָאֲהוּבִים שֶׁלּוֹ.

א. שְׁאֵלַת פְּתִיחָה (לְעוֹרֵר סַקְרָנוּת):

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “להלן עיבוד חינוכי, רגשי ומרתק לסיפורו המופלא של רבי אלעזר איש ברתותא , המותאם במיוחד לבנים בכיתות ב'-ד' (חינוך דתי-לאומי).” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “הַגַּבָּאִים בָּרְחוּ מִמֶּנּוּ כִּי רַבִּי אֶלְעָזָר אִישׁ בַּרְתּוֹתָא הָיָה צַדִּיק בְּרָמָה כָּזוֹ, שֶׁבְּכָל פַּעַם שֶׁרָאָה גַּבַּאי צְדָקָה...”?
ג. כיצד הגיב רבי אלעזר איש ברתותא למצב המתואר: “רַבִּי אֶלְעָזָר חָסַךְ אֶת הַכֶּסֶף הַזֶּה פְּרוּטָה לִפְרוּטָה מֶשֶׁךְ חֳדָשִׁים אֲרֻכִּים.”?
ד. מה השתנה בין “כְּשֶׁצַּדִּיק מְרִיחַ מִצְוָה, לִבּוֹ בּוֹעֵר כְּלַפִּיד שֶׁל אֵשׁ.” לבין “בְּוַדַּאי הוּא הָיָה נוֹתֵן מַטְבֵּעַ אֶחָד אוֹ שְׁנַיִם, אוֹמֵר 'תִּזְכּוּ לְמִצְווֹת' וְהוֹלֵךְ לַשּׁוּק כְּדֵי…”?
ה. כיצד הגיע הסיפור אל האירוע: “ילדים, תארו לכם שחסכתם המון זמן – יום אחרי יום אספתם מטבעות כדי לקנות את המשחק שאתם הכי רוצים.”?

3. מה היה ראשון?

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “"תן לו משלו שאתה ושלך שלו"” על רבי אלעזר איש ברתותא? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “כְּשֶׁהִתְקָרֵב לַשּׁוּק, הִכּוּ בְּאַפּוֹ הָרֵיחוֹת הַמְּתוּקִים שֶׁל תַּבְלִינִים קְלוּיִים, וְרֵיחַ שֶׁל לֶחֶם טָרִי שֶׁזֶּה עַתָּה יָצָא מֵהַתַּנּוּר.”.
ג. אילו רבי אלעזר איש ברתותא לא היה פועל כפי שמתואר ב־“בְּוַדַּאי הוּא הָיָה נוֹתֵן מַטְבֵּעַ אֶחָד אוֹ שְׁנַיִם, אוֹמֵר 'תִּזְכּוּ לְמִצְווֹת' וְהוֹלֵךְ לַשּׁוּק כְּדֵי לִקְנוֹת…” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “ילדים, תארו לכם שחסכתם המון זמן – יום אחרי יום אספתם מטבעות כדי לקנות את המשחק שאתם הכי רוצים.” מחזק את המסר של הסיפור?
