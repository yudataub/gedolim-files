דף עבודה בעקבות רבא
שם: כיתה:

המעשה — הסיפור המלא

פְּתִיחָה: קוֹל הַתּוֹרָה בְּעִיר מְחוֹזָא

יְלָדִים, אֲנַחְנוּ יוֹצְאִים עַכְשָׁו לְמַסָּע מְרַתֵּק. אֲנַחְנוּ עוֹזְבִים אֶת אֶרֶץ יִשְׂרָאֵל וְנוֹסְעִים הַרְחֵק הַרְחֵק, לְאֶרֶץ בָּבֶל, לִפְנֵי כְּאַלְפַּיִם שָׁנָה.

דַּמְיְנוּ אֶת הָעִיר הַגְּדוֹלָה 'מְחוֹזָא'. הַשֶּׁמֶשׁ בַּחוּץ חַמָּה מְאוֹד, קוֹפַחַת עַל גַּגּוֹת הַבָּתִּים וְעַל עֲצֵי הַתְּמָרִים. אֲבָל כְּשֶׁמִּתְקָרְבִים אֶל בֵּית הַמִּדְרָשׁ הַגָּדוֹל, כְּבָר אֶפְשָׁר לִשְׁמֹעַ אֶת הַקּוֹל הַמָּתוֹק וְהָאַדִּיר שֶׁל אֲלָפִים אֲלָפִים שֶׁל תַּלְמִידֵי חֲכָמִים! הַקּוֹל הַזֶּה הוֹמֶה כְּמוֹ גַּלִּים שֶׁל יָם. "אָמַר אַבַּיֵּי... אָמַר רָבָא...". הָאֲוִיר בְּתוֹךְ בֵּית הַמִּדְרָשׁ מָלֵא בְּרֵיחַ שֶׁל גְּוִילִים עַתִּיקִים וְסִפְרֵי קֹדֶשׁ.

בְּרֹאשׁ הַיְשִׁיבָה, עַל כִּסֵּא הַנְּשִׂיאוּת, יוֹשֵׁב רָבָא – גְּדוֹל הַדּוֹר! רָבָא לֹא רַק לִמֵּד תּוֹרָה כָּל הַיּוֹם וְהַלַּיְלָה, אֶלָּא הוּא הָיָה מַנְהִיגָם שֶׁל יִשְׂרָאֵל. הוּא הָיָה הָאַחְרַאי הָרָאשִׁי עַל "קֻפַּת הַצְּדָקָה". רָבָא הָיָה אָדָם יָשָׁר מְאוֹד. כַּאֲשֶׁר מְדֻבָּר הָיָה בְּכֶסֶף שֶׁל צְדָקָה, בְּכֶסֶף שֶׁהָאֲנָשִׁים הַפְּשׁוּטִים תָּרְמוּ מִמַּה שֶּׁחָסְכוּ – רָבָא שָׁמַר עָלָיו כְּמוֹ עַל אוֹצָר, כְּדֵי שֶׁיַּגִּיעַ בְּדִיּוּק לְמִי שֶׁצָּרִיךְ, בְּלִי לְבַזְבֵּז אֲפִלּוּ אֲגוֹרָה אַחַת מִיֻּתֶּרֶת.

הָאוֹרֵחַ הַמִּסְתּוֹרִי

בְּאֶחָד מִימֵי הַקַּיִץ הַלּוֹהֲטִים, נִפְתְּחָה לְאַט דֶּלֶת הָעֵץ הַכְּבֵדָה שֶׁל בֵּית הַמִּדְרָשׁ. עַל מִפְתַּן הַדֶּלֶת עָמַד אָדָם. הַנַּעֲלַיִם שֶׁלּוֹ הָיוּ מְכֻסּוֹת בְּאָבָק דַּק, וְהַבְּגָדִים הַקְּרוּעִים שֶׁלּוֹ הֶרְאוּ עַל עֲנִיּוּת וְחֹסֶר כֹּל. אֲבָל כְּשֶׁהִסְתַּכַּלְתָּ לוֹ בָּעֵינַיִם – רָאִיתָ מַשֶּׁהוּ מְיֻחָד. עֲמִידָתוֹ הָיְתָה זְקוּפָה וּמְכֻבֶּדֶת. הוּא לֹא הָיָה עָנִי רָגִיל; הוּא הָיָה "עָנִי בֶּן טוֹבִים" – אָדָם שֶׁבֶּעָבָר הָיָה עָשִׁיר גָּדוֹל מְאוֹד, וּבִגְלַל כָּל מִינֵי צָרוֹת, אִבֵּד אֶת כָּל רְכוּשׁוֹ בְּיּוֹם אֶחָד, וְנִשְׁאַר מַמָּשׁ בְּאֶפֶס יָדַיִם.

הָאִישׁ הִתְקָרֵב בַּעֲנָוָה אֶל שֻׁלְחָנוֹ שֶׁל רָבָא.

"שָׁלוֹם עָלֶיךָ, כְּבוֹד תּוֹרָתוֹ," אָמַר הָאִישׁ בְּקוֹל שָׁקֵט וְנָעִים, "רָעֵב אָנֹכִי. הַאִם תּוּכַל הַקְּהִלָּה לַעֲזֹר לִי וְלָתֵת לִי אֲרוּחָה?"

רָבָא הִבִּיט בּוֹ בְּעֵינַיִם טוֹבוֹת וְרַחְמָנִיּוֹת. "בְּוַדַּאי, אָחִי יָקָר. הַקֻּפָּה מְיֹעֶדֶת בְּדִיּוּק לַאֲנָשִׁים כָּמוֹךָ. אֱמֹר לִי, בְּנִי, בַּמֶּה אַתָּה רָגִיל לִסְעֹד אֶת לִבְּךָ? מָה לְהָכִין עֲבוּרְךָ?"

רָבָא שָׁאַל זֹאת כִּי הַהֲלָכָה הִיא שֶׁלְּאָדָם שֶׁיָּרַד מִנְּכָסָיו מִשְׁתַּדְּלִים לָתֵת דְּבָרִים שֶׁהוּא הָיָה רָגִיל בָּהֶם, כְּדֵי שֶׁלֹּא יִתְבַּיֵּשׁ וְלֹא יִתְעַצֵּב עַל הַמַּהְפָּךְ שֶׁקָּרָה לוֹ בַּחַיִּים.

הָעָנִי עָנָה לְלֹא הִסּוּס: "רָגִיל אֲנִי מֵאָז וּמִתָּמִיד לֶאֱכֹל כָּל יוֹם תַּרְנְגֹלֶת פְּטוּמָה הַצְּלוּיָה הֵיטֵב בְּתַנּוּר וּמְתֻבֶּלֶת בִּתְבָלִינִים, וְלִשְׁתּוֹת כּוֹס קְרִירָה שֶׁל יַיִן יָשָׁן מְשֻׁבָּח."

הַקּוֹנְפְלִיקְט וְהָאֱמוּנָה

יְלָדִים, אֲנַחְנוּ צְרִיכִים לְהָבִין: בְּאוֹתָם יָמִים, תַּרְנְגֹלֶת פְּטוּמָה (שְׁמֵנָה וִיקָרָה) וְיַיִן יָשָׁן הָיוּ מַאֲכָלִים נְדִירִים מְאוֹד! זֶה בְּעֵרֶךְ כְּמוֹ שֶׁעָנִי יְבַקֵּשׁ הַיּוֹם סְטֵיק יָקָר מְאוֹד וּמִיץ טִבְעִי נָדִיר מֵחוּץ לָאָרֶץ!

רָבָא קִמֵּט אֶת מִצְחוֹ הַקָּדוֹשׁ. הוּא אָמְנָם צָרִיךְ לַעֲזֹר לֶעָנִי, אֲבָל מִצַּד שֵׁנִי, קֻפַּת הַצְּדָקָה שַׁיֶּכֶת לַצִּבּוּר! כֶּסֶף שֶׁל אֲנָשִׁים שֶׁעוֹבְדִים קָשֶׁה!

רָבָא פָּנָה אֶל הָעָנִי בְּעֶדְנָה אַךְ בְּהִגָּיוֹן: "אָחִי הָאֲהוּב, תַּרְנְגֹלֶת פְּטוּמָה וְיַיִן יָשָׁן? כְּלוּם אֵינְךָ חוֹשֵׁשׁ שֶׁתַּטְרִיחַ מִדַּי עַל הַקְּהִלָּה? מֵאַיִן נִקַּח כֶּסֶף כָּל יוֹם לְמַאֲכָלִים כָּל כָּךְ יְקָרִים? אוּלַי תַּסְכִּים לְהִסְתַּפֵּק בְּלֶחֶם פָּשׁוּט וְתַבְשִׁיל יְרָקוֹת חַם שֶׁלְּכָל עֲנִיֵּי הָעִיר?"

אֲבָל הָעָנִי לֹא הִשְׁפִּיל אֶת מַבָּטוֹ, וְתָגוּבָתוֹ הָיְתָה עוֹצְמָתִית וּמְרַגֶּשֶׁת עַד דְּמָעוֹת. קוֹלוֹ צִלְצֵל בְּבֵית הַמִּדְרָשׁ מָלֵא בְּאֱמוּנָה טְהוֹרָה, אֱמוּנָה כָּזוֹ שֶׁאֶפְשָׁר לְהַרְגִּישׁ בַּלֵּב.

"כְּבוֹד תּוֹרָתוֹ, רַבִּי יַקִּירִי," אָמַר הָעָנִי, " וְכִי מִשֶּׁלָּהֶם אֲנִי אוֹכֵל?! וְכִי קֻפַּת הַצְּדָקָה שֶׁל הַקְּהִלָּה הִיא שֶׁזָּנָה אוֹתִי? לֹא! מִשֶּׁל הַקָּדוֹשׁ בָּרוּךְ הוּא אֲנִי אוֹכֵל! בּוֹרֵא עוֹלָם זָן אֶת הָעוֹלָם כֻּלּוֹ! הֲלֹא כָּתוּב בְּתְהִלִּים (קמה, טו): 'עֵינֵי כֹל אֵלֶיךָ יְשַׂבֵּרוּ, וְאַתָּה נוֹתֵן לָהֶם אֶת אָכְלָם בְּעִתּוֹ ' – לֹא כָּתוּב 'בְּעִתָּם' (בַּזְּמַן שֶׁל כֻּלָּם) אֶלָּא 'בְּעִתּוֹ' (בַּזְּמַן וּבַצֹּרֶךְ שֶׁל כָּל אֶחָד). הַקָּדוֹשׁ בָּרוּךְ הוּא, שֶׁהָאוֹצָר שֶׁלּוֹ אֵינוֹ נִגְמָר לְעוֹלָם, הוּא זֶה שֶׁדּוֹאֵג שֶׁלִּי יִהְיֶה בְּדִיּוּק אֶת מָה שֶׁאֲנִי רָגִיל מִתָּמִיד!"

הַהַפְתָּעָה מֵאֶרֶץ מֶרְחַקִּים

דִּבְרֵי הָעָנִי עֲדַיִן הִדְהֲדוּ בַּחֲלַל הַבַּיִת, כְּשֶׁלְּפֶתַע פִּתְאוֹם... טוּק... טוּק... טוּק.

דְּפִיקָה חֲזָקָה נִשְׁמְעָה מֵהַדֶּלֶת הַפְּנִימִית שֶׁל בֵּית הַמִּדְרָשׁ.

רָבָא פָּנָה לִבְדֹּק מִי שָׁם, וְהִנֵּה – לְתַדְהֲמָתוֹ הַעֲצוּמָה – הָיְתָה זוֹ אֲחוֹתוֹ !

הַאֲחוֹת שֶׁלּוֹ שֶׁמִּתְגּוֹרֶרֶת בְּעִיר כָּל כָּךְ רְחוֹקָה, עַד שֶׁהֵם לֹא הִתְרָאוּ וְלֹא שָׁמְעוּ זֶה מִזּוֹ בְּמֶשֶׁךְ שְׁלֹשׁ עֶשְׂרֵה שָׁנִים רְצוּפוֹת! שְׁלֹשׁ עֶשְׂרֵה שָׁנָה שֶׁלֹּא נִפְגְּשׁוּ, וְהִנֵּה בְּדִיּוּק הַיּוֹם, בְּדִיּוּק בַּשָּׁעָה הַזּוֹ, בְּדִיּוּק בַּדַּקָּה הַזּוֹ – הִיא הִגִּיעָה לְבַקֵּר אֶת אָחִיהָ הַגָּדוֹל הָרַב!

"אֲחוֹתִי יַקִּירָתִי!" שָׂמַח רָבָא וְדִמְעָה בְּעֵינָיו, "בָּרוּךְ בּוֹאֵךְ! שָׁנִים שֶׁלֹּא רָאִיתִי אֶת פָּנַיִךְ!".

אֲחוֹתוֹ חִיְּכָה וְאָמְרָה: "אָחִי הָאָהוּב, הִגַּעְתִּי מִמֶּרְחַקִּים לִרְאוֹתְךָ, וּכְדֵי שֶׁתִּשְׂמַח בְּבִיאָתִי, הֵבֵאתִי אִתִּי לִכְבוֹדְךָ מַתָּנָה מְיֻחֶדֶת מִבֵּיתִי לַאֲרוּחַת הַצָּהֳרַיִם. צִידָה אֵיכוּתִית בִּמְיֻחָד!"

הִיא הוֹרִידָה אֶת הַמִּכְסֶה מֵעַל הַסַּלְּסִלָּה הַגְּדוֹלָה שֶׁנָּשְׂאָה.

מִיָּד אֶפְשָׁר הָיָה לְהָרִיחַ מֵרָחוֹק רֵיחַ עָשִׁיר, עָסִיסִי וּמְגָרֶה הִתְפַּשֵּׁט בְּכָל רַחֲבֵי הַחֶדֶר. מָה יָשַׁב בְּתוֹךְ הַסַּלְּסִלָּה?

נְכוֹן מְאוֹד, יְלָדִים!

תַּרְנְגֹלֶת פְּטוּמָה לוֹהֶטֶת וְצְלוּיָה הֵיטֵב, וְלְצִדָּהּ קַנְקַן חֶרֶס צְנַנְקָר מָלֵא בְּיַיִן יָשָׁן וּמְשֻׁבָּח!

הַהֲבָנָה וְהַנִּצָּחוֹן שֶׁל הָאֱמוּנָה

רָבָא הִבִּיט בַּסַּלְּסִלָּה. אַחַר כָּךְ הִבִּיט בָּעָנִי שֶׁהָיָה בֶּעָבָר עָשִׁיר. שׁוּב בַּסַּלְּסִלָּה וְשׁוּב בָּעָנִי.

13 שָׁנָה אֲחוֹתוֹ לֹא הִגִּיעָה! מִכָּל הַיָּמִים בַּשָּׁנָה, הִיא הִגִּיעָה הַיּוֹם.

מכתב: כתבו לרבא בעקבות “הַאֲחוֹת שֶׁלּוֹ שֶׁמִּתְגּוֹרֶרֶת בְּעִיר כָּל כָּךְ רְחוֹקָה, עַד שֶׁהֵם לֹא הִתְרָאוּ וְלֹא שָׁמְעוּ זֶה מִזּוֹ בְּמֶשֶׁךְ…”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “תַּרְנְגֹלֶת פְּטוּמָה לוֹהֶטֶת וְצְלוּיָה הֵיטֵב, וְלְצִדָּהּ קַנְקַן חֶרֶס צְנַנְקָר מָלֵא בְּיַיִן יָשָׁן וּמְשֻׁבָּח!”.
דיווח: כתבו ידיעה קצרה שכותרתה “רבא”.
