צולע על ירכו בגלל מאמציו לגויים

דף עבודה בעקבות ר' יהושע בן לוי
שם: כיתה:

המעשה — הסיפור המלא

פְּתִיחָה: בְּשׂוֹרָה מָרָה בְּבֵית הַמִּדְרָשׁ

יְלָדִים, אֲנַחְנוּ נִמְצָאִים בְּאֶרֶץ יִשְׂרָאֵל, לִפְנֵי כִּמְעַט אַלְפַּיִם שָׁנָה. בְּבֵית הַמִּדְרָשׁ שֶׁל הָעִיר לוֹד שׂוֹרֶרֶת דְּמָמָה קְדוֹשָׁה. הָאֲוִיר קַר וְנָעִים, וְרֵיחַ שֶׁל גְּוִילֵי סְפָרִים עַתִּיקִים מְמַלֵּא אֶת הַחֶדֶר. עַל כִּסֵּא עֵץ פָּשׁוּט יוֹשֵׁב גְּדוֹל הַדּוֹר, רַבִּי יְהוֹשֻׁעַ בֶּן לֵוִי. עֵינָיו עֲצוּמוֹת מֶחֱצָה, וְהוּא שָׁקוּעַ בְּלִמּוּד מַעֲמִיק.

לְפֶתַע, הַשֶּׁקֶט נִקְרָע! צְעָדִים מְהִירִים וּבְהוּלִים נִשְׁמָעִים מִבַּחוּץ. דֶּלֶת הָעֵץ הַכְּבֵדָה נִפְתַּחַת בְּחָזְקָה וְחוֹרֶקֶת עַל צִירֶיהָ. בַּפֶּתַח עוֹמֵד יְהוּדִי מֵאַחַת הֶעָרִים הַסְּמוּכוֹת, מַזִּיעַ, מִתְנַשֵּׁף, וּבְגָדָיו קְרוּעִים מֵרֹב רִיצָה.

"רַבִּי!" הוּא צוֹעֵק, וְקוֹלוֹ רוֹעֵד מִפַּחַד, "אָסוֹן! הַשַּׂר הָרוֹמָאִי הָאַכְזָר הִטִּיל עָלֵינוּ מַס גָּדוֹל שֶׁאִי אֶפְשָׁר לְשַׁלֵּם. אִם לֹא נָבִיא אֶת הַכֶּסֶף עַד מָחָר – יַחֲרִיבוּ אֶת הָעִיר וְיִקְחוּ אֶת מִשְׁפְּחוֹתֵינוּ לַכֶּלֶא! רַק אַתָּה, רַבִּי יְהוֹשֻׁעַ, יָכוֹל לַעֲזֹר לָנוּ. הַשַּׂר מְכַבֵּד אֲנָשִׁים חֲכָמִים. אָנָּא, בּוֹא אִתִּי לַשַּׂר לְבַטֵּל אֶת הַגְּזֵרָה!"

הַהַחְלָטָה: הַכְּאֵב מוּל הָאַחֲרָיוּת

הַתַּלְמִידִים בְּבֵית הַמִּדְרָשׁ הִשְׁתַּתְּקוּ. הֵם הִבִּיטוּ עַל רַבָּם בִּדְאָגָה.

רַבִּי יְהוֹשֻׁעַ בֶּן לֵוִי לֹא הָיָה אָדָם צָעִיר וְחָזָק. הוּא הָיָה אִישׁ קָדוֹשׁ, אַךְ מְבֻגָּר מְאֹד. יוֹתֵר מִכָּךְ – הָיְתָה לוֹ בְּעָיָה קָשָׁה מְאֹד בְּרַגְלוֹ. הוּא צָלַע עַל יְרֵכוֹ, וְכָל צַעַד שֶׁעָשָׂה הָיָה מְלֻוֶּה בִּכְאֵב דּוֹקֵר, כְּמוֹ מַחַט עָמֹק בְּתוֹךְ הָעֶצֶם.

לִנְסֹעַ עַכְשָׁו בַּדְּרָכִים? לָלֶכֶת שָׁעוֹת אֲרֻכּוֹת עַד לְאַרְמוֹן הַנְּצִיב הָרוֹמָאִי? זֶה הָיָה כִּמְעַט לְמַעְלָה מִכֹּחוֹתָיו.

"רַבִּי," לָחַשׁ אֶחָד הַתַּלְמִידִים, "רַגְלְךָ הֲרֵי כּוֹאֶבֶת כָּל כָּךְ. מָזֶג הָאֲוִיר בַּחוּץ חַם וְקוֹפֵחַ. אֵיךְ תּוּכַל לָלֶכֶת? נִשְׁלַח מִישֶׁהוּ אַחֵר... עַכְשָׁו תִּשָּׁאֵר לָנוּחַ בְּבֵית הַמִּדְרָשׁ".

אֲבָל רַבִּי יְהוֹשֻׁעַ נִשְׁעַן עַל מַקְּלוֹ הָעָבֶה, וְקָם לְאַט לְאַט. עֵינָיו זָהֲרוּ.

"יְהוּדִים נִמְצָאִים בְּסַכָּנָה," אָמַר בְּקוֹל שָׁקֵט אַךְ תַּקִּיף, "וַאֲנִי אֶשְׁכַּב לִישֹׁן? הַכְּאֵב שֶׁלִּי לֹא חָשׁוּב עַכְשָׁו. כְּשֶׁהַגּוּף כּוֹאֵב – אַהֲבַת יִשְׂרָאֵל חַיֶּבֶת לְנַצֵּחַ!".

הַמַּסָּע הַקָּשֶׁה

רַבִּי יְהוֹשֻׁעַ יָצָא לַדֶּרֶךְ. הַשֶּׁמֶשׁ קָפְחָה עַל הָרֹאשׁ. רְגָבִים שֶׁל יֹבֶשׁ וְאֲבָק דָּבְקוּ לְבִגְדֵי הַקֹּדֶשׁ שֶׁלּוֹ. בְּכָל פַּעַם שֶׁרַגְלוֹ דָּרְכָה עַל סֶלַע אוֹ אֶבֶן בַּדֶּרֶךְ הַמְּשֻׁבֶּשֶׁת, הוּא עָצַר אֶת נְשִׁימָתוֹ. טַאק... קְלִיק... הַמַּקֵּל הֵקִישׁ עַל הָאֲדָמָה. הוּא צָלַע בִּכְבֵדוּת. גּוּפוֹ הַזָּקֵן נָטָה הַצִּדָּה עִם כָּל פְּסִיעָה. מִי שֶׁעָבַר וְרָאָה אוֹתוֹ, יָכוֹל הָיָה לַחֲשֹׁב שֶׁזֶּה סְתָם קַבְּצָן עָיֵף וּמִסְכֵּן שֶׁגּוֹרֵר אֶת עַצְמוֹ בֶּעָפָר.

"רַבִּי," בָּכָה הַיְּהוּדִי שֶׁהָלַךְ לְצִדּוֹ, "אֲנִי רוֹאֶה עַל הַפָּנִים שֶׁלְּךָ כַּמָּה זֶה כּוֹאֵב לְךָ. אוּלַי נַעֲצֹר?"

"לֹא נוֹעֲדִים בְּאֶמְצַע לְהַצִּיל יְהוּדִים," חִיֵּךְ אֵלָיו רַבִּי יְהוֹשֻׁעַ, "הַקָּדוֹשׁ בָּרוּךְ הוּא יִתֵּן כֹּחַ". הוּא מִלְמֵל פְּסוּקִים שֶׁל אֱמוּנָה מִסֵּפֶר תְּהִלִּים, וְכָךְ, מִתּוֹךְ מְסִירוּת נֶפֶשׁ אֲמִתִּית, הִמְשִׁיךְ לִצְלֹעַ כָּל הַדֶּרֶךְ עַד הָעִיר לָהּ מֻשָּׁב הַשַּׂר הָרוֹמָאִי.

מוּל אַרְמוֹן הַקֵּיסָר – פְּנֵי מַלְאָךְ ה'

הֵם הִגִּיעוּ אַל שַׁעֲרֵי הָאַרְמוֹן הַגָּדוֹל.

יְלָדִים, אִם הֱיִיתֶם רוֹאִים אֶת הַשּׁוֹמְרִים שָׁם, הֱיִיתֶם רוֹעֲדִים. חַיָּלִים עֲנָקִיִּים, לְבוּשִׁים בְּשִׁרְיוֹן בַּרְזֶל מַבְרִיק וְנוֹצֵץ. בִּידֵיהֶם חֲנִיתוֹת חַדּוֹת וַאֲרֻכּוֹת שֶׁנִּצְנְצוּ בַּשֶּׁמֶשׁ. הַשַּׂר יָשַׁב בִּפְנִים, עַל כִּסֵּא שֶׁל שַׁיִשׁ מְעֻטָּר מְרוֹמָם. הוּא הָיָה אָדָם גֵּאֶה וְאַכְזָר. מִי בִּכְלָל יַקְשִׁיב לְרַב יְהוּדִי זָקֵן שֶׁמַּגִּיעַ כָּל כָּךְ עָיֵף, מְכֻסֶּה אָבָק וּמַקֵּל עֵץ בְּיָדוֹ?

רַבִּי יְהוֹשֻׁעַ וְהַיְּהוּדִי אִתּוֹ עָמְדוּ לְהִכָּנֵס. הַשּׁוֹמְרִים רָאוּ שְׁנֵי יְהוּדִים מִתְקָרְבִים, וְרָצוּ לְגָרֵשׁ אוֹתָם.

אֲבָל כָּאן, הִתְעָרֵב בּוֹרֵא עוֹלָם!

הַקָּדוֹשׁ בָּרוּךְ הוּא רָאָה אֶת הַלֵּב הַטָּהוֹר שֶׁל רַבִּי יְהוֹשֻׁעַ. הוּא רָאָה אֶת הַכְּאֵב בָּרֶגֶל שֶׁהוּא וִתֵּר עָלָיו, אֶת הָאַהֲבָה לְעַם יִשְׂרָאֵל, אֶת הָעֲנָוָה שֶׁלּוֹ. כְּשֶׁצַּדִּיק מוֹסֵר אֶת נַפְשׁוֹ עֲבוּר אֲחֵרִים – ה' פּוֹרֵשׂ עָלָיו אֶת נְשִׂיאַת הַחֵן שֶׁלּוֹ!

בָּרֶגַע שֶׁרַבִּי יְהוֹשֻׁעַ בֶּן לֵוִי חָצָה אֶת הַדֶּלֶת, קָרָה דָּבָר נִפְלָא. הָאָבָק עַל פָּנָיו כְּאִלּוּ נֶעֱלַם. שֶׁפַע תּוֹרָתוֹ הִבְרִיק מִתּוֹךְ עֵינָיו. פִּתְאוֹם, פָּנָיו שֶׁל רַבִּי יְהוֹשֻׁעַ קָרְנוּ בְּאוֹר מַסְנְוֵר וְרַךְ שֶׁל הָדָר וּקְדֻשָּׁה, מַמָּשׁ "כִּפְנֵי מַלְאָךְ אֱלֹקִים צְבָאוֹת"!

הַחַיָּלִים הָרוֹמָאִים הָאַכְזָרִים שָׁמְטוּ אֶת הַחֲנִיתוֹת שֶׁלָּהֶם בְּהֶלֶם. הַשַּׂר הָרוֹמָאִי מִיָּד זָקַף אֶת מַבָּטוֹ, רָאָה אֶת הַצַּדִּיק נִכְנָס, וְכָל הַגַּאֲוָה שֶׁלּוֹ נִמְחֲקָה. רַעַד עָבַר בְּגֵו הַשַּׂר. עַל אַף שֶׁהָיָה הַשַּׁלִּיט, הוּא אֲפִלּוּ לֹא שָׂם לֵב לַמַּקֵּל וְלַצְּלִיעָה. הוּא קָם – קָם מִכִּסֵּא הַמַּלְכוּת שֶׁלּוֹ – וְעָמַד בְּיִרְאַת כָּבוֹד מוּל רַבִּי יְהוֹשֻׁעַ הַזָּקֵן!

"מַה יְבַקֵּשׁ כְּבוֹדוֹ?" שָׁאַל הַשַּׂר בְּקוֹל שָׁקֵט, כִּמְעַט מִתְחַנֵּן.

רַבִּי יְהוֹשֻׁעַ, בְּעֲנָוָה אֲבָל מִתּוֹךְ תֹּקֶף קְדֻשָּׁתוֹ, בִּקֵּשׁ לְבַטֵּל אֶת הַגְּזֵרָה וְאֶת הַמַּס הַכָּבֵד שֶׁהֻטַּל עַל הַיְּהוּדִים.

הַשַּׂר לֹא הִתְוַכֵּחַ, לֹא דָּרַשׁ כֶּסֶף כְּשֹׁחַד וְלֹא שָׁאַל שְׁאֵלוֹת. הָאוֹר הָאֱלֹקִי בְּפָנָיו שֶׁל הַצַּדִּיק הִכְנִיעַ אוֹתוֹ לְגַמְרֵי. "בַּקָּשָׁתְךָ מְקֻבֶּלֶת," קָבַע הַשַּׂר. "אַף פָּקִיד רוֹמָאִי לֹא יִפְגַּע בָּעִיר שֶׁלָּכֶם".

סִיּוּם: שִׂמְחָה עִם מַקֵּל

רַבִּי יְהוֹשֻׁעַ יָצָא מֵהָאַרְמוֹן. הַיְּהוּדִי שֶׁעָמַד לְצִדּוֹ בָּכָה – מִשִּׂמְחָה! הָעִיר נִצְּלָה!

הֵם יָצְאוּ לַדֶּרֶךְ חֲזָרָה. רַבִּי יְהוֹשֻׁעַ שׁוּב נִשְׁעַן עַל הַמַּקֵּל, הָרֶגֶל שׁוּב כָּאֲבָה כְּלַפֵּי חוּץ. טַאק... קְלִיק... הוּא צָלַע אֶת כָּל הַדֶּרֶךְ חֲזָרָה לְלוֹד. עָיֵף וְסוֹבֵל. אֲבָל בְּתוֹךְ הַלֵּב, עֲמֹק בִּפְנִים, הַלֵּב שֶׁל רַבִּי יְהוֹשֻׁעַ יָצָא בְּמָחוֹל שֶׁל שִׂמְחָה!

כִּי לִצְלֹעַ כְּדֵי שֶׁלִּיהוּדִים אֲחֵרִים יִהְיֶה טוֹב – זֶה לֹא כְּאֵב. זֶה קִדּוּשׁ ה' הֲכִי גָּדוֹל שֶׁיֵּשׁ בָּעוֹלָם!

מקורות: ירושלמי, מסכת תרומות פרק ח' הלכה ד': "עולא בר קושב תבעתיה מלכותא... עד דאתא ר' יהושע בן לוי ופייס עליה". ירושלמי, מסכת ברכות פרק ה' הלכה א': "אשכחון רבי יהושע בן לוי ורבי חנינא... דהוון אילין רומאי קיימין מן קומיהון. אמרין לון למלה אתון קיימין מקמי אלין יהודאי? אמרין לון, אפיהון דמלאכיא אנן חמיין" (כפני מלאך). ספר האגדה, סימן שפו/שפה: סיפורי מאמציו של ריב"ל אצל הרומאים כשגופו כואב וצולע על ירכו, ומאירים לו פניו כפני מלאך. (סעיפים המוזכרים מפורשות במקורות שסופקו).

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “יְלָדִים, אֲנַחְנוּ נִמְצָאִים בְּאֶרֶץ יִשְׂרָאֵל, לִפְנֵי כִּמְעַט אַלְפַּיִם שָׁנָה.” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “אִם לֹא נָבִיא אֶת הַכֶּסֶף עַד מָחָר – יַחֲרִיבוּ אֶת הָעִיר וְיִקְחוּ אֶת מִשְׁפְּחוֹתֵינוּ לַכֶּלֶא!”?
ג. כיצד הגיב ר' יהושע בן לוי למצב המתואר: “רְגָבִים שֶׁל יֹבֶשׁ וְאֲבָק דָּבְקוּ לְבִגְדֵי הַקֹּדֶשׁ שֶׁלּוֹ.”?
ד. מה השתנה בין “הַשַּׂר יָשַׁב בִּפְנִים, עַל כִּסֵּא שֶׁל שַׁיִשׁ מְעֻטָּר מְרוֹמָם.” לבין “פִּתְאוֹם, פָּנָיו שֶׁל רַבִּי יְהוֹשֻׁעַ קָרְנוּ בְּאוֹר מַסְנְוֵר וְרַךְ שֶׁל הָדָר וּקְדֻשָּׁה, מַמָּשׁ "כִּפְנֵי…”?
ה. כיצד הגיע הסיפור אל האירוע: “הָאוֹר הָאֱלֹקִי בְּפָנָיו שֶׁל הַצַּדִּיק הִכְנִיעַ אוֹתוֹ לְגַמְרֵי.”?

3. מה היה ראשון?

כתבו ליד כל אירוע מספר מ־1 עד 5 לפי סדר התרחשותו בסיפור.

□ הוּא מִלְמֵל פְּסוּקִים שֶׁל אֱמוּנָה מִסֵּפֶר תְּהִלִּים, וְכָךְ, מִתּוֹךְ מְסִירוּת נֶפֶשׁ אֲמִתִּית, הִמְשִׁיךְ לִצְלֹעַ כָּל הַדֶּרֶךְ עַד הָעִיר לָהּ…
□ יְלָדִים, אֲנַחְנוּ נִמְצָאִים בְּאֶרֶץ יִשְׂרָאֵל, לִפְנֵי כִּמְעַט אַלְפַּיִם שָׁנָה.
□ כִּי לִצְלֹעַ כְּדֵי שֶׁלִּיהוּדִים אֲחֵרִים יִהְיֶה טוֹב – זֶה לֹא כְּאֵב.
□ יוֹתֵר מִכָּךְ – הָיְתָה לוֹ בְּעָיָה קָשָׁה מְאֹד בְּרַגְלוֹ.
□ הַחַיָּלִים הָרוֹמָאִים הָאַכְזָרִים שָׁמְטוּ אֶת הַחֲנִיתוֹת שֶׁלָּהֶם בְּהֶלֶם.

לפני ואחרי

מה קרה לפני: “הוּא מִלְמֵל פְּסוּקִים שֶׁל אֱמוּנָה מִסֵּפֶר תְּהִלִּים, וְכָךְ, מִתּוֹךְ מְסִירוּת נֶפֶשׁ אֲמִתִּית, הִמְשִׁיךְ לִצְלֹעַ כָּל הַדֶּרֶךְ עַד הָעִיר לָהּ…”?
מה קרה אחרי: “הַחַיָּלִים הָרוֹמָאִים הָאַכְזָרִים שָׁמְטוּ אֶת הַחֲנִיתוֹת שֶׁלָּהֶם בְּהֶלֶם.”?

סיבה ותוצאה מתוך המעשה

האירוע הנתוןמה גרם לו?מה הייתה תוצאתו?
הוּא צָלַע עַל יְרֵכוֹ, וְכָל צַעַד שֶׁעָשָׂה הָיָה מְלֻוֶּה בִּכְאֵב דּוֹקֵר, כְּמוֹ מַחַט עָמֹק בְּתוֹךְ הָעֶצֶם.
הוּא רָאָה אֶת הַכְּאֵב בָּרֶגֶל שֶׁהוּא וִתֵּר עָלָיו, אֶת הָאַהֲבָה לְעַם יִשְׂרָאֵל, אֶת הָעֲנָוָה שֶׁלּוֹ.

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “צולע על ירכו בגלל מאמציו לגויים” על ר' יהושע בן לוי? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “רַבִּי," בָּכָה הַיְּהוּדִי שֶׁהָלַךְ לְצִדּוֹ, "אֲנִי רוֹאֶה עַל הַפָּנִים שֶׁלְּךָ כַּמָּה זֶה כּוֹאֵב לְךָ.”.
ג. אילו ר' יהושע בן לוי לא היה פועל כפי שמתואר ב־“פִּתְאוֹם, פָּנָיו שֶׁל רַבִּי יְהוֹשֻׁעַ קָרְנוּ בְּאוֹר מַסְנְוֵר וְרַךְ שֶׁל הָדָר וּקְדֻשָּׁה, מַמָּשׁ "כִּפְנֵי מַלְאָךְ…” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “הָאוֹר הָאֱלֹקִי בְּפָנָיו שֶׁל הַצַּדִּיק הִכְנִיעַ אוֹתוֹ לְגַמְרֵי.” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. יְלָדִים, אֲנַחְנוּ נִמְצָאִים בְּאֶרֶץ יִשְׂרָאֵל, לִפְנֵי כִּמְעַט…
2. אִם לֹא נָבִיא אֶת הַכֶּסֶף עַד מָחָר – יַחֲרִיבוּ אֶת הָעִיר וְיִקְחוּ…
3. בְּכָל פַּעַם שֶׁרַגְלוֹ דָּרְכָה עַל סֶלַע אוֹ אֶבֶן בַּדֶּרֶךְ…
4. הַקָּדוֹשׁ בָּרוּךְ הוּא רָאָה אֶת הַלֵּב הַטָּהוֹר שֶׁל רַבִּי…
5. עַל אַף שֶׁהָיָה הַשַּׁלִּיט, הוּא אֲפִלּוּ לֹא שָׂם לֵב לַמַּקֵּל…
6. כִּי לִצְלֹעַ כְּדֵי שֶׁלִּיהוּדִים אֲחֵרִים יִהְיֶה טוֹב – זֶה לֹא…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “רְגָבִים שֶׁל יֹבֶשׁ וְאֲבָק דָּבְקוּ לְבִגְדֵי הַקֹּדֶשׁ שֶׁלּוֹ.”.
מכתב: כתבו לר' יהושע בן לוי בעקבות “הוּא רָאָה אֶת הַכְּאֵב בָּרֶגֶל שֶׁהוּא וִתֵּר עָלָיו, אֶת הָאַהֲבָה לְעַם יִשְׂרָאֵל, אֶת הָעֲנָוָה שֶׁלּוֹ.”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “עַל אַף שֶׁהָיָה הַשַּׁלִּיט, הוּא אֲפִלּוּ לֹא שָׂם לֵב לַמַּקֵּל וְלַצְּלִיעָה.”.
דיווח: כתבו ידיעה קצרה שכותרתה “צולע על ירכו בגלל מאמציו לגויים”.
