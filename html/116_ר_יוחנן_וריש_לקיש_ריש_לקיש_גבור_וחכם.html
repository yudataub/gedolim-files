ריש לקיש — גבור וחכם

דף עבודה בעקבות ר' יוחנן וריש לקיש
שם: כיתה:

המעשה — הסיפור המלא

להלן עיבוד חינוכי, מותח, רגשי ועשיר לסיפור חזרתו בתשובה של ריש לקיש והפיכתו ממנהיג שודדים לגאון בתורה, יחד עם סיפור גבורתו למען רבו, רַבִּי יוֹחָנָן. הסיפור נכתב בדיוק לפי ההנחיות: שפה מכבדת, יצירת מתח, תיאורים חושיים ("Show, Don't Tell"), והתאמה מושלמת לבנים בכיתות ב'-ד' בחינוך הדתי-לאומי.

הסיפור מנוקד במלואו כנדרש.

פְּתִיחָה: הַקְּפִיצָה בְּנְהַר הַיַּרְדֵּן

יְלָדִים, אִם הֱיִיתֶם עוֹמְדִים לִפְנֵי שָׁנִים רַבּוֹת עַל גְּדַת נְהַר הַיַּרְדֵּן, הֱיִיתֶם רוֹאִים אֶת הַמַּיִם שׁוֹצְפִים וְקוֹצְפִים, שׁוֹמְעִים אֶת רַעַשׁ הַגַּלִּים מִתְנַפְּצִים עַל הַסְּלָעִים, וּמַרְגִּישִׁים אֶת הַשֶּׁמֶשׁ הַחַמָּה שֶׁל אֶרֶץ יִשְׂרָאֵל. בְּאוֹתָם יָמִים, הָיְתָה סַכָּנָה גְּדוֹלָה לְהִסְתּוֹבֵב שָׁם לְבַד. בַּמְּעָרוֹת וּבַיְּעָרוֹת הִסְתַּתְּרוּ לִסְטִים – שׁוֹדְדִים חֲמוּשִׁים שֶׁהִטִּילוּ אֵימָה עַל הָעוֹבְרִים וְהַשָּׁבִים.

מַנְהִיג הַשּׁוֹדְדִים הָיָה אָדָם בַּעַל גּוּף עֲנָק, חָזָק יוֹתֵר מִכָּל אָדָם אַחֵר, וּשְׁמוֹ שִׁמְצוֹן בֶּן לָקִישׁ. הוּא הָיָה מְסֻגָּל לְהָרִים סְלָעִים עֲנָקִיִּים וְלִקְפֹּץ מֵהַר לְהַר בְּלִי לְהִתְעַיֵּף! מֵרֹב כּוֹחוֹ הַגָּדוֹל, כֻּלָּם פָּחֲדוּ מִמֶּנּוּ.

יוֹם אֶחָד, רָחַץ שִׁמְעוֹן בֶּן לָקִישׁ בְּמֵימֵי הַיַּרְדֵּן. לְפֶתַע, סִנְוֵר אוֹתוֹ אוֹר חָזָק. הוּא הִבִּיט לַגָּדָה הַשְּׁנִיָּה שֶׁל הַנָּהָר, וְרָאָה שָׁם אָדָם כָּל כָּךְ יָפֶה וְזוֹרֵחַ, עַד שֶׁהַמַּיִם מִסְּבִיבוֹ הֵאִירוּ! זֶה הָיָה עֲנַק הַתּוֹרָה, גְּדוֹל הַדּוֹר – רַבִּי יוֹחָנָן . רַבִּי יוֹחָנָן הָיָה קָדוֹשׁ וְטָהוֹר, וְהַיֹּפִי הָרוּחָנִי שֶׁלּוֹ זָרַח מִפָּנָיו כְּמוֹ שֶׁמֶשׁ.

שִׁמְעוֹן בֶּן לָקִישׁ לֹא חָשַׁב פַּעֲמַיִם. הוּא לָקַח תְּנוּפָה, כִּוֵּץ אֶת רַגְלָיו הַחֲזָקוֹת – וּבִקְפִיצָה אַחַת אַדִּירָה רִחֵף בָּאֲוִיר מֵעַל הַנָּהָר הָרָחָב, עַד שֶׁנָּחַת מַמָּשׁ קָרוֹב לְרַבִּי יוֹחָנָן.

הָעִסְקָה שֶׁשִּׁנְּתָה אֶת הָעוֹלָם

רַבִּי יוֹחָנָן לֹא נִבְהַל מֵהָאִישׁ הַחָזָק וְהַמְּאַיֵּם שֶׁקָּפַץ מוּלוֹ. הוּא הִבִּיט בּוֹ בְּעֵינַיִם רַכּוֹת וַחֲכָמוֹת, רָאָה אֶת הַפּוֹטֶנְצְיָאל שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא נָתַן לוֹ, וְאָמַר לוֹ מִשְׁפָּט שֶׁהִדְהֵד בְּלִבּוֹ:

"חֵילָךְ לְאוֹרַיְתָא!" – כּוֹחֲךָ הֶעָצוּם צָרִיךְ לִהְיוֹת שַׁיָּךְ לַתּוֹרָה! חֲבָל עַל שְׁרִירִים כָּאֵלֶּה שֶׁיִּתְבַּזְבְּזוּ עַל מַעֲשִׂים רָעִים וְשֹׁד! אִם תַּשְׁקִיעַ אֶת כָּל הַמֶּרֶץ הַזֶּה בְּלִמּוּד תּוֹרָה, תִּהְיֶה מֵהַחֲכָמִים הַגְּדוֹלִים בָּעוֹלָם!

שִׁמְעוֹן עָנָה לוֹ: "וְיָפְיְךָ הַמֵּאִיר – רָאוּי לְנָשִׁים!". הָיָה כָּאן מִפְגָּשׁ בֵּין כּוֹחַ הַגּוּף (שִׁמְעוֹן) לְיֹפִי וְרוּחָנִיּוּת (רַבִּי יוֹחָנָן).

רַבִּי יוֹחָנָן חִיֵּךְ וְאָמַר: "אִם תַּחֲזֹר בִּתְשׁוּבָה וְתִלְמַד תּוֹרָה, אֶתֵּן לְךָ לְאִשָּׁה אֶת אֲחוֹתִי, שֶׁהִיא מְאִירָה וְיָפָה מִמֶּנִּי".

יְלָדִים, אוֹתוֹ רֶגַע הָיָה רֶגַע קְרִיטִי. שִׁמְעוֹן יָכוֹל הָיָה לִצְחֹק וְלַחְזֹר לַחֲבֵרָיו הַשּׁוֹדְדִים. אַךְ מַשֶּׁהוּ בַּקְּדֻשָּׁה שֶׁל רַבִּי יוֹחָנָן נָגַע בְּנִשְׁמָתוֹ. "אֲנִי מְקַבֵּל עָלַי!" הוּא קָרָא.

מִיָּד כַּאֲשֶׁר שִׁמְעוֹן קִבֵּל עָלָיו אֶת עֹל הַתּוֹרָה, קָרָה דָּבָר פֶּלֶא מֵהַשָּׁמַיִם. הוּא רָצָה לִקְפֹּץ בַּחֲזָרָה גַּדָּה כְּדֵי לְהָבִיא אֶת בְּגָדָיו, אַךְ רַגְלָיו בָּגְדוּ בּוֹ! הַכּוֹחַ הַגַּשְׁמִי, הַפְּרָאִי, נֶעֱלַם בְּבַת אַחַת. כָּל כּוֹחוֹ עָבַר עַכְשָׁו אֶל הַמֹּחַ וְהַלֵּב. רַבִּי יוֹחָנָן עָזַר לוֹ, הִלְבִּישׁ אוֹתוֹ, וְלָקַח אוֹתוֹ אִתּוֹ לְבֵית הַמִּדְרָשׁ. מֵאוֹתוֹ יוֹם קָרְאוּ לוֹ: רֵישׁ לָקִישׁ .

זֵעַת הַתּוֹרָה וְהַגְּבוּרָה הַחֲדָשָׁה

אֶל תַּחְשְׁבוּ שֶׁזֶּה הָיָה קַל! רֵישׁ לָקִישׁ יָשַׁב בְּבֵית הַמִּדְרָשׁ יוֹם וָלַיְלָה. הוּא חָשַׁק אֶת שְׂפָתָיו, קִמֵּט אֶת מִצְחוֹ בְּרִכּוּז. רֵיחַ עָתִיק שֶׁל גְּוִילֵי קְלַף מִלֵּא אֶת הָאֲוִיר. לִפְעָמִים כָּאֵב לוֹ הָרֹאשׁ מֵרֹב מַאֲמָץ. זֵעָה חַמָּה נָטְפָה עַל פָּנָיו כְּשֶׁחָזַר שׁוּב וָשׁוּב עַל הַמִּשְׁנָיוֹת. הוּא הִשְׁתַּמֵּשׁ בְּכָל כּוֹחַ הַסְּבֵלָנוּת הֶעָצוּם שֶׁלּוֹ כְּדֵי לַהֲבִין סְבָרָה קָשָׁה!

וְהוּא הִצְלִיחַ! בְּהַתְמָדָה עֲצוּמָה, הוּא הָפַךְ לַתַּלְמִיד-חָבֵר הֲכִי קָרוֹב שֶׁל רַבִּי יוֹחָנָן. כְּשֶׁהָיוּ לוֹמְדִים יַחַד, זֶה הָיָה כְּמוֹ אֵשׁ וָמַיִם! הָיוּ שׁוֹאֲלִים זֶה אֶת זֶה, חָקְרוּ וְתֵרְצוּ בַּהֲלָכָה, וְכָל הַדּוֹר הֵאִיר מִתּוֹרָתָם.

הַדְּמָעוֹת שֶׁל מַנְהִיג הַדּוֹר

עָבְרוּ שָׁנִים. עַכְשָׁו רֵישׁ לָקִישׁ הוּא תַּלְמִיד חָכָם עָצוּם שֶׁיּוֹשֵׁב קָבוּעַ בְּבֵית הַמִּדְרָשׁ הַמּוּגָן. לִכְאוֹרָה, לָמָּה צָרִיךְ אָדָם קָדוֹשׁ גְּבוּרָה שֶׁל חַיָּל אוֹ שׁוֹדֵד עָבָר? יְלָדִים, הַקָּדוֹשׁ בָּרוּךְ הוּא לֹא נָתַן אֶת הַכּוֹחַ הַזֶּה סְתָם!

יוֹם אֶחָד, הָלַךְ רַבִּי יוֹחָנָן הַגָּדוֹל לְבַדּוֹ בִּדְרָכִים, הָרְחוֹקִים מִן הָעִיר וְשׁוֹמְמִים.

לְפֶתַע... שְׁרִיקַת צְעָדִים וְרִשְׁרוּשׁ מַתֶּכֶת! מִתּוֹךְ הַשִּׂיחִים קָפְצוּ חֲבוּרָה שֶׁל עֲשָׂרָה לִסְטִים אֲכְזָרִיִּים וּבְיָדָם חֲרָבוֹת נוֹצְצוֹת מֵבַּרְזֶל.

רַבִּי יוֹחָנָן הַיָּשִׁישׁ לֹא יָכוֹל הָיָה לְהִתְנַגֵּד. הַשּׁוֹדְדִים דָּחֲפוּ אוֹתוֹ, אִיְּמוּ עָלָיו, וּלְלֹא רַחֲמִים שָׁדְדוּ אֶת אַרְנְקוֹ שֶׁהָיָה מָלֵא בְּכֶסֶף רַב שֶׁיּוֹעַד לִצְדָקָה וְלַהֲחְזָקַת הַיְּשִׁיבָה! רַבִּי יוֹחָנָן נִשְׁאַר בְּלִי כְּלוּם, עוֹמֵד עָצוּב וּכְאוּב תַּחַת הַשֶּׁמֶשׁ.

הוּא הִגִּיעַ לְבֵית הַמִּדְרָשׁ, חִוֵּר וְנָבוֹךְ. הוּא יָשַׁב עַל כִּסְאוֹ, וְתַלְמִידוֹ רֵישׁ לָקִישׁ שָׂם לֵב שֶׁרַבִּי יוֹחָנָן שׁוֹתֵק וּפָנָיו נְפוּלוֹת.

"כְּבוֹד הָרַב, מָה אֵרַע? מַדּוּעַ אֵינְךָ כְּהֶרְגֶּלְךָ הַיּוֹם?" שָׁאַל רֵישׁ לָקִישׁ בְּדַאֲגָה וְאַהֲבָה.

רַבִּי יוֹחָנָן שָׁתַק מְעַט, וְאָז סִפֵּר לוֹ אֶת אֲשֶׁר אֵרַע. "כְּשֶׁהָיִיתִי בַּדֶּרֶךְ," לָחַשׁ הָרַב, "הִקִּיפוּ אוֹתִי שׁוֹדְדִים. לֹא יָכֹלְתִּי לַעֲשׂוֹת מְאוּמָה. הֵם לָקְחוּ הַכֹּל, הַכֶּסֶף שֶׁל הַעֲנִיִּים... וְעַכְשָׁו, לֹא אוּכַל לְהַמְשִׁיךְ לִלְמֹד בְּשִׂמְחָה, שֶׁכֵּן הַדְּאָגָה הַזּוֹ רוֹבֶצֶת עַל לִבִּי."

הַמִּרְדָּף בֶּהָרִים וְנִצָּחוֹן הָרוּחַ

רֵישׁ לָקִישׁ שָׁמַע אֶת דִּבְרֵי רַבּוֹ, וְהַדָּם בְּעוֹרְקָיו הִתְחִיל לִרְתֹּחַ – אַךְ הַפַּעַם, זוֹ הָיְתָה רְתִיחָה שֶׁל תּוֹרָה! של קְדֻשָּׁה!

"לְאֵיזֶה כִּוּוּן הֵם בָּרְחוּ?" זָקַף רֵישׁ לָקִישׁ אֶת קוֹמָתוֹ. רַבִּי יוֹחָנָן הִצְבִּיעַ בְּיָד רְוֹעֶדֶת אֶל עֵבֶר הֶהָרִים הַחֲשׂוּפִים.

רֵישׁ לָקִישׁ, שֶׁעַכְשָׁו לָבַשׁ בִּגְדֵי חֲכָמִים צְנוּעִים וְיָשַׁב שָׁנִים עַל גַּבֵּי שָׁנִים בְּאֹהֶלָהּ שֶׁל תּוֹרָה, קָם כַּאֲרִי. אֵין לוֹ חֶרֶב, אֵין לוֹ חֲנִית וְאֵין לוֹ מָגֵן. יֵשׁ לוֹ רַק אֶת כּוֹחַ הַתּוֹרָה וֶאֱמוּנָה בְּצִדְקַת הַדֶּרֶךְ.

הוּא פָּתַח בְּרִיצָה אַדִּירָה, מְדַלֵּג מֵעַל סְלָעִים וְקוֹצִים כְּמוֹ נָמֵר בַּשָּׂדֶה. פַּעַם, הוּא קָפַץ אֶת הַיַּרְדֵּן כְּדֵי לִשְׁדֹּד. עַכְשָׁו הַגּוּף הַזֶּה הוּא קָדוֹשׁ – מֻקְדָּשׁ כֻּלּוֹ כְּדֵי לְהַצִּיל אֶת כְּבוֹדוֹ שֶׁל תַּלְמִיד חָכָם וּלְהָשִׁיב אֶת כֶּסֶף הַהֶקְדֵּשׁ!

עָבְרָה שָׁעָה קַלָּה, וְהִנֵּה הוּא רוֹאֶה אֶת הַשּׁוֹדְדִים יוֹשְׁבִים לְיַד מְדוּרָה, סוֹפְרִים אֶת מַטְבְּעוֹת הַכֶּסֶף שֶׁל רַבִּי יוֹחָנָן וְצוֹחֲקִים. הַחֲרָבוֹת שֶׁלָּהֶם נָחוּ עַל בִּרְכֵּיהֶם.

רֵישׁ לָקִישׁ לֹא הִתְחַבֵּא! הוּא הִזְדַּקֵּף מֵעֲלֵיהֶם, מַבָּטוֹ רוֹשֵׁף אֵשׁ!

הוּא קָרָא אֲלֵיהֶם בְּקוֹל רוֹעֵם שֶׁהִרְעִיד אֶת הָעֵצִים: "הַחְזִירוּ מִיָּד אֶת הַכֶּסֶף שֶׁל רַבִּי יוֹחָנָן!"

הַשּׁוֹדְדִים הִסְתּוֹבְבוּ, אָחֲזוּ בַּנֶּשֶׁק שֶׁלָּהֶם, אַךְ כְּשֶׁרָאוּ אֶת הָאִישׁ הָעוֹמֵד מוּלָם, הֵם קָפְאוּ בִּמְקוֹמָם. הֵם זִהוּ אֶת שִׁמְעוֹן! הֵם זָכְרוּ שֶׁאֵין אִישׁ שֶׁיָּכוֹל לוֹ! "אֲ... אֲבָל... הֲרֵי חָזַרְתָּ בִּתְשׁוּבָה!" הֵם גִּמְגְּמוּ בְּפַחַד, "לָמָּה בָּאתָ לְכָאן בְּלִי נֶשֶׁק? מָה, רֵישׁ לָקִישׁ לֹא מְפַחֵד מֵאִתָּנוּ?"

רֵישׁ לָקִישׁ הִבִּיט בָּהֶם בְּרַחֲמִים וּבְעָצְמָה, וְאָמַר לָהֶם בְּבִטָּחוֹן מֻחְלָט: "וְכִי אַתֶּם חוֹשְׁבִים שֶׁגִּבּוֹר אֲמִתִּי צָרִיךְ נֶשֶׁק?! אֲנִי בָּאתִי בְּכוֹחַ הַתּוֹרָה הַקְּדוֹשָׁה, שֶׁשּׁוֹמֶרֶת וּמְגִנָּה עָלַי! אִם לֹא תַּחְזִירוּ אֶת הַכֶּסֶף, לֹא תֵּצְאוּ מִפֹּה הַיּוֹם. אֲנִי לֹא מְפַחֵד מִיְּצוּר בָּעוֹלָם, מִלְּבַד מִבּוֹרֵא עוֹלָם!"

הַשּׁוֹדְדִים רָעֲדוּ מִפַּחַד. עֵינָיו שֶׁל רֵישׁ לָקִישׁ אָמְרוּ אֶת הָאֱמֶת. הֵם הֵבִינוּ שֶׁיֵּשׁ בּוֹ כּוֹחַ מְיֻחָד שֶׁאֵין לְאַף לוֹחֵם בָּעוֹלָם. מִיָּד הֵם הִגִּישׁוּ אֵלָיו אֶת כָּל כַּסְפּוֹ שֶׁל רַבִּי יוֹחָנָן. הֵם אֲפִלּוּ הִצִּיעוּ: "אוּלַי תִּקַּח רַק חֵצִי וְנִשְׁמֹר לָנוּ חֵצִי?"

אַךְ רֵישׁ לָקִישׁ עָנָה: "אַף לֹא פְּרוּטָה מֵהַהֶקְדֵּשׁ שֶׁל תַּלְמִיד חָכָם!"

הוּא אָסַף אֶת כָּל הַכֶּסֶף כְּשֶׁהוּא שָׁלֵם, וְחָזַר בְּשִׂמְחָה עֲצוּמָה וְרִיצָה מְהִירָה אֶל בֵּית הַמִּדְרָשׁ.

כְּשֶׁהִנִּיחַ אֶת הָאַרְנָק עַל שֻׁלְחָנוֹ שֶׁל רַבִּי יוֹחָנָן, קָמוּ כָּל תַּלְמִידֵי הַיְשִׁיבָה עַל רַגְלֵיהֶם בִּפְלִיאָה. הֵם הֵבִינוּ אֶת מָה שֶׁרַבִּי יוֹחָנָן יָדַע כְּבָר בַּיּוֹם הָרִאשׁוֹן בַּיַּרְדֵּן: גִּבּוֹר הוּא לֹא מִי שֶׁמַּרְבִּיץ, אֶלָּא מִי שֶׁכּוֹבֵשׁ אֶת יִצְרוֹ וּמַקְדִּישׁ אֶת כּוֹחוֹ לַעֲשׂוֹת צֶדֶק, לִלְמֹד תּוֹרָה, וּלְהָגֵן עַל קְדֻשַּׁת יִשְׂרָאֵל.

הַגִּבּוֹר שֶׁלָּנוּ (ME): רֵישׁ לָקִישׁ הָיָה הָאִישׁ הֲכִי חָזָק בַּדּוֹר. אַךְ בִּמְקוֹם לְהִשְׁתַּמֵּשׁ בַּכּוֹחַ שֶׁלּוֹ לְהַפְחִיד צְעִירִים אוֹ לְהַשְׁוִיץ, הוּא הֵבִין שֶׁכּוֹחַ הַגּוּף הוּא מַתָּנָה מֵה'. בְּרֶגַע אֶחָד שֶׁל תְּשׁוּבָה הוּא עָזַב הַכֹּל, הָפַךְ אֶת כּוֹחוֹ לְהַתְמָדָה בַּתּוֹרָה, וְהִשְׁתַּמֵּשׁ בַּגְּבוּרָה שֶׁלּוֹ שׁוּב, וְהַפַּעַם – רַק כְּדֵי לַעֲזֹר לְרַבּוֹ וּלְהַצִּיל עֲשׁוּקִים!

אֲנַחְנוּ (WE): לִפְעָמִים גַּם אֲנַחְנוּ מַרְגִּישִׁים כָּל כָּךְ הַרְבֵּה מֶרֶץ. יֵשׁ לָנוּ כּוֹחַ לָרוּץ מַהֵר בַּהַפְסָקָה, לִהְיוֹת הֲכִי חֲזָקִים בְּמִשְׂחַק כַּדּוּרַגְל, אוֹ לְדַבֵּר בְּקוֹל רָם. אַךְ אֵיפֹה הַנְּפִילָה שֶׁלָּנוּ? שֶׁלִּפְעָמִים כְּשֶׁלֹּא שָׂמִים לֵב, מִשְׁתַּמְּשִׁים בַּכּוֹחַ הַזֶּה כְּדֵי לִדְחֹף חָבֵר, לְהָצִיק לְיֶלֶד חַלָּשׁ אוֹ לְהַפְרִיעַ בַּכִּתָּה, בִּמְקוֹם לִהְיוֹת "גִּבּוֹרִים אֲמִתִּיִּים".

אַתֶּם (YOU): יְלָדִים, אֵיךְ נְיַשֵּׂם אֶת זֶה מָחָר בַּבֹּקֶר? כְּשֶׁאַתֶּם מַרְגִּישִׁים שֶׁיֵּשׁ לָכֶם כּוֹחַ רַב, "שְׁרִירִים" שֶׁל מֶרֶץ, תַּהַפְכוּ אֶת זֶה לִ"שְׁרִירִים שֶׁל מִצְוָה". רוּצוּ מַהֵר לְהָבִיא סִדּוּר לְסַבָּא בְּבֵית הַכְּנֶסֶת, הִשְׁתַּמְּשׁוּ בַּכּוֹחַ שֶׁלָּכֶם לַעֲזֹר לְסַחֵב שַׂקִּיּוֹת שֶׁל קְנִיּוֹת לְאִמָּא, וּכְשֶׁאַתֶּם מִתְיַשְּׁבִים לִלְמֹד תּוֹרָה – הַשְׁקִיעוּ אֶת הַמֶּרֶץ כְּדֵי לְהָבִין לַעֲמֹק, מַמָּשׁ כְּמוֹ רֵישׁ לָקִישׁ שֶׁהָפַךְ לְעֲנַק רוּחַ!

א. שְׁאֵלַת פְּתִיחָה (לִפְנֵי תְּחִלַּת הַסִּפּוּר):

"יְלָדִים, מִי כָּאן בַּכִּתָּה חוֹשֵׁב שֶׁהוּא מְאֹד חָזָק וְיָכוֹל לָרוּץ הֲכִי מַהֵר? וּמָה זֶה גִּבּוֹר בֶּאֱמֶת – הַאִם זֶה רַק מִי שֶׁיֵּשׁ לוֹ שְׁרִירִים, אוֹ גַּם מִי שֶׁיֵּשׁ לוֹ מַשֶּׁהוּ אַחֵר? הַיּוֹם נִשְׁמַע עַל אִישׁ שֶׁהָיָה אֲלוּף הַגִּבּוֹרִים בַּכּוֹחַ הַגּוּפָנִי שֶׁלּוֹ, אַךְ כְּשֶׁהוּא הִכִּיר אֶת הַתּוֹרָה – הוּא בָּחַר לְשַׁנּוֹת אֶת כָּל הַחַיִּים שֶׁלּוֹ מֵהַקָּצֶה לַקָּצֶה!"

ב. הֶעָרוֹת לְבִצּוּעַ וּסְטוֹרִיטֵלִינְג (שְׁלִיטָה קוֹלִית וְחָזוּתִית):

ג. הַמְלָצוֹת לְשִׁלּוּב בְּמַצֶּגֶת (תְּמוּנוֹת מְדֻיָּקוֹת וּבִקְדֻשָּׁה):

ד. פְּעִילוּת מְסַכֶּמֶת (לִרְצוֹן הַמּוֹרֶה):

מִשְׂחַק מִשְׁוְיוֹת הַגְּבוּרָה ("מִבְחַן הַשְּׁרִירִים"):

המורה מביא חפץ כבד יחסית (למשל, קופסת ספרים או ארגז הפתעות סגור).

מקורות: תלמוד בבלי, מסכת בבא מציעא, דף פ"ד עמוד א': סיפור הפגישה הראשונה של רבי יוחנן וריש לקיש בירדן. המעבר מכוח פיזי לכוח התורה ("חילך לאורייתא" – "שופרך לנשי"), ותיאור ההתעלות המשותפת. ספרי האגדה לילדים (מקורות החמ"ד למורה, דף קריאה - "ריש לקיש והשודדים"): עיבוד מסורתי של המרדף בו ריש לקיש מנצל את האומץ הטבעי שלו, הפעם כדי לצאת ולהציל את כספו של רבי יוחנן מפני שודדים בהרים, ללא נשק, רק מכוח אמונתו ותורתו. הדגשת העובדה שהפך את כוחו לחיובי.

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “להלן עיבוד חינוכי, מותח, רגשי ועשיר לסיפור חזרתו בתשובה של ריש לקיש והפיכתו ממנהיג שודדים לגאון בתורה, יחד עם סיפור גבורתו למען רבו, רַבִּי יוֹחָנָן.” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “רַבִּי יוֹחָנָן לֹא נִבְהַל מֵהָאִישׁ הַחָזָק וְהַמְּאַיֵּם שֶׁקָּפַץ מוּלוֹ.”?
ג. כיצד הגיב ר' יוחנן וריש לקיש למצב המתואר: “הוּא חָשַׁק אֶת שְׂפָתָיו, קִמֵּט אֶת מִצְחוֹ בְּרִכּוּז.”?
ד. מה השתנה בין “רַבִּי יוֹחָנָן נִשְׁאַר בְּלִי כְּלוּם, עוֹמֵד עָצוּב וּכְאוּב תַּחַת הַשֶּׁמֶשׁ.” לבין “הַשּׁוֹדְדִים הִסְתּוֹבְבוּ, אָחֲזוּ בַּנֶּשֶׁק שֶׁלָּהֶם, אַךְ כְּשֶׁרָאוּ אֶת הָאִישׁ הָעוֹמֵד מוּלָם, הֵם קָפְאוּ…”?
ה. כיצד הגיע הסיפור אל האירוע: “בְּרֶגַע אֶחָד שֶׁל תְּשׁוּבָה הוּא עָזַב הַכֹּל, הָפַךְ אֶת כּוֹחוֹ לְהַתְמָדָה בַּתּוֹרָה, וְהִשְׁתַּמֵּשׁ בַּגְּבוּרָה שֶׁלּוֹ שׁוּב, וְהַפַּעַם – רַק…”?

3. מה היה ראשון?

כתבו ליד כל אירוע מספר מ־1 עד 5 לפי סדר התרחשותו בסיפור.

□ מִתּוֹךְ הַשִּׂיחִים קָפְצוּ חֲבוּרָה שֶׁל עֲשָׂרָה לִסְטִים אֲכְזָרִיִּים וּבְיָדָם חֲרָבוֹת נוֹצְצוֹת מֵבַּרְזֶל.
□ להלן עיבוד חינוכי, מותח, רגשי ועשיר לסיפור חזרתו בתשובה של ריש לקיש והפיכתו ממנהיג שודדים לגאון בתורה, יחד עם סיפור גבורתו למען רבו, רַבִּי יוֹחָנָן.
□ המורה מביא חפץ כבד יחסית (למשל, קופסת ספרים או ארגז הפתעות סגור).
□ שִׁמְעוֹן עָנָה לוֹ: "וְיָפְיְךָ הַמֵּאִיר – רָאוּי לְנָשִׁים!".
□ אֲנִי בָּאתִי בְּכוֹחַ הַתּוֹרָה הַקְּדוֹשָׁה, שֶׁשּׁוֹמֶרֶת וּמְגִנָּה עָלַי!

לפני ואחרי

מה קרה לפני: “מִתּוֹךְ הַשִּׂיחִים קָפְצוּ חֲבוּרָה שֶׁל עֲשָׂרָה לִסְטִים אֲכְזָרִיִּים וּבְיָדָם חֲרָבוֹת נוֹצְצוֹת מֵבַּרְזֶל.”?
מה קרה אחרי: “אֲנִי בָּאתִי בְּכוֹחַ הַתּוֹרָה הַקְּדוֹשָׁה, שֶׁשּׁוֹמֶרֶת וּמְגִנָּה עָלַי!”?

סיבה ותוצאה מתוך המעשה

האירוע הנתוןמה גרם לו?מה הייתה תוצאתו?
הָיָה כָּאן מִפְגָּשׁ בֵּין כּוֹחַ הַגּוּף (שִׁמְעוֹן) לְיֹפִי וְרוּחָנִיּוּת (רַבִּי יוֹחָנָן).
רֵישׁ לָקִישׁ, שֶׁעַכְשָׁו לָבַשׁ בִּגְדֵי חֲכָמִים צְנוּעִים וְיָשַׁב שָׁנִים עַל גַּבֵּי שָׁנִים בְּאֹהֶלָהּ שֶׁל תּוֹרָה, קָם כַּאֲרִי.

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “ריש לקיש — גבור וחכם” על ר' יוחנן וריש לקיש? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “עַכְשָׁו רֵישׁ לָקִישׁ הוּא תַּלְמִיד חָכָם עָצוּם שֶׁיּוֹשֵׁב קָבוּעַ בְּבֵית הַמִּדְרָשׁ הַמּוּגָן.”.
ג. אילו ר' יוחנן וריש לקיש לא היה פועל כפי שמתואר ב־“הַשּׁוֹדְדִים הִסְתּוֹבְבוּ, אָחֲזוּ בַּנֶּשֶׁק שֶׁלָּהֶם, אַךְ כְּשֶׁרָאוּ אֶת הָאִישׁ הָעוֹמֵד מוּלָם, הֵם קָפְאוּ בִּמְקוֹמָם.” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “בְּרֶגַע אֶחָד שֶׁל תְּשׁוּבָה הוּא עָזַב הַכֹּל, הָפַךְ אֶת כּוֹחוֹ לְהַתְמָדָה בַּתּוֹרָה, וְהִשְׁתַּמֵּשׁ בַּגְּבוּרָה שֶׁלּוֹ…” מחזק את המסר של הסיפור?
