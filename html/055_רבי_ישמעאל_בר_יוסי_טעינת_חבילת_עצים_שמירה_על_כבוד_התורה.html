טעינת חבילת עצים — שמירה על כבוד התורה

דף עבודה בעקבות רבי ישמעאל ב"ר יוסי
שם: כיתה:

המעשה — הסיפור המלא

להלן מערך חינוכי, רגשי ומרתק, המבוסס על הסיפור התלמודי על רבי ישמעאל ברבי יוסי וחבילת העצים (בבא מציעא ל ע"ב). הסיפור עובד והותאם במיוחד לבנים בכיתות ב'-ד' בחינוך הדתי-לאומי, מתוך הקפדה על כבוד התורה, עקרונות הסטוריטלינג וההנחיות שביקשת.

הסיפור מנוקד במלואו כנדרש.

פְּתִיחָה: חֹם שֶׁל מִדְבָּר וְנֹד שֶׁל זֵעָה

יְלָדִים, עִצְמוּ רֶגַע אֶת הָעֵינַיִם. דַּמְיְנוּ עַצְמְכֶם הוֹלְכִים בְּדֶרֶךְ עָפָר רְחָבָה בְּאֶרֶץ יִשְׂרָאֵל, לִפְנֵי כְּאַלְפַּיִם שָׁנָה. הַשֶּׁמֶשׁ שֶׁל שְׁעַת הַצָּהֳרַיִם קוֹפַחַת וְלוֹהֶטֶת כְּמוֹ אֵשׁ, עֲנַן אָבָק מִתְרוֹמֵם עִם כָּל צַעַד שֶׁלָּכֶם, וּבָאֲוִיר עוֹמֵד רֵיחַ יָבֵשׁ שֶׁל קוֹצִים צְהֻבִּים וַאֲדָמָה חַמָּה.

בַּדֶּרֶךְ הַזּוֹ עָבַר תַּלְמִיד חָכָם עָצוּם, צַדִּיק יְסוֹד עוֹלָם – רַבִּי יִשְׁמָעֵאל בְּרַבִּי יוֹסֵי . הוּא פָּסַע לוֹ בְּנַחַת, לָבוּשׁ בְּבִגְדֵי תַּלְמִידֵי חֲכָמִים, נְקִיִּים וּמְכֻבָּדִים עַד לִמְאֹד. פָּנָיו הָיוּ מְאִירוֹת כְּמוֹ מַלְאָךְ, וְכָל מִי שֶׁרָאָה אוֹתוֹ מֵרָחוֹק, הִרְגִּישׁ מִיָּד רֶטֶט שֶׁל יִרְאַת שָׁמַיִם בַּלֵּב. רַבִּי יִשְׁמָעֵאל הָיָה שָׁקוּעַ כֻּלּוֹ בְּדִבְרֵי תּוֹרָה, מְמַלְמֵל פְּסוּקִים וַהֲלָכוֹת.

הַמִּפְגָּשׁ בַּדֶּרֶךְ

לְפֶתַע, הוּא שָׁמַע קוֹל נְשִׁימוֹת כְּבֵדוֹת. "הָא... הוּא... הָא..." . מִמּוּלוֹ עָמַד אִישׁ כַּפְרִי. הָאִישׁ הָיָה עָיֵף וּמֻזָּע. עַל שִׁכְמוֹ הוּא נָשָׂא חֲבִילַת עֵצִים עֲנָקִית – בּוּלֵי עֵץ רְחָבִים, מְלֵאִים בְּשָׂרָף דָּבִיק, בְּחֹל וּבְעָלִים יְבֵשִׁים שֶׁשָּׂרְטוּ אֶת עוֹרוֹ. הָאִישׁ כְּבָר לֹא הָיָה מְסֻגָּל לָשֵׂאת אֶת הַמַּשָּׂא הַכָּבֵד.

בּוּם! קוֹל נְפִילָה חָזָק הִדְהֵד בַּשָּׂדֶה.

הָאִישׁ זָרַק אֶת חֲבִילַת הָעֵצִים עַל חֲצַץ הַדֶּרֶךְ כְּדֵי לָנוּחַ מְעַט. הוּא נָשַׁם בִּכְבֵדוּת, נִגֵּב אֶת אֶגְלֵי הַזֵּעָה הַגְּדוֹלִים מִמִּצְחוֹ בְּיָד מְלֻכְלֶכֶת, וְהִבִּיט יָמִינָה וּשְׂמֹאלָה כְּדֵי לִמְצֹא מִישֶׁהוּ שֶׁיַּעֲזֹר לוֹ.

כְּשֶׁרָאָה אֶת רַבִּי יִשְׁמָעֵאל מִתְקָרֵב, קָרָא אֵלָיו בְּקוֹל עָבֶה וְצָרוּד: "הֵי! כְּבוֹד הָרַב! בּוֹא רֶגַע הֵנָּה! עֲשֵׂה מִצְוָה, תַּעֲזֹר לִי לְהָרִים אֶת חֲבִילַת הָעֵצִים הַזּוֹ הַחְזֵרָה עַל הַגַּב שֶׁלִּי. נָחְתִּי, וְעַכְשָׁו קָשֶׁה לִי לְהָרִים אוֹתָהּ לְבַד מֵהָרִצְפָּה."

הַהִתְלַבְּטוּת – כְּבוֹד הַתּוֹרָה אוֹ מִצְוָה?

יְלָדִים, רַבִּי יִשְׁמָעֵאל נֶעֱצַר. הַתּוֹרָה הֲרֵי מְצַוָּה עָלֵינוּ: "עָזֹב תַּעֲזֹב עִמּוֹ" – יֵשׁ מִצְוָה שֶׁנִּקְרֵאת 'מִצְוַת טְעִינָה', לַעֲזֹר לְאָדָם לְהַעֲמִיס אֶת הַמַּשָּׂא שֶׁלּוֹ. אֲבָל... הַהֲלָכָה הַקְּדוֹשָׁה קוֹבַעַת גַּם כְּלָל חָשׁוּב מְאֹד: "זָקֵן (תַּלְמִיד חָכָם) וְאֵינָהּ לְפִי כְּבוֹדוֹ – פָּטוּר" .

רַבִּי יִשְׁמָעֵאל הִבִּיט עַל חֲבִילַת הָעֵצִים. הִיא הָיְתָה מְטֻנֶּפֶת בְּבֹץ מִן הַבִּצָּה, רֵיחַ שֶׁל רָקָב וְעִפּוּשׁ עָלָה מִמֶּנָּה. אִם הוּא יִתְכּוֹפֵף, יְחַבֵּק אֶת הָעֵצִים הַמְּלֻכְלָכִים וְיֵאָבֵק אִתָּם עַל אֵם הַדֶּרֶךְ, הַבְּגָדִים שֶׁלּוֹ יִתְלַכְלְכוּ. אֲנָשִׁים יַעַבְרוּ בַּדֶּרֶךְ וְיֹאמְרוּ: "תִּרְאוּ אֶת הַצַּדִּיק הַגָּדוֹל, גְּדוֹל הַדּוֹר שֶׁלָּנוּ! הוּא נִרְאֶה עַכְשָׁו כְּמוֹ פּוֹעֵל שָׁחוֹר בַּשּׁוּק!".

זֶה לֹא בִּגְלַל שֶׁרַבִּי יִשְׁמָעֵאל הָיָה גַּאַוְתָן חָלִילָה! צַדִּיקִים הֵם הָאֲנָשִׁים הֲכִי עֲנָוִים בָּעוֹלָם. אֲבָל כְּשֶׁמְּבַזִּים תַּלְמִיד חָכָם – מְבַזִּים אֶת הַתּוֹרָה! לֹא אוֹתוֹ פְּרָטִית. סֵפֶר תּוֹרָה אָסוּר שֶׁיִּתְלַכְלֵךְ בַּבֹּץ. הַתּוֹרָה צְרִיכָה לִהְיוֹת יָפָה וּמְכֻבָּדֶת בְּעֵינֵי כֻּלָּם.

הַפִּתְרוֹן הַיָּקָר

הִבִּיט רַבִּי יִשְׁמָעֵאל עַל הָאִישׁ וְאָמַר בְּקוֹלוֹ הַנָּעִים: "אָחִי, כַּמָּה שָׁוָה חֲבִילַת הָעֵצִים שֶׁלְּךָ?"

הָאִישׁ בָּהָה בָּרַב. "כַּמָּה הִיא שָׁוָה? אֶמּ... חֲצִי זוּז!" (חֲצִי זוּז הָיָה סְכוּם נִכְבָּד לְפוֹעֵל פָּשׁוּט).

רַבִּי יִשְׁמָעֵאל שָׁלַח יָד לְכִיסוֹ, הוֹצִיא מִשָּׁם מַטְבֵּעַ כֶּסֶף נוֹצֵץ. שְׁקִשׁוּק הַמַּטְבֵּעַ נִשְׁמַע בָּאֲוִיר כַּאֲשֶׁר הִנִּיחַ אוֹתוֹ בְּיָדוֹ שֶׁל מוֹכֵר הָעֵצִים. הָאִישׁ קָפַץ מִשִּׂמְחָה! אֵיזֶה רֶוַח עָצוּם, בְּלִי לָלֶכֶת לַשּׁוּק עַד לְשָׁם!

"קַח אֶת הַכֶּסֶף," אָמַר הַצַּדִּיק, "כָּעֵת הָעֵצִים הֵם שֶׁלִּי. אֲבָל... הֲרֵינִי מַפְקִיר אוֹתָם! (הוֹפֵךְ אוֹתָם לְשֶׁל הֶפְקֵר, כָּל מִי שֶׁרוֹצֶה יָכוֹל לָקַחַת)."

בְּכָךְ, רַבִּי יִשְׁמָעֵאל פָּטַר אֶת עַצְמוֹ מִלַּעֲזֹר לוֹ (כִּי הָעֵצִים כְּבָר לֹא שֶׁל הָאִישׁ), וְגַם מָנַע אֶת בִּזּוּי הַתּוֹרָה. הוּא הָיָה מוּכָן לְשַׁלֵּם כֶּסֶף מִכִּיסוֹ הַפְּרָטִי, רַק כְּדֵי שֶׁכְּבוֹד הַתּוֹרָה לֹא יִפָּגַע.

רַבִּי יִשְׁמָעֵאל הִסְתּוֹבֵב וְהֵחֵל לָלֶכֶת.

הַהִתְחַכְּמוּת שֶׁל הָאִישׁ

אֲבָל תַּחְשְׁבוּ עַל הָאִישׁ מִן הַשּׁוּק... הָאִישׁ הַזֶּה הָיָה חַמְדָן. הוּא הִבִּיט בְּמַטְבֵּעַ הַכֶּסֶף, וְאָז בַּעֲרֵמַת הָעֵצִים הַמֻּפְקֶרֶת עַל הָרִצְפָּה.

"רֶגַע אֶחָד..." אָמַר בְּלִבּוֹ, "הָרַב הִפְקִיר אֶת הָעֵצִים? כְּלוֹמַר הֵם כָּעֵת שֶׁל אִישׁ מַעֲשֶׂה אֵין לָהֶם בְּעָלִים? מְצֻיָּן!"

הוּא קָפַץ קָדִימָה, נָגַע בָּעֵצִים וְאָמַר בְּקוֹל: "אֲנִי זוֹכֶה בָּהֶם מִן הַהֶפְקֵר! הָעֵצִים שׁוּב שֶׁלִּי!"

וְאָז הוּא רָץ אַחֲרֵי רַבִּי יִשְׁמָעֵאל וְצָעַק לְעֶבְרוֹ: "הֵי! כְּבוֹד הָרַב! הָעֵצִים כָּעֵת שֶׁלִּי! בַּקָּשָׁה מִמְּךָ, בּוֹא לַעֲזֹר לִי לִטְעֹן אוֹתָם עַל הַכְּתֵפַיִם!".

יְלָדִים, אֵיךְ אַתֶּם הֱיִיתֶם מַרְגִּישִׁים? בְּוַדַּאי הֱיִיתֶם כּוֹעֲסִים מְאֹד. "חוּצְפָּנִי מְנֻוָּל! אֲנִי נָתַתִּי לְךָ כֶּסֶף בְּמַתָּנָה כְּדֵי לְהַצִּיל אֶת כְּבוֹד הַתּוֹרָה, וְאַתָּה עוֹשֶׂה עָלַי תַּרְגִּיל?!"

הַנִּצָּחוֹן – שְׁמִירַת הַכָּבוֹד בְּאַהֲבָה

אֲבָל רַבִּי יִשְׁמָעֵאל בְּרַבִּי יוֹסֵי הָיָה צַדִּיק. הוּא לֹא עָנָה בְּכַעַס, הוּא לֹא הֵרִים קוֹל. הוּא יָדַע שֶׁהַמַּטָּרָה שֶׁלּוֹ הִיא לִשְׁמֹר עַל כְּבוֹד הַתּוֹרָה, וְכַעַס לֹא יוֹסִיף לַכָּבוֹד שֶׁל הַתּוֹרָה.

הוּא נֶעֱצַר. נָשַׁם עָמֹק. וּשְׁאֵלַת חֲכָמִים בְּפִיו: "תַּגִּיד לִי, אָחִי הַיָּקָר, כַּמָּה הָעֵצִים הָאֵלּוּ שָׁוִים עַכְשָׁו לְדַעְתְּךָ?"

הָאִישׁ חִיֵּךְ חִיּוּךְ רָחָב, הֶרְאָה שִׁנַּיִם לְבָנוֹת, וְאָמַר: "עוֹד חֲצִי זוּז!".

הִכְנִיס רַבִּי יִשְׁמָעֵאל שׁוּב יָד לְכִיסוֹ, הֵצִיץ בְּעֵינָיו הַפְּשׁוּטוֹת שֶׁל הָאִישׁ, וְהִנִּיחַ כֶּסֶף נוֹצֵץ שֵׁנִי בְּכַף יָדוֹ!. "קַח עַכְשָׁו, הֵם שׁוּב שֶׁלִּי."

וְכָעֵת, כְּדֵי שֶׁהָאִישׁ לֹא יוּכַל לַעֲשׂוֹת צְחוֹק אֶחָד שֵׁנִי, אָמַר רַבִּי יִשְׁמָעֵאל דָּבָר חָכָם לְפִי הַהֲלָכָה:

"הֲרֵינִי מַפְקִיר אוֹתָם לְכָל בְּנֵי הָעוֹלָם – חוּץ מִמְּךָ! לְךָ אֲנִי לֹא מַרְשֶׁה לִזְכּוֹת בָּהֶם שׁוּב!".

א. מה מלמד המעשה “טעינת חבילת עצים — שמירה על כבוד התורה” על רבי ישמעאל ב"ר יוסי? הביאו שתי ראיות מדויקות מן הסיפור.
