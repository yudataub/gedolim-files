🔹 שָׁלָב 1: בֵּרוּר מַקְדִּים וְנִתּוּחַ (לַמּוֹרֶה/לַמְסַפֵּר)

🔹 שָׁלָב 2: הַסִּפּוּר הַמָּלֵא – "הַסּוֹד מֵאֹהֶל הַכֹּהֵן הַגָּדוֹל"

פְּתִיחָה: הַשְּׁאֵלָה בְּבֵית הַמִּדְרָשׁ יְלָדִים, בְּבֵית הַמִּדְרָשׁ הַגָּדוֹל בִּירוּשָׁלַיִם, יָשַׁב נְשִׂיא יִשְׂרָאֵל, הִלֵּל הַזָּקֵן. סְבִיבוֹ יָשְׁבוּ שְׁמוֹנִים תַּלְמִידִים קְדוֹשִׁים, שֶׁשָּׁתוּ בְּצִמָּאוֹן כָּל מִלָּה וּמִלָּה שֶׁיָּצְאָה מִפִּיו. בְּאוֹתוֹ יוֹם, לִמֵּד הִלֵּל אֶת הַמִּשְׁנָה הַמְפֻרְסֶמֶת שֶׁלּוֹ. הוּא הִבִּיט בְּתַלְמִידָיו בְּעֵינַיִם מְאִירוֹת וְאָמַר: "הֱוֵי מִתַּלְמִידָיו שֶׁל אַהֲרֹן: אוֹהֵב שָׁלוֹם וְרוֹדֵף שָׁלוֹם, אוֹהֵב אֶת הַבְּרִיּוֹת וּמְקָרְבָן לַתּוֹרָה".

קָם אֶחָד הַתַּלְמִידִים הַצְּעִירִים וְשָׁאַל בְּהִסּוּס: "רַבֵּנוּ הִלֵּל, תּוֹרָתְךָ אֱמֶת, אַךְ קָשֶׁה לִי לְהָבִין. אַהֲרוֹן הַכֹּהֵן חַי בַּמִּדְבָּר כִּמְעַט אֶלֶף שָׁנִים לִפְנֵי שֶׁנּוֹלַדְנוּ! לְאַהֲרוֹן אֵין כַּיּוֹם בֵּית סֵפֶר אוֹ יְשִׁיבָה. אֵיךְ אֲנַחְנוּ יְכוֹלִים לִהְיוֹת הַיּוֹם 'תַּלְמִידִים שֶׁל אַהֲרוֹן'? הֲרֵי הוּא נִפְטַר לְעוֹלָמוֹ!".

חִיֵּךְ הִלֵּל הַזָּקֵן, זְקָנוֹ הַלָּבָן נָח עַל חָזֵהוּ, וְקוֹלוֹ הַנָּעִים הִדְהֵד בַּחֶדֶר: "תַּלְמִיד אֲמִתִּי הוּא מִי שֶׁלּוֹמֵד מֵהַמַּעֲשִׂים שֶׁל רַבּוֹ. בּוֹאוּ אֲסַפֵּר לָכֶם כֵּיצַד הָיָה אַהֲרוֹן הַכֹּהֵן מַשְׁכִּין שָׁלוֹם בְּמַחֲנֵה יִשְׂרָאֵל, וְכָךְ תָּבִינוּ אֵיךְ נוּכַל לִהְיוֹת תַּלְמִידָיו אֲפִלּוּ הַיּוֹם".

הַמְּרִיבָה בַּמִּדְבָּר וְכָךְ סִפֵּר הִלֵּל: דַּמְיְנוּ אֶת מַחֲנֵה יִשְׂרָאֵל בַּמִּדְבָּר. אֹהָלִים רַבִּים מְסֻדָּרִים לִשְׁבָטִים. בִּשְׁנֵי אֹהָלִים סְמוּכִים גָּרוּ שְׁנֵי חֲבֵרִים טוֹבִים, נִקְרָא לָהֶם שִׁמְעוֹן וְלֵוִי. יוֹם אֶחָד, הַיֵּצֶר הָרַע נִכְנַס בֵּינֵיהֶם. הֵם הִתְחִילוּ לְהִתְוַכֵּחַ עַל עִנְיָן כַּסְפִּי (אוּלַי עַל עֵסֶק שֶׁעָשׂוּ יַחַד). הַוִּכּוּחַ הָפַךְ לִצְעָקוֹת. הַפָּנִים שֶׁלָּהֶם אָדְמוּ מִכַּעַס. שִׁמְעוֹן הֵרִים אֶת קוֹלוֹ, וְלֵוִי צָעַק בַּחֲזָרָה: "אֲנִי לֹא רוֹצֶה לִרְאוֹת אֶת הַפָּנִים שֶׁלְּךָ לְעוֹלָם!" הֵם חָזְרוּ אִישׁ אִישׁ לְאָהֳלוֹ, כּוֹעֲסִים וּפְגוּעִים. לֹא דִּבְּרוּ זֶה עִם זֶה. שִׂנְאָה מָלְאָה אֶת הַלֵּב שֶׁהָיָה פַּעַם מָלֵא בְּאַהֲבָה.

וּמִי רָאָה זֹאת? אַהֲרוֹן, הַכֹּהֵן הַגָּדוֹל. הָאִישׁ שֶׁנּוֹשֵׂא אֶת חֹשֶׁן הַמִּשְׁפָּט עַל לִבּוֹ עִם כָּל שְׁמוֹת שִׁבְטֵי יִשְׂרָאֵל. כְּשֶׁאַהֲרוֹן רָאָה שְׁנֵי יְהוּדִים בְּמַחֲלֹקֶת, לִבּוֹ הַטָּהוֹר כָּאַב. הוּא יָדַע שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא עָצוּב כְּשֶׁהַבָּנִים שֶׁלּוֹ רָבִים.

אַהֲרוֹן יָכוֹל הָיָה לָשֶׁבֶת בְּאֹהֶל מוֹעֵד וּלְצַוּוֹת עֲלֵיהֶם: "בּוֹאוּ הֵנָּה, אֲנִי גּוֹזֵר עֲלֵיכֶם לַעֲשׂוֹת שָׁלוֹם!". אַךְ הוּא לֹא עָשָׂה זֹאת.

הַבִּקּוּר בָּאֹהֶל שֶׁל שִׁמְעוֹן בְּאִישׁוֹן לַיְלָה, כְּשֶׁהַמַּחֲנֶה הָיָה חָשׁוּךְ, קָם אַהֲרוֹן מִמִּטָּתוֹ. הוּא הָלַךְ בֶּחוֹל הַמִּדְבָּר, וְנִכְנַס בְּשֶׁקֶט לָאֹהֶל שֶׁל שִׁמְעוֹן. שִׁמְעוֹן קָם בְּבֶהָלָה. "אַהֲרוֹן הַכֹּהֵן הַגָּדוֹל בִּכְבוֹדוֹ וּבְעַצְמוֹ! מָה מֵבִיא אֶת כְּבוֹדוֹ אֵלַי?" אַהֲרוֹן יָשַׁב לְיָדוֹ, שָׂם אֶת יָדוֹ הַחַמָּה עַל הַכָּתֵף שֶׁל שִׁמְעוֹן וְאָמַר בְּקוֹל שָׁקֵט וְנִרְגָּשׁ: "שִׁמְעוֹן, בָּאתִי מֵהָאֹהֶל שֶׁל לֵוִי". שִׁמְעוֹן תֵּכֶף כָּעַס: "אַל תַּזְכִּיר לִי אוֹתוֹ! הוּא אִישׁ רַע!" "אַל תֹּאמַר כָּךְ," לָחַשׁ אַהֲרוֹן. "הָיִיתִי אֶצְלוֹ עַכְשָׁו... רָאִיתִי אוֹתוֹ יוֹשֵׁב בַּפִּנָּה וּבוֹכֶה בְּכִי תַּמְרוּרִים. הוּא קוֹרֵעַ אֶת הַשְּׂעָרוֹת שֶׁלּוֹ מֵרֹב צַעַר, וּמַכֶּה עַל הַלֵּב. שָׁאַלְתִּי אוֹתוֹ 'לָמָּה אַתָּה בוֹכֶה?' וְהוּא אָמַר לִי: 'אוֹי לִי שֶׁפָּגַעְתִּי בְּשִׁמְעוֹן חֲבֵרִי הַיָּקָר! אֵיךְ אָרִים עַכְשָׁו אֶת עֵינַי? בְּוַדַּאי שִׁמְעוֹן שׂוֹנֵא אוֹתִי עַכְשָׁו! אֲנִי מִתְבַּיֵּשׁ לָלֶכֶת לְבַקֵּשׁ סְלִיחָה...'" כְּשֶׁשִּׁמְעוֹן שָׁמַע שֶׁחֲבֵרוֹ לֵוִי כָּל כָּךְ עָצוּב וּמִתְחָרֵט, הַכַּעַס שֶׁבְּלִבּוֹ נָמַס מִיָּד כְּמוֹ קֶרַח בַּשֶּׁמֶשׁ. הַרַחֲמִים גָּבְרוּ עָלָיו, וְהוּא אָמַר: "הוּא מוּכָן לְבַקֵּשׁ סְלִיחָה? מִסְכֵּן, בְּוַדַּאי אֲנִי מוֹחֵל לוֹ!"

הַבִּקּוּר בָּאֹהֶל שֶׁל לֵוִי יָצָא אַהֲרוֹן מֵהָאֹהֶל שֶׁל שִׁמְעוֹן, אֲבָל לֹא הָלַךְ לִישׁוֹן. הוּא פָּנָה יָשָׁר לָאֹהֶל שֶׁל לֵוִי. "לֵוִי, בְּנִי," אָמַר לוֹ אַהֲרוֹן הַכֹּהֵן הַגָּדוֹל. "מָה הָעִנְיָן, רַבִּי?" שָׁאַל לֵוִי בְּתִמָּהוֹן. "בָּאתִי מֵהָאֹהֶל שֶׁל חֲבֵרְךָ שִׁמְעוֹן. רָאִיתִי אוֹתוֹ מְמָרֵר בְּבֶכִי, דְּמָעוֹת נוֹפְלוֹת עַל רִצְפַּת הָאֹהֶל! הוּא אָמַר לִי: 'כֵּיצַד פָּגַעְתִּי בְּלֵוִי צַדִּיק שֶׁכָּזֶה! כֵּיצַד אֶשְׁלַח אוֹתִי מֵהָאֹהֶל? אֲנִי רוֹצֶה לְהַשְׁלִים אִתּוֹ, אֲבָל אֲנִי פּוֹחֵד שֶׁהוּא יִטְרֹק לִי אֶת הַדֶּלֶת מֵרֹב כַּעַס'." כְּשֶׁלֵוִי שָׁמַע שֶׁשִּׁמְעוֹן אָהוּב כָּל כָּךְ מִתְחָרֵט, דְּמָעוֹת עָמְדוּ בְּעֵינָיו שֶׁלּוֹ. "מָה פִּתְאוֹם אֶטְרֹק דֶּלֶת? אֲנִי זֶה שֶׁאַשְׁלִים אִתּוֹ!" קָרָא.

הַפְּגִישָׁה שֶׁחִבְּרָה אֶת הַלְּבָבוֹת לְמָחֳרָת בַּבֹּקֶר, יָצְאוּ שִׁמְעוֹן וְלֵוִי לַהֲלִיכָה בַּמַּחֲנֶה. כָּל אֶחָד חִפֵּשׂ אֶת חֲבֵרוֹ. וְהִנֵּה – הֵם רוֹאִים זֶה אֶת זֶה מֵרָחוֹק בְּאֶמְצַע הַשְּׁבִיל. לוּ זֶה הָיָה אֶתְמוֹל, כָּל אֶחָד הָיָה סוֹבֵב אֶת הָרֹאשׁ לְצַד שֵׁנִי הַצִּדָּה מֵרֹב כַּעַס. אֲבָל עַכְשָׁו... עַכְשָׁו כָּל אֶחָד בָּטוּחַ שֶׁהַשֵּׁנִי כָּעוֹצֵב וּבוֹכֶה וּמִתְחָרֵט! הֵם רָצוּ אֶחָד לִקְרַאת הַשֵּׁנִי, פָּרְשׂוּ אֶת הַיָּדַיִם הַצִּדָּה, וּפָשׁוּט חִבְּקוּ וְנִשְּׁקוּ זֶה אֶת זֶה בְּשִׂמְחָה עֲצוּמָה! "סְלַח לִי אָחִי!" "לֹא, אַתָּה סְלַח לִי, אֲנִי הָיִיתִי לֹא בְּסֵדֶר!" וּמֵרָחוֹק, עָמַד אַהֲרוֹן הַכֹּהֵן הַקָּדוֹשׁ, הִבִּיט בָּהֶם שׁוֹלְבִים זְרוֹעוֹת וְחִיֵּךְ לַשָּׁמַיִם. כִּי הָיָה שָׁלוֹם בְּיִשְׂרָאֵל, וְכַאֲשֶׁר יֵשׁ שָׁלוֹם לְמַטָּה – הַשְּׁכִינָה יוֹרֶדֶת וְנָחָה לְמַעְלָה!

סִכּוּם הַסּוֹד – רוֹדֵף שָׁלוֹם סִיֵּם הִלֵּל הַזָּקֵן אֶת הַסִּפּוּר, הִבִּיט עַל פְנֵי תַּלְמִידָיו, וְהִסְבִּיר: "עַכְשָׁו, תַּלְמִידַי הַיְקָרִים, אַתֶּם מְבִינִים כֵּיצַד אֶפְשָׁר לִהְיוֹת תַּלְמִידִים שֶׁל אַהֲרוֹן? הַמִּשְׁנָה לֹא אוֹמֶרֶת רַק 'אוֹהֵב שָׁלוֹם'. לֶאֱהֹב שָׁלוֹם זֶה קַל – כֻּלָּם רוֹצִים שֶׁיִּהְיֶה שֶׁקֶט וְשָׁלוֹם כְּשֶׁהֵם יוֹשְׁבִים בַּכֻּרְסָא. אֲבָל אַהֲרוֹן הָיָה 'רוֹדֵף שָׁלוֹם'! הוּא הָיָה קָם בַּלַּיְלָה, הוֹלֵךְ, מֵזִיעַ, וּמִתְאַמֵּץ, כְּדֵי שֶׁשְּׁנֵי יְהוּדִים יַחְזְרוּ לֶאֱהֹב זֶה אֶת זֶה. מִי שֶׁעוֹשֶׂה כָּךְ כַּיּוֹם – שֶׁמַּשְׁלִים בֵּין שְׁנֵי יְלָדִים שֶׁרָבוּ בַּכִּתָּה אוֹ בַּמִּשְׂחָק בַּהַפְסָקָה – הוּא וַדַּאי נֶחְשָׁב כְּתַלְמִידוֹ הָאֲהוּב שֶׁל אַהֲרוֹן הַכֹּהֵן!".
