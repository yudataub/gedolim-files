המציבה שהתפרקה — גל אבנים על קברו

דף עבודה בעקבות רבי פנחס בן יאיר
שם: כיתה:

המעשה — הסיפור המלא

להלן עיבוד חינוכי, רגשי ומרתק לסיפורו המופלא של רבי פנחס בן יאיר והמצבה שהתפרקה , המעובד במיוחד לבנים בכיתות ב'-ד' (חינוך דתי-לאומי). הסיפור שם דגש על מידת הענווה העצומה שלו, על "שונא מתנות יחיה", ועל כך שגדולתו של אדם אינה נמדדת באבנים נוצצות, אלא במעשיו הטובים.

הסיפור מנוקד במלואו כנדרש.

פְּתִיחָה: הָאִישׁ שֶׁלֹּא לָקַח דָּבָר

יְלָדִים, נַסּוּ לְדַמְיֵן אֶת הָרֵיחַ הַמָּתוֹק שֶׁל לֶחֶם טָרִי שֶׁיּוֹצֵא מֵהַתַּנּוּר שֶׁל הַשָּׁכֵן. אוֹ אֶת הַתְּחוּשָׁה כְּשֶׁחָבֵר טוֹב מַצִּיעַ לָכֶם לִטְעֹם מֵהַמַּמְתָּק הַחָדָשׁ שֶׁהוּא קָנָה. זֶה מְפַתֶּה מְאֹד, נָכוֹן? כִּמְעַט כֻּלָּם שְׂמֵחִים לְקַבֵּל מַתָּנוֹת אוֹ כִּבּוּדִים מֵאֲחֵרִים.

אַךְ בִּימֵי חֲכָמֵינוּ זִכְרוֹנָם לִבְרָכָה, חַי תַּנָּא קָדוֹשׁ אֶחָד, צַדִּיק יְסוֹד עוֹלָם, שֶׁשְּׁמוֹ רַבִּי פִּינְחָס בֶּן יָאִיר . לְרַבִּי פִּינְחָס הָיָה כְּלַל בַּרְזֶל בַּחַיִּים. מִיּוֹם שֶׁעָמַד עַל דַּעְתּוֹ, הוּא קִבֵּל עַל עַצְמוֹ לְקַיֵּם אֶת הַפָּסוּק: "שׂוֹנֵא מַתָּנוֹת – יִחְיֶה".

הוּא אַף פַּעַם, מֵעוֹלָם, לֹא אָכַל מִשֻּׁלְחָנוֹ שֶׁל אָדָם אַחֵר! לֹא מִשֶּׁל עֲשִׁירִים וְלֹא מִשֶּׁל עֲנִיִּים. הוּא הָיָה עָנָיו בְּתַכְלִית, לָבַשׁ בְּגָדִים פְּשׁוּטִים מְאֹד עֲשׂוּיִים מִפִּשְׁתָּן, יָשַׁב בְּבֵיתוֹ הַקָּטָן, אָכַל רַק מֵהַמְּעַט שֶׁהִרְוִיחַ בְּעֶשֶׂר אֶצְבָּעוֹתָיו הַקְּדוֹשׁוֹת, וְעָסַק יוֹמָם וָלַיְלָה בְּתוֹרַת ה'. הוּא הָיָה אוֹמֵר: "אֲנִי לֹא רוֹצֶה לֵהָנוֹת מִשֶּׁל אֲחֵרִים. אוּלַי חָסֵר לָהֶם? אוּלַי הֵם נוֹתְנִים לִי מִתּוֹךְ כָּבוֹד, אֲבָל בַּלֵּב קָשֶׁה לָהֶם לָתֵת?".

לָכֵן, רַבִּי פִּינְחָס חַי בִּקְדֻשָּׁה עֲצוּמָה, נָקִי וְטָהוֹר, לְלֹא טִפָּה אַחַת שֶׁל טוֹבַת הֲנָאָה מִשּׁוּם אָדָם בָּעוֹלָם. אֲפִלּוּ כַּאֲשֶׁר רַבִּי יְהוּדָה הַנָּשִׂיא – עֲשִׁיר יִשְׂרָאֵל! – הִתְחַנֵּן אֵלָיו שֶׁיִּסְעַד עַל שֻׁלְחָנוֹ, רַבִּי פִּינְחָס עָנָה בְּנֹעַם וּבְחִיּוּךְ: "לֹא, תּוֹדָה".

יוֹם הַפְּרֵדָה בָּעִיר הַבּוֹכָה

אֲבָל כָּל אָדָם, אֲפִלּוּ צַדִּיק כָּל כָּךְ גָּדוֹל, מַגִּיעַ זְמַנּוֹ לַעֲלוֹת לַשָּׁמַיִם.

יוֹם אֶחָד, הַשֶּׁמֶשׁ לֹא הֵאִירָה בְּאוֹתָהּ עָצְמָה. רוּחַ שְׁקֵטָה נָשְׁבָה בִּרְחוֹבוֹת הָעִיר, וְקוֹל בְּכִי מַר נִשְׁמַע מִכָּל הַבָּתִּים. "נִצְּחוּ אֶרְאֶלִּים אֶת הַמְּצוּקִים". רַבִּי פִּינְחָס בֶּן יָאִיר נִפְטַר לְבֵית עוֹלָמוֹ, וְנִשְׁמָתוֹ הַטְּהוֹרָה עָלְתָה אֶל מִתַּחַת לְכִסֵּא הַכָּבוֹד.

כָּל אַנְשֵׁי הָעִיר יָצְאוּ לְלַוּוֹת אוֹתוֹ. מֵאוֹת וַאֲלָפִים שֶׁל יְהוּדִים בָּכוּ וְקָרְעוּ אֶת בִּגְדֵיהֶם. כְּשֶׁקָּבְרוּ אֶת הַצַּדִּיק בָּאֲדָמָה, נֶעֶמְדוּ הָעֲשִׁירִים וַחֲשׁוּבֵי הָעִיר, שָׂמוּ יָד עַל לִבָּם, וְאָמְרוּ: "רַבִּי פִּינְחָס לֹא הִסְכִּים לְקַבֵּל מֵאִתָּנוּ כְּלוּם כְּשֶׁהָיָה בַּחַיִּים. תָּמִיד הוּא הָיָה אִישׁ פָּשׁוּט וְצָנוּעַ. אֲבָל עַכְשָׁו? עַכְשָׁו הָאִישׁ הַגָּדוֹל הַזֶּה הוּא מֶלֶךְ! הוּא שַׂר הַתּוֹרָה! אֲנַחְנוּ נִבְנֶה לוֹ אֶת הַמַּצֵּבָה הֲכִי מְפֹאֶרֶת שֶׁנִּרְאֲתָה אֵי פַּעַם בְּאֶרֶץ יִשְׂרָאֵל! נַרְאֶה לְכֻלָּם כַּמָּה אָהַבְנוּ אוֹתוֹ!".

קְלוֹת הַפַּטִּישִׁים וְהַשַּׁיִשׁ הַלָּבָן

וְכָךְ, מִיָּד לְאַחַר שִׁבְעַת יְמֵי הָאֵבֶל, אָסְפוּ חֲשׁוּבֵי הָעִיר שַׂקִּים גְּדוֹלִים שֶׁל מַטְבְּעוֹת זָהָב. הֵם הֵבִיאוּ לְבֵית הֶעָלְמִין אֶת גְּדוֹלֵי הַבַּנָּאִים, חָצְבוּ בָּהָר אֲבָנִים עֲנָקִיּוֹת שֶׁל שַׁיִשׁ לָבָן וְנוֹצֵץ.

דּוּף! טְרַאח! טִיק-טַק! קוֹלוֹת שֶׁל מַכּוֹת פַּטִּישׁ וְאִזְמֵל (כְּלִי לַחְצִיבַת אֲבָנִים) נִשְׁמְעוּ עַל הַהַר בְּמֶשֶׁךְ שָׁבוּעוֹת. הֵם עִרְבְּבוּ מֶלֶט אָפֹר וְחָזָק שֶׁהֵרִיחַ כְּמוֹ רֶגֶב אֲדָמָה רָטֹב, הִנִּיחוּ אֶבֶן עַל אֶבֶן, וּבָנוּ מִבְנֶה מַרְשִׁים, מֵעֵין אַרְמוֹן קָטָן, סְבִיב קִבְרוֹ שֶׁל רַבִּי פִּינְחָס.

"תִּרְאוּ אֵיזֶה יֹפִי!" אָמַר אֶחָד מֵעֲשִׁירֵי הָעִיר כְּשֶׁהֵם סִיְּמוּ, מְנַגֵּב אֶת הַזֵּעָה מִמִּצְחוֹ. "הַשַּׁיִשׁ הַזֶּה נוֹצֵץ תַּחַת קַרְנֵי הַשֶּׁמֶשׁ. כָּל מִי שֶׁיַּעֲבֹר בַּדֶּרֶךְ יִרְאֶה אֶת הַמַּצֵּבָה הַזּוֹ וְיֵדַע: פֹּה שׁוֹכֵב צַדִּיק קָדוֹשׁ שֶׁאֲנַחְנוּ אָהַבְנוּ מְאֹד. נָתַנּוּ לוֹ מִכַּסְפֵּנוּ, מִשֶּׁלָּנוּ!". הֵם חָזְרוּ לְבָתֵּיהֶם שְׂמֵחִים וְגֵאִים, בְּלֵב גּוֹאֶה עַל שֶׁזָּכוּ לְכַבֵּד אֶת רַבָּם.

אֲבָל... הֵם שָׁכְחוּ דָּבָר אֶחָד חָשׁוּב. דָּבָר שֶׁרַבִּי פִּינְחָס זָעַק בְּכָל רְמַ"ח אֵבָרָיו בְּחַיָּיו.

הַלַּיְלָה שֶׁבּוֹ הָאֲבָנִים קָפְצוּ

הַלַּיְלָה יָרַד עַל הָהָר. חֹשֶׁךְ סָמִיךְ כִּסָּה אֶת בֵּית הֶעָלְמִין. דְּמָמָה.

לְפֶתַע, תַּחַת אוֹר הַיָּרֵחַ... קְרָאק... בּוּם! קוֹל נִפּוּץ אַדִּיר נִשְׁמַע. אֶבֶן שַׁיִשׁ אַחַת הִתְנַתְּקָה מִמְּקוֹמָהּ. וְאָז עוֹד אַחַת. וְעוֹד אַחַת! טְרָאח! טְרָאח! קְרָאשׁ! אֲבָנִים כְּבֵדוֹת בְּמִשְׁקָל שֶׁל מֵאוֹת קִילוֹגְרָמִים צָנְחוּ לָאֲדָמָה מֵעַצְמָן, כְּאִלּוּ מִישֶׁהוּ דּוֹחֵף אוֹתָן מִבִּפְנִים. הַמֶּלֶט הַחָזָק הִתְפּוֹרֵר לְאָבָק. עַד עֲלוֹת הַשַּׁחַר – מֵהַמַּצֵּבָה הַמְפֹאֶרֶת וְהַיְּקָרָה, לֹא נִשְׁאַר דָּבָר חָזָק. הַכֹּל שָׁכַב עַל הָרִצְפָּה מְפֹרָק!

בַּבֹּקֶר, קָמוּ חֲשׁוּבֵי הָעִיר וְעָלוּ לָהָר כְּדֵי לְהִתְפַּלֵּל אֵצֶל רַבָּם. כְּשֶׁרָאוּ אֶת הַהֶרֶס, עֵינֵיהֶם נִפְעֲרוּ מִתַּדְהֵמָה.

"מָה זֶה? מִי עָשָׂה אֶת זֶה?!" צָעַק אֶחָד מֵהֶם מִבַּעַד לִדְמָעוֹת הַכַּעַס. "הַאִם לִסְטִים (שׁוֹדְדִים) בָּאוּ בַּלַּיְלָה וְשָׁבְרוּ אֶת הַמַּצֵּבָה?! אוֹ שֶׁמָּא הָיְתָה פֹּה רְעִידַת אֲדָמָה גְּדוֹלָה?".

"לֹא מְשַׁנֶּה מָה קָרָה," עָנָה אַחֵר, קוֹמְצוֹ אֶת אֶגְרוֹפָיו. "אֲנַחְנוּ נִבְנֶה שׁוּב! וְהַפַּעַם – חָזָק יוֹתֵר, עָבֶה יוֹתֵר וְיָקָר יוֹתֵר!".

הַהִתְעַקְּשׁוּת וְהַסּוֹד הַגָּדוֹל

שׁוּב בָּאוּ הַפּוֹעֲלִים. הֵם הֵבִיאוּ אֲבָנִים חֲדָשׁוֹת, יָצְקוּ בַּרְזֶל סָבִיב, וּבָנוּ מַצֵּבָה נָאָה וַעֲצוּמָה אַף יוֹתֵר מֵהָרִאשׁוֹנָה.

אַךְ רְאוּ זֶה פֶּלֶא – בַּלַּיְלָה הַבָּא... הַמַּחֲזֶה חָזַר עַל עַצְמוֹ מְדֻיָּק! קְרָאשׁ! בּוּם! קוֹל קְרִיסָה, אָבָק אָפֹר וָסָמִיךְ הִתְעוֹפֵף בָּאֲוִיר, וּבַבֹּקֶר הַמַּצֵּבָה שׁוּב שָׁכְבָה שְׁבוּרָה לַחֲתִיכוֹת עַל הָאֲדָמָה הַלַּחָה.

הַפַּעַם, אַנְשֵׁי הָעִיר לֹא כָּעֲסוּ. הֵם פָּחֲדוּ. הֵם הֵבִינוּ שֶׁמַּשֶּׁהוּ שְׁמֵימִי, פִּלְאִי, קוֹרֶה כָּאן. הֵם עָמְדוּ סְבִיב אֲבָנֵי הַשַּׁיִשׁ הַשְּׁבוּרוֹת, שׁוֹתְקִים וּמֻבְהָלִים.

אָז, קָצַד זָקֵן חָכָם וַעֲנָו מִתּוֹךְ הַקָּהָל. הוּא הִבִּיט בָּאֲבָנִים הַיְקָרוֹת שֶׁנָּפְלוּ, לָטַף אֶת זְקָנוֹ הַלָּבָן, וְאָמַר בְּלַחַשׁ שֶׁנִּשְׁמַע בְּכָל בֵּית הֶעָלְמִין:

"אַחַי... עִצְרוּ. רַדְנוּ לַשָּׁוְא. הַאִם שְׁכַחְתֶּם מִי שׁוֹכֵב כָּאן? הַאִם שְׁכַחְתֶּם מִי הָיָה רַבִּי פִּינְחָס בֶּן יָאִיר?".

כֻּלָּם הֵרִימוּ אֶת עֵינֵיהֶם אֵלָיו.

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “המציבה שהתפרקה — גל אבנים על קברו” על רבי פנחס בן יאיר? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “הֵם הֵבִיאוּ לְבֵית הֶעָלְמִין אֶת גְּדוֹלֵי הַבַּנָּאִים, חָצְבוּ בָּהָר אֲבָנִים עֲנָקִיּוֹת שֶׁל שַׁיִשׁ לָבָן וְנוֹצֵץ.”.
ג. אילו רבי פנחס בן יאיר לא היה פועל כפי שמתואר ב־“קוֹל קְרִיסָה, אָבָק אָפֹר וָסָמִיךְ הִתְעוֹפֵף בָּאֲוִיר, וּבַבֹּקֶר הַמַּצֵּבָה שׁוּב שָׁכְבָה שְׁבוּרָה לַחֲתִיכוֹת עַל…” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “בִּמְקוֹמוֹ, כָּל תַּלְמִיד הִתְכּוֹפֵף, בְּשֶּׁקֶט וּבִדְמָעוֹת שֶׁל אֱמֶת, אָסַף בְּיָדָיו אֶבֶן פְּשׁוּטָה וַחֲלָקָה, רֶגֶב…” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. להלן עיבוד חינוכי, רגשי ומרתק לסיפורו המופלא של רבי פנחס בן יאיר והמצבה…
2. הוּא הָיָה אוֹמֵר: "אֲנִי לֹא רוֹצֶה לֵהָנוֹת מִשֶּׁל אֲחֵרִים.
3. אֲנַחְנוּ נִבְנֶה לוֹ אֶת הַמַּצֵּבָה הֲכִי מְפֹאֶרֶת שֶׁנִּרְאֲתָה אֵי…
4. אֲבָנִים כְּבֵדוֹת בְּמִשְׁקָל שֶׁל מֵאוֹת קִילוֹגְרָמִים צָנְחוּ…
5. כָּל יָמָיו שֶׁל רַבִּי פִּינְחָס בָּעוֹלָם הַזֶּה," הִמְשִׁיךְ…
6. (זְכֹר: אֵין צִיּוּרֵי אֱלֹהוּת, יֵשׁ כִּפּוֹת וּלְבוּשׁ מָלֵא.

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “מֵאוֹת וַאֲלָפִים שֶׁל יְהוּדִים בָּכוּ וְקָרְעוּ אֶת בִּגְדֵיהֶם.”.
מכתב: כתבו לרבי פנחס בן יאיר בעקבות “בַּבֹּקֶר, קָמוּ חֲשׁוּבֵי הָעִיר וְעָלוּ לָהָר כְּדֵי לְהִתְפַּלֵּל אֵצֶל רַבָּם.”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “הוּא מֵעוֹלָם לֹא הִסְכִּים לֵהָנוֹת מִשֶּׁל אֲחֵרִים, מֵעוֹלָם לֹא רָצָה כָּבוֹד, יֹפִי נוֹצֵץ אוֹ כֶּסֶף שֶׁל זָרִים.”.
דיווח: כתבו ידיעה קצרה שכותרתה “המציבה שהתפרקה — גל אבנים על קברו”.

קוֹל קְרִיסָה, אָבָק אָפֹר וָסָמִיךְ הִתְעוֹפֵף בָּאֲוִיר, וּבַבֹּקֶר הַמַּצֵּבָה שׁוּב שָׁכְבָה…

7. חיבור לחיים מתוך המעשה

א. מה אפשר ללמוד מן הדרך שבה רבי פנחס בן יאיר פעל באירוע “הֵם חָזְרוּ לְבָתֵּיהֶם שְׂמֵחִים וְגֵאִים, בְּלֵב גּוֹאֶה עַל שֶׁזָּכוּ לְכַבֵּד אֶת רַבָּם.”?
