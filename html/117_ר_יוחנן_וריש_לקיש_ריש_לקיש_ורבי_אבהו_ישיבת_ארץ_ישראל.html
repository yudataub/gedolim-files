ריש לקיש ורבי אבהו — ישיבת ארץ ישראל

דף עבודה בעקבות ר' יוחנן וריש לקיש
שם: כיתה:

המעשה — הסיפור המלא

להלן עיבוד חינוכי, רגשי ומרתק על אהבת ארץ ישראל ומסירות הנפש עליה , המשלב את גדולי האמוראים (רֵישׁ לָקִישׁ וְרַבִּי אַבָּהוּ) ואת הסיפור המפורסם מהמדרש (ספרי, ראה) על התנאים רַבִּי אֶלְעָזָר בֶּן שַׁמּוּעַ וְרַבִּי יוֹחָנָן הַסַּנְדְּלָר. הסיפור נתפר במיוחד עבור בנים בכיתות ב'-ד' בחינוך הדתי-לאומי, ומדגיש את ערך "ישיבת ארץ ישראל" מתוך קדושה והערצה לתורה.

הסיפור מנוקד במלואו כנדרש.

פְּתִיחָה: סוֹדָהּ שֶׁל יְשִׁיבַת אֶרֶץ יִשְׂרָאֵל

יְלָדִים, עִצְמוּ אֶת הָעֵינַיִם וְדַמְיְנוּ אֶת בֵּית הַמִּדְרָשׁ הַגָּדוֹל שֶׁל יְשִׁיבַת אֶרֶץ יִשְׂרָאֵל לִפְנֵי כְּאַלְפַּיִם שָׁנָה. קוֹלוֹת שֶׁל לִמּוּד תּוֹרָה מְמַלְאִים אֶת הָאֲוִיר, מִתְנַגְּנִים בְּשִׂמְחָה עֲצוּמָה. בְּמֶרְכַּז בֵּית הַמִּדְרָשׁ יָשְׁבוּ שְׁנֵי עֲנָקִים, קְדוֹשֵׁי עֶלְיוֹן: הָאָמוֹרָא הַגִּבּוֹר רֵישׁ לָקִישׁ, וְהֶחָכָם הַגָּדוֹל, רַבִּי אַבָּהוּ.

רַבִּי אַבָּהוּ הִבִּיט מִבַּעַד לַחַלּוֹן עַל הָרֵי הַגָּלִיל הַיְּרֻקִּים, עַל הַשֶּׁמֶשׁ הַזְּהוּבָה שֶׁל אֶרֶץ יִשְׂרָאֵל שֶׁלִּטְּפָה אֶת עֲצֵי הַזַּיִת, וְעֵינָיו בָּרְקוּ. "אֵין תּוֹרָה כְּתוֹרַת אֶרֶץ יִשְׂרָאֵל," אָמַר הָרַב בְּקוֹל רוֹעֵד מֵהִתְרַגְּשׁוּת. "הָאֲוִיר שֶׁל אֶרֶץ יִשְׂרָאֵל הוּא קָדוֹשׁ. הָאֲוִיר הַזֶּה – הוּא מַחְכִּים אֶת הָאָדָם וּמַכְנִיס בּוֹ יִרְאַת שָׁמַיִם אֲמִתִּית!".

רֵישׁ לָקִישׁ אָחַז בִּידוֹ שֶׁל רַבִּי אַבָּהוּ וְהִנְהֵן בְּרֹאשׁוֹ. "נָכוֹן מְאֹד," סָמַךְ רֵישׁ לָקִישׁ אֶת דְּבָרָיו. "וּכְדֵי שֶׁהַתַּלְמִידִים יָבִינוּ עַד כַּמָּה אֶרֶץ יִשְׂרָאֵל קְדוֹשָׁה, אֲסַפֵּר לָכֶם אֶת מָה שֶׁקָּרָה בְּדוֹרוֹת קוֹדְמִים, לִשְׁנַיִם מִגְּדוֹלֵי הַתַּנָּאִים: רַבִּי אֶלְעָזָר בֶּן שַׁמּוּעַ וְרַבִּי יוֹחָנָן הַסַּנְדְּלָר".

הַסַּפְסָלִים בְּבֵית הַמִּדְרָשׁ הוּדַמּוּ. כֻּלָּם רָצוּ לִשְׁמֹעַ. וְכָךְ פָּתַח וְסִפֵּר רֵישׁ לָקִישׁ...

הַצִּמָּאוֹן הָעַז לְתוֹרָה

הָיָה זֶה בְּדוֹר שֶׁלְּאַחַר חֻרְבַּן הַבַּיִת. עַם יִשְׂרָאֵל בְּאֶרֶץ יִשְׂרָאֵל הָיָה שָׁרוּי בְּצַעַר גָּדוֹל, וְהַרְבֵּה חֲכָמִים גָּלוּ לְחוּץ לָאָרֶץ עֵקֶב גְּזֵרוֹת הָרוֹמָאִים. הַחֲכָמִים רַבִּי אֶלְעָזָר בֶּן שַׁמּוּעַ וְרַבִּי יוֹחָנָן הַסַּנְדְּלָר אָהֲבוּ אֶת הַתּוֹרָה כָּל כָּךְ, עַד שֶׁהִרְגִּישׁוּ שֶׁנַּפְשָׁם מַמָּשׁ צְמֵאָה לְמַיִם.

"חַיָּבִים אֲנַחְנוּ לִלְמֹד תּוֹרָה מִפִּי רַבּוֹתֵינוּ הַגְּדוֹלִים בְּיוֹתֵר," אָמַר רַבִּי אֶלְעָזָר, "אַךְ גְּדוֹל הַדּוֹר, הַתַּנָּא הַקָּדוֹשׁ רַבִּי יְהוּדָה בֶּן בְּתֵירָא, יָשַׁב בָּעִיר 'נְצִיבִין' – וּנְצִיבִין נִמְצֵאת מֵעֵבֶר לַגְּבוּל, בְּחוּץ לָאָרֶץ!".

יְלָדִים, רַבִּי אֶלְעָזָר וְרַבִּי יוֹחָנָן יָדְעוּ שֶׁקָּשֶׁה מְאֹד לַעֲזֹב אֶת אֶרֶץ יִשְׂרָאֵל. הֵם אָהֲבוּ כָּל אַבֶן וְכָל רֶגֶב עָפָר בָּהּ. אֲבָל אָמְרוּ בְּלִבָּם: "הֲרֵי אֲנַחְנוּ לֹא יוֹצְאִים כְּדֵי לְטַיֵּל, וְלֹא כְּדֵי לַעֲשׂוֹת כֶּסֶף! אֲנַחְנוּ הוֹלְכִים נטו לִלְמֹד תּוֹרָה מִפִּי רַבִּי יְהוּדָה בֶּן בְּתֵירָא! בְּוַדַּאי שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא יִשְׂמַח בְּמַעֲשֵׂינוּ!".

בְּלֵב רוֹטֵט, הֵם אָרְזוּ אֶת חֶפְצֵיהֶם הַדַּלִּים בְּתוֹךְ שַׂקֵּי בַּד גַּסִּים, לָקְחוּ שְׁתֵּי כִּכְּרוֹת לֶחֶם וְנֹאדוֹת מַיִם, וְיָצְאוּ צָפוֹנָה, לְכִוּוּן גְּבוּל הָאָרֶץ.

הַהֲלִיכָה אֶל הַגְּבוּל

הַדֶּרֶךְ הָיְתָה אֲרֻכָּה וּמְעַיֶּפֶת. הֵם צָעֲדוּ בִּשְׁבִילֵי הֶעָפָר וְהָרִצְפָּה נָגְעָה בְּסַנְדְּלֵיהֶם. הַאֲדָמָה הָיְתָה חַמָּה וּמֻכֶּרֶת. בְּכָל פַּעַם שֶׁרָאוּ עֵץ רִמּוֹן אוֹ שְׂדֵה חִטָּה, לִבָּם הִתְרַחֵב מִשִּׂמְחָה – "פֵּרוֹת שֶׁל אֶרֶץ הַקֹּדֶשׁ!", אָמְרוּ. אֲבָל מִמּוּל עָמְדָה הַמַּטָּרָה הַגְּדוֹלָה: הַתּוֹרָה הַמְּחַכָּה לָהֶם בִּנְצִיבִין.

אַחֲרֵי יָמִים שֶׁל הֲלִיכָה, הֵם הִגִּיעוּ לָעִיר צִידוֹן . מִשָּׁם יָכְלוּ כְּבָר לִרְאוֹת אֶת אֲבָנִים הַגְּדוֹלוֹת שֶׁמְּסַמְּנוֹת אֶת גְּבוּל הָאָרֶץ. מִשָּׁם וָהָלְאָה – זֶה מְקוֹמָם שֶׁל הַגּוֹיִים. חוּץ לָאָרֶץ.

הֵם עָמְדוּ עַל הַחוֹף בְּצִידוֹן. רוּחַ חֲזָקָה נָשְׁבָה מִן הַיָּם. קוֹל הַגַּלִּים שֶׁהִתְנַפְּצוּ עַל הַסְּלָעִים נִשְׁמַע הֵיטֵב. רֵיחַ מַלּוּחַ וַחֲזָק שֶׁל מֵי יָם מִלֵּא אֶת אַפָּם. עוֹד כַּמָּה מֵאוֹת מֶטְרִים, וְהֵם בַּחוּץ. עוֹזְבִים אֶת נַחֲלַת ה', אֶת הָאָרֶץ שֶׁאֲבָהוֹתֵינוּ כָּל כָּךְ הִתְגַּעְגְּעוּ אֵלֶיהָ, וְיוֹצְאִים מִמֶּנָּה לְיָמִים, וְאוּלַי לְשָׁנִים אֲרֻכּוֹת.

הַקְּרִיעָה וְהַדְּמָעוֹת

פִּתְאוֹם... שְׁתִיקָה מֻחְלֶטֶת יָרְדָה עֲלֵיהֶם.

רַבִּי אֶלְעָזָר הָפַךְ אֶת רֹאשׁוֹ מָרְדָה (אָחוֹרָה), לְכִוּוּן דָּרוֹם, לְכִוּוּן הָרֵי מֵרוֹן וִירוּשָׁלַיִם. רַבִּי יוֹחָנָן הַסַּנְדְּלָר עָצַר גַּם הוּא וְהִבִּיט לְכִוּוּן עִיר הַקֹּדֶשׁ.

הֵם זָקְפוּ אֶת עֵינֵיהֶם כְּלַפֵּי מַעְלָה... וּפִתְאוֹם, לְלֹא מִלִּים... טִפָּה רוֹתַחַת זָלְגָה מֵעֵינוֹ שֶׁל רַבִּי אֶלְעָזָר. וְאָז דִּמְעָה נוֹסֶפֶת, וְעוֹד אַחַת. רַבִּי יוֹחָנָן גַּם הוּא הֵחֵל לִבְכּוֹת כְּיֶלֶד קָטָן שֶׁנִּפְרָד מֵאִמּוֹ. דְּמָעוֹת קְדוֹשׁוֹת, מְלֵאוֹת בְּגַעֲגּוּעַ וְאַהֲבָה לַשֵּׁם יִתְבָּרַךְ, נָשְׁרוּ עַל זְקָנָם שֶׁל שְׁנֵי עֲנָקֵי הָרוּחַ, וְטִפְטְפוּ אֶל חֲמִימוּת הַחֲמִימוּת שֶׁל אַדְמַת שְׂפַת הַיָּם.

טַעַם מַלּוּחַ שֶׁל דְּמָעוֹת אֲמִתִּיּוֹת הִתְעַרְבֵּב בָּרֵיחַ הַמַּלּוּחַ שֶׁל הַיָּם.

"אֵיךְ נַעֲזֹב אוֹתָךְ, אֶרֶץ הַקֹּדֶשׁ?" לָחַשׁ רַבִּי אֶלְעָזָר.

בְּאוֹתוֹ רֶגַע, רַגְלֵיהֶם מַמָּשׁ קָפְאוּ בִּמְקוֹמָן. הֵם לֹא הָיוּ מְסֻגָּלִים לָשִׂים וְלוּ צַעַד אֶחָד קָדִימָה. אוֹר הַשֶּׁמֶשׁ כְּאִלּוּ נֶעֱצַר רֶגַע לְהַבִּיט בָּהֶם.

מִתּוֹךְ כְּאֵב עָמוֹק שֶׁל פְּרֵדָה מֵאֶרֶץ יִשְׂרָאֵל, עָשׂוּ הַחֲכָמִים מַעֲשֶׂה מַדְהִים:

הֵם אָחֲזוּ בַּחֻלְצָה שֶׁלָּהֶם קָרוֹב לַצַּוָּאר וְ...

קְרַעעעע!

קוֹל שֶׁל קְרִיעַת הַבֶּגֶד נִשְׁמַע בָּאֲוִיר! הֵם קָרְעוּ קְרִיעָה בְּבִגְדֵיהֶם, כְּמוֹ אָדָם שֶׁנִּפְרָד מִדָּבָר שֶׁהוּא אוֹהֵב וְאַבֵּל עָלָיו, מִתּוֹךְ צַעַר גָּדוֹל עַל שֶׁהֵם נֶאֱלָצִים לַעֲזֹב אֶת אַדְמַת הַקֹּדֶשׁ.

הַהַכְרָעָה: שְׁקוּלָה כְּנֶגֶד הַכֹּל

רַבִּי יוֹחָנָן הַסַּנְדְּלָר פָּתַח אֶת תַּרְמִילוֹ וְהוֹצִיא חֻמָּשׁ קָטָן וְיָשָׁן. הוּא פָּתַח אוֹתוֹ בְּסֵפֶר דְּבָרִים, וְקָרָא בְּקוֹל בּוֹכֶה בְּפָרָשַׁת 'רְאֵה': "וִירִשְׁתֶּם אֹתָהּ... וִישַׁבְתֶּם בָּהּ. וּשְׁמַרְתֶּם לַעֲשׂוֹת אֵת כָּל הַחֻקִּים וְאֶת הַמִּשְׁפָּטִים" !

רַבִּי אֶלְעָזָר שָׁמַע אֶת הַפָּסוּק, וּמִיָּד הֵבִין אֶת דַּעַת הַשֵּׁם: כָּל הַמִּצְווֹת שֶׁל הַתּוֹרָה מְחֻבָּרוֹת לַיְּשִׁיבָה בְּאֶרֶץ יִשְׂרָאֵל. "אֲנַחְנוּ אֲמָנָם הוֹלְכִים לִלְמֹד תּוֹרָה, אֲבָל... הַיְּשִׁיבָה בְּאֶרֶץ יִשְׂרָאֵל, הִיא עַצְמָהּ הַמִּצְוָה הֲכִי קְדוֹשָׁה וְחֲשׁוּבָה!".

הֵם הִסְתַּכְּלוּ אֶחָד עַל הַשֵּׁנִי. הַדְּמָעוֹת שֶׁלָּהֶם הָפְכוּ פִּתְאוֹם לְדִמְעוֹת תִּקְוָה וְשִׂמְחָה עֲצוּמָה.

"אֲנַחְנוּ חוֹזְרִים!" קָרְאוּ יַחַד!

"אֲנַחְנוּ נִלְמַד בְּאֶרֶץ יִשְׂרָאֵל! נִמְצָא אֶת רַבּוֹתֵינוּ שֶׁנִּשְׁאֲרוּ בָּאָרֶץ! אֲנַחְנוּ לֹא נוֹטְשִׁים אֶת נַחֲלַת ה' אֲפִלּוּ לְרֶגַע אֶחָד!".

הֵם לָקְחוּ אֶת הַשַּׂקִּים שֶׁלָּהֶם, וְהַפַּעַם... סוֹבְבוּ אֶת הַגַּב לְצִידוֹן, וְאֶת הַפָּנִים אֶל תּוֹךְ אֶרֶץ יִשְׂרָאֵל, וְהֵחֵלּוּ לִצְעֹד דָּרוֹמָה, חֲזָרָה לְבָתֵּיהֶם. הַצַּעַד שֶׁלָּהֶם הָיָה פִּתְאוֹם קַל מְאֹד. הֵם הִרְגִּישׁוּ שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא שׁוֹמֵר עֲלֵיהֶם בְּתוֹךְ אַרְצוֹ. כְּשֶׁהֵם חָזְרוּ לִמְקוֹמָם, הֵם קָבְעוּ כְּלָל שֶׁכָּל הַדּוֹרוֹת יִזְכְּרוּ תָּמִיד, מִשְׁפָּט אֶחָד קָדוֹשׁ: "יְשִׁיבַת אֶרֶץ יִשְׂרָאֵל, שְׁקוּלָה כְּנֶגֶד כָּל הַמִּצְווֹת שֶׁבַּתּוֹרָה!" .

סִיֵּם רֵישׁ לָקִישׁ אֶת סִפּוּרוֹ, וְרַבִּי אַבָּהוּ קָם וְחִיֵּךְ אֶל תַּלְמִידָיו בִּישִׁיבַת אֶרֶץ יִשְׂרָאֵל. "אַתֶּם מְבִינִים, יְלָדִים?" אָמַר רַבִּי אַבָּהוּ, "אֲנַחְנוּ כָּאן. אֲנַחְנוּ קְשׁוּרִים בְּעֵץ הַחַיִּים. אֲנַחְנוּ שְׁקוּעִים בְּתוֹרָה – בְּתוֹךְ אֶרֶץ יִשְׂרָאֵל!".

כָּל בֵּית הַמִּדְרָשׁ פָּצַח בְּשִׁירָה אַדִּירָה שֶׁל הוֹדָיָה עַל הָאָרֶץ וְעַל הַתּוֹרָה.

ME (הם): רַבִּי אֶלְעָזָר וְרַבִּי יוֹחָנָן הָיוּ חֲכָמִים גְּדוֹלִים שֶׁצָּמְאוּ לְתוֹרָה. הֵם הָיוּ מוּכָנִים לוַתֵּר עַל הַלִּמּוּד אֵצֶל רַבּוֹתֵיהֶם הַגְּדוֹלִים בְּנְצִיבִין, וּבִלְבַד שֶׁלֹּא לַעֲזֹב אֶת קְדֻשַּׁת אֶרֶץ יִשְׂרָאֵל. הֵם הֵבִינוּ שֶׁהָאָרֶץ הִיא לֹא רַק מָקוֹם לָגוּר בּוֹ – אֶלָּא הִיא מִצְוָה, קְדֻשָּׁה, וּמַתָּנָה אֱלֹקִית.

WE (אנחנו): לִפְעָמִים לֹא נוֹחַ לָנוּ. חַם מְדַי, קַר מְדַי, לִפְעָמִים מִתְלוֹנְנִים עַל דְּבָרִים שֶׁקּוֹרִים פֹּה בָּאָרֶץ. לִפְעָמִים שׁוֹכְחִים אֵיזוֹ זְכוּת מַדְהִימָה וְעָצוּמָה נָפְלָה בְּחֶלְקֵנוּ, לָגוּר בָּאָרֶץ שֶׁעֲלֶיהָ חָלְמוּ וּבָכוּ דּוֹרוֹת שֶׁל גְּדוֹלֵי יִשְׂרָאֵל וּנְבִיאִים!

YOU (אתה): מָחָר בַּבֹּקֶר, כְּשֶׁתִּקְמוּ, לְכוּ אֶל הַחַלּוֹן. תִּנְשְׁמוּ נְשִׁימָה עֲמֻקָּה שֶׁל "אֲוִיר אֶרֶץ יִשְׂרָאֵל". תַּחְשְׁבוּ עַל הַדְּמָעוֹת שֶׁל הַתַּנָּאִים שֶׁלֹּא הָיוּ מְסֻגָּלִים לָצֵאת מִמֶּנָּה, וְתַגִּידוּ לַה': "תּוֹדָה, תּוֹדָה שֶׁנָּתַתָּ לִי לָגוּר בָּאָרֶץ הַקְּדוֹשָׁה שֶׁלְּךָ!".

א. שְׁאֵלַת פְּתִיחָה (מוּכָנוּת וּמֵעוֹרָבוּת):

"יְלָדִים, נַנִּיחַ שֶׁהָיוּ אוֹמְרִים לָכֶם: 'אִם תָּטוּסוּ עַכְשָׁו לְחוּץ לָאָרֶץ, תְּקַבְּלוּ אֶת הַתּוֹרָה הֲכִי נִפְלָאָה בָּעוֹלָם וְאֶת הָרַב הֲכִי חָכָם מֵאָז זְמַן נְבִיאִים'. הַאִם הֱיִיתֶם אוֹרְזִים מִזְוָדָה וְיוֹצְאִים?

הַיּוֹם נִשְׁמַע עַל שְׁנֵי גְּדוֹלֵי עֶלְיוֹן, שֶׁעָמְדוּ בְּדִיּוּק בַּמַּצָּב הַזֶּה, עַל שְׂפַת הַיָּם – וְקָבְעוּ עֲבוּרֵנוּ כְּלָל שֶׁנִּשְׁאַר לָנֶצַח!"

ג. הַמְלָצוֹת לְמַצֶּגֶת (תְּמוּנוֹת):

ד. פְּעִילוּת מְסַכֶּמֶת (לַכִּתָּה):

מִכְתָּב לְאֶרֶץ יִשְׂרָאֵל: קַח דַּפִּים שֶׁגְּזוּרִים בְּצוּרַת מַפַּת אֶרֶץ יִשְׂרָאֵל (אוֹ דַּף רָגִיל עִם סִמּוּן). בַּקֵּשׁ מִכָּל תַּלְמִיד לִכְתֹּב לְאֶרֶץ יִשְׂרָאֵל מִכְתָּב הֲלָל: "מָה אֲנִי הֲכִי אוֹהֵב בָּךְ אֶרֶץ קְדוֹשָׁה?", אוֹ "לָמָּה אֲנִי שָׂמֵחַ לִלְמֹד תּוֹרָה בְּתוֹכֵךְ?". אֶפְשָׁר שֶׁיְּצַיְּרוּ אֶת הַדְּמָעוֹת שֶׁל רַבִּי אֶלְעָזָר. תּוֹלִים אֶת הַתּוֹצָרִים עַל "קִיר אֶרֶץ יִשְׂרָאֵל" בַּכִּתָּה הַמְּקֻשָּׁט בִּירוֹק.

מקורות: עֶצֶם הַסִּפּוּר (הַקְּרִיעָה וְהַבֶּכִי בַּגְּבוּל): סִפְרִי עַל סֵפֶר דְּבָרִים, פָּרָשַׁת רְאֵה, פִּסְקָא פ' ("מַעֲשֶׂה בְּרַבִּי אֶלְעָזָר בֶּן שַׁמּוּעַ וְרַבִּי יוֹחָנָן הַסַּנְדְּלָר..."). הַמִּסְגֶּרֶת (רֵישׁ לָקִישׁ וְרַבִּי אַבָּהוּ וִישִׁיבַת אֶרֶץ יִשְׂרָאֵל): רֵישׁ לָקִישׁ וְרַבִּי אַבָּהוּ הָיוּ מִגְּדוֹלֵי הָאֲמוֹרָאִים שֶׁשִּׁבְּחוּ וְדָבְקוּ בְּאֶרֶץ יִשְׂרָאֵל וְלֹא עֲזָבוּהָ (עַל פִּי "כֹּה עָשׂוּ חֲכָמֵינוּ", ד', 35, כְּפִי שֶׁמּוּבָא בְּרְשִׁימַת הַמְּקוֹרוֹת בַּתָּכְנִית הַחִנּוּכִית). יָדוּעַ הַמַּאֲמָר מַסֶּכֶת בָּבָא בָּתְרָא (קנח:) בְּשֵׁם רַבִּי יוֹחָנָן/רֵישׁ לָקִישׁ "אֲוִירָא דְּאֶרֶץ יִשְׂרָאֵל מַחְכִּים". שִׁלּוּב אָמוֹרָאֵי אֶרֶץ יִשְׂרָאֵל הַמַּעֲבִירִים אֶת מָסֹרֶת הַתַּנָּאִים לְתַלְמִידֵיהֶם מַמְחִישׁ אֶת שַׁרְשֶׁרֶת הַתּוֹרָה וְאַהֲבַת הָאָרֶץ.

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “להלן עיבוד חינוכי, רגשי ומרתק על אהבת ארץ ישראל ומסירות הנפש עליה , המשלב את גדולי האמוראים (רֵישׁ לָקִישׁ וְרַבִּי אַבָּהוּ) ואת הסיפור המפורסם מהמדרש…” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “וּכְדֵי שֶׁהַתַּלְמִידִים יָבִינוּ עַד כַּמָּה אֶרֶץ יִשְׂרָאֵל קְדוֹשָׁה, אֲסַפֵּר לָכֶם אֶת מָה שֶׁקָּרָה בְּדוֹרוֹת קוֹדְמִים, לִשְׁנַיִם מִגְּדוֹלֵי…”?
ג. כיצד הגיב ר' יוחנן וריש לקיש למצב המתואר: “מִשָּׁם יָכְלוּ כְּבָר לִרְאוֹת אֶת אֲבָנִים הַגְּדוֹלוֹת שֶׁמְּסַמְּנוֹת אֶת גְּבוּל הָאָרֶץ.”?
ד. מה השתנה בין “הֵם קָרְעוּ קְרִיעָה בְּבִגְדֵיהֶם, כְּמוֹ אָדָם שֶׁנִּפְרָד מִדָּבָר שֶׁהוּא אוֹהֵב וְאַבֵּל עָלָיו, מִתּוֹךְ צַעַר…” לבין “הֵם הִרְגִּישׁוּ שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא שׁוֹמֵר עֲלֵיהֶם בְּתוֹךְ אַרְצוֹ.”?
ה. כיצד הגיע הסיפור אל האירוע: “YOU (אתה): מָחָר בַּבֹּקֶר, כְּשֶׁתִּקְמוּ, לְכוּ אֶל הַחַלּוֹן.”?

3. מה היה ראשון?

כתבו ליד כל אירוע מספר מ־1 עד 5 לפי סדר התרחשותו בסיפור.

□ הֵם לֹא הָיוּ מְסֻגָּלִים לָשִׂים וְלוּ צַעַד אֶחָד קָדִימָה.
□ להלן עיבוד חינוכי, רגשי ומרתק על אהבת ארץ ישראל ומסירות הנפש עליה , המשלב את גדולי האמוראים (רֵישׁ לָקִישׁ וְרַבִּי אַבָּהוּ) ואת הסיפור המפורסם מהמדרש…
□ תּוֹלִים אֶת הַתּוֹצָרִים עַל "קִיר אֶרֶץ יִשְׂרָאֵל" בַּכִּתָּה הַמְּקֻשָּׁט בִּירוֹק.
□ אֲבָל אָמְרוּ בְּלִבָּם: "הֲרֵי אֲנַחְנוּ לֹא יוֹצְאִים כְּדֵי לְטַיֵּל, וְלֹא כְּדֵי לַעֲשׂוֹת כֶּסֶף!
□ כְּשֶׁהֵם חָזְרוּ לִמְקוֹמָם, הֵם קָבְעוּ כְּלָל שֶׁכָּל הַדּוֹרוֹת יִזְכְּרוּ תָּמִיד, מִשְׁפָּט אֶחָד קָדוֹשׁ: "יְשִׁיבַת אֶרֶץ יִשְׂרָאֵל, שְׁקוּלָה…

לפני ואחרי

מה קרה לפני: “הֵם לֹא הָיוּ מְסֻגָּלִים לָשִׂים וְלוּ צַעַד אֶחָד קָדִימָה.”?
מה קרה אחרי: “כְּשֶׁהֵם חָזְרוּ לִמְקוֹמָם, הֵם קָבְעוּ כְּלָל שֶׁכָּל הַדּוֹרוֹת יִזְכְּרוּ תָּמִיד, מִשְׁפָּט אֶחָד קָדוֹשׁ: "יְשִׁיבַת אֶרֶץ יִשְׂרָאֵל, שְׁקוּלָה…”?

סיבה ותוצאה מתוך המעשה

האירוע הנתוןמה גרם לו?מה הייתה תוצאתו?
אֲנַחְנוּ הוֹלְכִים נטו לִלְמֹד תּוֹרָה מִפִּי רַבִּי יְהוּדָה בֶּן בְּתֵירָא!
רַבִּי אֶלְעָזָר שָׁמַע אֶת הַפָּסוּק, וּמִיָּד הֵבִין אֶת דַּעַת הַשֵּׁם: כָּל הַמִּצְווֹת שֶׁל הַתּוֹרָה מְחֻבָּרוֹת לַיְּשִׁיבָה בְּאֶרֶץ יִשְׂרָאֵל.

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “ריש לקיש ורבי אבהו — ישיבת ארץ ישראל” על ר' יוחנן וריש לקיש? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “רַבִּי יוֹחָנָן גַּם הוּא הֵחֵל לִבְכּוֹת כְּיֶלֶד קָטָן שֶׁנִּפְרָד מֵאִמּוֹ.”.
ג. אילו ר' יוחנן וריש לקיש לא היה פועל כפי שמתואר ב־“הֵם הִרְגִּישׁוּ שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא שׁוֹמֵר עֲלֵיהֶם בְּתוֹךְ אַרְצוֹ.” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “YOU (אתה): מָחָר בַּבֹּקֶר, כְּשֶׁתִּקְמוּ, לְכוּ אֶל הַחַלּוֹן.” מחזק את המסר של הסיפור?

1. להלן עיבוד חינוכי, רגשי ומרתק על אהבת ארץ ישראל ומסירות הנפש עליה ,…

4. וּשְׁמַרְתֶּם לַעֲשׂוֹת אֵת כָּל הַחֻקִּים וְאֶת הַמִּשְׁפָּטִים" !

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “מִשָּׁם יָכְלוּ כְּבָר לִרְאוֹת אֶת אֲבָנִים הַגְּדוֹלוֹת שֶׁמְּסַמְּנוֹת אֶת גְּבוּל הָאָרֶץ.”.
מכתב: כתבו לר' יוחנן וריש לקיש בעקבות “רַבִּי אֶלְעָזָר שָׁמַע אֶת הַפָּסוּק, וּמִיָּד הֵבִין אֶת דַּעַת הַשֵּׁם: כָּל הַמִּצְווֹת שֶׁל הַתּוֹרָה מְחֻבָּרוֹת…”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “הֵם הָיוּ מוּכָנִים לוַתֵּר עַל הַלִּמּוּד אֵצֶל רַבּוֹתֵיהֶם הַגְּדוֹלִים בְּנְצִיבִין, וּבִלְבַד שֶׁלֹּא לַעֲזֹב אֶת קְדֻשַּׁת…”.
דיווח: כתבו ידיעה קצרה שכותרתה “ריש לקיש ורבי אבהו — ישיבת ארץ ישראל”.
