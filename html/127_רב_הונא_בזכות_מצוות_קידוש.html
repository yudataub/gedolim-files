בזכות מצוות קידוש

דף עבודה בעקבות רב הונא
שם: כיתה:

המעשה — הסיפור המלא

להלן מערך חינוכי, רגשי ועשיר על דמותו של רב הונא ומסירות הנפש שלו על מצוות הקידוש . הסיפור מעובד בקפידה עבור בנים בכיתות ב'-ד' בחינוך הדתי-לאומי, תוך שימוש בכללי הסטוריטלינג, תיאורים חושיים, כבוד התורה, ואמונה ברורה בהשגחה פרטית.

הסיפור מנוקד במלואו כנדרש.

פְּתִיחָה: רֵיחַ שֶׁל שַׁבָּת – וְשֻׁלְחָן רֵיק

יְלָדִים, עִצְמוּ רֶגַע אֶת הָעֵינַיִם וְנַסּוּ לְדַמְיֵן שֶׁאַתֶּם הוֹלְכִים בִּרְחוֹב הַשְּׁכוּנָה שֶׁלָּכֶם בְּיוֹם שִׁישִּׁי בַּצָּהֳרַיִם. מָה אַתֶּם מְרִיחִים? בְּוַדַּאי שֶׁאֶת הָרֵיחַ הַמָּתוֹק שֶׁל הַחַלּוֹת הָאֲפוּיוֹת, מָרָק הָעוֹף הַמַּהְבִּיל וְדָגִים טְעִימִים. מִכָּל הַבָּתִּים שׁוֹמְעִים קוֹלוֹת שֶׁל נִקְיוֹנוֹת וְהַכָּנוֹת, כֻּלָּם מִתְרַגְּשִׁים לְקַבֵּל אֶת שַׁבָּת הַמַּלְכָּה!

כָּךְ הָיָה לִפְנֵי כְּאַלְפַּיִם שָׁנָה, בְּעִיר גְּדוֹלָה בְּבָבֶל (עִירַאק שֶׁל יָמֵינוּ). הַשֶּׁמֶשׁ נָטְתָה לָרֶדֶת וְצָבְעָה אֶת הַשָּׁמַיִם בִּצְבָעִים מַרְהִיבִים שֶׁל כָּתֹם, אָדֹם וְזָהָב.

בְּתוֹךְ בַּיִת אֶחָד קָטָן וְעָנִי, עָמַד אָדָם קָדוֹשׁ – רַב הוּנָא.

רַב הוּנָא הָיָה צַדִּיק יְסוֹד עוֹלָם וְתַלְמִיד חָכָם עָצוּם. כָּל הַיּוֹם וְכָל הַלַּיְלָה יָשַׁב בְּבֵית הַמִּדְרָשׁ וְלָמַד תּוֹרָה מִפִּי רַבּוֹ הַגָּדוֹל, אֲבָל בַּבַּיִת שֶׁלּוֹ... לֹא הָיָה כִּמְעַט כְּלוּם.

עֶרֶב שַׁבָּת הִגִּיעַ. רַב הוּנָא הִבִּיט עַל הַשֻּׁלְחָן הֶעָרוּךְ בִּפְשִׁיטוּת. הוּא הִכְנִיס אֶת יָדוֹ לַכִּיס, מְחַפֵּשׂ בְּתִקְוָה, אוּלַי, אוּלַי נִשְׁאֲרָה שָׁם אֵיזוֹ מַטְבֵּעַ נְחֹשֶׁת שְׁחוּקָה?

אַךְ הַכִּיס הָיָה רֵיק מִלִּפְנִים וּמֵאָחוֹר.

אֵין כֶּסֶף לִקְנוֹת יַיִן לְקִדּוּשׁ.

הַהַחְלָטָה הַקָּשָׁה – הַוִּתּוּר מֵאַהֲבָה

אֵיךְ אֶפְשָׁר לְקַדֵּשׁ אֶת הַשַּׁבָּת בְּלִי כּוֹס יַיִן אוֹ מִיץ עֲנָבִים תִּירוֹשׁ? הַלֵּב שֶׁל רַב הוּנָא נֶחְמַץ. הַשַּׁבָּת הִיא הַמַּתָּנָה הֲכִי גְּדוֹלָה שֶׁנָּתַן לָנוּ בּוֹרֵא עוֹלָם, מַתָּנָה מִבֵּית גְּנָזָיו! הַאִם אֵשֵׁב עָצוּב וְלֹא אֲקַיֵּם אֶת הַמִּצְוָה כָּרָאוּי? חָשַׁב בְּלִבּוֹ.

רַב הוּנָא הִסְתַּכֵּל סְבִיבוֹ. לֹא הָיָה שׁוּם דָּבָר שֶׁאֶפְשָׁר לִמְכֹּר. וְאָז, הִבְהִיק רַעְיוֹן בְּמֹחוֹ. הוּא הִבִּיט כְּלַפֵּי מַטָּה, וְרָאָה אֶת הַחֲגוֹרָה שֶׁעַל מָתְנָיו. קָרְאוּ לָהּ "אֵזוֹר". בְּאוֹתָם יָמִים, אֲנָשִׁים לָבְשׁוּ בֶּגֶד אָרֹךְ (כְּמוֹ חָלוּק), וּבְלִי הַחֲגוֹרָה הָעָבָה שֶׁתַּחְזִיק אֶת הַבֶּגֶד צָמוּד לַגּוּף, קָשֶׁה הָיָה לָלֶכֶת אוֹ לְהִתְכּוֹפֵף.

"אֲנִי אֶמְכֹּר אֶת הַחֲגוֹרָה שֶׁלִּי!" קָרָא רַב הוּנָא בְּשִׂמְחָה שֶׁהֵאִירָה אֶת פָּנָיו הַטְּהוֹרוֹת.

הוּא פָּתַח אֶת הַחֲגוֹרָה, קִפֵּל אוֹתָהּ בִּזְהִירוּת, וּמִהֵר לַשּׁוּק שֶׁל בָּבֶל.

רַעַשׁ... הֲמוּלָה... "קְנוּ תְּאֵנִים! תַּפּוּחִים! בָּשָׂר לְשַׁבָּת!" צָעֲקוּ הַמּוֹכְרִים סְבִיבוֹ.

רַב הוּנָא לֹא הִתְבַּיֵּשׁ. הוּא מָכַר אֶת חֲגוֹרָתוֹ לְאֶחָד הַסּוֹחֲרִים, וְקִבֵּל בִּתְמוּרָה כַּמָּה מַטְבְּעוֹת נוֹצְצוֹת. צִלְצוּל הַמַּטְבְּעוֹת בְּכַף יָדוֹ הָיָה בְּעֵינָיו כַּשִּׁירָה הַיָּפָה בְּיוֹתֵר – כִּי עַכְשָׁו יוּכַל לְקַדֵּשׁ אֶת שְׁמוֹ שֶׁל ה'!

בְּשִׂמְחָה עֲצוּמָה הוּא קָנָה כַּד יַיִן קָטָן, אָדֹם וּמָתוֹק.

אֲבָל... יְלָדִים, אֵיךְ רַב הוּנָא יַחְזִיק עַכְשָׁו אֶת בְּגָדָיו? הַבֶּגֶד פָּתוּחַ וְרָפוּי, וְהוּא תַּלְמִיד חָכָם חָשׁוּב, אָסוּר לוֹ לָלֶכֶת כָּךְ!

מָה עָשָׂה רַב הוּנָא? מָצָא עַל הָאֲדָמָה חֶבֶל דַּק, גִּבְעוֹל שֶׁל קַשׁ (גֶּמִי) קָשֶׁה וְדוֹקֵר! חֶבֶל שֶׁאוֹסְפִים בּוֹ חֲבִילוֹת. הוּא אָסַף אֶת הַחֶבֶל בְּאַהֲבָה, כָּרַךְ אוֹתוֹ סְבִיב הַמָּתְנַיִם שֶׁלּוֹ וְקָשַׁר קֶשֶׁר חָזָק. הַקַּשׁ שָׂרַט אוֹתוֹ מְעַט, אֲבָל רַב הוּנָא רַק חִיֵּךְ לְעַצְמוֹ בַּדֶּרֶךְ הַבַּיְתָה, חָבַק בִּשְׁתֵּי יָדָיו אֶת כַּד הַיַּיִן בַּעַל הָרֵיחַ הַמְּשַׁכֵּר, וְשָׁר זְמִירוֹת שַׁבָּת. הַקָּדוֹשׁ בָּרוּךְ הוּא רָאָה הַכֹּל מִלְמַעְלָה – וְשָׂמַח.

הַמִּפְגָּשׁ בְּבֵית הַמִּדְרָשׁ וּבִרְכַּת הָרַב

בְּאוֹתָהּ שַׁבָּת בַּבֹּקֶר, בְּבֵית הַמִּדְרָשׁ הַגָּדוֹל, עָמַד גְּדוֹל הַדּוֹר – רַב, וְלִמֵּד תּוֹרָה מֵאוֹת תַּלְמִידִים.

רַב הוּנָא יָשַׁב בַּשּׁוּרָה הָרִאשׁוֹנָה כְּהֶרְגֵּלוֹ, עֵינָיו בּוֹרְקוֹת מֵחָכְמָה, אַךְ סְבִיב מָתְנָיו הָיָה קָשׁוּר אוֹתוֹ חֶבֶל קַשׁ עָלוּב.

"רַב" (רַבִּי אַבָּא) מַבִּיט עַל תַּלְמִידוֹ הָאָהוּב וּמִתְפַּלֵּא: "הוּנָא בְּנִי, מַה זֶּה עַל מָתְנֶיךָ? אֵיפֹה הַחֲגוֹרָה שֶׁלְּךָ?".

רַב הוּנָא כִּמְעַט הִסְמִיק, אַךְ לֹא הִסְתִּיר אֶת הָאֱמֶת, וְסִפֵּר בַּעֲנָוָה: "רַבִּי הַקָּדוֹשׁ... לֹא הָיָה לִי כֶּסֶף לִקְנוֹת יַיִן לְקִדּוּשׁ לִכְבוֹד הַשַּׁבָּת. הָיִיתִי חַיָּב לְקַדֵּשׁ! לָכֵן מָכַרְתִּי אֶת חֲגוֹרָתִי וְקָנִיתִי יַיִן, וְעַכְשָׁו קָשַׁרְתִּי חֶבֶל קַשׁ בִּמְקוֹמָהּ כְּדֵי שֶׁאוּכַל לַעֲמֹד לְפָנֶיךָ מְכֻבָּד".

הַהַסְבָּר הַזֶּה הִרְעִיד אֶת לִבּוֹ שֶׁל "רַב". לִרְאוֹת תַּלְמִיד חָכָם שֶׁמּוֹסֵר אֶת נוֹחוּתוֹ, שֶׁהוֹלֵךְ עִם חֶבֶל שׂוֹרֵט, וּבִלְבַד לְקַדֵּשׁ עַל הַיַּיִן?

"רַב" הֵרִים אֶת יָדָיו, עָצַם אֶת עֵינָיו וְאָמַר לוֹ בְּרָכָה שֶׁבָּאָה מֵעֹמֶק הַנְּשָׁמָה:

"יְהִי רָצוֹן, שֶׁתִּזְכֶּה לִהְיוֹת מְעֻטָּף בִּבְגָדִים שֶׁל שִׁירָאִים!" (שִׁירָאִים = מֶשִׁי יָקָר וּמְפֹאָר מְאֹד).

כָּל בֵּית הַמִּדְרָשׁ עָנָה: אָמֵן!

הֶבְזֵק הַזְּמַן: הַבְּרָכָה הַנְּבִיאִית מִתְגַּשֶּׁמֶת

יְלָדִים, שָׁנִים עָבְרוּ. הַקָּדוֹשׁ בָּרוּךְ הוּא אֵינוֹ מְקַפֵּחַ שְׂכָרוֹ שֶׁל מִי שֶׁמּוֹסֵר נֶפֶשׁ לַמִּצְוָה.

רַב הוּנָא גָּדַל בַּתּוֹרָה וְהָפַךְ לְעַצְמוֹ לְרֹאשׁ יְשִׁיבַת סוּרָא! לָמְדוּ אֶצְלוֹ בְּכָל יְמֵי הַשָּׁנָה 800 תַּלְמִידִים! וּלְצַד הַתּוֹרָה, הוּא גַּם זָכָה לְעֹשֶׁר אַגָּדִי (הָיוּ לוֹ מֵאוֹת חָבִיּוֹת שֶׁל יַיִן, מֵאוֹת סְפִינוֹת וַעֲדָרִים).

וְהִנֵּה הִגִּיעַ הַיּוֹם שֶׁבּוֹ הִתְחַתֵּן בְּנוֹ שֶׁל רַב הוּנָא, צַדִּיק עָצוּם בִּפְנֵי עַצְמוֹ שֶׁשְּׁמוֹ "רַבָּה".

בְּיוֹם הַחֲתֻנָּה שֶׁל בְּנוֹ, מֵרֹב עֲיֵפוּת הַהֲכָנוֹת, רַב הוּנָא שָׁכַב לָנוּחַ בְּחַדְרוֹ, עַל גַּבֵּי מִטָּתוֹ. בְּאוֹתָהּ שָׁעָה, נִכְנְסוּ הַבָּנוֹת וְהַכַּלּוֹת שֶׁלּוֹ לַחֶדֶר, וְהֵן לֹא רָאוּ שֶׁהוּא יָשֵׁן שָׁם (כִּי הָיָה אָדָם שֶׁלֹּא בָּלַט בְּמִדּוֹת גּוּפוֹ). הֵן חִפְּשׂוּ מָקוֹם לְהַנִּיחַ אֶת בִּגְדֵי הַמֶּשִׁי הַיְּקָרִים וְהַיָּפִים שֶׁהֵכִינוּ לַחֲתֻנָּה... וְאֵיפֹה הֵן הִנִּיחוּ אוֹתָם? רַק וְרַק עַל הַמִּטָּה שֶׁלּוֹ!

עוֹד בֶּגֶד, וְעוֹד מְעִיל מֶשִׁי שֶׁל שַׁבָּת, מְצֻיָּץ בְּזָהָב וּכְסָפִים!

כְּשֶׁרַב הוּנָא פָּקַח אֶת עֵינָיו וְהִתְעוֹרֵר, הוּא גִּלָּה לְהַפְתַּעְתּוֹ, שֶׁהוּא מְכֻסֶּה – מִכַּף רֶגֶל וְעַד רֹאשׁ – בְּהַר שֶׁל בִּגְדֵי שִׁירָאִים! רַךְ, נָעִים, יָקָר וּמַבְרִיק מִמֶּשִׁי.

בְּאוֹתוֹ רֶגַע צָפוּ הַדְּמָעוֹת בְּעֵינָיו שֶׁל רַב הוּנָא. הוּא נִזְכַּר בְּאוֹתוֹ יוֹם שִׁישִּׁי עָנִי וְקָר, בְּחֶבֶל הַקַּשׁ שֶׁדָּקַר בְּמָתְנָיו מִפְּנֵי שֶׁכִּבֵּד אֶת הַשַּׁבָּת וּמָכַר אֶת חֲגוֹרָתוֹ לְיַיִן... וְהוּא נִזְכַּר בְּבִרְכָּתוֹ שֶׁל רַבּוֹ הַקָּדוֹשׁ: "שֶׁתִּזְכֶּה לִהְיוֹת מְעֻטָּף בְּבִגְדֵי מֶשִׁי". הַבְּרָכָה – מִלָּה בְּמִלָּה – הִתְקַיְּמָה!

הוּא (רַב הוּנָא): מָסַר אֶת נוֹחוּתוֹ, מָכַר אֶת חֲגוֹרָתוֹ וְהָלַךְ עִם חֶבֶל עָלוּב קָשׁוּר לְמָתְנָיו, הַכֹּל מֵאַהֲבַת כְּבוֹד הַשַּׁבָּת וּמִצְוַת הַקִּדּוּשׁ הַיְּקָרָה לוֹ יוֹתֵר מִגּוּפוֹ.

אֲנַחְנוּ (WE): לִפְעָמִים, קָשֶׁה לָנוּ לְוַתֵּר עַל מַשֶּׁהוּ קָטָן בִּשְׁבִיל מִצְוָה אוֹ לִכְבוֹד שַׁבָּת. לִפְעָמִים מִתְעַצְּלִים לָקוּם בְּיוֹם שִׁישִּׁי לְסַדֵּר אֶת הַשֻּׁלְחָן כִּי רָצִינוּ לְשַׂחֵק עִם חֲבֵרִים בַּחוּץ, אוֹ שֶׁקָּשֶׁה לָנוּ לִשְׁמֹעַ אֶת הַקִּדּוּשׁ עַד הַסּוֹף כִּי אֲנַחְנוּ כְּבָר רְעֵבִים עַד שֶׁהַבֶּטֶן מְקַרְקֶרֶת.

אַתֶּם (YOU): בְּיוֹם שִׁישִּׁי הַקָּרוֹב, אוֹ בְּסְעוּדַת הַשַּׁבָּת, כְּשֶׁאַתֶּם שׁוֹמְעִים אֶת כּוֹס הַקִּדּוּשׁ וּמֵרִיחִים אֶת הַיַּיִן – הִזָּכְרוּ בְּרַב הוּנָא. וַתְּרוּ וִתּוּר קָטָן לִכְבוֹד שַׁבָּת, עִמְדוּ זְקוּפִים כְּמוֹ חַיָּלִים שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא, עִם פָּנִים שְׂמֵחוֹת, וְתַרְגִּישׁוּ שֶׁאַתֶּם עוֹשִׂים אֶת רְצוֹן ה' בְּאַהֲבָה!

רַב הוּנָא גָּדַל בַּתּוֹרָה וְהָפַךְ לְעַצְמוֹ לְרֹאשׁ יְשִׁיבַת סוּרָא!

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “בזכות מצוות קידוש” על רב הונא? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “הַקָּדוֹשׁ בָּרוּךְ הוּא רָאָה הַכֹּל מִלְמַעְלָה – וְשָׂמַח.”.
ג. אילו רב הונא לא היה פועל כפי שמתואר ב־“הֵן חִפְּשׂוּ מָקוֹם לְהַנִּיחַ אֶת בִּגְדֵי הַמֶּשִׁי הַיְּקָרִים וְהַיָּפִים שֶׁהֵכִינוּ לַחֲתֻנָּה...” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “אַתֶּם (YOU): בְּיוֹם שִׁישִּׁי הַקָּרוֹב, אוֹ בְּסְעוּדַת הַשַּׁבָּת, כְּשֶׁאַתֶּם שׁוֹמְעִים אֶת כּוֹס הַקִּדּוּשׁ וּמֵרִיחִים…” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. להלן מערך חינוכי, רגשי ועשיר על דמותו של רב הונא ומסירות הנפש שלו על…
2. הַשַּׁבָּת הִיא הַמַּתָּנָה הֲכִי גְּדוֹלָה שֶׁנָּתַן לָנוּ בּוֹרֵא…
3. הוּא אָסַף אֶת הַחֶבֶל בְּאַהֲבָה, כָּרַךְ אוֹתוֹ סְבִיב הַמָּתְנַיִם…
4. יְהִי רָצוֹן, שֶׁתִּזְכֶּה לִהְיוֹת מְעֻטָּף בִּבְגָדִים שֶׁל…
5. בְּאוֹתוֹ רֶגַע צָפוּ הַדְּמָעוֹת בְּעֵינָיו שֶׁל רַב הוּנָא.
6. אֶת סִרְטֵי הַנְּיָר נִתְלֶה עַל הַקִּיר תַּחַת הַכּוֹתֶרֶת:…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “הַבֶּגֶד פָּתוּחַ וְרָפוּי, וְהוּא תַּלְמִיד חָכָם חָשׁוּב, אָסוּר לוֹ לָלֶכֶת כָּךְ!”.
מכתב: כתבו לרב הונא בעקבות “רַב הוּנָא גָּדַל בַּתּוֹרָה וְהָפַךְ לְעַצְמוֹ לְרֹאשׁ יְשִׁיבַת סוּרָא!”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “הוּא נִזְכַּר בְּאוֹתוֹ יוֹם שִׁישִּׁי עָנִי וְקָר, בְּחֶבֶל הַקַּשׁ שֶׁדָּקַר בְּמָתְנָיו מִפְּנֵי שֶׁכִּבֵּד אֶת הַשַּׁבָּת…”.
דיווח: כתבו ידיעה קצרה שכותרתה “בזכות מצוות קידוש”.

המחזה

רגע מן הסיפורהדמויותמה יאמרו או יעשו?
אֲנִי אֶמְכֹּר אֶת הַחֲגוֹרָה שֶׁלִּי!" קָרָא רַב הוּנָא בְּשִׂמְחָה שֶׁהֵאִירָה אֶת פָּנָיו…
הֵן חִפְּשׂוּ מָקוֹם לְהַנִּיחַ אֶת בִּגְדֵי הַמֶּשִׁי הַיְּקָרִים וְהַיָּפִים שֶׁהֵכִינוּ…

7. חיבור לחיים מתוך המעשה

א. מה אפשר ללמוד מן הדרך שבה רב הונא פעל באירוע “לָכֵן מָכַרְתִּי אֶת חֲגוֹרָתִי וְקָנִיתִי יַיִן, וְעַכְשָׁו קָשַׁרְתִּי חֶבֶל קַשׁ בִּמְקוֹמָהּ כְּדֵי שֶׁאוּכַל לַעֲמֹד…”?
ב. תארו מצב מחיי הכיתה שבו אפשר ליישם את המסר של “בזכות מצוות קידוש”.
