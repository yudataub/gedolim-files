כפני מלאך — מתאמץ אצל הרומאים

דף עבודה בעקבות ר' יהושע בן לוי
שם: כיתה:

המעשה — הסיפור המלא

להלן עיבוד חינוכי, רגשי ומרתק לסיפור המופלא על רבי יהושע בן לוי והשר הרומאי , המעובד במיוחד לבנים בכיתות ב'-ד' (חינוך דתי-לאומי). הסיפור מדגיש את כוחה של קדושת התורה, שאינה זקוקה לכלי נשק כדי לנצח, ואת העובדה שמי שמחובר לקדוש ברוך הוא – גם אומות העולם רוחשים לו כבוד ויראה.

הסיפור מנוקד במלואו, וערוך על פי כל כללי הסטוריטלינג שביקשת.

פְּתִיחָה: חֶרְדַת הַמַּגָּפַיִם הַכְּבֵדִים

יְלָדִים, עִצְמוּ אֶת הָעֵינַיִם וְנַסּוּ לְדַמְיֵן כְּפָר קָטָן בְּאֶרֶץ יִשְׂרָאֵל שֶׁל יְמֵי קֶדֶם. הַשֶּׁקֶט שֶׁל שְׁעַת הַצָּהֳרַיִם מֻפְרָע לְפֶתַע.

בּוּם... טְרַאח... בּוּם... קוֹל מַגְּפֵי הַבַּרְזֶל שֶׁל הַחַיָּלִים הָרוֹמָאִים מַרְעִיד אֶת הָאֲדָמָה. הֵם צוֹעֲדִים בַּסִּמְטָאוֹת. הַשִּׁרְיוֹנוֹת שֶׁלָּהֶם נוֹצְצִים בַּשֶּׁמֶשׁ הַחַמָּה, רֵיחַ שֶׁל עֲשַׁן לַפִּידִים וְאֵשׁ עוֹמֵד בָּאֲוִיר. שִׁקְשׁוּק הַחֲרָבוֹת שֶׁלָּהֶם מַטִּיל אֵימָה.

הַחַיָּלִים הָרוֹמָאִים תּוֹלִים שֶׁלֶט גָּדוֹל עַל לוח הַמּוֹדָעוֹת שֶׁל הָעִיר, וְהַכְּרוֹז צוֹעֵק תַּחַת הַחַלּוֹנוֹת: "בִּפְקֻדַּת הַשַּׂר הָרוֹמָאִי הַגָּדוֹל! מֵהַיּוֹם גְּזֵרָה חֲדָשָׁה עַל הַיְּהוּדִים!".

אִמָּהוֹת סוֹגְרוֹת מַהֵר אֶת תְּרִיסֵי הָעֵץ בְּפַחַד. הַיְלָדִים מִתְחַבְּאִים תַּחַת הַשְּׂמִיכוֹת. שׁוּב גְּזֵרָה? שׁוּב רוֹצִים לִקַּחַת כֶּסֶף? שׁוּב אוֹסְרִים עָלֵינוּ לִלְמֹד תּוֹרָה? הַלֵּב שֶׁל כָּל עַם יִשְׂרָאֵל רָעַד. לָרוֹמָאִים יֵשׁ צָבָא עָצוּם. אֵיךְ אֶפְשָׁר לְהִלָּחֵם בָּהֶם? לְמִי יֵשׁ כֹּחַ לַעֲמֹד מוּל שַׂר רוֹמָאִי אַכְזָר שֶׁיּוֹשֵׁב בְּאַרְמוֹן כָּל כָּךְ שָׁמוּר?

הָאוֹר בְּבֵית הַמִּדְרָשׁ

בְּאוֹתוֹ הַזְּמַן בְּדִיּוּק, בִּקְצֵה הָעִיר, מִתּוֹךְ מִבְנֶה פָּשׁוּט שֶׁל אֶבֶן, עָלָה קוֹל אַחֵר לְגַמְרֵי. לֹא קוֹל שֶׁל נֶשֶׁק, אֶלָּא קוֹל מָתוֹק מִדְּבַשׁ: "אָמַר רַבִּי... וְהַהֲלָכָה הִיא..." . זֶה הָיָה קוֹל סִלְסוּל לִמּוּד הַתּוֹרָה שֶׁל מַנְהִיג הַדּוֹר – רַבִּי יְהוֹשֻׁעַ בֶּן לֵוִי .

רַבִּי יְהוֹשֻׁעַ לֹא הָיָה אִישׁ צָבָא שְׁרִירִי. הוּא הָיָה צַדִּיק קָדוֹשׁ. רַבִּי יְהוֹשֻׁעַ הָיָה צולע מְעַט עַל יְרֵכוֹ, וְהוּא יָשַׁב רָכוּן עַל סִפְרֵי הַקֹּדֶשׁ יוֹמָם וָלַיְלָה. מֵרֹב קְדֻשָּׁה וְלִמּוּד תּוֹרָה, הָיָה נִרְאֶה כְּאִלּוּ פָּנָיו מְאִירוֹת בְּאוֹר מְיֻחָד.

כַּאֲשֶׁר נִכְנְסוּ הַיְהוּדִים לְבֵית הַמִּדְרָשׁ, בּוֹכִים וְרוֹעֲדִים, הֵם זָעֲקוּ: "רַבֵּנוּ! הַשַּׂר הָרוֹמָאִי גָּזַר עָלֵינוּ גְּזֵרָה אֲיֻמָּה! הוּא דּוֹרֵשׁ לְהָרַע לָנוּ. אֲנַחְנוּ אוֹבְדִים!".

רַבִּי יְהוֹשֻׁעַ הֵרִים אֶת עֵינָיו מִסֵּפֶר הַתּוֹרָה. פָּנָיו שִׁדְּרוּ רֹגַע עָמֹק. לֹא הָיָה שָׁם כַּעַס, וְלֹא הָיָה שָׁם פַּחַד.

"אַל תִּפְחֲדוּ, יְלָדַי," אָמַר רַבִּי יְהוֹשֻׁעַ בְּקוֹל נָעִים וְחָזָק. "הַקָּדוֹשׁ בָּרוּךְ הוּא עִמָּנוּ – לֹא נִירָא! אֲנִי אֵלֵךְ בְּעַצְמִי אֶל הַשַּׂר הָרוֹמָאִי וַאֲבַקֵּשׁ רַחֲמִים עַל עַם יִשְׂרָאֵל" .

הַתַּלְמִידִים הִשְׁתַּנְּקוּ מִבֶּהָלָה: "אַתָּה?! רַבֵּנוּ! הֲרֵי הוּא אַכְזָר! הָאַרְמוֹן שֶׁלּוֹ מָלֵא בְּשׁוֹמְרִים אִימְתָנִיִּים שֶׁיִּזְרְקוּ אוֹתְךָ הַחוּצָה! אֵיךְ תַּעֲמֹד מוּל רִשְׁעוּת כָּזוֹ?!".

רַבִּי יְהוֹשֻׁעַ חִיֵּךְ לְאַט. חִיּוּךְ מָלֵא בֶּאֱמוּנָה. הוּא מִשֵּׁשׁ בְּיָדוֹ אֶת מַקֵּל הַהֲלִיכָה עֵץ הַפָּשׁוּט שֶׁלּוֹ. "נֶשֶׁק אֵין לִי," הֵשִׁיב הַצַּדִּיק, "אֲבָל כֹּחַ הַתּוֹרָה שׁוֹמֵר עָלַי. ה' מְלַוֶּה אוֹתִי שָׁעָה שָׁעָה".

הַהֲלִיכָה אֶל גֹּב הָאֲרָיוֹת

הַיּוֹם גָּדוֹל, וְרַבִּי יְהוֹשֻׁעַ יוֹצֵא לַדֶּרֶךְ. הַהֲלִיכָה קָשָׁה מְעַט, שֶׁכֵּן רַגְלוֹ כּוֹאֶבֶת, אֲבָל שְׂפָתָיו מְמַלְמְלוֹת מִזְמוֹרֵי תְּהִלִּים כָּל הַדֶּרֶךְ.

הוּא מַגִּיעַ לְקֵיסָרְיָה הַבְּנוּיָה לְחוֹף הַיָּם. עַל גִּבְעָה רָמָה עוֹמֵד אַרְמוֹן עֲנָק בָּנוּי מִשַּׁיִשׁ לָבָן וְקַר. בְּפֶתַח הָאַרְמוֹן עוֹמְדִים עֲשָׂרוֹת שׁוֹמְרִים רְחָבִים כְּהָרִים, אוֹחֲזִים בְּחֲנִיתוֹת כְּבֵדוֹת שֶׁל בַּרְזֶל. רֵיחַ כָּבֵד שֶׁל קְטֹרֶת (רֵיחַ קָשֶׁה שֶׁהָיוּ מַדְלִיקִים הַגּוֹיִים) עָלָה בָּאֲוִיר. הַמָּקוֹם כֻּלּוֹ זָעַק 'אֲנַחְנוּ כֹּחַ, אֲנַחְנוּ הַשּׁוֹלְטִים'.

רַבִּי יְהוֹשֻׁעַ, עָטוּף בְּגַלִּימָתוֹ הַפְּשׁוּטָה, צוֹעֵד לְעֵבֶר הַדְּלָתוֹת הַגְּדוֹלוֹת. מְשָׁרְתֵי הַשַּׂר מַבִּיטִים בַּיְּהוּדִי הַמְּבֻגָּר וְהַצּוֹלֵעַ שֶׁמִּתְקָרֵב, וּמִתְחִילִים לְגַחֵךְ. "לְאָן אַתָּה חוֹשֵׁב שֶׁאַתָּה הוֹלֵךְ, זָקֵן?" אָמַר מְפַקֵּד הַמִּשְׁמָר בְּקוֹל גַּס, "הַשַּׂר הָרוֹמָאִי לֹא מְקַבֵּל בְּקָשׁוֹת מִיְּהוּדִים!".

אֲבָל מַשֶּׁהוּ בָּעֵינַיִם הַטְּהוֹרוֹת שֶׁל רַבִּי יְהוֹשֻׁעַ, מַשֶּׁהוּ בָּרֹגַע הַשְּׁמֵימִי שֶׁלּוֹ, גָּרַם לַשּׁוֹמֵר לְהִשְׁתַּתֵּק. הַשּׁוֹמֵר בָּלַע אֶת רֻקּוֹ, זָז הִצִּדָּה, וְהוֹרָה לִפְתֹּחַ אֶת עֲלֵי הַדֶּלֶת הַכְּבֵדִים.

חְרִיק... קְרָאק... הַדְּלָתוֹת נִפְתְּחוּ.

הָאוֹר נִכְנָס לָאַרְמוֹן הָאָפֵל

בְּתוֹךְ אוּלַם הַמַּלְכוּת יָשַׁב הַשַּׂר הָרוֹמָאִי עַל כִּסֵּא מַלְכוּת מֻגְבָּהּ, מְצֻפֶּה בְּעָלֵי זָהָב וּקְטִיפָה אֲדֻמָּה. סְבִיבוֹ עָמְדוּ יּוֹעֲצָיו שֶׁחִפְּשׂוּ רַק רָעָה לַיְּהוּדִים.

רַבִּי יְהוֹשֻׁעַ צָעַד פְּנִימָה. בְּרֶגַע שֶׁרַגְלוֹ דָּרְכָה עַל רִצְפַּת הַשַּׁיִשׁ שֶׁל הָאוּלָם, קָרָה דָּבָר שֶׁאִישׁ לֹא הֵבִין!

לְפֶתַע, הַשַּׂר הָרוֹמָאִי הַגֵּאֶה וְהֶחָזָק, זֶה שֶׁלֹּא קָם מִכִּסְאוֹ בִּשְׁבִיל אַף אָדָם בָּעוֹלָם, תָּפַס בְּיָדִיּוֹת כִּסְאוֹ, מָשַׁךְ אֶת עַצְמוֹ מַעְלָה... וְנֶעֱמַד עַל רַגְלָיו לְיֵרָאֶה!

הוּא קָד קִדָּה קְטַנָּה, כְּמִי שֶׁעוֹמֵד מוּל מֶלֶךְ.

הַיּוֹעֲצִים הָרוֹמָאִים מִשְּׁנֵי הַצְּדָדִים כִּמְעַט הִתְעַלְּפוּ מִתַּדְהֵמָה. "הַשַּׂר יָרַד מֵהַפַּסִּים?" לָחֲשׁוּ זֶה לָזֶה.

רַבִּי יְהוֹשֻׁעַ מִתְקָרֵב. הוּא מְדַבֵּר מִתּוֹךְ בִּטָּחוֹן בַּה'. הַקָּדוֹשׁ בָּרוּךְ הוּא נָתַן אֶת הַמִּלִּים הַנְּכוֹנוֹת בְּפִיו. הַשַּׂר מַקְשִׁיב לוֹ בְּיִרְאַת כָּבוֹד, וּמִיָּד מֵרִים אֶת יָדוֹ וּמַכְרִיז: "בַּקָּשָׁתְךָ הִתְקַבְּלָה, חָכָם יְהוּדִי! הַגְּזֵרָה מְבֻטֶּלֶת!".

סוֹד הַמַּלְאָךְ

רַבִּי יְהוֹשֻׁעַ מֵנִיד בְּרֹאשׁוֹ, מִסְתּוֹבֵב וְיוֹצֵא בְּנַחַת, נוֹשֵׂא תְּפִלַּת הוֹדָיָה לַה' כְּדֶרֶךְ שֶׁבָּא.

אֲבָל בְּתוֹךְ הָאוּלָם הַמַּלְכוּתִי הָיְתָה סְעָרָה. הַיּוֹעֲצִים נִגְּשׁוּ מַהֵר אֶל הַשַּׂר, כּוֹעֲסִים וְנִרְגָּשִׁים: "אֲדוֹנֵנוּ הַשַּׂר! מָה עָשִׂיתָ?! אַתָּה, הָאִישׁ הַחָזָק בְּיוֹתֵר בַּמָּחוֹז, שׁוֹלֵט בְּחַיָּלִים וְנֶשֶׁק, אַתָּה קָם וְעוֹמֵד מִכִּסְאֲךָ לִפְנֵי אוֹתוֹ יְהוּדִי זָקֵן וְצוֹלֵעַ?! מָה חָשַׁבְתָּ לְעַצְמְךָ? זֶה בִּזָּיוֹן לָאִמְפֶּרְיָה שֶׁלָּנוּ!".

הַשַּׂר הָרוֹמָאִי הֵנִיף יָדוֹ וְשִׁתֵּק אוֹתָם. פָּנָיו עֲדַיִן הָיוּ חִוְּרוֹת.

"שׁוֹטִים שֶׁכְּמוֹתְכֶם," הוּא אָמַר לָהֶם בְּקוֹל לְאוֹט. "אַתֶּם רְאִיתֶם סְתָם אָדָם קָשִׁישׁ. אַתֶּם רְאִיתֶם מַקֵּל עֵץ. אֲבָל כְּשֶׁהוּא נִכְנַס... אֲנִי רָאִיתִי מַשֶּׁהוּ אַחֵר לְגַמְרֵי".
