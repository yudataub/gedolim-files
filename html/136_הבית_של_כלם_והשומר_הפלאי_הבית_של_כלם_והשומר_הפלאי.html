הבית של כלם והשומר הפלאי

דף עבודה בעקבות הבית של כלם והשומר הפלאי
שם: כיתה:

המעשה — הסיפור המלא

🔹 שָׁלָב 1: בֵּרוּר מַקְדִּים וְנִתּוּחַ (לַמּוֹרֶה)

1. הַפַּחַד הַטִּבְעִי לַעֲזֹב בַּיִת רֵיק מֵאֲחוֹר כְּשֶׁיֵּשׁ אוֹיְבִים מִסָּבִיב.

2. הַצְּפִיפוּת הַגְּדוֹלָה בָּעֲזָרָה שֶׁנִּרְאֵית כְּבִלְתִּי אֶפְשָׁרִית לַעֲמִידָה, וְכָל שֶׁכֵּן לְהִשְׁתַּחֲוָיָה.

🔹 שָׁלָב 2: הַסִּפּוּר הַמָּלֵא – "הַבַּיִת שֶׁל כֻּלָּם וְהַשּׁוֹמֵר הַפִּלְאִי"

פְּתִיחָה: הַדֶּלֶת הַפְּתוּחָה יְלָדִים, נַסּוּ לְדַמְיֵן – הַאִם הֱיִיתֶם טָסִים לְטִיּוּל שֶׁל שְׁבוּעַיִם, וּמַשְׁאִירִים אֶת הַדֶּלֶת שֶׁל הַבַּיִת פְּתוּחָה לְרְוָחָה? בְּלִי מַנְעוּל? בְּלִי אַזְעָקָה? בִּזְמַן שֶׁבֵּית הַמִּקְדָּשׁ הָיָה קַיָּם, הָיְתָה מִצְוָה מְיֻחֶדֶת מְאֹד בַּחַגִּים: "עֲלִיָּה לָרֶגֶל". מֵאוֹת אַלְפֵי יְהוּדִים עָזְבוּ אֶת הַבָּתִּים שֶׁלָּהֶם בַּגָּלִיל וּבַדָּרוֹם, וְעָלוּ לִירוּשָׁלַיִם.

בִּכְפָר קָטָן אֶחָד, גָּר יְהוּדִי טוֹב וּתְמִים-לֵב. הוּא הֶעֱמִיס אֶת הַחֲמוֹר שֶׁלּוֹ בְּפֵרוֹת לְבִכּוּרִים, אָרַז צֵידָה לַדֶּרֶךְ וְקָרָא לִבְנֵי מִשְׁפַּחְתּוֹ עִם חִיּוּךְ גָּדוֹל: "קָדִימָה! יוֹצְאִים לְבֵית הַמִּקְדָּשׁ!". לְפֶתַע, קָפַץ אֶחָד הַשְּׁכֵנִים (שֶׁלֹּא הָיָה יְהוּדִי), צָחַק מִמֶּנּוּ וְאָמַר: "אַתָּה מְשֻׁגָּע! אַתָּה יוֹצֵא לִירוּשָׁלַיִם וּמַשְׁאִיר פֹּה אֶת הַתַּרְנְגֹלוֹת שֶׁלְּךָ? אֶת הַזָּהָב שֶׁלְּךָ? אֶת תְּבוּאַת הַשָּׂדֶה? אֲנִי בְּעַצְמִי אֶגְנֹב לְךָ הַכֹּל בָּרֶגַע שֶׁתֵּלֵךְ!". אֲבָל הַיְּהוּדִי הִבִּיט לַשָּׁמַיִם, לֹא פָּחַד כְּלָל, וְאָמַר: "אֲנִי מְקַיֵּם אֶת הַמִּצְוָה שֶׁל ה' יִתְבָּרַךְ. ה' בְּעַצְמוֹ הִבְטִיחַ בַּתּוֹרָה: 'וְלֹא יַחְמֹד אִישׁ אֶת אַרְצְךָ בַּעֲלֹתְךָ לֵרָאוֹת אֶת פְּנֵי ה''. יֵשׁ לִי אֶת הַשּׁוֹמֵר הֲכִי טוֹב בָּעוֹלָם!". וְכָךְ, הוּא הָלַךְ לִירוּשָׁלַיִם בְּלֵב שָׂמֵחַ.

הָעֲזָרָה הַצְּפוּפָה וְעֶשֶׂר הַנִּסִּים כְּשֶׁהִגִּיעוּ הַיְהוּדִים לִירוּשָׁלַיִם, הֵם נִכְנְסוּ לְבֵית הַמִּקְדָּשׁ. יְלָדִים, אֵיזֶה הָדָר! אֵיזוֹ קְדֻשָּׁה! הֵם יָדְעוּ שֶׁבְּבֵית הַמִּקְדָּשׁ יֵשׁ עֲשָׂרָה נִסִּים גְּלוּיִים שֶׁקּוֹרִים בְּכָל יוֹם, וְהֵם מַרְאִים כַּמָּה ה' אוֹהֵב אֶת יִשְׂרָאֵל: הֵם הִבִּיטוּ לַמִּזְבֵּחַ, הָיְתָה שָׁם רוּחַ סְעָרָה חֲזָקָה שֶׁנָּשְׁבָה בֶּהָרִים, אֲבָל – נֵס! – הֶעָשָׁן שֶׁל עֲצֵי הַמִּזְבֵּחַ עָלָה יָשָׁר לַשָּׁמַיִם כְּמוֹ עַמּוּד שֶׁל זְכוּכִית, וְהָרוּחַ בִּכְלָל לֹא הֵזִיזָה אוֹתוֹ! הֵם רָאוּ שֶׁשּׁוֹחֲטִים שָׁם אֲלָפִים שֶׁל כְּבָשִׂים לְקָרְבָּנוֹת. בְּדֶרֶךְ כְּלָל מָקוֹם כָּזֶה מָלֵא בִּזְבוּבִים, נָכוֹן? אֲבָל – נֵס! – לֹא נִרְאָה אֲפִלּוּ זְבוּב אֶחָד כָּל יְמֵי בֵּית הַמִּקְדָּשׁ! מָקוֹם נָקִי וְטָהוֹר לְגַמְרֵי. וְלַמְרוֹת שֶׁיָּרַד גֶּשֶׁם עַז פַּעַם אַחַת, גְּשָׁמִים לְעוֹלָם לֹא כִּבּוּ אֶת מְדוּרַת עֲצֵי הַמִּזְבֵּחַ הַבּוֹעֶרֶת!

אֲבָל הַנֵּס הֲכִי מְרַגֵּשׁ הָיָה הַנֵּס שֶׁל הָאֲנָשִׁים הֶחָמִים שֶׁבָּאוּ מִכָּל קְצוֹת הָאָרֶץ. הָיוּ שָׁם מֵאוֹת אַלְפֵי אֲנָשִׁים בְּתוֹךְ "עֶזְרַת יִשְׂרָאֵל". מָקוֹם לֹא גָּדוֹל. הַצְּפִיפוּת הָיְתָה כָּל כָּךְ גְּדוֹלָה, עַד שֶׁהַיְהוּדִים נִדְחֲפוּ זֶה בָּזֶה, מַמָּשׁ "כָּתֵף צְמוּדָה לְכָתֵף". כְּשֶׁהֵם עָמְדוּ, הֵם הָיוּ צְפוּפִים כְּמוֹ קֻפְסַת גַּפְרוּרִים. אִי אֶפְשָׁר הָיָה לְהָזִיז יָד! וְאָז, הִגִּיעַ הָרֶגַע שֶׁבּוֹ הַכֹּהֵן הַגָּדוֹל אָמַר אֶת הַשֵּׁם הַמְפֹרָשׁ. מָה עָשׂוּ כֻּלָּם? רָצוּ לִפֹּל עַל פְּנֵיהֶם עַל הָרִצְפָּה, לְהִשְׁתַּחֲווֹת. אֲבָל אֵיךְ אֶפְשָׁר?! מֵאוֹת אַלְפֵי אֲנָשִׁים יִפְּלוּ אֶחָד עַל הַשֵּׁנִי וְיִמְחֲצוּ זֶה אֶת זֶה!

אֲבָל כָּאן הִתְגַּלְּתָה אַהֲבַת ה'. בָּרֶגַע שֶׁכֻּלָּם הִתְכּוֹפְפוּ כְּדֵי לְהִשְׁתַּחֲווֹת – נֵס עָצוּם! – פִּתְאוֹם בֵּית הַמִּקְדָּשׁ כְּאִלּוּ נִמְתַּח וְהִתְרַחֵב! כָּל אֶחָד שֶׁהִשְׁתַּחֲוָה גִּלָּה לְפֶתַע שֶׁיֵּשׁ מִסְּבִיבוֹ מֶרְחָב גָּדוֹל (כְּאַרְבַּע אַמּוֹת), אַף אֶחָד לֹא נָגַע בּוֹ, אַף אֶחָד לֹא הִפְרִיעַ לוֹ וְאַף אֶחָד לֹא שָׁמַע אֶת הַתְּפִלָּה הַסּוֹדִית וְהַשְּׁקֵטָה שֶׁלּוֹ! כְּפִי שֶׁאָמְרוּ חֲזַ"ל בַּמִּשְׁנָה: "עוֹמְדִים צְפוּפִים – וּמִשְׁתַּחֲוִים רְוָחִים!". כְּשֶׁיְּהוּדִים מִתְבַּטְּלִים כְּלַפֵּי ה', לְלֹא גַּאֲוָה, פִּתְאוֹם יֵשׁ מָקוֹם לְכֻלָּם! וְאַף מִתְפַּלֵּל מֵעוֹלָם לֹא אָמַר שֶׁאֵין מָקוֹם לִישׁוֹן בִּירוּשָׁלַיִם.

הַשּׁוֹמֵר בַּדֶּלֶת הַחַג נִגְמַר. הַיְהוּדִי הַתָּמִים חָזַר לַכְּפָר שֶׁלּוֹ, רָכוּב עַל הַחֲמוֹר, מְלֵא הִתְרַגְּשׁוּת וּקְדֻשָּׁה מִבֵּית הַמִּקְדָּשׁ. אֲבָל... לְפֶתַע לִבּוֹ דָּפַק. מָה הוּא יִמְצָא בַּבַּיִת הַפָּתוּחַ? הַאִם הַשָּׁכֵן הָרָשָׁע רוֹקֵן לוֹ אֶת הַבַּיִת וְגָנַב הַכֹּל? הוּא מִתְקָרֵב לַשַּׁעַר וְנֶעֱצָר מֻפְתָּע מִמָּה שֶׁרָאוּ עֵינָיו. עַל מַנְעוּל דַּלְתּוֹ, הָיָה כָּרוּךְ לְסִיבּוּבִים – נָחָשׁ אֵימְתָנִי וְעָצוּם! מֵרִים אֶת רֹאשׁוֹ וּמוֹצִיא אֶת לְשׁוֹנוֹ קָדִימָה בִּלְחִישָׁה: חְסְסְס... מַמָּשׁ שׁוֹמֵר עַל הַדֶּלֶת! עַל עֲרֵמַת הַחִטִּים שֶׁלּוֹ יָשַׁב אֲרִי עָצוּם שֶׁשָּׁאַג! (לפי תיאורי המדרש). כָּל מִי שֶׁנִּסָּה לְהִתְקָרֵב, בָּרַח כְּשֶׁנַּפְשׁוֹ תּוֹבַעַת עַל יָדָיו. הַקָּדוֹשׁ בָּרוּךְ הוּא גִּיֵּס אֶת שׁוֹמְרֵי הַטֶּבַע שֶׁלּוֹ כְּדֵי שֶׁיִּשְׁמְרוּ עַל הַיְהוּדִי שֶׁשָּׁמַר אֶת הַמִּצְווֹת. מִיָּד כְּשֶׁבַּעַל הַבַּיִת הִגִּיעַ, הַנָּחָשׁ גָּלַשׁ אֶל הַשָּׂדֶה וְהָאַרְיֵה חָזַר לַיַּעַר. הַיְהוּדִי נִכְנַס לְבֵיתוֹ, מָצָא אֶת כָּל רְכוּשׁוֹ בִּשְׁלֵמוּת, נָשָׂא עֵינָיו לַשָּׁמַיִם וְאָמַר לְבוֹרֵא עוֹלָם: "תּוֹדָה רַבָּה לְךָ, הַשּׁוֹמֵר הַנֶּאֱמָן שֶׁל עַם יִשְׂרָאֵל!".

🔹 שָׁלָב 3: חִבּוּר עֶרְכִּי – ME-WE-YOU

🔹 שָׁלָב 4: מַדְרִיךְ לַמּוֹרֶה / לַמְסַפֵּר וְלַמֵּכִין (מַצֶּגֶת)

א. שְׁאֵלַת פְּתִיחָה (לְעוֹרֵר עִנְיָן): "יְלָדִים, אִם כָּל מִשְׁפַּחְתְּכֶם יוֹצֵאת לְחֻפְשָׁה שֶׁל שְׁבוּעַיִם, וְאַבָּא פִּתְאוֹם אוֹמֵר: 'עִזְבוּ אֶת הַדֶּלֶת פְּתוּחָה, אַל תִּנְעֲלוּ!' – מָה הֱיִיתֶם אוֹמְרִים לוֹ? בַּשִּׁעוּר הַיּוֹם נַחֲזֹר לִתְקוּפָה שֶׁכֻּלָּם עָשׂוּ אֶת זֶה בְּשִׂמְחָה!"

ב. הֶעָרוֹת בִּצּוּעַ (לַמְסַפֵּר/לַמַּצֶּגֶת):

ג. שְׁאֵלַת סִיּוּם (הֲבָנָה וְעֵרֶךְ): "לָמָּה לְדַעְתְּכֶם ה' עָשָׂה נֵס דַּוְקָא שֶׁיִּהְיֶה מָקוֹם לְכֻלָּם כְּשֶׁהִשְׁתַּחֲווּ?" (כִּי כְּשֶׁמִּשְׁתַּחֲוִים וְעוֹשִׂים לֵב עָנָו – מִתְבַּטֶּלֶת הַגַּאֲוָה, וְאָז יֵשׁ מָקוֹם וְאַחְדוּת בָּאֻמָּה).

ד. פְּעִילוּת מְסַכֶּמֶת ("נֵס מָקוֹם" מַעֲשִׂי):

מקורות: המקורות מופיעים בקובץ המקורי.

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “הַפַּחַד הַטִּבְעִי לַעֲזֹב בַּיִת רֵיק מֵאֲחוֹר כְּשֶׁיֵּשׁ אוֹיְבִים מִסָּבִיב.” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “לְפֶתַע, קָפַץ אֶחָד הַשְּׁכֵנִים (שֶׁלֹּא הָיָה יְהוּדִי), צָחַק מִמֶּנּוּ וְאָמַר: "אַתָּה מְשֻׁגָּע!”?
ג. כיצד הגיב הבית של כלם והשומר הפלאי למצב המתואר: “וְלַמְרוֹת שֶׁיָּרַד גֶּשֶׁם עַז פַּעַם אַחַת, גְּשָׁמִים לְעוֹלָם לֹא כִּבּוּ אֶת מְדוּרַת עֲצֵי הַמִּזְבֵּחַ הַבּוֹעֶרֶת!”?
ד. מה השתנה בין “מֵאוֹת אַלְפֵי אֲנָשִׁים יִפְּלוּ אֶחָד עַל הַשֵּׁנִי וְיִמְחֲצוּ זֶה אֶת זֶה!” לבין “הַיְהוּדִי הַתָּמִים חָזַר לַכְּפָר שֶׁלּוֹ, רָכוּב עַל הַחֲמוֹר, מְלֵא הִתְרַגְּשׁוּת וּקְדֻשָּׁה מִבֵּית הַמִּקְדָּשׁ.”?
ה. כיצד הגיע הסיפור אל האירוע: “מִיָּד כְּשֶׁבַּעַל הַבַּיִת הִגִּיעַ, הַנָּחָשׁ גָּלַשׁ אֶל הַשָּׂדֶה וְהָאַרְיֵה חָזַר לַיַּעַר.”?

בסיפור לא התרחש הדבר הבא: כְּפִי שֶׁאָמְרוּ חֲזַ"ל בַּמִּשְׁנָה: "עוֹמְדִים צְפוּפִים – וּמִשְׁתַּחֲוִים רְוָחִים!".□□
עַל עֲרֵמַת הַחִטִּים שֶׁלּוֹ יָשַׁב אֲרִי עָצוּם שֶׁשָּׁאַג!□□

3. מה היה ראשון?

כתבו ליד כל אירוע מספר מ־1 עד 5 לפי סדר התרחשותו בסיפור.

□ וְאָז, הִגִּיעַ הָרֶגַע שֶׁבּוֹ הַכֹּהֵן הַגָּדוֹל אָמַר אֶת הַשֵּׁם הַמְפֹרָשׁ.
□ הַפַּחַד הַטִּבְעִי לַעֲזֹב בַּיִת רֵיק מֵאֲחוֹר כְּשֶׁיֵּשׁ אוֹיְבִים מִסָּבִיב.
□ בַּשִּׁעוּר הַיּוֹם נַחֲזֹר לִתְקוּפָה שֶׁכֻּלָּם עָשׂוּ אֶת זֶה בְּשִׂמְחָה!

□ הוּא מִתְקָרֵב לַשַּׁעַר וְנֶעֱצָר מֻפְתָּע מִמָּה שֶׁרָאוּ עֵינָיו.

לפני ואחרי

מה קרה לפני: “וְאָז, הִגִּיעַ הָרֶגַע שֶׁבּוֹ הַכֹּהֵן הַגָּדוֹל אָמַר אֶת הַשֵּׁם הַמְפֹרָשׁ.”?
מה קרה אחרי: “הוּא מִתְקָרֵב לַשַּׁעַר וְנֶעֱצָר מֻפְתָּע מִמָּה שֶׁרָאוּ עֵינָיו.”?

סיבה ותוצאה מתוך המעשה

האירוע הנתוןמה גרם לו?מה הייתה תוצאתו?
– הֶעָשָׁן שֶׁל עֲצֵי הַמִּזְבֵּחַ עָלָה יָשָׁר לַשָּׁמַיִם כְּמוֹ עַמּוּד שֶׁל זְכוּכִית, וְהָרוּחַ בִּכְלָל לֹא הֵזִיזָה אוֹתוֹ!
כְּפִי שֶׁאָמְרוּ חֲזַ"ל בַּמִּשְׁנָה: "עוֹמְדִים צְפוּפִים – וּמִשְׁתַּחֲוִים רְוָחִים!".

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “הבית של כלם והשומר הפלאי” על הבית של כלם והשומר הפלאי? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “הַצְּפִיפוּת הָיְתָה כָּל כָּךְ גְּדוֹלָה, עַד שֶׁהַיְהוּדִים נִדְחֲפוּ זֶה בָּזֶה, מַמָּשׁ "כָּתֵף צְמוּדָה לְכָתֵף".”.
ג. אילו הבית של כלם והשומר הפלאי לא היה פועל כפי שמתואר ב־“הַיְהוּדִי הַתָּמִים חָזַר לַכְּפָר שֶׁלּוֹ, רָכוּב עַל הַחֲמוֹר, מְלֵא הִתְרַגְּשׁוּת וּקְדֻשָּׁה מִבֵּית הַמִּקְדָּשׁ.” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “מִיָּד כְּשֶׁבַּעַל הַבַּיִת הִגִּיעַ, הַנָּחָשׁ גָּלַשׁ אֶל הַשָּׂדֶה וְהָאַרְיֵה חָזַר לַיַּעַר.” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. הַפַּחַד הַטִּבְעִי לַעֲזֹב בַּיִת רֵיק מֵאֲחוֹר כְּשֶׁיֵּשׁ אוֹיְבִים…
2. אֲבָל הַיְּהוּדִי הִבִּיט לַשָּׁמַיִם, לֹא פָּחַד כְּלָל, וְאָמַר:…
3. אֲבָל הַנֵּס הֲכִי מְרַגֵּשׁ הָיָה הַנֵּס שֶׁל הָאֲנָשִׁים הֶחָמִים…
4. כָּל אֶחָד שֶׁהִשְׁתַּחֲוָה גִּלָּה לְפֶתַע שֶׁיֵּשׁ מִסְּבִיבוֹ…
5. מֵרִים אֶת רֹאשׁוֹ וּמוֹצִיא אֶת לְשׁוֹנוֹ קָדִימָה בִּלְחִישָׁה:…
6. בַּשִּׁעוּר הַיּוֹם נַחֲזֹר לִתְקוּפָה שֶׁכֻּלָּם עָשׂוּ אֶת זֶה…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “וְלַמְרוֹת שֶׁיָּרַד גֶּשֶׁם עַז פַּעַם אַחַת, גְּשָׁמִים לְעוֹלָם לֹא כִּבּוּ אֶת מְדוּרַת עֲצֵי הַמִּזְבֵּחַ הַבּוֹעֶרֶת!”.
מכתב: כתבו להבית של כלם והשומר הפלאי בעקבות “כְּפִי שֶׁאָמְרוּ חֲזַ"ל בַּמִּשְׁנָה: "עוֹמְדִים צְפוּפִים – וּמִשְׁתַּחֲוִים רְוָחִים!".”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “עַל עֲרֵמַת הַחִטִּים שֶׁלּוֹ יָשַׁב אֲרִי עָצוּם שֶׁשָּׁאַג!”.
דיווח: כתבו ידיעה קצרה שכותרתה “הבית של כלם והשומר הפלאי”.
