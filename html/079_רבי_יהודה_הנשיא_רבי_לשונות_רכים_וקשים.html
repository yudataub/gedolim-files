לשונות רכים וקשים

דף עבודה בעקבות רבי יהודה הנשיא (רבי)
שם: כיתה:

המעשה — הסיפור המלא

להלן עיבוד חינוכי, עמוק ומרתק לסיפור המפורסם על רבי יהודה הנשיא ותלמידיו (לשונות רכים וקשים) . הסיפור מעובד בדיוק רב עבור בנים בכיתות ב'-ד' בחינוך הדתי-לאומי. הוא משלב תיאורים חושיים, בניית מתח, ומעביר את המסר העצום של קדושת הדיבור ושמירת הלשון, מתוך יראת כבוד לגדולי ישראל.

הסיפור מנוקד במלואו כנדרש.

פְּתִיחָה: תּוֹרָה וּגְדֻלָּה בְּמָקוֹם אֶחָד

יְלָדִים, אִם נַחְזֹר כִּמְעַט אַלְפַּיִם שָׁנִים אָחוֹרָה בַּזְּמַן, נַגִּיעַ לְאֶרֶץ יִשְׂרָאֵל שֶׁל יְמֵי הַמִּשְׁנָה. בְּאוֹתָם יָמִים, הַמָּקוֹם הַחָשׁוּב בְּיוֹתֵר בָּעוֹלָם כֻּלּוֹ הָיָה בֵּית הַמִּדְרָשׁ שֶׁל נְשִׂיא יִשְׂרָאֵל, גְּדוֹל הַדּוֹר הַנּוֹדָע – רַבִּי יְהוּדָה הַנָּשִׂיא. חֲכָמִים אָהֲבוּ אוֹתוֹ וְהֶעֱרִיכוּ אֶת קְדֻשָּׁתוֹ כָּל כָּךְ, עַד שֶׁקָּרְאוּ לוֹ פָּשׁוּט "רַבִּי", אוֹ "רַבֵּנוּ הַקָּדוֹשׁ". כֻּלָּם יָדְעוּ – כְּשֶׁאוֹמְרִים "רַבִּי", מִתְכַּוְּנִים אֵלָיו.

הַבַּיִת שֶׁל רַבִּי יְהוּדָה הַנָּשִׂיא לֹא הָיָה בַּיִת רָגִיל. מִלְּבַד קְדֻשָּׁתוֹ הָעֲצוּמָה, רַבִּי הָיָה אָדָם עָשִׁיר כְּקֹרַח. "תּוֹרָה וּגְדֻלָּה בְּמָקוֹם אֶחָד," כָּךְ אָמְרוּ עָלָיו. הָאַרְמוֹן שֶׁלּוֹ הָיָה מָלֵא בְּמְשָׁרְתִים, בְּכֵלִים מִכֶּסֶף וּמִזָּהָב, וּבְעֹשֶׁר שֶׁאֲפִלּוּ מַלְכֵי רוֹמִי קִנְאוּ בּוֹ. אֲבָל רַבִּי מֵעוֹלָם לֹא לָקַח מֵהָעֹשֶׁר הַזֶּה לַהֲנָאָתוֹ הָאִישִׁית. הַכֹּל, מֵהַמַּטְבֵּעַ הָרִאשׁוֹן וְעַד הָאַחֲרוֹן, הֻקְדַּשׁ רַק לַדָּבָר שֶׁהָיָה חָשׁוּב לוֹ מִכֹּל – אֶחָד הַדְּבָרִים: כְּבוֹד הַתּוֹרָה וְתַלְמִידָיו הָאֲהוּבִים. בְּכָל יוֹם אָכְלוּ עַל שֻׁלְחָנוֹ תַּלְמִידֵי חֲכָמִים רַבִּים, וְרַבִּי דָּאַג לָהֶם כְּמוֹ אָב אוֹהֵב לְבָנָיו.

הַשִּׁיעוּר בְּבֵית הַמִּדְרָשׁ

יוֹם אֶחָד, יָשַׁב רַבִּי בְּבֵית הַמִּדְרָשׁ הַגָּדוֹל, וְלִמֵּד תּוֹרָה. קוֹל הַלִּמּוּד נִשְׁמַע לְמֵרָחוֹק, וּמֵאוֹת תַּלְמִידִים יָשְׁבוּ בְּצִמָּאוֹן וְהִקְשִׁיבוּ לְכָל הֲלָכָה שֶׁיָּצְאָה מִפִּיו הַקָּדוֹשׁ. רַבִּי הִבִּיט בַּתַּלְמִידִים וְחָשַׁב בְּלִבּוֹ: "הַתַּלְמִידִים הָאֵלֶּה גְּאוֹנִים וַחֲכָמִים גְּדוֹלִים. הֵם יוֹדְעִים הֲלָכוֹת סְבוּכוֹת בְּשַׁבָּת וּבְכַשְׁרוּת, הֵם מְפַלְפְּלִים בַּתּוֹרָה. אַךְ יֵשׁ דָּבָר אֶחָד שֶׁחָשׁוּב יוֹתֵר מֵהַכֹּל – אֵיךְ הֵם מְדַבְּרִים זֶה לָזֶה! שְׁמִירַת הַלָּשׁוֹן וּכְבוֹד הַחָבֵר קוֹדְמִים לַכֹּל".

רַבִּי רָצָה לְלַמֵּד אוֹתָם אֶת הַשִּׁיעוּר הֶחָשׁוּב בְּיוֹתֵר. הוּא הֵבִין שֶׁאִם רַק יִדְרֹשׁ לָהֶם דְּרָשָׁה, הֵם יִשְׁמְעוּ וְאוּלַי יִשְׁכְּחוּ. שִׁיעוּר אֲמִתִּי נִלְמָד דֶּרֶךְ חֲוָיָה, דֶּרֶךְ מַעֲשֶׂה שֶׁיַּפְתִּיעַ אוֹתָם וְיִחָּרֵט בְּנַפְשָׁם לָנֶצַח.

הַהוֹרָאָה הַסּוֹדִית שֶׁל הַנָּשִׂיא

בְּסוֹף יּוֹם הַלִּמּוּדִים, סָגַר רַבִּי אֶת הַגְּמָרָא שֶׁלּוֹ וְאָמַר: "תַּלְמִידַי הַיְקָרִים, הַלַּיְלָה אֲנִי מַזְמִין אֶתְכֶם אֵלַי לָאַרְמוֹן. תִּתְקַיֵּם סְעוּדָה גְּדוֹלָה לִכְבוֹדָהּ שֶׁל מִצְוָה, כִּסְעוּדַת שְׁלֹמֹה בְּשַׁעְתּוֹ!". הַתַּלְמִידִים הִתְרַגְּשׁוּ מְאֹד. זְכוּת גְּדוֹלָה הִיא לֶאֱכֹל עַל שֻׁלְחַן הַנָּשִׂיא.

כְּשֶׁחָזַר רַבִּי לְבֵיתוֹ, הוּא קָרָא מִיָּד לְשַׁר הַטַּבָּחִים, הָאָחְרַאי עַל הַמִּטְבָּח הַמַּלְכוּתִי.

"שְׁמַע נָא, הַטַּבָּח הָרָאשִׁי," קוֹלוֹ שֶׁל רַבִּי הָיָה חֲרִישִׁי אַךְ בָּרוּר, "הַלַּיְלָה יָבוֹאוּ הַתַּלְמִידִים שֶׁלִּי לִסְעוּדָה. אֲנִי מְבַקֵּשׁ שֶׁתָּכִין מַאֲכָל בָּשָׂר אֶחָד, וְרַק אוֹתוֹ: לְשׁוֹנוֹת בָּקָר. אֲבָל יֵשׁ לִי בַּקָּשָׁה סוֹדִית".

הַטַּבָּח הִרְכִּין אֶת רֹאשׁוֹ.

"אֲנִי דּוֹרֵשׁ שֶׁתַּכִין אֶת הַלְּשׁוֹנוֹת בִּשְׁתֵּי קְדֵרוֹת נִפְרָדוֹת," צִוָּה רַבִּי, "בִּקְדֵרָה אַחַת תְּבַשֵּׁל אֶת הַלְּשׁוֹנוֹת תְּבַשֵּׁל בְּאֵשׁ קְטַנָּה שָׁעוֹת רַבּוֹת, תְּתַבֵּל בְּתַבְלִינִים מְשֻׁבָּחִים, עַד שֶׁהַבָּשָׂר יִהְיֶה רַךְ, נִמּוֹחַ וְטָעִים כְּמוֹ חֶמְאָה מְתוּקָה. בִּקְדֵרָה שְׁנִיָּה, אַל תְּבַשֵּׁל כִּמְעַט כְּלָל. תַּשְׁאִיר אֶת הַלְּשׁוֹנוֹת קָשׁוֹת, קְשִׁיחוֹת, חַסְרוֹת טַעַם, כָּאֵלּוּ שֶׁקָּשֶׁה מְאֹד לִלְעֹס. וּכְשֶׁתַּגִּישׁ אוֹתָם – עָרֵב אֶת הַכֹּל בְּתוֹךְ קְעָרוֹת כֶּסֶף גְּדוֹלוֹת, כָּךְ שֶׁלְּשׁוֹנוֹת רַכּוֹת וְקָשׁוֹת יִהְיוּ יַחַד מוּל כָּל תַּלְמִיד".

הַטַּבָּח הִתְפַּלֵּא מְאֹד, בַּמִּטְבַּח הַנָּשִׂיא אָסוּר לְהַגִּישׁ בָּשָׂר שֶׁאֵינוֹ מְבֻשָּׁל כָּרָאוּי! אֲבָל מִצְוַת הַנָּשִׂיא – מִצְוָה הִיא, וְהוּא מִהֵר לְקַיְּמָהּ.

סְעוּדַת הַמְּלָכִים וְהַבְּחִירָה הַטִּבְעִית

בַּלַּיְלָה, טְרַקְלִין (סָלוֹן) הַנָּשִׂיא קֻשַּׁט בִּנְבָרְשׁוֹת זָהָב גְּדוֹלוֹת שֶׁהֵפִיצוּ אוֹר צָהֹב וְחַם. רֵיחוֹת מְטֹרָפִים שֶׁל תַּבְלִינִים עַתִּיקִים, קִנָּמוֹן וְעֲצֵי בֹּשֶׂם, מִלְּאוּ אֶת הָאֲוִיר. הַתַּלְמִידִים נָטְלוּ יָדַיִם בִּקְדֻשָּׁה, בֵּרְכוּ "הַמּוֹצִיא", וְהִתְיַשְּׁבוּ עַל כִּסְאוֹת מְרֻפָּדִים מִסָּבִיב לַשֻּׁלְחָנוֹת הָעֲרוּכִים.

הַתַּלְמִידִים צָפוּ לִסְעוּדָה גְּדוֹלָה, וְהִנֵּה נִכְנְסוּ הַמְשָׁרְתִים עִם קְעָרוֹת שֶׁל כֶּסֶף טָהוֹר, שֶׁעָלָה מֵהֶן הֶבֶל חַם. בְּתוֹכָן הָיְתָה עֲרֵמָה גְּדוֹלָה שֶׁל לְשׁוֹנוֹת בָּשָׂר צָלוּי וּמְבֻשָּׁל.

הַתַּלְמִידִים, כְּדֶרֶךְ אֶרֶץ, הִמְתִּינוּ שֶׁרַבִּי יְהוּדָה יַתְחִיל, וְאָז הוֹשִׁיטוּ אֶת הַמַּזְלְגוֹת וְהַסַּכִּינִים אֶל מֶרְכַּז הַשֻּׁלְחָן. אֶחָד הַתַּלְמִידִים שָׁלַח יָד, וְנָגַע בַּחֲתִיכַת בָּשָׂר. הוּא הִרְגִּישׁ מִיָּד שֶׁהִיא נוֹקְשָׁה וְקָשָׁה כְּסוּלִיָּה שֶׁל נַעַל. הוּא עָזַב אוֹתָהּ, וְשָׁלַח יָד לַחֲתִיכָה אַחֶרֶת. אָהּ! הַחֲתִיכָה הַזּוֹ הָיְתָה רַכָּה וְנִמּוֹחָה. הוּא לָקַח אוֹתָהּ לְצַלַּחְתּוֹ וְנֶהֱנָה מִכָּל בִּיס.

גַּם תַּלְמִיד שֵׁנִי וּשְׁלִישִׁי וּרְבִיעִי עָשׂוּ אֶת אוֹתוֹ הַדָּבָר בְּדִיּוּק. הֵם בָּרְרוּ בֵּין הַחֲתִיכוֹת. כָּל מִי שֶׁנָּגַע בְּ"לָשׁוֹן" קָשָׁה – זָז מִמֶּנָּה, וּמִיָּד חִפֵּשׂ בַּקְּעָרָה וּבָחַר לוֹ אֶת הַ"לָּשׁוֹן" הָרַכָּה, הַנְּעִימָה וְהַטְּעִימָה. בְּתוֹךְ דַּקּוֹת סְפוּרוֹת, נֶעֶלְמוּ כָּל חֲתִיכוֹת הַבָּשָׂר הָרַכּוֹת מֵהַקְּעָרוֹת. מָה נִשְׁאַר בָּאֶמְצַע הַשֻּׁלְחָן? רַק עֲרֵמָה שֶׁל לְשׁוֹנוֹת קָשׁוֹת, יְבֵשׁוֹת וְעִקְּשׁוֹת שֶׁאִישׁ לֹא רָצָה בָּהֶן.

סוֹד הַלָּשׁוֹן – הַשִּׁיעוּר שֶׁל הַנָּשִׂיא

הַתַּלְמִידִים סִיְּמוּ אֶת הַמָּנָה שֶׁלָּהֶם בַּהֲנָאָה. הֵם צִפּוּ לַמָּנָה הַבָּאָה, אוּלַי עוֹפוֹת, אוּלַי דָּגִים. אֲבָל... שֶׁקֶט עָמֹק וְכָבֵד הִשְׁתָּרֵר בָּאַרְמוֹן. רַבִּי יְהוּדָה הַנָּשִׂיא קָם מִכִּסְאוֹ הַגָּדוֹל.

עֵינָיו הַחֲכָמוֹת הִבִּיטוּ בַּתַּלְמִידִים. הוּא בָּחַן אֶת הַצַּלָּחוֹת הָרֵיקוֹת, וְאֶת קְעָרוֹת הַהַגָּשָׁה שֶׁנּוֹתְרוּ בָּהֶן רַק הַלְּשׁוֹנוֹת הַקָּשׁוֹת. חִיּוּךְ רַךְ וְקָדוֹשׁ הֵאִיר אֶת פָּנָיו הַטְּהוֹרוֹת.

הוּא פָּתַח אֶת פִּיו, וְקוֹלוֹ חָדַר יָשָׁר אֶל לִבּוֹת הַתַּלְמִידִים:

"תַּלְמִידַי וּבָנַי הָאֲהוּבִים. רְאוּ מָה עֲשִׂיתֶם כָּעֵת בִּסְעוּדַתְכֶם," הִצְבִּיעַ רַבִּי עַל הַקְּעָרָה שֶׁבְּמֶרְכַּז הַשֻּׁלְחָן. "כְּשֶׁהֻגָּשׁוּ לָכֶם הַלְּשׁוֹנוֹת, מָה עֲשִׂיתֶם? בְּרַרְתֶּם וּבְרַרְתֶּם, בְּחַרְתֶּם תָּמִיד רַק אֶת הַלְּשׁוֹנוֹת הָרַכּוֹת, הַטְּעִימוֹת וְהַנְּעִימוֹת, וְהִשְׁאַרְתֶּם הַרְחֵק מִכֶּם אֶת הַלְּשׁוֹנוֹת הַקָּשׁוֹת וְהָרָעוֹת שֶׁלֹּא נָעִים לָגַעַת בָּהֶן."

רַבִּי לָקַח נְשִׁימָה עֲמֻקָּה, הֵרִים אֶת אֶצְבָּעוֹ וְאָמַר אֶת מִשְׁפַּט הַמַּפְתֵּחַ:

"דְּעוּ מָה שֶׁאַתֶּם עוֹשִׂים! יְהִי רָצוֹן שֶׁכָּךְ יִהְיוּ לְשׁוֹנוֹתֵיכֶם רַכּוֹת אֵלּוּ לָאֵלּוּ! אַף אֶחָד לֹא אוֹהֵב לָשׁוֹן קָשָׁה כְּשֶׁמְּדֻבָּר בְּאֹכֶל – וְכָךְ בְּדִיּוּק בְּנֵי אָדָם לֹא אוֹהֲבִים 'לָשׁוֹן קָשָׁה' כְּשֶׁמְּדֻבָּר בְּדִבּוּר!" .

הַתַּלְמִידִים הָיוּ הֲמוּמִים. אֵיזֶה לֶקַח עָצוּם! פִּתְאוֹם הֵם הֵבִינוּ מַדּוּעַ רַבִּי הִגִּישׁ לָהֶם כָּךְ אֶת הַמָּזוֹן. כְּשֶׁתַּלְמִיד רָב אוֹ כּוֹעֵס עַל חֲבֵרוֹ – הוּא מִשְׁתַּמֵּשׁ בְּ'לָשׁוֹן קָשָׁה'. הוּא אוֹמֵר מִלִּים פוֹגְעוֹת, עִקְּשׁוֹת, שֶׁכּוֹאֵב לִשְׁמֹעַ. מִלִּים שֶׁאִישׁ לֹא רוֹצֶה לְקַבֵּל. אֲבָל כְּשֶׁאָדָם שׁוֹמֵר עַל הַפֶּה שֶׁלּוֹ, הוּא מְדַבֵּר בְּ'לָשׁוֹן רַכָּה'. בְּמִלִּים נְעִימוֹת, עֲדִינוֹת, אוֹהֲבוֹת. אֶת הַמִּלִּים הָרַכּוֹת – כֻּלָּם רוֹצִים לִשְׁמֹעַ, בְּדִיּוּק כְּמוֹ אֶת הַבָּשָׂר הָרַךְ!

הַשִּׁיעוּר שֶׁל רַבִּי יְהוּדָה הַנָּשִׂיא חָדַר עָמֹק, אֶל לִבּוֹ שֶׁל כָּל תַּלְמִיד, שִׁיעוּר אֲמִתִּי שֶׁל "אֲהַבְתָּ לְרֵעֲךָ כָּמוֹךָ".

ME (הַדְּמוּת): רַבִּי יְהוּדָה הַנָּשִׂיא חִפֵּשׂ לְהַעֲבִיר מֶסֶר מֻחֲשִׁי שֶׁל "שְׁמִירַת הַלָּשׁוֹן". הוּא לֹא סְתָם צָעַק וְאָמַר "אָסוּר לְכַנּוֹת שֵׁמוֹת! אָסוּר לְהַעֲלִיב!". הוּא הֶרְאָה לָהֶם דֶּרֶךְ סְעוּדָה, שֶׁכְּמוֹ שֶׁבָּשָׂר קָשֶׁה מַגְעִיל אוֹתָנוּ – כָּךְ מִלִּים קָשׁוֹת גּוֹרְמוֹת לָאָדָם הַשֵּׁנִי לְהִסְתַּיֵּג מֵאִתָּנוּ.

WE (אֲנַחְנוּ הַיּוֹם): כַּמָּה פְּעָמִים אֲנַחְנוּ מִתְעַצְבְּנִים בַּהַפְסָקָה אוֹ בַּבַּיִת? כְּשֶׁחָבֵר עוֹשֶׂה מַשֶּׁהוּ מַרְגִּיז אֲנַחְנוּ מִיָּד מוֹצִיאִים מֵהַפֶּה "לָשׁוֹן קָשָׁה": קְלָלוֹת, זִלְזוּל, שֵׁמוֹת גְּנַאי. אֲנַחְנוּ שׁוֹכְחִים שֶׁהַמִּלִּים הָאֵלּוּ הֵן כְּמוֹ כַּדּוּרֵי בַּרְזֶל שֶׁפּוֹגְעִים בַּנְּשָׁמָה שֶׁל הַשֵּׁנִי.

YOU (אַתָּה מָחָר בַּבֹּקֶר): מָחָר, כְּשֶׁתִּרְצֶה לְהָעִיר לֶחָבֵר עַל מַשֶּׁהוּ שֶׁהוּא טָעָה בּוֹ, אוֹ כְּשֶׁתִּרְצֶה עֶזְרָה מֵהוֹרֶה – תִּזְכֹּר אֶת "סְעוּדַת הַנָּשִׂיא". תַּעֲצֹר שְׁנִיָּה מִיּוֹתֶרֶת, תִּנְשֹׁם, וְתִבְחַר לְהוֹצִיא "לָשׁוֹן רַכָּה", דִּבּוּר נָעִים וּמְכַבֵּד. אַתָּה תִּרְאֶה שֶׁכָּךְ כֻּלָּם יִשְׂמְחוּ לְהַקְשִׁיב לְךָ וּלְאֱהֹב אוֹתְךָ.

א. שְׁאֵלַת פְּתִיחָה (לְעוֹרֵר זִהוּי):

"ילדים, אני רוצה לשאול אתכם שאלת חידה מעניינת על הגוף שלנו. מה לדעתכם האיבר הכי חזק שיש לנו בגוף? (לחכות לתשובות – שרירים בידיים? רגליים?) לא תאמינו – אבל חכמים מלמדים אותנו שהאיבר הכי חזק הוא... הלשון! כן, היא יכולה להרים אדם בשמחה, או לרסק אותו בעצב. היום נשמע על מבחן מסתורי שעשה גדול ישראל לתלמידיו כדי ללמד אותם על הכוח הזה."

ב. הֶעָרוֹת בִּצּוּעַ וּסְטוֹרִיטֶלִינְג לַמְסַפֵּר:

ג. הַמְלָצוֹת לְמַצֶּגֶת (תְּמוּנוֹת מְלַוּוֹת בִּקְדֻשָּׁה):

ד. פְּעִילוּת מְסַכֶּמֶת ("לָשׁוֹן רַכָּה, לָשׁוֹן קָשָׁה"):

מקורות: מדרש רבה: ויקרא רבה, פרשה ל"ג, אות א'. (וכן מופיע במדרש קהלת רבה ובספר האגדה, סימן שי"א). מסכת עבודה זרה (י"א ע"א): עושרו הגדול של רבי יהודה הנשיא ("לא פסק מעל שולחנו..."). "כה עשו חכמינו" (א', 148): עיבוד קלאסי לסיפור הלשונות הרכים והקשים, המדגיש את החינוך לשמירת הלשון.

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “להלן עיבוד חינוכי, עמוק ומרתק לסיפור המפורסם על רבי יהודה הנשיא ותלמידיו (לשונות רכים וקשים) .” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “בְּכָל יוֹם אָכְלוּ עַל שֻׁלְחָנוֹ תַּלְמִידֵי חֲכָמִים רַבִּים, וְרַבִּי דָּאַג לָהֶם כְּמוֹ אָב אוֹהֵב לְבָנָיו.”?
ג. כיצד הגיב רבי יהודה הנשיא (רבי) למצב המתואר: “שְׁמַע נָא, הַטַּבָּח הָרָאשִׁי," קוֹלוֹ שֶׁל רַבִּי הָיָה חֲרִישִׁי אַךְ בָּרוּר, "הַלַּיְלָה יָבוֹאוּ הַתַּלְמִידִים שֶׁלִּי לִסְעוּדָה.”?
ד. מה השתנה בין “הַתַּלְמִידִים, כְּדֶרֶךְ אֶרֶץ, הִמְתִּינוּ שֶׁרַבִּי יְהוּדָה יַתְחִיל, וְאָז הוֹשִׁיטוּ אֶת הַמַּזְלְגוֹת…” לבין “רַבִּי לָקַח נְשִׁימָה עֲמֻקָּה, הֵרִים אֶת אֶצְבָּעוֹ וְאָמַר אֶת מִשְׁפַּט הַמַּפְתֵּחַ:”?
ה. כיצד הגיע הסיפור אל האירוע: “כְּשֶׁחָבֵר עוֹשֶׂה מַשֶּׁהוּ מַרְגִּיז אֲנַחְנוּ מִיָּד מוֹצִיאִים מֵהַפֶּה "לָשׁוֹן קָשָׁה": קְלָלוֹת, זִלְזוּל, שֵׁמוֹת גְּנַאי.”?

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “לשונות רכים וקשים” על רבי יהודה הנשיא (רבי)? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “אֲבָל מִצְוַת הַנָּשִׂיא – מִצְוָה הִיא, וְהוּא מִהֵר לְקַיְּמָהּ.”.
ג. אילו רבי יהודה הנשיא (רבי) לא היה פועל כפי שמתואר ב־“רַבִּי לָקַח נְשִׁימָה עֲמֻקָּה, הֵרִים אֶת אֶצְבָּעוֹ וְאָמַר אֶת מִשְׁפַּט הַמַּפְתֵּחַ:” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “כְּשֶׁחָבֵר עוֹשֶׂה מַשֶּׁהוּ מַרְגִּיז אֲנַחְנוּ מִיָּד מוֹצִיאִים מֵהַפֶּה "לָשׁוֹן קָשָׁה": קְלָלוֹת, זִלְזוּל, שֵׁמוֹת…” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. להלן עיבוד חינוכי, עמוק ומרתק לסיפור המפורסם על רבי יהודה הנשיא…
2. יוֹם אֶחָד, יָשַׁב רַבִּי בְּבֵית הַמִּדְרָשׁ הַגָּדוֹל, וְלִמֵּד…
3. תַּשְׁאִיר אֶת הַלְּשׁוֹנוֹת קָשׁוֹת, קְשִׁיחוֹת, חַסְרוֹת טַעַם,…
4. כָּל מִי שֶׁנָּגַע בְּ"לָשׁוֹן" קָשָׁה – זָז מִמֶּנָּה, וּמִיָּד…
5. אֲבָל כְּשֶׁאָדָם שׁוֹמֵר עַל הַפֶּה שֶׁלּוֹ, הוּא מְדַבֵּר בְּ'לָשׁוֹן…
6. היום נשמע על מבחן מסתורי שעשה גדול ישראל לתלמידיו כדי ללמד אותם על הכוח…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “שְׁמַע נָא, הַטַּבָּח הָרָאשִׁי," קוֹלוֹ שֶׁל רַבִּי הָיָה חֲרִישִׁי אַךְ בָּרוּר, "הַלַּיְלָה יָבוֹאוּ הַתַּלְמִידִים שֶׁלִּי…”.
מכתב: כתבו לרבי יהודה הנשיא (רבי) בעקבות “רַק עֲרֵמָה שֶׁל לְשׁוֹנוֹת קָשׁוֹת, יְבֵשׁוֹת וְעִקְּשׁוֹת שֶׁאִישׁ לֹא רָצָה בָּהֶן.”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “אֶת הַמִּלִּים הָרַכּוֹת – כֻּלָּם רוֹצִים לִשְׁמֹעַ, בְּדִיּוּק כְּמוֹ אֶת הַבָּשָׂר הָרַךְ!”.
דיווח: כתבו ידיעה קצרה שכותרתה “לשונות רכים וקשים”.
