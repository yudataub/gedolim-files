"חטאים ולא חוטאים" — הגישה לחוטאים

דף עבודה בעקבות רבי מאיר וברוריה
שם: כיתה:

המעשה — הסיפור המלא

להלן מערך שיעור וסיפור חינוכי, רגשי ומרתק על רבי מאיר בעל הנס ואשתו ברוריה . הסיפור מעובד במיוחד לבנים בכיתות ב'-ד' (חינוך דתי-לאומי), תוך שימוש בעקרונות הסטוריטלינג (Show, Don't Tell), תיאורים חושיים, והמחשת מוסר ההשכל: לשנוא את החטא, אך לאהוב את החוטא ולהתפלל עליו שיחזור בתשובה.

הסיפור מנוקד במלואו, כנדרש.

פְּתִיחָה: הָאוֹר שֶׁל רַבִּי מֵאִיר וְהָרַעַשׁ שֶׁבַּחוּץ

יְלָדִים, דַּמְיְנוּ עֶרֶב יְרוּשַׁלְמִי שָׁקֵט. בַּחוּץ כְּבָר יָרַד הַחֹשֶׁךְ, אַךְ מֵחַלּוֹן אֶחָד קָטָן בָּקַע אוֹר צָהֹב וְחַם. זֶה הָיָה חַלּוֹן בֵּיתוֹ שֶׁל הַתַּנָּא הַקָּדוֹשׁ, רַבִּי מֵאִיר בַּעַל הַנֵּס .

בְּתוֹךְ הַחֶדֶר הַפָּשׁוּט הָיָה רֵיחַ נָעִים שֶׁל קְלָפִים עַתִּיקִים מֵעוֹר שֶׁל צְבִי (שֶׁכֵּן רַבִּי מֵאִיר הָיָה סוֹפֵר סְתָ"ם כָּתַב סִפְרֵי תּוֹרָה). עַל הַשֻּׁלְחָן דָּלַק נֵר קָטָן, וְרַבִּי מֵאִיר הָיָה רָכוּן עַל הַסְּפָרִים, לוֹמֵד בְּקוֹל עָרֵב וּמָתוֹק כִּדְבַשׁ: "אָמַר רַבִּי עֲקִיבָא... תָּנוּ רַבָּנַן...". הַקּוֹל שֶׁלּוֹ הָיָה כָּל כָּךְ נָעִים, עַד שֶׁמִּי שֶׁהָיָה שׁוֹמֵעַ אוֹתוֹ הָיָה מַרְגִּישׁ שֶׁהַלֵּב שֶׁלּוֹ מִתְמַלֵּא בְּשִׂמְחָה שֶׁל קְדֻשָּׁה.

לְצִדּוֹ, בַּחֶדֶר הַסָּמוּךְ, יָשְׁבָה אִשְׁתּוֹ, הָרַבָּנִית בְּרוּרְיָה . בְּרוּרְיָה הָיְתָה אִשָּׁה חֲכָמָה מְאֹד מְאֹד, בִּתּוֹ שֶׁל רַבִּי חֲנִינָא בֶּן תְּרַדְיוֹן. גַּם הִיא אָהֲבָה אֶת הַתּוֹרָה בְּכָל נַפְשָׁהּ, וְהָיְתָה מַקְשִׁיבָה בְּנַחַת לְקוֹל לִמּוּדוֹ שֶׁל בַּעֲלָהּ הַגָּדוֹל.

אֲבָל... הַשֶּׁקֶט הַזֶּה לֹא נִמְשַׁךְ זְמַן רַב.

פִּתְאוֹם – בּוּם! טְרָאח! קוֹלוֹת שֶׁל נִפּוּץ זְכוּכִיּוֹת נִשְׁמְעוּ מִבַּחוּץ. צְעָקוֹת רָמוֹת, שִׁירִים מְכֹעָרִים וּפְרָאִיִּים קָרְעוּ אֶת מְנוּחַת הַלַּיְלָה.

הַשְּׁכֵנִים הַבִּרְיוֹנִים

בַּשְּׁכוּנָה שֶׁל רַבִּי מֵאִיר הִתְגּוֹרְרָה קְבוּצָה שֶׁל "בִּרְיוֹנִים". אֵלֶּה הָיוּ אֲנָשִׁים יְהוּדִים, אֲבָל הֵם עָזְבוּ אֶת דֶּרֶךְ הַתּוֹרָה. הֵם הָיוּ מִשְׁתַּכְּרִים, כּוֹעֲסִים, מַרְבִּיצִים, וְעוֹשִׂים מַעֲשִׂים רָעִים מְאֹד.

רַבִּי מֵאִיר נִסָּה לַחְזֹר לִלְמֹד. "אָמַר רַבִּי עֲקִיבָא..." הוּא לָחַשׁ, אֲבָל קוֹלוֹ נִבְלַע בְּתוֹךְ קוֹלוֹת הַצְּחוֹק הָרַע שֶׁל הַבִּרְיוֹנִים מֵהָרְחוֹב. רֵיחַ רַע שֶׁל עָשָׁן וְיַיִן מְקֻלְקָל חָדַר דֶּרֶךְ הַחַלּוֹן. קִירוֹת הָאֶבֶן שֶׁל הַבַּיִת מַמָּשׁ רָעֲדוּ מֵהַדְּפִיקוֹת שֶׁלָּהֶם עַל הַדְּלָתוֹת בַּשְּׁכוּנָה.

זֶה לֹא קָרָה רַק לַיְלָה אֶחָד. זֶה קָרָה יוֹם אַחַר יוֹם! הַבִּרְיוֹנִים הָאֵלֶּה צִעֲרוּ אֶת כָּל תּוֹשָׁבֵי הַשְּׁכוּנָה, וּבִמְיֻחָד עָשׂוּ צַעַר גָּדוֹל לְרַבִּי מֵאִיר. הֵם הִפְרִיעוּ לוֹ לִלְמֹד, וְיוֹתֵר מִכָּךְ – לִבּוֹ שֶׁל רַבִּי מֵאִיר כָּאַב לִרְאוֹת יְהוּדִים מִתְנַהֲגִים בְּצוּרָה כָּל כָּךְ נְמוּכָה, כָּל כָּךְ לֹא מְכַבֶּדֶת אֶת ה' יִתְבָּרַךְ!

הַהַחְלָטָה שֶׁל הַצַּדִּיק

בְּאוֹתוֹ לַיְלָה, רַבִּי מֵאִיר סָגַר אֶת הַסֵּפֶר. הַפָּנִים שֶׁלּוֹ, שֶׁתָּמִיד הָיוּ מְאִירוֹת, נַעֲשׂוּ רְצִינִיּוֹת. הוּא הֵנִיחַ אֶת רֹאשׁוֹ בֵּין יָדָיו.

"אִי אֶפְשָׁר לְהַמְשִׁיךְ כָּךְ," לָחַשׁ רַבִּי מֵאִיר, "הֵם מְחַלְּלִים מַמָּשׁ אֶת שֵׁם ה'! הֵם מַפְרִיעִים לַתּוֹרָה! אֵין לָהֶם תַּקָּנָה".

רַבִּי מֵאִיר יָדַע שֶׁיֵּשׁ לוֹ כֹּחַ גָּדוֹל. מָה קוֹרֶה כְּשֶׁצַּדִּיק שֶׁל כָּל עַם יִשְׂרָאֵל, שֶׁלּוֹמֵד תּוֹרָה יוֹמָם וָלַיְלָה, מְבַקֵּשׁ בַּקָּשָׁה מֵהַקָּדוֹשׁ בָּרוּךְ הוּא? הַקָּדוֹשׁ בָּרוּךְ הוּא שׁוֹמֵעַ לוֹ! כְּמוֹ שֶׁכָּתוּב: "רְצוֹן יְרֵאָיו יַעֲשֶׂה".

רַבִּי מֵאִיר עָצַם אֶת עֵינָיו. הוּא נָשָׂא אֶת יָדָיו כְּלַפֵּי הַשָּׁמַיִם, וְהֶחְלִיט לְהִתְפַּלֵּל לַה' בַּקָּשָׁה קָשָׁה: שֶׁהַבִּרְיוֹנִים הָאֵלֶּה יָמוּתוּ וְיִסְתַּלְּקוּ מִן הָעוֹלָם, כְּדֵי שֶׁלֹּא יַעֲשׂוּ יוֹתֵר רַע.

"יְהִי רָצוֹן מִלְּפָנֶיךָ רִבּוֹנוֹ שֶׁל עוֹלָם..." הִתְחִיל רַבִּי מֵאִיר לְהִתְפַּלֵּל עֲלֵיהֶם דִּין רָע מִן הַשָּׁמַיִם. אִם הָרְשָׁעִים יֵעָלְמוּ – הַשְּׁכוּנָה תִּהְיֶה טְהוֹרָה!

הָעֵצָה הַגְּאוֹנִית שֶׁל בְּרוּרְיָה

אֲבָל אָז... פְּסִיעוֹת חֲרִישִׁיּוֹת . דֶּלֶת הַחֶדֶר נִפְתְּחָה בַּעֲדִינוּת.

בְּרוּרְיָה עָמְדָה בַּפֶּתַח. הִיא שָׁמְעָה אֶת תְּפִלָּתוֹ שֶׁל בַּעֲלָהּ, וְהַלֵּב הַטָּהוֹר שֶׁלָּהּ נִזְעַק. הִיא הִכִּירָה אֶת סִפְרֵי הַקֹּדֶשׁ עַל בֻּרְיָם, וְיָדְעָה שֶׁה' אֵינוֹ חָפֵץ בְּמוֹתָם שֶׁל רְשָׁעִים, אֶלָּא שֶׁיַּחְזְרוּ בִּתְשׁוּבָה.

"מָאִיר שֶׁלִּי," הִיא פָנְתָה אֵלָיו בְּקוֹל שָׁקֵט אַךְ בָּטוּחַ. "מָה שֶׁהִנְּךָ עוֹשֶׂה... מַדּוּעַ אַתָּה מִתְפַּלֵּל שֶׁיָּמוּתוּ?"

רַבִּי מֵאִיר פָּקַח אֶת עֵינָיו וְעָנָה בְּצַעַר: "כִּי הֵם אֲנָשִׁים מְרֻשָּׁעִים! דָּוִד הַמֶּלֶךְ אָמַר בְּסֵפֶר תְּהִלִּים (ק"ד, ל"ה): 'יִתַּמּוּ חַטָּאִים מִן הָאָרֶץ, וּרְשָׁעִים עוֹד אֵינָם'. זֶה מָה שֶׁכָּתוּב! דָּוִד הַמֶּלֶךְ מְבַקֵּשׁ שֶׁהַחוֹטְאִים יִתַּמּוּ, שֶׁיֵּעָלְמוּ הַחוֹטְאִים מִן הָאָרֶץ!".

חִיְּכָה בְּרוּרְיָה חִיּוּךְ שֶׁל אִמָּא הָאוֹהֶבֶת אֶת כָּל יַלְדֵי יִשְׂרָאֵל, וְאָמְרָה אֶת הַמִּשְׁפָּט הֶחָכָם בְּיוֹתֵר:

"רַבִּי מֵאִיר... תִּסְתַּכֵּל שׁוּב בַּפָּסוּק. הַאִם כָּתוּב 'יִתַּמּוּ חוֹטְאִים '?".

הִיא הִמְתִּינָה שְׁנִיָּה, וְאָז עָנְתָה בְּעַצְמָהּ: "לֹא! לֹא כָּתוּב 'חוֹטְאִים'. כָּתוּב 'יִתַּמּוּ חַטָּאִים' ! חֲטָאִים – בְּלִי וָא"ו. דָּוִד הַמֶּלֶךְ אֵינוֹ מְבַקֵּשׁ שֶׁיָּמוּתוּ הָאֲנָשִׁים! הָאֲנָשִׁים הֵם נְשָׁמוֹת קְדוֹשׁוֹת שֶׁל ה'. דָּוִד הַמֶּלֶךְ מְבַקֵּשׁ שֶׁיֵּעָלְמוּ הַ חֲטָאִים , שֶׁהַמַּעֲשִׂים הָרָעִים יֵעָלְמוּ מִן הָעוֹלָם!".

רַבִּי מֵאִיר הִקְשִׁיב בְּתַדְהֵמָה. בְּרוּרְיָה הִמְשִׁיכָה:

"וְתִרְאֶה אֶת סוֹף הַפָּסוּק: 'וּרְשָׁעִים – עוֹד אֵינָם'. לָמָּה הֵם כְּבָר אֵינָם רְשָׁעִים? כִּי בָּרֶגַע שֶׁהַ חֲטָאִים שֶׁלָּהֶם נֶעֱלָמִים, הֵם כְּבָר לֹא הַרְשָׁעִים, הֵם חוֹזְרִים לִהְיוֹת צַדִּיקִים!

לָכֵן," אָמְרָה בְּרוּרְיָה בְּקוֹל רַךְ וְאוֹהֵב, "אַל תִּתְפַּלֵּל שֶׁהֵם יָמוּתוּ חָלִילָה. תִּתְפַּלֵּל עֲלֵיהֶם שֶׁיַּחְזְרוּ בִּתְשׁוּבָה! אִם הֵם יַחְזְרוּ בִּתְשׁוּבָה, הֵם כְּבָר לֹא יִהְיוּ רְשָׁעִים, וְהַבְּעָיָה תִּפָּתֵר בְּצוּרָה הַרְבֵּה יוֹתֵר טוֹבָה!".

הַנִּצָּחוֹן שֶׁל הָאַהֲבָה

יְלָדִים, אָדָם רָגִיל שֶׁכּוֹעֵס וְעָבְרוּ לוֹ עַל הָעֲצַבִּים הָיָה אוֹמֵר: "מָה אַתְּ מִתְעָרֶבֶת? עֲזִיבִי אוֹתִי, אֲנִי כּוֹעֵס עֲלֵיהֶם!".

אֲבָל רַבִּי מֵאִיר בַּעַל הַנֵּס הָיָה עָנָיו וֶאֱמֶת תָּמִיד הָיְתָה נֵר לְרַגְלָיו. הוּא שָׁמַע אֶת הַסְּבָרָה שֶׁל אִשְׁתּוֹ, וּמִיָּד פָּנָיו הֵאִירוּ מֵחָדָשׁ. "צָדַקְתְּ, בְּרוּרְיָה. צָדַקְתְּ," הוּא לָחַשׁ.

רַבִּי מֵאִיר עָצַם שׁוּב אֶת עֵינָיו. קוֹלוֹת הַנֶּפֶּץ שֶׁל הַבִּרְיוֹנִים עֲדַיִן נִשְׁמְעוּ בַּחוּץ, אֲבָל עַכְשָׁו, בִּמְקוֹם לִרְאוֹת אוֹיְבִים, רַבִּי מֵאִיר רָאָה מִסְכֵּנִים. הוּא רָאָה יְהוּדִים אֲבוּדִים שֶׁהַיֵּצֶר הָרַע אָחַז בָּהֶם.

רַבִּי מֵאִיר הִתְפַּלֵּל בְּדִמְעוֹת שֶׁל אַהֲבָה: "רִבּוֹנוֹ שֶׁל עוֹלָם, אָב הָרַחֲמָן, אָנָּא מִמְּךָ – פְּתַח אֶת לִבָּם שֶׁל הַבִּרְיוֹנִים הָאֵלֶּה. הָסֵר אֶת יֵצֶר הָרַע מֵהֶם, זַכֵּה אוֹתָם לַחְזֹר לְדֶרֶךְ טוֹבָה, אֶל הַתּוֹרָה וְהַמִּצְווֹת".

וְיְלָדִים, אַתֶּם יוֹדְעִים מָה קָרָה?

הַשְּׁגָחָה הָעֶלְיוֹנָה לֹא אֵחֲרָה לָבוֹא. כַּאֲשֶׁר צַדִּיק מִתְפַּלֵּל מִתּוֹךְ אַהֲבַת יִשְׂרָאֵל אֲמִתִּית – זֶה פּוֹעֵל מִיָּד!

בְּאוֹתָם יָמִים הַבִּרְיוֹנִים שֶׁבַּחוּץ הִתְחִילוּ לְהַרְגִּישׁ מַשֶּׁהוּ מוּזָר. פִּתְאוֹם נִהְיָה לָהֶם רַע לַעֲשׂוֹת מַעֲשִׂים מְכֹעָרִים. הַלֵּב שֶׁלָּהֶם, שֶׁהָיָה קָשֶׁה כְּאֶבֶן, פִּתְאוֹם נִהְיָה רַךְ. הֵם הִתְחִילוּ לְהִתְחָרֵט.

הֵם זָרְקוּ אֶת כּוֹסוֹת הַיַּיִן מֵהַיָּדַיִם. הֵם הִפְסִיקוּ לִצְעֹק. הֵם חָזְרוּ בִּתְשׁוּבָה שְׁלֵמָה!

הַשְּׁכוּנָה נִהְיְתָה שְׁקֵטָה. יוֹתֵר מִכָּךְ, קוֹל תּוֹרָה וּתְפִלָּה נִשְׁמַע גַּם מֵהַבָּתִּים שֶׁלָּהֶם.

הַכֹּל – בִּזְכוּת מִשְׁפָּט אֶחָד שֶׁל בְּרוּרְיָה הַחֲכָמָה: "חֲטָאִים – וְלֹא חוֹטְאִים!" .

א. שְׁאֵלַת פְּתִיחָה (לְעוֹרֵר סַקְרָנוּת הַכִּתָּה):

"יְלָדִים, הַאִם קָרָה לָכֶם פַּעַם שֶׁמִּישֶׁהוּ כָּל כָּךְ הִצִּיק לָכֶם, הִפְרִיעַ לָכֶם לְשַׂחֵק אוֹ לִלְמֹד, עַד שֶׁרְצִיתֶם שֶׁהוּא פָּשׁוּט... יֵעָלֵם?! שֶׁיַּעֲבֹר בֵּית סֵפֶר וְלֹא תִּרְאוּ אוֹתוֹ יוֹתֵר?

הַיּוֹם נִשְׁמַע עַל צַדִּיק עָצוּם, רַבִּי מֵאִיר, שֶׁעָבַר דָּבָר דּוֹמֶה, וְעַל אִשְׁתּוֹ, הָרַבָּנִית בְּרוּרְיָה, שֶׁלִּמְּדָה אוֹתוֹ (וְאוֹתָנוּ) סוֹד עֲנָק לַחַיִּים."

ב. הֶעָרוֹת בִּצּוּעַ וּסְטוֹרִיטֶלִינְג (אֵיךְ לְסַפֵּר בַּכִּתָּה):

ג. הַמְלָצוֹת לְמַצֶּגֶת (4 שְׁקוּפִיּוֹת בְּהַקְפָּדָה עַל צְנִיעוּת):

ד. פְּעִילוּת מְסַכֶּמֶת (לַכִּתָּה): "מִשְׁקְפֵי מֵאִיר וּבְרוּרְיָה"

מביאים לכיתה זוג משקפיים (מפלסטיק או מצוירים על דף שמעבירים).

המשחק: מציגים בפני התלמידים "מעשה שלילי" פשוט של חבר (למשל: "ילד חטף לי את הכדור בכוונה בהפסקה").

הילד שמרכיב את ה"משקפיים" צריך להגיד שני דברים:

כך מפתחים את שריר הראייה שמפריד בין המעשה לבין האדם.

מקורות: תלמוד בבלי, מסכת ברכות, דף י', עמוד א': "הנהו בריוני דהוו בשבבותיה דרבי מאיר והוו קא מצערו ליה טובא. הוה קא בעי רבי מאיר רחמי עלויהו כי היכי דלימותו. אמרה ליה ברוריא דביתהו: מאי דעתך? משום דכתיב (תהלים קד, לה) 'יתמו חטאים מן הארץ'? מי כתיב חוטאים? חטאים כתיב! ועוד, שפיל לסיפיה דקרא 'ורשעים עוד אינם' - כיון דתמו חטאים ורשעים עוד אינם? אלא, בעי רחמי עלויהו דלהדרו בתשובה, ורשעים עוד אינם! בעא רחמי עלויהו והדרו בתשובה." האגדה רו / ספר עֹז 163-164: העוסקים בגדולתה של ברוריה והיחס הנכון לחוטאים בישראל (הפרדה בין מעשה לאדם).

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “להלן מערך שיעור וסיפור חינוכי, רגשי ומרתק על רבי מאיר בעל הנס ואשתו ברוריה .” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “גַּם הִיא אָהֲבָה אֶת הַתּוֹרָה בְּכָל נַפְשָׁהּ, וְהָיְתָה מַקְשִׁיבָה בְּנַחַת לְקוֹל לִמּוּדוֹ שֶׁל בַּעֲלָהּ הַגָּדוֹל.”?
ג. כיצד הגיב רבי מאיר וברוריה למצב המתואר: “הַבִּרְיוֹנִים הָאֵלֶּה צִעֲרוּ אֶת כָּל תּוֹשָׁבֵי הַשְּׁכוּנָה, וּבִמְיֻחָד עָשׂוּ צַעַר גָּדוֹל לְרַבִּי מֵאִיר.”?
ד. מה השתנה בין “דָּוִד הַמֶּלֶךְ אָמַר בְּסֵפֶר תְּהִלִּים (ק"ד, ל"ה): 'יִתַּמּוּ חַטָּאִים מִן הָאָרֶץ, וּרְשָׁעִים עוֹד אֵינָם'.” לבין “הוּא שָׁמַע אֶת הַסְּבָרָה שֶׁל אִשְׁתּוֹ, וּמִיָּד פָּנָיו הֵאִירוּ מֵחָדָשׁ.”?
ה. כיצד הגיע הסיפור אל האירוע: “הַכֹּל – בִּזְכוּת מִשְׁפָּט אֶחָד שֶׁל בְּרוּרְיָה הַחֲכָמָה: "חֲטָאִים – וְלֹא חוֹטְאִים!" .”?

לפני ואחרי

מה קרה לפני: “מָאִיר שֶׁלִּי," הִיא פָנְתָה אֵלָיו בְּקוֹל שָׁקֵט אַךְ בָּטוּחַ.”?
מה קרה אחרי: “קוֹלוֹת הַנֶּפֶּץ שֶׁל הַבִּרְיוֹנִים עֲדַיִן נִשְׁמְעוּ בַּחוּץ, אֲבָל עַכְשָׁו, בִּמְקוֹם לִרְאוֹת אוֹיְבִים, רַבִּי מֵאִיר רָאָה מִסְכֵּנִים.”?

סיבה ותוצאה מתוך המעשה

האירוע הנתוןמה גרם לו?מה הייתה תוצאתו?
הֵם הָיוּ מִשְׁתַּכְּרִים, כּוֹעֲסִים, מַרְבִּיצִים, וְעוֹשִׂים מַעֲשִׂים רָעִים מְאֹד.
כִּי בָּרֶגַע שֶׁהַ חֲטָאִים שֶׁלָּהֶם נֶעֱלָמִים, הֵם כְּבָר לֹא הַרְשָׁעִים, הֵם חוֹזְרִים לִהְיוֹת צַדִּיקִים!

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “"חטאים ולא חוטאים" — הגישה לחוטאים” על רבי מאיר וברוריה? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “הִיא שָׁמְעָה אֶת תְּפִלָּתוֹ שֶׁל בַּעֲלָהּ, וְהַלֵּב הַטָּהוֹר שֶׁלָּהּ נִזְעַק.”.
ג. אילו רבי מאיר וברוריה לא היה פועל כפי שמתואר ב־“הוּא שָׁמַע אֶת הַסְּבָרָה שֶׁל אִשְׁתּוֹ, וּמִיָּד פָּנָיו הֵאִירוּ מֵחָדָשׁ.” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “הַכֹּל – בִּזְכוּת מִשְׁפָּט אֶחָד שֶׁל בְּרוּרְיָה הַחֲכָמָה: "חֲטָאִים – וְלֹא חוֹטְאִים!" .” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. להלן מערך שיעור וסיפור חינוכי, רגשי ומרתק על רבי מאיר בעל הנס ואשתו…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “הַבִּרְיוֹנִים הָאֵלֶּה צִעֲרוּ אֶת כָּל תּוֹשָׁבֵי הַשְּׁכוּנָה, וּבִמְיֻחָד עָשׂוּ צַעַר גָּדוֹל לְרַבִּי מֵאִיר.”.
מכתב: כתבו לרבי מאיר וברוריה בעקבות “כִּי בָּרֶגַע שֶׁהַ חֲטָאִים שֶׁלָּהֶם נֶעֱלָמִים, הֵם כְּבָר לֹא הַרְשָׁעִים, הֵם חוֹזְרִים לִהְיוֹת צַדִּיקִים!”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “הָסֵר אֶת יֵצֶר הָרַע מֵהֶם, זַכֵּה אוֹתָם לַחְזֹר לְדֶרֶךְ טוֹבָה, אֶל הַתּוֹרָה וְהַמִּצְווֹת".”.
דיווח: כתבו ידיעה קצרה שכותרתה “"חטאים ולא חוטאים" — הגישה לחוטאים”.
