האיש המכוער — כבוד הבריות וחזרה בתשובה

דף עבודה בעקבות רבי שמעון בן ר' אלעזר
שם: כיתה:

המעשה — הסיפור המלא

פְּתִיחָה: רֵיחַ שֶׁל תּוֹרָה בָּאֲוִיר

יְלָדִים, צְאוּ אִתִּי לְמַסָּע בִּזְמַן. אֲנַחְנוּ נִמְצָאִים בְּאֶרֶץ יִשְׂרָאֵל, לִפְנֵי כְּאַלְפַּיִם שָׁנָה. הַיּוֹם בָּהִיר מְאֹד. רוּחַ קְרִירָה נוֹשֶׁבֶת מִן הֶהָרִים, מֵבִיאָה אִתָּהּ רֵיחַ מָתוֹק שֶׁל עֲצֵי תְּאֵנָה וְשָׂדוֹת יְרֻקִּים. עַל שְׂפַת הַנָּהָר הַזּוֹרֵם, רוֹכֵב לוֹ בְּנַחַת רַבִּי שִׁמְעוֹן בֶּן רַבִּי אֶלְעָזָר.

טְרִיק... טְרָק... טְרִיק... טְרָק... פַּרְסוֹת חֲמוֹרוֹ נוֹקְשׁוֹת עַל אַבְנֵי הַשְּׁבִיל. הַמַּיִם הַצְּלוּלִים מְפַכְפְּכִים לְצִדּוֹ, אֲבָל רַבִּי שִׁמְעוֹן כִּמְעַט וְאֵינוֹ שָׂם לֵב לַנּוֹף. הַלֵּב שֶׁלּוֹ כֻּלּוֹ שָׁקוּעַ בְּמַשֶּׁהוּ אַחֵר לְגַמְרֵי.

הוּא בְּדִיּוּק חוֹזֵר מִמִּגְדַּל גְּדוֹר, מִבֵּיתוֹ שֶׁל רַבּוֹ הַגָּדוֹל, שָׁם יָשַׁב יָמִים וְלֵילוֹת וְלָמַד תּוֹרָה בְּלִי הַפְסָקָה. הָרֹאשׁ שֶׁלּוֹ מָלֵא בַּהֲלָכוֹת שֶׁלָּמַד, בִּפְסוּקִים זְהוּבִים שֶׁהֵאִירוּ אֶת נִשְׁמָתוֹ. "מָה אָהַבְתִּי תוֹרָתֶךָ כָּל הַיּוֹם הִיא שִׂיחָתִי," הוּא לוֹחֵשׁ לְעַצְמוֹ בְּנִגּוּן חֲרִישִׁי, וְשִׂמְחָה עֲצוּמָה מְמַלֵּאת אֶת חָזֵהוּ.

אֲבָל... יְלָדִים, לִפְעָמִים, מִתּוֹךְ רֹב רֹבָהּ שֶׁל שִׂמְחָה, מִתּוֹךְ תְּחוּשָׁה שֶׁל 'כַּמָּה הַרְבֵּה דְּבָרִים נִפְלָאִים אֲנִי יוֹדֵעַ', יָכוֹל הַיֵּצֶר הָרַע לְהִתְגַּנֵּב פְּנִימָה הַשֶּׁקֶט. הוּא מְנַסֶּה לְלַחֵשׁ: 'תִּרְאֶה אוֹתְךָ, כַּמָּה אַתָּה גָּדוֹל. כַּמָּה אַתָּה חָכָם וְנַעֲלֶה מִכֻּלָּם'. חֲכָמֵינוּ כּוֹתְבִים עַל אוֹתוֹ הָרֶגַע: "הָיְתָה דַּעְתּוֹ גַּסָּה עָלָיו, מִפְּנֵי שֶׁלָּמַד תּוֹרָה הַרְבֵּה".

הַמִּפְגָּשׁ בַּדֶּרֶךְ וְהַמִּלִּים שֶׁבָּרְחוּ

בְּעוֹדוֹ רוֹכֵב שָׂמֵחַ וְזָקוּף עַל חֲמוֹרוֹ, הוּא מַבְחִין בִּדְמוּת הוֹלֶכֶת בַּשְּׁבִיל, בָּאָה מִמּוּלוֹ. זוֹהִי דְּמוּת אָדָם. כַּאֲשֶׁר הָאָדָם הַזֶּה מִתְקָרֵב, רַבִּי שִׁמְעוֹן מַבִּיט בּוֹ, וְעֵינָיו נִפְעָרוֹת.

זֶה הָיָה אָדָם מְכֹעָר בְּיוֹתֵר. מַרְאֵהוּ הַחִיצוֹנִי הָיָה קָשֶׁה מְאֹד לְמַרְאֶה. קוֹמָתוֹ הָיְתָה כְּפוּפָה, עֵינָיו בּוֹלְטוֹת בְּמִבְנֶה חָרִיג, וְעוֹר פָּנָיו – מוּזָר וְשׁוֹנֶה. זֶה הָיָה מַרְאֶה שֶׁגּוֹרֵם לְאָדָם לַעֲצֹר לְרֶגַע אֶת נְשִׁימָתוֹ.

הָאִישׁ עוֹצֵר לְיַד הַחֲמוֹר, מַבִּיט בְּרַבִּי שִׁמְעוֹן מִלְּמַטָּה לְמַעְלָה, מְחַיֵּךְ חִיּוּךְ רָחָב וְעָדִין, וְקוֹרֵא מִכָּל הַלֵּב בְּקוֹל אוֹהֵב: "שָׁלוֹם עָלֶיךָ, רַבִּי!".

זֶה מָה שֶׁהָאִישׁ אָמַר. הוּא כִּבֵּד אֶת הָרַב, נָתַן לוֹ שָׁלוֹם כְּדֶרֶךְ יְהוּדִים קְדוֹשִׁים.

אֲבָל רַבִּי שִׁמְעוֹן – הַתַּנָּא שֶׁתּוֹרָתוֹ מְאִירָה כַּשֶּׁמֶשׁ – נִתְקַף לְפֶתַע מֵעֵין הֲלֶם מֵהַמַּרְאֶה הַלֹּא שִׁגְרָתִי וּמִזְּחִיחוּת הַדַּעַת שֶׁלּוֹ. בִּמְקוֹם לָשִׁיב "עָלֶיךָ שָׁלוֹם", פִּיו נִפְתַּח, וְהוּא אָמַר לוֹ מִלִּים אֲיֻמּוֹת. הַקָּדוֹשׁ בָּרוּךְ הוּא סִבֵּב שֶׁהַמִּלִּים יֵצְאוּ מִפִּיו כְּדֵי לְלַמֵּד אוֹתוֹ וְאוֹתָנוּ שִׁעוּר לְדוֹרוֹת.

אָמַר לוֹ רַבִּי שִׁמְעוֹן: "רֵיקָה! (אָדָם רֵיק מִתּוֹרֶן וְלֹא חָשׁוּב) כַּמָּה מְכֹעָר אוֹתוֹ הָאִישׁ... שֶׁמָּא כָּל בְּנֵי עִירְךָ מְכֹעָרִים כְּמוֹתְךָ?".

הַשּׁוֹק וְהַהֲקָצָה: לֵךְ לָאוּמָן!

יְלָדִים, נַסּוּ לְהַרְגִּישׁ מָה עוֹבֵר עַל הָאִישׁ הַזֶּה, הָעוֹמֵד כָּעֵת מוּל נְהַר הַמַּיִם. הוּא בָּא לָתֵת שָׁלוֹם וְאַהֲבָה לְרַב כָּל כָּךְ חָשׁוּב, וְחוֹטֵף בִּיקָרְתִּיּוּת וְעֶלְבּוֹן פֻּמְבִּי עַל מַרְאֵהוּ שֶׁאֵין לוֹ שׁוּם שְׁלִיטָה עָלָיו! מָה הוּא אָשֵׁם שֶׁנּוֹלַד כָּךְ? הַאִם הוּא בָּחַר אֶת גּוּפוֹ?

הָאִישׁ לֹא כּוֹעֵס בְּצַעֲקוֹת. הוּא עוֹמֵד זָקוּף. הוּא מֵישִׁיר מַבָּט לְעֵינָיו שֶׁל הַתַּנָּא הַגָּדוֹל, קוֹלוֹ צוֹנֵן אַךְ חוֹדֵר כְּמַחַט דֶּרֶךְ נִשְׁמָתוֹ שֶׁל הָרַב, וְאוֹמֵר בְּלַחַשׁ רוֹעֵד:

"אֵינִי יוֹדֵעַ שֶׁמָּא כָּל בְּנֵי עִירִי כָּךְ... אַךְ הוֹאִיל וַאֲנִי מְכֹעָר בְּעֵינֶיךָ, לֵךְ נָא... וֶאֱמֹר לָאוּמָן שֶׁעֲשָׂאַנִי: כַּמָּה מְכֹעָר כְּלִי זֶה שֶׁעָשִׂיתָ!".

"לֵךְ וֶאֱמֹר לָאוּמָן شֶׁעֲשָׂאַנִי...". מִי זֶה הָאוּמָן? הָאוּמָן – הוּא הַקָּדוֹשׁ בָּרוּךְ הוּא! זֶה שֶׁצִּיֵּר, זֶה שֶׁבָּרָא, זֶה שֶׁעִצֵּב אֶת צוּרַת הָאָדָם הַזֶּה מִבֶּטֶן אִמּוֹ!

הַמִּלִּים הַלָּלוּ הִכּוּ בְּרַבִּי שִׁמְעוֹן כְּמוֹ רַעַם אֲדִיר בְּיוֹם בָּהִיר. פִּתְאוֹם, כָּל גַּאֲוָתוֹ עַל הַתּוֹרָה שֶׁלָּמַד בָּרְחָה כְּלֹא הָיְתָה. הֶחָכְמָה הָעֲצוּמָה שֶׁלּוֹ הֵבִינָה בִּשְׁנִיָּה אַחַת אֶת גֹּדֶל הַפְּגָם – הוּא פָּגַע בִּכְבוֹד שָׁמַיִם! הוּא לִגְלֵג עַל עֲבוֹדָתוֹ שֶׁל מֶלֶךְ מַלְכֵי הַמְּלָכִים! זֶה לֹא פְּגִיעָה בָּאִישׁ, זוֹ פְּגִיעָה בַּבּוֹרֵא יִתְבָּרַךְ!

דְּמָעוֹת שֶׁל חֲרָטָה עַל הַשְּׁבִיל

יְלָדִים, אָדָם רָגִיל שֶׁמֵּבִין שֶׁעָשָׂה טָעוּת כָּל כָּךְ נורָאָה, מָה הָיָה עוֹשֶׂה? אוּלַי מַסְמִיק, מְלַמְלֵם סְלִיחָה, מַכֶּה בַּחֲמוֹר וּבוֹרֵחַ מֵהַמָּקוֹם כְּדֵי שֶׁאַף אֶחָד לֹא יִרְאֶה אוֹתוֹ.

אֲבָל רַבִּי שִׁמְעוֹן הוּא קְדוֹשׁ עֶלְיוֹן. מִדַּת הָאֱמֶת שֶׁלּוֹ חֲזָקָה יוֹתֵר מֵהַבּוּשָׁה שֶׁלּוֹ. בְּלִי לַחְשֹׁב פַּעֲמַיִם, הוּא קוֹפֵץ וְיוֹרֵד מֵעַל הַחֲמוֹר הַמְּפֹאָר שֶׁלּוֹ. הוּא זָרַק אֶת עַצְמוֹ יְשִׁירוֹת עַל הָאֲדָמָה הַקָּשָׁה, עַל סַלְעֵי הַנָּהָר! קוֹמָתוֹ הַמְּכֻבֶּדֶת מִשְׁתַּטַּחַת עַד עָפָר. עֵינָיו דּוֹמְעוֹת וְשַׁרְווּלָיו מִתְלַכְלְכִים בַּבֹּץ.

"נַעֲנֵיתִי לְךָ! פָּשַׁעְתִּי, טָעִיתִי," זָעַק הָרַב מִתּוֹךְ בֶּכִי, "מְחַל לִי! אֲנִי מִתְחַנֵּן אֵלֶיךָ, אָחִי הַיָּקָר, מְחַל לִי!".

אַךְ הָאִישׁ הַמְּכֹעָר שִׁלֵּב אֶת יָדָיו. הַכְּאֵב בְּלִבּוֹ הָיָה מְאֹד עָמֹק. "לֹא," עָנָה בְּקֹשִׁי. "אֵינִי מוֹחֵל לְךָ, עַד שֶׁתֵּלֵךְ לָאוּמָן שֶׁעֲשָׂאַנִי, וְתֹאמַר לוֹ: כַּמָּה מְכֹעָר כְּלִי זֶה שֶׁעָשִׂיתָ!".

הָאִישׁ חָזַר עַל תְּשׁוּבָתוֹ, וְסֵרֵב לִסְלֹחַ. הוּא הִסְתּוֹבֵב וְהִתְחִיל לָלֶכֶת בַּהֲלִיכָה אִטִּית בַּדֶּרֶךְ הַמּוֹבִילָה אֶל מְקוֹם מְגוּרָיו שֶׁל הָרַב.

רַבִּי שִׁמְעוֹן לֹא וִתֵּר. הוּא קָם מֵהָאֲדָמָה. הוּא לֹא עָלָה חֲזָרָה עַל הַחֲמוֹר, אֶלָּא אָחַז בַּמּוֹשְׁכוֹת שֶׁלּוֹ, וְהִתְחִיל לָלֶכֶת בָּרֶגֶל, מֵאֲחוֹרֵי הָאִישׁ.

דַּמְיְנוּ אֶת הַמַּרְאֶה הַזֶּה: אִישׁ מְכֹעָר, עָנִי וּפָשׁוּט מוֹבִיל אֶת הַדֶּרֶךְ בִּשְׁתִיקָה, וּמֵאֲחוֹרָיו – נָשִׂיא בְּיִשְׂרָאֵל, רַב כָּל כָּךְ עָצוּם, הוֹלֵךְ מְשֻׁפָּל מֵאֲחוֹרָיו, כְּעֶבֶד הַמְבַקֵּשׁ רַחֲמִים, הַזֵּעָה כּוֹסָה מִצְחוֹ וְהוּא אֵינוֹ מַפְסִיק לְבַקֵּשׁ: "סְלַח לִי. אֲנִי בּוֹשׁ בְּמָה שֶׁעָשִׂיתִי. מְחַל לִי."

וְהָאִישׁ שׁוֹתֵק. וּמַמְשִׁיךְ לָלֶכֶת.

הַהַגָּעָה לָעִיר וְכֹחָהּ שֶׁל תּוֹרָה

כָּךְ הָלְכוּ מַהֲלָךְ רַב בַּדְּרָכִים הֶעָפְרוֹת, עַד אֲשֶׁר נִרְאוּ חוֹמוֹת עִירוֹ שֶׁל הָרַב מֵרָחוֹק. בַּחוּץ עָמְדוּ זִקְנֵי הָעִיר, סוֹחֲרִים בַּשּׁוּק וְעַשְׂרוֹת תַּלְמִידֵי חֲכָמִים שֶׁיָּצְאוּ לְקַבֵּל אֶת רַבָּם הָאָהוּב הַחוֹזֵר מִלִּמּוּדוֹ.

"שָׁלוֹם עָלֶיךָ רַבִּי! רַבִּי! מוֹרִי! מוֹרִי!" הִרְעִימוּ תּוֹשָׁבֵי הָעִיר בִּכְבוֹד מְלָכִים.

הָאִישׁ הַמְּכֹעָר הִבִּיט בָּהֶם וְשָׁאַל בְּתֵמַהּ: "סְלִיחָה... לְמִי אַתֶּם קוֹרְאִים 'רַבִּי, רַבִּי'?"

הֵם הִצְבִּיעוּ בִּפְלִיאָה עַל רַבִּי שִׁמְעוֹן: "לָזֶה שֶׁמְּטַיֵּל אַחֲרֶיךָ, כַּמּוּבָן! הוּא מוֹרֵנוּ וְרַבֵּנוּ!".

"אִם זֶהוּ רַב," צָעַק לְפֶתַע הָאִישׁ, בְּקוֹל שֶׁשָּׁבַר אֶת כָּל שִׂמְחַת הַקַּבָּלָה, "אִם זֶהוּ הָרַב שֶׁלָּכֶם, יְהִי רָצוֹן שֶׁלֹּא יִרְבּוּ כְּמוֹתוֹ בְּיִשְׂרָאֵל!".

הַקָּהָל נֶהֱרַד. הַשֶּׁקֶט בָּרְחוֹב יָכוֹל הָיָה לְהִחָתֵךְ בְּסַכִּין. "מִפְּנֵי מָה אַתָּה מְדַבֵּר כָּךְ עַל הַצַּדִּיק שֶׁלָּנוּ?" שָׁאֲלָה הֶהָמוֹן בַּחֲרָדָה.

"מִפְּנֵי שֶׁכָּךְ וְכָךְ דִּבֵּר אֵלַי!" וְהָאִישׁ שָׁפַךְ אֶת אֲשֶׁר עַל לִבּוֹ וְסִפֵּר בִּכְאֵב עַל הַמִּלִּים שֶׁהֵטִיחַ בּוֹ הָרַב, וְעַל כָּךְ שֶׁעָלָב בִּיצִירָתוֹ שֶׁל אֱלוֹקִים קְדוֹשׁ יִשְׂרָאֵל.

תּוֹשָׁבֵי הָעִיר שָׁתְקוּ. הֵם הִבִּיטוּ בְּרַבִּי שִׁמְעוֹן, שֶׁעָמַד שָׁם מֻשְׁפָּל, דְּמָעוֹת שֶׁל עֲנָוָה נוֹצְצוֹת בְּעֵינָיו. הוּא לֹא הִצְטַדֵּק, הוּא לֹא חִפֵּשׂ תֵּרוּצִים, אֶלָּא עָמַד שָׁם, שָׂא בְּבֶכִי אֶת בּוּשָׁתוֹ בָּרַבִּים.

"אַף עַל פִּי כֵן," בִּקְּשׁוּ אַנְשֵׁי הָעִיר מֵהָאִישׁ בְּקוֹל רַךְ וְאוֹהֵב, "מְחַל לוֹ! בְּבַקָּשָׁה מְחַל לוֹ! אַתָּה רוֹאֶה אֶת לִבּוֹ הַשָּׁבוּר. הוּא אָדָם גָּדוֹל בַּתּוֹרָה, אָדָם צַדִּיק הֶעָשׂוּי מִבָּשָׂר וָדָם, וְהוּא תִּקֵּן אֶת מַעֲשֵׂהוּ מִיָּד עִם עֲנָוָה רַבָּה".

נָשַׁם הָאִישׁ הַמְּכֹעָר נְשִׁימָה עֲמֻקָּה. הוּא רָאָה אֶת רַבִּי שִׁמְעוֹן, אֶת אֲמִתּוּת הַכְּאֵב שֶׁלּוֹ. נִשְׁמָתוֹ חָמְלָה עָלָיו, וְהוּא אָמַר לְאַנְשֵׁי הָעִיר:

"בִּשְׁבִילְכֶם... הֲרֵינִי מוֹחֵל לוֹ. אֲבָל בִּתְנַאי אֶחָד - שֶׁלֹּא יְהֵא רָגִיל לַעֲשׂוֹת כֵּן! שֶׁיִּזְכֹּר תָּמִיד שֶׁכָּל נִבְרָא הוּא נֵס וּפֶלֶא שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא!".

רַךְ כְּקָנֶה

מִיָּד נִכְנַס רַבִּי שִׁמְעוֹן בֶּן רַבִּי אֶלְעָזָר לְבֵית הַמִּדְרָשׁ הַגָּדוֹל. הוּא לֹא עָלָה לַמַּקְהֵלָה כְּדֵי לוֹמַר חִדּוּשִׁים אוֹ הֲלָכוֹת סְבוּכוֹת. הוּא פָּתַח אֶת הַדְּרָשָׁה שֶׁלּוֹ עִם הַשִּׁעוּר הֲכִי חָשׁוּב שֶׁלָּמַד בְּאוֹתוֹ בֹּקֶר מִיְּהוּדִי אֶחָד קָשֶׁה לְמַרְאֶה. קוֹלוֹ רָעַד כְּשֶׁהִכְרִיז בְּאֹזְנֵי עַשְׂרוֹת הַתַּלְמִידִים:

"תַּלְמִידַי הַיְקָרִים... לְעוֹלָם יְהֵא אָדָם רַךְ כְּקָנֶה, וְאַל יְהֵא קָשֶׁה כְּאֶרֶז!"

כְּלוֹמַר: הָעֵץ הֲכִי חָזָק, עֵץ הָאֶרֶז, מִתְגָּאֶה בְּחָזְקוֹ, אֲבָל כְּשֶׁבָּאָה סְעָרָה הוּא עָלוּל לְהִשָּׁבֵר לִשְׁנַיִם בְּבַת אַחַת. אֲבָל צֶמַח הַקָּנֶה שֶׁעַל שְׂפַת הַנָּהָר – הוּא עָנָו וְרַךְ. בָּאָה רוּחַ? הוּא מִתְכּוֹפֵף. חוֹלֶפֶת הָרוּחַ? הוּא מִזְדַּקֵּף מֵחָדָשׁ. מִתְכּוֹפֵף וְיוֹדֵעַ לְבַקֵּשׁ סְלִיחָה!

וְסִיֵּם רַבִּי שִׁמְעוֹן: "וְלָכֵן, יְלָדִים מְתוּקִים, כֵּיוָן שֶׁהַקָּנֶה כָּל כָּךְ עָנָיו וְרַךְ, מָה נָתַן לוֹ הַקָּדוֹשׁ בָּרוּךְ הוּא בְּמַתָּנָה? מֵעֵץ הַקָּנֶה מֵכִינִים אֶת הַקֻּלְמוֹס (הָעֵט), שֶׁבּוֹ כּוֹתְבִים הַסּוֹפְרִים – סִפְרֵי תּוֹרָה, תְּפִלִּין וּמְזוּזוֹת! רַק מִי שֶׁעָנָיו וִיכוֹל לַחְזֹר מִטָּעוּת – רָאוּי לְתַת אֶת הַתּוֹרָה לְעַם יִשְׂרָאֵל!".

מקורות: תלמוד בבלי, מסכת תענית, דף כ', עמוד א'. אבות דרבי נתן, פרק מ"א (הגירסה המדייקת כ"רבי שמעון בן רבי אלעזר"). ספר האגדה, פרק רע"ה (סיפור האיש המכוער וכבוד הבריות). ילקוט שמעוני והמדרשים העוסקים בקבלת האחר ("לעולם יהא אדם רך כקנה...").

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “רוּחַ קְרִירָה נוֹשֶׁבֶת מִן הֶהָרִים, מֵבִיאָה אִתָּהּ רֵיחַ מָתוֹק שֶׁל עֲצֵי תְּאֵנָה וְשָׂדוֹת יְרֻקִּים.” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “כַּאֲשֶׁר הָאָדָם הַזֶּה מִתְקָרֵב, רַבִּי שִׁמְעוֹן מַבִּיט בּוֹ, וְעֵינָיו נִפְעָרוֹת.”?
ג. כיצד הגיב רבי שמעון בן ר' אלעזר למצב המתואר: “הוּא בָּא לָתֵת שָׁלוֹם וְאַהֲבָה לְרַב כָּל כָּךְ חָשׁוּב, וְחוֹטֵף בִּיקָרְתִּיּוּת וְעֶלְבּוֹן פֻּמְבִּי עַל מַרְאֵהוּ שֶׁאֵין לוֹ שׁוּם שְׁלִיטָה…”?
ד. מה השתנה בין “בְּלִי לַחְשֹׁב פַּעֲמַיִם, הוּא קוֹפֵץ וְיוֹרֵד מֵעַל הַחֲמוֹר הַמְּפֹאָר שֶׁלּוֹ.” לבין “מִפְּנֵי שֶׁכָּךְ וְכָךְ דִּבֵּר אֵלַי!" וְהָאִישׁ שָׁפַךְ אֶת אֲשֶׁר עַל לִבּוֹ וְסִפֵּר בִּכְאֵב עַל הַמִּלִּים…”?
ה. כיצד הגיע הסיפור אל האירוע: “הוּא פָּתַח אֶת הַדְּרָשָׁה שֶׁלּוֹ עִם הַשִּׁעוּר הֲכִי חָשׁוּב שֶׁלָּמַד בְּאוֹתוֹ בֹּקֶר מִיְּהוּדִי אֶחָד קָשֶׁה לְמַרְאֶה.”?

לפני ואחרי

מה קרה לפני: “זֶה לֹא פְּגִיעָה בָּאִישׁ, זוֹ פְּגִיעָה בַּבּוֹרֵא יִתְבָּרַךְ!”?
מה קרה אחרי: “הֵם הִבִּיטוּ בְּרַבִּי שִׁמְעוֹן, שֶׁעָמַד שָׁם מֻשְׁפָּל, דְּמָעוֹת שֶׁל עֲנָוָה נוֹצְצוֹת בְּעֵינָיו.”?

סיבה ותוצאה מתוך המעשה

האירוע הנתוןמה גרם לו?מה הייתה תוצאתו?
אֲבָל רַבִּי שִׁמְעוֹן – הַתַּנָּא שֶׁתּוֹרָתוֹ מְאִירָה כַּשֶּׁמֶשׁ – נִתְקַף לְפֶתַע מֵעֵין הֲלֶם מֵהַמַּרְאֶה הַלֹּא שִׁגְרָתִי וּמִזְּחִיחוּת הַדַּעַת…
הוּא לֹא עָלָה חֲזָרָה עַל הַחֲמוֹר, אֶלָּא אָחַז בַּמּוֹשְׁכוֹת שֶׁלּוֹ, וְהִתְחִיל לָלֶכֶת בָּרֶגֶל, מֵאֲחוֹרֵי הָאִישׁ.

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “האיש המכוער — כבוד הבריות וחזרה בתשובה” על רבי שמעון בן ר' אלעזר? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “פִּתְאוֹם, כָּל גַּאֲוָתוֹ עַל הַתּוֹרָה שֶׁלָּמַד בָּרְחָה כְּלֹא הָיְתָה.”.
ג. אילו רבי שמעון בן ר' אלעזר לא היה פועל כפי שמתואר ב־“מִפְּנֵי שֶׁכָּךְ וְכָךְ דִּבֵּר אֵלַי!" וְהָאִישׁ שָׁפַךְ אֶת אֲשֶׁר עַל לִבּוֹ וְסִפֵּר בִּכְאֵב עַל הַמִּלִּים שֶׁהֵטִיחַ בּוֹ…” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “הוּא פָּתַח אֶת הַדְּרָשָׁה שֶׁלּוֹ עִם הַשִּׁעוּר הֲכִי חָשׁוּב שֶׁלָּמַד בְּאוֹתוֹ בֹּקֶר מִיְּהוּדִי אֶחָד קָשֶׁה לְמַרְאֶה.” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. רוּחַ קְרִירָה נוֹשֶׁבֶת מִן הֶהָרִים, מֵבִיאָה אִתָּהּ רֵיחַ מָתוֹק…
2. קוֹמָתוֹ הָיְתָה כְּפוּפָה, עֵינָיו בּוֹלְטוֹת בְּמִבְנֶה חָרִיג,…
3. וֶאֱמֹר לָאוּמָן שֶׁעֲשָׂאַנִי: כַּמָּה מְכֹעָר כְּלִי זֶה…
4. אֵינִי מוֹחֵל לְךָ, עַד שֶׁתֵּלֵךְ לָאוּמָן שֶׁעֲשָׂאַנִי, וְתֹאמַר…
5. הוּא אָדָם גָּדוֹל בַּתּוֹרָה, אָדָם צַדִּיק הֶעָשׂוּי מִבָּשָׂר וָדָם,…
6. רַק מִי שֶׁעָנָיו וִיכוֹל לַחְזֹר מִטָּעוּת – רָאוּי לְתַת אֶת…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “הוּא בָּא לָתֵת שָׁלוֹם וְאַהֲבָה לְרַב כָּל כָּךְ חָשׁוּב, וְחוֹטֵף בִּיקָרְתִּיּוּת וְעֶלְבּוֹן פֻּמְבִּי עַל מַרְאֵהוּ שֶׁאֵין…”.
מכתב: כתבו לרבי שמעון בן ר' אלעזר בעקבות “הוּא לֹא עָלָה חֲזָרָה עַל הַחֲמוֹר, אֶלָּא אָחַז בַּמּוֹשְׁכוֹת שֶׁלּוֹ, וְהִתְחִיל לָלֶכֶת בָּרֶגֶל, מֵאֲחוֹרֵי הָאִישׁ.”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “הוּא רָאָה אֶת רַבִּי שִׁמְעוֹן, אֶת אֲמִתּוּת הַכְּאֵב שֶׁלּוֹ.”.
דיווח: כתבו ידיעה קצרה שכותרתה “האיש המכוער — כבוד הבריות וחזרה בתשובה”.

המחזה

רגע מן הסיפורהדמויותמה יאמרו או יעשו?
אֲבָל רַבִּי שִׁמְעוֹן – הַתַּנָּא שֶׁתּוֹרָתוֹ מְאִירָה כַּשֶּׁמֶשׁ – נִתְקַף לְפֶתַע מֵעֵין הֲלֶם…
מִפְּנֵי שֶׁכָּךְ וְכָךְ דִּבֵּר אֵלַי!" וְהָאִישׁ שָׁפַךְ אֶת אֲשֶׁר עַל לִבּוֹ וְסִפֵּר בִּכְאֵב…

7. חיבור לחיים מתוך המעשה

א. מה אפשר ללמוד מן הדרך שבה רבי שמעון בן ר' אלעזר פעל באירוע “בְּלִי לַחְשֹׁב פַּעֲמַיִם, הוּא קוֹפֵץ וְיוֹרֵד מֵעַל הַחֲמוֹר הַמְּפֹאָר שֶׁלּוֹ.”?
