סיפור 107 • ר' אושעיא הגדול בן ר' חמא • ר' ממון עבור תלמידי חכמים

ר' ממון עבור תלמידי חכמים

דף עבודה בעקבות ר' אושעיא הגדול בן ר' חמא
שם: כיתה:

המעשה — הסיפור המלא

פְּתִיחָה: אֶבֶן אוֹ נֶפֶשׁ?

יְלָדִים, הַאִם יָצָא לָכֶם פַּעַם לְהִכָּנֵס לְבֵית כְּנֶסֶת חָדָשׁ וְעָנָק? כָּזֶה שֶׁיֵּשׁ בּוֹ אֲרוֹן קֹדֶשׁ מְגֻלָּף מֵעֵץ עִם פִּתּוּחֵי זָהָב יְפֵיפִיִּים, רִצְפַּת שַׁיִשׁ מַבְרִיקָה שֶׁאֶפְשָׁר לִרְאוֹת בָּהּ אֶת הַפָּנִים, וְסַפְסָלִים מְרֻפָּדִים וְנוֹחִים? כְּשֶׁנִּכְנָסִים לְמָקוֹם כָּזֶה, מִיָּד נִפְעָר הַפֶּה וְאוֹמְרִים: "וָאוּ! אֵיזֶה יֹפִי! אֵיזֶה הָדָר!".

גַּם לִפְנֵי אַלְפַּיִם שָׁנָה הָיוּ בָּתֵּי כְּנֶסֶת מְפֹאָרִים מְאֹד. אֲבָל הַיּוֹם, נִלְמַד מִגְּדוֹלֵי יִשְׂרָאֵל לְהִסְתַּכֵּל עַל הַיֹּפִי הַזֶּה... מִבִּפְנִים. עִם מִשְׁקָפַיִם שֶׁל תּוֹרָה.

הַסִּיּוּר שֶׁל גְּדוֹלֵי הַדּוֹר

הַשֶּׁמֶשׁ שֶׁל אֶרֶץ יִשְׂרָאֵל קָפְחָה בַּחֹם הַגָּדוֹל שֶׁל שְׁעַת צָהֳרַיִם בָּעִיר לוֹד. בִּרְחוֹבוֹת הָעִיר, שֶׁהָיוּ הוֹמֵי אָדָם וְסוֹחֲרִים, צָעֲדוּ שְׁנֵי צַדִּיקִים גְּדוֹלִים, עֲנָקֵי הָרוּחַ שֶׁל תְּקוּפַת הָאָמוֹרָאִים: רַבִּי חָמָא בַּר חֲנִינָא, וְחֲבֵרוֹ הָרַב הַגָּדוֹל, רַבִּי אוֹשַׁעְיָא.

כְּשֶׁהָלְכוּ שְׁנֵיהֶם יַחַד, כָּל אַנְשֵׁי הָעִיר עָמְדוּ בְּיִרְאַת כָּבוֹד. הַקְּדֻשָּׁה חָפְפָה עֲלֵיהֶם.

רַבִּי חָמָא בַּר חֲנִינָא בָּא מִמִּשְׁפָּחָה עֲשִׁירָה מְאֹד. אֲבוֹת אֲבוֹתָיו הָיוּ אֲנָשִׁים נְדִיבִים שֶׁאָהֲבוּ אֶת ה' יִתְבָּרַךְ אֲהָבָה גְּדוֹלָה, וְאֶת כָּל כַּסְפָּם הִשְׁקִיעוּ בִּבְנִיַּת בָּתֵּי כְּנֶסֶת לִכְבוֹד בּוֹרֵא עוֹלָם.

"בּוֹא, רַבִּי אוֹשַׁעְיָא," אָמַר רַבִּי חָמָא וְחִיּוּךְ מֵאִיר עַל פָּנָיו. "אֲנִי רוֹצֶה לְהַרְאוֹת לְךָ מַשֶּׁהוּ יָפֶה. בּוֹא נִסְתּוֹבֵב בְּבָתֵּי הַכְּנֶסֶת שֶׁל הָעִיר לוֹד".

הֵם הִתְקָרְבוּ אֶל הַמִּבְנֶה הַגָּדוֹל. עוֹד מִבַּחוּץ אֶפְשָׁר הָיָה לְהָרִיחַ אֶת הָרֵיחַ הַטּוֹב שֶׁל עֲצֵי הָאֶרֶז הַחֲדָשִׁים, מֵהֶם נִבְנוּ דַּלְתוֹת הָעֲנָק. רַבִּי חָמָא פָּתַח אֶת הַדֶּלֶת הַכְּבֵדָה, וְרוּחַ קְרִירָה וּנְעִימָה שֶׁל תּוֹכְכֵי בֵּית הַכְּנֶסֶת לִטְּפָה אֶת פְּנֵיהֶם. הֵם נִכְנְסוּ הַפְּנִימָה.

יְלָדִים, אֵיזֶה מַרְאֶה זֶה הָיָה! קַרְנֵי הַשֶּׁמֶשׁ שֶׁחָדְרוּ מֵהַחַלּוֹנוֹת הַגְּבוֹהִים פָּגְעוּ בְּקִירוֹת שַׁיִשׁ לָבָן וְטָהוֹר, שֶׁנָּצְצוּ בְּאוֹר מְסַנְוֵר. הָעַמּוּדִים הָיוּ חֲצוּבִים מֵאֶבֶן יְקָרָה, וּפָרוֹכֶת אֲרוֹן הַקֹּדֶשׁ הָיְתָה רְקוּמָה בְּחוּטֵי כֶּסֶף וְזָהָב. הַשֶּׁקֶט הַקָּדוֹשׁ בָּאוּלָם הֶעָצוּם נָתַן תְּחוּשָׁה שֶׁל כָּבוֹד אֲמִתִּי לַה'.

הַגַּאֲוָה שֶׁל הָעֲשִׁירִים

רַבִּי חָמָא בַּר חֲנִינָא עָמַד שָׁם, הִבִּיט בַּתִּקְרָה הַגְּבוֹהָה, וְלִבּוֹ הִתְרַחֵב מִשִּׂמְחָה שֶׁל מִצְוָה. הָאֲבוֹת שֶׁלּוֹ עָשׂוּ כָּאן דָּבָר עָצוּם לִכְבוֹד עַם יִשְׂרָאֵל! הוּא פָּנָה אֶל רַבִּי אוֹשַׁעְיָא, הִצְבִּיעַ בְּיָדוֹ אֶל הָעַמּוּדִים הַמְּפֹאָרִים וְאָמַר לְרַבּוֹ:

"תִּרְאֶה רַבִּי אוֹשַׁעְיָא! תִּרְאֶה כַּמָּה מָמוֹן (כֶּסֶף) שִׁקְּעוּ אֲבוֹתַי כָּאן! כַּמָּה אַלְפֵי מַטְבְּעוֹת שֶׁל זָהָב, כַּמָּה אֲבָנִים טוֹבוֹת, כַּמָּה מַאֲמָץ הֵם הִשְׁקִיעוּ כְּדֵי שֶׁבֵּית הַכְּנֶסֶת יִהְיֶה מְפֹאָר וְיָפֶה כָּל כָּךְ לִכְבוֹד ה' יִתְבָּרַךְ!".

רַבִּי חָמָא צִפָּה שֶׁרַבִּי אוֹשַׁעְיָא יַגִּיד לוֹ: 'אָכֵן, יִשַׁר כֹּחַ! אֵיזֶה זְכוּת גְּדוֹלָה יֵשׁ לַמִּשְׁפָּחָה שֶׁלְּךָ!'.

אֲבָל... שֶׁקֶט בַּחֶדֶר. רַבִּי אוֹשַׁעְיָא שָׁתַק.

הוּא הִסְתּוֹבֵב אִטִּית בְּבֵית הַכְּנֶסֶת הָרֵיק, הֶעֱבִיר אֶת יָדוֹ עַל הַשַּׁיִשׁ הַקַּר, וְעָצַם אֶת עֵינָיו.

הַקּוֹלוֹת שֶׁהָרַב שָׁמַע מֵרָחוֹק

יְלָדִים, בּוֹאוּ נַעֲצֹר רֶגַע וְנַחְשֹׁב – עַל מָה חָשַׁב רַבִּי אוֹשַׁעְיָא כְּשֶׁעָצַם אֶת הָעֵינַיִם? הוּא לֹא רָאָה זָהָב. הוּא רָאָה מַשֶּׁהוּ אַחֵר לְגַמְרֵי.

בְּעֵינֵי רוּחוֹ, דִּמְיֵן רַבִּי אוֹשַׁעְיָא אֶת בָּתֵּי הַמִּדְרָשׁ הַיְּשָׁנִים וְהַקְּטַנִּים בְּאֶרֶץ יִשְׂרָאֵל. הוּא רָאָה תַּלְמִידֵי חֲכָמִים אֲמִתִּיִּים, אֲנָשִׁים צַדִּיקִים, שֶׁיּוֹשְׁבִים לִלְמֹד תּוֹרָה מִבֹּקֶר עַד עֶרֶב בְּלִי לָנוּחַ. הוּא רָאָה אֶת הַבְּגָדִים שֶׁלָּהֶם – בְּגָדִים קְרוּעִים שֶׁנִּתְפְּרוּ שׁוּב וָשׁוּב מֵרֹב עֹנִי. הוּא מַמָּשׁ שָׁמַע אֶת קִרְקוּר הַבֶּטֶן שֶׁל אוֹתָם אֲבְרֵכִים קְדוֹשִׁים שֶׁאֵין לָהֶם כֶּסֶף לִקְנוֹת אֲפִלּוּ פְּרוּסַת לֶחֶם יְבֵשָׁה אוֹ שֶׁמֶן כְּדֵי לְהַדְלִיק נֵר בַּלַּיְלָה לִלְמֹד גְּמָרָא.

הוּא שָׁמַע אֶת נְשׁוֹתֵיהֶם הַצִּדְקָנִיּוֹת בּוֹכוֹת מִצַּעַר, כִּי אֵין אוֹכֶל לַיְּלָדִים... וְלַמְרוֹת הַכֹּל – אוֹתָם חֲכָמִים יָשְׁבוּ וְהָגוּ בְּתוֹרַת ה', כִּי הַתּוֹרָה הִיא חַיֵּינוּ.

וְכָאן... כָּאן יֵשׁ בִּנְיָן כָּל כָּךְ יָקָר. קִירוֹת שֶׁל זָהָב. אַךְ מָה שָׁוָה הָאֶבֶן אִם אֵין מִי שֶׁיִּלְמַד בְּתוֹכָהּ בִּמְנוּחָה? מָה שָׁוֶה הַשַּׁיִשׁ הַמַּבְרִיק, אִם לִמְשַׁמְּרֵי הַתּוֹרָה בְּעַם יִשְׂרָאֵל אֵין מָה לֶאֱכֹל? הַקָּדוֹשׁ בָּרוּךְ הוּא צָרִיךְ בִּנְיָנִים מִזָּהָב? הַלֹּא "הַשָּׁמַיִם כִּסְאִי וְהָאָרֶץ הֲדֹם רַגְלָי"! הַקָּדוֹשׁ בָּרוּךְ הוּא רוֹצֶה אֶת הַלֵּב הַלּוֹמֵד! אֶת הַנֶּפֶשׁ הַיְּהוּדִית!

נֶפֶשׁ מְעֻלָּה מֵאֶבֶן

פָּקַח רַבִּי אוֹשַׁעְיָא אֶת עֵינָיו. הוּא הִסְתּוֹבֵב אֶל רַבִּי חָמָא בַּר חֲנִינָא, שֶׁעֲדַיִן עָמַד שָׁם גֵּאֶה וְשָׂמֵחַ בְּמַעֲשֵׂי אֲבוֹתָיו. קוֹלוֹ שֶׁל רַבִּי אוֹשַׁעְיָא יָרַד, לַחַשׁ שֶׁקֶט אֲבָל חָדַר כְּמוֹ רַעַם דֶּרֶךְ עַמּוּדֵי הַבֵּית הֶעָצוּם.

"אָמַרְתָּ כַּמָּה מָמוֹן וְכֶסֶף שִׁקְּעוּ (הִשְׁקִיעוּ) אֲבוֹתֶיךָ כָּאן?" אָמַר רַבִּי אוֹשַׁעְיָא. "אֲנִי אוֹמֵר לְךָ מַשֶּׁהוּ אַחֵר: כַּמָּה נְפָשׁוֹת שִׁקְּעוּ אֲבוֹתֶיךָ כָּאן! ".

רַבִּי חָמָא נִבְהַל. "נְפָשׁוֹת? עַל מָה כְּבוֹד הָרַב מְדַבֵּר?".

"כֵּן, נְפָשׁוֹת!" הִמְשִׁיךְ רַבִּי אוֹשַׁעְיָא בִּכְאֵב. "וְכִי לֹא הָיוּ בְּנֵי אָדָם לַעֲסֹק בַּתּוֹרָה?! בַּכֶּסֶף הָרַב וְהֶעָצוּם הַזֶּה שֶׁנִּשְׁפַּךְ כָּאן עַל הָאֲבָנִים, הָיָה אֶפְשָׁר לְפַרְנֵס מֵאוֹת תַּלְמִידֵי חֲכָמִים! הָיָה אֶפְשָׁר לְקַיֵּם עַשְׂרוֹת בָּתֵּי מִדְרָשׁ! הָאֲנָשִׁים שֶׁהָיוּ יְכוֹלִים לִלְמֹד תּוֹרָה וְאֵין לָהֶם כֶּסֶף... נַפְשָׁם שָׁקְעָה בְּתוֹךְ קִירוֹת הַזָּהָב הָאֵלֶּה! הָאֶבֶן – לֹא לוֹמֶדֶת תּוֹרָה. תַּלְמִיד חָכָם – מַחְזִיק אֶת הָעוֹלָם!".

רַבִּי חָמָא בַּר חֲנִינָא עָמַד נִדְהָם. לְפֶתַע, כָּל הַזָּהָב וְהַכֶּסֶף נִרְאוּ לוֹ כָּל כָּךְ פְּשׁוּטִים, וְלִבּוֹ הֵבִין אֶת הָאֱמֶת הַטְּהוֹרָה וְהַקְּדוֹשָׁה.

בְּנִיַּת בַּיִת לַה' זֶה דָּבָר חָשׁוּב, אֲבָל מָה ה' הֲכִי רוֹצֶה? מָה בּוֹנֶה בֶּאֱמֶת אֶת אֶרֶץ יִשְׂרָאֵל וְאֶת עַם יִשְׂרָאֵל? לֹא הָאֲבָנִים הַקָּרוֹת. רַק אֲנָשִׁים חַיִּים, שֶׁעוֹסְקִים בַּתּוֹרָה, אֲנָשִׁים צַדִּיקִים, יְלָדִים טְהוֹרִים שֶׁמְּשַׁנְּנִים מִשְׁנָיוֹת, הֵם הַזָּהָב וְהַשַּׁיִשׁ הָאֲמִתִּי שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא. וְאִם יֵשׁ כֶּסֶף – הַהַשְׁקָעָה הֲכִי גְּדוֹלָה שֶׁנָּבִיא אִתָּנוּ לָעוֹלָם הַבָּא, הִיא הַחְזָקַת תּוֹרָה שֶׁל כְּלַל יִשְׂרָאֵל.

מקורות: תלמוד ירושלמי: מסכת פאה, פרק ח', הלכה ח' (ולפי גרסאות אחרות: ה"ה / ה"ט). תלמוד ירושלמי: מסכת שקלים, פרק ה', הלכה ד'. ספר האגדה: סיפור שפ"א (כפי שמצוין במקורות: "ר' אושעיא ור' חמא בר חנינא / ממון עבור תלמידי חכמים / חשיבות השקעה בת"ח כדי שילמדו יותר מאשר בבנין"). הערת דיוק: רבי חמא בן חנינא היה בנו של רבי חנינא בר חמא. רבי אושעיא רבו היה גדול ממנו (רבי אושעיא הגדול היה מחבורת רבי חייא מעתיקי השמועות מתקופת המעבר תנאים-אמוראים). שניהם למדו והיו קשורים לארץ ישראל וללימוד התורה בה.

ב. הסבירו במילים שלכם את המשפט: “הָאֲבוֹת שֶׁלּוֹ עָשׂוּ כָּאן דָּבָר עָצוּם לִכְבוֹד עַם יִשְׂרָאֵל!”.
ג. אילו ר' אושעיא הגדול בן ר' חמא לא היה פועל כפי שמתואר ב־“הוּא שָׁמַע אֶת נְשׁוֹתֵיהֶם הַצִּדְקָנִיּוֹת בּוֹכוֹת מִצַּעַר, כִּי אֵין אוֹכֶל לַיְּלָדִים...” — כיצד היה הסיפור עשוי להסתיים?

7. חיבור לחיים מתוך המעשה

א. מה אפשר ללמוד מן הדרך שבה ר' אושעיא הגדול בן ר' חמא פעל באירוע “רַבִּי חָמָא צִפָּה שֶׁרַבִּי אוֹשַׁעְיָא יַגִּיד לוֹ: 'אָכֵן, יִשַׁר כֹּחַ!”?
