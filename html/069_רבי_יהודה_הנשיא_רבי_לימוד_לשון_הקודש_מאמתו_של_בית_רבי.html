לימוד לשון הקודש מאמתו של בית רבי

דף עבודה בעקבות רבי יהודה הנשיא (רבי)
שם: כיתה:

המעשה — הסיפור המלא

פְּתִיחָה: אַרְמוֹנוֹ שֶׁל נְשִׂיא יִשְׂרָאֵל

יְלָדִים, עִצְמוּ רֶגַע אֶת הָעֵינַיִם. דַּמְיְנוּ שֶׁאַתֶּם עוֹמְדִים מוּל שַׁעַר עֵץ עֲנָק, מְגֻלָּף בְּצוּרוֹת שֶׁל פִּרְחֵי רִמּוֹן וְגַפְנֵי זָהָב. אַתֶּם דּוֹפְקִים עַל הַשַּׁעַר, וְהוּא נִפְתָּח בַּחֲרִיקָה חֲרִישִׁית. מִיָּד פּוֹגֵשׁ אוֹתְכֶם רֵיחַ מְיֻחָד – רֵיחַ שֶׁל קְלָפִים עַתִּיקִים מֵעוֹר כֶּבֶשׂ, מְעֹרָב בְּרֵיחַ שֶׁל עֲצֵי הֲדַס וּבְשָׂמִים. מִבִּפְנִים נִשְׁמָע קוֹל מְתוּק כִּדְבַשׁ: מֵאוֹת אֲנָשִׁים לוֹמְדִים תּוֹרָה בְּשִׁירָה עֲצוּמָה וּמְהַדְהֶדֶת.

אַתֶּם נִמְצָאִים בְּצִפּוֹרִי. בְּתוֹךְ הַבַּיִת הֲכִי חָשׁוּב בָּעוֹלָם הַיְּהוּדִי שֶׁל אוֹתָם יָמִים – בֵּיתוֹ שֶׁל רַבִּי יְהוּדָה הַנָּשִׂיא, הָאִישׁ שֶׁכָּתַב וְסִדֵּר עֲבוּרֵנוּ אֶת הַמִּשְׁנָה!

רַבִּי יְהוּדָה הַנָּשִׂיא הָיָה עָשִׁיר מְאֹד כְּמֶלֶךְ, אֲבָל הַלֵּב שֶׁלּוֹ הָיָה מָלֵא רַק בְּאַהֲבַת הַתּוֹרָה. בְּאוֹתָם יָמִים קָשִׁים שֶׁל שִׁלְטוֹן הָרוֹמָאִים, רַבִּים מֵהַיְּהוּדִים בָּרְחוֹב כְּבָר לֹא דִּבְּרוּ עִבְרִית. הֵם דִּבְּרוּ בְּשָׂפָה שֶׁנִּקְרֵאת "אֲרָמִית" – מִין שָׂפָה חֲדָשָׁה שֶׁנִּדְבְּקָה אֲלֵיהֶם מֵהַגָּלוּת וּמֵהָעַמִּים הַשְּׁכֵנִים.

אֲבָל בְּבֵיתוֹ שֶׁל הַנָּשִׂיא יֵשׁ חֹק בַּרְזֶל! "בְּאֶרֶץ יִשְׂרָאֵל – לָמָּה לְדַבֵּר לָשׁוֹן אֲרָמִית?" הָיָה שׁוֹאֵל רַבִּי יְהוּדָה הַנָּשִׂיא וּפָנָיו הַקְּדוֹשׁוֹת מְאִירוֹת כַּשֶּׁמֶשׁ. "אוֹ שֶׁנְּדַבֵּר בִּלְשׁוֹן הַקֹּדֶשׁ – הַשָּׂפָה שֶׁל הַתּוֹרָה וְהַנְּבִיאִים, אוֹ – מִי שֶׁכְּבָר חַיָּב, שֶׁיְּדַבֵּר יְוָנִית שֶׁהִיא שְׂפַת הַמַּלְכוּת. אֲבָל בַּבַּיִת שֶׁלִּי – עִבְרִית טְהוֹרָה!".

וְאָכֵן, בַּבַּיִת הַזֶּה, כֻּלָּם דִּבְּרוּ רַק עִבְרִית. אֲפִלּוּ הָעוֹבְדִים וְהַמְשָׁרְתִים הַפְּשׁוּטִים בַּמִּטְבָּח וּבֶחָצֵר. אִשָּׁה אַחַת מְיֻחֶדֶת, צְנוּעָה וְיִרְאַת שָׁמַיִם, שֶׁכֻּלָּם קָרְאוּ לָהּ "אַמַּת בֵּית רַבִּי" (הַשִּׁפְחָה אוֹ הַמְשָׁרֶתֶת שֶׁל הַבַּיִת), גָּדְלָה וְחָיְתָה בַּבַּיִת הַזֶּה שָׁנִים רַבּוֹת. הִיא אָפְתָה אֶת הַלֶּחֶם, טִאטְאָה אֶת הָרִצְפָּה וְדָאֲגָה לָאֵרוּחַ, וְהַכֹּל – בְּעִבְרִית צַחָה, שְׂפַת עֲבָרֵנוּ, עִם מִלִּים שֶׁהִיא זָכְרָה עוֹד מִיְּמֵי בֵּית הַמִּקְדָּשׁ שֶׁחָרַב.

הַתַּעֲלוּמָה הָרִאשׁוֹנָה: מַהִי "חֲלוּגְלְגוֹת"?

יוֹם אֶחָד יָשְׁבוּ חַכְמֵי הַסַּנְהֶדְרִין, הָרַבָּנִים הֲכִי גְּדוֹלִים בְּאוֹתוֹ דּוֹר, וְלָמְדוּ סוּגְיָה קָשָׁה מְאֹד. הֵם לָמְדוּ עַל סוּגֵי צְמָחִים בַּשָּׂדֶה. פִּתְאוֹם הֵם נִתְקְלוּ בְּמִלָּה עַתִּיקָה מֵהַמְּקוֹרוֹת. צֶמַח שֶׁנִּקְרָא "חֲלוּגְלְגוֹת".

הָרַבָּנִים הַגְּדוֹלִים קִמְּטוּ אֶת מִצְחָם. הֵם הֶעֱבִירוּ יָד שָׁתַּק עַל זְקָנָם הַלָּבָן. "מָה זֶה חֲלוּגְלְגוֹת?" שָׁאַל צַדִּיק אֶחָד וְהֶעֱבִיר אֶת יָדוֹ עַל מַגַּע הַקְּלָף הֶחָלָק.

"אֵינֶנִּי יוֹדֵעַ. בַּחוּץ כֻּלָּם קוֹרְאִים לַצְּמָחִים בַּשֵּׁמוֹת הָאֲרָמִיִּים שֶׁלָּהֶם. מִיסְכֵּנִים, הֵם שָׁכְחוּ לְשׁוֹן הַקֹּדֶשׁ," עָנָה רַב שֵׁנִי וְנֶאֱנַח אֲנָחָה עֲמֻקָּה שֶׁנִּשְׁמְעָה בַּחֶדֶר הַגָּדוֹל.

הֵם יָשְׁבוּ מְתֻסְכָּלִים. אֵיךְ יוּכְלוּ לָדַעַת אֵיזֶה צֶמַח פָּטוּר מִמַּעֲשֵׂר וְאֵיזֶה חַיָּב, אִם אֵינָם יוֹדְעִים לְזַהוֹת אוֹתוֹ?!

בְּאוֹתוֹ בֹּקֶר הָלַךְ אֶחָד מִתַּלְמִידֵי הַחֲכָמִים בַּחֲצַר בֵּיתוֹ שֶׁל הַנָּשִׂיא. הָיָה זֶה יוֹם אָבִיב נָעִים, וְקַרְנֵי שֶׁמֶשׁ זְהוּבוֹת נָפְלוּ עַל שְׁבִיל הָאֶבֶן. בְּיָדוֹ הֶחְזִיק הֶחָכָם צְרוֹר שֶׁל עֲשָׂבִים טְרִיִּים שֶׁקָּטַף. רֵיחָם הָרַעֲנָן וְהַיָּרֹק הִתְפַּזֵּר בָּאֲוִיר. לְפֶתַע, מָעֲדָה רַגְלוֹ בְּאֶבֶן קְטַנָּה, וְכַמָּה מֵהָעֲשָׂבִים צָנְחוּ הַרְצָה מִיָּדוֹ.

בְּדִיּוּק בְּאוֹתוֹ רֶגַע יָצְאָה בְּשֶׁקֶט "אַמַּת בֵּית רַבִּי", עֲטוּפָה בְּמִטְפַּחַת רֹאשָׁהּ הַפְּשׁוּטָה וְהַנְּקִיָּה. הִיא הִבִּיטָה בֶּחָכָם, מִיָּד נִגְּשָׁה לַעֲזֹר לוֹ, וְאָמְרָה בִּשְׂפָתָהּ הַטִּבְעִית, בְּעִבְרִית טְהוֹרָה מַמָּשׁ:

"רַבִּי! עֲצֹר! הִנֵּה פִּזַּרְתָּ אֶת... הַחֲלוּגְלְגוֹת שֶׁלְּךָ!".

הֶחָכָם כִּמְעַט קָפַץ מִמְּקוֹמוֹ. הוּא הִבִּיט בָּעֵשֶׂב שֶׁהָיָה פָּזוּר עַל הָרִצְפָּה (צֶמַח שֶׁהַיּוֹם נִקְרָא "רְגֵלַת הַגִּנָּה").

הוּא הֵרִים אֶת מַבָּטוֹ לַשָּׁמַיִם. "תּוֹדָה רַבָּה לְךָ, הַקָּדוֹשׁ בָּרוּךְ הוּא! לָמַדְתִּי עַכְשָׁו תּוֹרָה מִפִּיהָ שֶׁל פּוֹעֶלֶת פְּשׁוּטָה!". הוּא מִהֵר רָץ לְבֵית הַמִּדְרָשׁ וְקָרָא לַחֲכָמִים: "חֲבֵרִים! מָצָאתִי מַהִי חֲלוּגְלְגוֹת!".

הַתַּעֲלוּמָה הַשְּׁנִיָּה: הַכְּנִיסָה בְּ"סֵרוּגִין"

עָבְרוּ כַּמָּה חֳדָשִׁים. בֵּית הַמִּדְרָשׁ שׁוּב הוֹמֶה אָדָם. הַפַּעַם דָּנוּ הַחֲכָמִים בַּהֲלָכוֹת קְרִיאַת הַמְּגִלָּה. הַמִּשְׁנָה אוֹמֶרֶת: "הַקּוֹרֵא אֶת הַמְּגִלָּה לְמַפְרֵעַ (מֵהַסּוֹף לַהַתְחָלָה) – לֹא יָצָא. קְרָאָהּ סֵרוּגִין ... יָצָא יְדֵי חוֹבָה".

"מָה הַפֵּרוּשׁ שֶׁל הַמִּלָּה 'סֵרוּגִין' בְּעִבְרִית תַּנָ"כִית מְקוֹרִית?" שָׁאֲלוּ הַחֲכָמִים אֶחָד אֶת הַשֵּׁנִי. "הַאִם הַפֵּרוּשׁ הוּא מִי שֶׁמְּדַלֵּג עַל מִלִּים? אוֹ אוּלַי הַפֵּרוּשׁ הוּא מִי שֶׁקּוֹרֵא, עוֹצֵר וְנָח, וְאָז שׁוּב מַמְשִׁיךְ לִקְרֹא?"

הֵם יָשְׁבוּ וְחָשְׁבוּ. מִיָּמִים יָמִימָה לֹא שָׁמְעוּ אִישׁ בַּשּׁוּק מִשְׁתַּמֵּשׁ בַּמִּלָּה "סֵרוּגִין". הַתַּעֲלוּמָה שׁוּב עָמְדָה בַּאֲוִיר, כְּבֵדָה כֶּאֶבֶן.

בְּאוֹתוֹ בֹּקֶר הָיָה תּוֹר כָּבֵד וְאָרוֹךְ מְאֹד לְהִכָּנֵס לְחַדְרוֹ שֶׁל רַבִּי יְהוּדָה הַנָּשִׂיא. תַּלְמִידֵי חֲכָמִים רַבִּים רָצוּ לְדַבֵּר אִתּוֹ. הֵם עָמְדוּ בַּשַּׁעַר, נִכְנָס אֶחָד... אַחַר כָּךְ עוֹמְדִים כַּמָּה דַּקּוֹת... וְאָז נִכְנָסוּ עוֹד שְׁנַיִם.

אַמַּת בֵּית רַבִּי הָיְתָה צְרִיכָה לְנַקּוֹת אֶת אֲסַדְרַת הַכְּנִיסָה (הַמִּסְדְּרוֹן). הִיא עָמְדָה בְּסַבְלָנוּת, וְרָאֲתָה אֵיךְ תַּלְמִיד כָּאן וְתַלְמִיד שָׁם מַפְרִיעִים לָהּ לַעֲשׂוֹת סֵדֶר בִּתְנוּעָתָם הַקְּטוּעָה.

נִגְּשָׁה אֲלֵיהֶם הַמְשָׁרֶתֶת בְּנִמּוּס עָמֹק, אֲבָל בְּבַקָּשָׁה בְּרוּרָה, וְשָׁאֲלָה בִּשְׂפָתָהּ שֶׁצִּלְצְלָה כִּפְסוּקֵי תּוֹרָה: "רַבָּנַי הַנִּכְבָּדִים... עַד מָתַי אַתֶּם נִכְנָסִים סֵרוּגִין סֵרוּגִין ?! אוֹ שֶׁתִּכָּנְסוּ כֻּלְּכֶם יַחַד, אוֹ שֶׁתְּחַכּוּ בַּחוּץ עַד שֶׁאֲסַיֵּם!".

הַחֲכָמִים פָּעֲרוּ אֶת עֵינֵיהֶם מֵרֹב תַּדְהֵמָה. "סֵרוּגִין סֵרוּגִין"?! קָרָא אֶחָד מֵהֶם כְּשֶׁלִּבּוֹ הוֹלֵם. "הִיא הִשְׁתַּמְּשָׁה בַּמִּלָּה 'סֵרוּגִין' לִתְנוּעָה שֶׁנֶּעֱצֶרֶת וּמַמְשִׁיכָה עִם הַפְסָקוֹת בָּאֶמְצַע! מִי שֶׁנִּכְנָס בַּהַפְסָקוֹת הוּא נִכְנָס בְּסֵרוּגִין!".

הֵם שָׁבוּ מִיָּד לְבֵית הַמִּדְרָשׁ עִם הַתְּשׁוּבָה. עַכְשָׁו יָדְעוּ מַהִי הֲלָכָה: קוֹרֵא הַמְּגִלָּה שֶׁקָּרָא בְּהַפְסָקוֹת שֶׁל מְנוּחָה שָׁתַּק, יָצָא יְדֵי חוֹבָה. אֵיזוֹ שִׂמְחָה גְּדוֹלָה מִלְּאָה אֶת לִבָּם שֶׁל הַלּוֹמְדִים.

הַתַּעֲלוּמָה הַשְּׁלִישִׁית: הַמִּלָּה "טָאטֵא"

אֲבָל הַדָּבָר הַמַּדְהִים בְּיוֹתֵר קָרָה עִם פָּסוּק אֶחָד מִנְּבִיאֵינוּ. בְּסֵפֶר יְשַׁעְיָהוּ לוֹמְדִים הַחֲכָמִים נְבוּאָה קָשָׁה שֶׁנִּבָּא יְשַׁעְיָהוּ עַל בָּבֶל הָרְשָׁעָה: "וְטֵאטֵאתִיהָ בְּמַטְאֲטֵא הַשְׁמֵד".

מָה פֵּרוּשׁ הַמִּלָּה 'טֵאטֵאתִיהָ'? וּמַהוּ 'מַטְאֲטֵא'? הַיּוֹם בִּשְׁבִילֵנוּ זֶה פָּשׁוּט וּבָרוּר, כִּי חָזַרְנוּ לְדַבֵּר בִּלְשׁוֹן הַקֹּדֶשׁ. אֲבָל בִּימֵי חֲזַ"ל, הַמִּלָּה נִשְׁמְטָה מִפִּי הָעָם וְנִשְׁכְּחָה. הֵם קָרְאוּ לַמַּטְאֲטֵא 'מַכְבֵּד' אוֹ 'מוּקִימָה'. הֵם הָפְכוּ בַּסְּפָרִים וְשָׁאֲלוּ אֶת הַזְּקֵנִים. הֵם חָשְׁשׁוּ שֶׁמָּא הַכַּוָּנָה לְכֶלִי נֶשֶׁק שֶׁעוֹשֶׂה שֶׁמֶד? אוֹ אוּלַי חֶרֶב?

דְּמָמָה שָׂרְרָה בְּבֵית הַמִּדְרָשׁ. "עָלֵינוּ לְהָבִין דִּבְרֵי נְבִיאִים עַל בֻּרְיָם," מִלְמֵל רַבִּי יְהוּדָה הַנָּשִׂיא.

כְּדֵי שֶׁיּוּכְלוּ לְרַעֲנֵן אֶת מַחְשַׁבְתָּם, יָצְאוּ כַּמָּה מִן הַחֲכָמִים לֶחָצֵר. הָרִצְפָּה בֶּחָצֵר הָיְתָה מְלֵאָה בְּעָלִים נוֹשְׁרִים, אָבָק מֵהָרְחוֹב, וּפֵרוּרִים. וְהִנֵּה שׁוּב... קוֹל שְׁרִיקַת עֲנָפִים חֲשׁוּפִים נִשְׁמַע עַל רִצְפַּת הָאֶבֶן. שְׁרִיק... שְׁרִיק... שְׁרִיק...

הָיְתָה זוֹ אַמַּת בֵּית רַבִּי שֶׁעָבְדָה בֶּחָצֵר. הִיא הֶחְזִיקָה בְּיָדָהּ אֲגֻדָּה עָבָה שֶׁל עַנְפֵי סִירָה נַקְשָׁנִיִּים הַקְּשׁוּרִים זֶה בָּזֶה לַמַּקֵּל. הִיא רָאֲתָה אֶת הַלִּכְלוּךְ הַגָּדוֹל, וְאָז הִבְחִינָה בְּאֶחָד מִפּוֹעֲלֵי הַנִּקָּיוֹן הָאֲחֵרִים שֶׁשָּׁכַח בַּצַּד כְּלִי נוֹסָף לְנִקָּיוֹן.

עָמְדָה הָאִשָּׁה וְהִכְרִיזָה אֵלָיו בְּקוֹל עֵרָנִי מֵעֵבֶר לֶחָצֵר: "קַח אֶת הַמַּטְאֲטֵא , וְ טָאטֵא אֶת הַבַּיִת!".

כְּמוֹ בָּרָק בְּיוֹם גֶּשֶׁם, הִכְּתָה הַהֲבָנָה בְּרֹאשָׁם שֶׁל הַחֲכָמִים. "מַטְאֲטֵא! מַטְאֲטֵא הַשְׁמֵד שֶׁל הַנָּבִיא! הַכַּוָּנָה הִיא לְנִקָּיוֹן מֻחְלָט וִיסוֹדִי, כְּמוֹ אִשָּׁה הַמְטַאטְאָה אֶת הָרִצְפָּה וְלֹא מַשְׁאִירָה שׁוּם פֵּרוּר לִכְלוּךְ!".

הַחֲכָמִים הַקְּדוֹשִׁים, שֶׁיָּדְעוּ אֶת הַתּוֹרָה כֻּלָּהּ עַל פֶּה, נֶעֶמְדוּ בִּירְאַת כָּבוֹד וְהִבִּיטוּ זֶה בָּזֶה. אֵיזוֹ אֲוִירָה עֲצוּמָה וּקְדוֹשָׁה הָיְתָה בְּבֵיתוֹ שֶׁל רַבִּי יְהוּדָה הַנָּשִׂיא! אוֹתוֹ צַדִּיק שֶׁדָּאַג לִשְׁמֹר עַל תּוֹרַת יִשְׂרָאֵל, שָׁמַר גַּם עַל שְׂפַת יִשְׂרָאֵל, עַד שֶׁאֲפִלּוּ אִשָּׁה פְּשׁוּטָה הָעוֹבֶדֶת בְּבֵיתוֹ, יָדְעָה וְשָׁמְרָה אֶת שְׂפַת הַקֹּדֶשׁ שֶׁכֻּלָּם שָׁכְחוּ. דַּוְקָא הַמִּלִּים הַפְּשׁוּטוֹת הָאֵלֶּה הִמְשִׁיכוּ לִחְיוֹת בְּבֵית הַנָּשִׂיא, וּמִשָּׁם – חָזְרוּ אֵלֵינוּ עַד לְדוֹרֵנוּ זֶה.

מקורות: תלמוד ירושלמי, מסכת מגילה פרק ב' הלכה ב' ("סֵרוּגִין", "חֲלוּגְלְגוֹת", "מַטְאֲטֵא"). ספר האגדה: סימן ש"ה. מקבילה בבלית (להעשרת הרקע בלבד, אם המורה ירצה להרחיב לילדים גדולים יותר): בבלי, ראש השנה כ"ו ע"ב, ומגילה י"ח ע"א (מוזכרת שם שאלת החכמים לאמת בית רבי על אותן מילים ממש).

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “דַּמְיְנוּ שֶׁאַתֶּם עוֹמְדִים מוּל שַׁעַר עֵץ עֲנָק, מְגֻלָּף בְּצוּרוֹת שֶׁל פִּרְחֵי רִמּוֹן וְגַפְנֵי זָהָב.” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “אִשָּׁה אַחַת מְיֻחֶדֶת, צְנוּעָה וְיִרְאַת שָׁמַיִם, שֶׁכֻּלָּם קָרְאוּ לָהּ "אַמַּת בֵּית רַבִּי" (הַשִּׁפְחָה אוֹ הַמְשָׁרֶתֶת שֶׁל הַבַּיִת), גָּדְלָה…”?
ג. כיצד הגיב רבי יהודה הנשיא (רבי) למצב המתואר: “בְּדִיּוּק בְּאוֹתוֹ רֶגַע יָצְאָה בְּשֶׁקֶט "אַמַּת בֵּית רַבִּי", עֲטוּפָה בְּמִטְפַּחַת רֹאשָׁהּ הַפְּשׁוּטָה וְהַנְּקִיָּה.”?
ד. מה השתנה בין “נִגְּשָׁה אֲלֵיהֶם הַמְשָׁרֶתֶת בְּנִמּוּס עָמֹק, אֲבָל בְּבַקָּשָׁה בְּרוּרָה, וְשָׁאֲלָה בִּשְׂפָתָהּ שֶׁצִּלְצְלָה…” לבין “אֲבָל בִּימֵי חֲזַ"ל, הַמִּלָּה נִשְׁמְטָה מִפִּי הָעָם וְנִשְׁכְּחָה.”?
ה. כיצד הגיע הסיפור אל האירוע: “הַכַּוָּנָה הִיא לְנִקָּיוֹן מֻחְלָט וִיסוֹדִי, כְּמוֹ אִשָּׁה הַמְטַאטְאָה אֶת הָרִצְפָּה וְלֹא מַשְׁאִירָה שׁוּם פֵּרוּר לִכְלוּךְ!".”?

3. מה היה ראשון?

כתבו ליד כל אירוע מספר מ־1 עד 5 לפי סדר התרחשותו בסיפור.

□ אַמַּת בֵּית רַבִּי הָיְתָה צְרִיכָה לְנַקּוֹת אֶת אֲסַדְרַת הַכְּנִיסָה (הַמִּסְדְּרוֹן).
□ דַּמְיְנוּ שֶׁאַתֶּם עוֹמְדִים מוּל שַׁעַר עֵץ עֲנָק, מְגֻלָּף בְּצוּרוֹת שֶׁל פִּרְחֵי רִמּוֹן וְגַפְנֵי זָהָב.
□ דַּוְקָא הַמִּלִּים הַפְּשׁוּטוֹת הָאֵלֶּה הִמְשִׁיכוּ לִחְיוֹת בְּבֵית הַנָּשִׂיא, וּמִשָּׁם – חָזְרוּ אֵלֵינוּ עַד לְדוֹרֵנוּ זֶה.
□ מִיסְכֵּנִים, הֵם שָׁכְחוּ לְשׁוֹן הַקֹּדֶשׁ," עָנָה רַב שֵׁנִי וְנֶאֱנַח אֲנָחָה עֲמֻקָּה שֶׁנִּשְׁמְעָה בַּחֶדֶר הַגָּדוֹל.
□ עָלֵינוּ לְהָבִין דִּבְרֵי נְבִיאִים עַל בֻּרְיָם," מִלְמֵל רַבִּי יְהוּדָה הַנָּשִׂיא.

לפני ואחרי

מה קרה לפני: “אַמַּת בֵּית רַבִּי הָיְתָה צְרִיכָה לְנַקּוֹת אֶת אֲסַדְרַת הַכְּנִיסָה (הַמִּסְדְּרוֹן).”?
מה קרה אחרי: “עָלֵינוּ לְהָבִין דִּבְרֵי נְבִיאִים עַל בֻּרְיָם," מִלְמֵל רַבִּי יְהוּדָה הַנָּשִׂיא.”?

סיבה ותוצאה מתוך המעשה

האירוע הנתוןמה גרם לו?מה הייתה תוצאתו?
בְּאוֹתוֹ בֹּקֶר הָלַךְ אֶחָד מִתַּלְמִידֵי הַחֲכָמִים בַּחֲצַר בֵּיתוֹ שֶׁל הַנָּשִׂיא.
אֵיזוֹ שִׂמְחָה גְּדוֹלָה מִלְּאָה אֶת לִבָּם שֶׁל הַלּוֹמְדִים.

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “לימוד לשון הקודש מאמתו של בית רבי” על רבי יהודה הנשיא (רבי)? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “הַמִּשְׁנָה אוֹמֶרֶת: "הַקּוֹרֵא אֶת הַמְּגִלָּה לְמַפְרֵעַ (מֵהַסּוֹף לַהַתְחָלָה) – לֹא יָצָא.”.
ג. אילו רבי יהודה הנשיא (רבי) לא היה פועל כפי שמתואר ב־“אֲבָל בִּימֵי חֲזַ"ל, הַמִּלָּה נִשְׁמְטָה מִפִּי הָעָם וְנִשְׁכְּחָה.” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “הַכַּוָּנָה הִיא לְנִקָּיוֹן מֻחְלָט וִיסוֹדִי, כְּמוֹ אִשָּׁה הַמְטַאטְאָה אֶת הָרִצְפָּה וְלֹא מַשְׁאִירָה שׁוּם פֵּרוּר…” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. דַּמְיְנוּ שֶׁאַתֶּם עוֹמְדִים מוּל שַׁעַר עֵץ עֲנָק, מְגֻלָּף…
2. אִשָּׁה אַחַת מְיֻחֶדֶת, צְנוּעָה וְיִרְאַת שָׁמַיִם, שֶׁכֻּלָּם…
3. הוּא הִבִּיט בָּעֵשֶׂב שֶׁהָיָה פָּזוּר עַל הָרִצְפָּה (צֶמַח…
4. הִיא הִשְׁתַּמְּשָׁה בַּמִּלָּה 'סֵרוּגִין' לִתְנוּעָה שֶׁנֶּעֱצֶרֶת…
5. קוֹל שְׁרִיקַת עֲנָפִים חֲשׁוּפִים נִשְׁמַע עַל רִצְפַּת הָאֶבֶן.
6. דַּוְקָא הַמִּלִּים הַפְּשׁוּטוֹת הָאֵלֶּה הִמְשִׁיכוּ לִחְיוֹת בְּבֵית…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “בְּדִיּוּק בְּאוֹתוֹ רֶגַע יָצְאָה בְּשֶׁקֶט "אַמַּת בֵּית רַבִּי", עֲטוּפָה בְּמִטְפַּחַת רֹאשָׁהּ הַפְּשׁוּטָה וְהַנְּקִיָּה.”.
מכתב: כתבו לרבי יהודה הנשיא (רבי) בעקבות “אֵיזוֹ שִׂמְחָה גְּדוֹלָה מִלְּאָה אֶת לִבָּם שֶׁל הַלּוֹמְדִים.”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “קוֹל שְׁרִיקַת עֲנָפִים חֲשׁוּפִים נִשְׁמַע עַל רִצְפַּת הָאֶבֶן.”.
דיווח: כתבו ידיעה קצרה שכותרתה “לימוד לשון הקודש מאמתו של בית רבי”.
