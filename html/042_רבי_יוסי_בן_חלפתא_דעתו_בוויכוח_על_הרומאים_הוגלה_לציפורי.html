דעתו בוויכוח על הרומאים — הוגלה לציפורי

דף עבודה בעקבות רבי יוסי בן חלפתא
שם: כיתה:

המעשה — הסיפור המלא

הנה מערך חינוכי וסיפור מקיף ומרתק על רבי יוסי בן חלפתא וגלותו לציפורי, מותאם במיוחד לבנים בחינוך הדתי-לאומי בכיתות ב'-ד', על פי כל כללי הסטוריטלינג שביקשת.

פְּתִיחָה: הַפְּגִישָׁה הַסּוֹדִית בְּצֵל הָרוֹמָאִים

יְלָדִים, אֲנַחְנוּ נִמְצָאִים בְּאֶרֶץ יִשְׂרָאֵל שֶׁלְּאַחַר חֻרְבַּן בֵּית הַמִּקְדָּשׁ הַשֵּׁנִי. אוֹתָם יָמִים הָיוּ יָמִים לֹא פְּשׁוּטִים לְעַם יִשְׂרָאֵל. כַּאֲשֶׁר יְהוּדִי הָיָה הוֹלֵךְ בָּרְחוֹב, הוּא הָיָה יָכוֹל לִשְׁמֹעַ אֶת קוֹל צַעֲדֵיהֶם הַכְּבֵדִים שֶׁל נַעֲלֵי הַבַּרְזֶל שֶׁל חַיָּלֵי רוֹמִי, צְעָדוֹת שֶׁהִרְעִידוּ אֶת הָאֲדָמָה: טְרָאח... טְרָאח... טְרָאח...

הָרוֹמָאִים שָׁלְטוּ בָּאָרֶץ בְּיָד קָשָׁה, אַךְ כְּדֵי לְהַרְאוֹת שֶׁהֵם "שַׁלִּיטִים טוֹבִים", הֵם הִתְחִילוּ לִבְנוֹת בָּאָרֶץ כָּל מִינֵי בִּנְיָנִים וְשִׁפּוּצִים נוֹצְצִים מִשַּׁיִשׁ.

יוֹם אֶחָד, בְּחֶדֶר סָגוּר וְשָׁקֵט שֶׁבְּאַחַת מֵעָרֵי יְהוּדָה, הִתְקַבְּצוּ יַחַד שְׁלוֹשָׁה מִגְּדוֹלֵי הַתַּנָּאִים שֶׁבַּדּוֹר: רַבִּי יְהוּדָה בַּר אִלְעָאי, רַבִּי יוֹסֵי בֶּן חֲלַפְתָּא, וְרַבִּי שִׁמְעוֹן בַּר יוֹחַאי (רַשְׁבִּ"י). אִתָּם יָשַׁב אָדָם נוֹסָף שֶׁקָּרְאוּ לוֹ יְהוּדָה בֶּן גֵּרִים.

תּוֹךְ כְּדֵי דִּבּוּר בְּדִבְרֵי תּוֹרָה, עָלָה הַנּוֹשֵׂא שֶׁל הַשִּׁלְטוֹן הָרוֹמָאִי. הַתַּנָּא רַבִּי יְהוּדָה בַּר אִלְעָאי, שֶׁתָּמִיד חִפֵּשׂ צַד זְכוּת, אָמַר: "רְאוּ כַּמָּה נָאִים מַעֲשֵׂיהֶם שֶׁל הָרוֹמָאִים! הֵם בָּנוּ גְּשָׁרִים מְפֹאָרִים מֵאֶבֶן שֶׁמְּחַבְּרִים בֵּין מְקוֹמוֹת קָשִׁים, הֵם תִּקְּנוּ שְׁוָקִים יָפִים כְּדֵי שֶׁאֲנָשִׁים יוּכְלוּ לִקְנוֹת אֹכֶל בְּקַלּוּת, וְהֵם בָּנוּ בָּתֵּי מֶרְחָץ עִם מַיִם חַמִּים כְּדֵי שֶׁבְּנֵי הָאָדָם יוּכְלוּ לִרְחֹץ וְלִהְיוֹת בְּרִיאִים".

הַשְּׁתִיקָה הַגְּדוֹלָה וְהָאֵשׁ שֶׁל רַשְׁבִּ"י

כֻּלָּם הִפְנוּ אֶת מַבָּטָם אֶל הַתַּנָּא רַבִּי יוֹסֵי بֶּן חֲלַפְתָּא, מְצַפִּים לִשְׁמֹעַ מָה תִּהְיֶה דַּעְתּוֹ שֶׁל הַחָכָם הַגָּדוֹל, אָדָם שֶׁמֵּאוֹת מֵהֲלָכוֹת הַמִּשְׁנָה יָצְאוּ מִפִּיו. רַבִּי יוֹסֵי הִבִּיט בַּחֲבֵרָיו. בְּתוֹךְ לִבּוֹ, הוּא יָדַע הֵיטֵב שֶׁהָרוֹמָאִים לֹא עוֹשִׂים דָּבָר לְמַעַן הַיְּהוּדִים, אַךְ עֵינָיו רָאוּ גַּם אֶת הַסַּכָּנָה הַגְּדוֹלָה בְּהַמְרָדַת הַשִּׁלְטוֹן. מָה עָשָׂה רַבִּי יוֹסֵי? הוּא שָׁתַק. הוּא לֹא הִסְכִּים לְשַׁבֵּחַ גּוֹיִם רְשָׁעִים מִצַּד אֶחָד, אַךְ לֹא רָצָה לְהוֹצִיא בִּקֹּרֶת עֲלֵיהֶם מִצַּד שֵׁנִי. שְׁתִיקָה טְהוֹרָה וַחֲכָמָה שֶׁל תַּלְמִיד חָכָם.

אֲבָל אָז... קָם רַבִּי שִׁמְעוֹן בַּר יוֹחַאי! עֵינָיו הַקְּדוֹשׁוֹת בָּעֲרוּ כְּמוֹ לַפִּידִים שֶׁל אֵשׁ! הוּא לֹא הָיָה מְסֻגָּל לִשְׁמֹעַ שֶׁמְּשַׁבְּחִים אֶת מַחֲרִיבֵי הַמִּקְדָּשׁ. קוֹלוֹ הָרַעַם הִדְהֵד בַּחֶדֶר: "כָּל מָה שֶׁתִּקְּנוּ – לֹא תִּקְּנוּ אֶלָּא לְצֹרֶךְ עַצְמָם! אֶת הַשְּׁוָקִים בָּנוּ כְּדֵי שֶׁיּוּכְלוּ לֶאֱסֹף כֶּסֶף מִמִּסִּים, וְאֶת בָּתֵּי הַמֶּרְחָץ בָּנוּ רַק כְּדֵי לְפַנֵּק אֶת הַגּוּף שֶׁל עַצְמָם בְּמַיִם מְבַעְבְּעִים! אֵין מַעֲשֵׂיהֶם לְשֵׁם שָׁמַיִם, רַק לְשֵׁם עַצְמָם הָרְשָׁעִים!".

הַשִּׂיחָה הִסְתַּיְּמָה, וְכָל רַב חָזַר לְתַלְמוּדוֹ. אֲבָל יְלָדִים, סוֹדוֹת הֵם כְּמוֹ נוֹצוֹת בָּרוּחַ – מֻקְדָּם אוֹ מְאֻחָר, הֵם בּוֹרְחִים הַחוּצָה. יְהוּדָה בֶּן גֵּרִים, שֶׁיָּשַׁב אִתָּם, סִפֵּר בְּתֹם לֵב לְבִנֵי מִשְׁפַּחְתּוֹ עַל הַוִּכּוּחַ מַרְתִּיחַ הַדָּם שֶׁהָיָה בֵּין הַחֲכָמִים. הַשְּׁכֵנִים שָׁמְעוּ, וּמִשָּׁם לְעוֹבְרֵי הָאֹרַח, עַד שֶׁהַדְּבָרִים הִתְגַּלְגְּלוּ יָשָׁר לְאָזְנָיו שֶׁל הַקֵּיסָר הָרוֹמָאִי עַצְמוֹ!

גְּזַר הַדִּין שֶׁל הַקֵּיסָר

הַקֵּיסָר זָעַם. "יְהוּדִים יוֹשְׁבִים וּמְדַבְּרִים עָלַי?!" הוּא טָרַק בְּאֶגְרוֹפוֹ עַל הַשֻּׁלְחָן. "אֲנִי מְצַוֶּה לְהוֹצִיא גְּזֵרָה מִיָּדִית! רַבִּי יְהוּדָה שֶׁשִּׁבַּח אוֹתָנוּ – יִתְעַלֶּה. הוּא יִהְיֶה 'רֹאשׁ הַמְדַבְּרִים בְּכָל מָקוֹם'. רַבִּי שִׁמְעוֹן בַּר יוֹחַאי שֶׁגִּנָּה אוֹתָנוּ – דִּינוֹ לְמָוֶת! לַהֲרֹג אוֹתוֹ!" (וּכְמוֹ שֶׁאֲנַחְנוּ יוֹדְעִים, בִּגְלַל הַגְּזֵרָה הַזּוֹ נֶאֱלַץ רַשְׁבִּ"י לִבְרֹחַ לַמְּעָרָה עִם בְּנוֹ רַבִּי אֶלְעָזָר לְמֶשֶׁךְ שְׁלֹשׁ עֶשְׂרֵה שָׁנָה).

"וּמָה עִם הָרַב הַשְּׁלִישִׁי? הַהוּא שֶׁשָּׁתַק? רַבִּי יוֹסֵי?" שָׁאֲלוּ יֹעֲצֵי הַקֵּיסָר.

הַקֵּיסָר צִמְצֵם אֶת עֵינָיו. "מִי שֶׁשּׁוֹתֵק וְלֹא מְגֵן עָלֵינוּ... דִּינוֹ גָּלוּת! אֲנִי מְגָרֵשׁ אוֹתוֹ! יוֹסֵי יִגְלֶה מִבֵּיתוֹ שֶׁבַּדָּרוֹם הַרְחֵק צָפוֹנָה, אֶל הָעִיר צִפּוֹרִי שֶׁבַּגָּלִיל!"

הַהֲלִיכָה לַגָּלוּת – עֹנֶשׁ אוֹ מַתָּנָה?

חַיָּלִים רוֹמָאִים הִגִּיעוּ לְבֵיתוֹ שֶׁל רַבִּי יוֹסֵי בֶּן חֲלַפְתָּא. הֵם צִוּוּ עָלָיו לֶאֱרֹז תּוֹךְ זְמַן קָצָר אֶת מְעַט מִטַּלְטְלָיו – טַלִּיתוֹ, תְּפִלָּיו וּמִסְפַּר גְּוִילֵי סְפָרִים שֶׁהָיוּ לְיָדוֹ, וְגֵרְשׁוּ אוֹתוֹ מִבֵּיתוֹ לְלֹא רַחֲמִים.

יְלָדִים, דַּמְיְנוּ מָה עָבַר בְּלִבּוֹ שֶׁל רַבִּי יוֹסֵי. לַעֲזֹב אֶת קְהִלָּתוֹ הָאֲהוּבָה? אֶת בֵּית הַמִּדְרָשׁ שֶׁקּוֹלוֹת לִמּוּד הַתּוֹרָה בָּקְעוּ מִמֶּנּוּ כָּל דַּקָּה? הַהֲלִיכָה צָפוֹנָה הָיְתָה אֲרֻכָּה וּמְיַגַּעַת. הַשֶּׁמֶשׁ שֶׁל אֶרֶץ יִשְׂרָאֵל קָפְחָה עַל רֹאשׁוֹ, וַאֲבַק הַדְּרָכִים הָרֵיחָנִי נִדְבַּק לְבִגְדֵי אוֹת הָרַב. כָּל צַעַד הִרְחִיק אוֹתוֹ מֵחַיָּיו הַקּוֹדְמִים.

אָדָם רָגִיל הָיָה וַדַּאי בּוֹכֶה וּמִתְלוֹנֵן: "לָמָּה עָשׂוּ לִי אֶת זֶה?! רַק בִּגְלַל שֶׁשָּׁתַקְתִּי?! אֵיזֶה חֹסֶר צֶדֶק! אֵיזֶה חֹסֶר מַזָּל!".

אֲבָל תַּנָּא קָדוֹשׁ כְּרַבִּי יוֹסֵי לֹא הֶאֱמִין בְּ"מַזָּל" אוֹ בְּ"עֹנֶשׁ" שֶׁל רוֹמָאִים. תּוֹךְ כְּדֵי הֲלִיכָה מְעַיֶּפֶת הוּא אָמַר לְעַצְמוֹ: "אִם הַקָּדוֹשׁ בָּרוּךְ הוּא, אָבִי בַּשָּׁמַיִם הָאוֹהֵב שֶׁלִּי, גִּלְגֵּל אֶת הַדְּבָרִים כָּךְ שֶׁאֲנִי אֶסַּע לְצִפּוֹרִי – סִימָן שֶׁשָּׁם הַתַּפְקִיד שֶׁלִּי! סִימָן שֶׁיֵּשׁ לִי שְׁלִיחוּת שָׁם!". הַמַּחֲשָׁבָה הַזּוֹ מִלְּאָה אֶת רַגְלָיו הָעֲיֵפוֹת בְּכֹחַ חָדָשׁ.

אוֹר חָדָשׁ בְּצִפּוֹרִי

כַּאֲשֶׁר נִכְנַס רַבִּי יוֹסֵי לְצִפּוֹרִי, הוּא גִּלָּה עִיר יָפָה מְאֹד. אֲוִיר קָרִיר נָשַׁב מֵהָרֵי הַגָּלִיל, שְׂדוֹת זֵיתִים יְרֻקִּים הִקִּיפוּ אוֹתָהּ, וְאֶשְׁכּוֹלוֹת עֲנָבִים מְתוּקִים גָּדְלוּ בָּהּ מִכָּל עֵבֶר. אַךְ מַשֶּׁהוּ חָסֵר... הַיְּהוּדִים בְּצִפּוֹרִי הָיוּ עֲסוּקִים בַּעֲבוֹדַת הָאֲדָמָה וּבְמִסְחָר, אַךְ כִּמְעַט וְלֹא נִשְׁמַע בָּעִיר קוֹל סִלְסוּלָהּ הַמָּתוֹק שֶׁל הַתּוֹרָה. רַבִּי יוֹסֵי הֵבִין אֶת סוֹד גָּלוּתוֹ!

הוּא אָסַף סְבִיבוֹ אֶת תּוֹשָׁבֵי צִפּוֹרִי. בַּסַּבְלָנוּת הַגְּדוֹלָה שֶׁלּוֹ, בְּחִיּוּךְ רַךְ וּבְקוֹל עָמֹק וְחַם, הוּא הֵחֵל לְלַמֵּד אוֹתָם. הוּא פָּתַח יְשִׁיבָה גְּדוֹלָה! מִי כָּמוֹהוּ יָדַע אֶת הֲלָכוֹת הַתּוֹרָה? מִי כָּמוֹהוּ הָיָה בָּקִי בְּתַלְמוּד וּבְמִדְרָשׁ? תּוֹךְ זְמַן קָצָר, הַתּוֹשָׁבִים שֶׁאָהֲבוּ אוֹתוֹ מְאֹד הֵחֵלּוּ לִנְהֹר לְבֵית הַמִּדְרָשׁ שֶׁלּוֹ. צִפּוֹרִי, הָעִיר הַשְּׁקֵטָה, הָפְכָה לְעִיר שֶׁשּׁוֹקֶקֶת תּוֹרָה!

הָרוֹמָאִים חָשְׁבוּ שֶׁהֵם מַעֲנִישִׁים וּמַסְתִּירִים אֶת רַבִּי יוֹסֵי בֶּן חֲלַפְתָּא בְּחוֹר מְרֻחָק. אֲבָל דַּוְקָא שָׁם, בְּצִפּוֹרִי שֶׁבַּגָּלִיל, זָרַח אוֹרוֹ הַגָּדוֹל שֶׁל רַבִּי יוֹסֵי אֶל כָּל עַם יִשְׂרָאֵל. לֹא פָּחוֹת מִשְּׁלֹשׁ מֵאוֹת הֲלָכוֹת (!!!) בַּמִּשְׁנָה כְּתוּבוֹת עַל שְׁמוֹ שֶׁל רַבִּי יוֹסֵי (וּכְשֶׁכָּתוּב "רַבִּי יוֹסֵי" סְתָם בַּמִּשְׁנָה, זֶה הוּא!).

א. בתחילת הסיפור נאמר: “הנה מערך חינוכי וסיפור מקיף ומרתק על רבי יוסי בן חלפתא וגלותו לציפורי, מותאם במיוחד לבנים בחינוך הדתי-לאומי בכיתות ב'-ד', על פי כל כללי הסטוריטלינג שביקשת.” — מה קרה מיד לאחר מכן?

□ הַשֶּׁמֶשׁ שֶׁל אֶרֶץ יִשְׂרָאֵל קָפְחָה עַל רֹאשׁוֹ, וַאֲבַק הַדְּרָכִים הָרֵיחָנִי נִדְבַּק לְבִגְדֵי אוֹת הָרַב.
□ הנה מערך חינוכי וסיפור מקיף ומרתק על רבי יוסי בן חלפתא וגלותו לציפורי, מותאם במיוחד לבנים בחינוך הדתי-לאומי בכיתות ב'-ד', על פי כל כללי הסטוריטלינג שביקשת.
□ כך הילדים בונים "ישיבה/בית מדרש" יחד, בדיוק כפי שעשה רבי יוסי בציפורי.
□ קוֹלוֹ הָרַעַם הִדְהֵד בַּחֶדֶר: "כָּל מָה שֶׁתִּקְּנוּ – לֹא תִּקְּנוּ אֶלָּא לְצֹרֶךְ עַצְמָם!
□ לֹא פָּחוֹת מִשְּׁלֹשׁ מֵאוֹת הֲלָכוֹת (!!!) בַּמִּשְׁנָה כְּתוּבוֹת עַל שְׁמוֹ שֶׁל רַבִּי יוֹסֵי (וּכְשֶׁכָּתוּב "רַבִּי יוֹסֵי" סְתָם בַּמִּשְׁנָה, זֶה…

סיבה ותוצאה מתוך המעשה

האירוע הנתוןמה גרם לו?מה הייתה תוצאתו?
אֶת הַשְּׁוָקִים בָּנוּ כְּדֵי שֶׁיּוּכְלוּ לֶאֱסֹף כֶּסֶף מִמִּסִּים, וְאֶת בָּתֵּי הַמֶּרְחָץ בָּנוּ רַק כְּדֵי לְפַנֵּק אֶת הַגּוּף שֶׁל עַצְמָם…
הַיְּהוּדִים בְּצִפּוֹרִי הָיוּ עֲסוּקִים בַּעֲבוֹדַת הָאֲדָמָה וּבְמִסְחָר, אַךְ כִּמְעַט וְלֹא נִשְׁמַע בָּעִיר קוֹל סִלְסוּלָהּ הַמָּתוֹק שֶׁל…

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “דעתו בוויכוח על הרומאים — הוגלה לציפורי” על רבי יוסי בן חלפתא? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “חַיָּלִים רוֹמָאִים הִגִּיעוּ לְבֵיתוֹ שֶׁל רַבִּי יוֹסֵי בֶּן חֲלַפְתָּא.”.
ג. אילו רבי יוסי בן חלפתא לא היה פועל כפי שמתואר ב־“אֲבָל דַּוְקָא שָׁם, בְּצִפּוֹרִי שֶׁבַּגָּלִיל, זָרַח אוֹרוֹ הַגָּדוֹל שֶׁל רַבִּי יוֹסֵי אֶל כָּל עַם יִשְׂרָאֵל.” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “הַמְלָצוֹת לַמַּצֶּגֶת (4 שְׁקוּפִיּוֹת, בְּלִי לְהֵרָאוֹת חֲסַר צְנִיעוּת):” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. הנה מערך חינוכי וסיפור מקיף ומרתק על רבי יוסי בן חלפתא וגלותו לציפורי,…
2. בְּתוֹךְ לִבּוֹ, הוּא יָדַע הֵיטֵב שֶׁהָרוֹמָאִים לֹא עוֹשִׂים דָּבָר…
3. לַהֲרֹג אוֹתוֹ!" (וּכְמוֹ שֶׁאֲנַחְנוּ יוֹדְעִים, בִּגְלַל הַגְּזֵרָה…
4. כַּאֲשֶׁר נִכְנַס רַבִּי יוֹסֵי לְצִפּוֹרִי, הוּא גִּלָּה עִיר יָפָה…
5. רַבִּי יוֹסֵי לֹא הִתְאוֹנֵן עַל כָּךְ שֶׁהַתָּכְנִיּוֹת שֶׁלּוֹ…
6. כך הילדים בונים "ישיבה/בית מדרש" יחד, בדיוק כפי שעשה רבי יוסי בציפורי.

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “הַשְּׁכֵנִים שָׁמְעוּ, וּמִשָּׁם לְעוֹבְרֵי הָאֹרַח, עַד שֶׁהַדְּבָרִים הִתְגַּלְגְּלוּ יָשָׁר לְאָזְנָיו שֶׁל הַקֵּיסָר…”.
מכתב: כתבו לרבי יוסי בן חלפתא בעקבות “הַיְּהוּדִים בְּצִפּוֹרִי הָיוּ עֲסוּקִים בַּעֲבוֹדַת הָאֲדָמָה וּבְמִסְחָר, אַךְ כִּמְעַט וְלֹא נִשְׁמַע בָּעִיר קוֹל…”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “רַבִּי יוֹסֵי לֹא הִתְאוֹנֵן עַל כָּךְ שֶׁהַתָּכְנִיּוֹת שֶׁלּוֹ נֶהֶרְסוּ וְהוּא אֻלַּץ לַעֲבֹר דִּירָה עֵקֶב עֹנֶשׁ הַגָּלוּת.”.
דיווח: כתבו ידיעה קצרה שכותרתה “דעתו בוויכוח על הרומאים — הוגלה לציפורי”.
