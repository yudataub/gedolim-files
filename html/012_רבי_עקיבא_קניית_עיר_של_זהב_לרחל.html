סיפור 12 • רבי עקיבא • קניית עיר של זהב לרחל

קניית עיר של זהב לרחל

דף עבודה בעקבות רבי עקיבא
שם: כיתה:

המעשה — הסיפור המלא

פְּתִיחָה: לַיְלָה קָפוּא בַּמַּתְבֵּן

יְלָדִים, נַסּוּ לְדַמְיֵן לַיְלָה חָרְפִּי קַר מְאֹד. הָרוּחַ בַּחוּץ נוֹשֶׁבֶת בְּחָזְקָה, שׁוֹרֶקֶת דֶּרֶךְ סְדָקִים בְּקִירוֹת שֶׁל עֵץ, וְהַקֹּר – קֹר אֵימִים, מַמָּשׁ מַקְפִּיא אֶת קְצוֹת הָאֶצְבָּעוֹת.

בְּתוֹךְ מַחְסָן קָטָן וְיָשָׁן, מַחְסָן שֶׁמָּלֵא בַּעֲרֵמוֹת שֶׁל תֶּבֶן וְקַשׁ, יוֹשְׁבִים שְׁנֵי אֲנָשִׁים. אֵין לָהֶם תַּנּוּר מְחַמֵּם, אֵין לָהֶם מִטּוֹת נוֹחוֹת וְלֹא שְׂמִיכוֹת צֶמֶר עָבוֹת. הֵם יוֹשְׁבִים עַל הָאֲדָמָה הַקָּרָה, מִתְכַּסִּים בִּקְצָת תֶּבֶן וְקַשׁ כְּדֵי שֶׁלֹּא לִרְעֹד מִקֹּר.

אֵלּוּ הָיוּ רַבִּי עֲקִיבָא וְאִשְׁתּוֹ, הַצַּדֶּקֶת רָחֵל.

רָחֵל הָיְתָה רְגִילָה לְחַיִּים מְלֵאִים בְּעֹשֶׁר. אַבָּא שֶׁלָּהּ הָיָה "כַּלְבָּא שָׂבוּעַ", הָאִישׁ הֲכִי עָשִׁיר בִּירוּשָׁלַיִם. אֲבָל רָחֵל וִתְּרָה עַל הַכֹּל! עַל הַבְּגָדִים הַיָּפִים, עַל הָאֹכֶל הַטָּעִים וְעַל הַחֶדֶר הֶחָם, וְהַכֹּל כְּדֵי לִזְכּוֹת בְּדָבָר אֶחָד – שֶׁבַּעְלָהּ יֵלֵךְ לִלְמֹד תּוֹרָה. אָבִיהָ כָּעַס, סִלֵּק אוֹתָהּ מִבֵּיתוֹ, וְעַכְשָׁו אֵין לָהֶם כְּלוּם. רַק זֶה אֶת זוֹ, וֶאֱמוּנָה גְּדוֹלָה בַּה'.

הַהַבְטָחָה הַגְּדוֹלָה בַּחֹשֶׁךְ

בַּבֹּקֶר הַקַּר, קָם רַבִּי עֲקִיבָא. הַתֶּבֶן הַיָּבֵשׁ הָיָה עֲקָצָנִי וְדוֹקֵר, וּרְסִיסִים שֶׁל קַשׁ נִדְבְּקוּ לְשַׂעֲרוֹתֶיהָ הָאֲרֻכּוֹת שֶׁל רָחֵל.

רַבִּי עֲקִיבָא, בַּעֲדִינוּת רַבָּה, הִתְחִיל לְלַקֵּט אֶת הַתֶּבֶן וּלְהוֹצִיא אֶת הַקּוֹצִים מִשַּׂעֲרוֹתֶיהָ. הַלֵּב שֶׁלּוֹ כָּאַב. הוּא רָאָה מַה אִשְׁתּוֹ מַקְרִיבָה בִּשְׁבִילוֹ וּבִשְׁבִיל הַתּוֹרָה הַקְּדוֹשָׁה שֶׁהוּא עָתִיד לִלְמֹד. אֲבָל בְּדִיּוּק בָּרֶגַע הַזֶּה, כְּשֶׁלֹּא הָיָה לָהֶם אֲפִלּוּ מַטְבֵּעַ קָטָן שֶׁל נְחֹשֶׁת לִקְנוֹת בּוֹ לֶחֶם, רַבִּי עֲקִיבָא חִיֵּךְ חִיּוּךְ רָחָב, הִסְתַּכֵּל יָשָׁר לְעֵינֶיהָ שֶׁל רָחֵל וְאָמַר:

"רָחֵל, אִשְׁתִּי הַיְּקָרָה! אִלּוּ רַק הָיָה לִי כֶּסֶף, הָיִיתִי קוֹנֶה לָךְ אֶת הַתַּכְשִׁיט הֲכִי יָקָר שֶׁיֵּשׁ... הָיִיתִי שָׂם עַל רֹאשֵׁךְ תַּכְשִׁיט שֶׁל 'יְרוּשָׁלַיִם שֶׁל זָהָב'!" (זֶה הָיָה תַּכְשִׁיט נָדִיר וְיָקָר שֶׁרַק נְשׁוֹת הָעֲשִׁירִים הַגְּדוֹלִים בְּיוֹתֵר הָיוּ עוֹנְדוֹת עַל רֹאשָׁן, תַּכְשִׁיט מִזָּהָב טָהוֹר בְּצוּרַת חוֹמוֹת יְרוּשָׁלַיִם).

רָחֵל חִיְּכָה. הִיא לֹא בִּקְּשָׁה זָהָב. הִיא בִּקְּשָׁה תּוֹרָה. "לֵךְ וּלְמַד, עֲקִיבָא," לָחֲשָׁה בְּבִטָּחוֹן שָׁלֵם.

הַהַצְלָחָה בְּבֵית הַמִּדְרָשׁ וְהַזָּהָב שֶׁחָזַר

וְרַבִּי עֲקִיבָא הָלַךְ. הוּא הָלַךְ וְלָמַד עֶשְׂרִים וְאַרְבַּע שָׁנִים רְצוּפוֹת מִתּוֹךְ מְסִירוּת נֶפֶשׁ אֲדִירָה. וּבַבַּיִת, בְּמֶשֶׁךְ שָׁנִים אֲרֻכּוֹת, רָחֵל נִשְׁאֲרָה בְּדַלּוּת.

אֲבָל יְלָדִים, הַקָּדוֹשׁ בָּרוּךְ הוּא אֵינוֹ נִשְׁאָר חַיָּב לְמִי שֶׁמּוֹסֵר נַפְשׁוֹ עַל תּוֹרָתוֹ!

כַּאֲשֶׁר חָזַר רַבִּי עֲקִיבָא לִירוּשָׁלַיִם, הוּא כְּבָר הָיָה הַתַּנָּא הַגָּדוֹל בְּיוֹתֵר בַּדּוֹר, שַׂר הַתּוֹרָה! עֶשְׂרִים וְאַרְבָּעָה אֶלֶף תַּלְמִידִים הָלְכוּ אַחֲרָיו. רַעַשׁ לִמּוּד תּוֹרָתָם הָיָה אֲדִיר וַחֲזָק כְּמוֹ רַעַשׁ שֶׁל גַּלֵּי הַיָּם הַמִּתְנַפְּצִים אֶל הַחוֹף! הָאֲוִיר בְּבֵית הַמִּדְרָשׁ הָיָה סָפוּג בְּרֵיחַ מָתוֹק וְקָדוֹשׁ שֶׁל גְּוִילִים יְשָׁנִים וְסִפְרֵי קֹדֶשׁ.

לְשֵׁמַע גְּדֻלָּתוֹ שֶׁל רַבִּי עֲקִיבָא, חָזַר בּוֹ כַּלְבָּא שָׂבוּעַ (חֹתְנוֹ הֶעָשִׁיר) מִנִּדְרוֹ, וְנָתַן לְרַבִּי עֲקִיבָא וּלְרָחֵל חֲצִי מִכָּל עָשְׁרוֹ הָעָצוּם. רַבִּי עֲקִיבָא, שֶׁהָיָה עָנִי מָרוּד, נַעֲשָׂה עָשִׁיר גָּדוֹל מְאֹד בְּהַשְׁגָּחָה פְּרָטִית מֻפְלָאָה מֵאֵת ה'.

הַהֲלִיכָה לַצּוֹרֵף וַעֲשִׂיַּת הַכֶּתֶר

רַבִּי עֲקִיבָא לֹא שָׁכַח אֶת אִשְׁתּוֹ וְאֶת הַהַבְטָחָה מֵהַמַּתְבֵּן הַקַּר. גְּדוֹל הַדּוֹר שֶׁכָּל נְשִׂיאִים וַחֲכָמִים שׁוֹחֲרִים לְפִתְחוֹ, פָּנָה אֶל שׁוּק הַצּוֹרְפִים – הָאָמָּנִים הַמְּכִינִים תַּכְשִׁיטִים. הוּא נִגַּשׁ לַצּוֹרֵף הַמֻּמְחֶה בְּיוֹתֵר.

"אֲנִי מְבַקֵּשׁ שֶׁתָּכִין תַּכְשִׁיט שֶׁל יְרוּשָׁלַיִם שֶׁל זָהָב," אָמַר לוֹ הַתַּנָּא בְּנַחַת. "עֲשֵׂה אוֹתוֹ מֵהַזָּהָב הַטָּהוֹר וְהַמַּבְרִיק בְּיוֹתֵר. שֶׁכָּל חוֹמָה, וְכָל מִגְדָּל שֶׁל יְרוּשָׁלַיִם יִהְיוּ חֲקוּקִים בּוֹ בְּיוֹפִי רַב".

כְּשֶׁהָיָה הַכֶּתֶר מוּכָן, הוּא נָצַץ וְזָרַח לְאוֹר הַשֶּׁמֶשׁ. רַבִּי עֲקִיבָא הֵבִיא אֶת הַתַּכְשִׁיט הַמַּדְהִים הַזֶּה לְרָחֵל, אִשְׁתּוֹ הַצַּדֶּקֶת, וְהִנִּיחַ אוֹתוֹ עַל רֹאשָׁהּ בְּכָבוֹד וּבְיִרְאַת שָׁמַיִם. עַכְשָׁו, בַּמָּקוֹם שֶׁבּוֹ הָיוּ קוֹצִים שֶׁל תֶּבֶן שֶׁדָּקְרוּ – זָרַח כֶּתֶר שֶׁל זָהָב יְרוּשַׁלְמִי שֶׁהֵאִיר אֶת כָּל הַחֶדֶר.

תְּגוּבַת הַתַּלְמִידִים וְתְשׁוּבַת הַמַּחַץ

תַּלְמִידָיו שֶׁל רַבִּי עֲקִיבָא, וְאַף רַבָּן גַּמְלִיאֵל וְאִשְׁתּוֹ, רָאוּ אֶת הַכֶּתֶר הָעָצוּם. קְצָת מֵהַתַּלְמִידִים הִרְגִּישׁוּ קְצָת לֹא בְּנוֹחַ, וַאֲפִלּוּ אִשְׁתּוֹ שֶׁל רַבָּן גַּמְלִיאֵל קִנְּאָה מְעַט.

"רַבֵּנוּ," בִּקְּשׁוּ הַתַּלְמִידִים וַאֲחֵרִים לוֹמַר בְּשֶׁקֶט, "אוּלַי הַתַּכְשִׁיט הַזֶּה יָקָר מִדַּי? אוּלַי זֶה יוֹתֵר מִדַּי בּוֹלֵט לְאֵשֶׁת רַב הַיְשִׁיבָה?"

רַבִּי עֲקִיבָא לֹא גָּעַר בָּהֶם בְּכַעַס, אֲבָל קוֹלוֹ הָפַךְ מוּצָק וְעָמֹק, מָלֵא בְּרֶגֶשׁ. הוּא הִבִּיט עַל קְהַל קְדוֹשִׁים שֶׁעָמַד מוּלוֹ, הִצְבִּיעַ עַל אִשְׁתּוֹ הַצַּדֶּקֶת וְאָמַר לְכֻלָּם מִשְׁפָּט שֶׁיִּהְדְהֵד לְנֶצַח בְּלֵב יִשְׂרָאֵל:

"הַרְבֵּה צַעַר נִצְטַעֲרָה עִמִּי בַּתּוֹרָה!"

הוּא הִזְכִּיר לָהֶם אֶת הַקֹּר, אֶת הָרָעָב, אֶת הַתֶּבֶן וְאֶת הַבְּדִידוּת. "כְּשֶׁהָיִיתִי רוֹעֶה פָּשׁוּט – הִיא הֶאֱמִינָה בִּי. כְּשֶׁרָעַדְנוּ מִקֹּר – הִיא עָמְדָה מֵאֲחוֹרַי וּדְחָפַתְנִי לִלְמֹד. הַזָּהָב הַזֶּה אֵינוֹ סְתָם גַּאֲוָה שֶׁל תַּכְשִׁיטִים," הוֹסִיפָה שְׁתִיקָתוֹ לוֹמַר, "הַזָּהָב הַזֶּה הוּא כֶּתֶר הַכָּבוֹד שֶׁל הַתּוֹרָה עַצְמָהּ! מִי שֶׁמּוֹסֶרֶת נֶפֶשׁ שֶׁלְּמַעַן הַתּוֹרָה – רְאוּיָה לְכֶתֶר הַזָּהָב שֶׁל יְרוּשָׁלַיִם!"

וְכָל הַתַּלְמִידִים בְּתַדְהֵמָה וּבְיִרְאַת כָּבוֹד, הוֹרִידוּ אֶת רֹאשָׁם בְּשֶׁקֶט, לוֹמְדִים פֶּרֶק עָצוּם בְּהַכָּרַת הַטּוֹב וּבִכְבוֹד הַתּוֹרָה.

מקורות: תלמוד בבלי, מסכת נדרים, דף נ', עמוד א': "...בסיתוא הוה גנו בי תיבנא... אמר לה: אי הואי לי רמינא ליך ירושלים דדהבא..." (ההבטחה במתבן שייתן לה ירושלים של זהב). תלמוד בבלי, מסכת שבת, דף נ"ט, עמוד א': "רבי עקיבא עבד לה לאינתותיה עיר של זהב..." (הקמת התכשיט וענידת עיר של זהב). "...אמרו ליה תלמידיו: רבי, ביישתנו ממה שעשית לה. אמר להם: הרבה צער נצטערה עמי בתורה" (תגובתו לתלמידיו - מובא גם באבות דרבי נתן פרק ו' ).

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “הָרוּחַ בַּחוּץ נוֹשֶׁבֶת בְּחָזְקָה, שׁוֹרֶקֶת דֶּרֶךְ סְדָקִים בְּקִירוֹת שֶׁל עֵץ, וְהַקֹּר – קֹר אֵימִים, מַמָּשׁ מַקְפִּיא אֶת קְצוֹת הָאֶצְבָּעוֹת.” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “אָבִיהָ כָּעַס, סִלֵּק אוֹתָהּ מִבֵּיתוֹ, וְעַכְשָׁו אֵין לָהֶם כְּלוּם.”?
ג. כיצד הגיב רבי עקיבא למצב המתואר: “הוּא הָלַךְ וְלָמַד עֶשְׂרִים וְאַרְבַּע שָׁנִים רְצוּפוֹת מִתּוֹךְ מְסִירוּת נֶפֶשׁ אֲדִירָה.”?
ד. מה השתנה בין “רַבִּי עֲקִיבָא לֹא שָׁכַח אֶת אִשְׁתּוֹ וְאֶת הַהַבְטָחָה מֵהַמַּתְבֵּן הַקַּר.” לבין “עַכְשָׁו, בַּמָּקוֹם שֶׁבּוֹ הָיוּ קוֹצִים שֶׁל תֶּבֶן שֶׁדָּקְרוּ – זָרַח כֶּתֶר שֶׁל זָהָב יְרוּשַׁלְמִי שֶׁהֵאִיר…”?
ה. כיצד הגיע הסיפור אל האירוע: “כְּשֶׁרָעַדְנוּ מִקֹּר – הִיא עָמְדָה מֵאֲחוֹרַי וּדְחָפַתְנִי לִלְמֹד.”?

3. מה היה ראשון?

כתבו ליד כל אירוע מספר מ־1 עד 5 לפי סדר התרחשותו בסיפור.

□ לְשֵׁמַע גְּדֻלָּתוֹ שֶׁל רַבִּי עֲקִיבָא, חָזַר בּוֹ כַּלְבָּא שָׂבוּעַ (חֹתְנוֹ הֶעָשִׁיר) מִנִּדְרוֹ, וְנָתַן לְרַבִּי עֲקִיבָא וּלְרָחֵל חֲצִי מִכָּל…
□ הָרוּחַ בַּחוּץ נוֹשֶׁבֶת בְּחָזְקָה, שׁוֹרֶקֶת דֶּרֶךְ סְדָקִים בְּקִירוֹת שֶׁל עֵץ, וְהַקֹּר – קֹר אֵימִים, מַמָּשׁ מַקְפִּיא אֶת קְצוֹת הָאֶצְבָּעוֹת.
□ וְכָל הַתַּלְמִידִים בְּתַדְהֵמָה וּבְיִרְאַת כָּבוֹד, הוֹרִידוּ אֶת רֹאשָׁם בְּשֶׁקֶט, לוֹמְדִים פֶּרֶק עָצוּם בְּהַכָּרַת הַטּוֹב וּבִכְבוֹד הַתּוֹרָה.
□ רַבִּי עֲקִיבָא, בַּעֲדִינוּת רַבָּה, הִתְחִיל לְלַקֵּט אֶת הַתֶּבֶן וּלְהוֹצִיא אֶת הַקּוֹצִים מִשַּׂעֲרוֹתֶיהָ.
□ תַּלְמִידָיו שֶׁל רַבִּי עֲקִיבָא, וְאַף רַבָּן גַּמְלִיאֵל וְאִשְׁתּוֹ, רָאוּ אֶת הַכֶּתֶר הָעָצוּם.

לפני ואחרי

מה קרה לפני: “לְשֵׁמַע גְּדֻלָּתוֹ שֶׁל רַבִּי עֲקִיבָא, חָזַר בּוֹ כַּלְבָּא שָׂבוּעַ (חֹתְנוֹ הֶעָשִׁיר) מִנִּדְרוֹ, וְנָתַן לְרַבִּי עֲקִיבָא וּלְרָחֵל חֲצִי מִכָּל…”?
מה קרה אחרי: “תַּלְמִידָיו שֶׁל רַבִּי עֲקִיבָא, וְאַף רַבָּן גַּמְלִיאֵל וְאִשְׁתּוֹ, רָאוּ אֶת הַכֶּתֶר הָעָצוּם.”?

סיבה ותוצאה מתוך המעשה

האירוע הנתוןמה גרם לו?מה הייתה תוצאתו?
הוּא רָאָה מַה אִשְׁתּוֹ מַקְרִיבָה בִּשְׁבִילוֹ וּבִשְׁבִיל הַתּוֹרָה הַקְּדוֹשָׁה שֶׁהוּא עָתִיד לִלְמֹד.
שֶׁכָּל חוֹמָה, וְכָל מִגְדָּל שֶׁל יְרוּשָׁלַיִם יִהְיוּ חֲקוּקִים בּוֹ בְּיוֹפִי רַב".

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “קניית עיר של זהב לרחל” על רבי עקיבא? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “רַעַשׁ לִמּוּד תּוֹרָתָם הָיָה אֲדִיר וַחֲזָק כְּמוֹ רַעַשׁ שֶׁל גַּלֵּי הַיָּם הַמִּתְנַפְּצִים אֶל הַחוֹף!”.
ג. אילו רבי עקיבא לא היה פועל כפי שמתואר ב־“עַכְשָׁו, בַּמָּקוֹם שֶׁבּוֹ הָיוּ קוֹצִים שֶׁל תֶּבֶן שֶׁדָּקְרוּ – זָרַח כֶּתֶר שֶׁל זָהָב יְרוּשַׁלְמִי שֶׁהֵאִיר אֶת כָּל…” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “כְּשֶׁרָעַדְנוּ מִקֹּר – הִיא עָמְדָה מֵאֲחוֹרַי וּדְחָפַתְנִי לִלְמֹד.” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. הָרוּחַ בַּחוּץ נוֹשֶׁבֶת בְּחָזְקָה, שׁוֹרֶקֶת דֶּרֶךְ סְדָקִים…
2. הַתֶּבֶן הַיָּבֵשׁ הָיָה עֲקָצָנִי וְדוֹקֵר, וּרְסִיסִים שֶׁל קַשׁ…
3. כַּאֲשֶׁר חָזַר רַבִּי עֲקִיבָא לִירוּשָׁלַיִם, הוּא כְּבָר הָיָה…
4. גְּדוֹל הַדּוֹר שֶׁכָּל נְשִׂיאִים וַחֲכָמִים שׁוֹחֲרִים לְפִתְחוֹ,…
5. קְצָת מֵהַתַּלְמִידִים הִרְגִּישׁוּ קְצָת לֹא בְּנוֹחַ, וַאֲפִלּוּ…
6. וְכָל הַתַּלְמִידִים בְּתַדְהֵמָה וּבְיִרְאַת כָּבוֹד, הוֹרִידוּ אֶת…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “הוּא הָלַךְ וְלָמַד עֶשְׂרִים וְאַרְבַּע שָׁנִים רְצוּפוֹת מִתּוֹךְ מְסִירוּת נֶפֶשׁ אֲדִירָה.”.
מכתב: כתבו לרבי עקיבא בעקבות “שֶׁכָּל חוֹמָה, וְכָל מִגְדָּל שֶׁל יְרוּשָׁלַיִם יִהְיוּ חֲקוּקִים בּוֹ בְּיוֹפִי רַב".”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “רַבִּי עֲקִיבָא לֹא גָּעַר בָּהֶם בְּכַעַס, אֲבָל קוֹלוֹ הָפַךְ מוּצָק וְעָמֹק, מָלֵא בְּרֶגֶשׁ.”.
דיווח: כתבו ידיעה קצרה שכותרתה “קניית עיר של זהב לרחל”.

המחזה

רגע מן הסיפורהדמויותמה יאמרו או יעשו?
הוּא רָאָה מַה אִשְׁתּוֹ מַקְרִיבָה בִּשְׁבִילוֹ וּבִשְׁבִיל הַתּוֹרָה הַקְּדוֹשָׁה שֶׁהוּא עָתִיד…
עַכְשָׁו, בַּמָּקוֹם שֶׁבּוֹ הָיוּ קוֹצִים שֶׁל תֶּבֶן שֶׁדָּקְרוּ – זָרַח כֶּתֶר שֶׁל זָהָב…
