ר' יוחנן — אהבתו לתורה

דף עבודה בעקבות ר' יוחנן וריש לקיש
שם: כיתה:

המעשה — הסיפור המלא

להלן מערך חינוכי, רגשי ומרתק, המשלב שניים מהסיפורים הגדולים ביותר על רבי יוחנן ומסירותו לתורה , יחד עם המפגש המכונן עם ריש לקיש . הסיפור מעובד בקפידה עבור בנים בכיתות ב'-ד' (חינוך דתי-לאומי), תוך שימוש בתיאורים חושיים, הקפדה על כבוד התורה, והמחשת מוסר ההשכל: התורה היא האוצר הגדול ביותר בעולם, ובכוחה להפוך גם את הרחוק ביותר לגדול בישראל .

הסיפור מנוקד במלואו כנדרש.

פְּתִיחָה: מַסָּע בְּנוֹפֵי הַגָּלִיל

יְלָדִים יְקָרִים, עִצְמוּ רֶגַע אֶת הָעֵינַיִם. צְאוּ אִתִּי לְמַסָּע קָסוּם בְּאֶרֶץ יִשְׂרָאֵל שֶׁל יְמֵי קֶדֶם.

הַשֶּׁמֶשׁ שֶׁל הַגָּלִיל מְחַמֶּמֶת מֵעָלֵינוּ, וְרוּחַ קַלָּה וּנְעִימָה מְבִיאָה אִתָּהּ רֵיחַ מָתוֹק שֶׁל אֲדָמָה חֲרוּשָׁה וְשָׂדוֹת יְרֻקִּים. בַּשְּׁבִיל הֶעָפָר הַמִּתְפַּתֵּל הַיּוֹרֵד מֵהָעִיר טְבֶרְיָה וְעוֹלֶה לְצִפּוֹרִי, צוֹעֲדִים שְׁנֵי אֲנָשִׁים הֲדוּרִים בְּבִגְדֵיהֶם. הַגָּדוֹל שֶׁבָּהֶם הוּא רַבִּי יוֹחָנָן [i], רֹאשׁ הַיְּשִׁיבָה וּגְדוֹל אָמוֹרָאֵי אֶרֶץ יִשְׂרָאֵל, וְלִמְלוֹתָיו צוֹעֵד בְּיִרְאַת כָּבוֹד תַּלְמִידוֹ הַנֶּאֱמָן, רַבִּי חִיָּא בַּר אַבָּא [i].

קְרָאק, קְרָאק , נִשְׁמַע קוֹל סַנְדְּלֵיהֶם עַל הָאֲבָנִים וְהֶעָפָר [i].

הֵם הָלְכוּ בַּדֶּרֶךְ וְשׂוֹחֲחוּ בְּדִבְרֵי תּוֹרָה. לְפֶתַע, הֵם הִגִּיעוּ לְשָׂדֶה אָרֹךְ וְרָחָב, הַמָּלֵא בְּשִׁבֳּלֵי חִטָּה זְהֻבּוֹת. הַשִּׁבֳּלִים נָעוּ בָּרוּחַ כְּמוֹ יָם שֶׁל זָהָב, וְרֵיחַ מְרַעֲנֵן שֶׁל תְּבוּאָה עָמַד בָּאֲוִיר.

הִסְתַּכֵּל רַבִּי יוֹחָנָן עַל הַשָּׂדֶה וְאָמַר לְתַלְמִידוֹ: "חִיָּא בְּנִי, אַתָּה רוֹאֶה אֶת הַשָּׂדֶה הַזֶּה? הַשָּׂדֶה הַזֶּה... הָיָה שֶׁלִּי! אֲנִי הָיִיתִי הַבְּעָלִים שֶׁלּוֹ. אֲבָל... מָכַרְתִּי אוֹתוֹ. לָמָּה? כְּדֵי שֶׁאֶזְכֶּה לִלְמֹד תּוֹרָה בְּלִי הַפְרָעוֹת" [i].

רַבִּי חִיָּא הִנְהֵן בְּרֹאשׁוֹ בִּפְלִיאָה. אֵיזֶה וִתּוּר גָּדוֹל עָשָׂה הָרַב שֶׁלּוֹ! הֵם הִמְשִׁיכוּ לָלֶכֶת, וְהִנֵּה נִגְלָה לְעֵינֵיהֶם כֶּרֶם עֲנָבִים מֻפְלָא. אֶשְׁכּוֹלוֹת עֲנָבִים סְגֻלִּים, גְּדוֹלִים וּמְתוּקִים כַּדְּבַשׁ, הָיוּ תְּלוּיִים עַל הַגְּפָנִים. עָצַר רַבִּי יוֹחָנָן, הִצְבִּיעַ בְּאֶצְבָּעוֹ, וְאָמַר: "רַבִּי חִיָּא, אַתָּה רוֹאֶה אֶת כֶּרֶם הָעֲנָבִים הַזֶּה? אֲשֶׁר עֲנָבָיו כֹּה מְתוּקִים? הוּא הָיָה שֶׁלִּי. וּמָכַרְתִּי גַּם אוֹתוֹ, כְּדֵי לִקְנוֹת סְפָרִים וְלָשֶׁבֶת לִלְמֹד אֶת תּוֹרַת הַשֵּׁם יִתְבָּרַךְ" [i].

הַדְּמָעוֹת שֶׁל רַבִּי חִיָּא

הַהֲלִיכָה נִמְשְׁכָה שָׁעָה אֲרֻכָּה תַּחַת הַשֶּׁמֶשׁ. צִלָּם שֶׁל עֲצֵי הַזַּיִת הַגְּדוֹלִים הִזְמִין אוֹתָם לָנוּחַ. עֲצֵי הַזַּיִת נָטְפוּ שֶׁמֶן טָהוֹר, שֶׁמֶן שֶׁלְּפָנִים הָיוּ מַדְלִיקִים מִמֶּנּוּ אֶת מְנוֹרַת הַמִּקְדָּשׁ.

רַבִּי יוֹחָנָן נָעַץ מַבָּט בְּעֲצֵי הַזַּיִת וְאָמַר עִם חִיּוּךְ: "הָרְאֵה, תַּלְמִידִי, גַּם מַטַּע הַזֵּיתִים הָעֲנָק הַזֶּה – שֶׁלִּי הָיָה. וְגַם אוֹתוֹ מָכַרְתִּי. מָכַרְתִּי אֶת הַכֹּל כְּדֵי לִזְכּוֹת בְּתּוֹרָה" [i].

כְּשֶׁשָּׁמַע זֹאת רַבִּי חִיָּא בַּר אַבָּא, לֹא יָכוֹל הָיָה לַעֲצֹר אֶת רִגְשׁוֹתָיו. הַלֵּב שֶׁלּוֹ הִתְכַּוֵּץ, וּדְמָעוֹת עֲצוּמוֹת הֵחֵלּוּ לְהִתְגַּלְגֵּל עַל לְחָיָיו. מִי שֶׁאָהַב כָּל כָּךְ אֶת רַבּוֹ, לֹא יָכֹל הָיָה לִסְבֹּל אֶת הַמַּחֲשָׁבָה הַזּוֹ!

"רַבִּי יוֹחָנָן!" קָרָא רַבִּי חִיָּא מִבַּעַד לַדְּמָעוֹת, קוֹלוֹ חָנוּק, "לָמָּה אַתָּה גּוֹרֵם לִי לִבְכּוֹת? אֵיךְ לֹא שָׁמַרְתָּ לְעַצְמְךָ כְּלוּם? הֲרֵי תִּזְדַּקֵּן יוֹם אֶחָד... אֵיךְ תִּקְנֶה אֹכֶל? מָה תֹּאכַל כְּשֶׁכְּבָר לֹא יִהְיֶה לְךָ כֹּחַ? לֹא הִשְׁאַרְתָּ לְעַצְמְךָ שׁוּם דָּבָר, שׁוּם שָׂדֶה, שׁוּם מָקוֹם שֶׁיָּבִיא לְךָ כֶּסֶף לִימֵי זִקְנָתְךָ!" [i].

יְלָדִים, אָנָּא דַּמְיְנוּ אֶת רַבִּי יוֹחָנָן עוֹמֵד מוּל תַּלְמִידוֹ הַבּוֹכֶה. הַאִם הוּא הָיָה עָצוּב? הַאִם הִתְחָרֵט?

מַמָּשׁ לֹא! פָּנָיו הַטְּהוֹרוֹת שֶׁל רַבִּי יוֹחָנָן זָרְחוּ בְּאוֹר פְּנִימִי, חִיּוּךְ רָחָב וְשָׁלֵו הֵאִיר אֶת עֵינָיו. הוּא הֵנִיחַ אֶת יָדוֹ הַחַמָּה עַל כְּתֵפוֹ שֶׁל רַבִּי חִיָּא וְאָמַר לוֹ מִשְׁפָּט שֶׁיַּהֲדְהֵד לָנֶצַח.

"קַלָּה הִיא בְּעֵינֶיךָ מַה שֶּׁעָשִׂיתִי?"

וְאָז, הִצְבִּיעַ לַשָּׁמַיִם וְלָאֲדָמָה: "שֶׁמָּכַרְתִּי דָּבָר שֶׁנִּבְרָא בְּשִׁשָּׁה יָמִים – אֶת עֲצֵי הַזַּיִת, אֶת הָעֲנָבִים וְאֶת הָאֲדָמָה, וְקָנִיתִי דָּבָר שֶׁנִּתַּן לְמֹשֶׁה רַבֵּנוּ רַק אַחֲרֵי אַרְבָּעִים יוֹם וְאַרְבָּעִים לַיְלָה מֵהַקָּדוֹשׁ בָּרוּךְ הוּא – קָנִיתִי אֶת הַתּוֹרָה! [i] מַהוּ שָׂדֶה מִבָּשָׂר וָעֵשֶׂב לְעֻמַּת תּוֹרַת הַנֶּצַח?!"

רַבִּי יוֹחָנָן יָדַע שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא, שֶׁנּוֹתֵן אֶת הַתּוֹרָה, לֹא יַעֲזֹב אֶת מִי שֶׁאוֹהֵב אוֹתָהּ עַד כְּדֵי כָּךְ!

הַמִּפְגָּשׁ עַל שְׂפַת הַיַּרְדֵּן

הָאַהֲבָה הָעֲצוּמָה הַזּוֹ שֶׁל רַבִּי יוֹחָנָן לַתּוֹרָה, יְלָדִים, בָּרְאָה אֶצְלוֹ כֹּחַ קָסוּם שֶׁאִי אֶפְשָׁר לְתָאֵר בְּמִלִּים. הִיא נָתְנָה לוֹ כֹּחַ לְהַצִּיל אֲנָשִׁים אֲחֵרִים מִטֻּמְאָה לִקְדֻשָּׁה.

בְּיָמִים הָהֵם הָיָה בְּאֶרֶץ יִשְׂרָאֵל שׁוֹדֵד נוֹרָא וְחָזָק מְאֹד. אִישׁ שֶׁגּוּפוֹ הָיָה בָּנוּי כְּסֶלַע, וּבְיָדָיו שְׁרִירִים כְּשֶׁל עֲנָק. קָרְאוּ לוֹ שִׁמְעוֹן בֶּן לָקִישׁ. הוּא הָיָה רֹאשׁ הַשּׁוֹדְדִים! אֲנָשִׁים הָיוּ רוֹעֲדִים מִפַּחַד כַּאֲשֶׁר שָׁמְעוּ אֶת צַעֲדָיו [i].

יוֹם אֶחָד, יָשַׁב רַבִּי יוֹחָנָן לִטְבֹּל עַל שְׂפַת נְהַר הַיַּרְדֵּן. מֵי הַנָּהָר הַקְּרִירִים וְהַזַּכִּים שָׁטְפוּ אֶת פָּנָיו הַזּוֹרְחוֹת כְּמַלְאָךְ. וְהִנֵּה... סְפְּלָאשׁ! קְפִיצָה עֲצוּמָה מִן הַגָּדָה הַשְּׁנִיָּה! רֹאשׁ הַשּׁוֹדְדִים, שִׁמְעוֹן בֶּן לָקִישׁ, זִנֵּק מֵעַל כָּל מֵי הַיַּרְדֵּן בִּקְפִיצַת גְּבוּרָה אַחַת גְּדוֹלָה, וְנָחַת מַמָּשׁ מוּל רַבִּי יוֹחָנָן.

כָּל אָדָם רָגִיל הָיָה קוֹפֵא מִפַּחַד. הֲרֵי זֶה שׁוֹדֵד אֲכְזָרִי! אֲבָל רַבִּי יוֹחָנָן מָלֵא בְּכֹחַ שֶׁל קְדֻשָּׁה וֶאֱמוּנָה. הוּא הִבִּיט יָשָׁר לְתוֹךְ עֵינָיו הַזּוֹעֲפוֹת שֶׁל הַשּׁוֹדֵד, הִתְפַּעֵל מֵהַכּוֹחַ הַגּוּפָנִי הָעָצוּם שֶׁנָּתַן לוֹ הַשֵּׁם יִתְבָּרַךְ, וְאָמַר בְּקוֹל שֶׁקֶט וּבָטוּחַ:

"חֵילְךָ – לְאוֹרַיְתָא!" [i] (הַכֹּחַ הַגָּדוֹל שֶׁלְּךָ – נוֹעַד לִלְמֹד אֶת תּוֹרַת ה'!).

הַשּׁוֹדֵד הֶחָסוֹן הָיָה הָמוּם. מִימָיו לֹא הֵעֵז אִישׁ לְדַבֵּר אֵלָיו כָּךְ. אַךְ מֵהָעֵינַיִם שֶׁל רַבִּי יוֹחָנָן בָּקַע אוֹר שֶׁל אַהֲבַת תּוֹרָה כָּל כָּךְ טְהוֹרָה וַאֲמִתִּית, אוֹר שֶׁהֵמֵס בְּרֶגַע אֶחָד אֶת כָּל הַחֹשֶׁךְ שֶׁהָיָה בְּלִבּוֹ שֶׁל רֹאשׁ הַשּׁוֹדְדִים.

"אִם תַּסְכִּים לַחְזֹר בִּתְשׁוּבָה וְלָבוֹא אִתִּי לְבֵית הַמִּדְרָשׁ כְּדֵי לְהַקְדִּישׁ אֶת כּוֹחֲךָ לַתּוֹרָה," הִבְטִיחַ לוֹ רַבִּי יוֹחָנָן, "אֶתֵּן לְךָ אֶת אֲחוֹתִי לְאִשָּׁה, שֶׁהִיא יָפָה וְצַדֶּקֶת" [i].

בְּאוֹתוֹ רֶגַע, הַהַשְׁגָּחָה הַפְּרָטִית מִלְּמַעְלָה הֵאִירָה אֶת הַלֵּב שֶׁל רֵישׁ לָקִישׁ. הוּא לָקַח נְשִׁימָה עֲמֻקָּה, הִנִּיחַ אֶת עֲבָרוֹ מֵאָחוֹר, וְקִבֵּל עַל עַצְמוֹ עֹל תּוֹרָה. קַבָּלָתוֹ הָיְתָה כָּל כָּךְ חֲזָקָה, שֶׁלְּפֶתַע, כְּשֶׁנִּסָּה שׁוּב לִקְפֹּץ הַחוּצָה, הַגּוּף הַשְּׁרִירִי שֶׁלּוֹ נֶחֱלַשׁ – כִּי כָּל מַחְשְׁבוֹתָיו וְכֹחוֹתָיו עָבְרוּ לִהְיוֹת שֶׁל הַנְּשָׁמָה הַקְּדוֹשָׁה וְהַתּוֹרָה! [i]

כָּךְ קָרָה, יְלָדִים, שֶׁבִּזְכוּת אַהֲבַת הַתּוֹרָה חַסְרַת הַתְּנָאִים שֶׁל רַבִּי יוֹחָנָן, שֶׁמָּכַר אֶת כָּל רְכוּשׁוֹ בִּשְׁבִיל ה', הוּא זָכָה לְחַלֵּץ אֶת שִׁמְעוֹן הַשּׁוֹדֵד מִן הַטֻּמְאָה. וְשִׁמְעוֹן זֶה לֹא חָזַר שׁוּב לַיַּעַר לִשְׁדֹּד אֲנָשִׁים, אֶלָּא יָשַׁב יָמִים וְלֵילוֹת, שָׁנִים עַל גַּבֵּי שָׁנִים, עַד שֶׁהָפַךְ לְאֶחָד מִגְּדוֹלֵי תַּלְמִידֵי הַחֲכָמִים הֲכִי צַדִּיקִים וְעַנְקִיִּים שֶׁל דּוֹרוֹ – הָאָמוֹרָא הַקָּדוֹשׁ רֵישׁ לָקִישׁ [i].

וְעַל כָּךְ אָמְרוּ חֲכָמֵינוּ: כְּשֶׁנִּפְטַר רַבִּי יוֹחָנָן, אָמְרוּ עָלָיו אֶת הַפָּסוּק בְּשִׁיר הַשִּׁירִים: "אִם יִתֵּן אִישׁ אֶת כָּל הוֹן בֵּיתוֹ בָּאַהֲבָה" – כְּמוֹ רַבִּי יוֹחָנָן שֶׁנָּתַן אֶת שְׂדוֹתָיו בְּאַהֲבַת הַתּוֹרָה – לְעוֹלָם אִי אֶפְשָׁר לְבַזּוֹת אוֹתוֹ אוֹ לָקַחַת מִמֶּנּוּ אֶת נִצְחוֹן קְדֻשָּׁתוֹ! [i].

הוּא (ME): רַבִּי יוֹחָנָן הָיָה מְאֹד עָשִׁיר, הָיוּ לוֹ שְׂדוֹת וּכְרָמִים, אֲבָל הוּא הֵבִין שֶׁכֶּסֶף זֶה לֹא הַדָּבָר הֲכִי חָשׁוּב בָּעוֹלָם. הוּא הֵבִין שֶׁהַדָּבָר הֲכִי חָשׁוּב, הָעֹשֶׁר הָאֲמִתִּי, זֶה לִלְמֹד תּוֹרָה מִתּוֹךְ אַהֲבָה שְׁלֵמָה.

אֲנַחְנוּ (WE): לִפְעָמִים גַּם אֲנַחְנוּ חוֹשְׁבִים שֶׁצַּעֲצוּעַ חָדָשׁ, כַּדּוּר מֻשְׁלָם, אוֹ מִשְׂחַק מַחְשֵׁב נוֹצֵץ הֵם הַדָּבָר הֲכִי יָקָר. וְאִם חָסֵר לָנוּ אוֹתָם, מַרְגִּישׁ לָנוּ קְצָת עָצוּב, וַאֲנַחְנוּ שׁוֹכְחִים לִפְעָמִים שֶׁהַלִּמּוּד וְהַקְּדֻשָּׁה שָׁוִים יוֹתֵר מִכָּל זָהָב וָכֶסֶף.

אַתֶּם (YOU): מָחָר בַּבֹּקֶר, כְּשֶׁתִּצְטָרְכוּ לִבְחֹר אִם לְשַׂחֵק בַּהַפְסָקָה עוֹד רֶגַע אֶחָד שֶׁל מִשְׂחָק נִפְלָא, אוֹ לָרוּץ לְשִׁעוּר תּוֹרָה שֶׁמַּתְחִיל עַכְשָׁו, תִּזָּכְרוּ בְּרַבִּי יוֹחָנָן. תּוֹותְּרוּ קְצָת עַל הַדָּבָר הַגַּשְׁמִי, וְתָרוּצוּ לֶאֱסֹף אֶת הַיַּהֲלוֹמִים שֶׁל הַתּוֹרָה הַקְּדוֹשָׁה.

א. שְׁאֵלַת פְּתִיחָה:

"יְלָדִים, נַנִּיחַ שֶׁעַכְשָׁו אֲנִי בָּא וְנוֹתֵן לְכָל אֶחָד מִכֶּם מִזְוָדָה עִם מִילְיוֹן דּוֹלָר! מָה תַּעֲשׂוּ אִתָּהּ? [לָתֵת לַיְּלָדִים לַעֲנוֹת].

וּמָה הֱיִיתֶם אוֹמְרִים עַל אָדָם שֶׁהָיָה לוֹ מִילְיוֹן דּוֹלָר, עַצְמָאִי, עָשִׁיר מֻפְלָג... וְהוּא פָּשׁוּט וִתֵּר עַל זֶה מֵרָצוֹן? הַיּוֹם נַכִּיר אָדָם כָּזֶה. מִגְּדוֹלֵי יִשְׂרָאֵל, שֶׁמָּכַר אֶת הַכֹּל בִּשְׁבִיל לִקְנוֹת מַשֶּׁהוּ אֶחָד יָקָר מִכֹּל."

ב. שְׁאֵלַת סִיּוּם:

"אֵיזֶה דָּבָר יָקָר וּמְיֻחָד בַּתּוֹרָה אָנוּ רוֹאִים כָּאן?

[תֵּן לַיְּלָדִים לַעֲנוֹת]

נָכוֹן, שֶׁאַהֲבַת תּוֹרָה כָּל כָּךְ אֲמִתִּית לֹא רַק נוֹתֶנֶת חָכְמָה, הִיא גַּם יְכוֹלָה לְהַצִּיל אֲפִלּוּ אָדָם מִתְנַגֵּד אוֹ רָחוֹק – מַמָּשׁ כְּמוֹ רֹאשׁ שׁוֹדְדִים עֲנָק – וְלַהֲפֹךְ אוֹתוֹ לְתַלְמִיד חָכָם צַדִּיק!"

ג. הַמְלָצוֹת לְמַצֶּגֶת (תְּמוּנוֹת מֻקְפָּדוֹת ותרחישים בעלי הדרת קודש וצניעות):

ד. פְּעִילוּת מְסַכֶּמֶת:

מִשְׂחַק מִשְׁקָלוֹת: שִׂים בְּצַד אֶחָד שֶׁל "מֹאזְנַיִם" (אֲפִלּוּ דִּמְיוֹנִיִּים עִם הַיָּדַיִם) חֲפָצִים שׁוֹנִים (כֶּסֶף מְדֻמֶּה, צַעֲצוּעַ, צִיּוּר שֶׁל שָׂדֶה), וּבַצַּד הַשֵּׁנִי - חוּמָשׁ שֶׁל תּוֹרָה. תֵּן לַיְּלָדִים לִצְעֹק אֵיזֶה הַצַּד הַכָּבֵד יוֹתֵר. מַסְבִּירִים לָהֶם שֶׁזֶּה בְּדִיּוּק מָה שֶׁרַבִּי יוֹחָנָן עָשָׂה – הוּא שָׁקַל בְּדַעְתּוֹ, וְהֵבִין שֶׁהַתּוֹרָה כְּבֵדָה וַחֲשׁוּבָה יוֹתֵר מִכָּל הַשָּׂדוֹת שֶׁבָּעוֹלָם.

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “להלן מערך חינוכי, רגשי ומרתק, המשלב שניים מהסיפורים הגדולים ביותר על רבי יוחנן ומסירותו לתורה , יחד עם המפגש המכונן עם ריש לקיש .” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “הֵם הִמְשִׁיכוּ לָלֶכֶת, וְהִנֵּה נִגְלָה לְעֵינֵיהֶם כֶּרֶם עֲנָבִים מֻפְלָא.”?
ג. כיצד הגיב ר' יוחנן וריש לקיש למצב המתואר: “יְלָדִים, אָנָּא דַּמְיְנוּ אֶת רַבִּי יוֹחָנָן עוֹמֵד מוּל תַּלְמִידוֹ הַבּוֹכֶה.”?
ד. מה השתנה בין “יוֹם אֶחָד, יָשַׁב רַבִּי יוֹחָנָן לִטְבֹּל עַל שְׂפַת נְהַר הַיַּרְדֵּן.” לבין “קַבָּלָתוֹ הָיְתָה כָּל כָּךְ חֲזָקָה, שֶׁלְּפֶתַע, כְּשֶׁנִּסָּה שׁוּב לִקְפֹּץ הַחוּצָה, הַגּוּף הַשְּׁרִירִי שֶׁלּוֹ…”?
ה. כיצד הגיע הסיפור אל האירוע: “וּמָה הֱיִיתֶם אוֹמְרִים עַל אָדָם שֶׁהָיָה לוֹ מִילְיוֹן דּוֹלָר, עַצְמָאִי, עָשִׁיר מֻפְלָג...”?

3. מה היה ראשון?

כתבו ליד כל אירוע מספר מ־1 עד 5 לפי סדר התרחשותו בסיפור.

□ אִישׁ שֶׁגּוּפוֹ הָיָה בָּנוּי כְּסֶלַע, וּבְיָדָיו שְׁרִירִים כְּשֶׁל עֲנָק.
□ להלן מערך חינוכי, רגשי ומרתק, המשלב שניים מהסיפורים הגדולים ביותר על רבי יוחנן ומסירותו לתורה , יחד עם המפגש המכונן עם ריש לקיש .
□ מַסְבִּירִים לָהֶם שֶׁזֶּה בְּדִיּוּק מָה שֶׁרַבִּי יוֹחָנָן עָשָׂה – הוּא שָׁקַל בְּדַעְתּוֹ, וְהֵבִין שֶׁהַתּוֹרָה כְּבֵדָה וַחֲשׁוּבָה יוֹתֵר מִכָּל…
□ עֲצֵי הַזַּיִת נָטְפוּ שֶׁמֶן טָהוֹר, שֶׁמֶן שֶׁלְּפָנִים הָיוּ מַדְלִיקִים מִמֶּנּוּ אֶת מְנוֹרַת הַמִּקְדָּשׁ.
□ כָּךְ קָרָה, יְלָדִים, שֶׁבִּזְכוּת אַהֲבַת הַתּוֹרָה חַסְרַת הַתְּנָאִים שֶׁל רַבִּי יוֹחָנָן, שֶׁמָּכַר אֶת כָּל רְכוּשׁוֹ בִּשְׁבִיל ה', הוּא זָכָה…

לפני ואחרי

מה קרה לפני: “אִישׁ שֶׁגּוּפוֹ הָיָה בָּנוּי כְּסֶלַע, וּבְיָדָיו שְׁרִירִים כְּשֶׁל עֲנָק.”?
מה קרה אחרי: “כָּךְ קָרָה, יְלָדִים, שֶׁבִּזְכוּת אַהֲבַת הַתּוֹרָה חַסְרַת הַתְּנָאִים שֶׁל רַבִּי יוֹחָנָן, שֶׁמָּכַר אֶת כָּל רְכוּשׁוֹ בִּשְׁבִיל ה', הוּא זָכָה…”?

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “ר' יוחנן — אהבתו לתורה” על ר' יוחנן וריש לקיש? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “הִיא נָתְנָה לוֹ כֹּחַ לְהַצִּיל אֲנָשִׁים אֲחֵרִים מִטֻּמְאָה לִקְדֻשָּׁה.”.
ג. אילו ר' יוחנן וריש לקיש לא היה פועל כפי שמתואר ב־“קַבָּלָתוֹ הָיְתָה כָּל כָּךְ חֲזָקָה, שֶׁלְּפֶתַע, כְּשֶׁנִּסָּה שׁוּב לִקְפֹּץ הַחוּצָה, הַגּוּף הַשְּׁרִירִי שֶׁלּוֹ נֶחֱלַשׁ…” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “וּמָה הֱיִיתֶם אוֹמְרִים עַל אָדָם שֶׁהָיָה לוֹ מִילְיוֹן דּוֹלָר, עַצְמָאִי, עָשִׁיר מֻפְלָג...” מחזק את המסר של הסיפור?
