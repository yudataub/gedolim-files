דף עבודה בעקבות רבי יהודה הנשיא (רבי)
שם: כיתה:

המעשה — הסיפור המלא

פְּתִיחָה: הָאַרְמוֹן שֶׁל הַתּוֹרָה

יְלָדִים, נַסּוּ לְדַמְיֵן בְּרֹאשְׁכֶם אֶת בֵּית הַמִּדְרָשׁ הַגָּדוֹל בְּיוֹתֵר בָּעוֹלָם. שֶׁקֶט בַּחוּץ, אֲבָל בִּפְנִים – אוֹר גָּדוֹל! מֵאוֹת מְנוֹרוֹת שֶׁל זָהָב טָהוֹר דּוֹלְקוֹת עַל הַקִּירוֹת. אִם תִּקְּחוּ נְשִׁימָה עֲמֻקָּה, תּוּכְלוּ לְהָרִיחַ אֶת הָרֵיחַ הַמָּתוֹק שֶׁל שֶׁמֶן הַזַּיִת הַנִּשְׂרָף לְאִטּוֹ, יַחַד עִם רֵיחַ הַקְּלָפִים הַחֲדָשִׁים, מִגְּוִילֵי סִפְרֵי הַתּוֹרָה שֶׁמֻּנָּחִים עַל הַשֻּׁלְחָנוֹת.

וְהַקּוֹל? אוֹהוֹ, אֵיזֶה קוֹל! קוֹל עָצוּם וּמָתוֹק כִּדְבַשׁ שֶׁל אַלְפֵי תַּלְמִידִים שֶׁלּוֹמְדִים יַחַד.

בָּרֹאשׁ, עַל כִּסֵּא כָּבוֹד מְיֻחָד, יוֹשֵׁב רַבִּי יְהוּדָה הַנָּשִׂיא . כֻּלָּם קָרְאוּ לוֹ פָּשׁוּט "רַבֵּנוּ הַקָּדוֹשׁ" אוֹ שֶׁקָּרְאוּ לוֹ "רַבִּי". הוּא הָיָה אָדָם עָשִׁיר מְאֹד מְאֹד, חָבֵר שֶׁל קֵיסָר רוֹמָא, אֲבָל הָעֹשֶׁר לֹא עִנְיֵן אוֹתוֹ. הַיָּדַיִם שֶׁלּוֹ הָיוּ חֲזָקוֹת וְעֵינָיו בּוֹרְקוֹת כְּכוֹכָבִים מֵרֹב קְדֻשָּׁה וְאַהֲבַת ה'.

הַהִתְלַהֲבוּת שֶׁל אֶלְעָזָר

בֵּין כָּל תַּלְמִידֵי הַחֲכָמִים, יָשַׁב נַעַר צָעִיר. נִקְרָא לוֹ אֶלְעָזָר.

אֶלְעָזָר אָהַב אֶת הַתּוֹרָה בְּכָל לִבּוֹ. בְּכָל בֹּקֶר, מַמָּשׁ כְּשֶׁהַשֶּׁמֶשׁ עֲדַיִן יְשֵׁנָה וַחֲשֵׁכָה קָרָה מְכַסָּה אֶת הָרְחוֹבוֹת, אֶלְעָזָר הָיָה קוֹפֵץ מִמִּטָּתוֹ. הוּא הָיָה לָבוּשׁ בִּמְהִירוּת, נוֹעֵל אֶת סַנְדָּלֵי הָעֵץ הַכְּבֵדִים שֶׁלּוֹ, וְרָץ לְבֵית הַמִּדְרָשׁ כְּדֵי לֹא לְהַפְסִיד אַף דַּקָּה שֶׁל תּוֹרָה!

טְרַאק! – כָּךְ הָיְתָה נִטְרֶקֶת דֶּלֶת הָעֵץ הַגְּדוֹלָה שֶׁל בֵּיתוֹ.

טְרַאח... טְרַאח... טְרַאח... – כָּךְ נִשְׁמְעוּ סַנְדָּלֵי הָעֵץ שֶׁל אֶלְעָזָר כְּשֶׁהוּא דּוֹהֵר עַל אַבְנֵי הָרְחוֹב, מְדַלֵּג מֵעַל שְׁלוּלִיּוֹת.

וּבַלַּיְלָה? בַּלַּיְלָה הָיָה אֶלְעָזָר חוֹזֵר לְבֵיתוֹ מְאֻחָר מְאֹד. שׁוּב טְרִיקַת דֶּלֶת חֲזָקָה בְּחַדְרֵי הַמַּדְרֵגוֹת, שׁוּב צְעָדִים חֲזָקִים. אֶלְעָזָר הָיָה גֵּאֶה בְּעַצְמוֹ! הוּא הָיָה מְלַטֵּף אֶת סִפְרוֹ בְּשִׂמְחָה וְחוֹשֵׁב בְּלִבּוֹ: "אֵיזֶה צַדִּיק אֲנִי! כָּל הָעִיר כְּבָר יְשֵׁנָה, וְרַק אֲנִי קָם לִלְמֹד תּוֹרָה שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא!".

הַדְּמָעוֹת שֶׁל הַסַּנְדְּלָר

אֲבָל... יְלָדִים, מִי לֹא הָיָה גֵּאֶה בְּאֶלְעָזָר?

הַשְּׁכֵנִים שֶׁלּוֹ. בַּבַּיִת שֶׁלְּיָדוֹ גָּר סַנְדְּלָר זָקֵן. הַסַּנְדְּלָר הַזֶּה הָיָה עוֹבֵד כָּל הַיּוֹם וְכָל הַלַּיְלָה עִם פַּטִּישׁ קָטָן כְּדֵי לְהָבִיא לֶחֶם לַיְּלָדִים שֶׁלּוֹ. כְּשֶׁהוּא כְּבָר נִרְדַּם וְרָצָה לָנוּחַ, פִּתְאוֹם... טְרַאק! טְרַאח טְרַאח טְרַאח! הַסַּנְדָּלִים שֶׁל אֶלְעָזָר שֶׁרָץ לְבֵית הַמִּדְרָשׁ, הִרְעִידוּ אֶת הַקִּירוֹת.

הַתִּינוֹק שֶׁל הַשָּׁכֵן הָיָה מִתְעוֹרֵר וּמַתְחִיל לִבְכּוֹת. הַסַּנְדְּלָר הַזָּקֵן הָיָה תּוֹפֵס אֶת רֹאשׁוֹ וְנֶאֱנָח מֵרֹב עֲיֵפוּת. הוּא לֹא הָיָה לוֹמֵד תּוֹרָה כָּל הַיּוֹם כִּי עָבַד, אֲבָל נִשְׁמָתוֹ הָיְתָה כּוֹאֶבֶת: "לָמָּה הַבָּחוּר הַזֶּה לֹא מִתְחַשֵּׁב בָּנוּ?", לָחַשׁ בְּעֶצֶב. "כְּלוּם כָּךְ מְצַוָּה הַתּוֹרָה, לִרְמֹס אֶת שְׁנָתָם שֶׁל הָאֲחֵרִים?".

הַלַּיְלָה שֶׁל הַגִּלּוּי

רַבִּי יְהוּדָה הַנָּשִׂיא, "רַבֵּנוּ הַקָּדוֹשׁ", יָדַע מָה קוֹרֶה בַּלְּבָבוֹת שֶׁל עַם יִשְׂרָאֵל. הוּא הָיָה צַדִּיק שֶׁאוֹהֵב אֶת כָּל יְהוּדִי וִיהוּדִי. לַיְלָה אֶחָד, כְּשֶׁסִּיְּמוּ אֶת הַלִּמּוּד וְכֻלָּם קָמוּ לָלֶכֶת בַּחֲשֵׁכָה הַקָּרָה, קָרָא רַבִּי לְאֶלְעָזָר.

הָרַב קָרָא לַבָּחוּר אֶל חַדְרוֹ, אֶל הַמָּקוֹם שֶׁבּוֹ הוּא כּוֹתֵב אֶת הַמִּשְׁנָה.

"אֶלְעָזָר בְּנִי," פָתַח רַבִּי בְּקוֹל רַךְ וְנָעִים מְאֹד, כְּמוֹ שֶׁל אַבָּא אֹהֵב, "רוֹאֶה אֲנִי כַּמָּה אַתָּה אוֹהֵב אֶת הַתּוֹרָה וְאֶת הַמִּצְווֹת הַקְּדוֹשׁוֹת. אַשְׁרֶיךָ!".

אֶלְעָזָר הִסְמִיק, וְלִבּוֹ דָּפַק בְּגַאֲוָה.

"אֲבָל אֱמֹר לִי בְּנִי," הִמְשִׁיךְ רַבִּי, וְהִבִּיט עָמֹק לְתוֹךְ עֵינָיו שֶׁל הַנַּעַר, "כְּשֶׁאַתָּה לוֹמֵד, וְיוֹדֵעַ שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא שָׂמֵחַ בְּךָ – הַאִם גַּם הַסַּנְדְּלָר שֶׁגָּר לְיָדְךָ, וְהַתִּינוֹק שֶׁיָּשֵׁן שָׁם, שְׂמֵחִים בְּךָ?".

אֶלְעָזָר שָׁתַק. הוּא פִּתְאוֹם הֵבִין שֶׁהָרַב יוֹדֵעַ עַל הָרַעַשׁ שֶׁהוּא עוֹשֶׂה מִדֵּי לַיְלָה וָבֹקֶר. "אֲ... אֲנִי הָלַכְתִּי לִלְמֹד תּוֹרָה! זֶה מִצְוָה, כְּבוֹד הָרַב!" גִּמְגֵּם בַּהֲגַנָּה.

רַבִּי קָם מִכִּסְאוֹ הַגָּדוֹל, הִנִּיחַ אֶת יָדוֹ הַקְּדוֹשָׁה עַל כְּתֵפוֹ שֶׁל אֶלְעָזָר וְלִמֵּד אוֹתוֹ מִשְׁנָה שֶׁתִּשָּׁאֵר נֵצַח לְכָל עַם יִשְׂרָאֵל:

"אֶלְעָזָר. אֵיזוֹהִי דֶּרֶךְ יְשָׁרָה שֶׁיָּבוֹר לוֹ הָאָדָם? שֶׁיִּבְחַר לְעַצְמוֹ הָאָדָם? לֹא מַסְפִּיק שֶׁמַּה שֶּׁאַתָּה עוֹשֶׂה הוּא מִצְוָה לִפְנֵי שָׁמַיִם. כֹּל שֶׁהִיא תִּפְאֶרֶת לְעוֹשֶׂיהָ – וְתִפְאֶרֶת לוֹ מִן הָאָדָם! דֶּרֶךְ אֲמִתִּית הִיא דֶּרֶךְ שֶׁגַּם עוֹשָׂה נַחַת לַה', וְגַם נוֹתֶנֶת כָּבוֹד, שֶׁקֶט וְטוֹב לִבְנֵי אָדָם שֶׁלְּיָדְךָ. אִם מִצְוָה עוֹשָׂה בֶּכִי לְתִינוֹק שֶׁל הַשָּׁכֵן – זוֹ לֹא תִּפְאֶרֶת...".

אֶלְעָזָר הִרְגִּישׁ מְעַט מְבֻלְבָּל. "אֲבָל... כְּשֶׁאֲנִי בַּמַּדְרֵגוֹת הַחֲשׁוּכוֹת בַּלַּיְלָה... אַף אֶחָד לֹא רוֹאֶה אֶת הַכַּוָּנוֹת שֶׁלִּי. רַק אֲנִי מְמַהֵר...".

חִיֵּךְ הַנָּשִׂיא וְאָמַר אֶת סוֹפָהּ שֶׁל הַמִּשְׁנָה הַקְּדוֹשָׁה:

"אֶלְעָזָר בְּנִי, הִסְתַּכֵּל בִּשְׁלֹשָׁה דְּבָרִים וְאִי אַתָּה בָּא לִידֵי עֲבֵרָה: דַּע מַה לְּמַעְלָה מִמְּךָ: עַיִן רוֹאָה, וְאֹזֶן שׁוֹמַעַת, וְכָל מַעֲשֶׂיךָ בַּסֵּפֶר נִכְתָּבִין. גַּם כְּשֶׁאַתָּה לְבַד בַּחֲדַר הַמַּדְרֵגוֹת, עֵינוֹ שֶׁל ה' רוֹאָה הַכֹּל. וְהָאֹזֶן הַשּׁוֹמַעַת – זוֹ הָאֹזֶן שֶׁל ה' שֶׁשּׁוֹמַעַת אֶת קוֹל הַסַּנְדָּלִים שֶׁלְּךָ מַפְרִיעִים לַסַּנְדְּלָר הַזָּקֵן. תּוֹרָה צְרִיכָה לִהְיוֹת יָפָה מִבִּפְנִים וּמִבַּחוּץ!".

נִצָּחוֹן שֶׁל עֲנָוָה

אֶלְעָזָר יָצָא מִבֵּית הַמִּדְרָשׁ. הָרוּחַ בַּחוּץ הָיְתָה עוֹד יוֹתֵר קָרָה מִקֹּדֶם, אֲבָל בְּתוֹךְ הַלֵּב שֶׁל אֶלְעָזָר דָּלְקָה לֶהָבָה שֶׁל חֹם וֶאֱמֶת.

הוּא עָמַד מוּל דֶּלֶת הַבַּיִת שֶׁל הַסַּנְדְּלָר הַזָּקֵן בְּאֶמְצַע הַלַּיְלָה. הוּא לֹא פָּתַח אֶת דֶּלֶת בֵּיתוֹ בִּטְרִיקָה. הוּא הֶחֱזִיק בַּיָּדִית הָעֵץ בַּעֲדִינוּת... הוּא הוֹרִיד אֶת סַנְדָּלֵי הָעֵץ הַכְּבֵדִים מֵהָרַגְלַיִם, וְהָלַךְ רַק עַל קְצוֹת הָאֶצְבָּעוֹת! בְּשֶׁקֶט, בְּשֶׁקֶט עָצוּם. אַף צְעָדָה לֹא נִשְׁמְעָה.

בַּבַּיִת הַסָּמוּךְ, נָשַׁם הַתִּינוֹק נְשִׁימוֹת שְׁקֵטוֹת שֶׁל שֵׁנָה טוֹבָה, וְהַסַּנְדְּלָר הַיָּגֵעַ חָלַם חֲלוֹמוֹת טוֹבִים.

וּמִלְמַעְלָה, "עַיִן רוֹאָה וְאֹזֶן שׁוֹמַעַת", שָׁמַע הַקָּדוֹשׁ בָּרוּךְ הוּא אֶת הַשֶּׁקֶט שֶׁעָשָׂה אֶלְעָזָר – וְחִיֵּךְ. זֶה הָיָה הַלִּמּוּד תּוֹרָה הֲכִי יָפֶה שֶׁלּוֹ מֵעוֹלָם. תִּפְאֶרֶת מִן הָאָדָם!

מקורות: "אֵיזוֹהִי דֶרֶךְ יְשָׁרָה שֶׁיָּבוֹר לוֹ הָאָדָם? כֹּל שֶׁהִיא תִפְאֶרֶת לְעוֹשֶׂיהָ וְתִפְאֶרֶת לוֹ מִן הָאָדָם..." – משנה, מסכת אבות, פרק ב', משנה א'. "וְהִסְתַּכֵּל בִּשְׁלשָׁה דְבָרִים וְאִי אַתָּה בָא לִידֵי עֲבֵרָה: דַּע מַה לְמַעְלָה מִמְּךָ, עַיִן רוֹאָה וְאֹזֶן שׁוֹמַעַת וְכָל מַעֲשֶׂיךָ בַּסֵּפֶר נִכְתָּבִין" – משנה, מסכת אבות, פרק ב', משנה א'. הקפדתו של רבי להעיר על דברים המפריעים לשכנים (דוגמת הרעש בחדר המדרגות) – נשען על קובץ המקורות מתוך הנתונים שהמשתמש סיפק ("מס' דוגמאות שהילדים צריכים לכתוב אם זה הדרך הישרה... אדם שלומד תורה... ועושה רעש בחדר המדרגות"). הדגמה רוחנית של הלכה למעשה במידת ה"דרך ישרה" של רבי. עושרו, קדושתו והגותו של רבנו הקדוש – תלמוד בבלי, מסכת עבודה זרה (י:), מנחות (פד: - "רבי הקדוש"), גיטין (נט.).

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “יְלָדִים, נַסּוּ לְדַמְיֵן בְּרֹאשְׁכֶם אֶת בֵּית הַמִּדְרָשׁ הַגָּדוֹל בְּיוֹתֵר בָּעוֹלָם.” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “הַיָּדַיִם שֶׁלּוֹ הָיוּ חֲזָקוֹת וְעֵינָיו בּוֹרְקוֹת כְּכוֹכָבִים מֵרֹב קְדֻשָּׁה וְאַהֲבַת ה'.”?
ג. כיצד הגיב רבי יהודה הנשיא (רבי) למצב המתואר: “כָּל הָעִיר כְּבָר יְשֵׁנָה, וְרַק אֲנִי קָם לִלְמֹד תּוֹרָה שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא!".”?
ד. מה השתנה בין “הָרַב קָרָא לַבָּחוּר אֶל חַדְרוֹ, אֶל הַמָּקוֹם שֶׁבּוֹ הוּא כּוֹתֵב אֶת הַמִּשְׁנָה.” לבין “אִם מִצְוָה עוֹשָׂה בֶּכִי לְתִינוֹק שֶׁל הַשָּׁכֵן – זוֹ לֹא תִּפְאֶרֶת...".”?
ה. כיצד הגיע הסיפור אל האירוע: “הוּא הוֹרִיד אֶת סַנְדָּלֵי הָעֵץ הַכְּבֵדִים מֵהָרַגְלַיִם, וְהָלַךְ רַק עַל קְצוֹת הָאֶצְבָּעוֹת!”?

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “מאמרותיו: "איזוהי דרך ישרה...", "הוי זהיר במצוה קלה..."” על רבי יהודה הנשיא (רבי)? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “הַסַּנְדְּלָר הַזָּקֵן הָיָה תּוֹפֵס אֶת רֹאשׁוֹ וְנֶאֱנָח מֵרֹב עֲיֵפוּת.”.
ג. אילו רבי יהודה הנשיא (רבי) לא היה פועל כפי שמתואר ב־“אִם מִצְוָה עוֹשָׂה בֶּכִי לְתִינוֹק שֶׁל הַשָּׁכֵן – זוֹ לֹא תִּפְאֶרֶת...".” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “הוּא הוֹרִיד אֶת סַנְדָּלֵי הָעֵץ הַכְּבֵדִים מֵהָרַגְלַיִם, וְהָלַךְ רַק עַל קְצוֹת הָאֶצְבָּעוֹת!” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. יְלָדִים, נַסּוּ לְדַמְיֵן בְּרֹאשְׁכֶם אֶת בֵּית הַמִּדְרָשׁ הַגָּדוֹל…
2. בְּכָל בֹּקֶר, מַמָּשׁ כְּשֶׁהַשֶּׁמֶשׁ עֲדַיִן יְשֵׁנָה וַחֲשֵׁכָה…
3. הַסַּנְדְּלָר הַזֶּה הָיָה עוֹבֵד כָּל הַיּוֹם וְכָל הַלַּיְלָה עִם…
4. הוּא פִּתְאוֹם הֵבִין שֶׁהָרַב יוֹדֵעַ עַל הָרַעַשׁ שֶׁהוּא עוֹשֶׂה…
5. אֶלְעָזָר בְּנִי, הִסְתַּכֵּל בִּשְׁלֹשָׁה דְּבָרִים וְאִי אַתָּה בָּא…
6. זֶה הָיָה הַלִּמּוּד תּוֹרָה הֲכִי יָפֶה שֶׁלּוֹ מֵעוֹלָם.

6. כתיבה ויצירה בעקבות הסיפור

דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “גַּם כְּשֶׁאַתָּה לְבַד בַּחֲדַר הַמַּדְרֵגוֹת, עֵינוֹ שֶׁל ה' רוֹאָה הַכֹּל.”.
