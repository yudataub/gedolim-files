סוד התפלה בעלית הגג

המעשה — הסיפור המלא

הסיפור מנוקד במלואו כנדרש.

🔹 שָׁלָב 1: בֵּרוּר מַקְדִּים וְנִתּוּחַ (לַמּוֹרֶה/לַמְסַפֵּר)

🔹 שָׁלָב 2: הַסִּפּוּר הַמָּלֵא – "סוֹד הַתְּפִלָּה בַּעֲלִיַּת הַגַּג"

פְּתִיחָה: הָאַרְמוֹן הֶעָצוּב יְלָדִים, כְּשֶׁמִּישֶׁהוּ מַרְגִּישׁ לֹא טוֹב וְיֵשׁ לוֹ חֹם גָּבוֹהַּ, מָה עוֹשִׂים? לוֹקְחִים אוֹתוֹ מַהֵר לָרוֹפֵא, נוֹתְנִים לוֹ תְּרוּפָה וְקוֹנִים כַּדּוּרִים. אֲבָל מָה קוֹרֶה כְּשֶׁאַף רוֹפֵא לֹא מַצְלִיחַ לַעֲזֹר?

לִפְנֵי שָׁנִים רַבּוֹת, בְּבֵיתוֹ הַגָּדוֹל שֶׁל נְשִׂיא יִשְׂרָאֵל, הָרַב הַגָּדוֹל מִכֻּלָּם – רַבָּן גַּמְלִיאֵל – שָׂרַר שֶׁקֶט עָצוּב. הַבֵּן שֶׁל רַבָּן גַּמְלִיאֵל חָלָה בְּמַחֲלָה קָשָׁה מְאֹד. הַחֹם שֶׁלּוֹ בָּעַר כְּמוֹ אֵשׁ, וְהוּא שָׁכַב עַל הַמִּטָּה, חַלָּשׁ וְחִוֵּר. כָּל הָרוֹפְאִים הֲכִי חֲכָמִים בָּעִיר בָּאוּ לִבְדֹּק אוֹתוֹ. הֵם הֵכִינוּ שִׁקּוּיִים מֵעֲשָׂבִים מָרִים, שָׂמוּ מַגָּבוֹת קָרוֹת עַל הַמֵּצַח שֶׁלּוֹ, אֲבָל כְּלוּם לֹא עָזַר.

רַבָּן גַּמְלִיאֵל עָמַד מִחוּץ לַחֶדֶר. לִבּוֹ הָיָה שָׁבוּר. הוּא יָדַע שֶׁרוֹפְאִים הֵם רַק שְׁלִיחִים שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא, וְרַק מִי שֶׁעוֹשֶׂה אֶת רְצוֹן ה' בִּשְׁלֵמוּת, יָכוֹל לִפְתֹּחַ אֶת שַׁעֲרֵי הַשָּׁמַיִם בִּתְפִלָּתוֹ. לְפֶתַע קָרָא רַבָּן גַּמְלִיאֵל לִשְׁנֵי תַּלְמִידֵי חֲכָמִים צְעִירִים וְאָמַר לָהֶם בְּקוֹל דָּחוּף: "סְעוּ מַהֵר! עֲלוּ עַל סוּסִים וְרִכְבוּ לָעִיר שֶׁל רַבִּי חֲנִינָא בֶּן דּוֹסָא! תַּגִּידוּ לוֹ שֶׁבְּנִי שׁוֹכֵב חוֹלֶה, וּבַקְּשׁוּ מִמֶּנּוּ שֶׁיְּבַקֵּשׁ רַחֲמִים שֶׁיִּחְיֶה!".

הַדֶּרֶךְ אֶל הֶעָנִי הַקָּדוֹשׁ יְלָדִים, אַתֶּם זוֹכְרִים מִי הָיָה רַבִּי חֲנִינָא? אָדָם עָנִי שֶׁלֹּא הָיָה לוֹ כְּלוּם בַּבַּיִת חוּץ מִקַּב שֶׁל חֲרוּבִים יְבֵשִׁים! אֲבָל רַבָּן גַּמְלִיאֵל, הַנָּשִׂיא הֶעָשִׁיר, לֹא פָּנָה לְעוֹד רוֹפְאִים אוֹ עֲשִׁירִים – הוּא פָּנָה אֶל רַבִּי חֲנִינָא. מַדּוּעַ? כִּי רַבִּי חֲנִינָא הָיָה אוֹהֵב אֶת ה' אֲהָבָה מֻשְׁלֶמֶת! הוּא דִּקְדֵּק וְהִדֵּר בְּכָל מִצְוָה וּמִצְוָה, שָׁמַר עַל כְּבוֹד חֲבֵרָיו, נִזְהַר בְּרְכוּשׁ הַשְּׁכֵנִים, וְלָכֵן – כְּשֶׁרַבִּי חֲנִינָא הָיָה מְבַקֵּשׁ מַשֶּׁהוּ, הַקָּדוֹשׁ בָּרוּךְ הוּא הָיָה שׁוֹמֵעַ מִיָּד לִתְפִלָּתוֹ.

שְׁנֵי הַשְּׁלִיחִים מִהֲרוּ. הֵם רָכְבוּ שָׁעוֹת אֲרֻכּוֹת בַּדְּרָכִים הַמְאֻבָּקוֹת, כְּשֶׁהַלֵּב שֶׁלָּהֶם דּוֹפֵק מֵרֹב לַחַץ. הֵם הִגִּיעוּ לְבֵיתוֹ הַקָּטָן וְהַפָּשׁוּט שֶׁל רַבִּי חֲנִינָא, נִכְנְסוּ הַפֶּתְחָה וְאָמְרוּ בִּנְשִׁימָה קְטוּעָה: "רַבִּי חֲנִינָא! הַנָּשִׂיא שָׁלַח אוֹתָנוּ! הַבֵּן שֶׁלּוֹ כִּמְעַט גּוֹסֵס! תַּצִּיל אוֹתוֹ, אָנָּא! הִתְפַּלֵּל לַה'!".

רַבִּי חֲנִינָא שָׁמַע, וּפָנָיו הַטְּהוֹרוֹת הִרְצִינוּ. הוּא לֹא דִּבֵּר מִלָּה. הוּא פָּנָה לַמַּדְרֵגוֹת הָעֵץ הַצָּרוֹת שֶׁל בֵּיתוֹ, וְעָלָה לָ"עֲלִיָּה" – חֶדֶר עֶלְיוֹן עַל הַגַּג, מָקוֹם שָׁקֵט שֶׁמְּיֻחָד לְהִתְבּוֹדְדוּת מִשְּׁאָר אֲנָשִׁים, מָקוֹם שֶׁל תְּפִלָּה לַה'. וְשָׁם, לְמַעְלָה... נַעֲשָׂה שֶׁקֶט עָמֹק.

שְׁגוּרָה תְּפִלָּתִי בְּפִי הַשְּׁלִיחִים עָמְדוּ לְמַטָּה, כּוֹסְסִים צִפָּרְנַיִם מֵרֹב מֶתַח. דַּקָּה עָבְרָה, ועוֹד דַּקָּה. וְאָז, הֵם שָׁמְעוּ קוֹל צְעָדִים יוֹרְדִים. רַבִּי חֲנִינָא יָרַד. הַפָּנִים שֶׁלּוֹ שָׁפְעוּ אוֹר מְיֻחָד, אוֹר שֶׁל שִׂמְחָה שְׁקֵטָה. הוּא הִבִּיט בַּשְּׁלִיחִים וְאָמַר לָהֶם שְׁלוֹשׁ מִלִּים בִּלְבַד: "לְכוּ. חֲלָצַתְהוּ חַמָּה" (חַמָּה = חֹם שֶׁל הַמַּחֲלָה הָלַךְ מִמֶּנּוּ).

יְלָדִים, תָּאֲרוּ לָכֶם מָה הַשְּׁלִיחִים הִרְגִּישׁוּ! רַבִּי חֲנִינָא מִסְתַּכֵּל עֲלֵיהֶם, לֹא יָצָא מֵהַבַּיִת, הוּא נִמְצָא רָחוֹק מְאֹד מִירוּשָׁלַיִם, וְהוּא בָּטוּחַ שֶׁהַיֶּלֶד הִבְרִיא הָרֶגַע?! מֵאֵיפֹה הוּא יוֹדֵעַ? הֲרֵי בַּיָּמִים הָהֵם לֹא הָיָה טֶלֶפוֹן, לֹא וָוטְסְאַפּ, לֹא כְּלוּם! הַשְּׁלִיחִים פָּעֲרוּ אֶת הָעֵינַיִם בְּפֶלֶא: "רַבִּי! וְכִי נָבִיא אַתָּה?! שֶׁאַתָּה רוֹאֶה אֶת הֶעָתִיד הַמְּרֻחָק?".

עָנָה לָהֶם רַבִּי חֲנִינָא, וְדִבְרֵי הֶחָכְמָה שֶׁלּוֹ מְלַמְּדִים אוֹתָנוּ עַד הַיּוֹם: "לֹא נָבִיא אָנֹכִי וְלֹא בֶּן נָבִיא אָנֹכִי. אַךְ מַסֹּרֶת הִיא בְּיָדִי... מָתַי אֲנִי יוֹדֵעַ אִם הַתְּפִלָּה הִתְקַבְּלָה וְהַחוֹלֶה יִחְיֶה? 'אִם שְׁגוּרָה תְּפִלָּתִי בְּפִי!'. כְּלוֹמַר, כַּאֲשֶׁר אֲנִי מִתְפַּלֵּל, וַאֲנִי מַרְגִּישׁ שֶׁהַמִּלִּים יוֹצְאוֹת לִי מֵהַלֵּב בְּשִׂמְחָה, לְלֹא גִּמְגּוּם, מִלָּה מְחַבֶּרֶת לְמִלָּה בְּכַוָּנָה וּבִדְבֵקוּת – זֶה סִימָן מֵהַשָּׁמַיִם שֶׁהַשֵּׁם שׁוֹמֵעַ אֶת מִלּוֹתַי וּמְקַבֵּל אֶת בַּקָּשָׁתִי! וְעַכְשָׁו תְּפִלָּתִי עָבְרָה בְּלֵב שָׁלֵם וּבְנַחַת, אֲנִי יוֹדֵעַ שֶׁהַיֶּלֶד מַבְרִיא הָרֶגַע!".

הַהוֹכָחָה הַמַּדְהִימָה הַשְּׁלִיחִים הָיוּ חֲכָמִים מְאֹד. מָה הֵם עָשׂוּ מִיָּד? הֵם מִהֲרוּ לִכְתֹּב עַל פֶּתֶק אֶת הַשָּׁעָה הַמְּדֻיֶּקֶת שֶׁבָּהּ אָמַר לָהֶם רַבִּי חֲנִינָא שֶׁהַיֶּלֶד הִבְרִיא (לְפִי אוֹר הַשֶּׁמֶשׁ שֶׁבַּחוּץ). הֵם אָרְזוּ אֶת הַחֲפָצִים, טִפְּסוּ עַל הַסּוּסִים וְדָהֲרוּ חֲזָרָה כָּל הַדֶּרֶךְ עַד לְאַרְמוֹנוֹ שֶׁל נְשִׂיא יִשְׂרָאֵל.

הֵם הִגִּיעוּ לַבַּיִת, מְצַפִּים לְגַלּוֹת מָה קָרָה. רַבָּן גַּמְלִיאֵל יָצָא לִקְרָאתָם וּפָנָיו מְאִירוֹת! "בָּרוּךְ ה'! הַבֵּן שֶׁלִּי מַרְגִּישׁ טוֹב!" אָמַר. "בְּאֶמְצַע הַיּוֹם, פִּתְאוֹם הַחֹם הַנּוֹרָא שֶׁלּוֹ יָרַד, הוּא פָּקַח אֶת עֵינָיו וּבִקֵּשׁ קְצָת מַיִם צְנוּנִים לִשְׁתּוֹת!". הַשְּׁלִיחִים פָּתְחוּ אֶת הַפֶּתֶק, הִבִּיטוּ עַל הַשָּׁעָה הַכְּתוּבָה, וְצָעֲקוּ בִּפְלִיאָה עֲצוּמָה: "הָעֲבוֹדָה! (בִּלְשׁוֹן שְׁבוּעָה עַל שֵׁם עֲבוֹדַת הַמִּקְדָּשׁ) לֹא אִחַרְתֶּם וְלֹא הִקְדַּמְתֶּם וְלוּ בְּדַקָּה אַחַת! בְּדִיּוּק, מַמָּשׁ בְּאוֹתוֹ רֶגַע שֶׁאֲשֶׁר רַבִּי חֲנִינָא יָרַד מֵהָעֲלִיָּה מֵהַתְּפִלָּה הָעוֹצְמָתִית שֶׁלּוֹ – הַחֹם יָרַד לַיֶּלֶד בִּירוּשָׁלַיִם!".

מִכָּאן אָנוּ לוֹמְדִים עַד הַיּוֹם אֶת הַכֹּחַ הָאַדִּיר שֶׁל תְּפִלָּה מְכֻוֶּנֶת: אָדָם עוֹמֵד עָנִי וּפָשׁוּט בַּגָּלִיל, מִתְפַּלֵּל בְּדִיּוּק וּמִכָּל הַלֵּב מִתּוֹךְ מִצְווֹת שֶׁהוּא מְקַיֵּם בְּהִדּוּר, וּמִיָּד – הַקָּדוֹשׁ בָּרוּךְ הוּא בּוֹקֵעַ מֵרְחַקִּים עֲצוּמִים, וּמְרַפֵּא אֶת הַחוֹלֶה מִיָּד!

ד. כיצד האירוע “(בִּלְשׁוֹן שְׁבוּעָה עַל שֵׁם עֲבוֹדַת הַמִּקְדָּשׁ) לֹא אִחַרְתֶּם וְלֹא הִקְדַּמְתֶּם וְלוּ בְּדַקָּה אַחַת!” מחזק את המסר של הסיפור?
