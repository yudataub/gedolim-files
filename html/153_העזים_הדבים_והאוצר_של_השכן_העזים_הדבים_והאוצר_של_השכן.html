העזים, הדבים והאוצר של השכן

דף עבודה בעקבות העזים, הדבים והאוצר של השכן
שם: כיתה:

המעשה — הסיפור המלא

הסיפור מנוקד במלואו כנדרש.

🔹 שָׁלָב 1: בֵּרוּר מַקְדִּים וְנִתּוּחַ (לַמּוֹרֶה/לַמְסַפֵּר)

1. תרנגולות אובדות מגיעות לפתח ביתם הריק ממזון. הפיתוי: לאכול את הביצים. ההחלטה: לשמור עליהן בקנאות למען המאבד.

2. התרנגולות מתרבות ומאלצות אותם לקנות עיזים. השכנים חושדים שהעיזים אוכלות מהשדות שלהם (חשד גזל).

🔹 שָׁלָב 2: הַסִּפּוּר הַמָּלֵא – "הָעִזִּים, הַדֻּבִּים וְהָאוֹצָר שֶׁל הַשָּׁכֵן"

פְּתִיחָה: קַרְקוּרִים בַּפֶּתַח יְלָדִים, אֲנַחְנוּ כְּבָר מַכִּירִים אֶת הַבַּיִת שֶׁל רַבִּי חֲנִינָא בֶּן דּוֹסָא וְאִשְׁתּוֹ. בַּיִת קָטָן, פָּשׁוּט מְאֹד. אֵין בּוֹ מְקָרֵר מָלֵא בְּכָל טוּב, וְאֵין בּוֹ מַמְתַּקִּים. זָכוּר לָנוּ כִּי "חֲנִינָא בְּנִי, מִסְתַּפֵּק בְּקַב חֲרוּבִים מֵעֶרֶב שַׁבָּת לְעֶרֶב שַׁבָּת". הָרָעָב הָיָה אוֹרֵחַ קָבוּעַ בְּבֵיתָם, אַךְ הַשִּׂמְחָה עַל הַתּוֹרָה מִלְּאָה אֶת הַכֹּל.

יוֹם אֶחָד, הַשֶּׁקֶט שֶׁל הַבַּיִת הֻפְרַע. קוּקּוּ... קוּרִיקוּ... קוֹלוֹת שֶׁל קִרְקוּרִים נִשְׁמְעוּ מִבַּחוּץ. יָצְאָה אִשְׁתּוֹ שֶׁל רַבִּי חֲנִינָא לְפֶתַח הַבַּיִת, וְהִנֵּה, עַל הָאֲדָמָה מִתְרוֹצְצוֹת לָהֶן תַּרְנְגוֹלוֹת קְטַנּוֹת. הִיא הִבִּיטָה יָמִינָה וּשְׂמֹאלָה – אֵין אִישׁ בָּרְחוֹב. מִישֶׁהוּ כַּנִּרְאֶה עָבַר דֶּרֶךְ סִמְטְאוֹת הָעִיר הַצָּרוֹת, וְהַתַּרְנְגוֹלוֹת אָבְדוּ לוֹ וְהִגִּיעוּ לְכָאן.

כְּשֶׁחָזַר רַבִּי חֲנִינָא מִבֵּית הַמִּדְרָשׁ, סִפְּרָה לוֹ אִשְׁתּוֹ עַל הָאוֹרְחוֹת שֶׁהִגִּיעוּ. יְלָדִים, אָדָם רָגִיל שֶׁרָעֵב לְלֶחֶם וּמוֹצֵא בֵּיצִים טְרִיּוֹת שֶׁהַתַּרְנְגֹלֶת כְּבָר הִטִּילָה בֶּחָצֵר, מָה הָיָה חוֹשֵׁב? "וָאוּ! ה' שָׁלַח לִי אֲרוּחַת בֹּקֶר! הֲרֵי לֹא גָּנַבְתִּי, הֵן בָּאוּ לְבַד!". אֲבָל רַבִּי חֲנִינָא בֶּן דּוֹסָא הָיָה קָדוֹשׁ וְטָהוֹר. מָמוֹן שֶׁל מִישֶׁהוּ אַחֵר הָיָה בְּעֵינָיו כְּמוֹ אֵשׁ! "חָלִילָה לָנוּ מִלָּגַעַת בַּבֵּיצִים הָאֵלֶּה!" אָמַר רַבִּי חֲנִינָא בְּיִרְאַת שָׁמַיִם. "זוֹ הֲשָׁבַת אֲבֵדָה. בְּוַדַּאי יָבוֹא הָאִישׁ לְחַפֵּשׂ אֶת הַתַּרְנְגוֹלוֹת שֶׁלּוֹ. עַד שֶׁיָּבוֹא – אֲנַחְנוּ נַאֲכִיל אוֹתָן, נַשְׁקֶה אוֹתָן, וְנִשְׁמֹר עֲלֵיהֶן מִכָּל מִשְׁמָר!".

הָעֵדֶר שֶׁגָּדַל לִמְמַדִּים עֲצוּמִים עָבְרוּ יָמִים. הָאִישׁ שֶׁאִבֵּד אֶת הַתַּרְנְגוֹלוֹת לֹא בָּא. וּבֵינְתַיִם – הַתַּרְנְגוֹלוֹת הִטִּילוּ עוֹד וְעוֹד בֵּיצִים. מֵהַבֵּיצִים בָּקְעוּ אֶפְרוֹחִים צְהֻבִּים וּקְטַנִּים. הָאֶפְרוֹחִים גָּדְלוּ לְתַרְנְגוֹלוֹת, וְהֵן הִטִּילוּ עוֹד בֵּיצִים! הַקָּדוֹשׁ בָּרוּךְ הוּא שָׁלַח בְּרָכָה עֲצוּמָה בָּאֲבֵדָה הַזּוֹ! בְּתוֹךְ זְמַן קָצָר, הַבַּיִת הַקָּטָן שֶׁל רַבִּי חֲנִינָא וְאִשְׁתּוֹ הִתְמַלֵּא כֻּלּוֹ בְּנוֹצוֹת, קִרְקוּרִים וְצִיּוּצִים. אִי אֶפְשָׁר הָיָה לָשֶׁבֶת, אִי אֶפְשָׁר הָיָה לִישֹׁן! וְעֲדַיִן – הֵם לֹא נָגְעוּ אֲפִלּוּ בְּבֵיצָה אַחַת! אָסוּר לָגַעַת בְּמָה שֶׁשַּׁיָּךְ לַשֵּׁנִי.

"מָה נַעֲשֶׂה?" שָׁאֲלָה הָאִשָּׁה, "לֹא נִשְׁאַר לָנוּ מָקוֹם בַּבַּיִת!". חָשַׁב רַבִּי חֲנִינָא, וְאָז מָצָא פִּתְרוֹן מֻשְׁלָם: "נִקַּח אֶת כָּל הַתַּרְנְגוֹלוֹת לַשּׁוּק. נִמְכֹּר אוֹתָן כְּדֵי שֶׁלֹּא יֵהָרְסוּ, וּבַכֶּסֶף שֶׁנְּקַבֵּל... נִקְנֶה עִזִּים! עִזִּים תּוֹפְסוֹת פָּחוֹת מָקוֹם, וַאֲנַחְנוּ נִשְׁמֹר אֶת הָעִזִּים עֲבוּר הָאִישׁ, כִּי הַכֶּסֶף שֶׁל הַתַּרְנְגוֹלוֹת חַיָּב לַחְזֹר אֵלָיו!". כָּךְ הָיָה. רַבִּי חֲנִינָא הֵבִיא עֵדֶר שֶׁל עִזִּים לֶחָצֵר שֶׁלּוֹ, וְטִפֵּל בָּהֶן בִּמְסִירוּת שֶׁל רוֹעֶה צֹאן טוֹב.

הַהַקְפָּדָה עַל שְׂדוֹת הַשְּׁכֵנִים וְכָאן הַדָּבָר הַמַּדְהִים בְּיוֹתֵר יְלָדִים. מָה עִזִּים אוֹהֲבוֹת לַעֲשׂוֹת כְּשֶׁהֵן יוֹצְאוֹת לְרְעוֹת בַּשָּׂדֶה? הֵן אוֹהֲבוֹת לִלְעֹס כָּל תִּלְתַּן, לִקְפֹּץ מֵעַל הַגָּדֵר וְלִזְלֹל אֶת תְּבוּאַת הַשָּׂדֶה! אֲבָל הָעִזִּים שֶׁל רַבִּי חֲנִינָא בֶּן דּוֹסָא הָיוּ "עִזִּים עִם שְׁמִירָה מִשָּׁמַיִם". רַבִּי חֲנִינָא הָיָה כָּל כָּךְ נִזְהָר שֶׁלֹּא לִפְגֹּעַ בִּרְכוּשׁ הַשְּׁכֵנִים, עַד שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא שָׁמַר עַל הָעִזִּים שֶׁלּוֹ. כְּשֶׁהָעִזִּים הָלְכוּ, הֵן קָפְצוּ מֵעַל שְׂדוֹת הַשְּׁכֵנִים בְּלִי לָגַעַת בְּעָלֶה אֶחָד יָרֹק! הֵן רָעוּ רַק בְּשָׂדוֹת שֶׁל הֶפְקֵר – דְּבָרִים שֶׁשַּׁיָּכִים לְכֻלָּם.

אַךְ יוֹם אֶחָד הִגִּיעוּ עַד לְבֵיתוֹ שֶׁל רַבִּי חֲנִינָא כַּמָּה שְׁכֵנִים זַעֲמָנִיִּים. הֵם דָּפְקוּ עַל הַדֶּלֶת בְּזַעַם: "רַבִּי חֲנִינָא! הָעִזִּים שֶׁלְּךָ אוֹכְלוֹת לָנוּ אֶת הַשָּׂדוֹת! הֵן לוֹעֲסוֹת אֶת הַיְּבוּל שֶׁהִשְׁקַעְנוּ בּוֹ כָּל כָּךְ הַרְבֵּה!" רַבִּי חֲנִינָא הָיָה בְּהֶלֶם. הוּא אָדָם קָדוֹשׁ. שֶׁהָעִזִּים שֶׁבְּאַחֲרָיוּתוֹ יִגְזְלוּ? יַזִּיקוּ? הֲרֵי הוּא נִזְהָר עַל אֲגוֹרָה! הוּא יָצָא אֶל הַשְּׁכֵנִים, הִבִּיט בָּהֶם בְּעֵינָיו הַטְּהוֹרוֹת, וְאָמַר בְּקוֹל בּוֹטֵחַ וְשָׁלֵו: "שְׁכֵנַי הַיְקָרִים. אֲנִי בָּטוּחַ שֶׁהָעִזִּים שֶׁלִּי לֹא גָּזְלוּ תְּלוּשֶׁת-עֵשֶׂב שֶׁלֹּא שֶׁלָּהֶן. בּוֹאוּ נַעֲשֶׂה הֶסְכֵּם: הָעֶרֶב אֶשְׁלַח אֶת הָעִזִּים לַיַּעַר הַגָּדוֹל, אֵיפֹה שֶׁיֵּשׁ חַיּוֹת טֶרֶף. אִם אַתֶּם צוֹדְקִים, וְהָעִזִּים בֶּאֱמֶת אָכְלוּ מֵהַשָּׂדֶה שֶׁלָּכֶם לְלֹא רְשׁוּת וְגָרְמוּ נֶזֶק – יָבוֹאוּ זְאֵבִים וְדֻבִּים בַּיַּעַר וְיִטְרְפוּ אוֹתָן. עֹנֶשׁ הַגַּנָּב הוּא הֶפְסֵד. אַךְ... אִם סְתָם הֶאֱשַׁמְתֶּם אוֹתִי, וְהָעִזִּים שֶׁלִּי לְעוֹלָם לֹא נָגְעוּ בִּרְכוּשׁ הַזּוּלַת..." "אָז מָה?" צָחֲקוּ הַשְּׁכֵנִים. "אָז הָעִזִּים תַּחְזֹרְנָה הַבַּיְתָה בְּשָׁלוֹם, וְכָל עֵז... תָּבִיא עַל קַרְנֶיהָ דֹּב זְאֵבִי אֶחָד מִן הַיַּעַר!"

הַשְּׁכֵנִים חִיְּכוּ וְגִחֲכוּ. "אֵיזוֹ בְּדִיחָה! עֵז קְטַנָּה תָּבִיא דֹּב עַל הַקַּרְנַיִם? מָה הוּא מַמְצִיא?!". אֲבָל הֵם הִסְכִּימוּ.

הַשּׁוּבָה מִן הַיַּעַר הֶאָפֵל הַשֶּׁמֶשׁ שָׁקְעָה. עֶדֶר הָעִזִּים נִכְנַס לְתּוֹךְ מַעֲבֵה הַיַּעַר. הַשְּׁכֵנִים הִמְתִּינוּ, בְּטוּחִים שֶׁרַבִּי חֲנִינָא יַפְסִיד בַּהִתְעָרְבוּת. "כְּבָר נִטְרְפוּ אוֹתָן עִזִּים גַּזְלָנִיּוֹת," צָחֲקוּ בֵּינֵיהֶם.

אַךְ אָז... כְּשֶׁהֶחְשִׁיךְ לְגַמְרֵי... נִשְׁמְעוּ צְעָדִים עַל הַשְּׁבִיל. קְלִיק-קְלָק... קְלִיק-קְלָק... קוֹל פַּרְסוֹת עִזִּים פוֹסְעוֹת בְּשֶׁקֶט. הַשְּׁכֵנִים שִׁפְשְׁפוּ אֶת עֵינֵיהֶם כְּשֶׁרָאוּ אֶת מָה שֶׁמּוּלָם: הָעִזִּים חָזְרוּ בְּשָׁלוֹם! אַף שְׂעָרָה לֹא חָסְרָה מֵהֶן! וּמָה הָיָה יְלָדִים מֻנָּח עַל כָּל קֶרֶן שֶׁל עֵז? עַל קַרְנֶיהָ שֶׁל הָעֵז הָרִאשׁוֹנָה... הָיָה שָׁפוּף לוֹ דֹּב עֲנָקִי, נָכוֹן לְהִתְפַּס מֵרֹב פַּחַד מֵהַקְּדֻשָּׁה! וְעַל קַרְנֶיהָ שֶׁל הָעֵז הַשְּׁנִיָּה? עוֹד דֹּב פראי! כָּל עֵז חָזְרָה מֵהַיַּעַר שְׁלֵמָה, נוֹשֵׂאת כְּהוֹכָחָה אֶת חַיּוֹת הַטֶּרֶף שֶׁנִּכְנְעוּ מוּל צִדְקָתוֹ וְיָשְׁרוֹ שֶׁל רַבִּי חֲנִינָא בֶּן דּוֹסָא!

א. מה מלמד המעשה “העזים, הדבים והאוצר של השכן” על העזים, הדבים והאוצר של השכן? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “רַבִּי חֲנִינָא הָיָה כָּל כָּךְ נִזְהָר שֶׁלֹּא לִפְגֹּעַ בִּרְכוּשׁ הַשְּׁכֵנִים, עַד שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא שָׁמַר עַל הָעִזִּים שֶׁלּוֹ.”.
ג. אילו העזים, הדבים והאוצר של השכן לא היה פועל כפי שמתואר ב־“הָאִישׁ עָנָה: "כֵּן, הֵן הָיוּ קְשׁוּרוֹת בְּחוּט אָדֹם...” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “הַמְלָצוֹת שִׁלּוּב בְּמַצֶּגֶת (תְּמוּנוֹת צְנוּעוֹת, בְּהַקְפָּדָה עַל רוּחֲנִיּוּת):” מחזק את המסר של הסיפור?

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “רַבִּי חֲנִינָא הֵבִיא עֵדֶר שֶׁל עִזִּים לֶחָצֵר שֶׁלּוֹ, וְטִפֵּל בָּהֶן בִּמְסִירוּת שֶׁל רוֹעֶה צֹאן טוֹב.”.
מכתב: כתבו להעזים, הדבים והאוצר של השכן בעקבות “הַשְּׁכֵנִים שִׁפְשְׁפוּ אֶת עֵינֵיהֶם כְּשֶׁרָאוּ אֶת מָה שֶׁמּוּלָם: הָעִזִּים חָזְרוּ בְּשָׁלוֹם!”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “עָנִי מָרוּד, שֶׁשָּׁמַר לוֹ עַל הָרְכוּשׁ, דָּאַג שֶׁאִישׁ לֹא יִפָּגַע מִמֶּנּוּ, וְהֶחְזִיר לוֹ עֹשֶׁר מִשָּׁמַיִם!”.
דיווח: כתבו ידיעה קצרה שכותרתה “העזים, הדבים והאוצר של השכן”.
