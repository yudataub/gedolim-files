ויתור על ראשות הישיבה — ובמה זכה להאריך ימים

דף עבודה בעקבות ר' חנינא בר חמא
שם: כיתה:

המעשה — הסיפור המלא

להלן עיבוד חינוכי, רגשי ומרתק לסיפורו המופלא של רבי חנינא בר חמא , וויתורו העצום על משרת ראשות הישיבה, והשכר שקיבל על כך מאת ה'. הסיפור מעובד במיוחד לבנים בכיתות ב'-ד' (חינוך דתי-לאומי), תוך שימוש בכללי הסטוריטלינג, תיאורים חושיים, דגש על כבוד התורה, ענווה והשגחה פרטית.

הסיפור מנוקד במלואו כנדרש.

פְּתִיחָה: הַצַּוָּאָה שֶׁל רַבֵּינוּ הַקָּדוֹשׁ

יְלָדִים, אֲנַחְנוּ נִמְצָאִים הַיּוֹם בָּעִיר צִפּוֹרִי שֶׁבַּגָּלִיל. הָאֲוִיר קַר, וְרֵיחַ שֶׁל עֲצֵי זַיִת וּגְפָנִים נִשָּׂא בָּרוּחַ, אַךְ אָוִירָה שֶׁל עֶצֶב כָּבֵד עוֹטֶפֶת אֶת כָּל רְחוֹבוֹת הָעִיר.

בְּחֶדֶר קָטָן וְשָׁקֵט, שׁוֹכֵב עַל מִטָּתוֹ גְּדוֹל הַדּוֹר, נְשִׂיא יִשְׂרָאֵל, רַבִּי יְהוּדָה הַנָּשִׂיא. זֶהוּ אוֹתוֹ הַצַּדִּיק הַקָּדוֹשׁ שֶׁכָּתַב אֶת הַמִּשְׁנָה כְּדֵי שֶׁתּוֹרַת יִשְׂרָאֵל לֹא תִּשְׁתַּכַּח. עַכְשָׁו, רַבֵּינוּ הַקָּדוֹשׁ זָקֵן וְחוֹלֶה, וְהוּא יוֹדֵעַ שֶׁבְּקָרוֹב יַעֲלֶה לְיְשִׁיבָה שֶׁל מַעְלָה.

הוּא קוֹרֵא לִגְדוֹלֵי תַּלְמִידָיו אֵלָיו לַחֶדֶר. הַשֶּׁקֶט בַּחֶדֶר הָיָה כָּל כָּךְ עָמֹק, עַד שֶׁאֶפְשָׁר הָיָה לִשְׁמֹעַ אֶת נְשִׁימוֹתֵיהֶם שֶׁל הַחֲכָמִים הַבּוֹכִים.

רַבִּי יְהוּדָה אוֹסֵף אֶת כֹּחוֹתָיו הָאַחֲרוֹנִים וְלוֹחֵשׁ כַּמָּה הוֹרָאוֹת לְעַם יִשְׂרָאֵל. אַחַת הַהוֹרָאוֹת הַחֲשׁוּבוֹת בְּיוֹתֵר הָיְתָה תַּפְקִיד הַהַנְהָגָה בַּתּוֹרָה. וְכָךְ פּוֹסֵק רַבֵּינוּ הַקָּדוֹשׁ: "חֲנִינָא בַּר חָמָא יֵשֵׁב בָּרֹאשׁ!".

הַמַּשְׁמָעוּת הָיְתָה בְּרוּרָה: צַדִּיק הַדּוֹר, נְשִׂיא יִשְׂרָאֵל, קוֹבֵעַ כִּי הַתַּלְמִיד הַגָּדוֹל, רַבִּי חֲנִינָא, הוּא זֶה שֶׁיִּהְיֶה 'רֹאשׁ הַיְשִׁיבָה' מֵעַתָּה וָאֵילָךְ. זֶהוּ תַּפְקִיד רָם, קָדוֹשׁ וְאָהוּב. לְלַמֵּד תּוֹרָה לְכָל עַם יִשְׂרָאֵל!

בְּבֵית הַמִּדְרָשׁ: הַכִּסֵּא שֶׁמַּמְתִּין

יָמִים סְפוּרִים לְאַחַר מִכֵּן, הִסְתַּלֵּק רַבֵּינוּ הַקָּדוֹשׁ לְבֵית עוֹלָמוֹ. עַם יִשְׂרָאֵל בָּכָה וְהִתְאַבֵּל, אַךְ קוֹל הַתּוֹרָה חַיָּב לְהַמְשִׁיךְ לְהִשָּׁמַע.

בֵּית הַמִּדְרָשׁ הַגָּדוֹל בְּצִפּוֹרִי הָיָה הוֹמֶה. מֵאוֹת חֲכָמִים יָשְׁבוּ עַל סַפְסַלֵּי הָעֵץ הַכְּבֵדִים, הַסְּפָרִים פְּתוּחִים לִפְנֵיהֶם. רִפְרוּף קַל שֶׁל דַּפֵּי קְלָף עַתִּיקִים נִשְׁמַע בָּאֲוִיר, מַמְתִּינִים. כֻּלָּם הִמְתִּינוּ לְרַבִּי חֲנִינָא בַּר חָמָא שֶׁיִּכָּנֵס, יִתְיַשֵּׁב עַל כִּסֵּא רֹאשׁ הַיְשִׁיבָה וְיַתְחִיל לִדְרֹשׁ בַּתּוֹרָה, בְּדִיּוּק כְּפִי שֶׁהוֹרָה רַבֵּינוּ.

וְהִנֵּה, נִכְנַס רַבִּי חֲנִינָא. צַעֲדָיו עֲדִינִים, פָּנָיו מְאִירוֹת בְּאוֹר שֶׁל קְדֻשָּׁה וַעֲנָוָה, אֲבָל... הוּא לֹא מִתְקַדֵּם מִיָּד אֶל הַכִּסֵּא. עֵינָיו שֶׁל רַבִּי חֲנִינָא סוֹרְקוֹת אֶת הַשּׁוּרוֹת הָרִאשׁוֹנוֹת בְּבֵית הַמִּדְרָשׁ. הוּא רוֹאֶה אֶת הַחֲכָמִים הַצַּדִּיקִים מַמְתִּינִים לוֹ, אֲבָל בֵּינֵיהֶם הוּא הִבְחִין בְּאָדָם מְיֻחָד – רַבִּי אַפֵס!

רַבִּי אַפֵס הָיָה תַּלְמִיד חָכָם עָצוּם, צַדִּיק יְסוֹד עוֹלָם. רַבִּי חֲנִינָא יָדַע פְּרָט אֶחָד שֶׁהָיָה קָרוֹב מְאֹד לְלִבּוֹ: רַבִּי אַפֵס מְבֻגָּר מִמֶּנּוּ. בְּכַמָּה זְמַן? בִּשְׁנָתַיִם וָחֵצִי בִּלְבַד.

הַהַחְלָטָה הַקָּשָׁה מִכֹּל

יְלָדִים, אָדָם רָגִיל הָיָה אוֹמֵר: "נָכוֹן, הוּא קְצָת יוֹתֵר מְבֻגָּר מִמֶּנִּי. אֲבָל מָה לַעֲשׂוֹת? גְּדוֹל הַדּוֹר מִנָּה אוֹתִי! הַתַּפְקִיד שֶׁלִּי! אֲנִי אֶהְיֶה רֹאשׁ הַיְשִׁיבָה הֲכִי טוֹב שֶׁאֶפְשָׁר!".

אֲבָל רַבִּי חֲנִינָא עָצַר. לִבּוֹ זִנֵּק בְּקִרְבּוֹ. כְּבוֹד הַתּוֹרָה וּכְבוֹד הַזְּקֵנִים יָקָר לוֹ יוֹתֵר מִכָּל תַּפְקִיד וּמִכָּל כָּבוֹד אִישִׁי.

הוּא פָּנָה לַחֲכָמִים וּבְקוֹל שָׁקֵט וּבָטוּחַ אָמַר: "אֵינֶנִּי יָכוֹל לָשֶׁבֶת בָּרֹאשׁ וּלְלַמֵּד, כַּאֲשֶׁר רַבִּי אַפֵס מְבֻגָּר מִמֶּנִּי בִּשְׁנָתַיִם וּמֶחֱצָה. זֶהוּ כְּבוֹדוֹ שֶׁל זָקֵן שֶׁלָּמַד תּוֹרָה עוֹד לְפָנַי. רַבִּי אַפֵס הוּא זֶה שֶׁיֵּשֵׁב בָּרֹאשׁ!".

בֵּית הַמִּדְרָשׁ עָמַד נִדְהָם. רַבִּי אַפֵס מֻנָּה לְרֹאשׁ הַיְשִׁיבָה! וְהֵיכָן יָשַׁב רַבִּי חֲנִינָא, הַתַּלְמִיד שֶׁרַבֵּינוּ הַקָּדוֹשׁ בִּקֵּשׁ שֶׁיְּכַהֵן כְּרֹאשׁ הַיְשִׁיבָה?

רַבִּי חֲנִינָא יָצָא אֶל מִחוּץ לְבֵית הַמִּדְרָשׁ! מֵרֹב כָּבוֹד, הוּא וִתֵּר לַחֲלוּטִין עַל מְקוֹמוֹ הַמְכֻבָּד בַּפְּנִים, יָשַׁב בַּחוּץ כְּתַלְמִיד רָגִיל מִן הַשּׁוּרָה, וְלָמַד מִפִּיו שֶׁל רַבִּי אַפֵס מִתּוֹךְ עֲנָוָה עֲצוּמָה!

הַהַמְתָּנָה הָאֲרֻכָּה וְהַשָּׂכָר מִשָּׁמַיִם

חָלְפוּ מִסְפַּר שָׁנִים, וְרַבִּי אַפֵס הִסְתַּלֵּק לְבֵית עוֹלָמוֹ. עַתָּה, חָשְׁבוּ הַכֹּל, יַגִּיעַ זְמַנּוֹ שֶׁל רַבִּי חֲנִינָא. אַךְ בְּבֵית הַמִּדְרָשׁ הָיָה עֲדַיִן תַּלְמִיד חָכָם נוֹסָף – רַבִּי לֵוִי. אָמְנָם לֹא הָיָה גָּדוֹל כְּרַבִּי חֲנִינָא בַּהֲלָכָה, אַךְ יְלָדִים... גַּם הוּא הָיָה מְבֻגָּר יוֹתֵר!

וְשׁוּב, רַבִּי חֲנִינָא מוֹכִיחַ גְּדֻלַּת נֶפֶשׁ אֲמִתִּית. שׁוּב הוּא מוֹשֵׁךְ יָדָיו. "אֵינֶנִּי יָכוֹל לָקַחַת אֶת הַתַּפְקִיד מִמִּי שֶׁגָּדוֹל מִמֶּנִּי," פָּסַק שׁוּב בַּעֲנָוָה, וְאִפְשֵׁר לְרַבִּי לֵוִי לְהַנְהִיג. מַמָּשׁ כָּךְ! הַכֶּתֶר הִמְתִּין, אַךְ רַבִּי חֲנִינָא בָּחַר בְּכָבוֹד חֲבֵרוֹ עַל פְּנֵי תּוֹאַר וְכָבוֹד עַצְמִי.

רַק לְאַחַר שְׁנוֹתָיו שֶׁל רַבִּי לֵוִי, קִבֵּל לְבַסּוֹף רַבִּי חֲנִינָא אֶת מִשְׂרַת רֹאשׁ הַיְשִׁיבָה שֶׁנּוֹעֲדָה לוֹ, מִבְּלִי לִפְגֹּעַ בְּאִישׁ.

וְהַקָּדוֹשׁ בָּרוּךְ הוּא, שֶׁרוֹאֶה מַעֲשֵׂה הָאָדָם וּבוֹחֵן כְּלָיוֹת וָלֵב, בֵּרֵךְ אֶת רַבִּי חֲנִינָא מִדָּה כְּנֶגֶד מִדָּה! עַל שֶׁוִּתֵּר עַל הַשָּׁנִים בָּהֶן הָיָה אָמוּר לַעֲמֹד בָּרֹאשׁ לְטוֹבַת הַמְבֻגָּרִים מִמֶּנּוּ, הוּא זָכָה לַאֲרִיכוּת יָמִים נְדִירָה מְאֹד. וְלֹא סְתָם אֲרִיכוּת יָמִים – אֶלָּא כֹּחַ וּגְבוּרָה לִלְמֹד וּלְקַיֵּם מִצְווֹת לְלֹא כָּל חֻלְשָׁה!

תּאֲרוּ לָכֶם, רַבִּי חֲנִינָא הִגִּיעַ לְגִיל שְׁמוֹנִים שָׁנָה! הוּא הָיָה עָמֵל בַּתּוֹרָה, וְכוֹחוֹ הָיָה כְּשֶׁל בָּחוּר צָעִיר! מְסַפְּרִים חֲזַ"ל כִּי בְּגִיל שְׁמוֹנִים הָיָה עוֹמֵד עַל רֶגֶל אַחַת שֶׁלּוֹ, חוֹלֵץ וְנוֹעֵל אֶת נַעֲלוֹ שֶׁל הָרֶגֶל הַשְּׁנִיָּה לְלֹא צֹרֶךְ בְּהִשָּׁעֲנוּת אוֹ יְשִׁיבָה! הַשֶּׁמֶשׁ שֶׁל יְרוּשָׁלַיִם הֵאִירָה עַל פָּנָיו הַזְּקוּפוֹת.

כְּשֶׁרָאוּ זֹאת תַּלְמִידָיו, שָׁאֲלוּ אוֹתוֹ פַּעַם: "רַבֵּינוּ, בִּזְכוּת מָה זָכִיתָ לַאֲרִיכוּת יָמִים מֻפְלָאָה כָּזוֹ? לְכֹחַ וּלְבְרִיאוּת אֵיתָנָה?".

חִיֵּךְ הַצַּדִּיק, הֶעֱבִיר אֶת יָדוֹ עַל זְקָנוֹ הַלָּבָן כַּשֶּׁלֶג, וֶהֱשִׁיב: "בְּכָל יָמַי... מֵעוֹלָם לֹא הִקְדַּמְתִּי, וְלֹא תָּפַסְתִּי אֶת מְקוֹמוֹ שֶׁל אָדָם שֶׁהָיָה גָּדוֹל מִמֶּנִּי".

הוּא וִתֵּר עַל כְּבוֹדוֹ, וּמֶלֶךְ הַכָּבוֹד, הַקָּדוֹשׁ בָּרוּךְ הוּא, הֶעֱנִיק לוֹ כָּבוֹד וְחַיִּים מְלֵאִים בְּתּוֹרָה!

א. שְׁאֵלַת פְּתִיחָה:

"ילדים, תגידו את האמת – מי כאן ממש אוהב להיות ראשון? להיות מנהיג הקבוצה, הקפטן?

ד. פְּעִילוּת מְסַכֶּמֶת (הַפְנָמָה):

לוח 'גיבורי הוויתור':

תלו בכיתה פלקט בשם "רבי חנינא ויתר – גם אנחנו יכולים!". חלקו לכל ילד פתק דביק קטן. בקשו שכל ילד יחשוב ויכתוב/יצייר על הפתק מצב אחד שבו קשה לו לוותר (בתור לקיוסק, על השלט של המזגן, על מקום במשחק). כל ילד ידביק את הפתק על הלוח ויכריז בלבו שהוא לוקח על עצמו לנסות לוותר לפחות פעם אחת השבוע, כמו רבי חנינא.

מקורות: תלמוד ירושלמי, מסכת תענית פרק ד' הלכה ב': על הצוואה של רבי "חנינא בר חמא ישב בראש", ועל סירובו של ר' חנינא לקבל את המשרה כי ר' אפס היה גדול ממנו בשנתיים ומחצה. תלמוד בבלי, מסכת כתובות דף קג עמוד ב': צוואתו של רבי יהודה הנשיא על מחליפיו, וישיבתו של ר' חנינא מחוץ לבית המדרש בימי ר' אפס. תלמוד בבלי, מסכת חולין דף כד עמוד ב': על בריאותו המופלגת של רבי חנינא בגיל 80 (יכולתו לחלוץ ולנעול מנעליו בעמידה על רגל אחת), ועל תשובתו שזכה לכך בזכות שמעולם לא תפס מקום של אדם שהיה גדול ממנו. (כמו כן, הרקע מדוע רבי לא מינהו בחייו מופיע במקורות שונים המצביעים על הקפדה קלה שהייתה, כמפורט בחוברות החמ"ד הנלוות).

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “להלן עיבוד חינוכי, רגשי ומרתק לסיפורו המופלא של רבי חנינא בר חמא , וויתורו העצום על משרת ראשות הישיבה, והשכר שקיבל על כך מאת ה'.” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “אַחַת הַהוֹרָאוֹת הַחֲשׁוּבוֹת בְּיוֹתֵר הָיְתָה תַּפְקִיד הַהַנְהָגָה בַּתּוֹרָה.”?
ג. כיצד הגיב ר' חנינא בר חמא למצב המתואר: “צַעֲדָיו עֲדִינִים, פָּנָיו מְאִירוֹת בְּאוֹר שֶׁל קְדֻשָּׁה וַעֲנָוָה, אֲבָל...”?
ד. מה השתנה בין “מֵרֹב כָּבוֹד, הוּא וִתֵּר לַחֲלוּטִין עַל מְקוֹמוֹ הַמְכֻבָּד בַּפְּנִים, יָשַׁב בַּחוּץ כְּתַלְמִיד רָגִיל מִן…” לבין “וְהַקָּדוֹשׁ בָּרוּךְ הוּא, שֶׁרוֹאֶה מַעֲשֵׂה הָאָדָם וּבוֹחֵן כְּלָיוֹת וָלֵב, בֵּרֵךְ אֶת רַבִּי חֲנִינָא מִדָּה…”?
ה. כיצד הגיע הסיפור אל האירוע: “הוּא וִתֵּר עַל כְּבוֹדוֹ, וּמֶלֶךְ הַכָּבוֹד, הַקָּדוֹשׁ בָּרוּךְ הוּא, הֶעֱנִיק לוֹ כָּבוֹד וְחַיִּים מְלֵאִים בְּתּוֹרָה!”?

3. מה היה ראשון?

כתבו ליד כל אירוע מספר מ־1 עד 5 לפי סדר התרחשותו בסיפור.

□ כְּבוֹד הַתּוֹרָה וּכְבוֹד הַזְּקֵנִים יָקָר לוֹ יוֹתֵר מִכָּל תַּפְקִיד וּמִכָּל כָּבוֹד אִישִׁי.

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “ויתור על ראשות הישיבה — ובמה זכה להאריך ימים” על ר' חנינא בר חמא? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “רַבִּי חֲנִינָא יָדַע פְּרָט אֶחָד שֶׁהָיָה קָרוֹב מְאֹד לְלִבּוֹ: רַבִּי אַפֵס מְבֻגָּר מִמֶּנּוּ.”.
ג. אילו ר' חנינא בר חמא לא היה פועל כפי שמתואר ב־“וְהַקָּדוֹשׁ בָּרוּךְ הוּא, שֶׁרוֹאֶה מַעֲשֵׂה הָאָדָם וּבוֹחֵן כְּלָיוֹת וָלֵב, בֵּרֵךְ אֶת רַבִּי חֲנִינָא מִדָּה כְּנֶגֶד…” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “הוּא וִתֵּר עַל כְּבוֹדוֹ, וּמֶלֶךְ הַכָּבוֹד, הַקָּדוֹשׁ בָּרוּךְ הוּא, הֶעֱנִיק לוֹ כָּבוֹד וְחַיִּים מְלֵאִים בְּתּוֹרָה!” מחזק את המסר של הסיפור?
