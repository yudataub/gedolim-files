סיפור 95 • ר' חנינא בר חמא • דבש הדבורים שהוחלף

דבש הדבורים שהוחלף

דף עבודה בעקבות ר' חנינא בר חמא
שם: כיתה:

המעשה — הסיפור המלא

פְּתִיחָה: קוֹלוֹת שֶׁל מִסְחָר וְרֵיחַ שֶׁל תּוֹרָה

יְלָדִים, אִם הֱיִיתֶם מְטַיְּלִים לִפְנֵי שְׁנוֹת דּוֹר בָּרְחוֹבוֹת שֶׁל הָעִיר צִפּוֹרִי שֶׁבַּגָּלִיל, הֱיִיתֶם מַרְגִּישִׁים מִיָּד בָּאֲוִיר מַשֶּׁהוּ מְיֻחָד.

הָעִיר צִפּוֹרִי הָיְתָה עִיר עֲנָקִית. רֵיחוֹת עַזִּים שֶׁל תַּבְלִינִים, קִנָּמוֹן וְזַעְפְרָן בָּקְעוּ מֵהַשּׁוּק הַגָּדוֹל, קוֹלוֹת פַּרְסוֹת הַסּוּסִים נִשְׁמְעוּ עַל אֲבָנֵי הָרְחוֹב, וּסְוָחָרִים מִכָּל הָעוֹלָם בָּאוּ לִקְנוֹת וְלִמְכֹּר. אֲבָל מֵעַל כָּל הָרַעַשׁ הַזֶּה, אִם הֱיִיתֶם מַטִּים אֹזֶן, הֱיִיתֶם שׁוֹמְעִים אֶת הַקּוֹל הֲכִי נִפְלָא וּמָתוֹק בָּעוֹלָם – קוֹל לִמּוּד הַתּוֹרָה הַבּוֹקֵעַ מִבֵּית הַמִּדְרָשׁ הַגָּדוֹל.

שָׁם, בְּבֵית הַמִּדְרָשׁ, יָשַׁב מַנְהִיג הַדּוֹר, הָאָמוֹרָא הַקָּדוֹשׁ רַבִּי חֲנִינָא בַּר חָמָא .

רַבִּי חֲנִינָא הָיָה אָדָם מְיֻחָד בְּמִינוֹ. הוּא לֹא הָיָה רַק תַּלְמִיד חָכָם עָצוּם שֶׁיּוֹדֵעַ אֶת כָּל הַתּוֹרָה, הוּא הָיָה גַּם רוֹפֵא דָּגוּל שֶׁרִפֵּא אֲנָשִׁים, וְגַם אָדָם שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא בֵּרֵךְ אוֹתוֹ בְּעֹשֶׁר גָּדוֹל. אֲבָל הַדָּבָר שֶׁהֲכִי בָּלַט אֵצֶל רַבִּי חֲנִינָא, הָיָה הַיֹּשֶׁר שֶׁלּוֹ . אֶצְלוֹ, מַטְבֵּעַ שֶׁל מִישֶׁהוּ אַחֵר – הָיָה שָׁמוּר כְּמוֹ יַהֲלוֹם שֶׁל הַמֶּלֶךְ.

הַפִּקָּדוֹן הַמָּתוֹק

בֹּקֶר אֶחָד, הִגִּיעַ לְבֵיתוֹ שֶׁל רַבִּי חֲנִינָא סוֹחֵר יְהוּדִי, כְּשֶׁבְּיָדָיו קַד חֶרֶס כָּבֵד וְקַר.

הַסּוֹחֵר נָשַׁם בִּכְבֵדוּת מֵהַמַּשָּׂא, נִגֵּב אֶת הַזֵּעָה מִמִּצְחוֹ וְאָמַר: "כְּבוֹד הָרַב! אֲנִי חַיָּב לִנְסֹעַ לַמֶּרְחַקִּים לְצֹרֶךְ עֲסָקִים. יֵשׁ לִי כָּאן קַד מָלֵא בִּדְבַשׁ טָהוֹר וּמְשֻׁבָּח, צָהֹב וְסָמִיךְ כְּמוֹ זָהָב. אֲנִי לֹא מְעִיז לְהַשְׁאִיר אוֹתוֹ אֵצֶל אַף אֶחָד, מִלְּבַדְךָ. רַק עָלֶיךָ אֲנִי סוֹמֵךְ. הַאִם תַּסְכִּים לִשְׁמֹר לִי עַל הַדְּבַשׁ עַד שֶׁאָשׁוּב?".

רַבִּי חֲנִינָא חִיֵּךְ מִבַּעַד לִזְקָנוֹ הַהָדוּר וְהִנְהֵן בְּרֹאשׁוֹ. "סַע לְשָׁלוֹם, יְדִידִי. הַקַּד שֶׁלְּךָ שָׁמוּר עִמָּדִי. ה' יִשְׁמֹר צֵאתְךָ וּבוֹאֶךָ".

רַבִּי חֲנִינָא לָקַח אֶת כַּד הַחֶרֶס הַכָּבֵד, וְהִנִּיחַ אוֹתוֹ בְּפִנָּה קְרִירָה וּמוּצֶלֶת בְּתוֹךְ הַמַּחְסָן שֶׁלּוֹ. הַסּוֹחֵר הוֹדָה לָרַב, וְיָצָא לְדַרְכּוֹ.

הַדְּאָגָה הַגְּדוֹלָה

עָבְרוּ שָׁבוּעוֹת, עָבְרוּ גַּם חֳדָשִׁים, וְהַסּוֹחֵר – אֵינֶנּוּ קוֹרֵא וְאֵינֶנּוּ בָּא. אוּלַי הִתְעַכֵּב בְּמְדִינָה אֲחֶרֶת, אוּלַי הַסְּפִינָה שֶׁלּוֹ אֵחֲרָה. אִישׁ לֹא יָדַע.

יוֹם אֶחָד, יָרַד רַבִּי חֲנִינָא לַמַּחְסָן וְהִבִּיט בְּכַד הַדְּבַשׁ. בְּתוֹר רוֹפֵא וְחָכָם גָּדוֹל, רַבִּי חֲנִינָא הֵבִין בְּטֶבַע הַדְּבָרִים. הוּא הֵרִיחַ מְעַט מֵהַקַּד וְהֵבִין שֶׁמַּשֶּׁהוּ לֹא קַשׁוּר. כַּאֲשֶׁר דְּבַשׁ עוֹמֵד זְמַן רַב מִדַּי בְּלִי טִפּוּל, הוּא עָלוּל לְאַבֵּד מִטַּעְמוֹ, לְהַחְמִיץ אוֹ לְהִתְגַּבֵּשׁ לִגְוִישִׁים קָשִׁים. מְעַט נְמָלִים תּוֹעוֹת גַּם הֵחֵלּוּ לְהִתְאַסֵּף סְבִיב הַכַּד.

יְלָדִים, אָדָם רָגִיל הָיָה אוֹמֵר לְעַצְמוֹ: "מָה זֶה אִכְפַּת לִי? אֲנִי רַק שׁוֹמֵר! הִנַּחְתִּי אֶת הַקַּד פֹּה, וְאִם הוּא הִתְקַלְקֵל – זוֹ בְּעָיָה שֶׁל הַסּוֹחֵר שֶׁאֵחֵר לַחְזֹר! הֲרֵי לֹא נָגַעְתִּי בּוֹ!".

אֲבָל רַבִּי חֲנִינָא הָיָה קָדוֹשׁ. שׁוֹמֵר חִנָּם צָרִיךְ לִשְׁמֹר מִכָּל מִשְׁמָר! מָה יִהְיֶה עַל רְכוּשׁוֹ שֶׁל הַיְּהוּדִי? "אֲנִי חַיָּב לְהַצִּיל אֶת הַכֶּסֶף שֶׁלּוֹ מִקִּלְקוּל!" חָשַׁב הָרַב.

בְּלִי לְבַזְבֵּז רֶגַע, הֵרִים הָרַב אֶת הַכַּד הַכָּבֵד, וְצָעַד אִתּוֹ לְשׁוּק הַגָּדוֹל שֶׁל צִפּוֹרִי. הַשֶּׁמֶשׁ יָקְדָה, אֲבָל הָרַב לֹא עָצַר. עַד שֶׁמָּצָא קוֹנֶה יָשָׁר, מָכַר לוֹ אֶת הַדְּבַשׁ בְּמֶחִיר הָגוֹן, וְקִבֵּל לְיָדָיו מַטְבְּעוֹת כֶּסֶף מְצַלְצְלִים וּבּוֹרְקִים.

הַהַמְצָאָה הַגְּאוֹנִית: הַדְּבוֹרִים

עַכְשָׁו, הָיָה בְּיָדָיו שֶׁל רַבִּי חֲנִינָא כֶּסֶף שֶׁשַּׁיָּךְ לַסּוֹחֵר. אֲבָל כֶּסֶף שֶׁיּוֹשֵׁב בָּאָרוֹן – לֹא מוֹסִיף כְּלוּם לַבְּעָלִים. מָה עָשָׂה רַבִּי חֲנִינָא בְּחָכְמָתוֹ כְּדֵי שֶׁהַדְּבַשׁ יַמְשִׁיךְ לִחְיוֹת?

בַּכֶּסֶף הַזֶּה... הוּא הָלַךְ וְקָנָה דְּבוֹרִים! כֵּן, כַּמָּה דְּבוֹרִים חָרוּצוֹת עִם פַּסִּים צְהֻבִּים וּשְׁחוֹרִים.

הָרַב הֵבִיא אֶת הַדְּבוֹרִים אֶל חֲצַר בֵּיתוֹ. בְּמוֹ יָדָיו בָּנָה עֲבוּרָן כַּוֶּרֶת עֵץ נָאָה. כָּל בֹּקֶר הָיָה יוֹצֵא לֶחָצֵר, שׁוֹמֵעַ אֶת הַזִּמְזוּם הֶעָמֹק וְהַחֲרִישִׁי – זְזְזְזְ... זְזְזְזְ... – וְרוֹאֶה אֵיךְ הַדְּבוֹרִים עָפוֹת אֶל פִּרְחֵי הַגָּלִיל הָרֵיחָנִיִּים, יוֹנְקוֹת אֶת הַצּוּף הַמָּתוֹק וְחוֹזְרוֹת לַכַּוֶּרֶת כְּדֵי לְיַצֵּר דְּבַשׁ טָרִי.

הַשָּׁנִים חָלְפוּ. בְּהַשְׁגָּחָה פְּרָטִית מֻפְלָאָה, הַקָּדוֹשׁ בָּרוּךְ הוּא בֵּרֵךְ אֶת מַעֲשֵׂה יָדָיו שֶׁל הָרַב בִּבְרָכָה גְּדוֹלָה. הַדְּבוֹרִים פָּרוּ וְרָבוּ בְּקֶצֶב מְסַחְרֵר! מִכַּוֶּרֶת אַחַת הֵן הָפְכוּ לִשְׁתַּיִם, וְאָז לְעֶשֶׂר, וְאָז לְעֶשְׂרִים! הֶחָצֵר שֶׁל רַבִּי חֲנִינָא הָפְכָה לְחַוַּת רְדִיַּת דְּבַשׁ שְׁלֵמָה. רֵיחַ מְשַׁכֵּר וּמָתוֹק שֶׁל דְּבַשׁ עָמַד בָּאֲוִיר. רַבִּי חֲנִינָא טִפֵּל בָּהֶן בִּמְסִירוּת שֶׁל פּוֹעֵל עָמֵל. הוּא אָסַף אֶת הַדְּבַשׁ לְתוֹךְ כַּדִּים נוֹצְצִים וַחֲדָשִׁים, אֲבָל – הוּא וּבְנֵי בֵּיתוֹ לֹא טָעֲמוּ אֲפִלּוּ טִפָּה אַחַת! .

הֵם רָאוּ אֶת הַזָּהָב הַמָּתוֹק הַזֶּה קוֹרֵץ אֲלֵיהֶם, יָדְעוּ שֶׁטַּעֲמוֹ עָשִׁיר עַל הַלָּשׁוֹן, אַךְ יָדְעוּ שֶׁזֶּה גָּזֵל מֻחְלָט. "זֶה לֹא שֶׁלָּנוּ," אָמַר הָרַב, "זֶה שֶׁל הַיְהוּדִי שֶׁהִפְקִיד אֶצְלֵנוּ אֶת הַקַּד הַקָּטָן בְּאוֹתוֹ הַבֹּקֶר."

הַשּׁוּב הַמַּפְתִּיעַ

עָבְרוּ שָׁנִים אֲרֻכּוֹת. יוֹם אֶחָד, הָלַךְ רַבִּי חֲנִינָא בִּרְחוֹבוֹת צִפּוֹרִי. פִּתְאוֹם, הוּא זִהָה פָּנִים מֻכָּרוֹת. אָדָם מְבֻגָּר, שֶׁעוֹרוֹ צָרוּב מֵהַשֶּׁמֶשׁ שֶׁל שְׁנוֹת מַסָּע בַּמִּדְבָּר וּבַיָּם. זֶה הָיָה הַסּוֹחֵר מֵאוֹתוֹ הַיּוֹם!

הַסּוֹחֵר נִגַּשׁ לָרַב בְּבוּשָׁה, עֵינָיו מֻשְׁפָּלוֹת לָאֲדָמָה. "שָׁלוֹם עָלֶיךָ, כְּבוֹד הָרַב," לָחַשׁ. "אוּלַי הָרַב זוֹכֵר אוֹתִי? לִפְנֵי כָּךְ וְכָךְ שָׁנִים... הִשְׁאַרְתִּי אֵצֶל הָרַב קַד קָטָן שֶׁל דְּבַשׁ. אֲנִי יוֹדֵעַ שֶׁעָבַר זְמַן רַב. הַעֲסָקִים בַּנֵּכָר הִסְתַּבְּכוּ וְנִתְקַעְתִּי בַּגּוֹלָה לְשָׁנִים אֲרֻכּוֹת. בְּוַדַּאי הַדְּבַשׁ הַהוּא כְּבָר הִרְקִיב וְנִזְרַק, הֲרֵי מִי יָכוֹל לִשְׁמֹר עַל דְּבַשׁ כָּל כָּךְ הַרְבֵּה זְמַן? בָּאתִי רַק לְבַקֵּשׁ סְלִיחָה עַל הַטִּרְחָה שֶׁגָּרַמְתִּי..."

עֵינָיו שֶׁל רַבִּי חֲנִינָא הֵאִירוּ מִשִּׂמְחָה שֶׁל קְדֻשָּׁה.

"יְדִידִי הַיָּקָר!" קָרָא הָרַב בְּשִׂמְחָה עֲצוּמָה, "מָה פִּתְאוֹם נִזְרַק? הַדְּבַשׁ שֶׁלְּךָ חַי וְקַיָּם! בּוֹא, בּוֹא אַחֲרַי הַבַּיְתָה!".

הַסּוֹחֵר הָיָה הָמוּם. הַאִם הָרַב שָׁמַר לוֹ כַּד רָקוּב וּמָלֵא נְמָלִים? הוּא הָלַךְ אַחֲרֵי רַבִּי חֲנִינָא כְּשֶׁלִּבּוֹ פּוֹעֵם.

כְּשֶׁהִגִּיעוּ לֶחָצֵר בֵּיתוֹ שֶׁל הָרַב, נִעֲנֵעַ הָרַב בְּיָדוֹ לְעֵבֶר הֶחָצֵר. "הַבֵּט כָּאן," אָמַר הָרַב וְהִצְבִּיעַ.

הַסּוֹחֵר פָּעַר אֶת עֵינָיו הַגְּדוֹלוֹת. הוּא רָאָה עֲשָׂרוֹת כַּוָּרוֹת שֶׁל דְּבוֹרִים. מֵאוֹת כַּדִּים נְקִיִּים שֶׁבְּתוֹכָם מִתְנוֹצֵץ דְּבַשׁ זָהֹב, רֵיחָנִי וּמָתוֹק. עֲנָנִים שֶׁל דְּבוֹרִים זִמְזְמוּ בַּאֲוִיר הַפָּתוּחַ, בְּחַוָּה שֶׁשָּׁוָה הוֹן תּוֹעֵפוֹת!

"מָה הַדָּבָר הַזֶּה, רַבִּי?" גִּמְגֵּם הַסּוֹחֵר מִתּוֹךְ תַּדְהֵמָה. "אֵיזֶה עוֹשֶׁר כָּבֵד יֵשׁ כָּאן!".

"שֶׁלְּךָ הוּא!" אָמַר רַבִּי חֲנִינָא בְּפַשְׁטוּת. "שֶׁלְּךָ! רָאִיתִי שֶׁהַדְּבַשׁ שֶׁלְּךָ מַתְחִיל לְהִתְקַלְקֵל, אָז מָכַרְתִּי אוֹתוֹ וְקָנִיתִי דְּבוֹרִים. וְהַדְּבוֹרִים – הֵן הֵבִיאוּ אֶת הַבְּרָכָה הַזּוֹ! קַח יְדִידִי אֶת כָּל אֵלֶּה, כִּי זֶה הָרְכוּשׁ שֶׁלְּךָ!".

הַסּוֹחֵר נָפַל עַל צַוָּאר הָרַב וּבָכָה מֵהִתְרַגְּשׁוּת. הוּא שָׁמַע עַל צַדִּיקִים בָּעוֹלָם, אַךְ עַל כָּזֶה יֹשֶׁר, עַל אָדָם שֶׁעוֹבֵד שָׁנִים כְּדֵי שֶׁהַכֶּסֶף שֶׁל הֶחָבֵר שֶׁלּוֹ לֹא יֹאבַד וְאֵינוֹ לוֹקֵחַ לְעַצְמוֹ פְּרוּטָה – אֶת זֶה אֶפְשָׁר לִמְצֹא רַק אֵצֶל עֲנָקֵי הָרוּחַ, רַק אֵצֶל צַדִּיקָיו שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא!

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “דבש הדבורים שהוחלף” על ר' חנינא בר חמא? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “מְעַט נְמָלִים תּוֹעוֹת גַּם הֵחֵלּוּ לְהִתְאַסֵּף סְבִיב הַכַּד.”.
ג. אילו ר' חנינא בר חמא לא היה פועל כפי שמתואר ב־“הֶחָצֵר שֶׁל רַבִּי חֲנִינָא הָפְכָה לְחַוַּת רְדִיַּת דְּבַשׁ שְׁלֵמָה.” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “כְּשֶׁהִגִּיעוּ לֶחָצֵר בֵּיתוֹ שֶׁל הָרַב, נִעֲנֵעַ הָרַב בְּיָדוֹ לְעֵבֶר הֶחָצֵר.” מחזק את המסר של הסיפור?

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “יוֹם אֶחָד, יָרַד רַבִּי חֲנִינָא לַמַּחְסָן וְהִבִּיט בְּכַד הַדְּבַשׁ.”.

7. חיבור לחיים מתוך המעשה

א. מה אפשר ללמוד מן הדרך שבה ר' חנינא בר חמא פעל באירוע “עַד שֶׁמָּצָא קוֹנֶה יָשָׁר, מָכַר לוֹ אֶת הַדְּבַשׁ בְּמֶחִיר הָגוֹן, וְקִבֵּל לְיָדָיו מַטְבְּעוֹת כֶּסֶף מְצַלְצְלִים…”?
