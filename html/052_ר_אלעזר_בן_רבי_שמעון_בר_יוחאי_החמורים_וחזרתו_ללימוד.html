החמורים וחזרתו ללימוד

דף עבודה בעקבות ר' אלעזר בן רבי שמעון (בר יוחאי)
שם: כיתה:

המעשה — הסיפור המלא

להלן עיבוד חינוכי, רגשי ומרתק למדרש המופלא על רבי אלעזר בן רבי שמעון וחזרתו ללימוד התורה . הסיפור מעובד בדיוק לפי ההנחיות עבור בנים בכיתות ב'-ד' (חינוך דתי-לאומי), תוך שימוש בתיאורים חושיים, הקפדה על כבוד התורה וחכמיה, והמחשת מוסר ההשכל: הכוח האמיתי של האדם נועד לעבודת ה' וללימוד התורה .

הסיפור מנוקד במלואו כנדרש.

פְּתִיחָה: הַכֹּחַ הָעָצוּם שֶׁל בֶּן הַצַּדִּיק

יְלָדִים, מִי מִכֶּם הוּא הַיֶּלֶד הֲכִי חָזָק בַּכִּתָּה? מִי מַצְלִיחַ לְהָרִים דְּבָרִים כְּבֵדִים, לָרוּץ הֲכִי מַהֵר אוֹ לְנַצֵּחַ בְּהוֹרָדַת יָדַיִם?

הַיּוֹם אֲנַחְנוּ נִפְגֹּשׁ תַּנָּא קָדוֹשׁ שֶׁהָיָה אָדָם חָזָק יוֹתֵר מִכָּל מָה שֶׁאַתֶּם מְסֻגָּלִים לְדַמְיֵן! קָרְאוּ לוֹ רַבִּי אֶלְעָזָר . הוּא הָיָה הַבֵּן שֶׁל רַבִּי שִׁמְעוֹן בַּר יוֹחַאי, הַצַּדִּיק שֶׁהִסְתַּתֵּר בַּמְּעָרָה.

לְאַחַר שֶׁיָּצְאוּ מֵהַמְּעָרָה, גָּדַל רַבִּי אֶלְעָזָר וְהָפַךְ לְאִישׁ חָסוֹן מְאֹד. הַשְּׁרִירִים שֶׁלּוֹ הָיוּ חֲזָקִים כְּבַרְזֶל. הַשִּׁלְטוֹן הָרוֹמָאִי, שֶׁשָּׁלַט אָז בָּאָרֶץ, זִהָה אֶת הַכֹּחַ הָעָצוּם שֶׁלּוֹ, וְקָבַע שֶׁהוּא יִהְיֶה "מְמֻנֶּה עַל עֲבוֹדַת הַמֶּלֶךְ" – תַּפְקִיד קָשֶׁה שֶׁדָּרַשׁ הֲלִיכוֹת אֲרֻכּוֹת בַּדְּרָכִים, תְּפִיסַת שׁוֹדְדִים וּפְעֻלּוֹת כּוֹחָנִיּוֹת.

רַבִּי אֶלְעָזָר הָיָה מְנַצֵּחַ אֶת כֻּלָּם. בְּכָל מָקוֹם שֶׁאֵלָיו הִגִּיעַ, כֻּלָּם פָּחֲדוּ מִמֶּנּוּ וּמִכּוֹחוֹ הָרַב. אַךְ מֵרֹב עֲבוֹדָה, הִתְרַחֵק מְעַט מִסַּפְסְלֵי בֵּית הַמִּדְרָשׁ. הַיָּדַיִם הַקְּדוֹשׁוֹת שֶׁלּוֹ הָיוּ מְלֵאוֹת בַּעֲבוֹדַת הַמֶּלֶךְ, וּפָחוֹת בַּאֲחִיזַת סִפְרֵי הַתּוֹרָה.

וְהַקָּדוֹשׁ בָּרוּךְ הוּא שֶׁבַּשָּׁמַיִם, אוֹהֵב אֶת רַבִּי אֶלְעָזָר. "הַיָּדַיִם הָאֵלֶּה צְרִיכוֹת לְהַחְזִיק תּוֹרָה!" אָמַר כִּבְיָכוֹל הַקָּדוֹשׁ בָּרוּךְ הוּא, וְשָׁלַח אֵלָיו אוֹרֵחַ מְיֻחָד בְּמִינוֹ.

הַזָּקֵן, הַמַּשָּׂא וְהַהַבְטָחָה

יוֹם אֶחָד, הָלַךְ רַבִּי אֶלְעָזָר בְּדֶרֶךְ עָפָר רְחָבָה. הַשֶּׁמֶשׁ יָקְדָה בַּשָּׁמַיִם, וְרֵיחַ יָבֵשׁ שֶׁל אֲדָמָה חַמָּה עָמַד בָּאֲוִיר. לְפֶתַע, הוּא רָאָה מֵרָחוֹק אִישׁ זָקֵן. הַזָּקֵן נִשְׁעַן עַל מַקֵּל עֵץ, פָּנָיו הָיוּ חֲרוּשׁוֹת קְמָטִים, וְזָקָן לָבָן אָרֹךְ עִטֵּר אֶת פָּנָיו. לְרַגְלָיו שֶׁל הַזָּקֵן הָיוּ מֻנָּחוֹת כַּמָּה חֲבִילוֹת כְּבֵדוֹת.

רַבִּי אֶלְעָזָר לֹא יָדַע זֹאת, אֲבָל הַזָּקֵן הַזֶּה לֹא הָיָה אָדָם רָגִיל. זֶה הָיָה... אֵלִיָּהוּ הַנָּבִיא מַלְאַךְ הַבְּרִית!

הַזָּקֵן הֵרִים אֶת מַבָּטוֹ וְאָמַר לְרַבִּי אֶלְעָזָר בְּקוֹל רוֹעֵד: "צָעִיר חָזָק וְחָסוֹן, אֲנִי זָקֵן וְעָיֵף. הַאִם תּוּכַל לְהַשִּׂיג עֲבוּרִי בְּהֵמָה? חֲמוֹר שֶׁיּוּכַל לָשֵׂאת אֶת הַחֲבִילוֹת הַכְּבֵדוֹת שֶׁלִּי לַדֶּרֶךְ שֶׁלְּפָנַי?"

רַבִּי אֶלְעָזָר, שֶׁהָיָה בָּטוּחַ בְּכוֹחוֹ וְרָצָה לַעֲזֹר, חִיֵּךְ חִיּוּךְ רָחָב וְעָנָה בְּקוֹל בּוֹטֵחַ: "חֲמוֹר?! צְרִיכִים חֲמוֹר כְּשֶׁאֲנִי כָּאן? הַבֵּט נָא עָלַי, זָקֵן יָקָר. אֲנִי בְּעַצְמִי חָזָק כְּמוֹ שׁוֹר! עֲזֹב אֶת הַחֲמוֹרִים, אֲנִי אֶשָּׂא אֶת הַחֲבִילוֹת שֶׁלְּךָ, וְאִם תִּרְצֶה – אֶשָּׂא גַּם אוֹתְךָ הַרְחֵק עַד לְסוֹף הָעוֹלָם!".

הַזָּקֵן חִיֵּךְ חִיּוּךְ מִסְתּוֹרִי מִתַּחַת לִזְקָנוֹ. "בֶּאֱמֶת תַּסְכִּים לָשֵׂאת אוֹתִי?" שָׁאַל הַזָּקֵן.

"בְּוַדַּאי!" פָּסַק רַבִּי אֶלְעָזָר. "הַרְכֵּב אֶת הַחֲבִילוֹת עָלַי, וְגַם אַתָּה, עֲלֵה לִי עַל הַגַּב!".

הַמַּשָּׂא הַהוֹלֵךְ וְכָבֵד

וְכָךְ יָצְאוּ לַדֶּרֶךְ. רַבִּי אֶלְעָזָר נָשָׂא עַל גַּבּוֹ הָרָחָב אֶת כָּל הַחֲבִילוֹת, וְעַל כְּתֵפָיו אֶת הַזָּקֵן עַצְמוֹ. הֵם הִתְחִילוּ לָלֶכֶת.

בַּתְּחִלָּה, רַבִּי אֶלְעָזָר צָעַד בְּקַלּוּת. הֵם עָלוּ הָרִים גְּבוֹהִים, יָרְדוּ בִּבְקָעוֹת וַעֲמָקִים, עָבְרוּ בְּתוֹךְ שְׂדוֹת קוֹצִים שֶׁשָּׂרְטוּ אֶת הָרַגְלַיִם, וְרַבִּי אֶלְעָזָר כִּמְעַט לֹא הִזִּיעַ. "אֵיזֶה כֹּחַ יֵשׁ לִי!" חָשַׁב רַבִּי אֶלְעָזָר בְּגַאֲוָה, אֲנִי הֲכִי חָזָק בָּעוֹלָם.

אַךְ לְפֶתַע, קָרָה דָּבָר מוּזָר מְאֹד. אֵלִיָּהוּ הַנָּבִיא, הַזָּקֵן שֶׁהָיָה עַל גַּבּוֹ, הֵחֵל לְהַכְבִּיד אֶת עַצְמוֹ. זֶה הָיָה נֵס! עִם כָּל צַעַד שֶׁלֹּא הָלְכוּ, הַזָּקֵן שָׁקַל יוֹתֵר וְיוֹתֵר. מִמִּשְׁקָל שֶׁל אָדָם רָגִיל, הוּא הָפַךְ לִשְׁקֹל כְּמוֹ אֶבֶן רֵחַיִם כְּבֵדָה, וְאָז כְּמוֹ סֶלַע עָצוּם, וְאָז כְּמוֹ הַר שָׁלֵם!

רַבִּי אֶלְעָזָר הִתְחִיל לְהִתְאַמֵּץ. הַשְּׁרִירִים שֶׁלּוֹ רָעֲדוּ מִמַּאֲמָץ. נְשִׁימוֹתָיו הָפְכוּ כְּבֵדוֹת – כָּא... חָא... כָּא... אַגְלֵי זֵעָה נוֹצְצִים גָּלְשׁוּ עַל מִצְחוֹ וְנָטְפוּ עַל הָאֲדָמָה הַיְּבֵשָׁה. הָרַגְלַיִם שֶׁלּוֹ הִתְחִילוּ לִשְׁקֹעַ בְּתוֹךְ עֲפַר הַדֶּרֶךְ מֵרֹב כֹּבֶד הַמַּשָּׂא. הוּא הִרְגִּישׁ שֶׁהוּא כִּמְעַט מִתְעַלֵּף!

"סָבָּא!" זָעַק רַבִּי אֶלְעָזָר מִתּוֹךְ מַאֲמָץ, "הָקֵל אֶת עַצְמְךָ! לָמָּה אַתָּה כָּל כָּךְ כָּבֵד?! אִם לֹא תַּעֲשֶׂה עַצְמְךָ קַל יוֹתֵר, אֲנִי מַשְׁלִיךְ אוֹתְךָ כָּאן עַל הָאֲדָמָה!".

הַזָּקֵן שָׁתַק מְעַט, וְאָז שָׁאַל בְּקוֹל רַךְ וְשָׁקֵט: "הַאִם בִּרְצוֹנְךָ לָנוּחַ מְעַט? לָפוּשׁ וְלִצְבֹּר כֹּחַ?".

"כֵּן!" הִתְנַשֵּׁף רַבִּי אֶלְעָזָר.

הַזָּקֵן כִּוֵּן אוֹתוֹ לְשָׂדֶה שָׁקֵט שֶׁלְּצַד הַדֶּרֶךְ. שָׁם, מִתַּחַת לְעֵץ גָּדוֹל שֶׁפָּרַשׂ עֲנָפִים רְחָבִים מְלֵאִים בְּצֵל טוֹב וְקָרִיר, הִנִּיחַ רַבִּי אֶלְעָזָר אֶת הַזָּקֵן עַל קַרְקַע הָעֵשֶׂב הָרַכָּה. הַזָּקֵן הוֹצִיא מִתַּרְמִילוֹ לֶחֶם רַךְ וּמַיִם קָרִים, וְנָתַן לְרַבִּי אֶלְעָזָר הֶעָיֵף לֶאֱכֹל וְלִשְׁתּוֹת. הַמַּיִם הַקָּרִים הֶחֱזִירוּ לְרַבִּי אֶלְעָזָר מְעַט מִכּוֹחוֹ, וְהוּא יָשַׁב לָנוּחַ.

הַהִתְעוֹרְרוּת – מָה הוּא הַכֹּחַ הָאֲמִתִּי?

וְאָז, הִבִּיט בּוֹ הַזָּקֵן, וּבְקוֹל שֶׁחָדַר עָמֹק לְתוֹךְ הַנְּשָׁמָה שֶׁלּוֹ, אָמַר:

"רַבִּי אֶלְעָזָר... כָּל הַטֵּרוּף הַזֶּה – לָמָּה לְךָ? לָמָּה אַתָּה מְבַזְבֵּז אֶת מַתְּנַת הַכֹּחַ הָאֲדִירָה שֶׁה' נָתַן לְךָ, עַל סְחִיבַת מַשָּׂאוֹת וְרִיצוֹת אַחֲרֵי שׁוֹדְדִים עֲבוּר הַמֶּלֶךְ? הַאִם לֹא מוּטָב לְךָ לַחְזֹר לָאֻמָּנוּת שֶׁל אַבָּא שֶׁלְּךָ – רַבִּי שִׁמְעוֹן בַּר יוֹחַאי?"

"וּמָה הִיא הָאֻמָּנוּת שֶׁלּוֹ?" שָׁאַל רַבִּי אֶלְעָזָר, אַף שֶׁהֵבִין בְּלִבּוֹ הֵיטֵב.

"הִיא הַתּוֹרָה הַקְּדוֹשָׁה!" לָחַשׁ הַזָּקֵן. "זֶהוּ הַכֹּחַ הָאֲמִתִּי שֶׁל יְהוּדִי!".

הַמִּלִּים הָאֵלֶּה כָּהוּ כְּבָרָק בְּלִבּוֹ שֶׁל רַבִּי אֶלְעָזָר. הוּא נִזְכַּר בְּמִלִּים הַקְּדוֹשׁוֹת בַּמְּעָרָה, נִזְכַּר בְּאַבָּא שֶׁלּוֹ שֶׁלּוֹמֵד יָמִים וְלֵילוֹת. הֲרֵי הוּא צָרִיךְ לִהְיוֹת "תּוֹפֵס מִשְׁנָיוֹת" וְלֹא "תּוֹפֵס שׁוֹדְדִים"!

"וְ... וְיֵשׁ בְּךָ אֶת הַיְּכֹלֶת לְלַמֵּד אוֹתִי?" שָׁאַל רַבִּי אֶלְעָזָר אֶת הַזָּקֵן.

"כֵּן!" עָנָה אֵלִיָּהוּ הַנָּבִיא בִּמְאוֹר פָּנִים.

הַשְּׁרִירִים הַנֶּעֱלָמִים וְהַתּוֹרָה הַמְּאִירָה

מֵאוֹתוֹ יוֹם, הַכֹּל שְׁתַּנָּה. רַבִּי אֶלְעָזָר בְּרַבִּי שִׁמְעוֹן עָזַב אֶת עֲבוֹדַת הַמֶּלֶךְ. הוּא הִסְתַּגֵּר עִם אֵלִיָּהוּ הַנָּבִיא לְמֶשֶׁךְ שְׁלוֹשׁ עֶשְׂרֵה שָׁנִים רְצוּפוֹת! שְׁלוֹשׁ עֶשְׂרֵה שָׁנִים בָּהֶן הַפֶּה לֹא פָּסַק לְשַׁנֵּן אֶת פְּסוּקֵי סֵפֶר "וַיִּקְרָא" (הַנִּקְרָא "תּוֹרַת כֹּהֲנִים" אוֹ "סִיפְרָא"). קוֹל תּוֹרָה מָתוֹק מִדְּבַשׁ נִשְׁמַע יוֹמָם וָלַיְלָה.

יְלָדִים, אַתֶּם יוֹדְעִים מָה קָרָה לַגּוּף שֶׁלּוֹ בְּמֶשֶׁךְ הַשָּׁנִים הָאֵלֶּה?

כְּכָל שֶׁרַבִּי אֶלְעָזָר הִתְחַזֵּק בְּכוֹחוֹ הָרוּחָנִי – כָּךְ יָרַד כּוֹחוֹ הַגּוּפָנִי. כַּאֲשֶׁר סִיֵּם לִלְמֹד אֶת כָּל סֵפֶר הַ"סִיפְרָא" וְהָפַךְ לְתַלְמִיד חָכָם עָצוּם, הָיָה גּוּפוֹ חַלָּשׁ וְשָׁבִיר. עַד כְּדֵי כָּךְ שֶׁכְּשֶׁהָיָה צָרִיךְ לָשִׂים אֶת מְעִילוֹ עַל כְּתֵפָיו – לֹא הָיָה בּוֹ כֹּחַ כְּדֵי לָשֵׂאת אוֹתוֹ!

מִי שֶׁהָיָה יָכוֹל פַּעַם לָשֵׂאת זָקֵן עִם עֲשָׂרוֹת חֲבִילוֹת כְּבֵדוֹת עַל הַגַּב מֵעַל הָרִים וַעֲמָקִים, לֹא הָיָה מְסֻגָּל עַכְשָׁו לְהָרִים אֶת הַמְּעִיל שֶׁלּוֹ!

אֲבָל... הַאִם רַבִּי אֶלְעָזָר הָיָה עָצוּב בִּגְלַל שֶׁאִבֵּד אֶת כּוֹחוֹ?

הַהֵפֶךְ הַגָּמוּר! פָּנָיו זָרְחוּ בְּאוֹר שֶׁל קְדֻשָּׁה, כִּי עַכְשָׁו הוּא יָדַע: הַכֹּחַ שֶׁל הַגּוּף יָכוֹל לָשֵׂאת חֲבִילוֹת – אֲבָל הַכֹּחַ שֶׁל הַתּוֹרָה יָכוֹל לָשֵׂאת אֶת הָעוֹלָם כֻּלּוֹ! הַכֹּחַ שֶׁלּוֹ נִשְׁאַר אֶצְלוֹ, אַךְ עַכְשָׁו הוּא שָׁכַן כֻּלּוֹ בְּתוֹךְ הַשֵּׂכֶל, בְּתוֹךְ הַלֵּב וּבְתוֹךְ הַנְּשָׁמָה. הָאִישׁ הַחָזָק בְּיוֹתֵר, הָפַךְ לְעֲנַק הָעֲנָקִים שֶׁל עַם יִשְׂרָאֵל.

א. שְׁאֵלַת פְּתִיחָה (לְפְנֵי הַסִּפּוּר):

"בָּנִים, תְּנִיפוּ רֶגַע שְׁרִירִים. אֵיזֶה כֹּחַ! אֲבָל תַּגִּידוּ לִי, מָה עוֹשִׂים עִם שְׁרִירִים כָּאֵלֶּה חֲזָקִים? בְּמָה שָׁוֶה בֶּאֱמֶת לְהִשְׁתַּמֵּשׁ בַּכֹּחַ שֶׁלָּנוּ? הַיּוֹם נִפְגֹּשׁ אִישׁ שֶׁהָיָה לוֹ כֹּחַ שֶׁל גִּבּוֹרִים, אֲבָל הוּא גִּלָּה מָה הוּא הַכֹּחַ הֲכִי חָזָק בָּעוֹלָם."

ב. הֶעָרוֹת בִּצּוּעַ וּסְטוֹרִיטֶלִינְג:

ג. הַמְלָצוֹת לַמַּצֶּגֶת (4 שְׁקוּפִיּוֹת בִּקְדֻשָּׁה):

ד. פְּעִילוּת מְסַכֶּמֶת ("מִבְחַן הַכּוֹחוֹת"):

מִשְׂחָק קָצָר לַכִּתָּה: הַמּוֹרֶה יְבַקֵּשׁ מִשְּׁנֵי יְלָדִים חֲזָקִים לָבוֹא מִקָּדִימָה. יִתֵּן לָהֶם לְהַחְזִיק עֲרֵמַת סְפָרִים כְּבֵדָה בַּיָּדַיִם מְתוּחוֹת (מִבְחַן שְׁרִיר). לְאַחַר 10 שְׁנִיּוֹת אֶפְשָׁר לְהוֹרִיד. אָז שׁוֹאֵל הַמּוֹרֶה תַּלְמִיד אַחֵר: "אֵיזֶה פָּסוּק אוֹ מִשְׁנָה אַתָּה יוֹדֵעַ בְּעַל פֶּה?" כְּשֶׁהוּא אוֹמֵר, הַמּוֹרֶה יַסְבִּיר לַכִּתָּה: "הַיָּדַיִם נִתְפְּסוּ אַחֲרֵי חֲצִי דַּקָּה, אֲבָל הַפָּסוּק בְּעַל פֶּה שֶׁל הַיֶּלֶד... יִשָּׁאֵר אִתּוֹ גַּם עוֹד מֵאָה שָׁנִים. זֶה מָה שֶׁרַבִּי אֶלְעָזָר הֵבִין!"

מקורות: פסיקתא דרב כהנא: פרשה י"א (ויהי בשלח, פיסקא "ולא נחם אלקים"). שם מובא המעשה באליהו ז"ל שבא בדמות זקן, ביקש מרבי אלעזר בהמה, הרכיבו ועלה עמו הרים ובקעות עד שהכביד עליו, ולימדו את ספר תורת כהנים ("סיפרא") למשך 13 שנה עד שלא היה בו כוח אפילו לשאת את מעילו. עקרונות חינוכיים בסיפור צדיקים (שילוב מידות חסידות, ענווה ויגיעת תורה על פני כוח פיזי - כפי שמובא בעלון פרשת השבוע "סיפורי צדיקים" שהוזכר במאגרי התוכן).

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “להלן עיבוד חינוכי, רגשי ומרתק למדרש המופלא על רבי אלעזר בן רבי שמעון וחזרתו ללימוד התורה .” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “הַיָּדַיִם הָאֵלֶּה צְרִיכוֹת לְהַחְזִיק תּוֹרָה!" אָמַר כִּבְיָכוֹל הַקָּדוֹשׁ בָּרוּךְ הוּא, וְשָׁלַח אֵלָיו אוֹרֵחַ מְיֻחָד בְּמִינוֹ.”?
ג. כיצד הגיב ר' אלעזר בן רבי שמעון (בר יוחאי) למצב המתואר: “הַרְכֵּב אֶת הַחֲבִילוֹת עָלַי, וְגַם אַתָּה, עֲלֵה לִי עַל הַגַּב!".”?
ד. מה השתנה בין “סָבָּא!" זָעַק רַבִּי אֶלְעָזָר מִתּוֹךְ מַאֲמָץ, "הָקֵל אֶת עַצְמְךָ!” לבין “הֲרֵי הוּא צָרִיךְ לִהְיוֹת "תּוֹפֵס מִשְׁנָיוֹת" וְלֹא "תּוֹפֵס שׁוֹדְדִים"!”?
ה. כיצד הגיע הסיפור אל האירוע: “הַכֹּחַ שֶׁלּוֹ נִשְׁאַר אֶצְלוֹ, אַךְ עַכְשָׁו הוּא שָׁכַן כֻּלּוֹ בְּתוֹךְ הַשֵּׂכֶל, בְּתוֹךְ הַלֵּב וּבְתוֹךְ הַנְּשָׁמָה.”?

3. מה היה ראשון?

כתבו ליד כל אירוע מספר מ־1 עד 5 לפי סדר התרחשותו בסיפור.

□ אַגְלֵי זֵעָה נוֹצְצִים גָּלְשׁוּ עַל מִצְחוֹ וְנָטְפוּ עַל הָאֲדָמָה הַיְּבֵשָׁה.
□ להלן עיבוד חינוכי, רגשי ומרתק למדרש המופלא על רבי אלעזר בן רבי שמעון וחזרתו ללימוד התורה .
□ יִתֵּן לָהֶם לְהַחְזִיק עֲרֵמַת סְפָרִים כְּבֵדָה בַּיָּדַיִם מְתוּחוֹת (מִבְחַן שְׁרִיר).
□ הַזָּקֵן נִשְׁעַן עַל מַקֵּל עֵץ, פָּנָיו הָיוּ חֲרוּשׁוֹת קְמָטִים, וְזָקָן לָבָן אָרֹךְ עִטֵּר אֶת פָּנָיו.
□ רַבִּי אֶלְעָזָר בְּרַבִּי שִׁמְעוֹן עָזַב אֶת עֲבוֹדַת הַמֶּלֶךְ.

לפני ואחרי

מה קרה לפני: “אַגְלֵי זֵעָה נוֹצְצִים גָּלְשׁוּ עַל מִצְחוֹ וְנָטְפוּ עַל הָאֲדָמָה הַיְּבֵשָׁה.”?
מה קרה אחרי: “רַבִּי אֶלְעָזָר בְּרַבִּי שִׁמְעוֹן עָזַב אֶת עֲבוֹדַת הַמֶּלֶךְ.”?

סיבה ותוצאה מתוך המעשה

האירוע הנתוןמה גרם לו?מה הייתה תוצאתו?
לְרַגְלָיו שֶׁל הַזָּקֵן הָיוּ מֻנָּחוֹת כַּמָּה חֲבִילוֹת כְּבֵדוֹת.
הַמַּיִם הַקָּרִים הֶחֱזִירוּ לְרַבִּי אֶלְעָזָר מְעַט מִכּוֹחוֹ, וְהוּא יָשַׁב לָנוּחַ.

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “החמורים וחזרתו ללימוד” על ר' אלעזר בן רבי שמעון (בר יוחאי)? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “אֵלִיָּהוּ הַנָּבִיא, הַזָּקֵן שֶׁהָיָה עַל גַּבּוֹ, הֵחֵל לְהַכְבִּיד אֶת עַצְמוֹ.”.
ג. אילו ר' אלעזר בן רבי שמעון (בר יוחאי) לא היה פועל כפי שמתואר ב־“הֲרֵי הוּא צָרִיךְ לִהְיוֹת "תּוֹפֵס מִשְׁנָיוֹת" וְלֹא "תּוֹפֵס שׁוֹדְדִים"!” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “הַכֹּחַ שֶׁלּוֹ נִשְׁאַר אֶצְלוֹ, אַךְ עַכְשָׁו הוּא שָׁכַן כֻּלּוֹ בְּתוֹךְ הַשֵּׂכֶל, בְּתוֹךְ הַלֵּב וּבְתוֹךְ הַנְּשָׁמָה.” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. להלן עיבוד חינוכי, רגשי ומרתק למדרש המופלא על רבי אלעזר בן רבי שמעון…
2. הַיָּדַיִם הָאֵלֶּה צְרִיכוֹת לְהַחְזִיק תּוֹרָה!" אָמַר כִּבְיָכוֹל…
3. הֵם עָלוּ הָרִים גְּבוֹהִים, יָרְדוּ בִּבְקָעוֹת וַעֲמָקִים, עָבְרוּ…
4. שָׁם, מִתַּחַת לְעֵץ גָּדוֹל שֶׁפָּרַשׂ עֲנָפִים רְחָבִים מְלֵאִים…
5. כְּכָל שֶׁרַבִּי אֶלְעָזָר הִתְחַזֵּק בְּכוֹחוֹ הָרוּחָנִי – כָּךְ…
6. יִתֵּן לָהֶם לְהַחְזִיק עֲרֵמַת סְפָרִים כְּבֵדָה בַּיָּדַיִם מְתוּחוֹת…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “הַרְכֵּב אֶת הַחֲבִילוֹת עָלַי, וְגַם אַתָּה, עֲלֵה לִי עַל הַגַּב!".”.
מכתב: כתבו לר' אלעזר בן רבי שמעון (בר יוחאי) בעקבות “הַמַּיִם הַקָּרִים הֶחֱזִירוּ לְרַבִּי אֶלְעָזָר מְעַט מִכּוֹחוֹ, וְהוּא יָשַׁב לָנוּחַ.”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “כְּכָל שֶׁרַבִּי אֶלְעָזָר הִתְחַזֵּק בְּכוֹחוֹ הָרוּחָנִי – כָּךְ יָרַד כּוֹחוֹ הַגּוּפָנִי.”.
דיווח: כתבו ידיעה קצרה שכותרתה “החמורים וחזרתו ללימוד”.

המחזה

רגע מן הסיפורהדמויותמה יאמרו או יעשו?
לְרַגְלָיו שֶׁל הַזָּקֵן הָיוּ מֻנָּחוֹת כַּמָּה חֲבִילוֹת כְּבֵדוֹת.
הֲרֵי הוּא צָרִיךְ לִהְיוֹת "תּוֹפֵס מִשְׁנָיוֹת" וְלֹא "תּוֹפֵס שׁוֹדְדִים"!
