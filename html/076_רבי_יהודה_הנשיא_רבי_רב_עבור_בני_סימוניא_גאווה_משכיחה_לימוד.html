רב עבור בני סימוניא — גאווה משכיחה לימוד

דף עבודה בעקבות רבי יהודה הנשיא (רבי)
שם: כיתה:

המעשה — הסיפור המלא

פְּתִיחָה: בַּקָּשָׁה מֵרַבֵּנוּ הַקָּדוֹשׁ

יְלָדִים, בּוֹאוּ נִכָּנֵס יַחַד לְבֵית הַמִּדְרָשׁ הַגָּדוֹל שֶׁל רַבִּי יְהוּדָה הַנָּשִׂיא. הָאֲוִיר בַּחֶדֶר הָיָה מָלֵא בְּקוֹלוֹת שֶׁל מֵאוֹת תַּלְמִידֵי חֲכָמִים, הַמִּתְוַכְּחִים בְּהִתְלַהֲבוּת בְּהִלְכוֹת שַׁבָּת, כַּשְׁרוּת וּתְפִלָּה. רֵיחַ שֶׁל קְלָפִים יְשָׁנִים וּדְיוֹ טָרִי עָמַד בָּאֲוִיר. בְּרֹאשׁ הַשֻּׁלְחָן יָשַׁב רַבִּי יְהוּדָה הַנָּשִׂיא. פָּנָיו הָיוּ מְאִירוֹת בְּאוֹר שֶׁל קְדֻשָּׁה וְטָהֳרָה. לֹא לְחִנָּם קָרְאוּ לוֹ כֻּלָּם "רַבֵּנוּ הַקָּדוֹשׁ".

יוֹם אֶחָד, הִגִּיעָה לְבֵית הַמִּדְרָשׁ מִשְׁלַחַת שֶׁל יְהוּדִים עֲיֵפִים. מַגְפֵיהֶם הָיוּ מְכֻסִּים בְּאָבָק לָבָן שֶׁל דַּרְכֵי עָפָר. הֵם הִגִּיעוּ מֵעִיר הַנִּקְרֵאת "סִימוֹנְיָא". הֵם נִגְּשׁוּ בְּיִרְאַת כָּבוֹד אֶל נְשִׂיא יִשְׂרָאֵל, הִרְכִּינוּ אֶת רֹאשָׁם וְאָמְרוּ:

"יְלַמְּדֵנוּ רַבֵּנוּ... הָעִיר שֶׁלָּנוּ גְּדֵלָה, אֲבָל אֵין לָנוּ מַנְהִיג רוּחָנִי. אָנוּ מְבַקְשִׁים שֶׁהָרַב יִשְׁלַח לָנוּ אָדָם גָּדוֹל. אָדָם שֶׁיִּהְיֶה לָנוּ גַּם רַב הַפּוֹסֵק הֲלָכוֹת, גַּם דַּיָּן שֶׁשּׁוֹפֵט צֶדֶק, וְגַם תַּלְמִיד חָכָם שֶׁיְּלַמֵּד אֶת הַיְלָדִים מִקְרָא וּמִשְׁנָה".

רַבִּי יְהוּדָה הַנָּשִׂיא הִבִּיט בָּהֶם בְּאַהֲבָה. הוּא הֶעֱבִיר אֶת מַבָּטוֹ עַל מֵאוֹת תַּלְמִידָיו, מְחַפֵּשׂ אֶת הָאָדָם הַמַּתְאִים בְּיוֹתֵר לַמְּשִׂימָה הַחֲשׁוּבָה. מַבָּטוֹ נֶעֱצַר עַל תַּלְמִיד אָהוּב וּמְיֻחָד שֶׁקָּרְאוּ לוֹ לֵוִי בַּר סִיסִי . לֵוִי הָיָה צַדִּיק יְסוֹד עוֹלָם, תַּלְמִיד חָכָם עָצוּם שֶׁיָּדַע אֶת כָּל הַתּוֹרָה כֻּלָּהּ.

"אֶשְׁלַח לָכֶם כְּבָר עַכְשָׁו אָדָם גָּדוֹל שֶׁכָּמוֹנִי!" אָמַר רַבִּי יְהוּדָה לְאַנְשֵׁי סִימוֹנְיָא, "אֶת לֵוִי בַּר סִיסִי. שִׁמְרוּ עָלָיו וְכַבְּדוּ אוֹתוֹ".

בִּימָה שֶׁל עֵץ טָרִי וְכָבוֹד גָּדוֹל

אַנְשֵׁי סִימוֹנְיָא שָׂמְחוּ שִׂמְחָה גְּדוֹלָה. הֵם חָזְרוּ לְעִירָם וְהִכְרִיזוּ בָּרְחוֹבוֹת: "תִּתְכּוֹנְנוּ! רַבֵּנוּ הַקָּדוֹשׁ שׁוֹלֵחַ לָנוּ גְּאוֹן עוֹלָם! צָרִיךְ לְקַבֵּל אוֹתוֹ בְּכָבוֹד שֶׁל מְלָכִים!".

כַּאֲשֶׁר נִכְנַס לֵוִי בַּר סִיסִי לְשַׁעֲרֵי הָעִיר, כָּל הַתּוֹשָׁבִים יָצְאוּ לִקְרָאתוֹ מְחָא כַּפַּיִם וְשָׁרוּ שִׁירִים שֶׁל שִׂמְחָה. הֵם הוֹבִילוּ אוֹתוֹ אֶל בֵּית הַכְּנֶסֶת הַגָּדוֹל בְּיוֹתֵר בָּעִיר.

אֲבָל כְּשֶׁנִּכְנַס לֵוִי לְבֵית הַכְּנֶסֶת, הוּא נִדְהַם לְגַמְרֵי.

בְּמֶרְכַּז בֵּית הַכְּנֶסֶת בָּנוּ עֲבוּרוֹ בִּימָה גְּבוֹהָה עֲשׂוּיָה מֵעֵץ מְגֻלָּף הֵיטֵב. רֵיחַ שֶׁל עֵץ אֹרֶן טָרִי וְנֶסֶרֶת הִתְעַרְבֵּב עִם רֵיחַ הַבְּשָׂמִים. זוֹ לֹא הָיְתָה סְתָם בָּמָה, אֶלָּא מִבְנֶה אָרֹךְ עִם מַדְרֵגוֹת, וּבְרֹאשָׁהּ הוּנַח כִּסֵּא מְפֹאָר עִם כָּרִיּוֹת רַכּוֹת מִקְּטִיפָה אֲדֻמָּה.

"בְּבַקָּשָׁה, מוֹרֵנוּ וְרַבֵּנוּ," אָמְרוּ לוֹ פַּרְנְסֵי הָעִיר (מַנְהִיגֵי הָעִיר), "זֶהוּ מְקוֹמְךָ!".

לֵוִי בַּר סִיסִי הֵחֵל לַעֲלוֹת בַּמַּדְרֵגוֹת. מַדְרֵגָה אַחַר מַדְרֵגָה. הוּא הִתְיַשֵּׁב עַל הַכִּסֵּא וְהִבִּיט לְמַטָּה. מֵאוֹת אֲנָשִׁים – עֲשִׁירִים וַעֲנִיִּים, מְבֻגָּרִים וִילָדִים – עָמְדוּ וְהִסְתַּכְּלוּ עָלָיו, עֵינֵיהֶם בּוֹרְקוֹת בְּהַעֲרָצָה.

וּבְאוֹתוֹ רֶגַע... קָרָה מַשֶּׁהוּ קָטָן בְּלִבּוֹ שֶׁל לֵוִי.

כְּשֶׁהוּא יָשַׁב שָׁם לְמַעְלָה בִּמְרוֹמֵי הַבִּימָה וְרָאָה אֶת כֻּלָּם כָּל כָּךְ לְמַטָּה, נִכְנְסָה לְלִבּוֹ טִפָּה קְטַנְטַנָּה שֶׁל תְּחוּשַׁת חֲשִׁיבוּת גְּדוֹלָה וְגַאֲוָה. הוּא חָשַׁב בְּלִבּוֹ עַל הַכָּבוֹד הָעָצוּם שֶׁחָלְקוּ לוֹ.

שְׁתִיקָה מַחֲרִישַׁת אָזְנַיִם

אַנְשֵׁי סִימוֹנְיָא חִכּוּ לִשְׁמֹעַ דִּבְרֵי תּוֹרָה וְחָכְמָה. אֶחָד מֵהָאֲנָשִׁים הַמְבֻגָּרִים קָם וְשָׁאַל בְּקוֹל רוֹעֵד מֵהִתְרַגְּשׁוּת:

"רַבִּי, יֵשׁ לָנוּ שְׁאֵלָה בַּהֲלָכָה שֶׁמַּעֲסִיקָה אוֹתָנוּ (שְׁאֵלָה עַל דִּינֵי חֲלִיצָה, מַה הַדִּין אִם לָאִשָּׁה חָסֵר אֶצְבַּע אוֹ יָד, הֲרֵי לְפִי הַהֲלָכָה הִיא צְרִיכָה לַחְלֹץ אֶת הַנַּעַל מֵרַגְלוֹ שֶׁל אֲחִי בַּעֲלָהּ)... תּוּכַל לִפְסֹק לָנוּ מַה הַהֲלָכָה בְּמִקְרֶה כָּזֶה?"

לֵוִי בַּר סִיסִי חִיֵּךְ. שְׁאֵלָה קַלָּה. הוּא לָמַד אוֹתָהּ עַשְׂרוֹת פְּעָמִים אֵצֶל רַבִּי יְהוּדָה הַנָּשִׂיא! הוּא פָּתַח אֶת פִּיו כְּדֵי לַעֲנוֹת, אַךְ פִּתְאוֹם... שֶׁקֶט.

הַמִּלִּים פָּשׁוּט נֶעֶלְמוּ. מוֹחוֹ נַעֲשָׂה רֵיק מִלִּמּוּד. הוּא אִמֵּץ אֶת זִכְרוֹנוֹ, כִּוֵּץ אֶת מִצְחוֹ, אֲבָל הַתְּשׁוּבָה לֹא בָּאָה. טִפּוֹת קְטַנּוֹת שֶׁל זֵעָה קָרָה הֵחֵלּוּ לָרֶדֶת עַל מִצְחוֹ. לֵב שֶׁלּוֹ דָּפַק בְּחָזְקָה.

"אֲנִי... אֲנִי אוּכַל לַעֲנוֹת עַל זֶה מְאֻחָר יוֹתֵר," הוּא לָחַשׁ סוֹף סוֹף, קוֹלוֹ חַלָּשׁ וְרוֹעֵד.

אַנְשֵׁי הָעִיר הִבִּיטוּ זֶה בָּזֶה מֻפְתָּעִים. אֲבָל אִישׁ שֵׁנִי קָם וְשָׁאַל שְׁאֵלָה אַחֶרֶת. כָּל בֵּית הַכְּנֶסֶת הָיָה בְּמֶתַח. אֲבָל לֵוִי יָשַׁב עַל כִּסְאוֹ שֶׁבִּמְרוֹמֵי הַבִּימָה וּפָשׁוּט שָׁתַק. שְׁתִיקָה עֲצוּבָה שֶׁל אָדָם שֶׁשָּׁכַח אֶת כָּל תַּלְמוּדוֹ. מֵרֹב הַגַּאֲוָה שֶׁנִּכְנְסָה לְלִבּוֹ – הַתּוֹרָה פָּרְחָה וּבָרְחָה מִמֶּנּוּ!

אַנְשֵׁי הָעִיר סִימוֹנְיָא הִרְגִּישׁוּ אַכְזָבָה מְרִירָה. הֵם לָחֲשׁוּ בֵּינֵיהֶם: "זֶה הָאָדָם הַגָּדוֹל שֶׁרַבֵּנוּ שָׁלַח לָנוּ? הוּא אֲפִלּוּ לֹא יוֹדֵעַ לַעֲנוֹת עַל שְׁאֵלוֹת פְּשׁוּטוֹת בַּהֲלָכָה!". הֵם חָשְׁבוּ שֶׁהוּא אָדָם שֶׁאֵינוֹ יְדַעְיָן כְּלָל.

הַשִּׁיבָה אֶל מְקוֹר הָעֲנָוָה

בְּלֵב שָׁבוּר, יָרַד לֵוִי בַּר סִיסִי מֵהַבִּימָה הַמְפֹאֶרֶת, לָקַח אֶת חֲפָצָיו וְשָׁב לְבֵית מִדְרָשׁוֹ שֶׁל רַבִּי יְהוּדָה הַנָּשִׂיא. גַּם הַמִּשְׁלַחַת מִסִּימוֹנְיָא עָשְׂתָה אֶת הַדֶּרֶךְ חֲזָרָה.

כַּאֲשֶׁר נִכְנְסוּ הָאֲנָשִׁים לַחֲדַר הַלִּמּוּד שֶׁל רַבִּי יְהוּדָה הַנָּשִׂיא, הֵם אָמְרוּ בְּצַעַר: "רַבֵּנוּ, זֶה הָאָדָם שֶׁשָּׁלַחְתָּ לָנוּ שֶׁהוּא גָּדוֹל כְּמוֹתְךָ? הוּא לֹא עָנָה אֲפִלּוּ עַל שְׁאֵלָה אַחַת שֶׁשָּׁאַלְנוּ אוֹתוֹ!".

רַבִּי יְהוּדָה לֹא מִהֵר לִכְעֹס. הוּא יָדַע מִי זֶה לֵוִי בַּר סִיסִי. הוּא הִבִּיט בְּלֵוִי, שֶׁעָמַד שָׁם, בְּפִנַּת הַחֶדֶר, עֵינָיו בָּרִצְפָּה מֵרֹב בּוּשָׁה.

"לֵוִי בְּנִי," קָרָא לוֹ רַבִּי הַנָּשִׂיא בְּחֹם, וְאָז שָׁאַל אוֹתוֹ אֶת אוֹתָהּ שְׁאֵלָה בְּדִיּוּק שֶׁשָּׁאֲלוּ אוֹתוֹ בְּסִימוֹנְיָא.

לְלֹא הִסּוּס קַל, מִיָּד, כְּמַעְיָן נוֹבֵעַ, מִלִּים שֶׁל תּוֹרָה זָרְמוּ מִפִּיו שֶׁל לֵוִי! הוּא עָנָה בְּפִירוּט רַב, הֵבִיא רְאָיוֹת וְנִמּוּקִים וּפָסַק אֶת הַהֲלָכָה הַמְדֻיֶּקֶת. הָאֲנָשִׁים מִסִּימוֹנְיָא עָמְדוּ פְּעוּרֵי פֶּה. הוּא הֲרֵי גָּאוֹן!

הַתְּשׁוּבָה שֶׁל לֵוִי: הַבִּימָה וְהַגַּאֲוָה

"אִם כָּךְ," שָׁאַל רַבִּי יְהוּדָה הַנָּשִׂיא בְּרַכּוּת, כְּשֶׁלִּטֵּף אֶת זְקָנוֹ הַלָּבָן, "מַדּוּעַ לֹא עָנִיתָ לָהֶם כַּאֲשֶׁר שָׁאֲלוּ אוֹתְךָ שָׁם בְּעִירָם?"

הִרְכִּין לֵוִי בַּר סִיסִי אֶת רֹאשׁוֹ. הוּא הָיָה אִישׁ שֶׁל אֱמֶת, וְהוּא יָדַע שֶׁהוּא חַיָּב לְהוֹדוֹת בְּחֻלְשָׁתוֹ בִּפְנֵי כֻּלָּם. הוּא לֹא הִמְצִיא תֵּרוּצִים, לֹא אָמַר שֶׁהָיָה כּוֹאֵב לוֹ הָרֹאשׁ וְלֹא הֶאֱשִׁים אִישׁ.

הוּא לָחַשׁ: "רַבִּי... עָשׂוּ לִי שָׁם בִּימָה גְּדוֹלָה כְּדֵי שֶׁאֵשֵׁב מֵעַל כֻּלָּם... כְּשֶׁהִתְיַשַּׁבְתִּי עָלֶיהָ, זָחָה דַּעְתִּי עָלַי! הִרְגַּשְׁתִּי בְּלִבִּי מְעַט תְּחוּשָׁה שֶׁל גַּאֲוָה עַל הַכָּבוֹד הַגָּדוֹל. וּבָרֶגַע שֶׁהַגַּאֲוָה נִכְנְסָה – שָׁכַחְתִּי תַּלְמוּדִי! הַתּוֹרָה שֶׁלִּי פּוֹרְחָה לָהּ וְנֶעֶלְמָה".

רַבִּי יְהוּדָה הַנָּשִׂיא הִסְתַּכֵּל עַל תַּלְמִידוֹ בְּאַהֲבָה רַבָּה עַל הַכֵּנוּת שֶׁלּוֹ וְאֲמִירַת הָאֱמֶת, וְאָז הִסְבִּיר לְכָל בֵּית הַמִּדְרָשׁ כְּלָל חָשׁוּב מְאֹד בַּחַיִּים.

"יְלָדִים, שִׂימוּ לֵב," הוּא אָמַר, "הַתּוֹרָה הַקְּדוֹשָׁה נִמְשְׁלָה בְּנָבִיא לְמַיִם. שֶׁנֶּאֱמַר: 'הוֹי כָּל צָמֵא לְכוּ לַמַּיִם'. וְלָמָּה לְמַיִם? כִּי מַיִם הֵם הַדָּבָר הַיָּחִיד בַּטֶּבַע שֶׁעוֹזֵב אֶת הַמָּקוֹם הַגָּבוֹהַּ, וְתָמִיד יוֹרֵד, נוֹזֵל וְזוֹרֵם אֶל הַמָּקוֹם הַנָּמוּךְ בְּיוֹתֵר. כָּךְ הִיא הַתּוֹרָה – הִיא לְעוֹלָם לֹא תִּשָּׁאֵר לְשָׁכֵן בְּלִבּוֹ שֶׁל אָדָם שֶׁהוּא "גָּבוֹהַּ", אָדָם שֶׁמִּתְגָּאֶה בְּחָכְמָתוֹ אוֹ יָשׁוּב עַל בִּימָה וְחוֹשֵׁב שֶׁהוּא טוֹב מֵאֲחֵרִים. הַתּוֹרָה תָּמִיד תִּמְצָא אֶת מְקוֹמָהּ בְּלֵב נָמוּךְ, בְּלִבּוֹ שֶׁל אָדָם עָנָו. גַּאֲוָה, תַּלְמִידַי הַיְקָרִים, מַשְׁכִּיחָה אֶת הַלִּמּוּד!".

לֵוִי בַּר סִיסִי קִבֵּל בְּהַכְנָעָה אֶת הַשִּׁעוּר, וּמֵאוֹתוֹ יוֹם, הוּא נִזְהַר שִׁבְעָתַיִם מִכָּל טִפָּה שֶׁל כָּבוֹד, וְחָזַר לִהְיוֹת אֶחָד מִגְּדוֹלֵי תַּלְמִידֵי הַחֲכָמִים הַזְּכוּרִים בְּתּוֹלְדוֹת עַם יִשְׂרָאֵל, שֶׁעַנְוְתָנוּתוֹ הִיא שֶׁשָּׁמְרָה עַל תּוֹרָתוֹ לָנֶצַח.

מקורות: תלמוד ירושלמי, מסכת יבמות, פרק י"ב הלכה ו'. מדרש בראשית רבה, פרשה פ"א. אזכור ב"ספר האגדה" של ביאליק ורבניצקי. "כה עשו חכמינו" (מאת יוכבד סגל), כרך ד', עמוד 142. (ראה מסמך מקורות שסופק ע"י המשתמש: "רבי יהודה הנשיא ותלמידו ר' לוי בר סיסי - רב עבור בני סימוניא. כע"ח ד 142. ירושלמי יבמות פי"ב ה"ו. בראשית רבה פרשה פ"א. דאגתו לתלמידיו. גאוה משכיחה את הלימוד").

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “יְלָדִים, בּוֹאוּ נִכָּנֵס יַחַד לְבֵית הַמִּדְרָשׁ הַגָּדוֹל שֶׁל רַבִּי יְהוּדָה הַנָּשִׂיא.” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “מַבָּטוֹ נֶעֱצַר עַל תַּלְמִיד אָהוּב וּמְיֻחָד שֶׁקָּרְאוּ לוֹ לֵוִי בַּר סִיסִי .”?
ג. כיצד הגיב רבי יהודה הנשיא (רבי) למצב המתואר: “זוֹ לֹא הָיְתָה סְתָם בָּמָה, אֶלָּא מִבְנֶה אָרֹךְ עִם מַדְרֵגוֹת, וּבְרֹאשָׁהּ הוּנַח כִּסֵּא מְפֹאָר עִם כָּרִיּוֹת רַכּוֹת מִקְּטִיפָה אֲדֻמָּה.”?
ד. מה השתנה בין “הוּא אִמֵּץ אֶת זִכְרוֹנוֹ, כִּוֵּץ אֶת מִצְחוֹ, אֲבָל הַתְּשׁוּבָה לֹא בָּאָה.” לבין “הוּא לֹא עָנָה אֲפִלּוּ עַל שְׁאֵלָה אַחַת שֶׁשָּׁאַלְנוּ אוֹתוֹ!".”?
ה. כיצד הגיע הסיפור אל האירוע: “רַבִּי יְהוּדָה הַנָּשִׂיא הִסְתַּכֵּל עַל תַּלְמִידוֹ בְּאַהֲבָה רַבָּה עַל הַכֵּנוּת שֶׁלּוֹ וְאֲמִירַת הָאֱמֶת, וְאָז הִסְבִּיר לְכָל בֵּית הַמִּדְרָשׁ…”?

3. מה היה ראשון?

כתבו ליד כל אירוע מספר מ־1 עד 5 לפי סדר התרחשותו בסיפור.

□ הוּא לָמַד אוֹתָהּ עַשְׂרוֹת פְּעָמִים אֵצֶל רַבִּי יְהוּדָה הַנָּשִׂיא!
□ יְלָדִים, בּוֹאוּ נִכָּנֵס יַחַד לְבֵית הַמִּדְרָשׁ הַגָּדוֹל שֶׁל רַבִּי יְהוּדָה הַנָּשִׂיא.
□ הַתּוֹרָה תָּמִיד תִּמְצָא אֶת מְקוֹמָהּ בְּלֵב נָמוּךְ, בְּלִבּוֹ שֶׁל אָדָם עָנָו.
□ כַּאֲשֶׁר נִכְנַס לֵוִי בַּר סִיסִי לְשַׁעֲרֵי הָעִיר, כָּל הַתּוֹשָׁבִים יָצְאוּ לִקְרָאתוֹ מְחָא כַּפַּיִם וְשָׁרוּ שִׁירִים שֶׁל שִׂמְחָה.
□ הוּא הִבִּיט בְּלֵוִי, שֶׁעָמַד שָׁם, בְּפִנַּת הַחֶדֶר, עֵינָיו בָּרִצְפָּה מֵרֹב בּוּשָׁה.

לפני ואחרי

מה קרה לפני: “הוּא לָמַד אוֹתָהּ עַשְׂרוֹת פְּעָמִים אֵצֶל רַבִּי יְהוּדָה הַנָּשִׂיא!”?
מה קרה אחרי: “הוּא הִבִּיט בְּלֵוִי, שֶׁעָמַד שָׁם, בְּפִנַּת הַחֶדֶר, עֵינָיו בָּרִצְפָּה מֵרֹב בּוּשָׁה.”?

סיבה ותוצאה מתוך המעשה

האירוע הנתוןמה גרם לו?מה הייתה תוצאתו?
הֵם הוֹבִילוּ אוֹתוֹ אֶל בֵּית הַכְּנֶסֶת הַגָּדוֹל בְּיוֹתֵר בָּעִיר.
שְׁתִיקָה עֲצוּבָה שֶׁל אָדָם שֶׁשָּׁכַח אֶת כָּל תַּלְמוּדוֹ.

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “רב עבור בני סימוניא — גאווה משכיחה לימוד” על רבי יהודה הנשיא (רבי)? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “הוּא חָשַׁב בְּלִבּוֹ עַל הַכָּבוֹד הָעָצוּם שֶׁחָלְקוּ לוֹ.”.
ג. אילו רבי יהודה הנשיא (רבי) לא היה פועל כפי שמתואר ב־“הוּא לֹא עָנָה אֲפִלּוּ עַל שְׁאֵלָה אַחַת שֶׁשָּׁאַלְנוּ אוֹתוֹ!".” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “רַבִּי יְהוּדָה הַנָּשִׂיא הִסְתַּכֵּל עַל תַּלְמִידוֹ בְּאַהֲבָה רַבָּה עַל הַכֵּנוּת שֶׁלּוֹ וְאֲמִירַת הָאֱמֶת, וְאָז…” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. יְלָדִים, בּוֹאוּ נִכָּנֵס יַחַד לְבֵית הַמִּדְרָשׁ הַגָּדוֹל שֶׁל…
2. מַבָּטוֹ נֶעֱצַר עַל תַּלְמִיד אָהוּב וּמְיֻחָד שֶׁקָּרְאוּ לוֹ לֵוִי…
3. מֵאוֹת אֲנָשִׁים – עֲשִׁירִים וַעֲנִיִּים, מְבֻגָּרִים וִילָדִים –…
4. אֲנִי אוּכַל לַעֲנוֹת עַל זֶה מְאֻחָר יוֹתֵר," הוּא לָחַשׁ סוֹף סוֹף,…
5. הוּא עָנָה בְּפִירוּט רַב, הֵבִיא רְאָיוֹת וְנִמּוּקִים וּפָסַק אֶת…
6. הַתּוֹרָה תָּמִיד תִּמְצָא אֶת מְקוֹמָהּ בְּלֵב נָמוּךְ, בְּלִבּוֹ שֶׁל…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “זוֹ לֹא הָיְתָה סְתָם בָּמָה, אֶלָּא מִבְנֶה אָרֹךְ עִם מַדְרֵגוֹת, וּבְרֹאשָׁהּ הוּנַח כִּסֵּא מְפֹאָר עִם כָּרִיּוֹת רַכּוֹת…”.
מכתב: כתבו לרבי יהודה הנשיא (רבי) בעקבות “שְׁתִיקָה עֲצוּבָה שֶׁל אָדָם שֶׁשָּׁכַח אֶת כָּל תַּלְמוּדוֹ.”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “הוּא עָנָה בְּפִירוּט רַב, הֵבִיא רְאָיוֹת וְנִמּוּקִים וּפָסַק אֶת הַהֲלָכָה הַמְדֻיֶּקֶת.”.
