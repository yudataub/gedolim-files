כיצד שלח אנטונינוס זהב לרבי

דף עבודה בעקבות רבי יהודה הנשיא (רבי)
שם: כיתה:

המעשה — הסיפור המלא

להלן עיבוד חינוכי, עשיר ומרתק לסיפור המופלא על קשריו של רבי יהודה הנשיא (רבינו הקדוש) ואנטונינוס קיסר רומא , ובמיוחד על הדרך שבה בחר הקיסר לשלוח זהב לרבי. הסיפור מותאם לבנים בכיתות ב'-ד' (חינוך דתי-לאומי), תוך שימוש בעקרונות הסטוריטלינג שביקשת, והוא מנוקד במלואו.

פְּתִיחָה: שְׁנֵי עוֹלָמוֹת שׁוֹנִים

יְלָדִים, בּוֹאוּ נַעֲצֹם עֵינַיִם וְנַחְזֹר בַּזְּמַן כְּאַלְפַּיִם שָׁנָה אָחוֹרָה. דַּמְיְנוּ עִיר גְּדוֹלָה, הוֹמָה וְרוֹעֶשֶׁת. זוֹהִי רוֹמָא הַגְּדוֹלָה. מִסָּבִיב אַרְמוֹנוֹת עֲנָקִיִּים מִשַּׁיִשׁ לָבָן וְקַר, פְּסָלִים מִזָּהָב, וְקוֹל נְקִישַׁת חֲרָבוֹת הַבַּרְזֶל שֶׁל חַיָּלֵי רוֹמָא נִשְׁמָע בְּכָל פִּנָּה. רֵיחַ שֶׁל אֵשׁ, בָּשָׂר צָלוּי וְגַאֲוָה גְּדוֹלָה מְמַלֵּא אֶת הָאֲוִיר. בְּתוֹךְ הָאַרְמוֹן הַגָּדוֹל בְּיוֹתֵר יוֹשֵׁב הַקֵּיסָר אַנְטוֹנִינוּס, שַׁלִּיט הָעוֹלָם, אִישׁ שֶׁכֻּלָּם רוֹעֲדִים מִפָּנָיו.

וְעַכְשָׁו, בּוֹאוּ נָטוּס מִשָּׁם הַרְחֵק הַרְחֵק, אֶל אֶרֶץ יִשְׂרָאֵל. לָעִיר צִפּוֹרִי בַּגָּלִיל.

כָּאן שָׁקֵט. רֵיחַ עָדִין שֶׁל עֲצֵי זַיִת וְלֶחֶם טָרִי נִשָּׂא בָּרוּחַ. אֲבָל בְּתוֹךְ הַשֶּׁקֶט הַזֶּה, אֶפְשָׁר לִשְׁמֹעַ הֶמְיָה מְתוּקָה, קוֹל עָרֵב שֶׁל רְבָבוֹת תַּלְמִידִים לוֹמְדִים תּוֹרָה. קוֹלֵי קוֹלוֹת שֶׁל מִשְׁנָיוֹת! בְּרֹאשׁ הַיְּשִׁיבָה יוֹשֵׁב הַנָּשִׂיא שֶׁל עַם יִשְׂרָאֵל, הַגָּאוֹן וְהַצַּדִּיק – רַבִּי יְהוּדָה הַנָּשִׂיא , שֶׁכֻּלָּם קוֹרְאִים לוֹ פָּשׁוּט "רַבִּי" אוֹ "רַבֵּינוּ הַקָּדוֹשׁ". עֵינָיו שֶׁל רַבִּי מְאִירוֹת כְּכוֹכָבִים, וּפָנָיו קְדוֹשׁוֹת. הַקָּדוֹשׁ בָּרוּךְ הוּא נָתַן לְרַבִּי גְּדֻלָּה עֲצוּמָה, תּוֹרָה עֲמוּקָה, וְגַם עֹשֶׁר כַּבִּיר, כְּדֵי שֶׁיּוּכַל לְסַדֵּר לְעַם יִשְׂרָאֵל אֶת כָּל שִׁשָּׁה סִדְרֵי מִשְׁנָה בְּשֶׁקֶט וּבְשַׁלְוָה.

הַקֶּשֶׁר הַסּוֹדִי

וְהִנֵּה, דָּבָר פֶּלֶא קָרָה. מִי הָיָה אוֹהֲבוֹ הַגָּדוֹל בְּיוֹתֵר שֶׁל רַבִּי? לֹא אַחֵר מֵאֲשֶׁר אַנְטוֹנִינוּס הַקֵּיסָר! אַנְטוֹנִינוּס הֵבִין שֶׁהַכֹּחַ הָאֲמִתִּי בָּעוֹלָם הוּא לֹא הַחֲרָבוֹת וְהַבַּרְזֶל שֶׁל רוֹמָא, אֶלָּא הַתּוֹרָה הַקְּדוֹשָׁה שֶׁל הַיְּהוּדִים וְחָכְמָתוֹ הָאֱלֹקִית שֶׁל רַבִּי יְהוּדָה הַנָּשִׂיא.

פְּעָמִים רַבּוֹת, בְּאִישׁוֹן לַיְלָה מְאֻחָר, כְּשֶׁהַשּׁוֹמְרִים בָּאַרְמוֹן כְּבָר נִמְנְמוּ, הָיָה אַנְטוֹנִינוּס יוֹרֵד דֶּרֶךְ מִנְהָרָה סוֹדִית אֶל בֵּיתוֹ שֶׁל רַבִּי, וְלוֹמֵד מִמֶּנּוּ חָכְמָה. הוּא הָיָה רוֹאֶה מֵרָחוֹק אֶת פָּנָיו הַמְּאִירוֹת שֶׁל רַבִּי רוֹכְנוֹת עַל קְלָפֵי הַמִּשְׁנָה, וְלִבּוֹ שֶׁל הַקֵּיסָר הָיָה מִתְמַלֵּא בְּיִרְאַת כָּבוֹד. הַקֵּיסָר יָדַע שֶׁרַבִּי הוּא מְפַרְנֵס אֶת כָּל תַּלְמִידֵי הַחֲכָמִים שֶׁל עַם יִשְׂרָאֵל, שֶׁהוּא דּוֹאֵג לָעֲנִיִּים, לָאַלְמָנוֹת וְלַיְתוֹמִים שֶׁהָיוּ צְרִיכִים לֶחֶם לֶאֱכֹל.

הַמְּצִיקוּת בְּאַרְמוֹן הַקֵּיסָר

יוֹם אֶחָד, בְּחַדְרוֹ הַפְּרָטִי שֶׁל הַקֵּיסָר בְּרוֹמָא, פָּתַח אַנְטוֹנִינוּס אֶת תֵּבַת אוֹצָרוֹ הָאִישִׁי. מַטְבְּעוֹת תְּפוּחִים שֶׁל זָהָב טָהוֹר, שֶׁהוּבְאוּ מִקְּצוֹת הָעוֹלָם, נָצְצוּ בָּאוֹר. אַנְטוֹנִינוּס הֶחְלִיט בְּלִבּוֹ: "אֲנִי רוֹצֶה לִשְׁלֹחַ אֶת כָּל הַזָּהָב הַזֶּה לְרַבִּי יְהוּדָה הַנָּשִׂיא! שֶׁיִּבְנֶה בּוֹ עוֹד יְשִׁיבוֹת, שֶׁיְּחַלֵּק לָעֲנִיִּים, שֶׁהַתּוֹרָה שֶׁל הַיְּהוּדִים תִּפְרַח מִן הַזָּהָב הַזֶּה".

אֲבָל... הָיְתָה בְּעָיָה קָשָׁה.

אַנְטוֹנִינוּס הֵצִיץ מִבַּעַד לַדֶּלֶת הַכְּבֵדָה עֲשׂוּיַת הָעֵץ. מִבַּחוּץ עָמְדוּ הַשָּׂרִים שֶׁל רוֹמָא וְחַיָּלָיו הָאַכְזָרִיִּים. עֵינֵיהֶם תָּמִיד הָיוּ צָרוֹת וְרָעוֹת. הֵם שָׂנְאוּ אֶת הַיְּהוּדִים וְתָמִיד חִפְּשׂוּ אֵיךְ לְהַזִּיק לָהֶם.

הַקֵּיסָר חָשַׁב בְּדְאָגָה: "אִם יִרְאוּ הַשָּׂרִים שֶׁלִּי שֶׁאֲנִי מוֹצִיא שַׂקִּים מְלֵאִים בְּמַטְבְּעוֹת זָהָב וְשׁוֹלֵחַ לִיהוּדָה, לָרַב הַגָּדוֹל שֶׁל הַיְּהוּדִים... הֵם יִכְעֲסוּ מְאֹד. הֵם עֲלוּלִים לִשְׁדֹּד אֶת הַזָּהָב בַּדֶּרֶךְ! אֲנָשִׁים הֲרֵי אוֹהֲבִים כֶּסֶף. וְגָרוּעַ מִזֶּה, הֵם יָשׁוּבוּ וְיָסִיתוּ אוֹתִי נֶגֶד הַיְּהוּדִים, יַעֲלִילוּ עֲלֵיהֶם עֲלִילוֹת שֶׁקֶר כְּדֵי לְהַחֲרִיב לָהֶם אֶת הַמְּדִינָה, חָלִילָה. אֵיךְ אוּכַל לַעֲשׂוֹת זֹאת בַּסֵּתֶר?"

הַתַּחְבּוּלָה שֶׁל הַקֵּיסָר

אַנְטוֹנִינוּס יָשַׁב עַל כִּסֵּא הַמַּלְכוּת וְהִפְעִיל אֶת חָכְמָתוֹ. הוּא יָדַע שֶׁחַיָּבִים לִמְצֹא תַּחְבּוּלָה, מַשֶּׁהוּ שֶׁיֵּרָאֶה טִבְעִי לַחֲלוּטִין.

"מְשָׁרְתִים!" קָרָא הַקֵּיסָר. קוֹלוֹ הִדְהֵד בָּאַרְמוֹן.

הַשּׁוֹמְרִים רָצוּ פְּנִימָה וְהִשְׁתַּחֲווּ.

"הָבִיאוּ לִי לְכָאן מִיָּד עֲשָׂרוֹת שַׂקִּים רֵיקִים! וּמְלֹא הָעֲגָלָה בְּחִטָּה טוֹבָה וּמֻבְחֶרֶת!" פָּקַד הַקֵּיסָר.

הַמְּשָׁרְתִים הָיוּ מֻפְתָּעִים. חִטָּה? מָה לְקֵיסָר רוֹמָא שֶׁאוֹכֵל בְּשַׂר צְבָאִים וְשׁוֹתֶה יַיִן מְשֻׁבָּח, לְבַקֵּשׁ שַׂקֵּי חִטָּה הַמַּתְאִימִים לְלֶחֶם פָּשׁוּט? אֲבָל בְּרוֹמָא אֵין שׁוֹאֲלִים שְׁאֵלוֹת.

כְּשֶׁהוּבְאוּ הַשַּׂקִּים וְהַחִטָּה לְחַדְרוֹ, סִלֵּק הַקֵּיסָר אֶת כֻּלָּם וְנָעַל אֶת הַדֶּלֶת מִבִּפְנִים. עַכְשָׁו נִשְׁאֲרוּ בַּחֶדֶר רַק הַקֵּיסָר, תֵּבַת הַזָּהָב, וְשַׂקֵּי הַחִטָּה הָרֵיקִים.

בַּחֲשַׁאי, בְּדִמְמַת אֶלֶף דְּמָמוֹת, פָּתַח הַקֵּיסָר אֶת תֵּבַת הַזָּהָב. הַמַּטְבְּעוֹת הִשְׁמִיעוּ צִלְצוּל קַל: קְלִינְג... צְלִינְג... אֲבָל אַנְטוֹנִינוּס מִהֵר לְעַטֵּף אוֹתָם בְּבַד כְּדֵי שֶׁלֹּא יַרְעִישׁוּ. הוּא לָקַח שַׂק אֶחָד, וְהִכְנִיס לְקַרְקָעִיתוֹ (לַתַּחְתִּית) חֲפָנִים שֶׁל זָהָב אֲדַמְדַּם מַבְרִיק. לְאַחַר מִכֵּן, קָפַץ אֶת יָדָיו וּמִלֵּא מֵעַל הַזָּהָב עוֹד חִטָּה, וְעוֹד חִטָּה, עַד שֶׁהַזָּהָב נֶעֱלַם לְגַמְרֵי מִן הָעַיִן, וְכָל הַשַּׂק נִרְאָה מִבַּחוּץ כְּשַׂקָּהּ שֶׁל חִטָּה רְגִילָה הַנִּמְכֶּרֶת בַּשּׁוּק הַגָּדוֹל. מֵעַל הַכֹּל כָּתַב: "מִנְחַת חִטִּים עֲבוּר אֲהוּבִי, רַבִּי יְהוּדָה".

כָּךְ עָשָׂה אַנְטוֹנִינוּס בְּשֶׁקֶט עִם שַׂק שֵׁנִי, וּשְׁלִישִׁי וּרְבִיעִי... עַד שֶׁסִּיֵּם אֶת כָּל תֵּבַת הַזָּהָב הָעֲצוּמָה.

הוּא פָּתַח אֶת הַדֶּלֶת וְקָרָא לְשָׂרָיו: "קְחוּ מִיָּד אֶת הַשַּׂקִּים הַלָּלוּ שֶׁל הַחִטָּה, וְהוֹבִילוּ אוֹתָם עַד לְאַרְמוֹנוֹ שֶׁל רַבִּי יְהוּדָה הַנָּשִׂיא בְּאֶרֶץ יִשְׂרָאֵל. דֶּרֶךְ צְלֵחָה!".

הַחַיָּלִים הָרוֹמָאִים צָחֲקוּ בַּסֵּתֶר. "אוֹהוֹ," הֵם לָחֲשׁוּ זֶה לָזֶה, "הַקֵּיסָר הַגָּדוֹל שׁוֹלֵחַ לְיְהוּדָה חִטָּה. שֶׁיֵּלְכוּ לַאֲפוֹת לֶחֶם הַיְּהוּדִים הָאֵלֶּה... אֵין בַּשַּׂקִּים הָאֵלּוּ מַשֶּׁהוּ יָקָר. אֲפִלּוּ אֵין טַעַם לִפְתֹּחַ לְהָצִיץ!". הֵם לֹא חָשְׁדוּ בִּכְלוּם, וְהַשַּׁיָּרָה יָצְאָה לַדֶּרֶךְ בְּבִטְחָה.

הַהֲפְתָּעָה בְּבֵית הַמִּדְרָשׁ

עָבְרוּ שָׁבוּעוֹת, וְשַׁיֶּרֶת הָעֲגָלוֹת שֶׁל צְבָא רוֹמָא הִגִּיעָה אֶל חֲצַר רַבִּי יְהוּדָה הַנָּשִׂיא. חַיָּלִים נוֹטְפֵי זֵעָה הֵרִימוּ אֶת הַשַּׂקִּים הַכְּבֵדִים (הֵם הָיוּ כְּבֵדִים בִּמְיֻחָד בִּגְלַל הַזָּהָב, אֲבָל הַחַיָּלִים חָשְׁבוּ שֶׁזּוֹ חִטָּה שֶׁנָּרְטְבָה אוֹ שֶׁהֵם עֲיֵפִים מִן הַמַּסָּע) וְהִנִּיחוּ אוֹתָם בְּקּוֹל חֲבָטָה עַמְמוּמִית עַל רִצְפַּת הָאֶבֶן שֶׁל פְּרוֹזְדוֹר בֵּית הַנָּשִׂיא.

הַשְּׁלִיחִים מָסְרוּ לְרַבִּי בַּקָּשָׁה שֶׁל אַנְטוֹנִינוּס: "הַקֵּיסָר צִוָּה לוֹמַר לְךָ, כִּי יֶשְׁנָם שׁוֹדְדִים וְרוֹאִים רָעִים, וְלָכֵן בָּחַר לִשְׁלֹחַ אֶת מַתְּנָתוֹ תַּחַת מַסְוֶה שֶׁל חִטָּה, כְּדֵי שֶׁתַּגִּיעַ בְּשָׁלוֹם לַיְּשִׁיבָה הַקְּדוֹשָׁה שֶׁלְּךָ". הֵם הִשְׁתַּחֲווּ וְחָזְרוּ לְרוֹמָא.

הַתַּלְמִידִים בַּיְּשִׁיבָה הִבִּיטוּ בַּשַּׂקִּים בִּתְמִיהָה. רַבִּי יְהוּדָה הַנָּשִׂיא כְּבָר הָיָה אָדָם עָשִׁיר מְאֹד. מְסֻפָּר עָלָיו שֶׁאַף לֹא פָּסַק צְנוֹן, חֲזֶרֶת וְקִשּׁוּאִים מֵעַל שֻׁלְחָנוֹ אֲפִלּוּ בִּתְקוּפוֹת קָשׁוֹת. "לָמָּה שׁוֹלֵחַ אַנְטוֹנִינוּס שַׂקִּים שֶׁל חִטָּה לְרַבֵּינוּ?" לָחֲשׁוּ הַתַּלְמִידִים. "וְכִי חָסֵר לְרַבֵּינוּ תְּבוּאָה לֶאֱפוֹת לֶחֶם? וּמַה זֶה הַשֶּׁקֶר הַזֶּה שֶׁאָמְרוּ הַחַיָּלִים?".

אֲבָל רַבִּי יְהוּדָה חִיֵּךְ. עֵינָיו הַקְּדוֹשׁוֹת הֵבִינוּ אֶת הַסּוֹד הַמִּסְתַּתֵּר מִתַּחַת לַפְּשׁוּטוּת.

"פִּתְחוּ אֶת הַשַּׂקִּים," צִוָּה רַבִּי בְּשִׂמְחָה שְׁקֵטָה.

אֶחָד מִשַּׁמָּשָׁיו שֶׁל רַבִּי נִגַּשׁ, וּפָתַח אֶת שְׂרוֹךְ הַפִּשְׁתָּן בְּרֹאשׁ הַשַּׂק הָרִאשׁוֹן. הוּא הֵחֵל לְהוֹרִיק אֶת תָּכְנוֹ אֶל מִכְלֵי הָאֲגִירָה שֶׁל הַיְּשִׁיבָה. בַּתְּחִלָּה יָרְדוּ גַּרְגִּירֵי חִטָּה צְהֻבִּים וְרֵיחָנִיִּים... רַחַשׁ נָעִים שֶׁל תְּבוּאָה. וְאָז... לְפֶתַע...

קְלַאנְג! צְלִינְג! קְלִינְג!

נִשְׁמְעָה סִדְרַת דְּפִיקוֹת מַתֶּכֶת מַבְרִיקוֹת עַל הָאֶבֶן. צֶבַע שֶׁל שֶׁמֶשׁ נָפַל אֶל הָרִצְפָּה. הַתַּלְמִידִים הִשְׁתַּתְּקוּ. מַטְבְּעוֹת קֵיסָר עֲשׂוּיִים זָהָב טָהוֹר צָנְחוּ בְּעֶרְמָה עֲנָקִית מִתּוֹךְ תַּחְתִּית שַׂק הַחִטָּה!

הַשַּׁמָּשׁ עָבַר לַשַּׂק הַשֵּׁנִי וְשָׁפַךְ – וְשׁוּב, תַּחַת הַחִטָּה – זָהָב! כָּךְ גַּם בַּשְּׁלִישִׁי וּבָרְבִיעִי! עוֹד וְעוֹד אוֹצָרוֹת נוֹפְלִים מִתּוֹךְ הַהַסְתָּרָה. רַבִּי קָם מִכִּסְאוֹ וְהִבִּיט בַּזָּהָב. אַף לֹא פְּרוּטָה אַחַת נוֹעֲדָה כְּדֵי שֶׁרַבִּי יִקְנֶה לְעַצְמוֹ מִשְׁחַקִּים אוֹ דְּבָרִים יְקָרִים בִּשְׁבִיל נוֹחִיּוּתוֹ. רַבִּי הָיָה קָדוֹשׁ. וְחָכְמָתוֹ שֶׁל אַנְטוֹנִינוּס מִלְּאָה אֶת הַיְּשִׁיבָה בְּחֶדְוָה, שֶׁכֵּן הַזָּהָב הַזֶּה הוּפְנָה כֻּלּוֹ כְּדֵי לְהַאֲכִיל יְתוֹמִים, וְלִבְנוֹת תַּלְמוּדֵי תּוֹרָה בְּכָל אֶרֶץ יִשְׂרָאֵל שֶׁהָיְתָה פְּגוּעָה וַחֲרֵבָה אַחֲרֵי צָרוֹת הָרוֹמָאִים.

וּמֵאָז יָדְעוּ הַכֹּל – הַקֶּשֶׁר בֵּין רַבִּי לְאַנְטוֹנִינוּס הוּא סוֹד גָּדוֹל וְטָהוֹר. דַּוְקָא הַדְּבָרִים שֶׁעוֹשִׂים בַּסֵּתֶר, בְּלִי רַעַשׁ וּצְלִצּוּלִים שֶׁל גַּאֲוָה כְּלַפֵּי חוּץ – הֵם אֵלּוּ שֶׁמַּעֲמִידִים אֶת הַתּוֹרָה, וְהֵם שֶׁמּוֹבִילִים לַנִּצָּחוֹן הָאֲמִתִּי מֵאֵת הַשֵּׁם יִתְבָּרַךְ!

ME (הדמות): אַנְטוֹנִינוּס הַקֵּיסָר וְרַבִּי יְהוּדָה הַנָּשִׂיא עָשׂוּ אֶת הַחֶסֶד וְהַחְזָקַת הַתּוֹרָה בְּאֹפֶן נִסְתָּר. הַקֵּיסָר יָכוֹל הָיָה לְהִתְגָּאוֹת בַּמַּתָּנָה שֶׁלּוֹ, אֲבָל כְּדֵי שֶׁזֶּה יַצְלִיחַ - הוּא כִּסָּה אֶת הַזָּהָב הַיָּקָר בְּחִטָּה פְּשׁוּטָה. וְרַבִּי – קִבֵּל בְּקְדֻשָּׁה וַעֲנָוָה אֶת הַכֶּסֶף שֶׁיְּקַיֵּם אֶת הַיְּשִׁיבָה.

WE (אנחנו): לִפְעָמִים, כְּשֶׁאֲנַחְנוּ עוֹשִׂים מַשֶּׁהוּ טוֹב – עוֹזְרִים בַּבַּיִת, מְוַתְּרִים לְחָבֵר, אוֹ נּוֹתְנִים מַטְבֵּעַ לַצְּדָקָה – אֲנַחְנוּ דּוֹאֲגִים שֶׁכֻּלָּם יִרְאוּ. אֲנַחְנוּ צוֹעֲקִים "אִמָּא, תִּרְאִי אֵיךְ עָזַרְתִּי!", אוֹ מַבְלִיטִים אֶת מָה שֶׁעָשִׂינוּ.

YOU (התלמיד): מָחָר בַּבֹּקֶר, אוֹ תּוֹךְ כְּדֵי הַיּוֹם, נַסּוּ לַעֲשׂוֹת "מִבְצָע אֲנְטוֹנִינוּס". תַּעֲשׂוּ חֶסֶד אֶחָד – לְסַדֵּר מַשֶּׁהוּ לְאָח אוֹ לְאָחוֹת, לַעֲזֹר קְצָת לַמּוֹרָה, אוֹ לְוַתֵּר עַל מַשֶּׁהוּ – בְּלִי שֶׁאַף אֶחָד יֵדַע מִי עָשָׂה אֶת זֶה . הַקָּדוֹשׁ בָּרוּךְ הוּא כְּבָר יִרְאֶה אֶת זְהַב הַמִּדּוֹת שֶׁלָּכֶם, גַּם אִם הוּא מְכֻסֶּה!

א. שְׁאֵלַת פְּתִיחָה (לפני הסיפור):

"ילדים, תתארו לכם שאתם מקבלים מתנה בקופסה ענקית ומהודרת, ואחרי שאתם פותחים אותה בזהירות... אתם מגלים בפנים שקית של חול. מבאס, נכון? אבל מה יקרה אם מתחת לחול מסתתר משהו שמשנה את הכול? היום נשמע על שקית שלא הייתה מלאה בחול, אלא במשהו פשוט מאוד... שקרה איתו הנס הגדול ביותר לרבינו הקדוש!"

ב. הֶעָרוֹת בִּצּוּעַ וּסְטוֹרִיטֶלִינְג:

ג. הַמְלָצוֹת לְמַצֶּגֶת (4 שְׁקוּפִיּוֹת עִם שְׁמִירָה עַל קְדֻשָּׁה וּצְנִיעוּת):

ד. פְּעִילוּת מְסַכֶּמֶת:

"קופסת חסד החיטה" –

הבא לכיתה קופסת פלסטיק (או שקית בד) מלאה בגרעיני חומוס, פסטה יבשה או חול קינטי. החבא בתוכה פתקים מקופלים או מטבעות שוקולד קטנות.

הזמן 3-4 ילדים בכל פעם לחפש בפנים ולהוציא פתק/מטבע. על כל פתק יהיה רשום "חסד בסתר" (למשל: לאסוף את הניירות מתחת לשולחן בסוף השיעור בלי שהמורה יראה, להכניס 10 אגורות לקופת צדקה בבוקר בשקט, לומר מילה טובה למישהו שעצוב). זה ימחיש שהדברים הכי יקרים באמת – נמצאים בפנים, איפה שאף אחד לא רואה!

ה. מְקוֹרוֹת לַסִּפּוּר:

מקורות: המקורות מופיעים בקובץ המקורי.

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “להלן עיבוד חינוכי, עשיר ומרתק לסיפור המופלא על קשריו של רבי יהודה הנשיא (רבינו הקדוש) ואנטונינוס קיסר רומא , ובמיוחד על הדרך שבה בחר הקיסר לשלוח זהב לרבי.” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “אַנְטוֹנִינוּס הֵבִין שֶׁהַכֹּחַ הָאֲמִתִּי בָּעוֹלָם הוּא לֹא הַחֲרָבוֹת וְהַבַּרְזֶל שֶׁל רוֹמָא, אֶלָּא הַתּוֹרָה הַקְּדוֹשָׁה שֶׁל הַיְּהוּדִים…”?
ג. כיצד הגיב רבי יהודה הנשיא (רבי) למצב המתואר: “וְגָרוּעַ מִזֶּה, הֵם יָשׁוּבוּ וְיָסִיתוּ אוֹתִי נֶגֶד הַיְּהוּדִים, יַעֲלִילוּ עֲלֵיהֶם עֲלִילוֹת שֶׁקֶר כְּדֵי לְהַחֲרִיב לָהֶם אֶת הַמְּדִינָה,…”?
ד. מה השתנה בין “הוּא פָּתַח אֶת הַדֶּלֶת וְקָרָא לְשָׂרָיו: "קְחוּ מִיָּד אֶת הַשַּׂקִּים הַלָּלוּ שֶׁל הַחִטָּה, וְהוֹבִילוּ אוֹתָם…” לבין “אַף לֹא פְּרוּטָה אַחַת נוֹעֲדָה כְּדֵי שֶׁרַבִּי יִקְנֶה לְעַצְמוֹ מִשְׁחַקִּים אוֹ דְּבָרִים יְקָרִים בִּשְׁבִיל…”?
ה. כיצד הגיע הסיפור אל האירוע: “ילדים, תתארו לכם שאתם מקבלים מתנה בקופסה ענקית ומהודרת, ואחרי שאתם פותחים אותה בזהירות...”?

3. מה היה ראשון?

כתבו ליד כל אירוע מספר מ־1 עד 5 לפי סדר התרחשותו בסיפור.

□ מֵעַל הַכֹּל כָּתַב: "מִנְחַת חִטִּים עֲבוּר אֲהוּבִי, רַבִּי יְהוּדָה".
□ להלן עיבוד חינוכי, עשיר ומרתק לסיפור המופלא על קשריו של רבי יהודה הנשיא (רבינו הקדוש) ואנטונינוס קיסר רומא , ובמיוחד על הדרך שבה בחר הקיסר לשלוח זהב לרבי.
□ זה ימחיש שהדברים הכי יקרים באמת – נמצאים בפנים, איפה שאף אחד לא רואה!
□ יוֹם אֶחָד, בְּחַדְרוֹ הַפְּרָטִי שֶׁל הַקֵּיסָר בְּרוֹמָא, פָּתַח אַנְטוֹנִינוּס אֶת תֵּבַת אוֹצָרוֹ הָאִישִׁי.
□ וּמֵאָז יָדְעוּ הַכֹּל – הַקֶּשֶׁר בֵּין רַבִּי לְאַנְטוֹנִינוּס הוּא סוֹד גָּדוֹל וְטָהוֹר.

לפני ואחרי

מה קרה לפני: “מֵעַל הַכֹּל כָּתַב: "מִנְחַת חִטִּים עֲבוּר אֲהוּבִי, רַבִּי יְהוּדָה".”?
מה קרה אחרי: “וּמֵאָז יָדְעוּ הַכֹּל – הַקֶּשֶׁר בֵּין רַבִּי לְאַנְטוֹנִינוּס הוּא סוֹד גָּדוֹל וְטָהוֹר.”?

סיבה ותוצאה מתוך המעשה

האירוע הנתוןמה גרם לו?מה הייתה תוצאתו?
מַטְבְּעוֹת תְּפוּחִים שֶׁל זָהָב טָהוֹר, שֶׁהוּבְאוּ מִקְּצוֹת הָעוֹלָם, נָצְצוּ בָּאוֹר.
עֵינָיו הַקְּדוֹשׁוֹת הֵבִינוּ אֶת הַסּוֹד הַמִּסְתַּתֵּר מִתַּחַת לַפְּשׁוּטוּת.

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “כיצד שלח אנטונינוס זהב לרבי” על רבי יהודה הנשיא (רבי)? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “אֲבָל אַנְטוֹנִינוּס מִהֵר לְעַטֵּף אוֹתָם בְּבַד כְּדֵי שֶׁלֹּא יַרְעִישׁוּ.”.
ג. אילו רבי יהודה הנשיא (רבי) לא היה פועל כפי שמתואר ב־“אַף לֹא פְּרוּטָה אַחַת נוֹעֲדָה כְּדֵי שֶׁרַבִּי יִקְנֶה לְעַצְמוֹ מִשְׁחַקִּים אוֹ דְּבָרִים יְקָרִים בִּשְׁבִיל נוֹחִיּוּתוֹ.” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “ילדים, תתארו לכם שאתם מקבלים מתנה בקופסה ענקית ומהודרת, ואחרי שאתם פותחים אותה בזהירות...” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. להלן עיבוד חינוכי, עשיר ומרתק לסיפור המופלא על קשריו של רבי יהודה הנשיא…
2. פְּעָמִים רַבּוֹת, בְּאִישׁוֹן לַיְלָה מְאֻחָר, כְּשֶׁהַשּׁוֹמְרִים…
3. כְּשֶׁהוּבְאוּ הַשַּׂקִּים וְהַחִטָּה לְחַדְרוֹ, סִלֵּק הַקֵּיסָר אֶת…
4. עָבְרוּ שָׁבוּעוֹת, וְשַׁיֶּרֶת הָעֲגָלוֹת שֶׁל צְבָא רוֹמָא הִגִּיעָה…
5. הַקֵּיסָר יָכוֹל הָיָה לְהִתְגָּאוֹת בַּמַּתָּנָה שֶׁלּוֹ, אֲבָל כְּדֵי…
6. זה ימחיש שהדברים הכי יקרים באמת – נמצאים בפנים, איפה שאף אחד לא רואה!

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “וְגָרוּעַ מִזֶּה, הֵם יָשׁוּבוּ וְיָסִיתוּ אוֹתִי נֶגֶד הַיְּהוּדִים, יַעֲלִילוּ עֲלֵיהֶם עֲלִילוֹת שֶׁקֶר כְּדֵי לְהַחֲרִיב…”.
מכתב: כתבו לרבי יהודה הנשיא (רבי) בעקבות “עֵינָיו הַקְּדוֹשׁוֹת הֵבִינוּ אֶת הַסּוֹד הַמִּסְתַּתֵּר מִתַּחַת לַפְּשׁוּטוּת.”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “וְרַבִּי – קִבֵּל בְּקְדֻשָּׁה וַעֲנָוָה אֶת הַכֶּסֶף שֶׁיְּקַיֵּם אֶת הַיְּשִׁיבָה.”.
דיווח: כתבו ידיעה קצרה שכותרתה “כיצד שלח אנטונינוס זהב לרבי”.
