המעיל המשותף עם אשתו

דף עבודה בעקבות רבי יהודה בר אלעאי
שם: כיתה:

המעשה — הסיפור המלא

יְלָדִים אֲהוּבִים, הַאִם הִרְגַּשְׁתֶּם פַּעַם אֶת קֹר הַחֹרֶף חודר לָעֲצָמוֹת? כְּשֶׁהָרוּחַ שׁוֹרֶקֶת בַּחוּץ, הַגֶּשֶׁם מַכֶּה עַל הַחַלּוֹנוֹת, וְרֵיחַ עָמֹק שֶׁל אֲדָמָה רְטֻבָּה וּבוֹץ נִכְנַס לְתוֹךְ הַבַּיִת? בְּיָמִים כָּאֵלֶּה, אֲנַחְנוּ רָצִים לָאָרוֹן, מוֹצִיאִים מְעִיל פּוּךְ עָבֶה, אוֹ מִתְכַּרְבְּלִים מִתַּחַת לִשְׂמִיכַת הַצֶּמֶר הַנְּעִימָה שֶׁלָּנוּ.

אֲבָל בְּבֵיתוֹ שֶׁל רַבִּי יְהוּדָה בַּר אִלְעָאי, הַתַּנָּא הַקָּדוֹשׁ שֶׁגָּר בְּאֶרֶץ הַגָּלִיל הַקָּרָה, לֹא הָיָה אָרוֹן מָלֵא בִּמְעִילִים. לְמַעֲשֶׂה, כִּמְעַט וְלֹא הָיָה בַּבַּיִת הַזֶּה שׁוּם דָּבָר!

רַבִּי יְהוּדָה וְאִשְׁתּוֹ הַצַּדֶּקֶת הָיוּ עֲנִיִּים מְרוּדִים. קִירוֹת בֵּיתָם הָיוּ עֲשׂוּיִים אֲבָנִים פְּשׁוּטוֹת וְטִיט שֶׁהִתְקַשָּׁה, וּבַמִּטְבָּח לֹא הָיָה בָּשָׂר אוֹ דָּגִים אֲפִלּוּ לְשַׁבָּת. מָה הֵם אָכְלוּ? יְרָקוֹת פְּשׁוּטִים, עָלִים, וְלִפְעָמִים תְּאֵנִים יְבֵשׁוֹת אוֹ מְעַט תְּרָדִים (סוּג שֶׁל סֶלֶק יָרֹק וּפָשׁוּט) שֶׁרֵיחָם הַמִּתְבַּשֵּׁל מִלֵּא אֶת הַבַּיִת הַקָּטָן [1, 2]. טַעַם הָאֹכֶל הָיָה דַּל, אַךְ אִם הֱיִיתֶם נִכְנָסִים לַחֶדֶר, הֱיִיתֶם רוֹאִים דָּבָר מַדְהִים: פָּנָיו שֶׁל רַבִּי יְהוּדָה הָיוּ מְאִירוֹת! אוֹר זָהוּב וְנָעִים מַמָּשׁ קָרַן מֵהַפָּנִים שֶׁלּוֹ. מִי שֶׁלֹּא יָדַע, חָשַׁב שֶׁרַבִּי יְהוּדָה הוּא אָדָם עָשִׁיר הָאוֹכֵל בְּכָל יוֹם סְעוּדוֹת מְלָכִים. אֲבָל הָאֱמֶת הָיְתָה אַחֶרֶת. שָׁאֲלוּ אוֹתוֹ פַּעַם: "רַבִּי, מַדּוּעַ פָּנֶיךָ מְאִירוֹת כָּל כָּךְ?" עָנָה לָהֶם הַצַּדִּיק עִם חִיּוּךְ מָלֵא אַהֲבָה: "זוֹ הַתּוֹרָה! חָכְמַת הַתּוֹרָה שֶׁאֲנִי לוֹמֵד – הִיא שֶׁמְּאִירָה אֶת פָּנַי!" [1, 2].

מְעִיל צֶמֶר אֶחָד – לִשְׁנַיִם

הַחֹרֶף הִגִּיעַ, וְאִתּוֹ הַקֹּר הַקָּשֶׁה. הַגֶּשֶׁם תּוֹפֵף בְּחָזְקָה עַל הַגַּג, וְרוּחוֹת עַזּוֹת נָשְׁבוּ בָּרְחוֹבוֹת. כָּל תּוֹשָׁבֵי הָעִיר הוֹצִיאוּ אֶת מְעִילֵיהֶם הָעָבִים, הִתְעַטְּפוּ בְּשָׁאלִים שֶׁל צֶמֶר וּבְכוֹבָעִים חַמִּים.

וּמָה הָיָה בְּבֵיתוֹ שֶׁל הַתַּנָּא רַבִּי יְהוּדָה בַּר אִלְעָאי?

בְּכָל הַבַּיִת כֻּלּוֹ הָיָה מְעִיל חוֹרְפִּי אֶחָד וְיָחִיד. בֶּגֶד זֶה נֶאֱרַג מִצֶּמֶר גַּס וּמְחֻסְפָּס, עָבֶה מַסְפִּיק כְּדֵי לִשְׁמֹר עַל חוֹם הַגּוּף בְּיָמִים קָפְאִים, אַךְ הוּא הָיָה הַבֶּגֶד הַחַם הַיְּחִידִי בָּעוֹלָם שֶׁהָיָה שַׁיָּךְ לְזֶה הַזּוּג הַקָּדוֹשׁ.

כֵּיצַד חַיִּים שְׁנֵי בְּנֵי אָדָם בַּחֹרֶף עִם מְעִיל אֶחָד? מִי יִלְבַּשׁ אוֹתוֹ?

יְלָדִים, אִם זֶה הָיָה קוֹרֶה הַיּוֹם בְּבַיִת רָגִיל, אוּלַי הָיְתָה מַתְחִילָה מְרִיבָה. אֲבָל בְּבֵיתוֹ שֶׁל רַבִּי יְהוּדָה הָיָה שָׁלוֹם נִפְלָא הַמָּלֵא בְּהִתְחַשְּׁבוּת [3].

הֵם קָבְעוּ בֵּינֵיהֶם הֶסְכֵּם שֶׁל אַהֲבָה.

כַּאֲשֶׁר אִשְׁתּוֹ שֶׁל רַבִּי יְהוּדָה הָיְתָה חַיֶּבֶת לָצֵאת אֶל הַשּׁוּק הַהוֹמֶה, לִקְנוֹת אֶת מְעַט תְּרָדִים לִסְעוּדָתָם, אוֹ לְבַקֵּר שְׁכֵנָה דַּלָּה כְּדֵי לַעֲזֹר לָהּ – הִיא הָיְתָה לוֹקַחַת אֶת מְעִיל הַצֶּמֶר. הִיא הָיְתָה מִתְעַטֶּפֶת בּוֹ הֵיטֵב מֵהַקֹּר, וְיוֹצֵאת לָרְחוֹב. וּמָה עָשָׂה רַבִּי יְהוּדָה בְּאוֹתוֹ זְמַן? הוּא נִשְׁאַר בַּבַּיִת, עָטוּף בִּבְגָדִים דַּקִּים, יוֹשֵׁב בַּפִּנָּה, וְקוֹלוֹ הַנָּעִים וְהָרַךְ מְנַגֵּן אֶת אֲמָרוֹת הַתּוֹרָה שֶׁל רַבּוֹ – רַבִּי עֲקִיבָא. קוֹל הַתּוֹרָה מִלֵּא אֶת הַבַּיִת בְּחֹם פְּנִימִי עַד שֶׁהוּא כִּמְעַט שָׁכַח מֵהַקֹּר הַצּוֹרֵב [2-4].

וְכַאֲשֶׁר אִשְׁתּוֹ חָזְרָה מֵהַשּׁוּק הַבַּיְתָה, יָדַיִם אֲדֻמּוֹת מֵהַקֹּר וְרֵיחַ שֶׁל כְּפוֹר בִּבְגָדֶיהָ, הִיא הָיְתָה מוֹרִידָה אֶת מְעִיל הַצֶּמֶר מִכְּתֵפֶיהָ וּמוֹסֶרֶת אוֹתוֹ לְבַעֲלָהּ הַצַּדִּיק. רַבִּי יְהוּדָה הָיָה צָרִיךְ לָצֵאת לְהִתְפַּלֵּל בְּבֵית הַכְּנֶסֶת.

הוּא הָיָה קָם, מַנִּיחַ אֶת הַמְּעִיל עַל גּוּפוֹ וְרוֹכֵס אוֹתוֹ הֵיטֵב. וְאָז, לִפְנֵי שֶׁדָּרַךְ מִחוּץ לַדֶּלֶת, הָיָה עוֹצֵם אֶת עֵינָיו בְּהִתְרַגְּשׁוּת וּמְבָרֵךְ בְּקוֹל עָמֹק וְשָׂמֵחַ:

"בָּרוּךְ שֶׁעָטָנִי מְעִיל!" (כְּלוֹמַר – בָּרוּךְ ה' שֶׁנָּתַן לִי מְעִיל לְהִתְכַּסּוֹת בּוֹ!).

יְלָדִים, שִׂימוּ לֵב! הוּא לֹא אָמַר לְאִשְׁתּוֹ: "אוף, לָמָּה הַמְּעִיל הַזֶּה רָטוֹב מֵהַגֶּשֶׁם?" וְהוּא לֹא צָעַק: "לָמָּה יֵשׁ לָנוּ רַק מְעִיל אֶחָד וְלַשְּׁכֵנִים יֵשׁ עֲשָׂרָה?!" לֹא מִנֵּיהּ וְלֹא מִקְצָתֵיהּ! רַבִּי יְהוּדָה הָיָה "עָשִׁיר הַשָּׂמֵחַ בְּחֶלְקוֹ" [2]. כַּאֲשֶׁר לָבַשׁ אֶת הַמְּעִיל וְחָשׁ בְּחוֹם הַצֶּמֶר, הַשִּׂמְחָה בְּלִבּוֹ הָיְתָה כָּל כָּךְ גְּדוֹלָה, כְּאִלּוּ הוּא לוֹבֵשׁ בֶּגֶד-מַלְכוּת שֶׁל נָשִׂיא גָּדוֹל [3]!

הַדְּפִיקָה בַּדֶּלֶת וְהַשָּׁלִיחַ שֶׁל הַנָּשִׂיא

וַיְהִי הַיּוֹם, וְרַבָּן שִׁמְעוֹן בֶּן גַּמְלִיאֵל, שֶׁהָיָה נְשִׂיא יִשְׂרָאֵל וְאָדָם עָשִׁיר וְנִכְבָּד, הִכְרִיז עַל כִּנּוּס חָשׁוּב שֶׁל כָּל גְּדוֹלֵי יִשְׂרָאֵל לִתְפִלָּה וְלִקְבִיעַת הֲלָכוֹת (בִּגְלַל שֶׁלֹּא יָרְדוּ גְּשָׁמִים, אוֹ צָרָה שֶׁבָּאָה לָעוֹלָם). כָּל הַחֲכָמִים אָסְפוּ אֶת חֲפָצֵיהֶם, לָבְשׁוּ אֶת מְעִילֵיהֶם הַיָּפִים, וְיָצְאוּ אֶל בֵּית הַמִּדְרָשׁ הַגָּדוֹל.

רַק אָדָם אֶחָד חָסַר בָּאוּלָם. רַבִּי יְהוּדָה בַּר אִלְעָאי, הַחָכָם מִתַּלְמִידֵי רַבִּי עֲקִיבָא, לֹא יָכוֹל הָיָה לְהַגִּיעַ. בְּאוֹתוֹ יוֹם, רַעֲיָתוֹ נִזְקְקָה לָצֵאת מִן הַבַּיִת, וְהַמְּעִיל שֶׁלָּהֶם הָיָה עַל כְּתֵפֶיהָ [3].

רַבָּן שִׁמְעוֹן בֶּן גַּמְלִיאֵל שָׁאַל אֶת הַחֲכָמִים: "הֵיכָן רַבִּי יְהוּדָה? חָכְמָתוֹ חֲסֵרָה לָנוּ מְאֹד הַיּוֹם!".

הִשְׁפִּילוּ הַחֲכָמִים אֶת עֵינֵיהֶם וְסִפְּרוּ לַנָּשִׂיא אֶת הַסּוֹד הַכּוֹאֵב: "כְּבוֹד הַנָּשִׂיא... אֵין לְרַבִּי יְהוּדָה מָה לִלְבֹּשׁ... הַמְּעִיל מְשֻׁתָּף לוֹ וּלְאִשְׁתּוֹ, וְעַכְשָׁו הוּא אֶצְלָהּ".

הַנָּשִׂיא הִזְדַּעֲזֵעַ! אָדָם כָּל כָּךְ קָדוֹשׁ, הָרַב שֶׁכָּתַב שֵׁשׁ מֵאוֹת הֲלָכוֹת בַּמִּשְׁנָה [1], לֹא יָכוֹל לָבוֹא לִלְמֹד בִּגְלַל שֶׁאֵין לוֹ בֶּגֶד לְהִתְעַטֵּף בּוֹ?!

מִיָּד קָרָא רַבָּן שִׁמְעוֹן לְאַחַר מִשְּׁלִיחָיו.

"קַח אֶת הַמְּעִיל הַזֶּה," צִוָּה עָלָיו הַנָּשִׂיא, וּמָסַר לְיָדָיו גְּלִימָה מְפֹאֶרֶת, עֲשׂוּיָה בַּד יָקָר וּמְחַמֵּם מְאֹד, תְּפוּרָה לְהַפְלִיא וּמְלֵאָה בְּחֹם. "רוּץ מַהֵר לְבֵיתוֹ שֶׁל רַבִּי יְהוּדָה בַּר אִלְעָאי, וְתֵן לוֹ אוֹתָהּ בְּמַתָּנָה כְּדֵי שֶׁיּוּכַל לָבוֹא אֵלֵינוּ לְבֵית הַמִּדְרָשׁ!".

טּוּק, טּוּק, טּוּק.

הַדְּפִיקָה עַל הַדֶּלֶת הַחֲלוּשָׁה נִשְׁמְעָה כְּמוֹ תֹּף רָעַם. בְּפֶתַח הַבַּיִת הַקָּטָן וְהַקַּר עָמַד שְׁלִיחוֹ שֶׁל הַנָּשִׂיא, כְּשֶׁלְּיָדָיו תְּלוּיָה הַגְּלִימָה הַמְפֹאֶרֶת.

"שָׁלוֹם כְּבוֹד הָרַב," בֵּרֵךְ הַשָּׁלִיחַ, "נְשִׂיא יִשְׂרָאֵל, רַבָּן שִׁמְעוֹן בֶּן גַּמְלִיאֵל, שָׁמַע שֶׁנִּבְצַר מִמְּךָ לְהַגִּיעַ לְבֵית הַמִּדְרָשׁ. הַנָּשִׂיא מְבַקֵּשׁ שֶׁתִּקַּח אֶת הַמְּעִיל הַזֶּה בְּמַתָּנָה, שֶׁיִּהְיֶה לְךָ וּלְרַעְיָתְךָ כָּל אֶחָד מְעִיל מִשֶּׁלּוֹ".

הָעֹשֶׁר הָאֲמִתִּי נֶחְשָׂף

יְלָדִים! דַּמְיְנוּ אֶת הָרֶגַע הַזֶּה. הַקֹּר בַּחוּץ מַמָּשׁ חָתַךְ בַּבָּשָׂר. הַגְּלִימָה שֶׁל הַנָּשִׂיא נִרְאֲתָה כָּל כָּךְ נְעִימָה וּמַזְמִינָה. אָדָם רָגִיל הָיָה קוֹפֵץ עַל הַמְּצִיאָה וּמְחַבֵּק אֶת הַשָּׁלִיחַ. "תּוֹדָה לַה'! תּוֹדָה לַנָּשִׂיא! נִצַּלְנוּ מֵהַקֹּר!".

אַךְ רַבִּי יְהוּדָה עָמַד בְּשֶׁקֶט. פָּנָיו הֵאִירוּ כָּרָגִיל. לֹא הָיָה בָּהֶן שׁוּם צַעַר וְלֹא כְּאֵב. עֵינָיו נָצְצוּ מִתּוֹךְ אַהֲבָה לְרִבּוֹנוֹ שֶׁל עוֹלָם. הוּא אָהַב אֶת הַקָּדוֹשׁ בָּרוּךְ הוּא בְּכָל לִבּוֹ, וְרָצָה דָּבָר אֶחָד וְיָחִיד בָּעוֹלָם הַזֶּה: לִהְיוֹת קָרוֹב לַה' מִתּוֹךְ לִמּוּד תּוֹרָה, וְלֹא לֵהָנוֹת וְלוּ קְצָת מִמַּתְּנוֹת בָּשָׂר וָדָם ("לֹא רוֹצֶה לֵהָנוֹת מִשֶּׁל אֲחֵרִים") [2, 3].

בַּחִיּוּךְ גָּדוֹל שֶׁהֵאִיר אֶת כָּל הַחֶדֶר, הֵרִים רַבִּי יְהוּדָה יָד דַּקָּה וַעֲדִינָה, וְאָחַז בְּקָצֶהָ שֶׁל הַמַּחְצֶלֶת הַיְשָׁנָה עָלֶיהָ הָיָה יוֹשֵׁב כְּשֶׁלָּמַד. הוּא לֹא רָצָה שֶׁהַשָּׁלִיחַ יַחְשֹׁב שֶׁהוּא מִסְכֵּן שֶׁשָּׁכְחוּ אוֹתוֹ מִן הַשָּׁמַיִם.

רַבִּי יְהוּדָה אָחַז בַּמַּחְצֶלֶת, וְהֵרִים אוֹתָהּ קְצָת כְּלַפֵּי מַעְלָה.

יְלָדִים, אֲנִי עוֹמֵד לוֹמַר לָכֶם מַשֶּׁהוּ פֶּלֶא! הַשָּׁלִיחַ פָּעַר אֶת עֵינָיו, וּפִיו נִשְׁאַר פָּתוּחַ מֵרֹב תַּדְהֵמָה!

מִתַּחַת לַמַּחְצֶלֶת הַקְּרוּעָה וְהַיְשָׁנָה שֶׁל הַתַּנָּא הַקָּדוֹשׁ... כָּל הָרִצְפָּה שֶׁל הַבַּיִת הָיְתָה פִּתְאוֹם מְלֵאָה, אֲבָל מְלֵאָה עַד הַתִּקְרָה, בְּמַטְבְּעוֹת שֶׁל זָהָב טָהוֹר! עֲרֵמוֹת עֲרֵמוֹת שֶׁל זָהָב וַאֲבָנִים טוֹבוֹת הִתְגַּלּוּ מִתַּחַת לַמַּחְצֶלֶת, נוֹצְצוֹת כְּמוֹ שֶׁמֶשׁ שְׁנִיָּה מִתּוֹךְ הָאֲדָמָה! (הַקָּדוֹשׁ בָּרוּךְ הוּא רָצָה לְהַרְאוֹת שֶׁרַבִּי יְהוּדָה רָאוּי לְכָל הָעֹשֶׁר שֶׁבָּעוֹלָם אִם יִבְקַשׁ!).

אָמַר רַבִּי יְהוּדָה לַשָּׁלִיחַ בְּרֹגַע קָדוֹשׁ: "הַבֵּט נָא... רְאֵה כַּמָּה עֹשֶׁר זָמִין וּמוּכָן עֲבוּרִי אִם רַק אֶרְצֶה וְאֶקַּח. אֲבָל אֲנִי לֹא מְבַקֵּשׁ דָּבָר! אֲנִי וְאִשְׁתִּי בָּחַרְנוּ בַּתּוֹרָה, בַּחַיִּים הַפְּשׁוּטִים, מִבְּחִירָה, כִּי זֶה מָה שֶׁמְּקָרֵב אוֹתָנוּ לַה'. אֵינֶנִּי רוֹצֶה לֵהָנוֹת מִמַּתְּנוֹת בָּשָׂר וָדָם בָּעוֹלָם הַזֶּה. אֱמֹר תּוֹדָה לַנָּשִׂיא עַל לִבּוֹ הַטּוֹב, אַךְ הַחֲזֵר אֶת הַגְּלִימָה שֶׁלּוֹ".

רַבִּי יְהוּדָה מָשַׁךְ מַטָּה אֶת הַמַּחְצֶלֶת, וְהַזָּהָב נֶעֱלַם כְּלֹא הָיָה. הַשָּׁלִיחַ הֶמוּם וְנִפְעָם חָזַר לַנָּשִׂיא, וּמִהֵר לְסַפֵּר אֶת עָצְמַת אֱמוּנָתוֹ וְשִׂמְחָתוֹ שֶׁל הַצַּדִּיק – הָעָשִׁיר הָאֲמִתִּי. וְרַבִּי יְהוּדָה? הוּא הִמְשִׁיךְ לְחַכּוֹת שֶׁאִשְׁתּוֹ תַּחֲזֹר עִם הַמְּעִיל, וְחִיּוּךְ רָחָב וּמְאִיר פָּרוּשׂ עַל פָּנָיו [1-3].

א. שְׁאֵלַת פְּתִיחָה (לְפְנֵי שֶׁמַּתְחִילִים הַסִּפּוּר):

"יְלָדִים, אֵיזֶה מְעִיל יֵשׁ לָכֶם בַּחֹרֶף? (תֵּן דֻּגְמָאוֹת מֵהַכִּתָּה). עַכְשָׁו, דַּמְיְנוּ שֶׁיֵּשׁ לָכֶם אַח גָּדוֹל אוֹ חָבֵר קָרוֹב, וְיֵשׁ לָכֶם רַק מְעִיל אֶחָד לִשְׁנֵיכֶם! בֹּקֶר אֶחָד הוּא יוֹצֵא בַּקֹּר עִם הַמְּעִיל, וְאַתֶּם חוֹכִים לוֹ 3 שָׁעוֹת שֶׁיַּחֲזֹר. הֲיִיתֶם כּוֹעֲסִים?

הַיּוֹם נִשְׁמַע עַל צַדִּיק שֶׁזֶּה הָיָה מַצָּבוֹ, וְאַף עַל פִּי כֵן הוּא הָיָה הָאָדָם הַשָּׂמֵחַ בָּעוֹלָם!"

ב. הֶעָרוֹת לְבִצּוּעַ וּסְטוֹרִיטֶלִינְג (קול וְתְנוּעָה):

ג. הַמְלָצוֹת לִמְצֶגֶת שֶׁלֹּא קוֹפֶצֶת שׁוֹגֶגֶת (4 שְׁקוּפִיּוֹת מוּמְלָצוֹת):

ד. פְּעִילוּת מְסַכֶּמֶת ("שְׂמֵחִים בְּחֶלְקֵנוּ"):

חלק לכל ילד פתק קטן שעליו כתוב "עָשִׁיר שָׂמֵחַ בְּחֶלְקוֹ". בקש מכל תלמיד לרשום על הפתק 3 דברים פשוטים בחיים שלו שהוא שמח שיש לו (למשל: סנדוויץ' לארוחת עשר, המיטה שלי, החבר שיושב לידי).

כל מי שסיים – קם ושם את הפתק ב'תיבת האוצר' של הכיתה.

מקורות: תַּלְמוּד בַּבְלִי, מַסֶּכֶת נְדָרִים, דַּף מ"ט עַמּוּד ב' – נ' עַמּוּד א' : סִיפּוּר הַמְּעִיל הַמְּשֻׁתָּף, הַהֲבָאָה שֶׁל מַעֲטֶפֶת הַנָּשִׂיא, וְחָשִׂיפַת הַזָּהָב מִתַּחַת לַמַּחְצֶלֶת כְּדֵי לְהוֹכִיחַ שֶׁאֵינוֹ רוֹצֶה לֵהָנוֹת מִן הָעוֹלָם הַזֶּה. תֵּאוּרֵי עֲנִיּוּתוֹ שֶׁל רַבִּי יְהוּדָה וְהַתְּרָדִים שֶׁאָכַל, שֶׁעָשׂוּ אֶת פָּנָיו "צְהֻבּוֹת/מְאִירוֹת" מִתּוֹךְ תּוֹרָה – מוּבָאִים כֵּן בַּמְּקוֹרוֹת הָעוֹסְקִים בַּתַּנָּא מִסֵּפֶר "הָאַגָּדָה" (רס"א, רס"ג) [1, 2].

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “בְּיָמִים כָּאֵלֶּה, אֲנַחְנוּ רָצִים לָאָרוֹן, מוֹצִיאִים מְעִיל פּוּךְ עָבֶה, אוֹ מִתְכַּרְבְּלִים מִתַּחַת לִשְׂמִיכַת הַצֶּמֶר הַנְּעִימָה שֶׁלָּנוּ.” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “בְּכָל הַבַּיִת כֻּלּוֹ הָיָה מְעִיל חוֹרְפִּי אֶחָד וְיָחִיד.”?
ג. כיצד הגיב רבי יהודה בר אלעאי למצב המתואר: “הוּא הָיָה קָם, מַנִּיחַ אֶת הַמְּעִיל עַל גּוּפוֹ וְרוֹכֵס אוֹתוֹ הֵיטֵב.”?
ד. מה השתנה בין “בְּפֶתַח הַבַּיִת הַקָּטָן וְהַקַּר עָמַד שְׁלִיחוֹ שֶׁל הַנָּשִׂיא, כְּשֶׁלְּיָדָיו תְּלוּיָה הַגְּלִימָה הַמְפֹאֶרֶת.” לבין “עֲרֵמוֹת עֲרֵמוֹת שֶׁל זָהָב וַאֲבָנִים טוֹבוֹת הִתְגַּלּוּ מִתַּחַת לַמַּחְצֶלֶת, נוֹצְצוֹת כְּמוֹ שֶׁמֶשׁ שְׁנִיָּה…”?
ה. כיצד הגיע הסיפור אל האירוע: “עַכְשָׁו, דַּמְיְנוּ שֶׁיֵּשׁ לָכֶם אַח גָּדוֹל אוֹ חָבֵר קָרוֹב, וְיֵשׁ לָכֶם רַק מְעִיל אֶחָד לִשְׁנֵיכֶם!”?

3. מה היה ראשון?

כתבו ליד כל אירוע מספר מ־1 עד 5 לפי סדר התרחשותו בסיפור.

□ קַח אֶת הַמְּעִיל הַזֶּה," צִוָּה עָלָיו הַנָּשִׂיא, וּמָסַר לְיָדָיו גְּלִימָה מְפֹאֶרֶת, עֲשׂוּיָה בַּד יָקָר וּמְחַמֵּם מְאֹד, תְּפוּרָה לְהַפְלִיא…
□ בְּיָמִים כָּאֵלֶּה, אֲנַחְנוּ רָצִים לָאָרוֹן, מוֹצִיאִים מְעִיל פּוּךְ עָבֶה, אוֹ מִתְכַּרְבְּלִים מִתַּחַת לִשְׂמִיכַת הַצֶּמֶר הַנְּעִימָה שֶׁלָּנוּ.
□ בקש מכל תלמיד לרשום על הפתק 3 דברים פשוטים בחיים שלו שהוא שמח שיש לו (למשל: סנדוויץ' לארוחת עשר, המיטה שלי, החבר שיושב לידי).
□ כַּאֲשֶׁר אִשְׁתּוֹ שֶׁל רַבִּי יְהוּדָה הָיְתָה חַיֶּבֶת לָצֵאת אֶל הַשּׁוּק הַהוֹמֶה, לִקְנוֹת אֶת מְעַט תְּרָדִים לִסְעוּדָתָם, אוֹ לְבַקֵּר שְׁכֵנָה…
□ (הַקָּדוֹשׁ בָּרוּךְ הוּא רָצָה לְהַרְאוֹת שֶׁרַבִּי יְהוּדָה רָאוּי לְכָל הָעֹשֶׁר שֶׁבָּעוֹלָם אִם יִבְקַשׁ!).

האירוע הנתוןמה גרם לו?מה הייתה תוצאתו?

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “המעיל המשותף עם אשתו” על רבי יהודה בר אלעאי? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “רַבִּי יְהוּדָה בַּר אִלְעָאי, הַחָכָם מִתַּלְמִידֵי רַבִּי עֲקִיבָא, לֹא יָכוֹל הָיָה לְהַגִּיעַ.”.
ג. אילו רבי יהודה בר אלעאי לא היה פועל כפי שמתואר ב־“עֲרֵמוֹת עֲרֵמוֹת שֶׁל זָהָב וַאֲבָנִים טוֹבוֹת הִתְגַּלּוּ מִתַּחַת לַמַּחְצֶלֶת, נוֹצְצוֹת כְּמוֹ שֶׁמֶשׁ שְׁנִיָּה מִתּוֹךְ…” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “עַכְשָׁו, דַּמְיְנוּ שֶׁיֵּשׁ לָכֶם אַח גָּדוֹל אוֹ חָבֵר קָרוֹב, וְיֵשׁ לָכֶם רַק מְעִיל אֶחָד לִשְׁנֵיכֶם!” מחזק את המסר של הסיפור?
