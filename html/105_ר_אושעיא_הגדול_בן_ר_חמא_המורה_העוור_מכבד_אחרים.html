המורה העוור — מכבד אחרים

דף עבודה בעקבות ר' אושעיא הגדול בן ר' חמא
שם: כיתה:

המעשה — הסיפור המלא

פְּתִיחָה: אִישׁ שֶׁל כָּבוֹד

יְלָדִים, נַסּוּ לְדַמְיֵן אֶת בֵּית הַמִּדְרָשׁ הַגָּדוֹל שֶׁל אֶרֶץ יִשְׂרָאֵל לִפְנֵי כְּאַלְפַּיִם שָׁנָה. קוֹלוֹת שֶׁל לִמּוּד תּוֹרָה מִתְנַגְּנִים בָּאֲוִיר מִבֹּקֶר וְעַד עֶרֶב, מֵאוֹת תַּלְמִידִים מְפַלְפְּלִים בַּהֲלָכָה, וְרֵיחַ נָעִים שֶׁל קְלָפִים יְשָׁנִים עֲשׂוּיִים מֵעוֹר עַתִּיק נוֹדֵף בֶּחָלָל.

בְּמֶרְכַּז בֵּית הַמִּדְרָשׁ יָשַׁב אֶחָד מִגְּדוֹלֵי הַדּוֹר – רַבִּי אוֹשַׁעְיָה הַגָּדוֹל .

רַבִּי אוֹשַׁעְיָה הָיָה כָּל כָּךְ חָכָם וְגָדוֹל, עַד שֶׁכֻּלָּם קָרְאוּ לוֹ "אֲבִיהֶן שֶׁל מִשְׁנָיוֹת". כָּל תַּלְמִידֵי הַחֲכָמִים הָיוּ עוֹמְדִים לִקְרָאתוֹ כְּשֶׁהָיָה נִכְנָס, וְכֻלָּם רָצוּ לִשְׁמוֹעַ אֶת דִּבְרֵי הַתּוֹרָה שֶׁלּוֹ. אֲבָל יוֹתֵר מִשֶּׁרַבִּי אוֹשַׁעְיָה הָיָה חָכָם, הוּא הָיָה מָלֵא בְּאַהֲבַת יִשְׂרָאֵל וּבְמִדּוֹת טוֹבוֹת וַאֲצִילוֹת. וּבִמְיֻחָד – בְּכָבוֹד לְמִי שֶׁחַלָּשׁ וְזָקוּק לעֶזְרָה.

הַסְּעוּדָה הַקְּבוּעָה בַּחֶדֶר הַחֲשׁוּךְ

לְרַבִּי אוֹשַׁעְיָה הָיָה רַב זָקֵן מְאֹד. זָקֵן שֶׁלִּמֵּד אוֹתוֹ תּוֹרָה בְּיַלְדוּתוֹ. אוֹתוֹ רַב זָקֵן וְצַדִּיק חָלָה, וּלְמַרְבֵּה הַצַּעַר, הוּא אִבֵּד אֶת מְאוֹר עֵינָיו וְהָפַךְ לְעִוֵּר לַחֲלוּטִין. הוּא יָשַׁב בְּחֶדֶר פָּשׁוּט וַאֲפַלְפּוּל, וְלֹא יָכוֹל הָיָה לָצֵאת לְבֵית הַמִּדְרָשׁ.

אָדָם רָגִיל אוּלַי הָיָה שׁוֹלֵחַ מְשָׁרֵת פַּעַם בְּשָׁבוּעַ כְּדֵי לְהָבִיא תַּפּוּחִים לָרַב הַזָּקֵן, אֲבָל רַבִּי אוֹשַׁעְיָה? מַמָּשׁ לֹא!

רַבִּי אוֹשַׁעְיָה הַגָּדוֹל, נְשִׂיא הַתּוֹרָה שֶׁמֵּאוֹת אֲנָשִׁים מְחַכִּים רַק לִשְׁמוֹעַ מִלָּה מִפִּיו, קָבַע הַנְהָגָה סוֹדִית קְבוּעָה. בְּכָל יוֹם, אֲבָל בְּכָל יוֹם, כְּשֶׁהִגִּיעָה שְׁעַת סְעוּדַת הַצָּהֳרַיִם, הוּא עָזַב אֶת בֵּיתוֹ הַמְּרֻוָּח, לָקַח אִתּוֹ שְׁנֵי כִּכְּרוֹת לֶחֶם רַכִּים וּמְעַט תַּבְשִׁיל, וְהָלַךְ לֶאֱכוֹל אֶת סְעוּדָתוֹ מַמָּשׁ בְּתוֹךְ חַדְרוֹ שֶׁל הָרַב הָעִוֵּר!

רַבִּי אוֹשַׁעְיָה הָיָה יוֹשֵׁב קָרוֹב-קָרוֹב אֶל הַזָּקֵן. הוּא הָיָה מַחְזִיק אֶת כַּפּוֹת יָדָיו הַחֲמוּקוֹת וְהַמְּקֻמָּטוֹת, מְגִישׁ לוֹ אֶת הָאֹכֶל כְּדֵי שֶׁלֹּא יִתְלַכְלֵךְ, וּמְדַבֵּר אִתּוֹ דִּבְרֵי תּוֹרָה מְתוּקִים. כְּשֶׁרַבִּי אוֹשַׁעְיָה הָיָה נִכְנָס, הָיוּ פָּנָיו שֶׁל הָרַב הָעִוֵּר מוּאָרוֹת בְּשִׂמְחָה. זוֹ הָיְתָה הַשָּׁעָה הֲכִי מְאֻשֶּׁרֶת בַּיּוֹם שֶׁל שְׁנֵיהֶם!

הָאוֹרְחִים הַחֲשׁוּבִים וְהַהִתְלַבְּטוּת

יוֹם אֶחָד, הָלְתָה תְּכֻנָּה גְּדוֹלָה בָּעִיר. אוֹרְחִים רָמֵי יַחַס וַחֲשׁוּבִים מְאֹד מִיְּרוּשָׁלַיִם וְהַסְּבִיבָה הִגִּיעוּ לְבַקֵּר אֶת רַבִּי אוֹשַׁעְיָה הַגָּדוֹל. אֵלּוּ הָיוּ חֲכָמִים גְּדוֹלִים בַּתּוֹרָה מִשּׁוֹשֶׁלֶת הַנְּשִׂיאוּת.

בְּבֵיתוֹ שֶׁל רַבִּי אוֹשַׁעְיָה עָרְכוּ סְעוּדַת מְלָכִים. רֵיחוֹת מַטְרִיפִים שֶׁל צְלִי בָּשָׂר, דָּגִים טְרִיִּים שֶׁנִּצְלוּ עַל הָאֵשׁ וּמְרָקִים עֲשִׁירִים מִלְּאוּ אֶת חֲלַל הַבַּיִת. הַשֻּׁלְחָן כֻּסָּה בְּמַפָּה לְבָנָה כַּשֶּׁלֶג, וְכֻלָּם הִמְתִּינוּ לַסְּעוּדָה לִכְבוֹד קְדֻשַּׁת הַתּוֹרָה.

וְאָז, הִגִּיעָה הַשָּׁעָה לָשֶׁבֶת וְלֶאֱכֹל.

רַבִּי אוֹשַׁעְיָה עָצַר לְרֶגַע. לִבּוֹ דָּפַק בְּחָזְקָה. "מָה אֶעֱשֶׂה עַכְשָׁו אֶת סְעוּדָתִי אֵצֶל רַבִּי הָעִוֵּר?" חָשַׁב לְעַצְמוֹ.

וְאָז עָלְתָה בְּלִבּוֹ מַחֲשָׁבָה דּוֹאֶגֶת: "אִם אוֹשִׁיב אֶת הָרַב הַזָּקֵן כָּאן אִתָּנוּ בַּשֻּׁלְחָן... מָה יִקְרֶה? הֲרֵי הוּא אֵינוֹ רוֹאֶה. אֲנָשִׁים זְקֵנִים וְעִוְרִים מִתְקַשִּׁים לֶאֱכוֹל, אוּלַי יִפּוֹל לוֹ מְעַט מָרָק עַל הַבֶּגֶד? אוּלַי כּוֹס הַיַּיִן תִּשָּׁפֵךְ עַל מַפַּת הַשֻּׁלְחָן הַלְּבָנָה? הָאוֹרְחִים הַמְּכֻבָּדִים עֲלוּלִים לְהִסְתַּכֵּל עָלָיו, וְאוּלַי... אוּלַי הַזָּקֵן שֶׁלִּי יַרְגִּישׁ רַע? אוּלַי הוּא יַרְגִּישׁ שֶׁהוּא מִתְבַּיֵּשׁ?".

מֵרֹב שֶׁרַבִּי אוֹשַׁעְיָה אָהַב וְכִבֵּד אֶת הָרַב הַזָּקֵן, הוּא לֹא רָצָה לָקַחַת שׁוּם סִכּוּי שֶׁהַזָּקֵן יִתְבַּיֵּשׁ לְיַד אֲנָשִׁים זָרִים.

לָכֵן, הוּא קִבֵּל הַחְלָטָה קָשָׁה. הַיּוֹם בְּאֹפֶן חָרִיג, מְשָׁרְתֵי הַבַּיִת יָבִיאוּ לָרַב הַזָּקֵן אֶת הַמָּנוֹת הֲכִי מְשֻׁבָּחוֹת אֵלָיו לַחֶדֶר, וְרַבִּי אוֹשַׁעְיָה יֵשֵׁב וְיֹאכַל אֶת הַסְּעוּדָה עִם הָאוֹרְחִים, כְּדֵי שֶׁלֹּא לִפְגּוֹעַ בִּכְבוֹדָם, וּכְדֵי לִשְׁמוֹר עַל כְּבוֹדוֹ שֶׁל הָרַב הָעִוֵּר.

הַצַּעַר וְהַבַּקָּשָׁה

הַסְּעוּדָה שֶׁל הָאוֹרְחִים הָיְתָה יָפָה. שָׁרוּ בָּהּ שִׁירִים וְאָמְרוּ דִּבְרֵי תּוֹרָה. אֲבָל יְלָדִים, רַבִּי אוֹשַׁעְיָה לֹא טָעַם כִּמְעַט מְאוּמָה מֵהַצְּלִי וְהַדָּגִים הַמְּשֻׁבָּחִים. מַשֶּׁהוּ הֵצִיק לוֹ בַּלֵּב. נַפְשׁוֹ שֶׁל הַצַּדִּיק אָמְרָה לוֹ שֶׁהוּא עָשָׂה טָעוּת.

"הַיּוֹם הָרַב שֶׁלִּי לֹא שָׁמַע אֶת קוֹלִי... הַיּוֹם לֹא הֶחֱזַקְתִּי בְּיָדוֹ..." כָּל דַּקָּה שֶׁעָבְרָה נִרְאֲתָה לוֹ כְּמוֹ נֵצַח. כְּשֶׁהָאוֹרְחִים סוֹף-סוֹף יָצְאוּ וְנִפְרְדוּ לְשָׁלוֹם, רַבִּי אוֹשַׁעְיָה אֲפִלּוּ לֹא נָח לְרֶגַע.

הוּא קָם וּמִהֵר יָשָׁר אֶל הַחֶדֶר הַחֲשׁוּךְ שֶׁל הָרַב הַזָּקֵן.

הוּא פָּתַח בַּעֲדִינוּת אֶת דֶּלֶת הָעֵץ הַיְּשָׁנָה. הַחֶדֶר הָיָה שָׁקֵט. הָרַב הָעִוֵּר יָשַׁב בְּפִנָּתוֹ, פָּנָיו שְׁלֵווֹת. רַבִּי אוֹשַׁעְיָה יָרַד עַל בִּרְכָּיו מוּל מִטַּת רַבּוֹ, לָקַח אֶת יָדוֹ שֶׁל הַזָּקֵן, וְדִמְעָה אַחַת חַמָּה צָנְחָה מֵעֵינוֹ שֶׁל הַצַּדִּיק עַל יָדוֹ שֶׁל הָעִוֵּר.

"מוֹרִי וְרַבִּי..." לָחַשׁ רַבִּי אוֹשַׁעְיָה בְּקוֹל דּוֹמֵעַ וְנִשְׁבָּר, "אַל יִכְעַס עָלַי אֲדוֹנִי כִּי הַיּוֹם לֹא הִגַּעְתִּי לֶאֱכוֹל יַחַד אִתְּךָ. אוֹרְחִים חֲשׁוּבִים בָּאוּ לְבֵיתִי לִסְעוֹד, וְחָשַׁשְׁתִּי בְּלִבִּי. יָדַעְתִּי שֶׁכְּבוֹדוֹ אֵינוֹ רוֹאֶה אֶת מַה שֶּׁהוּא אוֹכֵל, וּפָחַדְתִּי שֶׁמָּא הָאוֹרְחִים יִסְתַּכְּלוּ בְּצוּרָה לֹא יָפָה, וְשֶׁמָּא מַשֶּׁהוּ יִשָּׁפֵךְ וּכְבוֹדוֹ יִתְבַּיֵּשׁ לִפְנֵיהֶם וְיִצְטַעֵר בְּנַפְשׁוֹ. בִּגְלַל זֶה לֹא הִזְמַנְתִּי אוֹתְךָ. אֲנִי מְבַקֵּשׁ סְלִיחָה עַל כָּךְ שֶׁהִשְׁאַרְתִּי אוֹתְךָ לְבַד... מְחַל לִי, רַבִּי!".

הַבְּרָכָה הֶעֲצוּמָה מִשָּׁמַיִם

יְלָדִים אֲהוּבִים, מָה אַתֶּם חוֹשְׁבִים עָנָה הָרַב הַזָּקֵן? הַאִם הוּא כָּעַס?

הָרַב הָעִוֵּר עָצַם אֶת עֵינָיו הַסְּגוּרוֹת בְּלָאו-הָכִי. הוּא הִרְגִּישׁ אֶת הַדִּמְעָה שֶׁל רַבִּי אוֹשַׁעְיָה הַגָּדוֹל, וְהֵבִין שֶׁכָּל מַה שֶּׁרַבִּי אוֹשַׁעְיָה עָשָׂה הָיָה רַק מֵרֹב אַהֲבָה וְרֹב דְּאָגָה שֶׁלֹּא לְבַיֵּשׁ אוֹתוֹ! אֵיזוֹ מַחֲשָׁבָה מַדְהִימָה!

הָרַב הַזָּקֵן הֶעֱבִיר אֶת יָדוֹ הַמְּקֻמֶּטֶת עַל רֹאשׁוֹ שֶׁל תַּלְמִידוֹ הַגָּדוֹל, חִיֵּךְ חִיּוּךְ מָלֵא הָאוֹר, וְאָמַר אֶת אוֹתָם מִשְׁפָּטִים שֶׁיִּשָּׁאֲרוּ חֲקוּקִים לָנֶצַח בַּתַּלְמוּד:

"בְּנִי הָאָהוּב! אַתָּה פִּיַּסְתָּ אֶת מִי שֶׁהוּא נִרְאֶה וְאֵינוֹ רוֹאֶה" (כְּלוּמַר, אַתָּה בָּאתָ לְבַקֵּשׁ סְלִיחָה מִמֶּנִּי – מֵאָדָם שֶׁכֻּלָּם רוֹאִים אוֹתוֹ אַךְ הוּא בְּעַצְמוֹ עִוֵּר וְאֵינוֹ רוֹאֶה) – "יְהִי רָצוֹן, שֶׁמִּי שֶׁרוֹאֶה אֶת הַכֹּל וְאֵינוֹ נִרְאֶה – הוּא זֶה שֶׁיְּקַבֵּל אֶת סְלִיחָתְךָ וְיִתֵּן לְךָ אֶת שְׂכָרְךָ הַגָּדוֹל!" (כְּלוּמַר, הַקָּדוֹשׁ בָּרוּךְ הוּא, שֶׁרוֹאֶה אֶת הַלֵּב הַטּוֹב שֶׁלְּךָ וְאֶת הָרְגִישׁוּת שֶׁלְּךָ, יְשַׁלֵּם לְךָ שָׂכָר טוֹב מִשָּׁמַיִם!).

רַבִּי אוֹשַׁעְיָה יָצָא מֵהַחֶדֶר כְּשֶׁלִּבּוֹ שָׂמֵחַ. הוּא יָדַע שֶׁהַדָּבָר הֲכִי חָשׁוּב בָּעוֹלָם, וְהַכָּבוֹד הֲכִי מְיֻחָד בַּתּוֹרָה – הוּא לִשְׁמוֹר עַל כְּבוֹדוֹ שֶׁל הַזּוּלַת, וּבִמְיֻחָד שֶׁל מִי שֶׁקָּשֶׁה לוֹ, מִי שֶׁחָלָשׁ אוֹ פָּגוּעַ.

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “יְלָדִים, נַסּוּ לְדַמְיֵן אֶת בֵּית הַמִּדְרָשׁ הַגָּדוֹל שֶׁל אֶרֶץ יִשְׂרָאֵל לִפְנֵי כְּאַלְפַּיִם שָׁנָה.” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “אֲבָל יוֹתֵר מִשֶּׁרַבִּי אוֹשַׁעְיָה הָיָה חָכָם, הוּא הָיָה מָלֵא בְּאַהֲבַת יִשְׂרָאֵל וּבְמִדּוֹת טוֹבוֹת וַאֲצִילוֹת.”?
ג. כיצד הגיב ר' אושעיא הגדול בן ר' חמא למצב המתואר: “כְּשֶׁרַבִּי אוֹשַׁעְיָה הָיָה נִכְנָס, הָיוּ פָּנָיו שֶׁל הָרַב הָעִוֵּר מוּאָרוֹת בְּשִׂמְחָה.”?
ד. מה השתנה בין “וְאָז עָלְתָה בְּלִבּוֹ מַחֲשָׁבָה דּוֹאֶגֶת: "אִם אוֹשִׁיב אֶת הָרַב הַזָּקֵן כָּאן אִתָּנוּ בַּשֻּׁלְחָן...” לבין “כְּשֶׁהָאוֹרְחִים סוֹף-סוֹף יָצְאוּ וְנִפְרְדוּ לְשָׁלוֹם, רַבִּי אוֹשַׁעְיָה אֲפִלּוּ לֹא נָח לְרֶגַע.”?
ה. כיצד הגיע הסיפור אל האירוע: “אֲנִי מְבַקֵּשׁ סְלִיחָה עַל כָּךְ שֶׁהִשְׁאַרְתִּי אוֹתְךָ לְבַד...”?

3. מה היה ראשון?

כתבו ליד כל אירוע מספר מ־1 עד 5 לפי סדר התרחשותו בסיפור.

□ הַשֻּׁלְחָן כֻּסָּה בְּמַפָּה לְבָנָה כַּשֶּׁלֶג, וְכֻלָּם הִמְתִּינוּ לַסְּעוּדָה לִכְבוֹד קְדֻשַּׁת הַתּוֹרָה.
□ יְלָדִים, נַסּוּ לְדַמְיֵן אֶת בֵּית הַמִּדְרָשׁ הַגָּדוֹל שֶׁל אֶרֶץ יִשְׂרָאֵל לִפְנֵי כְּאַלְפַּיִם שָׁנָה.
□ הוּא יָדַע שֶׁהַדָּבָר הֲכִי חָשׁוּב בָּעוֹלָם, וְהַכָּבוֹד הֲכִי מְיֻחָד בַּתּוֹרָה – הוּא לִשְׁמוֹר עַל כְּבוֹדוֹ שֶׁל הַזּוּלַת, וּבִמְיֻחָד שֶׁל מִי…
□ הוּא יָשַׁב בְּחֶדֶר פָּשׁוּט וַאֲפַלְפּוּל, וְלֹא יָכוֹל הָיָה לָצֵאת לְבֵית הַמִּדְרָשׁ.
□ הוּא קָם וּמִהֵר יָשָׁר אֶל הַחֶדֶר הַחֲשׁוּךְ שֶׁל הָרַב הַזָּקֵן.

לפני ואחרי

מה קרה לפני: “הַשֻּׁלְחָן כֻּסָּה בְּמַפָּה לְבָנָה כַּשֶּׁלֶג, וְכֻלָּם הִמְתִּינוּ לַסְּעוּדָה לִכְבוֹד קְדֻשַּׁת הַתּוֹרָה.”?
מה קרה אחרי: “הוּא קָם וּמִהֵר יָשָׁר אֶל הַחֶדֶר הַחֲשׁוּךְ שֶׁל הָרַב הַזָּקֵן.”?

סיבה ותוצאה מתוך המעשה

האירוע הנתוןמה גרם לו?מה הייתה תוצאתו?
רַבִּי אוֹשַׁעְיָה הַגָּדוֹל, נְשִׂיא הַתּוֹרָה שֶׁמֵּאוֹת אֲנָשִׁים מְחַכִּים רַק לִשְׁמוֹעַ מִלָּה מִפִּיו, קָבַע הַנְהָגָה סוֹדִית קְבוּעָה.
נַפְשׁוֹ שֶׁל הַצַּדִּיק אָמְרָה לוֹ שֶׁהוּא עָשָׂה טָעוּת.

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “המורה העוור — מכבד אחרים” על ר' אושעיא הגדול בן ר' חמא? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “רֵיחוֹת מַטְרִיפִים שֶׁל צְלִי בָּשָׂר, דָּגִים טְרִיִּים שֶׁנִּצְלוּ עַל הָאֵשׁ וּמְרָקִים עֲשִׁירִים מִלְּאוּ אֶת חֲלַל הַבַּיִת.”.
ג. אילו ר' אושעיא הגדול בן ר' חמא לא היה פועל כפי שמתואר ב־“כְּשֶׁהָאוֹרְחִים סוֹף-סוֹף יָצְאוּ וְנִפְרְדוּ לְשָׁלוֹם, רַבִּי אוֹשַׁעְיָה אֲפִלּוּ לֹא נָח לְרֶגַע.” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “אֲנִי מְבַקֵּשׁ סְלִיחָה עַל כָּךְ שֶׁהִשְׁאַרְתִּי אוֹתְךָ לְבַד...” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. יְלָדִים, נַסּוּ לְדַמְיֵן אֶת בֵּית הַמִּדְרָשׁ הַגָּדוֹל שֶׁל אֶרֶץ…
2. אוֹתוֹ רַב זָקֵן וְצַדִּיק חָלָה, וּלְמַרְבֵּה הַצַּעַר, הוּא אִבֵּד…
3. זוֹ הָיְתָה הַשָּׁעָה הֲכִי מְאֻשֶּׁרֶת בַּיּוֹם שֶׁל שְׁנֵיהֶם!
4. אֲבָל יְלָדִים, רַבִּי אוֹשַׁעְיָה לֹא טָעַם כִּמְעַט מְאוּמָה…
5. רַבִּי אוֹשַׁעְיָה יָרַד עַל בִּרְכָּיו מוּל מִטַּת רַבּוֹ, לָקַח אֶת…
6. הוּא יָדַע שֶׁהַדָּבָר הֲכִי חָשׁוּב בָּעוֹלָם, וְהַכָּבוֹד הֲכִי…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “כְּשֶׁרַבִּי אוֹשַׁעְיָה הָיָה נִכְנָס, הָיוּ פָּנָיו שֶׁל הָרַב הָעִוֵּר מוּאָרוֹת בְּשִׂמְחָה.”.
מכתב: כתבו לר' אושעיא הגדול בן ר' חמא בעקבות “נַפְשׁוֹ שֶׁל הַצַּדִּיק אָמְרָה לוֹ שֶׁהוּא עָשָׂה טָעוּת.”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “מוֹרִי וְרַבִּי..." לָחַשׁ רַבִּי אוֹשַׁעְיָה בְּקוֹל דּוֹמֵעַ וְנִשְׁבָּר, "אַל יִכְעַס עָלַי אֲדוֹנִי כִּי הַיּוֹם לֹא…”.
דיווח: כתבו ידיעה קצרה שכותרתה “המורה העוור — מכבד אחרים”.

המחזה

רגע מן הסיפורהדמויותמה יאמרו או יעשו?
רַבִּי אוֹשַׁעְיָה הַגָּדוֹל, נְשִׂיא הַתּוֹרָה שֶׁמֵּאוֹת אֲנָשִׁים מְחַכִּים רַק לִשְׁמוֹעַ…
