רבי עקיבא ורחל — תחילת לימודו

דף עבודה בעקבות רבי עקיבא
שם: כיתה:

המעשה — הסיפור המלא

להלן עיבוד חינוכי, רגשי ומרתק לסיפורם המופלא של רבי עקיבא ורחל אשתו ותחילת לימודו , המעובד במיוחד לבנים בכיתות ב'-ד' (חינוך דתי-לאומי). הסיפור שם דגש על כוח ההתמדה, מסירות הנפש של רחל, וההבנה שאף פעם לא מאוחר להתחיל.

הסיפור מנוקד במלואו כנדרש.

פְּתִיחָה: הָרוֹעֶה בְּאַרְמוֹן הֶעָשִׁיר

יְלָדִים, עִצְמוּ אֶת הָעֵינַיִם וְדַמְיְנוּ אַרְמוֹן עֲנָק בִּירוּשָׁלַיִם שֶׁל יְמֵי קֶדֶם. זֶהוּ בֵּיתוֹ שֶׁל אֶחָד מִגְּדוֹלֵי עֲשִׁירֵי הָעִיר, אָדָם מְכֻבָּד מְאֹד שֶׁנִּקְרָא "כַּלְבָּא שָׂבוּעַ". מַדּוּעַ קָרְאוּ לוֹ כָּךְ? כִּי מִמִּטְבְּחוֹ עָלוּ תָּמִיד רֵיחוֹת שֶׁל תַּבְשִׁילִים חַמִּים, בָּשָׂר הַנִּצְלֶה עַל הָאֵשׁ וְלֶחֶם טָרִי. כָּל אָדָם שֶׁנִּכְנַס אֵלָיו רָעֵב "כַּכֶּלֶב", הָיָה יוֹצֵא מִמֶּנּוּ "שָׂבוּעַ" וּמְאֻשָּׁר.

לְכַלְבָּא שָׂבוּעַ הָיוּ שָׂדוֹת רַבִּים וַעֲדָרִים עֲצוּמִים שֶׁל כְּבָשִׂים לְבָנוֹת. מִי שֶׁשָּׁמַר עַל הַכְּבָשִׂים הָאֵלֶּה הָיָה רוֹעֵה צֹאן פָּשׁוּט וְעָנִי שֶׁשְּׁמוֹ עֲקִיבָא. עֲקִיבָא הָיָה כְּבָר כִּמְעַט בֶּן אַרְבָּעִים. הוּא מֵעוֹלָם לֹא לָמַד תּוֹרָה. הוּא לֹא יָדַע לִקְרֹא, לֹא יָדַע לִכְתֹּב, וַאֲפִלּוּ אֶת אוֹתִיּוֹת הָאָלֶף-בֵּית לֹא הִכִּיר. אַךְ בְּתוֹךְ לִבּוֹ פְנִימָה, עֲקִיבָא הָיָה אָדָם מְלֵא עֲנָוָה, אִישׁ אֱמֶת, שֶׁמִּדּוֹתָיו טוֹבוֹת וּמְעֻלּוֹת.

לְכַלְבָּא שָׂבוּעַ הָיְתָה בַּת יְחִידָה, צַדֶּקֶת וַחֲכָמָה מְאֹד, וּשְׁמָהּ רָחֵל. רָחֵל חָיְתָה בְּעֹשֶׁר שֶׁאִי אֶפְשָׁר לְתָאֵר – בְּגָדִים מִמֶּשִׁי, תַּכְשִׁיטִים מִזָּהָב וּמְשָׁרְתוֹת סְבִיבָהּ. חֲתָנִים עֲשִׁירִים רַבִּים בִּקְּשׁוּ לָשֵׂאת אוֹתָהּ לְאִשָּׁה, אַךְ הִיא הִבִּיטָה דַּוְקָא עַל עֲקִיבָא, רוֹעֵה הַצֹּאן שֶׁל אָבִיהָ. הִיא הִבְחִינָה בַּצְּנִיעוּת שֶׁלּוֹ, בַּיֹּשֶׁר שֶׁלּוֹ, וְאָמְרָה בְּלִבָּהּ: "אִם רַק יֵלֵךְ לִלְמֹד תּוֹרָה, הַמִּדּוֹת הַטּוֹבוֹת שֶׁלּוֹ יַהַפְכוּ אוֹתוֹ לְגָדוֹל בְּיִשְׂרָאֵל!"

נִגְּשָׁה רָחֵל לְעֲקִיבָא וְשָׁאֲלָה בְּקוֹל עָדִין: "עֲקִיבָא, אִם אֶנָּשֵׂא לְךָ, הַאִם תֵּלֵךְ לִלְמֹד תּוֹרָה?" עֲקִיבָא הֻפְתַּע, אַךְ הִנְהֵן בְּרֹאשׁוֹ וְאָמַר: "כֵּן."

הֵם הִתְחַתְּנוּ בַּסֵּתֶר, בְּשֶׁקֶט בְּשֶׁקֶט, בְּלִי תִּזְמֹרֶת וּבְלִי אֲרוּחוֹת פְּאֵר.

הַגֵּרוּשׁ וְהַקַּשׁ שֶׁבַּמַּתְבֵּן

כְּשֶׁשָּׁמַע כַּלְבָּא שָׂבוּעַ שֶׁבִּתּוֹ הָאֲהוּבָה הִתְחַתְּנָה עִם רוֹעֵה הַצֹּאן – כַּעֲסוֹ בָּעַר כְּאֵשׁ! הוּא לֹא הֵבִין כֵּיצַד וִתְּרָה עַל הַכֹּל לְמַעַן עַם הָאָרֶץ עָנִי. בְּקוֹל רוֹעֵם הוּא הִכְרִיז וְנָדַר נֶדֶר: "צְאִי מִבֵּיתִי! אַתְּ לֹא תְּקַבְּלִי אֲפִלּוּ פְּרוּטָה אַחַת מִכָּל הָרְכוּשׁ שֶׁלִּי לְעוֹלָמִים!"

רָחֵל וְעֲקִיבָא מָצְאוּ אֶת עַצְמָם קוֹפְאִים בַּחוּץ, בְּלִי קוֹרַת גַּג, בְּלִי כֶּסֶף לְלֶחֶם. הֵם הָלְכוּ וּמָצְאוּ מַתְבֵּן – מַחְסָן יָשָׁן שֶׁל קַשׁ שֶׁמְּיֹעָד לְמַאֲכַל סוּסִים. שָׁם הֵם יָשְׁנוּ. בַּבֹּקֶר הַקַּר, נָשְׁבָה רוּחַ מִבַּעַד לַסְּדָקִים בַּקִּירוֹת. עֲקִיבָא קָם, מַבִּיט בְּאִשְׁתּוֹ שֶׁהָיְתָה בַּת עֲשִׁירִים וְעַכְשָׁו יְשֵׁנָה עַל הָרִצְפָּה. הוּא הֵחֵל לְלַקֵּט וּלְהוֹצִיא בַּעֲדִינוּת אֶת חֲתִיכוֹת הַקַּשׁ שֶׁנִּדְבְּקוּ בְּשַׂעֲרוֹתֶיהָ. עֵינָיו דָּמְעוּ וְהוּא לָחַשׁ: "רָחֵל תִּרְאִי... עַכְשָׁו מָלֵא קַשׁ בְּשַׂעֲרוֹתַיִךְ. אַךְ אִלּוּ הָיָה לִי כֶּסֶף, הָיִיתִי קוֹנֶה לָךְ תַּכְשִׁיט עֲטֶרֶת זָהָב שֶׁנִּקְרָא 'יְרוּשָׁלַיִם שֶׁל זָהָב'."

פִּתְאוֹם נִשְׁמְעוּ דְּפִיקוֹת חַלָּשׁוֹת בַּדֶּלֶת. בַּפֶּתַח עָמַד עָנִי רוֹעֵד מִקֹּר, בְּגָדָיו קְרוּעִים. "אֲנָשִׁים טוֹבִים," הִתְחַנֵּן, "אִשְׁתִּי יָלְדָה הַלַּיְלָה תִּינוֹק קָטָן. אֵין לָנוּ כְּלוּם, גַּם לֹא קַשׁ לְהַשְׁכִּיב עָלָיו אֶת הַתִּינוֹק. אֶפְשָׁר קְצָת תֶּבֶן?"

עֲקִיבָא מִיָּד אָסַף כַּמָּה אֲגֻדּוֹת קַשׁ וְנָתַן לָאִישׁ. הוּא חִיֵּךְ אֶל רָחֵל וְאָמַר: "רוֹאָה? הַקָּדוֹשׁ בָּרוּךְ הוּא מַרְאֶה לָנוּ שֶׁיֵּשׁ אֲנָשִׁים שֶׁגַּם קַשׁ אֵין לָהֶם." וִילָדִים, אֲנַחְנוּ יוֹדְעִים שֶׁאוֹתוֹ אִישׁ לֹא הָיָה אֶלָּא אֵלִיָּהוּ הַנָּבִיא שֶׁהִגִּיעַ לְנַחֵם אוֹתָם!

סוֹד הַמַּיִם עַל הַסֶּלַע

הִגִּיעַ הַזְּמַן לְקַיֵּם אֶת הַהַבְטָחָה. עֲקִיבָא עָמַד לָצֵאת לִלְמֹד תּוֹרָה, אַךְ רַגְלָיו הָיוּ כְּבֵדוֹת. הוּא הָיָה בֶּן אַרְבָּעִים! אֵיךְ אִישׁ מְבֻגָּר שֶׁיָּדָיו קָשׁוֹת מֵעֲבוֹדָה עִם כְּבָשִׂים יָכוֹל לָשֶׁבֶת עִם יְלָדִים קְטַנִּים וְלִלְמֹד 'אָלֶף-בֵּית'? הַמֹּחַ שֶׁלּוֹ כְּבָר סָגוּר. זֶה מְאֻחָר מִדַּי.

מִתּוֹךְ מַחֲשָׁבוֹת אֵלֶּה, עָצַר עֲקִיבָא לְיַד בְּאֵר מַיִם.

הוּא הִבִּיט עַל סֶלַע גָּדוֹל וּמֻצָּק שֶׁעָמַד שָׁם, וְרָאָה מַשֶּׁהוּ מַדְהִים. בְּמֶרְכַּז הַסֶּלַע הַקָּשֶׁה הָיָה... חוֹר, נֶקֶב עָמֹק וְחָלָק!

"מִי חָקַק אֶת הָאֶבֶן הַזּוֹ?" שָׁאַל עֲקִיבָא בִּפְלִיאָה.

עָנוּ לוֹ הָאֲנָשִׁים שֶׁהָיוּ שָׁם וְצָחֲקוּ: "עֲקִיבָא, אֵינְךָ יוֹדֵעַ? הַמַּיִם שֶׁנּוֹפְלִים עָלֶיהָ תָּדִיר, יָמִים עַל יָמִים, הֵם חָפְרוּ בָּהּ. מָה, לֹא קָרָאתָ אֶת הַפָּסוּק 'אֲבָנִים שָׁחֲקוּ מַיִם'?"

בְּאוֹתוֹ רֶגַע, הַקָּדוֹשׁ בָּרוּךְ הוּא הֵאִיר אֶת לִבּוֹ שֶׁל עֲקִיבָא. הוּא אָמַר לְעַצְמוֹ שְׁאֵלָה שֶׁשִּׁנְּתָה אֶת הָעוֹלָם כֻּלּוֹ:

"מָה הֲכִי קָשֶׁה? אֶבֶן. מָה הֲכִי רַךְ? מַיִם. אִם מַיִם שֶׁהֵם כָּל כָּךְ רַכִּים, יְכוֹלִים לַחְפֹּר חוֹר בְּאֶבֶן כָּל כָּךְ קָשָׁה כִּי הֵם מְטַפְטְפִים עוֹד שְׁנִיָּה וְעוֹד שְׁנִיָּה, וְלֹא מוְותְּרִים... הֲרֵי הַתּוֹרָה שֶׁהִיא אֵשׁ נֶחֱצֶבֶת, בְּוַדַּאי תַּחְדֹּר לַלֵּב הַבָּשָׂר שֶׁלִּי!"

שְׁתֵּים עֶשְׂרֵה הַשָּׁנִים שֶׁל רָחֵל

עֲקִיבָא חָזַר לְרָחֵל בְּעֵינַיִם בּוֹרְקוֹת. הוּא עָזַב לְבֵית הַסֵּפֶר וְהִתְחִיל לִלְמֹד – הוּא וּבְנוֹ הַקָּטָן מִסֵּפֶר אֶחָד. אַחַר כָּךְ, הָלַךְ לִישִׁיבָתָם שֶׁל חֲכָמֵי הַדּוֹר, רַבִּי אֱלִיעֶזֶר וְרַבִּי יְהוֹשֻׁעַ, שָׁם לָמַד שְׁתֵּים עֶשְׂרֵה שָׁנִים בִּרְצִיפוּת! הוּא חָי בְּעֹנִי, אַךְ לֹא וִתֵּר עַל תָּג בַּתּוֹרָה, מְדַיֵּק וְלָמַד יוֹמָם וָלַיְלָה.

וּבֵינְתַיִם, רָחֵל הִמְשִׁיכָה לִחְיוֹת בְּעֹנִי נוֹרָא, מְכַבֶּסֶת בְּגָדִים שֶׁל אֲחֵרִים בַּיָּדַיִם, שְׂעָרָהּ הַיָּפֶה מְדֻבְלָל, מְחַכָּה לְבַעֲלָהּ הַצַּדִּיק שֶׁיִּגְדַּל בַּתּוֹרָה.

אַחֲרֵי שְׁתֵּים עֶשְׂרֵה שָׁנִים, חָזַר עֲקִיבָא – עַכְשָׁו כְּבָר "רַבִּי עֲקִיבָא" עִם 12,000 תַּלְמִידִים – לְבֵיתוֹ. הוּא עָמַד מֵאֲחוֹרֵי דֶּלֶת הַמַּתְבֵּן כְּדֵי לִשְׁמֹעַ הַאִם רָחֵל שְׁלוֹמָהּ טוֹב. לְפֶתַע, שָׁמַע שָׁכֵן רָשָׁע לוֹעֵג לָהּ:

"כַּמָּה יָפֶה שֶׁאָבִיךְ נָדַר עָלַיִךְ אֶת הַנֶּדֶר הַזֶּה! הִתְחַתַּנְתְּ עִם אִישׁ זָר שֶׁעָזַב אוֹתָךְ עֲנִיָּה כְּבָר שְׁתֵּים עֶשְׂרֵה שָׁנִים!"

עָנְתָה רָחֵל, בְּקוֹל עָדִין שֶׁבָּקַע מִלֵּב אוֹהֵב: "אִם מִישֶׁהוּ הָיָה שׁוֹמֵעַ לִי, הָיִיתִי רוֹצָה שֶׁיֵּשֵׁב וְיִלְמַד שְׁתֵּים עֶשְׂרֵה שָׁנִים נוֹסָפוֹת! כִּי אֵין דָּבָר חָשׁוּב יוֹתֵר מֵהַתּוֹרָה."

רַבִּי עֲקִיבָא, שֶׁשָּׁמַע שֶׁהִיא נוֹתֶנֶת לוֹ רְשׁוּת, לֹא דָּפַק בַּדֶּלֶת. הוּא הִסְתּוֹבֵב אָחוֹרָה, וְחָזַר לְבֵית הַמִּדְרָשׁ לְעוֹד שְׁתֵּים עֶשְׂרֵה שָׁנִים!

שֶׁלִּי וְשֶׁלָּכֶם – שֶׁלָּהּ הוּא

חָלְפוּ בְּסַךְ הַכֹּל 24 שָׁנִים. עַכְשָׁו, לֹא מַגִּיעַ רוֹעֵה צֹאן פָּשׁוּט. מַגִּיעַ הָרַב הֲכִי גָּדוֹל שֶׁהָיָה לְעַם יִשְׂרָאֵל דּוֹרוֹת רַבִּים! רַבִּי עֲקִיבָא! וּמֵאֲחוֹרָיו – נַחְשׁוֹל שֶׁל אֲנָשִׁים, תּוֹרָה שֶׁאִי אֶפְשָׁר לִסְפֹּר – עֶשְׂרִים וְאַרְבָּעָה אֶלֶף תַּלְמִידִים! כָּל יְרוּשָׁלַיִם רָעֲדָה!

רָחֵל שָׁמְעָה שֶׁגְּדוֹל הַדּוֹר מַגִּיעַ, וְיָצְאָה לִקְרָאתוֹ. בְּגָדֶיהָ הָיוּ קְרוּעִים וּבְלוּיִים, פָּנֶיהָ עֲיֵפוֹת מִשָּׁנִים שֶׁל עֲבוֹדָה קָשָׁה.

"אַל תֵּלְכִי כָּכָה לִפְגֹּשׁ אֶת הָרַב הַגָּדוֹל!" לָחֲשׁוּ לָהּ נְשׁוֹת הָעִיר, וְרָצוּ לְהַשְׁאִיל לָהּ בְּגָדִים יָפִים. אַךְ רָחֵל הֵשִׁיבָה בְּפָסוּק: "יוֹדֵעַ צַדִּיק נֶפֶשׁ בְּהֶמְתּוֹ." רַבִּי עֲקִיבָא שֶׁלִּי, יַכִּיר אוֹתִי וְיֹאהַב אוֹתִי כְּפִי שֶׁאֲנִי.

רָחֵל נִדְחֲקָה בֵּין הַהֲמוֹנִים וְרָצָה לִפֹּל עַל רַגְלָיו שֶׁל הָרַב בִּבְכִי גָּדוֹל שֶׁל הִתְרַגְּשׁוּת. הַתַּלְמִידִים הַצְּעִירִים שֶׁלֹּא הִכִּירוּ אוֹתָהּ, רָאוּ אִשָּׁה פְּשׁוּטָה בִּבְגָדִים עֲלוּבִים דּוֹחֶפֶת אֶת כֻּלָּם כְּדֵי לְהַגִּיעַ לָרַב. הֵם בָּאוּ לְהַדּוֹף אוֹתָהּ לְאָחוֹר.

בְּאוֹתוֹ רֶגַע, רַבִּי עֲקִיבָא זִהָה אוֹתָהּ. עֵינָיו זָלְגוּ דְּמָעוֹת. הוּא הֵרִים אֶת הַיָּד, וְדַמְמַת אֱלֹקִים מִלְּאָה אֶת הָרְחוֹב. עֶשְׂרִים וְאַרְבָּעָה אֶלֶף תַּלְמִידִים שָׁתְקוּ וְהִקְשִׁיבוּ.

וְאָז זָעַק רַבִּי עֲקִיבָא בְּקוֹל שֶׁנִּשְׁמַע לַדּוֹרוֹת כֻּלָּם:

"הַנִּיחוּ לָהּ! חַכְמֵי יִשְׂרָאֵל, דְּעוּ לָכֶם: שֶׁלִּי וְשֶׁלָּכֶם... שֶׁלָּהּ הוּא!"

הַתּוֹרָה שֶׁלִּי, וְהַתּוֹרָה שֶׁאֲנִי מְלַמֵּד אֶתְכֶם – הַכֹּל שַׁיָּךְ לָהּ, בִּזְכוּת מְסִירוּת הַנֶּפֶשׁ שֶׁלָּהּ!

כַּאֲשֶׁר שָׁמַע כַּלְבָּא שָׂבוּעַ שֶׁאָדָם כָּל כָּךְ גָּדוֹל הִגִּיעַ לָעִיר, הוּא בָּא אֵלָיו בְּכָבוֹד, וּבִקֵּשׁ שֶׁיַּתִּיר לוֹ אֶת הַנֶּדֶר שֶׁנָּדַר עַל בִּתּוֹ. הוּא כְּבָר בָּכָה וְהִתְחָרֵט שֶׁגֵּרֵשׁ אוֹתָהּ.

"אִלּוּ יָדַעְתָּ שֶׁעֲקִיבָא יִהְיֶה תַּלְמִיד חָכָם... הָיִיתָ נוֹדֵר?" שָׁאַל אוֹתוֹ הָרַב.

"חָלִילָה!" בָּכָה כַּלְבָּא שָׂבוּעַ, "אִם הָיָה לוֹמֵד אֲפִלּוּ הֲלָכָה אַחַת!"

הִבִּיט בּוֹ הָרַב בְּחִיּוּךְ מֵאִיר וְאָמַר: "אֲנִי הוּא עֲקִיבָא חֲתָנְךָ."

כַּלְבָּא שָׂבוּעַ נָפַל מִתַּדְהֵמָה וְנָשַׁק לְרַגְלָיו. הַשָּׁלוֹם חָזַר לַמִּשְׁפָּחָה, וְכַלְבָּא שָׂבוּעַ נָתַן לָהֶם אֶת חֲצִי רְכוּשׁוֹ.

וּמָה עָשָׂה רַבִּי עֲקִיבָא בַּכֶּסֶף הָרִאשׁוֹן? יְלָדִים, צַדִּיק דּוֹבֵר אֱמֶת תָּמִיד מוֹדֶה עַל הַטּוֹב. הַדָּבָר הָרִאשׁוֹן שֶׁעָשָׂה הָיָה לְקַיֵּם אֶת הַהַבְטָחָה מֵהַמַּתְבֵּן בַּלַּיְלָה הַקַּר הַהוּא. הוּא יָצַק לְרָחֵל תַּכְשִׁיט עָצוּם מִזָּהָב טָהוֹר, שֶׁצִּיּוּר יְרוּשָׁלַיִם מֻטְבָּע עָלָיו, וְהִנִּיחַ עַל רֹאשָׁהּ, בִּזְכוּת אַהֲבָתָהּ לַתּוֹרָה.

א. שְׁאֵלַת פְּתִיחָה (Hook):

"יְלָדִים, הַאִם מִישֶׁהוּ מִכֶּם נִסָּה פַּעַם לִשְׁבֹּר בּוּל עֵץ אוֹ אֶבֶן? עִם מָה נִסִּיתֶם, עִם פַּטִּישׁ? (תֵּן לַיְלָדִים לַעֲנוֹת). הַאִם מִישֶׁהוּ פַּעַם שָׁבַר אֶבֶן עֲנָקִית... עִם טִפָּה קְטַנָּה שֶׁל מַיִם?

הַיּוֹם נִשְׁמַע עַל אִישׁ בֶּן אַרְבָּעִים מַמָּשׁ, שֶׁעוֹד לֹא יָדַע לִקְרֹא מִלָּה אַחַת, אֲבָל מָה שֶׁהוּא לָמַד מִטִּפּוֹת שֶׁל מַיִם הָפַךְ אוֹתוֹ לְאָדָם הֲכִי גָּדוֹל בְּעַם יִשְׂרָאֵל..."

ב. שְׁאֵלַת סִיּוּם:

"אִם רַבִּי עֲקִיבָא הָיָה יוֹשֵׁב לְיַד הַבְּאֵר וְאוֹמֵר "עֲזֹב, אֲנִי זָקֵן מִדַּי, מְאֻחָר מִדַּי" – כַּמָּה תַּלְמִידִים הָיוּ לוֹ? רֶבָע עַם יִשְׂרָאֵל הָיָה נֶעֱלָם! הַכֹּחַ הֲכִי גָּדוֹל הוּא פָּשׁוּט – כֹּחַ הַהַתְמָדָה."

ג. הַמְלָצוֹת לְמַצֶּגֶת (תְמוּנוֹת מְדֻיָּקוֹת וּצְנוּעוֹת!):

ד. הֶעָרוֹת לְבִצּוּעַ וְסְטוֹרִיטֶלִינְג לַמְּסַפֵּר:

ה. פְּעִילוּת מְסַכֶּמֶת (הַמְלָצָה):

"נִיסוּי טִפּוֹת הַמַּיִם". לְהָבִיא לַכִּתָּה אֶבֶן גִּיר גְּדוֹלָה מְעַט כֵּהָה אוֹ גּוּשׁ חֵמָר קָשֶׁה. לָקַחַת טַפְטֶפֶת עִם מַיִם, וּלְטַפְטֵף עַל נְקֻדָּה אַחַת. לָתֵת לַיְלָדִים לְטַפְטֵף גַּם (אֶפְשָׁר בְּמֶשֶׁךְ כָּל הַשָּׁבוּעַ בַּכִּתָּה הַטִּפְטוּף יַמְשִׁיךְ). תּוֹךְ כַּמָּה יָמִים יֵרָאֶה שֶׁקַע קָטָן, לְהַדְגָּמַת כֹּחַ הַהַתְמָדָה.

ו. מְקוֹרוֹת (הָוֶי אוֹמֵר תּוֹרַת אֱמֶת):

מקורות: המקורות מופיעים בקובץ המקורי.

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “להלן עיבוד חינוכי, רגשי ומרתק לסיפורם המופלא של רבי עקיבא ורחל אשתו ותחילת לימודו , המעובד במיוחד לבנים בכיתות ב'-ד' (חינוך דתי-לאומי).” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “הִיא הִבְחִינָה בַּצְּנִיעוּת שֶׁלּוֹ, בַּיֹּשֶׁר שֶׁלּוֹ, וְאָמְרָה בְּלִבָּהּ: "אִם רַק יֵלֵךְ לִלְמֹד תּוֹרָה, הַמִּדּוֹת הַטּוֹבוֹת שֶׁלּוֹ יַהַפְכוּ…”?
ג. כיצד הגיב רבי עקיבא למצב המתואר: “אֵין לָנוּ כְּלוּם, גַּם לֹא קַשׁ לְהַשְׁכִּיב עָלָיו אֶת הַתִּינוֹק.”?
ד. מה השתנה בין “הוּא חָי בְּעֹנִי, אַךְ לֹא וִתֵּר עַל תָּג בַּתּוֹרָה, מְדַיֵּק וְלָמַד יוֹמָם וָלַיְלָה.” לבין “אַל תֵּלְכִי כָּכָה לִפְגֹּשׁ אֶת הָרַב הַגָּדוֹל!" לָחֲשׁוּ לָהּ נְשׁוֹת הָעִיר, וְרָצוּ לְהַשְׁאִיל לָהּ בְּגָדִים…”?
ה. כיצד הגיע הסיפור אל האירוע: “הוּא יָצַק לְרָחֵל תַּכְשִׁיט עָצוּם מִזָּהָב טָהוֹר, שֶׁצִּיּוּר יְרוּשָׁלַיִם מֻטְבָּע עָלָיו, וְהִנִּיחַ עַל רֹאשָׁהּ, בִּזְכוּת אַהֲבָתָהּ לַתּוֹרָה.”?

בסיפור לא התרחש הדבר הבא: עָנְתָה רָחֵל, בְּקוֹל עָדִין שֶׁבָּקַע מִלֵּב אוֹהֵב: "אִם מִישֶׁהוּ הָיָה שׁוֹמֵעַ לִי, הָיִיתִי רוֹצָה שֶׁיֵּשֵׁב וְיִלְמַד שְׁתֵּים עֶשְׂרֵה שָׁנִים נוֹסָפוֹת!□□
הַתּוֹרָה שֶׁלִּי, וְהַתּוֹרָה שֶׁאֲנִי מְלַמֵּד אֶתְכֶם – הַכֹּל שַׁיָּךְ לָהּ, בִּזְכוּת מְסִירוּת הַנֶּפֶשׁ שֶׁלָּהּ!□□

3. מה היה ראשון?

כתבו ליד כל אירוע מספר מ־1 עד 5 לפי סדר התרחשותו בסיפור.

□ אִם מַיִם שֶׁהֵם כָּל כָּךְ רַכִּים, יְכוֹלִים לַחְפֹּר חוֹר בְּאֶבֶן כָּל כָּךְ קָשָׁה כִּי הֵם מְטַפְטְפִים עוֹד שְׁנִיָּה וְעוֹד שְׁנִיָּה, וְלֹא…
□ להלן עיבוד חינוכי, רגשי ומרתק לסיפורם המופלא של רבי עקיבא ורחל אשתו ותחילת לימודו , המעובד במיוחד לבנים בכיתות ב'-ד' (חינוך דתי-לאומי).
□ תּוֹךְ כַּמָּה יָמִים יֵרָאֶה שֶׁקַע קָטָן, לְהַדְגָּמַת כֹּחַ הַהַתְמָדָה.
□ בְּקוֹל רוֹעֵם הוּא הִכְרִיז וְנָדַר נֶדֶר: "צְאִי מִבֵּיתִי!
□ רָחֵל נִדְחֲקָה בֵּין הַהֲמוֹנִים וְרָצָה לִפֹּל עַל רַגְלָיו שֶׁל הָרַב בִּבְכִי גָּדוֹל שֶׁל הִתְרַגְּשׁוּת.

לפני ואחרי

מה קרה לפני: “אִם מַיִם שֶׁהֵם כָּל כָּךְ רַכִּים, יְכוֹלִים לַחְפֹּר חוֹר בְּאֶבֶן כָּל כָּךְ קָשָׁה כִּי הֵם מְטַפְטְפִים עוֹד שְׁנִיָּה וְעוֹד שְׁנִיָּה, וְלֹא…”?
מה קרה אחרי: “רָחֵל נִדְחֲקָה בֵּין הַהֲמוֹנִים וְרָצָה לִפֹּל עַל רַגְלָיו שֶׁל הָרַב בִּבְכִי גָּדוֹל שֶׁל הִתְרַגְּשׁוּת.”?

סיבה ותוצאה מתוך המעשה

האירוע הנתוןמה גרם לו?מה הייתה תוצאתו?
רָחֵל וְעֲקִיבָא מָצְאוּ אֶת עַצְמָם קוֹפְאִים בַּחוּץ, בְּלִי קוֹרַת גַּג, בְּלִי כֶּסֶף לְלֶחֶם.
עָנְתָה רָחֵל, בְּקוֹל עָדִין שֶׁבָּקַע מִלֵּב אוֹהֵב: "אִם מִישֶׁהוּ הָיָה שׁוֹמֵעַ לִי, הָיִיתִי רוֹצָה שֶׁיֵּשֵׁב וְיִלְמַד שְׁתֵּים עֶשְׂרֵה שָׁנִים…

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “רבי עקיבא ורחל — תחילת לימודו” על רבי עקיבא? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “הַמַּיִם שֶׁנּוֹפְלִים עָלֶיהָ תָּדִיר, יָמִים עַל יָמִים, הֵם חָפְרוּ בָּהּ.”.
ג. אילו רבי עקיבא לא היה פועל כפי שמתואר ב־“אַל תֵּלְכִי כָּכָה לִפְגֹּשׁ אֶת הָרַב הַגָּדוֹל!" לָחֲשׁוּ לָהּ נְשׁוֹת הָעִיר, וְרָצוּ לְהַשְׁאִיל לָהּ בְּגָדִים יָפִים.” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “הוּא יָצַק לְרָחֵל תַּכְשִׁיט עָצוּם מִזָּהָב טָהוֹר, שֶׁצִּיּוּר יְרוּשָׁלַיִם מֻטְבָּע עָלָיו, וְהִנִּיחַ עַל רֹאשָׁהּ,…” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. להלן עיבוד חינוכי, רגשי ומרתק לסיפורם המופלא של רבי עקיבא ורחל אשתו…
2. הֵם הִתְחַתְּנוּ בַּסֵּתֶר, בְּשֶׁקֶט בְּשֶׁקֶט, בְּלִי תִּזְמֹרֶת…
3. עֲקִיבָא עָמַד לָצֵאת לִלְמֹד תּוֹרָה, אַךְ רַגְלָיו הָיוּ כְּבֵדוֹת.
4. הוּא עָמַד מֵאֲחוֹרֵי דֶּלֶת הַמַּתְבֵּן כְּדֵי לִשְׁמֹעַ הַאִם רָחֵל…
5. וְאָז זָעַק רַבִּי עֲקִיבָא בְּקוֹל שֶׁנִּשְׁמַע לַדּוֹרוֹת כֻּלָּם:
6. תּוֹךְ כַּמָּה יָמִים יֵרָאֶה שֶׁקַע קָטָן, לְהַדְגָּמַת כֹּחַ…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “אֵין לָנוּ כְּלוּם, גַּם לֹא קַשׁ לְהַשְׁכִּיב עָלָיו אֶת הַתִּינוֹק.”.
מכתב: כתבו לרבי עקיבא בעקבות “עָנְתָה רָחֵל, בְּקוֹל עָדִין שֶׁבָּקַע מִלֵּב אוֹהֵב: "אִם מִישֶׁהוּ הָיָה שׁוֹמֵעַ לִי, הָיִיתִי רוֹצָה שֶׁיֵּשֵׁב וְיִלְמַד…”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “הַתּוֹרָה שֶׁלִּי, וְהַתּוֹרָה שֶׁאֲנִי מְלַמֵּד אֶתְכֶם – הַכֹּל שַׁיָּךְ לָהּ, בִּזְכוּת מְסִירוּת הַנֶּפֶשׁ שֶׁלָּהּ!”.
דיווח: כתבו ידיעה קצרה שכותרתה “רבי עקיבא ורחל — תחילת לימודו”.

המחזה

אַל תֵּלְכִי כָּכָה לִפְגֹּשׁ אֶת הָרַב הַגָּדוֹל!" לָחֲשׁוּ לָהּ נְשׁוֹת הָעִיר, וְרָצוּ…

7. חיבור לחיים מתוך המעשה
