מי גנב את הכוס?

דף עבודה בעקבות מר זוטרא
שם: כיתה:

המעשה — הסיפור המלא

פְּתִיחָה: אוֹרְחִים שֶׁל כָּבוֹד

יְלָדִים, עִצְמוּ אֶת הָעֵינַיִם וְנַסּוּ לְדַמְיֵן כְּפָר עַתִּיק בְּבָבֶל שֶׁל יְמֵי הָאֲמוֹרָאִים.

הָרְחוֹבוֹת הָיוּ עֲשׂוּיִים עָפָר וַאֲבָנִים, וְרֵיחַ שֶׁל עֲצֵי תְּמָרִים מִלֵּא אֶת הָאֲוִיר הַחַם. בְּאוֹתוֹ יוֹם, הִגִּיעוּ לַכְּפָר אוֹרְחִים חֲשׁוּבִים מְאֹד. הָיָה זֶה הָאֲמוֹרָא הַגָּדוֹל, מָר זוּטְרָא הֶחָסִיד , יַחַד עִם קְבוּצָה גְּדוֹלָה שֶׁל תַּלְמִידָיו. הֵם הָלְכוּ בַּדֶּרֶךְ יָמִים רַבִּים כְּדֵי לְקַיֵּם מִצְוָה חֲשׁוּבָה שֶׁהָיְתָה בְּאוֹתָהּ עִיר.

בָּעִיר הַזּוֹ גָּר יְהוּדִי עָשִׁיר וְטוֹב לֵב. כַּאֲשֶׁר שָׁמַע שֶׁמָּר זוּטְרָא וְתַלְמִידָיו נִמְצָאִים בָּעִיר, שָׂמַח שִׂמְחָה עֲצוּמָה. "אֵיזֶה כָּבוֹד הוּא זֶה לְאָרֵחַ אֶת עֲמַלֵי הַתּוֹרָה בְּבֵיתִי!" אָמַר לְאִשְׁתּוֹ. הֵם מִהֲרוּ לַהֲכִין סְעוּדַת מְלָכִים.

כַּאֲשֶׁר נִכְנְסוּ מָר זוּטְרָא וְתַלְמִידָיו לַבַּיִת, עָלָה בְּאַפָּם רֵיחַ נִפְלָא שֶׁל לֶחֶם חַם, תַּבְשִׁילֵי בָּשָׂר וּדְגִים עֲרֵבִים לַחֵךְ. הַשֻּׁלְחָן הָיָה עָרוּךְ בְּכָל טוּב.

וּבְמֶרְכַּז הַשֻּׁלְחָן, לְצַד הָרַב, נִצְּבָה כּוֹס יְקָרָה וּמְיֻחֶדֶת מְאֹד. זוֹ הָיְתָה כּוֹס שֶׁל כֶּסֶף טָהוֹר, שֶׁעֻטְּרָה בְּצִיּוּרִים עֲדִינִים. לְאוֹר הַנֵּרוֹת שֶׁדָּלְקוּ בַּחֶדֶר, הַכּוֹס זָרְחָה וְנִצְנְצָה כְּמוֹ יַהֲלוֹם. בַּעַל הַבַּיִת הוֹצִיא אוֹתָהּ בִּמְיֻחָד מִן הַכַּסֶּפֶת שֶׁלּוֹ, כְּדֵי שֶׁמָּר זוּטְרָא יוּכַל לִשְׁתּוֹת מִמֶּנָּה וּלְבָרֵךְ עָלֶיהָ.

הַהֵעָלְמוּת הַמִּסְתּוֹרִית

הַסְּעוּדָה הָיְתָה רוּחָנִית וּמְרוֹמֶמֶת. שִׁירֵי קֹדֶשׁ נִשְׁמְעוּ בַּחֶדֶר, וְדִבְרֵי תּוֹרָה עֲמוּקִים וּמְתוּקִים מִדְּבַשׁ יָצְאוּ מִפִּיו שֶׁל מָר זוּטְרָא.

אַךְ כַּאֲשֶׁר הִסְתַּיְּמָה הַסְּעוּדָה וְהָאוֹרְחִים קָמוּ לִתְפִלַּת עַרְבִית, קָרָה דָּבָר מוּזָר וּמַפְחִיד.

בַּעֲלַת הַבַּיִת נִגְּשָׁה לַשֻּׁלְחָן כְּדֵי לְפַנּוֹת אֶת הַכֵּלִים וּלְהַחְזִיר אֶת הַכּוֹס הַיְקָרָה לַמָּקוֹם הַבָּטוּחַ. הִיא הִבִּיטָה יָמִינָה, הִבִּיטָה שְׂמֹאלָה... אַךְ הַכּוֹס אֵינֶנָּה!

"הֵיכָן כּוֹס הַכֶּסֶף שֶׁלָּנוּ?" שָׁאֲלָה אֶת בַּעֲלָהּ, וְקוֹלָהּ רָעַד.

"הַכּוֹס?" עָנָה הַבַּעַל בִּמְבוּכָה עֲצוּמָה, "הֲרֵי לִפְנֵי רְגָעִים אֲחָדִים הִיא עָמְדָה כָּאן עַל הַשֻּׁלְחָן! לְאָן הִיא נֶעֶלְמָה?".

מָר זוּטְרָא וְתַלְמִידָיו, שֶׁשָּׁמְעוּ אֶת הַהֶמְיָה וְהַלַּחַשׁ, הִצְטָרְפוּ מִיָּד לַחִיפּוּשִׂים. הֵם הִבִּיטוּ תַּחַת הַשֻּׁלְחָנוֹת, הֵזִיזוּ מַפּוֹת, חיפְּשׂוּ בַּפִּנּוֹת הַחֲשׁוּכוֹת שֶׁל הַחֶדֶר. אַךְ לַשָּׁוְא. כּוֹס הַכֶּסֶף כְּאִלּוּ נָמֹגָה וְהִתְאַדְּתָה בָּאֲוִיר!

בַּעַל הַבַּיִת הֵלִיט אֶת פָּנָיו בְּיָדָיו וְאָמַר בְּעֶצֶב נִכָּר: "אֲנִי חוֹשֵׁשׁ... אֲנִי חוֹשֵׁשׁ שֶׁהַכּוֹס פָּשׁוּט נִגְנְבָה!".

שֶׁקֶט שֶׁל פַּחַד

יְלָדִים, אֵיזוֹ הַרְגָּשָׁה נוֹרָאָה זוֹ הָיְתָה. הַשֶּׁקֶט בַּחֶדֶר הָיָה כָּבֵד כְּמוֹ אֶבֶן.

מָר זוּטְרָא הֶחָסִיד הִרְגִּישׁ צַעַר עָמֹק פּוֹלֵחַ אֶת לִבּוֹ. לֹא רַק עַל שֶׁוְיָהּ שֶׁל הַכּוֹס. הוּא הֵבִין אֶת הַסַּכָּנָה הַגְּדוֹלָה: הֲרֵי בַּעַל הַבַּיִת נָעַל אֶת הַדְּלָתוֹת! שׁוּם אָדָם זָר לֹא נִכְנַס לַבַּיִת זִלַּת הָרַב וְתַלְמִידָיו. מִי יָכוֹל לִהְיוֹת הַגַּנָּב? הַחֲשָׁד הַנּוֹרָא נוֹפֵל, חָלִילָה, עַל לְוֹמְדֵי הַתּוֹרָה הַקְּדוֹשָׁה!

הַתַּלְמִידִים יָשְׁבוּ עַל מְקוֹמָם, חִוְּרִים כַּסִּיד. אִישׁ הִבִּיט בְּרֵעֵהוּ, קוֹוִים בְּלִבָּם שֶׁרַבָּם יִמְצָא פִּתְרוֹן וְיַגִּיד שֶׁזּוֹ סְתָם טָעוּת. הֲרֵי כָּל אֶחָד מֵהֶם צַדִּיק, אֵיךְ אֶפְשָׁר לַחְשֹׁד בָּהֶם?

אָדָם רָגִיל הָיָה מִתְחִיל לִצְעֹק, אוֹ דּוֹרֵשׁ מִכֻּלָּם לְרוֹקֵן אֶת הַכִּיסִים. אַךְ מָר זוּטְרָא סֵרֵב לַעֲשׂוֹת זֹאת. חִיפּוּשׂ בְּכִיסָיו שֶׁל תַּלְמִיד חָכָם הוּא הַלְבָּנַת פָּנִים שֶׁאֵין כְּמוֹתָהּ! "הַקָּדוֹשׁ בָּרוּךְ הוּא בְּוַדַּאי יָאִיר אֶת עֵינַי," מִלְמֵל הָרַב, וְהוֹרָה לְכֻלָּם לָלֶכֶת לִישׁוֹן בַּחֲדָרִים שֶׁהֵכִינוּ עֲבוּרָם.

קֶרֶן שֶׁמֶשׁ שֶׁל תַּגְלִית

לְמָחֳרָת בַּבֹּקֶר, עִם עֲלוֹת הַשַּׁחַר, קָמוּ הַתַּלְמִידִים מִשְּׁנָתָם כְּדֵי לְהִתְכּוֹנֵן לִתְפִלַּת שַׁחֲרִית. הֵם מִהֲרוּ אֶל הֶחָצֵר, שָׁם עָמַד כַּד גָּדוֹל שֶׁל מַיִם צוֹנְנִים לִנְטִילַת יָדַיִם.

שְׁלִיק... שְׁלָאק... קוֹל הַמַּיִם הַנִּשְׁפָּכִים מִן הַנַּטְלָה נִשְׁמַע בֶּחָצֵר. מָר זוּטְרָא עָמַד בַּצַּד, מְסַדֵּר אֶת טַלִּיתוֹ, וְהִבִּיט בְּתַלְמִידָיו בְּחִיבָּה, כֵּיצַד הֵם מִזְדָּרְזִים לְקַיֵּם אֶת הַמִּצְוָה.

אֲבָל פִּתְאוֹם – עֵינָיו שֶׁל הָרַב צֻמְצְמוּ. הַמַּבָּט שֶׁלּוֹ נִהְיָה חַד כְּתַעַר. הוּא הִבְחִין בְּמַשֶּׁהוּ שֶׁאִישׁ לֹא חָשַׁב שֶׁהוּא חָשׁוּב.

אֶחָד הַתַּלְמִידִים נָטַל אֶת יָדָיו. הַיָּדַיִם שֶׁלּוֹ הָיוּ רְטוּבוֹת וְקָרוֹת. הוּא חִיפֵּשׂ סְבִיבוֹ אֶת מַגַּבְתּוֹ הָאִישִׁית לְנַגֵּב אוֹתָן, אַךְ לֹא מָצָא אוֹתָהּ. יְלָדִים, מָה עוֹשֶׂה יְהוּדִי שֶׁמּוֹצֵא עַצְמוֹ בְּלִי מַגֶּבֶת? מְמַתִּין. תּוֹפֵס בָּאֲוִיר. מוֹצֵא נְיָר אוֹ מְבַקֵּשׁ רְשׁוּת.

אַךְ מָה עָשָׂה אוֹתוֹ אָדָם? לְלֹא שׁוּם בּוּשָׁה, הוּא נִגַּשׁ לַמְּעִיל שֶׁל חֲבֵרוֹ – שֶׁהָיָה מֻנָּח עַל כִּסֵּא לְצִדּוֹ – תָּפַס בְּשׁוּלֵי הַבֶּגֶד, וּמִיָּד... נִגֵּב אֶת יָדָיו הָרְטוּבוֹת בַּבֶּגֶד הַיָּבֵשׁ וְהַנָּקִי שֶׁל חֲבֵרוֹ!

בָּרֶגַע שֶׁרָאָה זֹאת, הִשְׁתַּנּוּ פָּנָיו שֶׁל מָר זוּטְרָא. לִבּוֹ זִנֵּק. רוּחַ הַקֹּדֶשׁ אִירָה אֶת שִׂכְלוֹ וְהוּא אָמַר לְעַצְמוֹ הֲלָכָה פְּסוּקָה: "זֶהוּ הַגַּנָּב!"

מַדּוּעַ? כִּי מִי שֶׁלֹּא אִכְפַּת לוֹ לְלַכְלֵךְ וּלְקַלְקֵל אֶת הַבֶּגֶד שֶׁל חֲבֵרוֹ כְּאִלּוּ זֶה רְכוּשׁוֹ הַפְּרָטִי – מוֹכִיחַ שֶׁאֵין לוֹ שׁוּם כָּבוֹד לְמָמוֹן שֶׁל אֲנָשִׁים אֲחֵרִים. וּמִי שֶׁלֹּא אִכְפַּת לוֹ מִמָּמוֹן שֶׁל אֲחֵרִים, הוּא אָדָם הַמְסֻגָּל לִגְנֹב!

הָאֱמֶת מִתְגַּלֵּית

בְּעוֹד הַתַּלְמִידִים הוֹלְכִים לִתְפִלַּת בֹּקֶר, קָרָא מָר זוּטְרָא לְאוֹתוֹ אָדָם הַצִּדָּה, לְתוֹךְ חֲדָרוֹ הַפְּרָטִי כְּדֵי שֶׁלֹּא לְבַיְּשׁוֹ מוּל כֻּלָּם.

הוּא עָמַד מוּלוֹ בְּקוֹמָה זְקוּפָה וּבְקוֹל עָמֹק, אָמַר לוֹ בְּחֻמְרָה: "אַתָּה הוּא זֶה שֶׁלָּקַח אֶת הַכּוֹס! אַל תְּנַסֶּה לְהַסְתִּיר מִמֶּנִּי. הַדָּבָר בָּרוּר לִי!".

הָאִישׁ נִבְהַל. "רַבִּי! אֲנִי?! חָלִילָה!"

אַךְ מָר זוּטְרָא הִמְשִׁיךְ: "רָאִיתִי מָה עָשִׂיתָ הַבֹּקֶר. רָאִיתִי אוֹתְךָ מְנַגֵּב אֶת יָדֶיךָ עַל בֶּגֶד חֲבֵרְךָ בְּלִי רְשׁוּת וּבְלִי שֶׁיִּהְיֶה אִכְפַּת לְךָ. מִי שֶׁנּוֹגֵעַ כָּךְ בְּבֶגֶד זָר – מְסֻגָּל גַּם לְשַׁלְשֵׁל כּוֹס כֶּסֶף לְכִיסוֹ!".

הָאִישׁ חִוֵּר וְאָדַם הַחֲלִיפוֹת. הוּא נִסָּה לְהִתְנַצֵּל: "רַבִּי... רַק לֹא מָצָאתִי אֶת מַגַּבְתִּי... מָה כְּבָר עָשִׂיתִי? הַבֶּגֶד קְצָת נִרְטַב, הוּא יִתְיַבֵּשׁ בַּשֶּׁמֶשׁ. לֹא קָרָה לוֹ נֶזֶק שֶׁל מַמָּשׁ!".

אַךְ מָר זוּטְרָא לֹא וִתֵּר. הוּא שָׁלַח אֶת עוֹזְרָיו לִלְחֹץ וְלַחְקֹר אוֹתוֹ בְּחָכְמָה, עַד שֶׁהַגַּנָּב נִשְׁבַּר. הוּא פָּרַץ בִּבְכִי, וְהוֹדָה עַל הָאֱמֶת. "אָכֵן," לָחַשׁ, "הַיֵּצֶר הָרַע גָּבַר עָלַי. חָמַדְתִּי אֶת הַכּוֹס הַיָּפָה, וְהִטְמַנְתִּי אוֹתָהּ בְּתִיקִי".

הַגַּנָּב נֶאֱלַץ לְהַחְזִיר אֶת הַכּוֹס לְבַעַל הַבַּיִת הַשָּׂמֵחַ, וְקִבֵּל אֶת עָנְשׁוֹ. שְׁמָם שֶׁל תַּלְמִידֵי הַחֲכָמִים הָאֲמִתִּיִּים נִנְקָה וְטֹהַר, וְהַכֹּל הוֹדוּ לְחוֹכְמָתוֹ וּלְעֵינָיו הַבּוֹחֲנוֹת שֶׁל מָר זוּטְרָא שֶׁהֵבִין אֶת נֶפֶשׁ הָאָדָם דֶּרֶךְ פְּרָט כָּל כָּךְ קָטָן.

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “יְלָדִים, עִצְמוּ אֶת הָעֵינַיִם וְנַסּוּ לְדַמְיֵן כְּפָר עַתִּיק בְּבָבֶל שֶׁל יְמֵי הָאֲמוֹרָאִים.” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “זוֹ הָיְתָה כּוֹס שֶׁל כֶּסֶף טָהוֹר, שֶׁעֻטְּרָה בְּצִיּוּרִים עֲדִינִים.”?
ג. כיצד הגיב מר זוטרא למצב המתואר: “בַּעַל הַבַּיִת הֵלִיט אֶת פָּנָיו בְּיָדָיו וְאָמַר בְּעֶצֶב נִכָּר: "אֲנִי חוֹשֵׁשׁ...”?
ד. מה השתנה בין “חִיפּוּשׂ בְּכִיסָיו שֶׁל תַּלְמִיד חָכָם הוּא הַלְבָּנַת פָּנִים שֶׁאֵין כְּמוֹתָהּ!” לבין “נִגֵּב אֶת יָדָיו הָרְטוּבוֹת בַּבֶּגֶד הַיָּבֵשׁ וְהַנָּקִי שֶׁל חֲבֵרוֹ!”?
ה. כיצד הגיע הסיפור אל האירוע: “רָאִיתִי אוֹתְךָ מְנַגֵּב אֶת יָדֶיךָ עַל בֶּגֶד חֲבֵרְךָ בְּלִי רְשׁוּת וּבְלִי שֶׁיִּהְיֶה אִכְפַּת לְךָ.”?

3. מה היה ראשון?

כתבו ליד כל אירוע מספר מ־1 עד 5 לפי סדר התרחשותו בסיפור.

□ אָדָם רָגִיל הָיָה מִתְחִיל לִצְעֹק, אוֹ דּוֹרֵשׁ מִכֻּלָּם לְרוֹקֵן אֶת הַכִּיסִים.
□ יְלָדִים, עִצְמוּ אֶת הָעֵינַיִם וְנַסּוּ לְדַמְיֵן כְּפָר עַתִּיק בְּבָבֶל שֶׁל יְמֵי הָאֲמוֹרָאִים.
□ שְׁמָם שֶׁל תַּלְמִידֵי הַחֲכָמִים הָאֲמִתִּיִּים נִנְקָה וְטֹהַר, וְהַכֹּל הוֹדוּ לְחוֹכְמָתוֹ וּלְעֵינָיו הַבּוֹחֲנוֹת שֶׁל מָר זוּטְרָא שֶׁהֵבִין אֶת…
□ שִׁירֵי קֹדֶשׁ נִשְׁמְעוּ בַּחֶדֶר, וְדִבְרֵי תּוֹרָה עֲמוּקִים וּמְתוּקִים מִדְּבַשׁ יָצְאוּ מִפִּיו שֶׁל מָר זוּטְרָא.
□ בָּרֶגַע שֶׁרָאָה זֹאת, הִשְׁתַּנּוּ פָּנָיו שֶׁל מָר זוּטְרָא.

לפני ואחרי

מה קרה לפני: “אָדָם רָגִיל הָיָה מִתְחִיל לִצְעֹק, אוֹ דּוֹרֵשׁ מִכֻּלָּם לְרוֹקֵן אֶת הַכִּיסִים.”?
מה קרה אחרי: “בָּרֶגַע שֶׁרָאָה זֹאת, הִשְׁתַּנּוּ פָּנָיו שֶׁל מָר זוּטְרָא.”?

סיבה ותוצאה מתוך המעשה

האירוע הנתוןמה גרם לו?מה הייתה תוצאתו?
אַךְ כַּאֲשֶׁר הִסְתַּיְּמָה הַסְּעוּדָה וְהָאוֹרְחִים קָמוּ לִתְפִלַּת עַרְבִית, קָרָה דָּבָר מוּזָר וּמַפְחִיד.
מָר זוּטְרָא עָמַד בַּצַּד, מְסַדֵּר אֶת טַלִּיתוֹ, וְהִבִּיט בְּתַלְמִידָיו בְּחִיבָּה, כֵּיצַד הֵם מִזְדָּרְזִים לְקַיֵּם אֶת הַמִּצְוָה.

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “מי גנב את הכוס?” על מר זוטרא? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “הַחֲשָׁד הַנּוֹרָא נוֹפֵל, חָלִילָה, עַל לְוֹמְדֵי הַתּוֹרָה הַקְּדוֹשָׁה!”.
ג. אילו מר זוטרא לא היה פועל כפי שמתואר ב־“נִגֵּב אֶת יָדָיו הָרְטוּבוֹת בַּבֶּגֶד הַיָּבֵשׁ וְהַנָּקִי שֶׁל חֲבֵרוֹ!” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “רָאִיתִי אוֹתְךָ מְנַגֵּב אֶת יָדֶיךָ עַל בֶּגֶד חֲבֵרְךָ בְּלִי רְשׁוּת וּבְלִי שֶׁיִּהְיֶה אִכְפַּת לְךָ.” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. יְלָדִים, עִצְמוּ אֶת הָעֵינַיִם וְנַסּוּ לְדַמְיֵן כְּפָר עַתִּיק…
2. לְאוֹר הַנֵּרוֹת שֶׁדָּלְקוּ בַּחֶדֶר, הַכּוֹס זָרְחָה וְנִצְנְצָה…
3. מָר זוּטְרָא הֶחָסִיד הִרְגִּישׁ צַעַר עָמֹק פּוֹלֵחַ אֶת לִבּוֹ.
4. הֵם מִהֲרוּ אֶל הֶחָצֵר, שָׁם עָמַד כַּד גָּדוֹל שֶׁל מַיִם צוֹנְנִים…
5. כִּי מִי שֶׁלֹּא אִכְפַּת לוֹ לְלַכְלֵךְ וּלְקַלְקֵל אֶת הַבֶּגֶד שֶׁל…
6. שְׁמָם שֶׁל תַּלְמִידֵי הַחֲכָמִים הָאֲמִתִּיִּים נִנְקָה וְטֹהַר,…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “בַּעַל הַבַּיִת הֵלִיט אֶת פָּנָיו בְּיָדָיו וְאָמַר בְּעֶצֶב נִכָּר: "אֲנִי חוֹשֵׁשׁ...”.
מכתב: כתבו למר זוטרא בעקבות “מָר זוּטְרָא עָמַד בַּצַּד, מְסַדֵּר אֶת טַלִּיתוֹ, וְהִבִּיט בְּתַלְמִידָיו בְּחִיבָּה, כֵּיצַד הֵם מִזְדָּרְזִים לְקַיֵּם אֶת…”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “וּמִי שֶׁלֹּא אִכְפַּת לוֹ מִמָּמוֹן שֶׁל אֲחֵרִים, הוּא אָדָם הַמְסֻגָּל לִגְנֹב!”.
דיווח: כתבו ידיעה קצרה שכותרתה “מי גנב את הכוס?”.

המחזה

רגע מן הסיפורהדמויותמה יאמרו או יעשו?
אַךְ כַּאֲשֶׁר הִסְתַּיְּמָה הַסְּעוּדָה וְהָאוֹרְחִים קָמוּ לִתְפִלַּת עַרְבִית, קָרָה דָּבָר…
נִגֵּב אֶת יָדָיו הָרְטוּבוֹת בַּבֶּגֶד הַיָּבֵשׁ וְהַנָּקִי שֶׁל חֲבֵרוֹ!

7. חיבור לחיים מתוך המעשה
