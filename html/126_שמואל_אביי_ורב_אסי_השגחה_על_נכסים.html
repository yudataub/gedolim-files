השגחה על נכסים

דף עבודה בעקבות שמואל, אביי ורב אסי
שם: כיתה:

המעשה — הסיפור המלא

להלן עיבוד חינוכי, רגשי ומרתק מתוך התלמוד הבבלי (מסכת חולין, דף ק"ה עמוד א'), המביא את סיפורם של האמוראים הקדושים שמואל, אביי ורב אסי, בנושא "השגחה על נכסים" (שמירה על הרכוש). הסיפור נכתב בהתאמה מדויקת לבנים בכיתות ב'-ד' בחינוך הדתי-לאומי, על פי כל כללי הסטוריטלינג הנדרשים.

פְּתִיחָה: קוֹלוֹ שֶׁל הָאָמוֹרָא שְׁמוּאֵל

יְלָדִים, אֲנַחְנוּ בְּבָבֶל שֶׁל יְמֵי קֶדֶם. בְּתוֹךְ יְשִׁיבַת 'נְהַרְדְּעָא' הַגְּדוֹלָה, אֶפְשָׁר הָיָה לִשְׁמֹעַ אֶת קוֹל הַלִּמּוּד נִשָּׂא לְמֵרָחוֹק, מָתוֹק וְעַז כְּמַיִם אֲדִירִים.

בְּרֹאשׁ הַיְשִׁיבָה יָשַׁב הָאָמוֹרָא הַקָּדוֹשׁ, הָרַב שְׁמוּאֵל. הָרַב שְׁמוּאֵל יָדַע אֶת סוֹדוֹת הַתּוֹרָה וַאֲפִלּוּ אֶת סוֹדוֹת הַכּוֹכָבִים בַּשָּׁמַיִם, אַךְ הוּא יָדַע דָּבָר חָשׁוּב לֹא פָּחוֹת – אֵיךְ יְהוּדִי צָרִיךְ לְהִתְנַהֵג בְּחַיֵּי הַיּוֹם־יוֹם שֶׁלּוֹ בָּאָרֶץ.

בְּאוֹתוֹ בֹּקֶר, הֵרִים שְׁמוּאֵל אֶת קוֹלוֹ. תַּלְמִידֵי הַחֲכָמִים שָׁתְקוּ וְהִקְשִׁיבוּ.

"דְּעוּ לָכֶם, תַּלְמִידַי הַיְקָרִים," אָמַר שְׁמוּאֵל, "אָדָם לֹא יָכוֹל לִהְיוֹת צַדִּיק רַק בְּבֵית הַכְּנֶסֶת. אָדָם חַיָּב לִשְׁמֹר עַל מָה שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא נָתַן לוֹ. מָאן דְּסָיֵיר נִכְסֵיהּ כָּל יוֹמָא – מַשְׁכַּח אִסְתֵּירָא! ".

יְלָדִים, מָה פֵּרוּשׁ הַמִּלִּים הָאֵלּוּ בַּאֲרָמִית? "מִי שֶׁמְּסַיֵּר וּבוֹדֵק אֶת הַשָּׂדוֹת שֶׁלּוֹ וְאֶת הָרְכוּשׁ שֶׁלּוֹ בְּכָל יוֹם, מוֹצֵא בְּוַדָּאוּת מַטְבֵּעַ קָטָן שֶׁל כֶּסֶף ('אִסְתֵּירָא')". תַּלְמִיד חָכָם הוּא אָדָם אַחֲרָאִי. הוּא לֹא אוֹמֵר "יִהְיֶה בְּסֵדֶר, ה' יַעֲזֹר", אֶלָּא פּוֹתֵחַ עֵינַיִם, שׁוֹמֵר, מְסַדֵּר, וְשָׁם רְכוּשׁוֹ מִתְבָּרֵךְ.

אַבַּיֵי בּוֹדֵק אֶת הַשָּׂדֶה

אֶחָד מִגְּדוֹלֵי הַתַּלְמִידִים הָיָה אַבַּיֵי. כְּשֶׁשָּׁמַע אַבַּיֵי אֶת דִּבְרֵי רַבּוֹ, הוּא חָקַק אוֹתָם עַל לִבּוֹ. "אִם שְׁמוּאֵל אָמַר – חַיָּבִים לְקַיֵּם!" אָמַר אַבַּיֵי לְעַצְמוֹ.

וְאָכֵן, בְּכָל יוֹם, אַחֲרֵי שְׁעוֹת הַלִּמּוּד, הָיָה אַבַּיֵי סוֹגֵר אֶת הַגְּמָרָא הַגְּדוֹלָה שֶׁלּוֹ, וְיוֹצֵא אֶל הַשָּׂדֶה.

הַשֶּׁמֶשׁ הַזְּהוּבָה שֶׁל שְׁעַת מִנְחָה לִטְּפָה אֶת הָעֵצִים. רֵיחַ עָמֹק וְנָעִים שֶׁל אֲדָמָה רְטֻבָּה עָלָה מִן הָרְגָבִים. אַבַּיֵי הָלַךְ בֵּין הַשּׁוּרוֹת, בָּדַק עֵץ, בָּדַק עָלֶה, רָאָה שֶׁהַגָּדֵר תְּקִינָה.

וְהִנֵּה, מֵרָחוֹק, הוּא קוֹלֵט אֶת הָאָרִיס שֶׁלּוֹ (הַפּוֹעֵל שֶׁשָּׂכַר כְּדֵי לְטַפֵּל בַּשָּׂדֶה). הַפּוֹעֵל סָחַב עַל כְּתֵפוֹ חֲבִילָה כְּבֵדָה וַעֲנָקִית שֶׁל עֵצִים יָפִים וַחֲתוּכִים לְהַסָּקָה. אַבַּיֵי הִתְקָרֵב וּבָחַן אֶת הָעֵצִים. הַחֲסְפּוּס שֶׁל הַגֶּזַע, כַּמּוּת הָעֵץ... מַשֶּׁהוּ כָּאן לֹא הִסְתַּדֵּר! אֵלֶּה הָיוּ עֵצִים שֶׁהָאָרִיס בִּכְלָל לֹא הָיָה אָמוּר לָקַחַת.

אַבַּיֵי הַקָּדוֹשׁ לֹא כָּעַס וְלֹא צָעַק. בְּקוֹל שָׁקֵט מְלֵא חָכְמָה הוּא נִגַּשׁ לַפּוֹעֵל וְשָׁאַל: "תַּגִּיד לִי, מַיִם רַבִּים שֶׁהִשְׁקִיתָ אֶתְמוֹל אֶת הַשָּׂדֶה... שֶׁל מִי הָעֵצִים הַלָּלוּ עַל כְּתֵפְךָ?"

הַפּוֹעֵל שָׁמַט עֵינַיִם הַמֻּפְתָּעוֹת. "אֶה... שֶׁל מָר, כְּבוֹד הָרַב. אֵלֶּה עֵצִים שֶׁלְּךָ. רַק לָקַחְתִּי אוֹתָם קְצָת לְהַעֲבִיר לְמָקוֹם אַחֵר..." מִלְמֵל הַפּוֹעֵל בִּמְבוּכָה, כְּשֶׁהוּא תּוֹפֵס שֶׁנִּתְפַּס מְנַסֶּה לָקַחַת הַרְבֵּה מֵעֵבֶר לַמֻּתָּר לוֹ.

חִיֵּךְ אַבַּיֵי. "כְּבָר קִדְּמוּ אוֹתְךָ חֲכָמִים! הֵם כְּבָר יָדְעוּ שֶׁאִם לֹא אַשְׁגִּיחַ עַל רְכוּשִׁי, הַדְּבָרִים יֵעָלְמוּ. הַנַּח לָעֵצִים".

אַבַּיֵי צָעַד הַבַּיְתָה וְלִבּוֹ מָלֵא הוֹדָיָה לַה'. "דִּבְרֵי שְׁמוּאֵל מְדֻיָּקִים. בָּדַקְתִּי אֶת שָׂדִי, וְהִנֵּה נִצְּלוּ שָׁוְויָם שֶׁל הָעֵצִים שֶׁכִּמְעַט נִלְקְחוּ. מָצָאתִי הַיּוֹם 'אִסְתֵּירָא'!".

רַב אַסִּי לֹא מוֹצֵא דָּבָר

אֲבָל, יְלָדִים... לֹא אֵצֶל כֻּלָּם זֶה הָיָה מִיָּדִי.

חָכָם צַדִּיק אַחֵר בַּיְשִׁיבָה הָיָה רַב אַסִּי. גַּם רַב אַסִּי שָׁמַע אֶת דִּבְרֵי שְׁמוּאֵל: "מִי שֶׁבּוֹדֵק אֶת נְכָסָיו יוֹם־יוֹם, יִמְצָא מַטְבֵּעַ שֶׁל בְּרָכָה!". עָזַב רַב אַסִּי הַכֹּל, יָצָא אֶל חֶלְקַת הַשָּׂדֶה שֶׁלּוֹ.

יוֹם רִאשׁוֹן הוּא הָלַךְ. הַשֶּׁמֶשׁ קָפְחָה עַל גַּבּוֹ. אַגְלֵי זֵעָה צָצוּ עַל מִצְחוֹ. הוּא בָּדַק אֶת הָעֵצִים. נָדָה. הַכֹּל כְּשׁוּרָה. אַף פּוֹעֵל לֹא לָקַח כְּלוּם, אַף גָּדֵר לֹא נִפְרְצָה.

יוֹם שֵׁנִי, הוּא שׁוּב טָרַח וְהָלַךְ. נָדָה.

עָבַר שָׁבוּעַ שָׁלֵם! שְׁבוּעַיִם! רַב אַסִּי מְכַתֵּת רַגְלַיִם שָׁעָה שְׁלֵמָה בְּכָל יוֹם, וְהַשָּׂדֶה שָׁקֵט וְנָחוֹם. אֵין אֲפִלּוּ אֲגוֹרָה חֲדָשָׁה שֶׁנִּמְצֵאת מִתַּחַת לַעֲלֵה גֶּפֶן.

"אֵיפֹה הַמַּטְבֵּעַ שֶׁהִבְטִיחַ שְׁמוּאֵל הַגָּדוֹל?" חָשַׁב רַב אַסִּי בְּלִבּוֹ. "דִּבְרֵי תּוֹרָה אֵינָם שֶׁקֶר חָלִילָה! אֲבָל מַדּוּעַ אֲנִי, אַחֲרֵי שֶׁטָּרַחְתִּי כָּל כָּךְ הַרְבֵּה וְהִשְׁגַּחְתִּי בְּמוֹ עֵינַי, וְהִזַּעְתִּי הַרְבֵּה... לֹא רָאִיתִי שׁוּם רֶוַח? לָמָּה הַקָּדוֹשׁ בָּרוּךְ הוּא לֹא שׁוֹלֵחַ לִי אֶת הַבְּרָכָה שֶׁהֻבְטְחָה עַל יְדֵי רַבּוֹתֵינוּ?". זֶה הָיָה מְאֹד מְתַסְכֵּל לְהַשְׁקִיעַ כָּל יוֹם וְלַחְשֹׁב שֶׁזֶּה אולי זְמַן מְבֻזְבָּז מִלִּמּוּד תּוֹרָה.

אֲבָל רַב אַסִּי לֹא וִתֵּר. דִּבְרֵי חֲכָמִים הֵם קֹדֶשׁ. בְּכָל עֶרֶב, הוּא צָעַד אֶל הַשָּׂדֶה.

הַצִּנּוֹר הַמִּתְפּוֹצֵץ!

הִגִּיעַ זְמַן בֵּין הַשְּׁמָשׁוֹת. הַשָּׁמַיִם נִצְבְּעוּ בִּצְבָעִים שֶׁל כָּחֹל-כֵּהֶה וּסְגֹל מִצַּד אֶחָד, וְכָתֹם-אָדֹם מִן הַצַּד הַשֵּׁנִי. הַצְּרָצָרִים הֵחֵלּוּ לְנַסֵּר אֶת הָאֲוִיר בְּשִׁירָתָם הָאַחַטִית.

רַב אַסִּי עָשָׂה אֶת הַסִּיּוּר הַקָּבוּעַ שֶׁלּוֹ מִסָּבִיב לַחֶלְקָה הַחַקְלָאִית.

לְפֶתַע...

בּוּּּּּּּּּּּּּּּוּם!!! קוֹל מַחֲרִיד בָּקַע מִקְּצֵה הַשָּׂדֶה! קוֹל רַעַם שֶׁל מַשֶּׁהוּ נִשְׁבָּר וְנִסְדָּק, וְאָז – שְׁשְשְׁשְׁשְׁשְׁ! – קוֹל אַדִּיר שֶׁל מַיִם רַבִּים.

רַב אַסִּי רָץ מַהֵר אֶל מְקוֹם הָרַעַשׁ וְחָשַׁכוּ עֵינָיו!

צִנּוֹר הַהַשְׁקָיָה הָרָאשִׁי, אֲשֶׁר הוֹבִיל מַיִם מֵהַנָּהָר אֶל הַתְּעָלוֹת, הִתְפּוֹצֵץ מִלַּחַץ! חֹר עֲנָק נִפְעַר בְּדֹפֶן חֵמָר שֶׁל תְּעָלַת הַמַּיִם ('בִּדְקָא דְּמַיָּא'). זֶרֶם מַיִם אֵימְתָנִי וְחוּם מִבּוֹץ חָזַר לִשְׁטֹף הַכֹּל!

אִם הַמַּיִם הָאֵלֶּה יַמְשִׁיכוּ לִשְׁצֹף בְּעָצְמָה הַזּוֹ לְעוֹד חֲצִי שָׁעָה, כָּל הַשָּׂדֶה יֻצַּף! הַשֳּׁרָשִׁים שֶׁל הַפֵּרוֹת יֵרָקְבוּ! כָּל יְבוּל הַשָּׁנָה שְׁלֵמָה יִהְיֶה מוּכָן לִזְרִיקָה. רְכוּשׁ עָצוּם עוֹמֵד לָרֶדֶת לְטִמְיוֹן.

אֵין אִישׁ בַּשָּׂדֶה. הַפּוֹעֲלִים כְּבָר שָׁבוּ לְבָתֵּיהֶם. מָה עוֹשִׂים?! מָה שָׂמִים כְּדֵי לִסְתֹּם אֶת הַחֹר עַד שֶׁיָּבוֹאוּ לְתַקֵּן?

רַב אַסִּי שָׁלַח יָד אֶל דּוֹפְנוֹת הַתְּעָלָה וְנִסָּה לְכַסּוֹת בְּעָפָר וּבַאֲבָנִים, אֲבָל הַמַּיִם הַשּׁוֹצְפִים מִיָּד פּוֹרְרִים אֶת הַבּוֹץ הָרַךְ. יָדָיו שֶׁל הָרַב קָפְאוּ בַּמַּיִם הַקָּרִים, אֲבָל הוּא לֹא נִכְנַע. הוּא הֵבִין שֶׁהוּא חַיָּב מַשֶּׁהוּ עָבֶה, חָזָק וְסוֹפֵג מַיִם...

רַב אַסִּי גִּלְגֵּל בִּמְהִירוּת אֶת מְעִיל הַצֶּמֶר הַחַם, גִּלְגֵּל וְגִלְגֵּל כְּמוֹ כַּדּוּר, רָכַן קָדִימָה וּדְחַף אֶת כָּל מְעִילוֹ לְתוֹךְ פִּרְצַת הַמַּיִם!

הַצֶּמֶר הֶעָבֶה סָפַג אֶת הַמַּיִם, תָּפַח, וְהוּא שֶׁעָצַר וְאָטַם אֶת הַחֹר!

אֶלֶף מַטְבְּעוֹת שֶׁל בְּרָכָה!

רַב אַסִּי הִבִּיט עַל הַשָּׂדֶה הַנִּפְלָא שֶׁלּוֹ. כָּל הַגְּפָנִים וְעֵצֵי הַפְּרִי עָמְדוּ בִּמְקוֹמָם יְצִיבִים. אַף לֹא אֶחָד נֶהֱרַס!

וְאָז... לְפֶתַע צָץ עַל פָּנָיו הַקְּדוֹשׁוֹת חִיּוּךְ גָּדוֹל, חִיּוּךְ עָצוּם! הֶאָרָה יָרְדָה אֶל לִבּוֹ.

הוּא הֵרִים אֶת עֵינָיו לַשָּׁמַיִם וְאָמַר: "רִבּוֹנוֹ שֶׁל עוֹלָם! כַּמָּה מְדֻיָּקִים וּנְכוֹנִים דִּבְרֵי רַבִּי וּמוֹרִי שְׁמוּאֵל בַּיְשִׁיבָה! הוּא אָמַר שֶׁמִּי שֶׁבּוֹדֵק נְכָסָיו כָּל יוֹם, מוֹצֵא מַטְבֵּעַ כֶּסֶף!

אֲנִי חִפַּשְׂתִּי כָּל הַיָּמִים הָאֵלֶּה לִמְצֹא כֶּסֶף זָרוּק מִתַּחַת לַאֲבָנִים, אֲבָל... הַמַּטְבֵּעַ הָאֲמִתִּי הוּא מָה שֶׁנִּצַּלְתִּי בִּזְכוּת הַהַשְׁגָּחָה וְהָאַחֲרָיוּת! אִלְמָלֵא הָיִיתִי בָּא לְכָאן הַיּוֹם בְּסִיּוּר הָעֶרֶב שֶׁלִּי לִבְדֹּק, הַשָּׂדֶה כֻּלּוֹ הָיָה הוֹלֵךְ לְאִבּוּד. מַחֲצִית שֶׁל הַכֶּסֶף שֶׁלִּי הָיְתָה כָּלָה! הוּא אָמַר שֶׁאֶמְצָא מַטְבֵּעַ, וַאֲנִי... מָצָאתִי הַיּוֹם אַלְפֵי מַטְבְּעוֹת שֶׁנִּשְׁאֲרוּ בְּכִיסִי!"

הַקָּדוֹשׁ בָּרוּךְ הוּא דּוֹאֵג לָנוּ וּמְבָרֵךְ אֶת מַעֲשֵׂה יָדֵינוּ – אַךְ אָנוּ צְרִיכִים מִצִּדֵּנוּ לִשְׁמֹר וְלִפְקֹחַ עַיִן אוֹהֶבֶת עַל כָּל הַשֶּׁפַע שֶׁהוּא שָׁלַח לָנוּ.

ME (אֲנַחְנוּ וְהַגִּבּוֹר):

רַב אַסִּי יָכוֹל הָיָה לְהִשָּׁאֵר בַּבַּיִת. הוּא יָכוֹל הָיָה לוֹמַר "ה' כְּבָר יִשְׁמֹר עַל הַשָּׂדֶה שֶׁלִּי, מָה אֲנִי צָרִיךְ לָלֶכֶת בַּחֹם?". אֲבָל רַב אַסִּי יָדַע שֶׁאִם קִבַּלְנוּ מַשֶּׁהוּ - חוֹבָתֵנוּ לְהַשְׁגִּיחַ עָלָיו!

3. מה היה ראשון?

כתבו ליד כל אירוע מספר מ־1 עד 5 לפי סדר התרחשותו בסיפור.

□ רַב אַסִּי רָץ מַהֵר אֶל מְקוֹם הָרַעַשׁ וְחָשַׁכוּ עֵינָיו!
□ להלן עיבוד חינוכי, רגשי ומרתק מתוך התלמוד הבבלי (מסכת חולין, דף ק"ה עמוד א'), המביא את סיפורם של האמוראים הקדושים שמואל, אביי ורב אסי, בנושא "השגחה על…
□ אִם הוּא הִשְׁגִּיחַ הֵיטֵב – הַ'מַּטְבֵּעַ' שֶׁלּוֹ הוּא שֶׁהַקַּלְמָר שֶׁלּוֹ נִשְׁאַר מָלֵא!
□ הַפּוֹעֵל סָחַב עַל כְּתֵפוֹ חֲבִילָה כְּבֵדָה וַעֲנָקִית שֶׁל עֵצִים יָפִים וַחֲתוּכִים לְהַסָּקָה.
□ הוּא הֵרִים אֶת עֵינָיו לַשָּׁמַיִם וְאָמַר: "רִבּוֹנוֹ שֶׁל עוֹלָם!

לפני ואחרי

מה קרה לפני: “רַב אַסִּי רָץ מַהֵר אֶל מְקוֹם הָרַעַשׁ וְחָשַׁכוּ עֵינָיו!”?
מה קרה אחרי: “הוּא הֵרִים אֶת עֵינָיו לַשָּׁמַיִם וְאָמַר: "רִבּוֹנוֹ שֶׁל עוֹלָם!”?

סיבה ותוצאה מתוך המעשה

האירוע הנתוןמה גרם לו?מה הייתה תוצאתו?
אֵלֶּה הָיוּ עֵצִים שֶׁהָאָרִיס בִּכְלָל לֹא הָיָה אָמוּר לָקַחַת.
תַּלְמִיד חָכָם צָרִיךְ לִשְׁמֹר עַל מַלְבּוּשָׁיו שֶׁיִּהְיוּ נְקִיִּים לְלֹא כֶּתֶם, אַךְ עַכְשָׁו, פִּקּוּחַ מָמוֹן וְנֶזֶק עָצוּם נֶגֶד עֵינָיו!

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “השגחה על נכסים” על שמואל, אביי ורב אסי? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “הַשָּׁמַיִם נִצְבְּעוּ בִּצְבָעִים שֶׁל כָּחֹל-כֵּהֶה וּסְגֹל מִצַּד אֶחָד, וְכָתֹם-אָדֹם מִן הַצַּד הַשֵּׁנִי.”.
ג. אילו שמואל, אביי ורב אסי לא היה פועל כפי שמתואר ב־“לְפֶתַע צָץ עַל פָּנָיו הַקְּדוֹשׁוֹת חִיּוּךְ גָּדוֹל, חִיּוּךְ עָצוּם!” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “הָרְכוּשׁ שֶׁיֵּשׁ לָנוּ הוּא כֶּסֶף שֶׁל הַהוֹרִים, כֶּסֶף מֵהַקָּדוֹשׁ בָּרוּךְ הוּא, וְאִם נְזַלְזֵל – לֹא נִמְצָא אֶת…” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. להלן עיבוד חינוכי, רגשי ומרתק מתוך התלמוד הבבלי (מסכת חולין, דף ק"ה…
2. רֵיחַ עָמֹק וְנָעִים שֶׁל אֲדָמָה רְטֻבָּה עָלָה מִן הָרְגָבִים.
3. אֵין אֲפִלּוּ אֲגוֹרָה חֲדָשָׁה שֶׁנִּמְצֵאת מִתַּחַת לַעֲלֵה גֶּפֶן.
4. יָדָיו שֶׁל הָרַב קָפְאוּ בַּמַּיִם הַקָּרִים, אֲבָל הוּא לֹא נִכְנַע.
5. אֲנִי חִפַּשְׂתִּי כָּל הַיָּמִים הָאֵלֶּה לִמְצֹא כֶּסֶף זָרוּק…
6. אִם הוּא הִשְׁגִּיחַ הֵיטֵב – הַ'מַּטְבֵּעַ' שֶׁלּוֹ הוּא שֶׁהַקַּלְמָר…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “עָזַב רַב אַסִּי הַכֹּל, יָצָא אֶל חֶלְקַת הַשָּׂדֶה שֶׁלּוֹ.”.
מכתב: כתבו לשמואל, אביי ורב אסי בעקבות “תַּלְמִיד חָכָם צָרִיךְ לִשְׁמֹר עַל מַלְבּוּשָׁיו שֶׁיִּהְיוּ נְקִיִּים לְלֹא כֶּתֶם, אַךְ עַכְשָׁו, פִּקּוּחַ מָמוֹן וְנֶזֶק…”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “הַמַּטְבֵּעַ הָאֲמִתִּי הוּא מָה שֶׁנִּצַּלְתִּי בִּזְכוּת הַהַשְׁגָּחָה וְהָאַחֲרָיוּת!”.
דיווח: כתבו ידיעה קצרה שכותרתה “השגחה על נכסים”.

המחזה

רגע מן הסיפורהדמויותמה יאמרו או יעשו?
אֵלֶּה הָיוּ עֵצִים שֶׁהָאָרִיס בִּכְלָל לֹא הָיָה אָמוּר לָקַחַת.
לְפֶתַע צָץ עַל פָּנָיו הַקְּדוֹשׁוֹת חִיּוּךְ גָּדוֹל, חִיּוּךְ עָצוּם!

7. חיבור לחיים מתוך המעשה

א. מה אפשר ללמוד מן הדרך שבה שמואל, אביי ורב אסי פעל באירוע “זֶרֶם מַיִם אֵימְתָנִי וְחוּם מִבּוֹץ חָזַר לִשְׁטֹף הַכֹּל!”?
ב. תארו מצב מחיי הכיתה שבו אפשר ליישם את המסר של “השגחה על נכסים”.
