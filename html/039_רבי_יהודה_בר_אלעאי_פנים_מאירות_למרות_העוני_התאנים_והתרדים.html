פנים מאירות למרות העוני — התאנים והתרדים

דף עבודה בעקבות רבי יהודה בר אלעאי
שם: כיתה:

המעשה — הסיפור המלא

פְּתִיחָה: בַּיִת קָטָן, קֹר גָּדוֹל

יְלָדִים, עִצְמוּ אֶת הָעֵינַיִם וְנַסּוּ לְדַמְיֵן לַיְלָה קַר וְחָשׁוּךְ בָּעִיר אוּשָׁא, שֶׁבִּצְפוֹן אֶרֶץ יִשְׂרָאֵל. שְׁשְּׁשְׁ... הַאִם אַתֶּם שׁוֹמְעִים אֶת קוֹל הָרוּחַ הַקָּרָה הַשּׁוֹרֶקֶת בַּחוּץ?

בִּקְצֵה הָרְחוֹב, עָמַד בַּיִת קָטָן בָּנוּי מֵאֲבָנִים פְּשׁוּטוֹת. זֶהוּ בֵּיתוֹ שֶׁל גְּדוֹל הַדּוֹר, הַתַּנָּא הַקָּדוֹשׁ רַבִּי יְהוּדָה בַּר אִלְעָאי .

אִם נָצִיץ פְּנִימָה אֶל תּוֹךְ הַבַּיִת, לֹא נִרְאֶה תַּנּוּר חִמּוּם מְשֻׁכְלָל אוֹ שְׂמִיכוֹת עָבוֹת. הָעֹנִי בְּבֵיתוֹ שֶׁל רַבִּי יְהוּדָה הָיָה כָּל כָּךְ עָצוּם, שֶׁלּוֹ וּלְאִשְׁתּוֹ הָיָה רַק מְעִיל צֶמֶר אֶחָד, גְּלִימָה מְחֻסְפֶּסֶת אַחַת שֶׁשִּׁמְּשָׁה אֶת שְׁנֵיהֶם! כַּאֲשֶׁר אֵשֶׁת רַבִּי יְהוּדָה הָיְתָה צְרִיכָה לָצֵאת לַשּׁוּק – הִיא הָיְתָה מִתְעַטֶּפֶת בּוֹ. וְכַאֲשֶׁר הַצַּדִּיק הָיָה צוֹעֵד לְבֵית הַמִּדְרָשׁ לִלְמֹד תּוֹרָה – הוּא הָיָה לוֹבֵשׁ אוֹתוֹ עַל עַצְמוֹ.

וּמָה הֵם אָכְלוּ? הָאָרוֹן הָיָה רֵיק. אֵין לֶחֶם, אֵין בָּשָׂר, אֵין גְּבִינוֹת וְאֵין מַמְתַּקִּים.

סְעוּדַת הַמְּלָכִים שֶׁל הַצַּדִּיק

יוֹם אֶחָד, חָזַר רַבִּי יְהוּדָה מִבֵּית הַמִּדְרָשׁ. גּוּפוֹ הָיָה עָיֵף, וְהַבֶּטֶן שֶׁלּוֹ מַמָּשׁ כָּאֲבָה מֵרָעָב. הִגִּיעָה שְׁעַת הַסְּעוּדָה.

אֵשֶׁת רַבִּי יְהוּדָה הַצַּדֶּקֶת הִגִּישָׁה לַשֻּׁלְחָן צַלַּחַת עֵץ פְּשׁוּטָה. מָה הָיָה בְּתוֹכָהּ? יְלָדִים, שָׁם לֹא הָיָה הַמְבּוּרְגֶּר חַם, גַּם לֹא אוֹרֶז. בַּצַּלַּחַת הָיוּ עֲלֵי 'תְּרָדִים' (עָלִים יְרֻקִּים שֶׁדּוֹמִים מְעַט לִתְרֶד שֶׁל יָמֵינוּ), שֶׁהִיא קָטְפָה מִן הַשָּׂדֶה. הִיא בִּשְּׁלָה אוֹתָם בְּמַיִם. זֶהוּ! אֲפִלּוּ מֶלַח, אוֹ טִפַּת שֶׁמֶן, לֹא הָיוּ לָהֶם כְּדֵי לָתֵת טַעַם לַיָּרָק הֶעָנִי הַזֶּה. הָרֵיחַ שֶׁעָלָה מִן הַצַּלַּחַת הָיָה רֵיחַ שֶׁל עֲשָׂבִים שְׁלוּקִים וְתָפֵלִים בְּמַיִם חַמִּים.

אָדָם רָגִיל הָיָה דּוֹחֵף אֶת הַצַּלַּחַת בְּכַעַס: "זֶה מָה שֶׁנֹּאכַל שׁוּב?! הַעֲלִים הַמָּרִים הָאֵלֶּה?!".

אֲבָל רַבִּי יְהוּדָה... אוֹי, הֱיִיתֶם צְרִיכִים לִרְאוֹת אֶת רַבִּי יְהוּדָה.

הוּא נָטַל אֶת יָדָיו בְּשִׂמְחָה עֲצוּמָה. הוּא יָשַׁב לַשֻּׁלְחָן, אָכַל אֶת הַתְּרָדִים כְּאִלּוּ הָיוּ הַמַּעֲדָן הַטָּעִים וְהַיָּקָר בְּיוֹתֵר בָּעוֹלָם. מִדֵּי פַּעַם הָיָה לוֹקֵחַ שְׁלוּק שֶׁל מַיִם צוֹנְנִים לִשְׁטֹף אֶת הַטַּעַם, וְאָז, תּוֹךְ כְּדֵי לְעִיסָה, נִשְׁמַע קוֹלוֹ הַמָּתוֹק וְהַעָרֵב – הוּא חָזַר בְּלֶחֶשׁ עַל הַמִּשְׁנָיוֹת וְהַהֲלָכוֹת שֶׁלָּמַד.

וְאָז, קָרָה דָּבָר נִפְלָא. קָסֶם רוּחָנִי שֶׁרַק הַקָּדוֹשׁ בָּרוּךְ הוּא יָכוֹל לַעֲשׂוֹת!

הָיִינוּ מְצַפִּים שֶׁאָדָם שֶׁאוֹכֵל רַק עָלִים בְּמַיִם, פָּנָיו יִהְיוּ לְבָנוֹת, רָזוֹת, חִוְּרוֹת וַעֲצוּבוֹת. אֲבָל פָּנָיו שֶׁל רַבִּי יְהוּדָה – הֵחֵלּוּ לִזְרֹחַ! אוֹר זָהוּב וְרַךְ מִלֵּא אֶת לְחָיָיו. מִישֶׁהוּ שֶׁהָיָה מַבִּיט בּוֹ, הָיָה בָּטוּחַ שֶׁזֶּה עַתָּה סִיֵּם סְעוּדַת בָּשָׂר וְיַיִן אֵצֶל הַמֶּלֶךְ. פָּנָיו צָהֲלוּ וְהֵאִירוּ כְּאוֹר הַחַמָּה!

הַהִתְקָלוּת בַּשּׁוּק

לְמָחֳרָת בַּבֹּקֶר, לָבַשׁ רַבִּי יְהוּדָה אֶת מְעִילוֹ הַיָּשָׁן וְצָעַד לְכִוּוּן בֵּית הַמִּדְרָשׁ. הַשֶּׁמֶשׁ שֶׁל הַבֹּקֶר לִטְּפָה אֶת זְקָנוֹ הַלָּבָן, וְהוּא הָלַךְ בִּקְפִיצוֹת קְטַנּוֹת שֶׁל שִׂמְחָה.

בְּאוֹתָהּ שָׁעָה, עָבַר בָּרְחוֹב אָדָם שֶׁלֹּא אָהַב אֶת תַּלְמִידֵי הַחֲכָמִים. אִישׁ חָצוּף זֶה הָיָה קָשׁוּר לַשִּׁלְטוֹן בָּרוֹמָאִי וְלִכְתוֹת שֶׁעָזְבוּ אֶת הַתּוֹרָה.

הָאִישׁ הַזֶּה עָצַר אֶת עֶגְלָתוֹ וְהִבִּיט בְּרַבִּי יְהוּדָה בַּר אִלְעָאי בִּתְמִיהָה גְּדוֹלָה.

הוּא סָקַר בְּמַבָּטוֹ אֶת הַמְּעִיל הֶעָנִי, אֶת הַנַּעֲלַיִם הַבְּלוּיוֹת. אֲבָל אָז, הוּא הִבִּיט בַּפָּנִים שֶׁל הָרַב. הַפָּנִים! הֵן הָיוּ מְלֵאוֹת, וְרֻדּוֹת-צְהֻבּוֹת מִבְּרִיאוּת, מְאִירוֹת כְּלַפִּיד שֶׁל אֵשׁ.

"הָהּ!" קָרָא הָאִישׁ בְּקוֹל עָבֶה וּמְזַלְזֵל מוּל כָּל עָבְרֵי הָאֹרַח בַּשּׁוּק. "אֲנִי יוֹדֵעַ מָה זֶה הָרַב הַזֶּה! אַתֶּם חוֹשְׁבִים שֶׁהוּא עָנִי וְצַדִּיק? שֶׁקֶר וְכָזָב!".

הָאִישׁ חָסַם אֶת הַדֶּרֶךְ לִפְנֵי רַבִּי יְהוּדָה וְהִכְרִיז בְּקוֹל לַעֲגָנִי: "פָּנִים מְאִירוֹת כָּאֵלֶּה בָּאוֹת רַק מִשְּׁלוֹשָׁה דְּבָרִים! אוֹ שֶׁאַתָּה נוֹשֵׁךְ בְּנֶשֶׁךְ (מַלְוֶה הַכֶּסֶף בְּרִבִּית אֲסוּרָה לַעֲנִיִּים, וְלָכֵן אַתָּה עָשִׁיר בַּסֵּתֶר), אוֹ שֶׁאַתָּה מְגַדֵּל חֲזִירִים בַּמַּחְתֶּרֶת (שֶׁמְּבִיאִים מָמוֹן רַב), אוֹ... שֶׁאַתָּה פָּשׁוּט שׁוֹתֶה הֲמוֹן יַיִן בַּלֵּילוֹת, וְלָכֵן פָּנֶיךָ מְאִירוֹת מִשִּׁכְרוּת!".

הַשּׁוּק כֻּלּוֹ הִשְׁתַּתֵּק. אֲנָשִׁים נֶעֶצְרוּ וְהִבִּיטוּ. מָה יַעֲנֶה גְּדוֹל יִשְׂרָאֵל לְהַעֲלָבָה הַנּוֹרָאָה הַזּוֹ? הַאִם יִכְעַס? הַאִם יִצְעַק?

אֵשׁ הַתּוֹרָה מְנַצַּחַת

רַבִּי יְהוּדָה בַּר אִלְעָאי לֹא נִבְהַל, וְלֹא כָּעַס. הוּא הִזְדַּקֵּף, וְהָאוֹר שֶׁעַל פָּנָיו לֹא נִכְבָּה אֲפִלּוּ לְרֶגַע אֶחָד. גַּאֲוָה שֶׁל יְהוּדִי בָּעֲרָה בְּעֵינָיו הַטְּהוֹרוֹת.

הוּא עָנָה לָאִישׁ בְּקוֹל רָגוּעַ אֲבָל חָזָק וְחַד שֶׁנִּשְׁמַע בְּכָל הַשּׁוּק:

"תִּפַּח רוּחֲךָ, מִינָא! (תִּבָּטֵל מַחֲשַׁבְתְּךָ הָרָעָה!). שְׁלֹשֶׁת הַדְּבָרִים שֶׁאָמַרְתָּ – אֲנִי רָחוֹק מֵהֶם מִזְרָח מִמַּעֲרָב! לְגַדֵּל חֲזִירִים אוֹ לְהַלְווֹת בְּרִבִּית – זֶה אָסוּר מִן הַתּוֹרָה הַקְּדוֹשָׁה וְאָסוּר לִיהוּדִי כְּלָל לְהִתְקָרֵב לְכָךְ. וְלִשְׁתּוֹת יַיִן? אֲנִי לֹא שׁוֹתֶה יַיִן לְעוֹלָם, מִלְּבַד אַרְבַּע הַכּוֹסוֹת בְּלֵיל הַסֵּדֶר שֶׁל פֶּסַח, שֶׁאֲנִי מֻכְרָח לִשְׁתּוֹת כְּדֵי לְקַיֵּם מִצְוָה, וַאֲפִלּוּ מֵהֶן – הָרֹאשׁ שֶׁלִּי כּוֹאֵב מִפֶּסַח וְעַד חַג הַשָּׁבוּעוֹת!".

הָאִישׁ צִמְצֵם אֶת עֵינָיו בְּבִלְבּוּל וְשָׁאַל: "אִם גַּם אֵין לְךָ מָמוֹן, וְאַתָּה גַּם לֹא שׁוֹתֶה שׁוּם דְּבַר מְתִיקָה אוֹ אֲכִילָה טוֹבָה – מֵאֵיפֹה יֵשׁ לְךָ פָּנִים שְׁמֵנוֹת, מְאִירוֹת וּשְׂמֵחוֹת כָּל כָּךְ?! מָה מֵאִיר לְךָ עַל הַפָּנִים?!".

רַבִּי יְהוּדָה צָעַד צַעַד אֶחָד קָדִימָה. חִיּוּךְ גָּדוֹל, קָדוֹשׁ וְאֲמִתִּי הִתְפַּשֵּׁט עַל פָּנָיו, וְהוּא לָחַשׁ מִשְׁפָּט אֶחָד, שֶׁכָּל הַשּׁוֹמְעִים יִזְכְּרוּ תָּמִיד. הוּא הִצְבִּיעַ עַל סֵפֶר הַתּוֹרָה, וְאָמַר אֶת פְּסוּקוֹ שֶׁל שְׁלֹמֹה הַמֶּלֶךְ:

"חָכְמַת אָדָם – תָּאִיר פָּנָיו" .

יְלָדִים, רַבִּי יְהוּדָה אָמַר לוֹ: שֶׁבֶּאֱמֶת אֵין לִי כֶּסֶף, וַאֲנִי אוֹכֵל רַק עָלִים שְׁלוּקִים וְתאֵנִים פְּשׁוּטוֹת. אֲבָל כְּשֶׁיְּהוּדִי מָלֵא בַּתּוֹרָה – הַתּוֹרָה מַכְנִיסָה בּוֹ שִׂמְחָה שֶׁלֹּא נִתָּן לִקְנוֹת בְּשׁוּם סוּפֶּרְמַרְקֶט וּבְשׁוּם חֲנוּת צַעֲצוּעִים! הַנְּשָׁמָה שֶׁלִּי אוֹכֶלֶת מִדִּבְרֵי חֲזַ"ל, הַנֶּפֶשׁ שֶׁלִּי רְגוּעָה בְּבִטָּחוֹן בַּה', וְהַשִּׂמְחָה הַזּוֹ הִיא אֵשׁ שֶׁבּוֹקַעַת הַחוּצָה וּמְאִירָה אֶת הַפָּנִים!

הָאִישׁ חָזַר לַעֲגַלְתּוֹ בְּבֹשֶׁת פָּנִים וּבְשֶׁקֶט, בְּעוֹד שֶׁרַבִּי יְהוּדָה בַּר אִלְעָאי, הֶעָנִי הַמְרֻדָּל אֲבָל הָאִישׁ הַמְּאֻשָּׁר בָּעוֹלָם, הִמְשִׁיךְ לְזַמְזֵם שִׁיר מָתוֹק לִכְבוֹד בּוֹרֵא עוֹלָם, בְּדַרְכּוֹ לְאוֹרָהּ שֶׁל תּוֹרָה בְּבֵית הַמִּדְרָשׁ.

ג. כיצד הגיב רבי יהודה בר אלעאי למצב המתואר: “מִדֵּי פַּעַם הָיָה לוֹקֵחַ שְׁלוּק שֶׁל מַיִם צוֹנְנִים לִשְׁטֹף אֶת הַטַּעַם, וְאָז, תּוֹךְ כְּדֵי לְעִיסָה, נִשְׁמַע קוֹלוֹ הַמָּתוֹק וְהַעָרֵב – הוּא…”?

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “פנים מאירות למרות העוני — התאנים והתרדים” על רבי יהודה בר אלעאי? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “מִישֶׁהוּ שֶׁהָיָה מַבִּיט בּוֹ, הָיָה בָּטוּחַ שֶׁזֶּה עַתָּה סִיֵּם סְעוּדַת בָּשָׂר וְיַיִן אֵצֶל הַמֶּלֶךְ.”.
ג. אילו רבי יהודה בר אלעאי לא היה פועל כפי שמתואר ב־“שֶׁאַתָּה פָּשׁוּט שׁוֹתֶה הֲמוֹן יַיִן בַּלֵּילוֹת, וְלָכֵן פָּנֶיךָ מְאִירוֹת מִשִּׁכְרוּת!".” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “חִיּוּךְ גָּדוֹל, קָדוֹשׁ וְאֲמִתִּי הִתְפַּשֵּׁט עַל פָּנָיו, וְהוּא לָחַשׁ מִשְׁפָּט אֶחָד, שֶׁכָּל הַשּׁוֹמְעִים יִזְכְּרוּ…” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. יְלָדִים, עִצְמוּ אֶת הָעֵינַיִם וְנַסּוּ לְדַמְיֵן לַיְלָה קַר…
2. גּוּפוֹ הָיָה עָיֵף, וְהַבֶּטֶן שֶׁלּוֹ מַמָּשׁ כָּאֲבָה מֵרָעָב.
3. קָסֶם רוּחָנִי שֶׁרַק הַקָּדוֹשׁ בָּרוּךְ הוּא יָכוֹל לַעֲשׂוֹת!
4. הוּא סָקַר בְּמַבָּטוֹ אֶת הַמְּעִיל הֶעָנִי, אֶת הַנַּעֲלַיִם…
5. הוּא עָנָה לָאִישׁ בְּקוֹל רָגוּעַ אֲבָל חָזָק וְחַד שֶׁנִּשְׁמַע…
6. הַנְּשָׁמָה שֶׁלִּי אוֹכֶלֶת מִדִּבְרֵי חֲזַ"ל, הַנֶּפֶשׁ שֶׁלִּי…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “מִדֵּי פַּעַם הָיָה לוֹקֵחַ שְׁלוּק שֶׁל מַיִם צוֹנְנִים לִשְׁטֹף אֶת הַטַּעַם, וְאָז, תּוֹךְ כְּדֵי לְעִיסָה, נִשְׁמַע קוֹלוֹ…”.
מכתב: כתבו לרבי יהודה בר אלעאי בעקבות “הֵן הָיוּ מְלֵאוֹת, וְרֻדּוֹת-צְהֻבּוֹת מִבְּרִיאוּת, מְאִירוֹת כְּלַפִּיד שֶׁל אֵשׁ.”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “שְׁלֹשֶׁת הַדְּבָרִים שֶׁאָמַרְתָּ – אֲנִי רָחוֹק מֵהֶם מִזְרָח מִמַּעֲרָב!”.
דיווח: כתבו ידיעה קצרה שכותרתה “פנים מאירות למרות העוני — התאנים והתרדים”.

המחזה

רגע מן הסיפורהדמויותמה יאמרו או יעשו?

שֶׁאַתָּה פָּשׁוּט שׁוֹתֶה הֲמוֹן יַיִן בַּלֵּילוֹת, וְלָכֵן פָּנֶיךָ מְאִירוֹת מִשִּׁכְרוּת!".

7. חיבור לחיים מתוך המעשה

א. מה אפשר ללמוד מן הדרך שבה רבי יהודה בר אלעאי פעל באירוע “בְּאוֹתָהּ שָׁעָה, עָבַר בָּרְחוֹב אָדָם שֶׁלֹּא אָהַב אֶת תַּלְמִידֵי הַחֲכָמִים.”?
