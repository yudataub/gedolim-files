זכה לראות תלמידו ר' יוחנן דורש בחייו

דף עבודה בעקבות ר' חנינא בר חמא
שם: כיתה:

המעשה — הסיפור המלא

פְּתִיחָה: הָרַב הַגָּדוֹל שֶׁל אֶרֶץ יִשְׂרָאֵל

יְלָדִים, עִצְמוּ אֶת הָעֵינַיִם. דַּמְיְנוּ אֶת הָעִיר צִפּוֹרִי שֶׁבַּגָּלִיל, לִפְנֵי כִּמְעַט אַלְפַּיִם שָׁנָה. הַשֶּׁמֶשׁ שֶׁל אֶרֶץ יִשְׂרָאֵל מְחַמֶּמֶת אֶת הָאֲבָנִים הַלְּבָנוֹת. הָאֲוִיר מָלֵא בְּרֵיחַ מָתוֹק שֶׁל תְּאֵנִים דְּבַשׁ וְעֵצִים שֶׁל זֵיתִים. בְּתוֹךְ בֵּית הַמִּדְרָשׁ, יוֹשֵׁב אֶחָד מִגְּדוֹלֵי עַם יִשְׂרָאֵל – רַבִּי חֲנִינָא בַּר חָמָא.

רַבִּי חֲנִינָא הָיָה אִישׁ פֶּלֶא. הַקָּדוֹשׁ בָּרוּךְ הוּא נָתַן לוֹ כֹּחוֹת עֲצוּמִים. הוּא הָיָה חָכָם גָּדוֹל בַּתּוֹרָה, תַּלְמִידוֹ הָאָהוּב שֶׁל רַבִּי יְהוּדָה הַנָּשִׂיא. הוּא גַּם הָיָה אִישׁ חָזָק וּבָרִיא מְאֹד. אֲפִלּוּ כְּשֶׁהָיָה בֶּן שְׁמוֹנִים שָׁנָה, הוּא הָיָה יָכוֹל לַעֲמֹד עַל רֶגֶל אַחַת, לַחֲלֹץ אֶת הַנַּעַל מֵהָרֶגֶל הַשְּׁנִיָּה, וְלִנְעֹל אוֹתָהּ בַּחֲזָרָה בְּלִי לִפֹּל! וְהָכִי חָשׁוּב – הוּא אָהַב אֶת אֶרֶץ יִשְׂרָאֵל אַהֲבָה שֶׁאִי אֶפְשָׁר לְתָאֵר, עַד כְּדֵי כָּךְ שֶׁהָיָה יורֵד לַשְּׁבִילִים וּמְסַדֵּר אֶת הָאֲבָנִים בְּיָדָיו, כְּדֵי שֶׁאֲנָשִׁים לֹא יִכָּשְׁלוּ בְּאֶרֶץ הַקֹּדֶשׁ.

וִתּוּר שֶׁל עֲנָוָה

אֲבָל מֵעַל הַכֹּל, יְלָדִים, רַבִּי חֲנִינָא הָיָה עָנָיו בְּתַכְלִית. כְּשֶׁרַבִּי יְהוּדָה הַנָּשִׂיא נִפְטַר, הִצִּיעוּ לְרַבִּי חֲנִינָא לִהְיוֹת "רֹאשׁ הַיְשִׁיבָה", לִהְיוֹת הָרַב הֲכִי גָּדוֹל וּמְפֻרְסָם. זֶה כָּבוֹד עָצוּם, נָכוֹן? כָּל אָדָם הָיָה קוֹפֵץ עַל הַזְּכוּת הַזּוֹ!

אֲבָל רַבִּי חֲנִינָא יָדַע שֶׁיֵּשׁ חָכָם גָּדוֹל אַחֵר בַּדּוֹר, שֶׁגָּדוֹל מִמֶּנּוּ בִּשְׁנֵי שָׁנִים קְצָרוֹת מְאֹד – רַבִּי אֶפְרַיִם מַקְשָׁאָה. בְּלִי רֶגַע שֶׁל הִסּוּס, רַבִּי חֲנִינָא וִתֵּר עַל הַתַּפְקִיד וְנָתַן אֶת הַכָּבוֹד לַחֲבֵרוֹ. "לֹא יִתָּכֵן שֶׁאֲנִי אֶשְׁלֹט וְאֶעֱמֹד בָּרֹאשׁ, בְּשָׁעָה שֶׁיֵּשׁ מִישֶׁהוּ שֶׁמַּגִּיעַ לוֹ קֹדֶם," אָמַר בְּשֶׁקֶט, וְהִמְשִׁיךְ לִלְמֹד בַּסַּפְסָל הָרָגִיל בְּבֵית הַמִּדְרָשׁ. חֲכָמֵינוּ אוֹמְרִים שֶׁבִּזְכוּת הַוִּתּוּר הֶעָצוּם הַזֶּה – הַקָּדוֹשׁ בָּרוּךְ הוּא שִׁלֵּם לוֹ שָׂכָר כָּפוּל וְהֶאֱרִיךְ אֶת שְׁנוֹת חַיָּיו! הוּא חַי שָׁנִים רַבּוֹת וְטוֹבוֹת, וּפָנָיו זָרְחוּ בְּאוֹר הַתּוֹרָה.

הַתַּלְמִיד הַצָּעִיר שֶׁעוֹלֶה וְגוֹדֵל

הַשָּׁנִים חָלְפוּ. בֵּין מֵאוֹת הַתַּלְמִידִים שֶׁיָּשְׁבוּ מוּל רַבִּי חֲנִינָא, הָיָה בָּחוּר צָעִיר וּמְיֻחָד בְּמִינוֹ. קָרְאוּ לוֹ רַבִּי יוֹחָנָן.

רַבִּי יוֹחָנָן הָיָה כְּלִי קִבּוּל מֻשְׁלָם לְתוֹרָה. הוּא הָיָה לְלֹא כֶּסֶף כְּלָל. הָיוּ לוֹ שָׂדוֹת, וְהוּא מָכַר אוֹתָם אֶחָד אַחֲרֵי הַשֵּׁנִי רַק כְּדֵי שֶׁיּוּכַל לָשֶׁבֶת וְלִלְמֹד! "מָכַרְתִּי מַשֶּׁהוּ שֶׁנִּבְרָא בְּשִׁשָּׁה יָמִים (שָׂדֶה וָאֲדָמָה), כְּדֵי לִקְנוֹת מַשֶּׁהוּ שֶׁנִּנְתַּן לָנוּ בְּאַרְבָּעִים יוֹם וְאַרְבָּעִים לַיְלָה (אֶת הַתּוֹרָה)!" כָּךְ הָיָה אוֹמֵר בְּחִיּוּךְ מֵאִיר.

רַבִּי חֲנִינָא דָּאַג לְרַבִּי יוֹחָנָן כְּמוֹ שֶׁאָב דּוֹאֵג לִבְנוֹ. הוּא לִמֵּד אוֹתוֹ, הִסְבִּיר לוֹ פְסוּקִים שֶׁל אֲגָדָה שֶׁנָּתְנוּ לַתַּלְמִיד כֹּחַ אֱמוּנִי וְתִקְוָה, וּפָתַח בְּפָנָיו אֶת אֲרוֹנוֹת הַסְּפָרִים שֶׁל הַהֲלָכָה הֲכִי עֲמֻקָּה. הוּא הִשְׁקָה אוֹתוֹ מִמֵּי הַתּוֹרָה כְּפִי שֶׁמַּשְׁקִים שְׁתִיל עֵץ זַיִת צָעִיר, שֶׁיִּגְדַּל לִהְיוֹת חָזָק וְיַצִּיב.

וְרַבִּי יוֹחָנָן אָכֵן צָמַח. קוֹלוֹ הִתְחַזֵּק, חָכְמָתוֹ עָבְרָה מִפֶּה לְאֹזֶן. מֵאוֹת בַּחוּרִים, אֲבְרֵכִים וְזְקֵנִים הֵחֵלּוּ לְהִתְאַסֵּף סְבִיבוֹ. הֵם רָצוּ לִשְׁמֹעַ אֶת הַתּוֹרָה שֶׁיּוֹצֵאת מִפִּיו שֶׁל רַבִּי יוֹחָנָן!

מִבְחַן הַלֵּב

יְלָדִים, כָּאן עָמַד רַבִּי חֲנִינָא בְּמִבְחָן עָצוּם.

הַאִם לַמּוֹרֶה יֵשׁ רֶגַע שֶׁבּוֹ קָשֶׁה לוֹ? בְּוַדַּאי. הֲרֵי רַבִּי חֲנִינָא זָקֵן וְחָכָם, וְהִנֵּה הַתַּלְמִיד שֶׁלּוֹ, אוֹתוֹ נַעַר צָעִיר שֶׁהוּא בְּעַצְמוֹ לִמֵּד, תּוֹפֵס אֶת הַמָּקוֹם הַמֶּרְכָּזִי! פִתְאוֹם אֲנָשִׁים הוֹלְכִים לִשְׁמֹעַ אֶת רַבִּי יוֹחָנָן דּוֹרֵשׁ. הַאִם רַבִּי חֲנִינָא יִרְגִּישׁ תְּחוּשָׁה שֶׁל עֶצֶב? הַאִם יַרְגִּישׁ שֶׁשָּׁכְחוּ אוֹתוֹ?

יוֹם אֶחָד, הַשֶּׁמֶשׁ נָטְתָה לִשְׁקֹעַ וְצָבְעָה אֶת עַנְנֵי טְבֶרְיָה וְצִפּוֹרִי בְּוָרֹד וְזָהָב. רַבִּי חֲנִינָא הַזָּקֵן, עִם מַקְּלוֹ הָעָבֶה, הָלַךְ לְאַט לְכִוּוּן בֵּית הַמִּדְרָשׁ. מֵרָחוֹק, הוּא כְּבָר שָׁמַע אֶת הַקּוֹל.

לֹא סְתָם קוֹל, אֶלָּא קוֹל אַדִּיר, קוֹל שֶׁל רַעַם מָתוֹק וְעָרֵב. זֶה הָיָה קוֹלוֹ שֶׁל רַבִּי יוֹחָנָן הַדּוֹרֵשׁ לְפָנֵי קָהָל שֶׁל אֲלָפִים!

רַבִּי חֲנִינָא הִתְקָרֵב. הוּא הִגִּיעַ לְפֶתַח בֵּית הַמִּדְרָשׁ. הַמָּקוֹם הָיָה הָמוּם מֵאָדָם. חֲכָמִים יָשְׁבוּ צְפוּפִים אֶחָד לַשֵּׁנִי, מַזִּיעִים מֵחֹם אֲבָל מִסְתַּכְּלִים בִּפְלִיאָה עַל הַבָּמָה. וְשָׁם, בַּמֶּרְכָּז, עָמַד רַבִּי יוֹחָנָן, וְהֵאִיר אֶת דִּבְרֵי הַתּוֹרָה כְּמוֹ נֵר זוֹהֵר מֵרֹאשׁ חָשׁוּךְ.

"וְזֶהוּ פֵּרוּשׁ הַפָּסוּק!" נִשְׁמַע קוֹלוֹ הַבָּרוּר שֶׁל רַבִּי יוֹחָנָן. קוֹל אָהוּב. קוֹל שֶׁל תּוֹרַת אֱמֶת.

הַדְּמָעוֹת שֶׁל הַזָּקֵן

רַבִּי חֲנִינָא עָמַד בַּפֶּתַח, נִשְׁעָן עַל מַקְּלוֹ. הוּא הִבִּיט בַּתַּלְמִיד שֶׁלּוֹ. אִישׁ מֵהַיּוֹשְׁבִים עֲדַיִן לֹא שָׂם לֵב שֶׁזְּקַן הַדּוֹר עוֹמֵד בַּדֶּלֶת.

וְאָז, מַשֶּׁהוּ קָרָה לְרַבִּי חֲנִינָא. הַגָּרוֹן שֶׁלּוֹ נֶחְנַק. הַלֵּב שֶׁלּוֹ הִתְמַלֵּא בְּהַרְגָּשָׁה חַמָּה עֲצוּמָה. שְׁתֵּי דְּמָעוֹת זַכּוֹת וַחַמּוֹת יָרְדוּ מֵעֵינָיו שֶׁל הָרַב הַיָּשִׁישׁ, וְהִתְגַּלְגְּלוּ לְתוֹךְ זְקָנוֹ הַלָּבָן כַּשֶּׁלֶג.

הַאִם אֵלֶּה הָיוּ דְּמָעוֹת שֶׁל עֶצֶב? שֶׁל קִנְאָה? חַס וְשָׁלוֹם! יְלָדִים, אֵלֶּה הָיוּ דְּמָעוֹת שֶׁל תּוֹדָה עֲצוּמָה לְבּוֹרֵא עוֹלָם!

"אָנָּא ה'," לָחַשׁ רַבִּי חֲנִינָא בְּקוֹל רוֹטֵט שֶׁרַק מַלְאֲכֵי הַשָּׁמַיִם שָׁמְעוּ. "תּוֹדָה לְךָ, הַקָּדוֹשׁ בָּרוּךְ הוּא! תּוֹדָה רַבָּה לְךָ שֶׁלֹּא נָטַלְתָּ אֶת נִשְׁמָתִי לַשָּׁמַיִם בְּעוֹדִי צָעִיר. תּוֹדָה לְךָ שֶׁנָּתַתָּ לִי חַיִּים אֲרֻכִּים."

מַדּוּעַ הוֹדָה רַבִּי חֲנִינָא לְה'?

הוּא הֵרִים אֶת מַבָּטוֹ, חִיֵּךְ חִיּוּךְ גָּדוֹל, וְאָמַר לְעַצְמוֹ, וּמְאֻחָר יוֹתֵר גַּם גִּלָּה זֹאת לְכֻלָּם: "אָשְׁרִי וּמַה טּוֹב חֶלְקִי! בָּרוּךְ הַשֵּׁם שֶׁזָּכִיתִי לִרְאוֹת בְּעֵינַי שֶׁהַתַּלְמִיד שֶׁלִּי גּוֹדֵל, וְדוֹרֵשׁ בְּחַיַּי!"

זֶה הָיָה נִצָּחוֹן גָּדוֹל מִכָּל נִצָּחוֹן שֶׁל צָבָא אוֹ מֶלֶךְ. זֶה הָיָה הַנִּצָּחוֹן שֶׁל הַתּוֹרָה וְהַלֵּב הַטּוֹב. רַבִּי חֲנִינָא לֹא חִפֵּשׂ לִהְיוֹת צוֹדֵק אוֹ מְפֻרְסָם. הוּא חִפֵּשׂ שֶׁהַתּוֹרָה שֶׁל ה' תִּגְדַּל. וּכְשֶׁרָאָה אֶת רַבִּי יוֹחָנָן תַּלְמִידוֹ מַרְבֶּה תּוֹרָה בְּעַם יִשְׂרָאֵל, הַשִּׂמְחָה שֶׁלּוֹ הָיְתָה גְּדוֹלָה מֵהַשִּׂמְחָה שֶׁל אָדָם הַמּוֹצֵא מִיליוֹנֵי מַטְבְּעוֹת שֶׁל זָהָב.

יְלָדִים, עַד סוֹף יָמָיו רַבִּי חֲנִינָא בַּר חָמָא שָׂמַח בְּהַצְלָחָתוֹ שֶׁל רַבִּי יוֹחָנָן. כָּךְ נוֹהֵג תַּלְמִיד חָכָם אֲמִתִּי, וְכָךְ בּוֹנִים אֶת עַם יִשְׂרָאֵל.

מקורות: המקורות מופיעים בקובץ המקורי.

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “דַּמְיְנוּ אֶת הָעִיר צִפּוֹרִי שֶׁבַּגָּלִיל, לִפְנֵי כִּמְעַט אַלְפַּיִם שָׁנָה.” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “אֲבָל מֵעַל הַכֹּל, יְלָדִים, רַבִּי חֲנִינָא הָיָה עָנָיו בְּתַכְלִית.”?
ג. כיצד הגיב ר' חנינא בר חמא למצב המתואר: “הָיוּ לוֹ שָׂדוֹת, וְהוּא מָכַר אוֹתָם אֶחָד אַחֲרֵי הַשֵּׁנִי רַק כְּדֵי שֶׁיּוּכַל לָשֶׁבֶת וְלִלְמֹד!”?
ד. מה השתנה בין “פִתְאוֹם אֲנָשִׁים הוֹלְכִים לִשְׁמֹעַ אֶת רַבִּי יוֹחָנָן דּוֹרֵשׁ.” לבין “אִישׁ מֵהַיּוֹשְׁבִים עֲדַיִן לֹא שָׂם לֵב שֶׁזְּקַן הַדּוֹר עוֹמֵד בַּדֶּלֶת.”?
ה. כיצד הגיע הסיפור אל האירוע: “זֶה הָיָה נִצָּחוֹן גָּדוֹל מִכָּל נִצָּחוֹן שֶׁל צָבָא אוֹ מֶלֶךְ.”?

3. מה היה ראשון?

כתבו ליד כל אירוע מספר מ־1 עד 5 לפי סדר התרחשותו בסיפור.

□ הֵם רָצוּ לִשְׁמֹעַ אֶת הַתּוֹרָה שֶׁיּוֹצֵאת מִפִּיו שֶׁל רַבִּי יוֹחָנָן!
□ דַּמְיְנוּ אֶת הָעִיר צִפּוֹרִי שֶׁבַּגָּלִיל, לִפְנֵי כִּמְעַט אַלְפַּיִם שָׁנָה.
□ כָּךְ נוֹהֵג תַּלְמִיד חָכָם אֲמִתִּי, וְכָךְ בּוֹנִים אֶת עַם יִשְׂרָאֵל.
□ בְּלִי רֶגַע שֶׁל הִסּוּס, רַבִּי חֲנִינָא וִתֵּר עַל הַתַּפְקִיד וְנָתַן אֶת הַכָּבוֹד לַחֲבֵרוֹ.
□ שְׁתֵּי דְּמָעוֹת זַכּוֹת וַחַמּוֹת יָרְדוּ מֵעֵינָיו שֶׁל הָרַב הַיָּשִׁישׁ, וְהִתְגַּלְגְּלוּ לְתוֹךְ זְקָנוֹ הַלָּבָן כַּשֶּׁלֶג.

לפני ואחרי

מה קרה לפני: “הֵם רָצוּ לִשְׁמֹעַ אֶת הַתּוֹרָה שֶׁיּוֹצֵאת מִפִּיו שֶׁל רַבִּי יוֹחָנָן!”?
מה קרה אחרי: “שְׁתֵּי דְּמָעוֹת זַכּוֹת וַחַמּוֹת יָרְדוּ מֵעֵינָיו שֶׁל הָרַב הַיָּשִׁישׁ, וְהִתְגַּלְגְּלוּ לְתוֹךְ זְקָנוֹ הַלָּבָן כַּשֶּׁלֶג.”?

סיבה ותוצאה מתוך המעשה

האירוע הנתוןמה גרם לו?מה הייתה תוצאתו?
לֹא יִתָּכֵן שֶׁאֲנִי אֶשְׁלֹט וְאֶעֱמֹד בָּרֹאשׁ, בְּשָׁעָה שֶׁיֵּשׁ מִישֶׁהוּ שֶׁמַּגִּיעַ לוֹ קֹדֶם," אָמַר בְּשֶׁקֶט, וְהִמְשִׁיךְ לִלְמֹד בַּסַּפְסָל…
זֶה הָיָה קוֹלוֹ שֶׁל רַבִּי יוֹחָנָן הַדּוֹרֵשׁ לְפָנֵי קָהָל שֶׁל אֲלָפִים!

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “זכה לראות תלמידו ר' יוחנן דורש בחייו” על ר' חנינא בר חמא? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “הוּא הִשְׁקָה אוֹתוֹ מִמֵּי הַתּוֹרָה כְּפִי שֶׁמַּשְׁקִים שְׁתִיל עֵץ זַיִת צָעִיר, שֶׁיִּגְדַּל לִהְיוֹת חָזָק וְיַצִּיב.”.
ג. אילו ר' חנינא בר חמא לא היה פועל כפי שמתואר ב־“אִישׁ מֵהַיּוֹשְׁבִים עֲדַיִן לֹא שָׂם לֵב שֶׁזְּקַן הַדּוֹר עוֹמֵד בַּדֶּלֶת.” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “זֶה הָיָה נִצָּחוֹן גָּדוֹל מִכָּל נִצָּחוֹן שֶׁל צָבָא אוֹ מֶלֶךְ.” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. דַּמְיְנוּ אֶת הָעִיר צִפּוֹרִי שֶׁבַּגָּלִיל, לִפְנֵי כִּמְעַט…
2. כְּשֶׁרַבִּי יְהוּדָה הַנָּשִׂיא נִפְטַר, הִצִּיעוּ לְרַבִּי חֲנִינָא…
3. מָכַרְתִּי מַשֶּׁהוּ שֶׁנִּבְרָא בְּשִׁשָּׁה יָמִים (שָׂדֶה וָאֲדָמָה),…
4. לֹא סְתָם קוֹל, אֶלָּא קוֹל אַדִּיר, קוֹל שֶׁל רַעַם מָתוֹק וְעָרֵב.
5. אָנָּא ה'," לָחַשׁ רַבִּי חֲנִינָא בְּקוֹל רוֹטֵט שֶׁרַק מַלְאֲכֵי…
6. כָּךְ נוֹהֵג תַּלְמִיד חָכָם אֲמִתִּי, וְכָךְ בּוֹנִים אֶת עַם…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “הָיוּ לוֹ שָׂדוֹת, וְהוּא מָכַר אוֹתָם אֶחָד אַחֲרֵי הַשֵּׁנִי רַק כְּדֵי שֶׁיּוּכַל לָשֶׁבֶת וְלִלְמֹד!”.
מכתב: כתבו לר' חנינא בר חמא בעקבות “זֶה הָיָה קוֹלוֹ שֶׁל רַבִּי יוֹחָנָן הַדּוֹרֵשׁ לְפָנֵי קָהָל שֶׁל אֲלָפִים!”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “תּוֹדָה רַבָּה לְךָ שֶׁלֹּא נָטַלְתָּ אֶת נִשְׁמָתִי לַשָּׁמַיִם בְּעוֹדִי צָעִיר.”.
דיווח: כתבו ידיעה קצרה שכותרתה “זכה לראות תלמידו ר' יוחנן דורש בחייו”.
