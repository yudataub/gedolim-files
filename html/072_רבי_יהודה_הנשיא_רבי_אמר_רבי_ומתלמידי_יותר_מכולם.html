"אמר רבי: ...ומתלמידי יותר מכולם"

דף עבודה בעקבות רבי יהודה הנשיא (רבי)
שם: כיתה:

המעשה — הסיפור המלא

להלן מערך חינוכי, רגשי ומרתק על רבי יהודה הנשיא (רבינו הקדוש) , המעובד במיוחד לבנים בכיתות ב'-ד' בחינוך הדתי-לאומי. הסיפור בנוי על פי כללי הסטוריטלינג הנדרשים, תוך דגש על שפה מכבדת, קדושת התורה, תיאורים חושיים ומבנה רגשי.

הסיפור מנוקד במלואו כנדרש.

פְּתִיחָה: אַרְמוֹנוֹ שֶׁל הַנָּשִׂיא

יְלָדִים, בּוֹאוּ נֵצֵא לְמַסָּע קָסוּם בַּזְּמַן, לִפְנֵי כְּאַלְפַּיִם שָׁנָה, לָעִיר שְׁפַעַת הַהוֹד 'בֵּית שְׁעָרִים' (וְאַחַר כָּךְ 'צִפּוֹרִי') שֶׁבַּגָּלִיל.

דַּמְיְנוּ שֶׁאַתֶּם עוֹמְדִים מוּל שַׁעֲרֵי בַּרְזֶל גְּדוֹלִים. מֵעֵבֶר לַשְּׁעָרִים, נִמְצָא אַרְמוֹן אֲמִתִּי. מְשָׁרְתִים מִתְרוֹצְצִים, עֲגָלוֹת עֲמוּסוֹת בִּיבוּל מִכָּל רַחֲבֵי הָאָרֶץ נִכְנָסוֹת פְּנִימָה, וּרְחוֹבוֹת הָעִיר מְלֵאִים בַּאֲנָשִׁים חֲשׁוּבִים. זֶהוּ בֵּיתוֹ שֶׁל רַבִּי יְהוּדָה הַנָּשִׂיא!

"רַבִּי", כָּךְ כֻּלָּם קָרְאוּ לוֹ. הוּא הָיָה מַנְהִיגָם שֶׁל כָּל יְהוּדֵי הָעוֹלָם, עָשִׁיר עָצוּם שֶׁאֲפִלּוּ קֵיסַר רוֹמָא, אַנְטוֹנִינוּס, הָיָה חֲבֵרוֹ הַטּוֹב וְהָיָה שׁוֹלֵחַ לוֹ שַׂקִּים מְלֵאִים בְּמַטְבְּעוֹת זָהָב.

אֲבָל... הַאִם רַבִּי חָשַׁב עַל כֶּסֶף? מַמָּשׁ לֹא!

אִם הֱיִיתֶם נִכְנָסִים לְתוֹךְ אֶמְצַע הָאַרְמוֹן, לֹא הֱיִיתֶם שׁוֹמְעִים קוֹלוֹת שֶׁל מְסִבּוֹת. הֱיִיתֶם שׁוֹמְעִים קוֹל אַדִּיר, קוֹל שֶׁנִּשְׁמַע כְּמוֹ גַּלִּים שֶׁל יָם שֶׁמִּתְנַפְּצִים עַל הַסְּלָעִים: אֵלּוּ הָיוּ קוֹלוֹתֵיהֶם שֶׁל מֵאוֹת תַּלְמִידֵי יְשִׁיבָה! רֵיחַ נָעִים שֶׁל קְלָפִים יְשָׁנִים וּדְיוֹ שֶׁל סוֹפְרִים מִלֵּא אֶת הָאֲוִיר. בְּמֶרְכַּז הָאוּלָם, עַל כִּסֵּא מְכֻבָּד מֵעֵץ מְגֻלָּף, יָשַׁב רַבִּי יְהוּדָה הַנָּשִׂיא בִּכְבוֹדוֹ וּבְעַצְמוֹ, וְלִמֵּד תּוֹרָה. רַבִּי יָדַע שֶׁהַגָּלוּת מִתְקָרֶבֶת, וְלָכֵן אָסַף אֶת כָּל הַהֲלָכוֹת שֶׁבָּעוֹלָם כְּדֵי לִכְתֹּב אֶת סֵפֶר "הַמִּשְׁנָה". הוּא הָיָה חָכָם בְּיוֹתֵר, קָדוֹשׁ בְּיוֹתֵר, וְכָל גְּדוֹלֵי הַדּוֹר יָשְׁבוּ לְפָנָיו בְּיִרְאַת כָּבוֹד וְשָׁתוּ אֶת דְּבָרָיו בְּצָמָא.

הַקֻּשְׁיָה בְּבֵית הַמִּדְרָשׁ וְהַשְּׁתִיקָה

יוֹם אֶחָד, הַשֶּׁמֶשׁ הַזְּהוּבָה שֶׁל הַצָּהֳרַיִם אִירָה מִבַּעַד לְחַלּוֹנוֹת בֵּית הַמִּדְרָשׁ, אַךְ בִּפְנִים – הָיָה מֶתַח עָצוּם.

רַבִּי יְהוּדָה הַנָּשִׂיא לִמֵּד סוּגְיָה (הֲלָכָה) עֲמֻקָּה וּמְסֻבֶּכֶת מְאֹד. זֶה הָיָה כָּל כָּךְ קָשֶׁה, עַד שֶׁלְּפֶתַע, קָמוּ כַּמָּה מֵהָרַבָּנִים הַמְבֻגָּרִים וְהֶעֱלוּ "קֻשְׁיָה" – שְׁאֵלָה קָשָׁה שֶׁנִּרְאָה שֶׁאֵין לָהּ שׁוּם פִּתְרוֹן!

"רַבֵּינוּ," בִּקְּשׁוּ הַחֲכָמִים הַזְּקֵנִים, "אֵיךְ נָבִין אֶת הַהֲלָכָה הַזּוֹ? הֲרֵי יֵשׁ פֹּה סְתִירָה מֻחְלֶטֶת!"

שִׂפְתוֹתָיו שֶׁל רַבִּי נָעוּ בַּחֲשַׁאי. מְנַסֶּה לַחְשֹׁב, לְחַבֵּר, לִמְצֹא אֶת מְקוֹר הַדְּבָרִים. שֶׁקֶט הִשְׁתָּרֵר בָּאוּלָם. שֶׁקֶט כָּל כָּךְ כָּבֵד, שֶׁאֶפְשָׁר הָיָה לִשְׁמֹעַ זְבוּב מְזַמְזֵם בַּאֲוִיר. כָּל הָרַבָּנִים שָׁתְקוּ וְהִמְתִּינוּ לַתְּשׁוּבָה. אֲפִלּוּ מֵזִיגֵי הַמַּיִם מִחוּץ לַדֶּלֶת עָצְרוּ אֶת נְשִׁימָתָם. הַקֻּשְׁיָה נִשְׁאֲרָה בָּאֲוִיר, קָשָׁה כְּסֶלַע.

הַתַּלְמִיד שֶׁבַּפִּנָּה

בַּשּׁוּרָה הָאַחֲרוֹנָה, הַרְחֵק-הַרְחֵק מֵהַכִּסֵּא שֶׁל הַנָּשִׂיא, יָשַׁב תַּלְמִיד אֶחָד, צָעִיר וּבַיְשָׁן. קָרְאוּ לוֹ לֵוִי. הוּא הָיָה מֵהַתַּלְמִידִים הַצְּעִירִים בְּיוֹתֵר בַּיְשִׁיבָה. עֵינָיו נָצוֹצוּ מִתּוֹךְ סְפָרָיו, וּמוֹחוֹ הַמָּהִיר עָבַד לְלֹא הֶרֶף. וְהִנֵּה, כְּשֶׁהוּא חָשַׁב עַל הַשְּׁאֵלָה הַקָּשָׁה – לְפֶתַע, הִבְרִיק אוֹר גָּדוֹל בְּשִׂכְלוֹ! הוּא מָצָא אֶת הַתְּשׁוּבָה! הָיָה לוֹ כִּוּוּן מֻשְׁלָם שֶׁהִסְבִּיר הַכֹּל!

אֲבָל... לֵוִי רָעַד. הוּא הִבִּיט יָמִינָה, וְרָאָה אֶת רַבִּי שִׁמְעוֹן הַגָּדוֹל יוֹשֵׁב בִּדְמָמָה. הִבִּיט שְׂמֹאלָה, וְרָאָה רַבָּנִים שֶׁזְּקָנָם הִגִּיעַ לְבִרְכֵּיהֶם שׁוֹתְקִים. "אֵיךְ אֲנִי, הַתַּלְמִיד הַקָּטָן, אָעֵז לְדַבֵּר כָּאן?" חָשַׁב לֵוִי בְּפַחַד. "אִם גְּדוֹלֵי הַדּוֹר לֹא מוֹצְאִים תְּשׁוּבָה, מִי אֲנִי שֶׁאָקוּם וַאֲלַמֵּד אוֹתָם? בֶּטַח יִצְחֲקוּ עָלַי. יַגִּידוּ שֶׁאֲנִי חָצוּף שֶׁאֲנִי מְדַבֵּר לִפְנֵי הַנָּשִׂיא!".

לִבּוֹ דָּפַק בְּחָזְקָה בְּתוֹךְ חָזֵהוּ, כְּמוֹ תֹּף מִלְחָמָה: בּוּם, בּוּם, בּוּם. מֵרֹב מֶתַח, כַּפּוֹת יָדָיו הֵזִיעוּ וְנִדְבְּקוּ לִקְלַף הַסֵּפֶר. הוּא הִשְׁפִּיל אֶת מַבָּטוֹ לָרִצְפָּה וְהֶחְלִיט לִשְׁתֹּק.

מַבָּטוֹ שֶׁל נְשִׂיא יִשְׂרָאֵל

אֲבָל רַבִּי יְהוּדָה הַנָּשִׂיא לֹא הָיָה רַק תַּלְמִיד חָכָם, הוּא הָיָה רוֹעֶה צֹאן רַחֲמָן שֶׁרָאָה כָּל תַּלְמִיד וְתַלְמִיד. עֵינָיו הַחַכִּימוֹת שֶׁל רַבִּי, שֶׁהָיוּ צְלוּלוֹת כִּבְדֹלַח, שׁוֹטְטוּ עַל פְּנֵי הַשּׁוּרוֹת. הוּא סָרַק אֶת פְּנֵי כָּל תַּלְמִידָיו, מְחַפֵּשׂ תְּשׁוּבָה אוֹ שְׁבִיב שֶׁל אוֹר.

כַּאֲשֶׁר מַבָּטוֹ הִגִּיעַ לַשּׁוּרָה הָאַחֲרוֹנָה, הוּא עָצַר לְרֶגַע עַל לֵוִי הַצָּעִיר. רַבִּי קָלַט אֶת תְּנוּעוֹת הַגּוּף הַנִּרְגָּשׁוֹת, אֶת הַשְּׂפָתַיִם הַנָּעוֹת בְּשֶׁקֶט. רַבִּי יָדַע: לַנַּעַר הַזֶּה יֵשׁ מַשֶּׁהוּ לְהַגִּיד!

"דַּבֵּר, בְּנִי," נִשְׁמַע פִּתְאוֹם קוֹלוֹ הֶעָמֹק וְהָעָדִין שֶׁל הַנָּשִׂיא בְּחָלָל. קוֹל מָלֵא בְּאַהֲבָה. "אַל תִּפְחַד. דִּבְרֵי תּוֹרָה מְחַפְּשִׂים לֵב פָּתוּחַ, וְלֹא רַק זָקָן לָבָן. אֱמֹר מָה שֶׁיֵּשׁ בְּלִבְּךָ!".

כָּל עֵינֵי תַּלְמִידֵי הַחֲכָמִים הוּפְנוּ אֲחוֹרָה, אֶל לֵוִי הַצָּעִיר.

לֵוִי קָם עַל רַגְלָיו בִּרְעִידָה קַלָּה. הוּא כִּחְכֵּחַ בִּגְרוֹנוֹ. "מוֹרִי וְרַבִּי..." הוּא לָחַשׁ, וְאָז נָשַׁם עָמֹק וְהִתְחִיל לְדַבֵּר. מִלָּה אַחַר מִלָּה, מִשְׁפָּט אַחַר מִשְׁפָּט. הוּא פָּרַס אֶת הַתְּשׁוּבָה שֶׁלּוֹ. הוּא הִבִּיעַ סְבָרָה עֲמֻקָּה וּמַדְהִימָה, שֶׁפָּתְרָה אֶת הַקֻּשְׁיָה כְּמוֹ מַפְתֵּחַ מֻשְׁלָם שֶׁפּוֹתֵחַ מַנְעוּל זָהָב סָגוּר. הַמִּלִּים שֶׁיָּצְאוּ מִפִּיו הָיוּ מְתוּקוֹת כִּדְבַשׁ!

גְּדֻלָּתָהּ שֶׁל עֲנָוָה

כְּשֶׁלֵּוִי סִיֵּם, שׁוּב הִשְׁתָּרֵר שֶׁקֶט בָּאוּלָם. אֲבָל הַפַּעַם, זֶה לֹא הָיָה שֶׁקֶט שֶׁל קוֹשִׁי, אֶלָּא שֶׁקֶט שֶׁל הִשְׁתָּאוּת וּפְלִיאָה!

רַבִּי יְהוּדָה הַנָּשִׂיא הִבִּיט בַּתַּלְמִיד. אָדָם רָגִיל, שֶׁגַּאֲוָה בְּלִבּוֹ, אוּלַי הָיָה אוֹמֵר: "בְּסֵדֶר, תְּשׁוּבָה נֶחְמָדָה יֶלֶד, אֲבָל אֲנִי הָרַב כָּאן וַאֲנִי אֶפְסֹק אֶת הַהֲלָכָה".

אֲבָל רַבִּי? רַבִּי לֹא כָּזֶה!

חִיּוּךְ רָחָב, קוֹרֵן מֵאוֹר שֶׁל קְדֻשָּׁה, הִתְפַּשֵּׁט עַל פָּנָיו שֶׁל הַנָּשִׂיא. הוּא הִזְדַּקֵּף עַל כִּסְאוֹ, וְעֵינָיו זָלְגוּ דִּמְעוֹת גִּיל וְשִׂמְחָה. מֵרֹב הִתְרַגְּשׁוּת וְהַעֲרָכָה, עָמַד רַבִּי יְהוּדָה הַנָּשִׂיא עַל רַגְלָיו לִכְבוֹד תְּשׁוּבָתוֹ שֶׁל הַתַּלְמִיד הַצָּעִיר!

הוּא פָּנָה לְכָל עֲדַת הַחֲכָמִים, הֵרִים אֶת קוֹלוֹ כְּדֵי שֶׁכֻּלָּם יִשְׁמְעוּ לָנֶצַח אֶת דְּבָרָיו, וְהִכְרִיז אֶת הַמִּשְׁפָּט הַמְפֻרְסָם:

"אָמַר רַבִּי: הַרְבֵּה תּוֹרָה לָמַדְתִּי מֵרַבּוֹתַי, וּמֵחֲבֵרַי לָמַדְתִּי יוֹתֵר מֵהֶם... אֲבָל מִתַּלְמִידַי – לָמַדְתִּי יוֹתֵר מִכֻּלָּם!"

אֵיזוֹ גְּדֻלָּה! אֵיזוֹ עֲנָוָה! רַבִּי, נְשִׂיא יִשְׂרָאֵל הָעֲנָק, מֵעִיד לִפְנֵי כֻּלָּם שֶׁהוּא לוֹמֵד הֲכִי הַרְבֵּה דַּוְקָא מֵהַתַּלְמִידִים הַצְּעִירִים. כִּי הַתּוֹרָה שַׁיֶּכֶת לְכֻלָּם, וְהַקָּדוֹשׁ בָּרוּךְ הוּא שָׂם אֶת חָכְמָתוֹ בְּלֵב מִי שֶׁעָנָיו – אֲפִלּוּ אִם הוּא הַצָּעִיר בַּחֲבוּרָה.

א. שְׁאֵלַת פְּתִיחָה (לְעוֹרֵר סַקְרָנוּת):

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “"אמר רבי: ...ומתלמידי יותר מכולם"” על רבי יהודה הנשיא (רבי)? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “לִבּוֹ דָּפַק בְּחָזְקָה בְּתוֹךְ חָזֵהוּ, כְּמוֹ תֹּף מִלְחָמָה: בּוּם, בּוּם, בּוּם.”.
ג. אילו רבי יהודה הנשיא (רבי) לא היה פועל כפי שמתואר ב־“אָדָם רָגִיל, שֶׁגַּאֲוָה בְּלִבּוֹ, אוּלַי הָיָה אוֹמֵר: "בְּסֵדֶר, תְּשׁוּבָה נֶחְמָדָה יֶלֶד, אֲבָל אֲנִי הָרַב כָּאן וַאֲנִי…” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “הַיּוֹם נִשְׁמַע סִפּוּר עַל הָרַב הֲכִי גָּדוֹל וְהֲכִי עָשִׁיר בָּעוֹלָם, נְשִׂיא עַם יִשְׂרָאֵל!” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. להלן מערך חינוכי, רגשי ומרתק על רבי יהודה הנשיא (רבינו הקדוש) , המעובד…
2. רַבִּי יָדַע שֶׁהַגָּלוּת מִתְקָרֶבֶת, וְלָכֵן אָסַף אֶת כָּל…
3. הוּא הִבִּיט יָמִינָה, וְרָאָה אֶת רַבִּי שִׁמְעוֹן הַגָּדוֹל יוֹשֵׁב…
4. דַּבֵּר, בְּנִי," נִשְׁמַע פִּתְאוֹם קוֹלוֹ הֶעָמֹק וְהָעָדִין שֶׁל…
5. הוּא פָּנָה לְכָל עֲדַת הַחֲכָמִים, הֵרִים אֶת קוֹלוֹ כְּדֵי שֶׁכֻּלָּם…
6. בּוֹאוּ נְתַרְגֵּל – שֶׁגַּם הַגְּדוֹלִים יְכוֹלִים לִלְמֹד…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “עֵינָיו נָצוֹצוּ מִתּוֹךְ סְפָרָיו, וּמוֹחוֹ הַמָּהִיר עָבַד לְלֹא הֶרֶף.”.
מכתב: כתבו לרבי יהודה הנשיא (רבי) בעקבות “כָּל עֵינֵי תַּלְמִידֵי הַחֲכָמִים הוּפְנוּ אֲחוֹרָה, אֶל לֵוִי הַצָּעִיר.”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “הוּא פָּנָה לְכָל עֲדַת הַחֲכָמִים, הֵרִים אֶת קוֹלוֹ כְּדֵי שֶׁכֻּלָּם יִשְׁמְעוּ לָנֶצַח אֶת דְּבָרָיו, וְהִכְרִיז אֶת…”.
דיווח: כתבו ידיעה קצרה שכותרתה “"אמר רבי: ...ומתלמידי יותר מכולם"”.
