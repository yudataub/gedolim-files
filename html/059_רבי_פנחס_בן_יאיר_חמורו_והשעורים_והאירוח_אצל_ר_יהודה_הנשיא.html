חמורו והשעורים והאירוח אצל ר' יהודה הנשיא

דף עבודה בעקבות רבי פנחס בן יאיר
שם: כיתה:

המעשה — הסיפור המלא

פְּתִיחָה: הָאִישׁ שֶׁקָּדוֹשׁ בְּכָל מַעֲשָׂיו

יְלָדִים, יֵשׁ אֲנָשִׁים שֶׁהֵם צַדִּיקִים גְּדוֹלִים בְּבֵית הַכְּנֶסֶת. כְּשֶׁהֵם מִתְפַּלְלִים, רוֹאִים עֲלֵיהֶם אוֹר מְיֻחָד. אֲבָל כְּשֶׁהֵם יוֹצְאִים לָרְחוֹב, הֵם נִרְאִים כְּמוֹ כֻּלָּם. אֲבָל הַתַּנָּא הַקָּדוֹשׁ, רַבִּי פִּינְחָס בֶּן יָאִיר , הָיָה קָדוֹשׁ בְּכָל רֶגַע וָרֶגַע שֶׁל חַיָּיו. אֲפִלּוּ כְּשֶׁהָלַךְ בַּדֶּרֶךְ, אֲפִלּוּ כְּשֶׁאָכַל, וַאֲפִלּוּ... הַחֲמוֹר שֶׁלּוֹ! הָיָה קָדוֹשׁ וְטָהוֹר.

רַבִּי פִּינְחָס הָיָה אָדָם שֶׁמֵּעוֹלָם – מֵאָז שֶׁהָיָה יֶלֶד קָטָן וְעַד שֶׁהִזְדַּקֵּן – לֹא אָכַל פְּרוּסַת לֶחֶם בְּבַיִת שֶׁל מִישֶׁהוּ אַחֵר. הוּא לֹא רָצָה לֵהָנוֹת וְלָקַחַת מֵאֲנָשִׁים שֶׁמָּא אֵין לָהֶם מַסְפִּיק וְהֵם מִתְבַּיְּשִׁים לְהַגִּיד, אוֹ שֶׁמָּא הֵם נוֹתְנִים לֹא בְּלֵב שָׁלֵם. הוּא סָמַךְ רַק עַל הַקָּדוֹשׁ בָּרוּךְ הוּא שֶׁיְּפַרְנֵס אוֹתוֹ.

הַהֲלִיכָה בַּדֶּרֶךְ וְרֵיחַ הַשְּׂעוֹרִים

יוֹם אֶחָד, יָצָא רַבִּי פִּינְחָס בֶּן יָאִיר לְמַסָּע חָשׁוּב מְאֹד. מַסָּע שֶׁל "פִּדְיוֹן שְׁבוּיִים" – לְהַצִּיל יְהוּדִים עֲנִיִּים שֶׁנָּפְלוּ בַּשֶּׁבִי, וְהוּא אָסַף עֲבוּרָם מָמוֹן. הַשֶּׁמֶשׁ קָפְחָה עַל רֹאשׁוֹ, רוּחַ מְלֵאַת אָבָק מִדְבָּרִי נָשְׁבָה בִּשְׂעָרוֹ, וְרַגְלֵי הַחֲמוֹר הַנֶּאֱמָן שֶׁלּוֹ הִדְהֲדוּ עַל אַבְנֵי הַדֶּרֶךְ – טְרָאק, טְרָאק, טְרָאק. הַחֲמוֹר נָשָׂא עַל גַּבּוֹ אֶת חֲפָצָיו שֶׁל הָרַב בִּמְסִירוּת, עַד שֶׁהִגִּיעוּ לְאֻכְסַנְיָה (מָלוֹן שֶׁל יָמִים עָבָרוּ) כְּדֵי לָנוּחַ מְעַט.

הָרַב יָשַׁב בַּצֵּל לִלְמֹד תּוֹרָה, וּבַעֲלֵי הָאֻכְסַנְיָה – שֶׁרָאוּ שֶׁמְּדֻבָּר בְּרַב גָּדוֹל וְצַדִּיק – רָצוּ מְאֹד לְכַבֵּד אֶת הַחֲמוֹר שֶׁלּוֹ. הֵם הֵבִיאוּ קְעָרַת עֵץ גְּדוֹלָה, וּמִלְּאוּ אוֹתָהּ בִּשְׂעוֹרִים טְרִיּוֹת וְזֹהֲבוֹת. רֵיחַ מָתוֹק וְנָעִים שֶׁל תְּבוּאָה מִלֵּא אֶת הָאֲוִיר. "בְּתֵאָבוֹן, חֲמוֹר נִכְבָּד!" אָמְרוּ לוֹ בַּחִיּוּךְ.

אֲבָל לְהַפְתָּעָתָם... הַחֲמוֹר, שֶׁהָיָה עָיֵף וְרָעֵב מְאֹד, רִחְרַח אֶת הַשְּׂעוֹרִים, סוֹבֵב אֶת הָרֹאשׁ שֶׁלּוֹ לַצַּד, וְסֵרֵב לֶאֱכֹל!

"אוּלַי יֵשׁ שָׁם לִכְלוּךְ?" חָשְׁבוּ הָעוֹבְדִים. הֵם לָקְחוּ אֶת הַשְּׂעוֹרִים, בֵּרְרוּ אוֹתָן מֵאֲבָנִים קְטַנּוֹת, נִקּוּ כָּל גַּרְגִּיר אָבָק, וְהִגִּישׁוּ שׁוּב לַחֲמוֹר. עַכְשָׁו הַשְּׂעוֹרִים הָיוּ נְקִיּוֹת וּמוּכָנוֹת לִסְעוּדַת מְלָכִים.

אַךְ שׁוּב – הַחֲמוֹר עָצַם עֵינַיִם, אָטַם אֶת פִּיו מִבַּעַד לַשִּׁנַּיִם הַגְּדוֹלוֹת שֶׁלּוֹ, וְלֹא נָגַע בְּאוֹכֶל!

סוֹדוֹ שֶׁל הַחֲמוֹר

מְבֻלְבָּלִים וּמֻדְאָגִים, נִגְּשׁוּ בַּעֲלֵי הָאֻכְסַנְיָה אֶל רַבִּי פִּינְחָס. "רַבֵּנוּ," אָמְרוּ בְּצַעַר, "הַחֲמוֹר שֶׁלְּךָ בַּרְרָן מְאֹד. אוּלַי הוּא חוֹלֶה? הוּא מְמָאֵן (מְסָרֵב) לִטְעֹם מֵהַשְּׂעוֹרִים הֲכִי טוֹבוֹת שֶׁלָּנוּ!".

חִיֵּךְ רַבִּי פִּינְחָס בֶּן יָאִיר חִיּוּךְ רָגוּעַ וְאוֹהֵב, וְשָׁאַל אוֹתָם בְּקוֹל שָׁקֵט: "בָּנַי הַיְקָרִים, הַאִם עִשַּׂרְתֶּם אֶת הַשְּׂעוֹרִים? הַאִם הוֹצֵאתֶם מַעֲשֵׂר עָנִי כְּפִי שֶׁצִּוְּתָה הַתּוֹרָה?".

בַּעֲלֵי הַמָּקוֹם תָּפְסוּ אֶת רֹאשָׁם בְּבֶהָלָה. "אוֹי וַאֲבוֹי! שָׁכַחְנוּ לְהַפְרִישׁ מַעַשְׂרוֹת מֵהַשַּׂק הַזֶּה! אֲבָל... מֵאֵיפֹה הַחֲמוֹר יוֹדֵעַ?!".

הֵם מִהֲרוּ, הִפְרִישׁוּ תְּרוּמוֹת וּמַעַשְׂרוֹת כַּהֲלָכָה, וְהִגִּישׁוּ אֶת הָאֹכֶל מֵחָדָשׁ. וְרְאוּ זֶה פֶּלֶא! הַחֲמוֹר נָגַס בַּשְּׂעוֹרִים בְּתֵאָבוֹן רַב וְאָכַל בְּשִׂמְחָה!

יְלָדִים, אָמְרוּ עַל כָּךְ חֲכָמִים מִשְׁפָּט מַדְהִים: שֶׁבִּגְלַל שֶׁרַבִּי פִּינְחָס בֶּן יָאִיר הָיָה כָּל כָּךְ צַדִּיק, כָּל כָּךְ יָשָׁר, שֶׁמֵּעוֹלָם לֹא לָקַח מַשֶּׁהוּ שֶׁלֹּא שֶׁלּוֹ אוֹ שֶׁלֹּא מַתְאִים לַהֲלָכָה – הַקָּדוֹשׁ בָּרוּךְ הוּא שָׁמַר אֲפִלּוּ עַל הַבְּהֵמָה שֶׁלּוֹ, שֶׁלֹּא תֹּאכַל אֹכֶל שֶׁלֹּא עֻשַּׂר הֵיטֵב! אִם כָּךְ נוֹהֵג הַחֲמוֹר, תּאֲרוּ לָכֶם מִי הָיָה הָרַב בְּעַצְמוֹ!

הַהַזְמָנָה שֶׁל הַנָּשִׂיא

הַמַּסָּע הִמְשִׁיךְ. הַשְּׁמוּעָה שֶׁרַבִּי פִּינְחָס בֶּן יָאִיר עוֹבֵר בַּסְּבִיבָה הִגִּיעָה עַד לְאַרְמוֹנוֹ שֶׁל נְשִׂיא יִשְׂרָאֵל, הָאִישׁ הֶחָשׁוּב בְּיוֹתֵר בַּדּוֹר – רַבִּי יְהוּדָה הַנָּשִׂיא (שֶׁקּוֹרְאִים לוֹ גַּם "רַבִּי"). רַבִּי יְהוּדָה הַנָּשִׂיא, שֶׁהָיָה עָשִׁיר גָּדוֹל וְלָמַד תּוֹרָה מִתּוֹךְ כָּבוֹד וָעֹשֶׁר, כָּל כָּךְ שָׂמַח! הוּא מִהֵר לְקַבֵּל אֶת פָנָיו שֶׁל רַבִּי פִּינְחָס בֶּן יָאִיר, וְאָמַר לוֹ מִתּוֹךְ אַהֲבָה: "רַבִּי פִּינְחָס הַקָּדוֹשׁ, כַּמָּה אֲנִי שָׂמֵחַ שֶׁבָּאתָ! אֲנִי מִתְחַנֵּן אֵלֶיךָ, הִכָּנֵס לְבֵיתִי! סְעַד (תֹּאכַל) אֶצְלִי בְּבֵיתִי!".

רַבִּי פִּינְחָס הִסְתַּכֵּל עַל רַבִּי יְהוּדָה הַנָּשִׂיא. הוּא יָדַע שֶׁאֵצֶל רַבִּי יְהוּדָה יֵשׁ שֶׁפַע גָּדוֹל, וְהוּא נוֹתֵן מִכָּל הַלֵּב וּמֵאַהֲבַת תּוֹרָה. לָכֵן, לַמְרוֹת שֶׁרַבִּי פִּינְחָס מֵעוֹלָם לֹא אָכַל אֵצֶל אֲנָשִׁים אֲחֵרִים, הַפַּעַם הִנְהֵן בְּרֹאשׁוֹ וְאָמַר: "הֵן. מֵאַחַר שֶׁאַתָּה נוֹתֵן בְּלֵב שָׁלֵם וְיֵשׁ לְךָ מִשֶּׁלְּךָ כָּל כָּךְ הַרְבֵּה, אָבוֹא הַיּוֹם לֶאֱכֹל בְּבֵיתְךָ".

רַבִּי יְהוּדָה הַנָּשִׂיא זָרַח מִשִּׂמְחָה. זוֹ זְכוּת עֲצוּמָה לְאָרֵחַ אֶת רַבִּי פִּינְחָס!

הַקּוֹלוֹת שֶׁבַּחֲצֵר הַנָּשִׂיא

הִגִּיעָה שְׁעַת הַסְּעוּדָה. רַבִּי פִּינְחָס הִתְקָרֵב אֶל חֲצַר הָאַרְמוֹן הַיָּפֶה שֶׁל רַבִּי יְהוּדָה הַנָּשִׂיא. שַׁעֲרֵי בַּרְזֶל מְעֻטָּרִים עָמְדוּ בַּכְּנִיסָה, וּמִבִּפְנִים נִשְׁמַע רַעַשׁ נַהֲמוֹת סוּסִים.

רַבִּי פִּינְחָס צָעַד צַעַד אֶחָד קָדִימָה. לְפֶתַע, עֵינָיו נִפְעֲרוּ.

בְּתוֹךְ הֶחָצֵר שֶׁל רַבִּי עָמְדוּ קְשׁוּרוֹת... כֻּדְנְיָתָא חִוָּרָתָא! יְלָדִים, אֵלּוּ מִין פְּרָדוֹת (חַיּוֹת שֶׁדּוֹמוֹת לְסוּס וַחֲמוֹר יַחַד) בְּצֶבַע לָבָן.

לִכְאוֹרָה, מָה הַבְּעָיָה בִּפְרָדוֹת לְבָנוֹת? הֵן חַיּוֹת חֲזָקוֹת וִיקָרוֹת מְאֹד. אֲבָל... רַבִּי פִּינְחָס יָדַע סוֹד וַהֲלָכָה הַקְּשׁוּרִים אֲלֵיהֶן: פְּרָדוֹת לְבָנוֹת בְּאוֹתָם יָמִים הָיוּ מְסֻכָּנוֹת מְאֹד! כְּשֶׁהֵן הָיוּ בּוֹעֲטוֹת עִם הַפְּרָסוֹת (הָרַגְלַיִם) שֶׁלָּהֶן, הֵן הָיוּ פוֹצְעוֹת אֲנָשִׁים, וְלִפְעָמִים הַפְּצִיעָה הָיְתָה כָּל כָּךְ קָשָׁה שֶׁהַפֶּצַע לֹא הָיָה מִתְרַפֵּא!

הַתּוֹרָה אוֹמֶרֶת: "וְלֹא תָשִׂים דָּמִים בְּבֵיתֶךָ" – אָסוּר לְאָדָם לְהַחְזִיק בְּבֵיתוֹ דָּבָר שֶׁמְּסַכֵּן אֲנָשִׁים אֲחֵרִים, כְּמוֹ סֻלָּם רָעוּעַ אוֹ חַיָּה מְסֻכָּנֶת.

כְּשֶׁרַבִּי פִּינְחָס רָאָה אֶת הַפְּרָדוֹת הַמְּסֻכָּנוֹת מְהַדְּקוֹת אֶת רַגְלֵיהֶן בַּקַּרְקַע, הוּא נֶעֱצַר בַּמָּקוֹם. מַבָּטוֹ הִרְצִין. "מַלְאַךְ הַמָּוֶת בְּבֵיתוֹ שֶׁל זֶה, וַאֲנִי אֶסְעַד אֶצְלוֹ?!" אָמַר בְּקוֹל כּוֹאֵב, וּמִיָּד הִסְתּוֹבֵב וְהָלַךְ לְאָחוֹר! הוּא לֹא נִכְנַס!

"לִפְנִים מִשּׁוּרַת הַדִּין"

רַבִּי יְהוּדָה הַנָּשִׂיא שָׁמַע שֶׁהָרַב מִסְתּוֹבֵב וְעוֹזֵב. הוּא מִהֵר, רָץ מִתּוֹךְ הַבַּיִת, מְגִלַּת בְּגָדָיו הַמְכֻבָּדִים מִתְנַפְנֶפֶת מֵאֲחוֹרָיו.

"רַבִּי פִּינְחָס! רַבִּי פִּינְחָס, לָמָּה אַתָּה עוֹזֵב?" קָרָא אֵלָיו.

"בִּגְלַל הַפְּרָדוֹת הַלְּבָנוֹת. הֵן מְסַכְּנוֹת אֶת הָאֲנָשִׁים. אֵינֶנִּי נִכְנָס לְמָקוֹם שֶׁיֵּשׁ בּוֹ סַכָּנָה כָּזוֹ לִיהוּדִים!" עָנָה רַבִּי פִּינְחָס.

רַבִּי יְהוּדָה הַנָּשִׂיא (שֶׁכַּנִּרְאֶה קָנָה אוֹתָן אוֹ קִבֵּל אוֹתָן וְדָאַג לִשְׁמֹר עֲלֵיהֶן הֵיטֵב שֶׁלֹּא יַזִּיקוּ), רָצָה כָּל כָּךְ שֶׁהָרַב יִשָּׁאֵר, עַד שֶׁהוּא זָעַק: "אִם הֵן מַפְרִיעוֹת לְךָ, הֲרֵינִי מוֹכֵר אוֹתָן לְגוֹיִים מִיָּד! מָחָר הֵן לֹא כָּאן!".

אֲבָל רַבִּי פִּינְחָס סֵרֵב: "לֹא. אָסוּר לִמְכֹּר חַיָּה מַזִּיקָה. וְלִפְנֵי עִוֵּר לֹא תִתֵּן מִכְשׁוֹל. זֶה יַזִּיק לְאֶחָד אַחֵר!".

"אָז... אָז אַפְקִיר אוֹתָן! אֲשַׁחְרֵר אוֹתָן!" הוֹסִיף רַבִּי יְהוּדָה הַנָּשִׂיא.

"לֹא," עָנָה רַבִּי פִּינְחָס, "הֵן יִסְתּוֹבְבוּ בָּרְחוֹבוֹת וְיַזִּיקוּ יְהוּדִים עַל יְדֵי בְּעִיטָה! זֶה גָּרוּעַ יוֹתֵר!".

"אָז אֲנִי אֶעֱקֹר אֶת הַפְּרָסוֹת שֶׁלָּהֶן כְּדֵי שֶׁלֹּא יוּכְלוּ לִבְעֹט כְּלָל!" הִתְחַנֵּן נְשִׂיא יִשְׂרָאֵל. הוּא הָיָה מוּכָן לְהַפְסִיד מָמוֹן רַב כְּדֵי שֶׁרַבִּי פִּינְחָס יֹאכַל אֶצְלוֹ.

אֲבָל כָּאן הִבִּיט רַבִּי פִּינְחָס בְּעֵינָיו הַטְּהוֹרוֹת. "צַעַר בַּעֲלֵי חַיִּים," אָמַר לוֹ. "אָסוּר לְצַעֵר אֶת הַחַיָּה סְתָם וְלַעֲקֹר לָהּ אֶת הַפַּרְסָה שֶׁכּוֹאֶבֶת לָהּ. זֶה אָסוּר מִן הַתּוֹרָה".

רַבִּי יְהוּדָה הַנָּשִׂיא נֶאֱנַח. "אָז... אֲנִי אֶהֱרֹג אוֹתָן מִיָּד, שֶׁלֹּא יִסְבְּלוּ וְשֶׁלֹּא יַזִּיקוּ".

וְאָז, מִשֶּׁהֵבִין רַבִּי פִּינְחָס עַד כַּמָּה רַבִּי יְהוּדָה הַנָּשִׂיא מוּכָן לְוַתֵּר עֲבוּרוֹ, הוּא חָתַךְ בְּקוֹל שָׁקֵט וּבָרוּר: "אָסוּר, בַּל תַּשְׁחִית. אָסוּר לְהַשְׁחִית דָּבָר שֶׁיֵּשׁ בּוֹ עֵרֶךְ בָּעוֹלָם סְתָם כָּךְ".

כְּשֶׁרָאָה רַבִּי יְהוּדָה שֶׁרַבִּי פִּינְחָס מַקְפִּיד וְעוֹמֵד עַל שֶׁלּוֹ כְּמוֹ חוֹמָה בְּצוּרָה, הֵחֵל רַבִּי יְהוּדָה לְהַפְצִיר בּוֹ עוֹד וְעוֹד: "רַבִּי, אֲנִי אֶדְאַג לְכָךְ בַּדֶּרֶךְ הֲכִי נְכוֹנָה! בּוֹא לְתוֹךְ בֵּיתִי, רַק לִסְעוּדָה אַחַת!".

רַבִּי פִּינְחָס רָאָה כַּמָּה נְשִׂיא יִשְׂרָאֵל מִצְטַעֵר. מָה אַתֶּם מַצִּיעִים לַעֲשׂוֹת? אָדָם רָגִיל הָיָה חוֹשֵׁב: "טוֹב, הַנָּשִׂיא כָּל כָּךְ מְבַקֵּשׁ, לֹא נָעִים מִמֶּנּוּ, אֶכָּנֵס רַק לְחָמֵשׁ דַּקּוֹת...".

אֲבָל רַבִּי פִּינְחָס בֶּן יָאִיר הָיָה קוֹמַת קֹדֶשׁ. בִּמְקוֹם לַעֲנוֹת לְרַבִּי יְהוּדָה עַל הַפְצָרוֹתָיו, קָרָא רַבִּי פִּינְחָס דְּבָרִים שֶׁהָיוּ נִשְׁמָעִים מַמָּשׁ כְּדִבְרֵי תּוֹכֵחָה סוֹדִיִּים. בְּאוֹתוֹ הָרֶגַע, הִבִּיט רַבִּי יְהוּדָה הַנָּשִׂיא סְבִיבוֹ. הַהַד הִשְׁתַּתֵּק. הָרַב סִבֵּב אֶת הַחֲמוֹר הַנֶּאֱמָן שֶׁלּוֹ, וְצָעַד לְאַט-לְאַט, מִתְרַחֵק מֵהָאַרְמוֹן שֶׁל הַנָּשִׂיא, מַשְׁאִיר מֵאֲחוֹרָיו אֶת הַסְּעוּדָה הַמְפֹאֶרֶת בְּיוֹתֵר שֶׁאֶפְשָׁר לְדַמְיֵן, רַק כְּדֵי שֶׁלֹּא לְהִתְקָרֵב לְמָקוֹם שֶׁמּוּנָח בּוֹ דָּבָר שֶׁיָּכוֹל אֲפִלּוּ קְצָת לְהַזִּיק לְיְהוּדִי אַחֵר!

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “יְלָדִים, יֵשׁ אֲנָשִׁים שֶׁהֵם צַדִּיקִים גְּדוֹלִים בְּבֵית הַכְּנֶסֶת.” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “הַשֶּׁמֶשׁ קָפְחָה עַל רֹאשׁוֹ, רוּחַ מְלֵאַת אָבָק מִדְבָּרִי נָשְׁבָה בִּשְׂעָרוֹ, וְרַגְלֵי הַחֲמוֹר הַנֶּאֱמָן שֶׁלּוֹ הִדְהֲדוּ עַל אַבְנֵי הַדֶּרֶךְ…”?
ג. כיצד הגיב רבי פנחס בן יאיר למצב המתואר: “מְבֻלְבָּלִים וּמֻדְאָגִים, נִגְּשׁוּ בַּעֲלֵי הָאֻכְסַנְיָה אֶל רַבִּי פִּינְחָס.”?
ד. מה השתנה בין “מֵאַחַר שֶׁאַתָּה נוֹתֵן בְּלֵב שָׁלֵם וְיֵשׁ לְךָ מִשֶּׁלְּךָ כָּל כָּךְ הַרְבֵּה, אָבוֹא הַיּוֹם לֶאֱכֹל בְּבֵיתְךָ".” לבין “הוּא מִהֵר, רָץ מִתּוֹךְ הַבַּיִת, מְגִלַּת בְּגָדָיו הַמְכֻבָּדִים מִתְנַפְנֶפֶת מֵאֲחוֹרָיו.”?
ה. כיצד הגיע הסיפור אל האירוע: “אָסוּר לְהַשְׁחִית דָּבָר שֶׁיֵּשׁ בּוֹ עֵרֶךְ בָּעוֹלָם סְתָם כָּךְ".”?

□ אֵינֶנִּי נִכְנָס לְמָקוֹם שֶׁיֵּשׁ בּוֹ סַכָּנָה כָּזוֹ לִיהוּדִים!" עָנָה רַבִּי פִּינְחָס.

לפני ואחרי

מה קרה לפני: “הוּא מִהֵר לְקַבֵּל אֶת פָנָיו שֶׁל רַבִּי פִּינְחָס בֶּן יָאִיר, וְאָמַר לוֹ מִתּוֹךְ אַהֲבָה: "רַבִּי פִּינְחָס הַקָּדוֹשׁ, כַּמָּה אֲנִי שָׂמֵחַ…”?
מה קרה אחרי: “אֵינֶנִּי נִכְנָס לְמָקוֹם שֶׁיֵּשׁ בּוֹ סַכָּנָה כָּזוֹ לִיהוּדִים!" עָנָה רַבִּי פִּינְחָס.”?

סיבה ותוצאה מתוך המעשה

האירוע הנתוןמה גרם לו?מה הייתה תוצאתו?
רֵיחַ מָתוֹק וְנָעִים שֶׁל תְּבוּאָה מִלֵּא אֶת הָאֲוִיר.
רַבִּי פִּינְחָס יָדַע סוֹד וַהֲלָכָה הַקְּשׁוּרִים אֲלֵיהֶן: פְּרָדוֹת לְבָנוֹת בְּאוֹתָם יָמִים הָיוּ מְסֻכָּנוֹת מְאֹד!

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “חמורו והשעורים והאירוח אצל ר' יהודה הנשיא” על רבי פנחס בן יאיר? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “הַשְּׁמוּעָה שֶׁרַבִּי פִּינְחָס בֶּן יָאִיר עוֹבֵר בַּסְּבִיבָה הִגִּיעָה עַד לְאַרְמוֹנוֹ שֶׁל נְשִׂיא יִשְׂרָאֵל, הָאִישׁ הֶחָשׁוּב בְּיוֹתֵר בַּדּוֹר…”.
ג. אילו רבי פנחס בן יאיר לא היה פועל כפי שמתואר ב־“הוּא מִהֵר, רָץ מִתּוֹךְ הַבַּיִת, מְגִלַּת בְּגָדָיו הַמְכֻבָּדִים מִתְנַפְנֶפֶת מֵאֲחוֹרָיו.” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “אָסוּר לְהַשְׁחִית דָּבָר שֶׁיֵּשׁ בּוֹ עֵרֶךְ בָּעוֹלָם סְתָם כָּךְ".” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. יְלָדִים, יֵשׁ אֲנָשִׁים שֶׁהֵם צַדִּיקִים גְּדוֹלִים בְּבֵית…
2. הַחֲמוֹר נָשָׂא עַל גַּבּוֹ אֶת חֲפָצָיו שֶׁל הָרַב בִּמְסִירוּת, עַד…
3. הֵם מִהֲרוּ, הִפְרִישׁוּ תְּרוּמוֹת וּמַעַשְׂרוֹת כַּהֲלָכָה,…
4. שַׁעֲרֵי בַּרְזֶל מְעֻטָּרִים עָמְדוּ בַּכְּנִיסָה, וּמִבִּפְנִים…
5. אָז אֲנִי אֶעֱקֹר אֶת הַפְּרָסוֹת שֶׁלָּהֶן כְּדֵי שֶׁלֹּא יוּכְלוּ…
6. בִּמְקוֹם לַעֲנוֹת לְרַבִּי יְהוּדָה עַל הַפְצָרוֹתָיו, קָרָא רַבִּי…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “מְבֻלְבָּלִים וּמֻדְאָגִים, נִגְּשׁוּ בַּעֲלֵי הָאֻכְסַנְיָה אֶל רַבִּי פִּינְחָס.”.
מכתב: כתבו לרבי פנחס בן יאיר בעקבות “רַבִּי פִּינְחָס יָדַע סוֹד וַהֲלָכָה הַקְּשׁוּרִים אֲלֵיהֶן: פְּרָדוֹת לְבָנוֹת בְּאוֹתָם יָמִים הָיוּ מְסֻכָּנוֹת מְאֹד!”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “הוּא הָיָה מוּכָן לְהַפְסִיד מָמוֹן רַב כְּדֵי שֶׁרַבִּי פִּינְחָס יֹאכַל אֶצְלוֹ.”.
דיווח: כתבו ידיעה קצרה שכותרתה “חמורו והשעורים והאירוח אצל ר' יהודה הנשיא”.

המחזה

רגע מן הסיפורהדמויותמה יאמרו או יעשו?
רֵיחַ מָתוֹק וְנָעִים שֶׁל תְּבוּאָה מִלֵּא אֶת הָאֲוִיר.
הוּא מִהֵר, רָץ מִתּוֹךְ הַבַּיִת, מְגִלַּת בְּגָדָיו הַמְכֻבָּדִים מִתְנַפְנֶפֶת מֵאֲחוֹרָיו.

7. חיבור לחיים מתוך המעשה

א. מה אפשר ללמוד מן הדרך שבה רבי פנחס בן יאיר פעל באירוע “מֵאַחַר שֶׁאַתָּה נוֹתֵן בְּלֵב שָׁלֵם וְיֵשׁ לְךָ מִשֶּׁלְּךָ כָּל כָּךְ הַרְבֵּה, אָבוֹא הַיּוֹם לֶאֱכֹל בְּבֵיתְךָ".”?
