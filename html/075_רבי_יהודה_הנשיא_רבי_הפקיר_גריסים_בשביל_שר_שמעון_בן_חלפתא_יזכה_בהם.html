הפקיר גריסים בשביל שר' שמעון בן חלפתא יזכה בהם

דף עבודה בעקבות רבי יהודה הנשיא (רבי)
שם: כיתה:

המעשה — הסיפור המלא

להלן מערך חינוכי וסיפור מעובד, מרגש ומרתק, על רבי יהודה הנשיא (רבנו הקדוש) ורבי שמעון בן חלפתא . הסיפור בנוי בדיוק לפי ההנחיות עבור בנים בכיתות ב'-ד' בחינוך הדתי-לאומי, תוך שימוש בעקרונות סטוריטלינג, תיאורים חושיים, כבוד התורה, ומוסר השכל ישים.

הסיפור מנוקד במלואו כנדרש.

פְּתִיחָה: הָעֹשֶׁר הַגָּדוֹל וְהָעֹנִי הַמָּרוּד

יְלָדִים, אִם נִכָּנֵס לְרֶגַע אֶל בֵּיתוֹ שֶׁל נְשִׂיא יִשְׂרָאֵל, רַבִּי יְהוּדָה הַנָּשִׂיא, מָה נִרְאֶה?

נִרְאֶה אַרְמוֹן שֶׁל תּוֹרָה, מָלֵא בְּהָדָר וּבְכָבוֹד. רַבִּי יְהוּדָה לֹא הָיָה רַק גְּדוֹל הַדּוֹר שֶׁכָּתַב אֶת הַמִּשְׁנָה, אֶלָּא הָיָה גַּם אָדָם עָשִׁיר מְאֹד. בְּבֵיתוֹ תָּמִיד נָשַׁב רֵיחַ מָתוֹק וְחַם שֶׁל לֶחֶם טָרִי שֶׁזֶּה עַתָּה יָצָא מִן הַתַּנּוּר, תַּבְשִׁילֵי בָּשָׂר וִירָקוֹת עֲסִיסִיִּים שֶׁבִּעְבְּעוּ בַּסִּירִים. רַבִּי הָיָה מְאָרֵחַ תַּלְמִידֵי חֲכָמִים רַבִּים עַל שֻׁלְחָנוֹ, וְדָאַג לְפַרְנָסָתָם שֶׁל עֲנִיִּים רַבִּים.

אַךְ לֹא הַרְחֵק מִשָּׁם, בְּבַיִת קָטָן עָשׂוּי מֵאֲבָנִים פְּשׁוּטוֹת, גָּר צַדִּיק אַחֵר. קָרְאוּ לוֹ רַבִּי שִׁמְעוֹן בֶּן חֲלַפְתָּא. רַבִּי שִׁמְעוֹן הָיָה צַדִּיק נִפְלָא, אָדָם שֶׁכָּל דִּבּוּרוֹ הָיָה תּוֹרָה וְקְדֻשָּׁה, אֲבָל בְּבֵיתוֹ – הָיָה עֹנִי נורא. אִם הֱיִיתֶם עוֹמְדִים בַּפֶּתַח שֶׁל בֵּיתוֹ, לֹא הֱיִיתֶם מְרִיחִים לֶחֶם חַם. הֱיִיתֶם שׁוֹמְעִים אֶת קוֹל הָרוּחַ הַקָּרָה נוֹשֶׁבֶת דֶּרֶךְ סְדָקִים בַּקִּירוֹת, וְרוֹאִים אֶת הַשֻּׁלְחָן הָרֵיק. פְּעָמִים רַבּוֹת לֹא הָיָה לְרַבִּי שִׁמְעוֹן וְלְמִשְׁפַּחְתּוֹ אֲפִלּוּ פֵּרוּר לֶחֶם לֶאֱכֹל כְּדֵי לְהַשְׂבִּיעַ אֶת בִּטְנָם הַמְקַרְקֶרֶת מֵרָעָב. וּבְכָל זֹאת – חִיּוּךְ גָּדוֹל הָיָה תָּמִיד עַל שְׂפָתָיו שֶׁל רַבִּי שִׁמְעוֹן. תּוֹרַת ה' הָיְתָה הַשִּׂמְחָה שֶׁלּוֹ. אֲבָל הַגּוּף... הַגּוּף דָּרַשׁ אֶת שֶׁלּוֹ, וְהַיְלָדִים בַּבַּיִת הָיוּ רְעֵבִים.

הַדְּאָגָה בְּלֵב הַנָּשִׂיא

דָּבָר אֵינוֹ נִסְתָּר מֵעֵינָיו וּמִלִּבּוֹ שֶׁל מַנְהִיג אֲמִתִּי. רַבִּי יְהוּדָה הַנָּשִׂיא רָאָה מֵרָחוֹק שֶׁרַבִּי שִׁמְעוֹן תַּלְמִידוֹ נִרְאֶה חִוֵּר וְחַלָּשׁ. הוּא הֵבִין שֶׁבְּבֵיתוֹ שֶׁל רַבִּי שִׁמְעוֹן שׂוֹרֵר חֹסֶר עָצוּם.

לִבּוֹ שֶׁל רַבִּי יְהוּדָה נִחְמַץ בְּקִרְבּוֹ. "כֵּיצַד יִתָּכֵן שֶׁאֲנִי יוֹשֵׁב בְּאַרְמוֹן, וְצַדִּיק יְסוֹד עוֹלָם סוֹבֵל חֶרְפַּת רָעָב?" חָשַׁב רַבִּי. הַיָּד שֶׁלּוֹ כְּבָר נָגְעָה בְּכִיסוֹ הַמָּלֵא בְּמַטְבְּעוֹת שֶׁל זָהָב וָכֶסֶף שֶׁצִּלְצְלוּ בְּקוֹל עָמוּם. קַל מְאֹד הָיָה לְרַבִּי לִקְרֹא לְמְשָׁרְתָיו וְלוֹמַר: "קְחוּ מֵאָה מַטְבְּעוֹת זָהָב לְבֵיתוֹ שֶׁל רַבִּי שִׁמְעוֹן!"

אֲבָל רַבִּי עָצַר.

הוּא יָדַע מִי זֶה רַבִּי שִׁמְעוֹן בֶּן חֲלַפְתָּא. רַבִּי שִׁמְעוֹן שָׂנֵא מַתָּנוֹת. "שׂוֹנֵא מַתָּנוֹת יִחְיֶה!" כָּךְ הָיָה אוֹמֵר. הוּא לֹא הָיָה מַסְכִּים בְּשׁוּם אֹפֶן לְקַבֵּל צְדָקָה, כִּי יָדַע שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא זָן וּמְפַרְנֵס כָּל בְּרִיָּה. אִם רַבִּי יְהוּדָה יִשְׁלַח אֵלָיו כֶּסֶף, רַבִּי שִׁמְעוֹן יַסְמִיק, עֵינָיו יִמָּלְאוּ דְּמָעוֹת שֶׁל בּוּשָׁה, וְהוּא יַחֲזִיר אֶת הַכֶּסֶף לְאַלְתַּר.

"כְּבוֹדוֹ שֶׁל יְהוּדִי יָקָר לִי יוֹתֵר מִכָּל זְהַב הָעוֹלָם," לָחַשׁ רַבִּי לְעַצְמוֹ. "אֲנִי חַיָּב לִמְצֹא דֶּרֶךְ לָתֵת לוֹ אֹכֶל, מִבְּלִי שֶׁיֵּדַע שֶׁזּוֹ עֶזְרָה מִמֶּנִּי. הוּא חַיָּב לַחְשֹׁב שֶׁקִּבֵּל זֹאת יְשִׁירוֹת מִן הַשָּׁמַיִם, מִבְּלִי שֶׁאָדָם יְרַחֵם עָלָיו".

הַתָּכְנִית הַסּוֹדִית קוֹרֶמֶת עוֹר וְגִידִים

לְמָחֳרַת בַּבֹּקֶר, רַבִּי יְהוּדָה קָרָא לִשְׁנֵי מְשָׁרְתִים נֶאֱמָנִים אֶל חֲדַר הַסּוֹדוֹת שֶׁבְּאַרְמוֹנוֹ.

"שִׂימוּ לֵב הֵיטֵב אֶל הַמְּשִׂימָה הַזּוֹ," לָחַשׁ רַבִּי וּפָנָיו רְצִינִיּוֹת. "גְּשׁוּ לַמַּחְסָן הַגָּדוֹל, וּקְחוּ מִשָּׁם שַׂק אֶחָד כָּבֵד וְגָדוֹל, מָלֵא בְּגְרִיסִים".

יְלָדִים, אַתֶּם יוֹדְעִים מָה זֶה גְּרִיסִים? אֵלּוּ זְרָעִים וְקִטְנִיּוֹת, עֲגוּלִים, צְהֻבִּים וְנָאִים, שֶׁכַּאֲשֶׁר מְבַשְּׁלִים אוֹתָם הֵם מְכִינִים מָרָק סָמִיךְ, טָעִים, חַם וּמַשְׂבִּיעַ מְאֹד. זֶה הָיָה אֹכֶל שֶׁהָיָה יָכוֹל לְהַחְזִיק מִשְׁפָּחָה שְׁלֵמָה בְּמֶשֶׁךְ שָׁבוּעוֹת!

"קְחוּ אֶת הַשַּׂק הַזֶּה," הִמְשִׁיךְ רַבִּי יְהוּדָה, וְעֵינָיו בָּרְקוּ. "אֲנִי יוֹדֵעַ שֶׁבְּעוֹד שָׁעָה קַלָּה, יַעֲבֹר רַבִּי שִׁמְעוֹן בֶּן חֲלַפְתָּא בַּשְּׁבִיל שֶׁמִּחוּץ לָעִיר, שְׁבִיל חָרֵב שֶׁאֵין בּוֹ אֲנָשִׁים. לְכוּ לְשָׁם לִפְנֵיו. כַּאֲשֶׁר תִּרְאוּ אוֹתוֹ מִתְקָרֵב מֵרָחוֹק, עֲשׂוּ עַצְמְכֶם כְּאִלּוּ שֶׁנָּפַל לָכֶם הַשַּׂק. הַנִּיחוּ אוֹתוֹ עַל הָאֲדָמָה הַלָּבָה מִקַּרְנֵי הַשֶּׁמֶשׁ. וְאָז – הַכְרִיזוּ בְּקוֹל שֶׁהוּא יִשְׁמַע: 'הַשַּׂק הַזֶּה קָשֶׁה לָנוּ מִדַּי! אֲנַחְנוּ מַפְקִירִים אוֹתוֹ! שֶׁיִּקַּח אוֹתוֹ מִי שֶׁרוֹצֶה! הֶפְקֵר! הֶפְקֵר!'. וּמִיָּד שׁוּבוּ לְכָאן בִּמְרוּצָה".

הַמְשָׁרְתִים הִנְהֲנוּ. הֵם הֶעֱמִיסוּ עַל כִּתְפֵיהֶם שַׂק עָצוּם. יָכְלוּ לְהַרְגִּישׁ בְּיָדָם אֶת הַבַּד הַמְחֻסְפָּס שֶׁל הַשַּׂק, וְלִשְׁמֹעַ אֶת הַגְּרִיסִים הָרַבִּים מִתְחַכְּכִים זֶה בָּזֶה כְּלַחַשׁ שֶׁל נֶחָמָה.

הַשַּׂק הֶעָזוּב בָּרְחוֹב

בַּחוּץ, בַּדֶּרֶךְ הֶעָפָר, הָלַךְ לוֹ רַבִּי שִׁמְעוֹן בֶּן חֲלַפְתָּא. קוֹמָתוֹ הָיְתָה זְקוּפָה אֲבָל צַעֲדָיו הָיוּ אִטִּיִּים. הוּא הָיָה חַלָּשׁ. עוֹד מְעַט תֵּרֵד הַשַּׁבָּת, וּבְבֵיתוֹ אֵין אֲפִלּוּ כַּמָּה פְּרוּטוֹת נְחֹשֶׁת לִקְנוֹת קֶמַח.

הוּא נָשָׂא עֵינָיו לַשָּׁמַיִם וְהִתְפַּלֵּל בְּלַחַשׁ: "רִבּוֹנוֹ שֶׁל עוֹלָם, פּוֹתֵחַ אֶת יָדֶךָ וּמַשְׂבִּיעַ לְכָל חַי רָצוֹן. עֲשֵׂה חֶסֶד עִם עֲנִיֵּי עַמְּךָ, שֶׁנּוּכַל לְקַבֵּל אֶת הַשַּׁבָּת בְּשִׂמְחָה".

בְּאוֹתוֹ רֶגַע מַמָּשׁ, הִבְחִין רַבִּי שִׁמְעוֹן מֵרָחוֹק בִּשְׁנֵי אֲנָשִׁים מְמַהֲרִים, כְּשֶׁלְּפֶתַע, "בּוּם!" - שַׂק כָּבֵד וְגָדוֹל נָפַל מֵעַל כִּתְפֵיהֶם וְנֶחְבַּט בָּאֲדָמָה הַחֲשׂוּפָה.

הָאֲנָשִׁים הִבִּיטוּ בַּשַּׂק וּבִמְקוֹם לְהָרִים אוֹתוֹ, אֶחָד מֵהֶם זָעַק בְּקוֹל: "אוּף! זֶה כָּבֵד מִדַּי! כּוֹאֵב לִי הַגַּב! אֲנִי מַפְקִיר אֶת הַשַּׂק הַזֶּה! הֶפְקֵר לַכֹּל! מִי שֶׁמּוֹצֵא – שֶׁלּוֹ הַדָּבָר!".

וּמִיָּד, שְׁנֵי הָאֲנָשִׁים נָסוּ עַל נַפְשָׁם וְנֶעֶלְמוּ מֵאֲחוֹרֵי הַגְּבָעוֹת, כְּשֶׁהֵם מוֹתִירִים אֶת הַשַּׂק לְבַדּוֹ, בְּאֶמְצַע הַשְּׁבִיל.

תְּשׁוּבָה מִן הַשָּׁמַיִם

רַבִּי שִׁמְעוֹן בֶּן חֲלַפְתָּא נֶעֱצַר. הַשֶּׁקֶט חָזַר לַשָּׂדֶה. הוּא הִתְקָרֵב בִּזְהִירוּת.

כְּשֶׁרָאָה שֶׁאֵין אִישׁ מִסָּבִיב, הוּא גָּחַן אַרְצָה. הוּא הִתִּיר אֶת הַחֶבֶל הֶעָבֶה שֶׁקָּשַׁר אֶת פִּי הַשַּׂק. גַּל שֶׁל גְּרִיסִים צְהֻבִּים, בּוֹהֲקִים, מְלֵאֵי בְּרָכָה נִגְלָה לְעֵינָיו. זֶה הָיָה אֹצָר שֶׁל מָזוֹן! הָרֵיחַ שֶׁל אֲדָמָה חֲדָשָׁה וְקִטְנִיּוֹת טְרִיּוֹת מִלֵּא אֶת אַפּוֹ.

אֲבָל רַבִּי שִׁמְעוֹן הָיָה תַּלְמִיד חָכָם. הוּא הִרְהֵר בַּהֲלָכָה. "הַאִם מֻתָּר לִי לָקַחַת אֶת זֶה? וְאוּלַי זֶה שַׁיָּךְ לְמִישֶׁהוּ וְדִינוֹ כְּהֲשָׁבַת אֲבֵדָה? חָלִילָה לִי לָגַעַת בְּמָמוֹן שֶׁאֵינוֹ שֶׁלִּי!".

אַךְ אָז נִזְכַּר בַּמִּלִּים הַמְפֹרָשׁוֹת שֶׁצָּעֲקוּ הָאֲנָשִׁים: 'הֶפְקֵר! מִי שֶׁמּוֹצֵא – שֶׁלּוֹ!'. אָדָם שֶׁמַּפְקִיר כְּלִי, הַכְּלִי הוֹפֵךְ לִהְיוֹת שֶׁל הָרִאשׁוֹן שֶׁלּוֹקֵחַ אוֹתוֹ.

"הַלְלוּיָהּ!" קָרָא רַבִּי שִׁמְעוֹן וְהֵרִים אֶת עֵינָיו לַשָּׁמַיִם הַכְּחֻלִּים, כְּשֶׁדִּמְעַת אֹשֶׁר זָלְגָה עַל קְצֵה זְקָנוֹ. "בָּרוּךְ מַשְׁגִּיחַ עַל עֲנִיֵּי עַמּוֹ! לֹא מִתּוֹךְ בּוּשָׁה אוֹ מַתְּנַת בָּשָׂר וָדָם, אֶלָּא מִתּוֹךְ הֶפְקֵר, מִיָּדוֹ הַפְּתוּחָה שֶׁל בּוֹרֵא עוֹלָם!"

הוּא הֶעֱמִיס אֶת הַשַּׂק מְלֵא הַגְּרִיסִים עַל כְּתֵפוֹ הַחַלָּשָׁה, אֲבָל לְפֶתַע, הַשַּׂק כְּלָל לֹא הִרְגִּישׁ כָּבֵד. הַשִּׂמְחָה נָתְנָה לוֹ כּוֹחוֹת שֶׁל אַרְיֵה. כְּשֶׁהִגִּיעַ לְבֵיתוֹ, בְּנֵי מִשְׁפַּחְתּוֹ קָפְצוּ מֵאֹשֶׁר. כַּעֲבֹר שָׁעוֹת בּוֹדְדוֹת, כְּבָר בִּעְבְּעוּ עַל הָאֵשׁ סִירִים מְלֵאִים בְּמָרָק גְּרִיסִים מְהֻבָּל וְטָעִים מִנְּשֹׂא. זֶה הָיָה טַעַם שֶׁל שַׁבָּת, טַעַם שֶׁל כָּבוֹד.

בְּאוֹתָהּ שָׁעָה, בְּאַרְמוֹנוֹ, יָשַׁב רַבִּי יְהוּדָה הַנָּשִׂיא. שְׁנֵי הַמְשָׁרְתִים נִכְנְסוּ וְקָדּוּ קִידָה: "רַבֵּנוּ, רַבִּי שִׁמְעוֹן בֶּן חֲלַפְתָּא מָצָא אֶת הַשַּׂק וְלָקַח אוֹתוֹ בְּשִׂמְחָה עֲצוּמָה וּבְכָבוֹד גָּדוֹל".

חִיּוּךְ טָהוֹר וְרָחָב, חִיּוּךְ שֶׁמֵּאִיר אֶת כָּל תּוֹרַת יִשְׂרָאֵל, הֵאִיר עַל פָּנָיו שֶׁל רַבִּי. הוּא הִצְלִיחַ לַעֲזֹר לֶעָנִי, וְהַדָּבָר הַחָשׁוּב בְּיוֹתֵר – הוּא שָׁמַר לוֹ עַל הַכָּבוֹד.

א. שְׁאֵלַת פְּתִיחָה:

"יְלָדִים, שְׁאֵלָה קָשָׁה לִי אֲלֵיכֶם: דַּמְיְנוּ שֶׁיֵּשׁ לָכֶם חָבֵר שֶׁאֵין לוֹ אֹכֶל כְּלָל, אֲבָל... הוּא לֹא מַסְכִּים בְּשׁוּם אֹפֶן לְקַבֵּל מַתָּנוֹת, כִּי הוּא מִתְבַּיֵּשׁ וְרוֹצֶה רַק מִמָּה שֶׁה' נוֹתֵן לוֹ. אֵיךְ נָבִיא לוֹ עֶזְרָה בְּלִי שֶׁהוּא יִפָּגַע?

הַיּוֹם נִשְׁמַע עַל הַמִּבְצָע הַסּוֹדִי וְהַגְּאוֹנִי שֶׁל הַנָּשִׂיא הֲכִי גָּדוֹל בְּעַם יִשְׂרָאֵל, רַבִּי יְהוּדָה הַנָּשִׂיא!"

ב. הֶעָרוֹת בִּצּוּעַ וּסְטוֹרִיטֶלִינְג (קול ותנועה):

ג. הַמְלָצוֹת לְמַצֶּגֶת (4 שְׁקוּפִיּוֹת מוּבְנוֹת):

שקופית 1: חציית המסך לשניים. בצד אחד שולחן עץ מהודר מלא מטעמים בסגנון קדום (לחם, בשר), בצד שני שולחן עץ רעוע ויבש בבית עשוי אבן. (ללא דמויות, להעברת אווירת העושר מול העוני).

שקופית 2: שני משרתים יהודיים עם כיפות ושרוולים ארוכים, בגדי ארץ ישראל הקדומה, סוחבים על גבם יחד שק גדול של בד המלא ב"גריסים", על רקע רחוב אבנים ירושלמי עתיק.

שקופית 3: שדה פתוח ומואר בשמש. שק הגריסים מונח במרכז הדרך, ושני המשרתים נראים בורחים בריצה אל מאחורי הגבעה.

שקופית 4: יהודי מבוגר עטוף טלית/מעיל צנוע, כיפה נראית לעין, זקן לבן, עומד ליד השק העצום ומביט לשמיים ביראת כבוד והכרת תודה, ידיו מונפות קלות באוויר. קרן אור עדינה נופלת עליו מהשמיים (ללא ציור של ה' אלא רק אור רוחני).

ד. פְּעִילוּת מְסַכֶּמֶת - "הַמַּתָּנָה הַסּוֹדִית":

נשחק משחק: כל תלמיד בוחר (בלי לגלות!) תלמיד אחר בכיתה. עליו לחשוב על דרך "להפקיד" מילה טובה או פתק עידוד אצל אותו חבר (למשל: להגניב לקלמר, להשאיר מתחת למחברת, לשתול לו בתא) מבלי שהחבר יידע ממי זה הגיע. מי ש"מוצא" את הפתק מרגיש שמחה גדולה מהשמיים! זה ממחיש איך נתינה אמיתית היא כזו שהנותן מוותר על ה"כבוד" שלו בשביל השמחה של המקבל.

הסיפור מבוסס בדייקנות על המקורות בחז"ל, מתוך הקפדה שלא לשנות מרוח הדברים:

(ראה מקורות מספר 39 ומספר 91 במסמכים שנמסרו: "רבי יהודה הנשיא - הפקיר גריסים בשביל שר' שמעון בן חלפתא יזכה בהם, האגדה שי, רות רבה ה', עושרו ודאגתו לתלמידו").

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “להלן מערך חינוכי וסיפור מעובד, מרגש ומרתק, על רבי יהודה הנשיא (רבנו הקדוש) ורבי שמעון בן חלפתא .” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “הַגּוּף דָּרַשׁ אֶת שֶׁלּוֹ, וְהַיְלָדִים בַּבַּיִת הָיוּ רְעֵבִים.”?
ג. כיצד הגיב רבי יהודה הנשיא (רבי) למצב המתואר: “שִׂימוּ לֵב הֵיטֵב אֶל הַמְּשִׂימָה הַזּוֹ," לָחַשׁ רַבִּי וּפָנָיו רְצִינִיּוֹת.”?
ד. מה השתנה בין “הוּא נָשָׂא עֵינָיו לַשָּׁמַיִם וְהִתְפַּלֵּל בְּלַחַשׁ: "רִבּוֹנוֹ שֶׁל עוֹלָם, פּוֹתֵחַ אֶת יָדֶךָ וּמַשְׂבִּיעַ…” לבין “לֹא מִתּוֹךְ בּוּשָׁה אוֹ מַתְּנַת בָּשָׂר וָדָם, אֶלָּא מִתּוֹךְ הֶפְקֵר, מִיָּדוֹ הַפְּתוּחָה שֶׁל בּוֹרֵא עוֹלָם!”?
ה. כיצד הגיע הסיפור אל האירוע: “שק הגריסים מונח במרכז הדרך, ושני המשרתים נראים בורחים בריצה אל מאחורי הגבעה.”?

לפני ואחרי

מה קרה לפני: “יָכְלוּ לְהַרְגִּישׁ בְּיָדָם אֶת הַבַּד הַמְחֻסְפָּס שֶׁל הַשַּׂק, וְלִשְׁמֹעַ אֶת הַגְּרִיסִים הָרַבִּים מִתְחַכְּכִים זֶה בָּזֶה כְּלַחַשׁ שֶׁל נֶחָמָה.”?
מה קרה אחרי: “הוּא הֶעֱמִיס אֶת הַשַּׂק מְלֵא הַגְּרִיסִים עַל כְּתֵפוֹ הַחַלָּשָׁה, אֲבָל לְפֶתַע, הַשַּׂק כְּלָל לֹא הִרְגִּישׁ כָּבֵד.”?

הוּא הִתִּיר אֶת הַחֶבֶל הֶעָבֶה שֶׁקָּשַׁר אֶת פִּי הַשַּׂק.

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “הפקיר גריסים בשביל שר' שמעון בן חלפתא יזכה בהם” על רבי יהודה הנשיא (רבי)? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “אֲנִי יוֹדֵעַ שֶׁבְּעוֹד שָׁעָה קַלָּה, יַעֲבֹר רַבִּי שִׁמְעוֹן בֶּן חֲלַפְתָּא בַּשְּׁבִיל שֶׁמִּחוּץ לָעִיר, שְׁבִיל חָרֵב שֶׁאֵין בּוֹ אֲנָשִׁים.”.
ג. אילו רבי יהודה הנשיא (רבי) לא היה פועל כפי שמתואר ב־“לֹא מִתּוֹךְ בּוּשָׁה אוֹ מַתְּנַת בָּשָׂר וָדָם, אֶלָּא מִתּוֹךְ הֶפְקֵר, מִיָּדוֹ הַפְּתוּחָה שֶׁל בּוֹרֵא עוֹלָם!” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “שק הגריסים מונח במרכז הדרך, ושני המשרתים נראים בורחים בריצה אל מאחורי הגבעה.” מחזק את המסר של הסיפור?

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “שִׂימוּ לֵב הֵיטֵב אֶל הַמְּשִׂימָה הַזּוֹ," לָחַשׁ רַבִּי וּפָנָיו רְצִינִיּוֹת.”.
מכתב: כתבו לרבי יהודה הנשיא (רבי) בעקבות “הוּא הִתִּיר אֶת הַחֶבֶל הֶעָבֶה שֶׁקָּשַׁר אֶת פִּי הַשַּׂק.”.
