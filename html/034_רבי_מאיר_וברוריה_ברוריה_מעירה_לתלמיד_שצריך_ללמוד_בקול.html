ברוריה מעירה לתלמיד שצריך ללמוד בקול

דף עבודה בעקבות רבי מאיר וברוריה
שם: כיתה:

המעשה — הסיפור המלא

פְּתִיחָה: רֵיחַ שֶׁל בֹּקֶר וְקוֹל שֶׁל תּוֹרָה

יְלָדִים יְקָרִים, הַאִם עֲמַדְתֶּם פַּעַם לְיַד בֵּית כְּנֶסֶת גָּדוֹל אוֹ יְשִׁיבָה בִּשְׁעַת בֹּקֶר מֻקְדֶּמֶת? אִם תַּעַצְמוּ אֶת הָעֵינַיִם, תּוּכְלוּ לְדַמְיֵן אֶת זֶה: רֵיחַ טָרִי שֶׁל טַל הַבֹּקֶר עוֹלֶה מִן הָאֲדָמָה, קַרְנֵי שֶׁמֶשׁ זְהֻבּוֹת מְאִירוֹת אֶת הַחַלּוֹנוֹת, וּמִבִּפְנִים – בּוֹקֵעַ קוֹל אַדִּיר! קוֹל מָתוֹק מִדְּבַשׁ! מֵאוֹת אֲנָשִׁים וִילָדִים לוֹמְדִים יַחַד אַלֶ"ף בֵּי"ת, חוּמָשׁ, מִשְׁנָה וּגְמָרָא. הַקּוֹלוֹת שֶׁלָּהֶם מִתְחַבְּרִים יַחַד לְשִׁירָה גְּדוֹלָה שֶׁעוֹלָה יָשָׁר לְכִסֵּא הַכָּבוֹד בַּשָּׁמַיִם.

כָּךְ בְּדִיּוּק הָיָה נִשְׁמָע בֵּית הַמִּדְרָשׁ שֶׁל הַתַּנָּא הַקָּדוֹשׁ, רַבִּי מֵאִיר בַּעַל הַנֵּס , שֶׁחַי בְּאֶרֶץ יִשְׂרָאֵל לִפְנֵי כְּאַלְפַּיִם שָׁנָה [1]. רַבִּי מֵאִיר הָיָה אוֹר גָּדוֹל, כָּל חַכְמֵי דּוֹרוֹ הָיוּ בָּאִים לִשְׁמֹעַ אֶת שִׁעוּרָיו, וּכְשֶׁהוּא הָיָה דּוֹרֵשׁ, הַתּוֹרָה הָיְתָה מְאִירָה כְּמוֹ בְּהַר סִינַי.

אִשְׁתּוֹ שֶׁל רַבִּי מֵאִיר הָיְתָה אִשָּׁה חֲכָמָה וּקְדוֹשָׁה מְאֹד. קָרְאוּ לָהּ בְּרוּרְיָה . בְּרוּרְיָה הָיְתָה בִּתּוֹ שֶׁל רַבִּי חֲנִינָא בֶּן תְּרַדְיוֹן, שֶׁמָּסַר אֶת נַפְשׁוֹ עַל סֵפֶר הַתּוֹרָה. הִיא לָמְדָה תּוֹרָה מֵאָבִיהָ וּמִבַּעֲלָהּ, וְהָיְתָה יוֹדַעַת הֲלָכוֹת רַבּוֹת וְחָכְמָה עֲצוּמָה [1, 2]. כֻּלָּם כִּבְּדוּ אוֹתָהּ וְהִקְשִׁיבוּ לַעֲצוֹתֶיהָ.

הַשֶּׁקֶט שֶׁבַּפִּנָּה וְהַתִּסְכּוּל שֶׁל הַתַּלְמִיד

יוֹם אֶחָד, בְּאַחַד הַחֲדָרִים שֶׁסָּמוּךְ לְבֵית הַמִּדְרָשׁ, יָשַׁב תַּלְמִיד בָּחוּר מִישִׁיבָתוֹ שֶׁל רַבִּי מֵאִיר. נִקְרָא לוֹ אֶלְעָזָר. אֶלְעָזָר הָיָה בָּחוּר רְצִינִי שֶׁרָצָה בְּכָל לִבּוֹ לִהְיוֹת תַּלְמִיד חָכָם.

הוּא יָשַׁב מוּל סֵפֶר פָּתוּחַ מִקְּלַף יָשָׁן. רֵיחַ הַדְּיוֹ וְהַקְּלַף הֶעָתִיק מִלֵּא אֶת הָאֲוִיר. הַתַּלְמִיד שָׂם אֶת שְׁתֵּי יָדָיו עַל רֹאשׁוֹ, קֵרֵב אֶת מַבָּטוֹ אֶל הָאוֹתִיּוֹת הַשְּׁחֹרוֹת, וְקָרָא.

אֲבָל יְלָדִים, מַשֶּׁהוּ הָיָה מוּזָר מְאֹד. אִם הֱיִיתֶם עוֹמְדִים לְיָדוֹ, לֹא הֱיִיתֶם שׁוֹמְעִים כְּלוּם. מַמָּשׁ כְּלוּם! שֶׁקֶט מֻחְלָט.

אֶלְעָזָר לָמַד אֶת הַתּוֹרָה בְּשֶׁקֶט. רַק הָעֵינַיִם שֶׁלּוֹ זָזוּ מִיָּמִין לִשְׂמֹאל. מִדֵּי פַּעַם שְׂפָתָיו זָזוּ וְהוּא לָחַשׁ בְּקוֹל חַלָּשׁ-חַלָּשׁ, כְּמוֹ רִשְרוּשׁ שֶׁל רוּחַ בַּעֲלֵי הָעֵצִים: "פְּשִׁיטָא... מַאי קָא מַשְׁמַע לָן... הוֹאִיל וְכָךְ..." .

מַדּוּעַ הוּא לָמַד בְּשֶׁקֶט? אוּלַי מִפְּנֵי שֶׁהִתְבַּיֵּשׁ? אוּלַי מִפְּנֵי שֶׁחָשַׁב שֶׁכָּךְ לֹא יַפְרִיעַ לְאִישׁ? אוֹ אוּלַי חָשַׁב שֶׁהַתּוֹרָה הִיא כְּמוֹ עִתּוֹן אוֹ סִפּוּר מֶתַח שֶׁקּוֹרְאִים בְּשֶׁקֶט לִפְנֵי הַשֵּׁנָה?

אֲבָל יְלָדִים, אֶלְעָזָר הָיָה עָצוּב. הָיָה לוֹ קָשֶׁה!

הוּא קָרָא אֶת הַדַּף הָרִאשׁוֹן, הֵבִין הַכֹּל, וְעָבַר לַדַּף הַשֵּׁנִי. כְּשֶׁהִגִּיעַ לַדַּף הַשְּׁלִישִׁי... אוֹי לֹא! הוּא כְּבָר שָׁכַח מָה לָמַד בַּדַּף הָרִאשׁוֹן!

הוּא חָזַר לְהַתְחָלָה, עָצַם עֵינַיִם מִתּוֹךְ תִּסְכּוּל, מָעַךְ קְצָת אֶת פָּנָיו בְּיָדָיו, וְנִסָּה לְהִזָּכֵר. "מָה אָמַר הַתַּנָּא מִקֹּדֶם? אוּף, שׁוּב שָׁכַחְתִּי!".

בְּנוֹסָף לְכָךְ, הַשֶּׁקֶט הַזֶּה הִרְדִּים אוֹתוֹ. הָעַפְעַפַּיִם שֶׁלּוֹ נִהְיוּ כְּבֵדִים עַל הָעֵינַיִם. הוּא פִּהֵק פִּהוּק גָּדוֹל, קוֹל שֶׁל יֵאוּשׁ יָצָא מִפִּיו, וְהוּא כִּמְעַט סָגַר אֶת הַסֵּפֶר וְהָלַךְ.

הַהִתְעָרְבוּת שֶׁל בְּרוּרְיָה

בְּאוֹתָהּ שָׁעָה, עָבְרָה שָׁם בְּרוּרְיָה אֵשֶׁת רַבִּי מֵאִיר [1, 2]. הִיא הִבִּיטָה מִן הַפֶּתַח וְרָאֲתָה אֶת הַבָּחוּר. אִשָּׁה חֲכָמָה כָּמוֹהָ כְּבָר רָאֲתָה אֲלָפִים שֶׁל תַּלְמִידִים. הִיא קָלְטָה מִיָּד מָה הַבְּעָיָה שֶׁלּוֹ.

הִיא רָאֲתָה בָּחוּר שֶׁאוֹהֵב אֶת הַתּוֹרָה, שֶׁמִּתְאַמֵּץ לִלְמֹד, אֲבָל עוֹשֶׂה טָעוּת אַחַת גְּדוֹלָה: הוּא מַשְׁאִיר אֶת הַתּוֹרָה כְּלוּאָה בְּתוֹךְ הַשֶּׁקֶט, וְלֹא מוֹצִיא אוֹתָהּ לָאֲוִיר! הִיא חָשָׁה כְּאֵב בְּלִבָּהּ עַל הַצַּעַר שֶׁל הַתַּלְמִיד, וְיָדְעָה שֶׁהִיא חַיֶּבֶת לַעֲזֹר לוֹ.

אִלּוּ הָיְתָה נִגֶּשֶׁת וְאוֹמֶרֶת סְתָם: "לֹא כָּכָה לוֹמְדִים", אוּלַי לֹא הָיָה מֵבִין בֶּאֱמֶת אֶת גֹּדֶל הַטָּעוּת. בְּרוּרְיָה נִגְּשָׁה אֵלָיו בִּצְעָדִים מְדוּדִים וְעָצְרָה בַּפֶּתַח. כָּךְ מְסַפֶּרֶת לָנוּ הַגְּמָרָא (בְּמַסֶּכֶת עֵירוּבִין).

הִיא בָּטְשָׁה בּוֹ, כְּלוֹמַר - נָתְנָה דְּפִיקָה קַלָּה וְאַסֶרְטִיבִית סָמוּךְ לִמְקוֹמוֹ, כְּדֵי שֶׁיִּתְעוֹרֵר מֵהַתַּרְדֵּמָה שֶׁנָּפְלָה עָלָיו וְכִי לֹא מְכֻבָּד שֶׁתַּלְמִיד חָכָם יִהְיֶה מְיֹאָשׁ וְיָשֵׁן מוּל סֵפֶר פָּתוּחַ!

הַתַּלְמִיד קָפַץ קְצָת בְּבֶהָלָה, הֵרִים אֶת הָרֹאשׁ וְהִבִּיט לְעֶבְרָהּ.

"לָמָּה אַתָּה גּוֹרֵס בִּלְחִישָׁה?" שָׁאֲלָה בְּרוּרְיָה בְּקוֹל עָמֹק וּמַלְכוּתִי [1, 2]. "הַאִם כָּכָה הַתּוֹרָה מְצַוָּה עָלֵינוּ לִלְמֹד?! שָׁכַחְתָּ אֶת מָה שֶׁכָּתוּב בַּפָּסוּק בְּסֵפֶר שְׁמוּאֵל עַל הַתּוֹרָה?".

הַתַּלְמִיד מִצְמֵץ בְּעֵינָיו בְּהִסּוּס. "אֵיזֶה פָּסוּק?"

בְּרוּרְיָה הִבִּיטָה בּוֹ וְאָמְרָה בְּחֹם אִמָּהִי שֶׁל אִשָּׁה גְּדוֹלָה שֶׁאוֹהֶבֶת תּוֹרָה: "וְכִי לֹא כָּךְ כָּתוּב (בשמואל ב, כג): 'עֲרוּכָה בַכֹּל וּשְׁמוּרָה' ? זֶה הַסּוֹד הַגָּדוֹל שֶׁל הַזִּכָּרוֹן, בְּנִי! כְּדֵי שֶׁהַתּוֹרָה תִּהְיֶה 'שְׁמוּרָה' בְּתוֹךְ הַלֵּב וְהָרֹאשׁ שֶׁלְּךָ, הִיא חַיֶּבֶת לִהְיוֹת 'עֲרוּכָה בַּכֹּל'!"

"מָה זֶה אוֹמֵר עֲרוּכָה בַּכֹּל?" שָׁאַל הַתַּלְמִיד.

"הַתּוֹרָה אֵינָהּ רַק עִנְיָן לַעֲשָׂרָה אִבְרֵי הָרְאִיָּה וְהַמַּחֲשָׁבָה", הִסְבִּירָה לוֹ, "אִם הַתּוֹרָה עֲרוּכָה בְּכָל רְמַ"ח אֵיבָרֶיךָ (248 הָאֵיבָרִים שֶׁל הָאָדָם) – כְּלוֹמַר: אִם תִּפְתַּח אֶת שְׂפָתֶיךָ בְּקוֹל, אִם תָּזִיז אֶת יָדֶיךָ בַּהֲבָנָה, אִם תְּשַׁלֵּב אֶת הַשְּׁמִיעָה וְאֶת הַדִּבּוּר וְאֶת הַשִּׂמְחָה שֶׁל הַגּוּף כֻּלּוֹ בַּלִּמּוּד – אָז 'וּשְׁמוּרָה' – הַתּוֹרָה תִּשָּׁאֵר זְכוּרָה אֶצְלְךָ לָנֶצַח! אַךְ אִם לָאו, אִם תִּלְמַד רַק בָּעֵינַיִם וּבְהִרְהוּר הַלֵּב... אֵינָהּ מִשְׁתַּמֶּרֶת. הִיא תִּפְרַח כְּמוֹ צִפּוֹר הַחוּצָה וְתֵעָלֵם!".

שִׁירַת הַתּוֹרָה בּוֹקַעַת לְבַסּוֹף

יְלָדִים, הַמִּלִּים הַקְּדוֹשׁוֹת שֶׁל בְּרוּרְיָה חָדְרוּ כְּמוֹ חֵץ שֶׁל אוֹר יָשָׁר לְלִבּוֹ שֶׁל אֶלְעָזָר.

הוּא הֵבִין! הַתּוֹרָה הִיא מֵהַשָּׁמַיִם. אִי אֶפְשָׁר לִקְלֹט מַשֶּׁהוּ אֱלֹקִי כָּזֶה רַק בְּעֵינַיִם מְרַפְרְפוֹת. צָרִיךְ לַחְבֹּק אוֹתָהּ עִם כָּל הַגּוּף. צָרִיךְ שֶׁמֵּיתְרֵי הַקּוֹל יִרְעֲדוּ מִקְּדֻשָּׁתָהּ, שֶׁהָאָזְנַיִם יִשְׁמְעוּ אֶת הַמְּתִיקוּת.

הוּא זָקַף אֶת גַּבּוֹ, לָקַח נְשִׁימָה אֲרֻכָּה, וּפָתַח בְּנִגּוּן...

בַּהַתְחָלָה זֶה הָיָה קְצָת בְּבַיְשָׁנוּת: "אָמַר רַבִּי...". אֲבָל אַחֲרֵי חָמֵשׁ דַּקּוֹת הַקּוֹל שֶׁלּוֹ כְּבָר הִדְהֵד בַּחֶדֶר! הוא הֵזִיז אֶת הָאֲגוּדָל הַבּוֹחֶנֶת קָדִימָה וְאָחוֹרָה, הַקּוֹל שֶׁלּוֹ עָלָה וְיָרַד בְּשִׁירָה מְתוּקָה: "מַאי קָא מַשְׁמַע לָן הַתַּנָּא?!"

פִּתְאוֹם נֶעֶלְמָה הָעֲיֵפוּת! הָעֵינַיִם שֶׁלּוֹ נִצְּצוּ בְּאוֹר חָדָשׁ! הַשִּׂמְחָה מִלְּאָה אֶת הַחֶדֶר כֻּלּוֹ. אֶלְעָזָר הִמְשִׁיךְ לִקְרֹא דַּף אַחַר דַּף, וְהַפֶּלֶא קָרָה - הַמִּלִּים מַמָּשׁ נֶחְרְטוּ בְּתוֹךְ לִבּוֹ [1, 2]. הוּא זָכַר כָּל סְבָרָה וְכָל קֻשְׁיָה.

בִּזְכוּת מַה? בִּזְכוּת אוֹתָהּ הֶעָרָה חֲכָמָה שֶׁל בְּרוּרְיָה צִדְקָנִית, שֶׁלִּמְּדָה אֶת תַּלְמִידֵי יִשְׂרָאֵל, עַד לְיָמֵינוּ אָנוּ, שֶׁהַתּוֹרָה מַכְנִיסָה חַיִּים וְשִׂמְחָה - רַק מִי שֶׁמּוֹצִיא אוֹתָהּ מֵהַפֶּה שֶׁלּוֹ בְּקוֹל!

מקורות: תלמוד בבלי, מסכת עירובין דף נג עמוד ב – עד דף נד עמוד א (המעשה בברוריה שמצאה תלמיד שגורס בלחישה). מקור פדגוגי לעיבוד הערכי - מקורות מספרות [1] ו-[2]: סיפור חשיבות לימוד התורה בקול, והערתה של ברוריה לתלמיד על כך. (ספר האגדה רט').

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “אִם תַּעַצְמוּ אֶת הָעֵינַיִם, תּוּכְלוּ לְדַמְיֵן אֶת זֶה: רֵיחַ טָרִי שֶׁל טַל הַבֹּקֶר עוֹלֶה מִן הָאֲדָמָה, קַרְנֵי שֶׁמֶשׁ זְהֻבּוֹת מְאִירוֹת אֶת…” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “בְּרוּרְיָה הָיְתָה בִּתּוֹ שֶׁל רַבִּי חֲנִינָא בֶּן תְּרַדְיוֹן, שֶׁמָּסַר אֶת נַפְשׁוֹ עַל סֵפֶר הַתּוֹרָה.”?
ג. כיצד הגיב רבי מאיר וברוריה למצב המתואר: “אִם הֱיִיתֶם עוֹמְדִים לְיָדוֹ, לֹא הֱיִיתֶם שׁוֹמְעִים כְּלוּם.”?
ד. מה השתנה בין “אִשָּׁה חֲכָמָה כָּמוֹהָ כְּבָר רָאֲתָה אֲלָפִים שֶׁל תַּלְמִידִים.” לבין “כְּדֵי שֶׁהַתּוֹרָה תִּהְיֶה 'שְׁמוּרָה' בְּתוֹךְ הַלֵּב וְהָרֹאשׁ שֶׁלְּךָ, הִיא חַיֶּבֶת לִהְיוֹת 'עֲרוּכָה בַּכֹּל'!”?
ה. כיצד הגיע הסיפור אל האירוע: “הוּא זָקַף אֶת גַּבּוֹ, לָקַח נְשִׁימָה אֲרֻכָּה, וּפָתַח בְּנִגּוּן...”?

□ בְּאוֹתָהּ שָׁעָה, עָבְרָה שָׁם בְּרוּרְיָה אֵשֶׁת רַבִּי מֵאִיר [1, 2].
□ אִם תַּעַצְמוּ אֶת הָעֵינַיִם, תּוּכְלוּ לְדַמְיֵן אֶת זֶה: רֵיחַ טָרִי שֶׁל טַל הַבֹּקֶר עוֹלֶה מִן הָאֲדָמָה, קַרְנֵי שֶׁמֶשׁ זְהֻבּוֹת מְאִירוֹת אֶת…

לפני ואחרי

מה קרה לפני: “בְּאוֹתָהּ שָׁעָה, עָבְרָה שָׁם בְּרוּרְיָה אֵשֶׁת רַבִּי מֵאִיר [1, 2].”?
מה קרה אחרי: “כְּדֵי שֶׁהַתּוֹרָה תִּהְיֶה 'שְׁמוּרָה' בְּתוֹךְ הַלֵּב וְהָרֹאשׁ שֶׁלְּךָ, הִיא חַיֶּבֶת לִהְיוֹת 'עֲרוּכָה בַּכֹּל'!”?

סיבה ותוצאה מתוך המעשה

האירוע הנתוןמה גרם לו?מה הייתה תוצאתו?
יוֹם אֶחָד, בְּאַחַד הַחֲדָרִים שֶׁסָּמוּךְ לְבֵית הַמִּדְרָשׁ, יָשַׁב תַּלְמִיד בָּחוּר מִישִׁיבָתוֹ שֶׁל רַבִּי מֵאִיר.
אִלּוּ הָיְתָה נִגֶּשֶׁת וְאוֹמֶרֶת סְתָם: "לֹא כָּכָה לוֹמְדִים", אוּלַי לֹא הָיָה מֵבִין בֶּאֱמֶת אֶת גֹּדֶל הַטָּעוּת.

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “ברוריה מעירה לתלמיד שצריך ללמוד בקול” על רבי מאיר וברוריה? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “הוּא חָזַר לְהַתְחָלָה, עָצַם עֵינַיִם מִתּוֹךְ תִּסְכּוּל, מָעַךְ קְצָת אֶת פָּנָיו בְּיָדָיו, וְנִסָּה לְהִזָּכֵר.”.
ג. אילו רבי מאיר וברוריה לא היה פועל כפי שמתואר ב־“כְּדֵי שֶׁהַתּוֹרָה תִּהְיֶה 'שְׁמוּרָה' בְּתוֹךְ הַלֵּב וְהָרֹאשׁ שֶׁלְּךָ, הִיא חַיֶּבֶת לִהְיוֹת 'עֲרוּכָה בַּכֹּל'!” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “הוּא זָקַף אֶת גַּבּוֹ, לָקַח נְשִׁימָה אֲרֻכָּה, וּפָתַח בְּנִגּוּן...” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. אִם תַּעַצְמוּ אֶת הָעֵינַיִם, תּוּכְלוּ לְדַמְיֵן אֶת זֶה: רֵיחַ טָרִי…
2. בְּרוּרְיָה הָיְתָה בִּתּוֹ שֶׁל רַבִּי חֲנִינָא בֶּן תְּרַדְיוֹן,…
3. מִדֵּי פַּעַם שְׂפָתָיו זָזוּ וְהוּא לָחַשׁ בְּקוֹל חַלָּשׁ-חַלָּשׁ,…
4. הִיא חָשָׁה כְּאֵב בְּלִבָּהּ עַל הַצַּעַר שֶׁל הַתַּלְמִיד, וְיָדְעָה…
5. יְלָדִים, הַמִּלִּים הַקְּדוֹשׁוֹת שֶׁל בְּרוּרְיָה חָדְרוּ כְּמוֹ חֵץ…
6. בִּזְכוּת אוֹתָהּ הֶעָרָה חֲכָמָה שֶׁל בְּרוּרְיָה צִדְקָנִית,…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “אִם הֱיִיתֶם עוֹמְדִים לְיָדוֹ, לֹא הֱיִיתֶם שׁוֹמְעִים כְּלוּם.”.
מכתב: כתבו לרבי מאיר וברוריה בעקבות “אִלּוּ הָיְתָה נִגֶּשֶׁת וְאוֹמֶרֶת סְתָם: "לֹא כָּכָה לוֹמְדִים", אוּלַי לֹא הָיָה מֵבִין בֶּאֱמֶת אֶת גֹּדֶל הַטָּעוּת.”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “יְלָדִים, הַמִּלִּים הַקְּדוֹשׁוֹת שֶׁל בְּרוּרְיָה חָדְרוּ כְּמוֹ חֵץ שֶׁל אוֹר יָשָׁר לְלִבּוֹ שֶׁל אֶלְעָזָר.”.
דיווח: כתבו ידיעה קצרה שכותרתה “ברוריה מעירה לתלמיד שצריך ללמוד בקול”.
