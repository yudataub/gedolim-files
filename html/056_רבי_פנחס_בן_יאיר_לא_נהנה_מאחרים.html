לא נהנה מאחרים

דף עבודה בעקבות רבי פנחס בן יאיר
שם: כיתה:

המעשה — הסיפור המלא

פְּתִיחָה: הַכְּלָל שֶׁל מִשְׁפַּחַת בֶּן יָאִיר

יְלָדִים, נַסּוּ לְדַמְיֵן רֶגַע שֶׁבּוֹ חָבֵר שֶׁלָּכֶם פּוֹתֵחַ שַׂקִּית שֶׁל מַמְתַּקִּים טְעִימִים, הָרֵיחַ הַמָּתוֹק מִתְפַּשֵּׁט בַּכִּתָּה, וְכֻלָּם מְבַקְשִׁים: "תָּבִיא לִי קְצָת! תְּכַבֵּד אוֹתִי!". קָשֶׁה מְאֹד לֹא לְבַקֵּשׁ, נָכוֹן? קָשֶׁה מְאֹד לְהַגִּיד: "אֲנִי שָׂמֵחַ בְּמָה שֶׁאִמָּא הֵבִיאָה לִי, וְאֲנִי לֹא מְבַקֵּשׁ כְּלוּם מֵאֲחֵרִים".

אֲבָל בְּאֶרֶץ יִשְׂרָאֵל, לִפְנֵי שָׁנִים רַבּוֹת, חַי תַּנָּא קָדוֹשׁ שֶׁקָּרְאוּ לוֹ רַבִּי פִּנְחָס בֶּן יָאִיר . לְרַבִּי פִּנְחָס הָיָה כְּלָל בַּרְזֶל שֶׁעָלָיו הוּא לֹא וִתֵּר מֵעוֹלָם. מֵהָרֶגַע שֶׁבּוֹ הוּא הָפַךְ לְאָדָם מְבֻגָּר, הוּא מֵעוֹלָם – אֲבָל מֵעוֹלָם! – לֹא בָּצַע אֲפִלּוּ פְּרוּסַת לֶחֶם אַחַת שֶׁל מִישֶׁהוּ אַחֵר! הוּא לֹא נֶהֱנָה מִשּׁוּם דָּבָר שֶׁלֹּא הָיָה שֶׁלּוֹ. הוּא הֶאֱמִין בֶּאֱמוּנָה שְׁלֵמָה שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא בְּעַצְמוֹ מְפַרְנֵס אוֹתוֹ, וְאָדָם שֶׁבּוֹטֵחַ בַּה' לֹא צָרִיךְ לְחַפֵּשׂ מַתָּנוֹת מִבְּנֵי אָדָם.

הַבְּשׂוֹרָה הַקָּשָׁה וְהַיְצִיאָה לַדֶּרֶךְ

יוֹם אֶחָד, הַשֶּׁקֶט בְּבֵית הַמִּדְרָשׁ הֻפְרַע. שָׁלִיחַ מְבֹהָל, מָלֵא בְּזֵעָה וְאָבָק מֵהַדֶּרֶךְ, נִכְנַס פְּנִימָה. "רַבִּי פִּנְחָס! רַבִּי פִּנְחָס!" הוּא צָעַק, וְקוֹלוֹ רָעַד. "יְהוּדִים יְקָרִים נִלְקְחוּ בַּשֶּׁבִי! הַשּׁוֹדְדִים כָּלְאוּ אוֹתָם בְּתוֹךְ בּוֹר חָשׁוּךְ, קַר וְטָחוּב. הֵם קְשׁוּרִים בְּשַׁרְשְׁרָאוֹת כְּבֵדוֹת שֶׁל בַּרְזֶל. הַשּׁוֹדְדִים דּוֹרְשִׁים כֶּסֶף רַב מְאֹד – וְאִם לֹא נָבִיא אוֹתוֹ, הֵם יַהַרְגוּ אוֹתָם חָלִילָה!".

רַבִּי פִּנְחָס נֶעֱמַד מִיָּד. צָרַת עַם יִשְׂרָאֵל הָיְתָה תָּמִיד בְּרֹאשׁ מַעֲיָנָיו. הוּא לָקַח עִמּוֹ שַׂק עֲנָק מָלֵא בְּמַטְבְּעוֹת תְּרוּמָה לְפִדְיוֹן שְׁבוּיִים – צְלִיל שֶׁל 'קְלִינְג, קְלִינְג' נִשְׁמַע מִן הַשַּׂק הַכָּבֵד – וְיָצָא בִּמְהִירוּת לַדֶּרֶךְ, לְהַצִּיל אֶת הַשְּׁבוּיִים. הוּא לֹא חָשַׁב עַל מְנוּחָה, וְלֹא עָצַר לֶאֱכֹל אוֹ לִשְׁתּוֹת. הַזְּמַן דּוֹחֵק!

עוֹמְדִים מוּל הַגַּלִּים

הַדֶּרֶךְ הָיְתָה אֲרֻכָּה וְקָשָׁה. יַחַד עִם רַבִּי פִּנְחָס הָלְכוּ עוֹד שְׁנֵי מְלַוִּים: יְהוּדִי שֶׁסָּחַב שַׂקֵּי חִטָּה שֶׁנִּשְׁמְרוּ לִכְבוֹד חַג הַפֶּסַח (כְּדֵי שֶׁלֹּא יֵרְדוּ עֲלֵיהֶם מַיִם וְיַחְמִיצוּ), וְסוֹחֵר יִשְׁמְעֵאלִי (עַרְבִי) שֶׁרָכַב עַל גָּמָל.

לְפֶתַע, קוֹל שְׁאָגָה אַדִּירָה נִשְׁמַע מֵרָחוֹק. שששש-בוּם! שששש-בוּם! רֵיחַ חָזָק שֶׁל בּוֹץ וּמַיִם הִכָּה בְּאַפָּם.

הֵם הִגִּיעוּ אֶל שְׂפַת הַנָּהָר שֶׁנִּקְרָא "גִּינַאי". זֶה לֹא הָיָה נָהָר שָׁקֵט וְרָגוּעַ. הַמַּיִם שָׁם קָפְצוּ, עָלוּ וְיָרְדוּ בְּגַלִּים זְעוּמִים מְלֵאִים בְּקֶצֶף לָבָן!

"אוֹי וַאֲבוֹי!" זָעַק הַיְּהוּדִי עִם שַׂקֵּי הַחִטָּה, "אֵין כָּאן גֶּשֶׁר! אִם נִכָּנֵס, הַמַּיִם יַטְבִּיעוּ אוֹתָנוּ, אוֹ שֶׁהַחִטָּה שֶׁל פֶּסַח תֵּהָרֵס לְגַמְרֵי! אֵיךְ נַעֲבֹר?! הַשְּׁבוּיִים יָמוּתוּ בַּבּוֹר!".

רַבִּי פִּנְחָס בֶּן יָאִיר נֶעֱמַד מַמָּשׁ עַל שְׂפַת הַמַּיִם הַקָּפְאִים. זְקָנוֹ הַלָּבָן הִתְבַּדֵּר בָּרוּחַ הַחֲזָקָה. הוּא לֹא בָּכָה וְלֹא נִבְהַל. הוּא הֵרִים אֶת מַטֵּהוּ, הִבִּיט יָשָׁר אֶל תּוֹךְ הַגַּלִּים הַזּוֹעֲמִים, וְקָרָא בְּקוֹל שֶׁרָעַד אֶת הָאֲוִיר:

"גִּינַאי, גִּינַאי! חֲלֹק מֵימֶיךָ, וְאֶעֱבֹר בְּךָ!" .

יְלָדִים, נָהָר יוֹדֵעַ לְדַבֵּר? וַדַּאי שֶׁלֹּא! אֲבָל מִפְּנֵי קְדֻשָּׁתוֹ שֶׁל הַצַּדִּיק הַזֶּה שֶׁשָּׁמַר עַל עַצְמוֹ בְּכָל רֶגַע וָרֶגַע, הַקָּדוֹשׁ בָּרוּךְ הוּא נָתַן בַּנָּהָר כֹּחַ לַעֲנוֹת.

מִתּוֹךְ רַעַשׁ הַמַּיִם נִשְׁמַע קוֹל: "אַתָּה הָולֵךְ לַעֲשׂוֹת אֶת רְצוֹן בּוֹרַאֲךָ לִפְדּוֹת שְׁבוּיִים – וְגַם אֲנִי הוֹלֵךְ לַעֲשׂוֹת אֶת רְצוֹן בּוֹרְאִי שֶׁצִּוָּה עָלַי לִזְרֹם לַיָּם. אֲבָל אוּלַי אַתָּה לֹא תַּצְלִיחַ בַּמְּשִׂימָה שֶׁלְּךָ? אֲנִי בְּוַדַּאי מַצְלִיחַ! לָמָּה שֶׁאֲפַנֶּה לְךָ מָקוֹם?".

הַסּוֹחֵר הָעַרְבִי רָעַד מִפַּחַד כְּשֶׁשָּׁמַע אֶת זֶה. אֲבָל רַבִּי פִּנְחָס יָדַע שֶׁהַתּוֹרָה גְּדוֹלָה מֵהַטֶּבַע! כְּלֵי מִלְחָמָה לֹא מַפְחִידִים אוֹתוֹ, וְגַם לֹא גַּלִּים.

"אִם אֵינְךָ חוֹלֵק אֶת מֵימֶיךָ לִפְנֵי מִי שֶׁהוֹלֵךְ לְהַצִּיל עַם יִשְׂרָאֵל," זָעַק רַבִּי פִּנְחָס כְּשֶׁעֵינָיו בּוֹרְקוֹת כְּאֵשׁ, "אֲנִי אֶגְזֹר עָלֶיךָ שֶׁלְּעוֹלָם, לְעוֹלָם מַיִם לֹא יַעַבְרוּ בְּךָ שׁוּב! אַתָּה תִּתְיַבֵּשׁ כְּמוֹ מִדְבָּר!".

בְּאוֹתוֹ רֶגַע... סְוִוישׁ! הַמַּיִם מִיָּמִין קָפְאוּ וְעָמְדוּ כְּמוֹ חוֹמַת בֶּטוֹן! הַמַּיִם מִשְּׂמֹאל נָסוּגּוּ אָחוֹרָה! בְּאֶמְצַע הַנָּהָר הַגּוֹעֵשׁ נוֹצַר שְׁבִיל יָבֵשׁ. רַבִּי פִּנְחָס פָּסַע עַל הָאֲדָמָה וְעָבַר.

"רַבִּי!" קָרָא הַיְּהוּדִי עִם מַצּוֹת הַפֶּסַח, "מָה אִתִּי? קְרַע אֶת הַנָּהָר גַּם עֲבוּרִי שֶׁלֹּא יֵהָרֵס לִי חַג הַפֶּסַח!". וְרַבִּי פִּנְחָס צִוָּה עַל הַנָּהָר לְהִקָּרַע שׁוּב, וְהַיְּהוּדִי עָבַר.

"וּמָה אִתִּי?" זָעַק הַסּוֹחֵר הַיִּשְׁמְעֵאלִי, "אֲנִי מְלַוֶּה אֶתְכֶם!". אָמַר רַבִּי פִּנְחָס לַנָּהָר: "אִם תַּשְׁאִיר אוֹתוֹ בַּחוּץ, יַגִּידוּ שֶׁכָּךְ יְהוּדִים מִתְנַהֲגִים לְאוֹרְחִים מִגּוֹיִים שֶׁהוֹלְכִים אִתָּם!". וְהַנָּהָר נִבְקַע בַּפַּעַם הַשְּׁלִישִׁית! יְלָדִים, שָׁלֹשׁ פְּעָמִים נִקְרַע הַנָּהָר עֲבוּר הַצַּדִּיק שֶׁשָּׁמַר עַל כְּבוֹדוֹ שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא!

הַפְּגִישָׁה עִם הַנָּשִׂיא

רַבִּי פִּנְחָס בֶּן יָאִיר הִמְשִׁיךְ בַּדֶּרֶךְ, עַד שֶׁהִגִּיעַ לָעִיר הַגְּדוֹלָה שֶׁבָּהּ גָּר גְּדוֹל הַדּוֹר וּנְשִׂיא יִשְׂרָאֵל, רַבִּי יְהוּדָה הַנָּשִׂיא (רַבֵּנוּ הַקָּדוֹשׁ). כְּשֶׁשָּׁמַע רַבִּי שֶׁרַבִּי פִּנְחָס בָּעִיר, הוּא יָצָא לִקְרָאתוֹ בְּכָבוֹד גָּדוֹל שֶׁל נְשִׂיאִים.

"רַבִּי פִּנְחָס הַצַּדִּיק!" קָרָא הַנָּשִׂיא, וְחִיּוּךְ רָחָב עַל פָּנָיו, "שִׂמְחָה עֲצוּמָה נָפְלָה בְּחֶלְקִי. אֲנִי מַפְצִיר בְּךָ, בּוֹא לָסוּר לְבֵיתִי! סְעוֹד עִמִּי סְעוּדָה לִכְבוֹד מִצְוַת פִּדְיוֹן שְׁבוּיִים!".

רַבִּי פִּנְחָס עָצַר. לְהִזָּמֵן לִסְעוּדָה אֵצֶל הַנָּשִׂיא עַצְמוֹ! זוֹ זְכוּת גְּדוֹלָה שֶׁכָּל יְהוּדִי הָיָה רוֹצֶה. אֲבָל זִכְרוּ אֶת הַכְּלָל שֶׁל רַבִּי פִּנְחָס!

הוּא הִבִּיט בַּנָּשִׂיא בְּאַהֲבָה וְאָמַר לוֹ: "כְּבוֹד הַנָּשִׂיא, תֵּדַע לְךָ שֶׁיְּהוּדִים הֵם קְדוֹשִׁים. כָּל יְהוּדִי יִשְׂמַח לָתֵת אֹכֶל, אֲבָל יֵשׁ שְׁתֵּי סִבּוֹת שֶׁאֲנִי מֵעוֹלָם לֹא לוֹקֵחַ שׁוּם דָּבָר מֵאִישׁ: חֵלֶק מֵהָאֲנָשִׁים – רוֹצִים לָתֵת לִי לֶאֱכֹל, אֲבָל פָּשׁוּט 'אֵין לָהֶם מסְפִּיק'. וְאִם אֶקַּח, זֶה יֶחְסַר לַיְּלָדִים שֶׁלָּהֶם בַּבַּיִת! וְחֵלֶק אֲחֵר מֵהָאֲנָשִׁים – יֵשׁ לָהֶם שֶׁפַע גָּדוֹל בַּבַּיִת, אֲבָל 'הֵם אֵינָם נוֹתְנִים בְּלֵב שָׁלֵם', הֵם לֹא בֶּאֱמֶת רוֹצִים לְאָרֵחַ אוֹתִי, אֶלָּא מִתְבַּיְּשִׁים אוֹ רוֹצִים כָּבוֹד".

חִיֵּךְ רַבִּי יְהוּדָה הַנָּשִׂיא: "צָרִיךְ לַחֲשֹׁשׁ! אֲבָל אֶצְלִי? הֲרֵי הַקָּדוֹשׁ בָּרוּךְ הוּא בֵּרֵךְ אוֹתִי בְּעוֹשֶׁר שֶׁל מְלָכִים! יֵשׁ לִי שֶׁפַע תָּמִיד – וַאֲנִי גַּם רוֹצֶה בְּלֵב שָׁלֵם לְאָרֵחַ אוֹתְךָ! הִכָּנֵס, הַשֻּׁלְחָן עָרוּךְ!".

שָׁתַק רַבִּי פִּנְחָס. הוּא רָאָה שֶׁרַבִּי דּוֹבֵר אֱמֶת. "אִם כָּךְ," הוּא הִסְכִּים, "אֶכָּנֵס בְּדַרְכִּי חֲזָרָה אַחֲרֵי שֶׁאֶפְדֶּה אֶת הַשְּׁבוּיִים".

הַסִּיּוּם - שׁוֹנֵא מַתָּנוֹת יִחְיֶה

רַבִּי פִּנְחָס יָצָא, חִלֵּק אֶת הַמַּטְבְּעוֹת שֶׁקִּשְׁקְשׁוּ בַּשַּׂק שֶׁלּוֹ לַשּׁוֹדְדִים, פָּתַח אֶת דַּלְתוֹת הַבַּרְזֶל וְהוֹצִיא אֶת הַיְּהוּדִים הַבּוֹכִים חֲזָרָה לַחֹפֶשׁ! לִבּוֹ רָחַב מִשִּׂמְחָה.

כְּשֶׁחָזַר בַּדֶּרֶךְ וְעָבַר לְיַד הָאַרְמוֹן שֶׁל רַבִּי יְהוּדָה הַנָּשִׂיא, הוּא נִגַּשׁ לַפֶּתַח לְקַיֵּם אֶת הַבְטָחָתוֹ. אֲבָל כְּשֶׁעָמַד בַּכְּנִיסָה, נָפְלוּ עֵינָיו עַל חֲצַר הָאַרְמוֹן. רָעוּ שָׁם פְּרָדוֹת (חיות הדומות לסוסים וחמורים) לבָנוֹת וְעַזּוֹת. הַפְּרָדוֹת הָאֵלּוּ הָיוּ מְסֻכָּנוֹת מְאֹד – הֵן הָיוּ בּוֹעֲטוֹת וּפוֹגְעוֹת בַּאֲנָשִׁים.

"מַלְאַךְ הַמָּוֶת בְּבֵיתוֹ שֶׁל זֶה – וַאֲנִי אֶסְעַד אֶצְלוֹ מִסְּעוּדָתוֹ?!" אָמַר רַבִּי פִּנְחָס. (אדם צדיק שומר על עצמו מסכנות, ואינו נכנס למקום שיש בו סכנה לאחרים).

רַבִּי יְהוּדָה הַנָּשִׂיא שָׁמַע מִיָּד עַל כָּךְ. הוּא יָצָא בִּמְהִירוּת, עָצוּב שֶׁהַצַּדִּיק מְסָרֵב. "רַבִּי פִּנְחָס! אַתָּה צוֹדֵק! תֵּכֶף וּמִיָּד אֲנִי אוֹמֵר לַשּׁוֹמְרִים שֶׁלִּי שֶׁיִּמְכְּרוּ אֶת הַפְּרָדוֹת! אוֹ שֶׁיַּעֲשׂוּ לָהֶן פְּגָם בַּפַּרְסוֹת כְּדֵי שֶׁלֹּא יוּכְלוּ לִבְעֹט!".

אֲבָל רַבִּי פִּנְחָס עָמַד עַל עֶקְרוֹנוֹ. "אִם תִּמְכֹּר, הֵן יַזִּיקוּ לְמִישֶׁהוּ אַחֵר (לפני עיוור לא תיתן מכשול)! אִם תַּפְגִּים אוֹתָן (לעקור את פרסותיהן) תַּעֲבֹר עַל 'צַעַר בַּעֲלֵי חַיִּים'! שַׁחְרֵר אוֹתָן!".

כְּשֶׁהֵבִין רַבִּי פִּנְחָס שֶׁהַדָּבָר עָלוּל לְהָבִיא לְנֵזֶק אוֹ מַחֲלֹקֶת, הוּא אָמַר: "כַּנִּרְאֶה הַקָּדוֹשׁ בָּרוּךְ הוּא רָצָה לְכַוֵּן אוֹתִי שֶׁאֶשְׁמֹר עַל הַמִּנְהָג שֶׁלִּי כָּל יָמַי – שֶׁלֹּא אֵהָנֶה מֵאִישׁ עֲלֵי אֲדָמוֹת". הוּא בֵּרֵךְ אֶת הַנָּשִׂיא בְּאַהֲבָה רַבָּה, שָׂם אֶת כַּפּוֹת יָדָיו עַל מַטֵּהוּ, חִיֵּךְ וְהִמְשִׁיךְ לְעִירוֹ. אָדָם גָּדוֹל עֲנָק בַּתּוֹרָה, שֶׁשָּׂמַח בְּמָה שֶׁיֵּשׁ לוֹ, סָמַךְ רַק עַל מֶלֶךְ מַלְכֵי הַמְּלָכִים, וְקָרַע אֶת הַנְּהָרוֹת.

מקורות: תלמוד בבלי, מסכת חולין דף ז עמוד א - עמוד ב: "אמר ר' פנחס בן יאיר כל ימי לא בצעתי על פרוסה שאינה שלי... אמר לו לנהר גינאי: 'חלוֹק לי מימיך...'". תלמוד ירושלמי, מסכת דמאי פרק א, הלכה ג: תיאור הנהר שנקרע ומידת החסידות של רבי פנחס, וכן תיאור המפגש של האירוח אצל רבי יהודה הנשיא ("חמורו והשעורים והאירוח אצל ר' יהודה הנשיא"). ספר האגדה, רפא-רפג: סיכום של הסיפורים בדבר סרובו העקרוני של ר' פנחס ליהנות ממתנת בשר ודם, וההתמודדות עם הפקדת הממון וארוחת הנשיא.

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “יְלָדִים, נַסּוּ לְדַמְיֵן רֶגַע שֶׁבּוֹ חָבֵר שֶׁלָּכֶם פּוֹתֵחַ שַׂקִּית שֶׁל מַמְתַּקִּים טְעִימִים, הָרֵיחַ הַמָּתוֹק מִתְפַּשֵּׁט בַּכִּתָּה,…” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “הַשּׁוֹדְדִים כָּלְאוּ אוֹתָם בְּתוֹךְ בּוֹר חָשׁוּךְ, קַר וְטָחוּב.”?
ג. כיצד הגיב רבי פנחס בן יאיר למצב המתואר: “רַבִּי פִּנְחָס בֶּן יָאִיר נֶעֱמַד מַמָּשׁ עַל שְׂפַת הַמַּיִם הַקָּפְאִים.”?
ד. מה השתנה בין “יְלָדִים, שָׁלֹשׁ פְּעָמִים נִקְרַע הַנָּהָר עֲבוּר הַצַּדִּיק שֶׁשָּׁמַר עַל כְּבוֹדוֹ שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא!” לבין “אִם כָּךְ," הוּא הִסְכִּים, "אֶכָּנֵס בְּדַרְכִּי חֲזָרָה אַחֲרֵי שֶׁאֶפְדֶּה אֶת הַשְּׁבוּיִים".”?
ה. כיצד הגיע הסיפור אל האירוע: “אִם תִּמְכֹּר, הֵן יַזִּיקוּ לְמִישֶׁהוּ אַחֵר (לפני עיוור לא תיתן מכשול)!”?

לפני ואחרי

מה קרה לפני: “וְרַבִּי פִּנְחָס צִוָּה עַל הַנָּהָר לְהִקָּרַע שׁוּב, וְהַיְּהוּדִי עָבַר.”?
מה קרה אחרי: “רַבִּי פִּנְחָס יָצָא, חִלֵּק אֶת הַמַּטְבְּעוֹת שֶׁקִּשְׁקְשׁוּ בַּשַּׂק שֶׁלּוֹ לַשּׁוֹדְדִים, פָּתַח אֶת דַּלְתוֹת הַבַּרְזֶל וְהוֹצִיא אֶת…”?

סיבה ותוצאה מתוך המעשה

האירוע הנתוןמה גרם לו?מה הייתה תוצאתו?
יַחַד עִם רַבִּי פִּנְחָס הָלְכוּ עוֹד שְׁנֵי מְלַוִּים: יְהוּדִי שֶׁסָּחַב שַׂקֵּי חִטָּה שֶׁנִּשְׁמְרוּ לִכְבוֹד חַג הַפֶּסַח (כְּדֵי שֶׁלֹּא יֵרְדוּ…
כָּל יְהוּדִי יִשְׂמַח לָתֵת אֹכֶל, אֲבָל יֵשׁ שְׁתֵּי סִבּוֹת שֶׁאֲנִי מֵעוֹלָם לֹא לוֹקֵחַ שׁוּם דָּבָר מֵאִישׁ: חֵלֶק מֵהָאֲנָשִׁים – רוֹצִים לָתֵת לִי…

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “לא נהנה מאחרים” על רבי פנחס בן יאיר? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “אִם אֵינְךָ חוֹלֵק אֶת מֵימֶיךָ לִפְנֵי מִי שֶׁהוֹלֵךְ לְהַצִּיל עַם יִשְׂרָאֵל," זָעַק רַבִּי פִּנְחָס כְּשֶׁעֵינָיו בּוֹרְקוֹת כְּאֵשׁ, "אֲנִי אֶגְזֹר…”.
ג. אילו רבי פנחס בן יאיר לא היה פועל כפי שמתואר ב־“אִם כָּךְ," הוּא הִסְכִּים, "אֶכָּנֵס בְּדַרְכִּי חֲזָרָה אַחֲרֵי שֶׁאֶפְדֶּה אֶת הַשְּׁבוּיִים".” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “אִם תִּמְכֹּר, הֵן יַזִּיקוּ לְמִישֶׁהוּ אַחֵר (לפני עיוור לא תיתן מכשול)!” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. יְלָדִים, נַסּוּ לְדַמְיֵן רֶגַע שֶׁבּוֹ חָבֵר שֶׁלָּכֶם פּוֹתֵחַ…
2. הַשּׁוֹדְדִים דּוֹרְשִׁים כֶּסֶף רַב מְאֹד – וְאִם לֹא נָבִיא אוֹתוֹ,…
3. אֲבָל מִפְּנֵי קְדֻשָּׁתוֹ שֶׁל הַצַּדִּיק הַזֶּה שֶׁשָּׁמַר עַל…
4. רַבִּי פִּנְחָס הַצַּדִּיק!" קָרָא הַנָּשִׂיא, וְחִיּוּךְ רָחָב עַל…
5. אֲבָל כְּשֶׁעָמַד בַּכְּנִיסָה, נָפְלוּ עֵינָיו עַל חֲצַר הָאַרְמוֹן.
6. אָדָם גָּדוֹל עֲנָק בַּתּוֹרָה, שֶׁשָּׂמַח בְּמָה שֶׁיֵּשׁ לוֹ, סָמַךְ…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “רַבִּי פִּנְחָס בֶּן יָאִיר נֶעֱמַד מַמָּשׁ עַל שְׂפַת הַמַּיִם הַקָּפְאִים.”.
מכתב: כתבו לרבי פנחס בן יאיר בעקבות “כָּל יְהוּדִי יִשְׂמַח לָתֵת אֹכֶל, אֲבָל יֵשׁ שְׁתֵּי סִבּוֹת שֶׁאֲנִי מֵעוֹלָם לֹא לוֹקֵחַ שׁוּם דָּבָר מֵאִישׁ: חֵלֶק…”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “רָעוּ שָׁם פְּרָדוֹת (חיות הדומות לסוסים וחמורים) לבָנוֹת וְעַזּוֹת.”.
דיווח: כתבו ידיעה קצרה שכותרתה “לא נהנה מאחרים”.

המחזה

רגע מן הסיפורהדמויותמה יאמרו או יעשו?
יַחַד עִם רַבִּי פִּנְחָס הָלְכוּ עוֹד שְׁנֵי מְלַוִּים: יְהוּדִי שֶׁסָּחַב שַׂקֵּי חִטָּה…
אִם כָּךְ," הוּא הִסְכִּים, "אֶכָּנֵס בְּדַרְכִּי חֲזָרָה אַחֲרֵי שֶׁאֶפְדֶּה אֶת הַשְּׁבוּיִים".

7. חיבור לחיים מתוך המעשה

א. מה אפשר ללמוד מן הדרך שבה רבי פנחס בן יאיר פעל באירוע “יְלָדִים, שָׁלֹשׁ פְּעָמִים נִקְרַע הַנָּהָר עֲבוּר הַצַּדִּיק שֶׁשָּׁמַר עַל כְּבוֹדוֹ שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא!”?
ב. תארו מצב מחיי הכיתה שבו אפשר ליישם את המסר של “לא נהנה מאחרים”.
