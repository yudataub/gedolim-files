גדולתו — "ש"י עולמות לעולם הבא"

דף עבודה בעקבות ר' יהושע בן לוי
שם: כיתה:

המעשה — הסיפור המלא

פְּתִיחָה: רֵיחַ שֶׁל גְּוִילִים וְקוֹלוֹת שֶׁל מַלְאָכִים

יְלָדִים, אִם הֱיִיתֶם נִכְנָסִים לְבֵית הַמִּדְרָשׁ שֶׁל רַבִּי יְהוֹשֻׁעַ בֶּן לֵוִי בָּעִיר לוֹד, הֱיִיתֶם מַרְגִּישִׁים מִיָּד שֶׁאַתֶּם בְּמָקוֹם קָדוֹשׁ. בָּאֲוִיר הָיָה עוֹמֵד רֵיחַ מְיֻחָד – רֵיחַ עַתִּיק וְנָעִים שֶׁל גְּוִילֵי סִפְרֵי תּוֹרָה יְשָׁנִים וּקְלָפִים שֶׁל מִשְׁנָיוֹת. הָאָזְנַיִם הָיוּ מִתְמַלְּאוֹת בְּזִמְזוּם אֲדִיר, כְּמוֹ שֶׁל כַּוֶּרֶת דְּבוֹרִים חֲרוּצוֹת. מֵאוֹת תַּלְמִידֵי חֲכָמִים יָשְׁבוּ שָׁם, מִתְנַדְנְדִים קָדִימָה וְאָחוֹרָה, קוֹרְאִים בְּקוֹל נָעִים וְלָמְדוּ אֶת תּוֹרַת ה'.

וּבְרֹאשׁ הַיְשִׁיבָה יָשַׁב רַבִּי יְהוֹשֻׁעַ בֶּן לֵוִי (שֶׁאֲנַחְנוּ קוֹרְאִים לוֹ בְּקִצּוּר: רִיבָּ"ל). רַבִּי יְהוֹשֻׁעַ הָיָה צַדִּיק יְסוֹד עוֹלָם. פָּנָיו הֵאִירוּ כְּאוֹר הַשֶּׁמֶשׁ. חֲזַ"ל מְסַפְּרִים שֶׁמֵּרֹב קְדֻשָּׁתוֹ, אֵלִיָּהוּ הַנָּבִיא הָיָה נִגְלֶה אֵלָיו וְלוֹמֵד אִתּוֹ תּוֹרָה תָּמִיד! אֲבָל רַבִּי יְהוֹשֻׁעַ הָיָה גַּם סַבְלָן גָּדוֹל, וְאָהַב אֶת הַתַּלְמִידִים הַקְּטַנִּים בְּאַהֲבָה שֶׁל אַבָּא הָאוֹהֵב אֶת בָּנָיו.

הַמִּשְׁנָה הַקָּשָׁה וְהָעֵינַיִם הַנֶּעֱצָמוֹת

בְּאוֹתוֹ לַיְלָה מְאֻחָר, יָשַׁב בְּבֵית הַמִּדְרָשׁ תַּלְמִיד צָעִיר וּמָתוֹק. נִקְרָא לוֹ אֶלְעָזָר. אֶלְעָזָר יָשַׁב עַל סַפְסַל הָאֶבֶן הַקַּר, מִתַּחַת לְאוֹר שֶׁל נֵר שֶׁמֶן קָטָן, וְנִסָּה לְשַׁנֵּן אֶת הַמִּשְׁנָה. הוּא לָמַד אֶת אוֹתָהּ הַמִּשְׁנָה כְּבָר חָמֵשׁ פְּעָמִים, אַךְ עֲדַיִן לֹא הֵבִין אוֹתָהּ עַד הַסּוֹף.

הוּא הָיָה עָיֵף מְאֹד. עֵינָיו צָרְבוּ. מֵרֹב עֲיֵפוּת, שׁוּרוֹת הָאוֹתִיּוֹת שֶׁעַל הַקְּלָף נִרְאוּ לוֹ כְּמוֹ נְמָלִים שְׁחֹרוֹת שֶׁרָצוֹת מִצַּד לְצַד. אֶלְעָזָר הִשְׁעִין אֶת רֹאשׁוֹ הַכָּבֵד עַל שְׁתֵּי יָדָיו וְנֶאֱנַח אֲנָחָה קְטַנָּה שֶׁל יֵאוּשׁ.

"מַה יִּצֵּא לִי מִזֶּה?" מִלְמֵל לְעַצְמוֹ בְּלַחַשׁ. "לָמָּה אֲנִי צָרִיךְ לַעֲמֹל כָּל כָּךְ קָשֶׁה? בַּחוּץ יֵשׁ לַיְלָה יָפֶה, חֲבֵרַי יְשֵׁנִים בְּמִטּוֹתֵיהֶם הַחַמּוֹת. הַשָּׂדוֹת מְלֵאִים בְּפֵרוֹת לַקָּטִיף, סוֹחֲרִים בַּשּׁוּק מַרְוִיחִים מַטְבְּעוֹת זָהָב שֶׁמְּצַלְצְלִים לָהֶם בַּכִּיס... וְרַק אֲנִי יוֹשֵׁב פֹּה וְשׁוֹבֵר אֶת הָרֹאשׁ עַל מִלָּה שֶׁאֲנִי לֹא מֵבִין... אוּלַי מָחָר אַפְסִיק לִלְמֹד וְאֵלֵךְ לַעֲבֹד בַּשָּׂדֶה?".

הַיָּד שֶׁעַל הַכָּתֵף

פִּתְאוֹם, אֶלְעָזָר הִרְגִּישׁ יָד חַמָּה וַעֲדִינָה מֻנַּחַת לוֹ עַל הַכָּתֵף.

הוּא הֵרִים אֶת רֹאשׁוֹ בְּבֶהָלָה. מוּלוֹ עָמַד לֹא אַחֵר מֵאֲשֶׁר רַבִּי יְהוֹשֻׁעַ בֶּן לֵוִי בִּכְבוֹדוֹ וּבְעַצְמוֹ! עֵינָיו הַחֲכָמוֹת הִבִּיטוּ בְּאֶלְעָזָר בַּהֲבָנָה כָּל כָּךְ עֲמֻקָּה, כְּאִלּוּ קָרְאוּ אֶת הַמַּחֲשָׁבוֹת שֶׁלּוֹ.

"קָשֶׁה לְךָ, בְּנִי?" שָׁאַל הָרַב בְּקוֹל נָעִים וְרַךְ.

אֶלְעָזָר הִשְׁפִּיל מַבָּט. "כֵּן, רַבִּי," בָּלַע אֶת הָרֹק. "קָשֶׁה לִי כָּל כָּךְ. וְלִפְעָמִים... לִפְעָמִים הַיֵּצֶר הָרַע לוֹחֵשׁ לִי שֶׁעָדִיף לְהִתְעַסֵּק בְּדְבָרִים אֲחֵרִים, לֶאֱסֹף כֶּסֶף, לְטַיֵּל, לִקְנוֹת שָׂדוֹת עֲנָקִיִּים. הַתּוֹרָה אוֹמֶרֶת לָנוּ כָּל הַזְּמַן לְהִתְאַמֵּץ וּלְהִתְעַיֵּף... זֶה נִרְאֶה לִי עָמָל עָצוּם."

רַבִּי יְהוֹשֻׁעַ חִיֵּךְ. "בּוֹא אִתִּי, אֶלְעָזָר."

הַצַּדִּיק הַגָּדוֹל לָקַח אֶת הַתַּלְמִיד הַקָּטָן בְּיָדוֹ, וְהֵם יָצְאוּ מִבֵּית הַמִּדְרָשׁ אֶל הָאֲוִיר הַפָּתוּחַ שֶׁל הַלַּיְלָה.

רוּחַ קְרִירָה נָשְׁבָה עַל פְּנֵיהֶם. רֵיחַ מָתוֹק שֶׁל עֲצֵי תְּאֵנָה וְהָדָר הִגִּיעַ מֵהַפַּרְדֵּסִים שֶׁל לוֹד. רַבִּי יְהוֹשֻׁעַ עָצַר, נָשַׁם נְשִׁימָה עֲמֻקָּה וְאָז הִצְבִּיעַ עַל הַשָּׁמַיִם.

תִּסְתַּכֵּל שָׁמַיְמָה

"תִּסְתַּכֵּל, אֶלְעָזָר," לָחַשׁ הָרַב. אֶלְעָזָר הֵרִים אֶת מַבָּטוֹ וְנֶעֱצַר לִנְשׁוֹם. הַשָּׁמַיִם הָיוּ עֲנָקִיִּים, שְׁחֹרִים וַעֲמֻקִּים, וְזְרוּעִים בְּמִילְיוֹנֵי כּוֹכָבִים נוֹצְצִים שֶׁרִצְּדוּ כְּמוֹ יַהֲלוֹמִים מַבְרִיקִים עַל גַּבֵּי קְטִיפָה שְׁחוֹרָה.

"הָעוֹלָם הַזֶּה, אֶלְעָזָר, הוּא גָּדוֹל וְיָפֶה," אָמַר רַבִּי יְהוֹשֻׁעַ בֶּן לֵוִי. "אֲנָשִׁים רָבִים עַל חֶלְקַת אֲדָמָה כָּאן. מְלָכִים יוֹצְאִים לְמִלְחָמוֹת עַל חֲתִיכַת עִיר. סוֹחֲרִים שָׁטִים בַּיָּם כְּדֵי לְהָבִיא תַּבְלִינִים. כֻּלָּם חוֹשְׁבִים שֶׁהָעוֹלָם הַזֶּה הוּא הַדָּבָר הֲכִי גָּדוֹל שֶׁיֵּשׁ, וְשֶׁשָּׁוֶה לַעֲבֹד רַק כְּדֵי לְהַשִּׂיג מִמֶּנּוּ קְצָת."

הָרַב הִסְתּוֹבֵב אֶל הַתַּלְמִיד, וְעֵינָיו בָּרְקוּ מֵהִתְרַגְּשׁוּת קְדוֹשָׁה.

"אֲבָל דַּע לְךָ סוֹד, סוֹד שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא גִּלָּה לְלוֹמְדֵי הַתּוֹרָה. אֲנִי כָּתַבְתִּי אֶת הַסּוֹד הַזֶּה בַּמִּשְׁנָה, וְכָךְ הוּא אוֹמֵר: 'עָתִיד הַקָּדוֹשׁ בָּרוּךְ הוּא לְהַנְחִיל לְכָל צַדִּיק וְצַדִּיק – שְׁלֹשׁ מֵאוֹת וַעֲשָׂרָה (ש"י) עוֹלָמוֹת!' "

אֶלְעָזָר הַקָּטָן פָּתַח תֵּכֶף אֶת פִּיו מֵרֹב תַּדְהֵמָה. "שְׁלֹשׁ מֵאוֹת וַעֲשָׂרָה עוֹלָמוֹת?!" קָרָא בְּתַדְהֵמָה.

"כֵּן!" עָנָה רִיבָּ"ל. "תְּדַמְיֵן אֶת כָּל כַּדּוּר הָאָרֶץ עִם כָּל הֶהָרִים שֶׁלּוֹ, הָאוֹקְיָנוֹסִים, הַיְּעָרוֹת וְהַיַּהֲלוֹמִים... עוֹלָם שָׁלֵם. הַשֵּׁם יִתְבָּרַךְ לֹא מֵכִין לַצַּדִּיק רַק 'עוֹלָם אֶחָד כָּזֶה'. הוּא מֵכִין לְכָל מִי שֶׁלּוֹמֵד תּוֹרָה וּמִתְאַמֵּץ – שְׁלֹשׁ מֵאוֹת וַעֲשָׂרָה עוֹלָמוֹת קְדוֹשִׁים, טְהוֹרִים וּמְלֵאִים בְּאוֹר וּבְשִׂמְחָה! מֵאֵיפֹה אֲנִי יוֹדֵעַ אֶת זֶה? מֵהַפָּסוּק בְּסֵפֶר מִשְׁלֵי: 'לְהַנְחִיל אֹהֲבַי יֵשׁ – וְאֹצְרֹתֵיהֶם אֲמַלֵּא'. הַמִּלָּה 'יֵשׁ' כְּתוּבָה בָּאוֹתִיּוֹת י' (עֶשֶׂר) וְ-ש' (שְׁלֹשׁ מֵאוֹת). סַךְ הַכֹּל: שְׁלֹשׁ מֵאוֹת וָעֶשֶׂר!"

הַנִּצָּחוֹן בְּבֵית הַמִּדְרָשׁ

רַבִּי יְהוֹשֻׁעַ הֵנִיחַ אֶת שְׁתֵּי יָדָיו עַל רֹאשׁוֹ שֶׁל הַיֶּלֶד וּבֵרְכוֹ.

"אֶלְעָזָר בְּנִי, כְּשֶׁאַתָּה לוֹמֵד, כְּשֶׁאַתָּה חוֹזֵר עַל הַמִּשְׁנָה בַּפַּעַם הַשִּׁשִּׁית וְעֵינֶיךָ עֲיֵפוֹת... אַתָּה לֹא סְתָם קוֹרֵא מִלִּים. אַתָּה בּוֹנֶה עוֹלָם בַּשָּׁמַיִם! עַל כָּל מִשְׁנָה יֵשׁ שָׂכָר כָּל כָּךְ עָצוּם, עַד שֶׁעַיִן אֱנוֹשִׁית אֵינָהּ יְכוֹלָה בִּכְלָל לִרְאוֹת אוֹתָהּ וּלְדַמְיֵן כַּמָּה זֶה נִפְלָא. הַאִם שָׁוֶה לִמְכֹּר שְׁלֹשׁ מֵאוֹת וַעֲשָׂרָה עוֹלָמוֹת שֶׁל קְדֻשָּׁה נִצְחִית מֵהַשֵּׁם – בִּשְׁבִיל שָׂדֶה אֶחָד וְכַמָּה לִירוֹת שֶׁל מַטְבְּעוֹת זָהָב בָּעוֹלָם הַזֶּה?"

אֶלְעָזָר שָׁתַק. עֵינָיו נִפְקְחוּ לִרְוָחָה וְדִמְעָה קְטַנָּה שֶׁל הִתְרַגְּשׁוּת יָרְדָה עַל לֶחְיוֹ. לְפֶתַע, הַכֹּל נִרְאָה שׁוֹנֶה. הָעֲיֵפוּת נֶעֶלְמָה כְּלֹא הָיְתָה! הָרַגְלַיִם שֶׁלּוֹ כְּבָר רָצוּ לָרוּץ.

הוּא הֵבִין: הַתּוֹרָה שֶׁהוּא לוֹמֵד יְקָרָה יוֹתֵר מִכָּל הַכֶּסֶף שֶׁיֵּשׁ בָּעוֹלָם, מִכָּל הַיַּהֲלוֹמִים וְהַזָּהָב שֶׁבַּיָּם וּבַיַּבָּשָׁה!

"תּוֹדָה לְךָ, מוֹרִי וְרַבִּי!" מִלְמֵל אֶלְעָזָר מִתּוֹךְ שִׂמְחָה עֲצוּמָה. הוּא הִסְתּוֹבֵב וְרָץ חֲזָרָה אֶל תּוֹךְ בֵּית הַמִּדְרָשׁ. אוֹר הַנֵּר הֵאִיר עַל דַּף הַמִּשְׁנָה, אֲבָל הַפַּעַם, הָאוֹתִיּוֹת כְּבָר לֹא הָיוּ מְטֻשְׁטָשׁוֹת. הֵן זָרְחוּ אֵלָיו כְּמוֹ שְׁלֹשׁ מֵאוֹת וַעֲשָׂרָה כּוֹכָבִים שֶׁל קְדֻשָּׁה בַּשָּׁמַיִם, מַמְתִּינוֹת רַק לוֹ שֶׁיִּלְמַד אוֹתָן, שֶׁיִּבְנֶה דַּרְכָּן אֶת עוֹלָמוֹ בָּנֶצַח.

מקורות: משנה, מסכת עוקצין, פרק ג' משנה י"ב: "אמר רבי יהושע בן לוי, עתיד הקדוש ברוך הוא להנחיל לכל צדיק וצדיק שלש מאות ועשרה עולמות, שנאמר (משלי ח) לְהַנְחִיל אֹהֲבַי יֵשׁ וְאֹצְרֹתֵיהֶם אֲמַלֵּא". תלמוד בבלי, מסכת סנהדרין דף ק. ע"א: מאמרו על שכר הצדיקים. תלמוד בבלי, מסכת כתובות דף עז ע"ב: על כך שריב"ל היה לומד תורה עם יושבי בית המדרש ומתמסר אליהם למרות סכנות, ומעלת לימודו שהגנה עליו (גדולתו של ריב"ל שחיבר תורה למעשה ולימד תלמידים במסירות).

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “יְלָדִים, אִם הֱיִיתֶם נִכְנָסִים לְבֵית הַמִּדְרָשׁ שֶׁל רַבִּי יְהוֹשֻׁעַ בֶּן לֵוִי בָּעִיר לוֹד, הֱיִיתֶם מַרְגִּישִׁים מִיָּד שֶׁאַתֶּם בְּמָקוֹם…” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “אֶלְעָזָר יָשַׁב עַל סַפְסַל הָאֶבֶן הַקַּר, מִתַּחַת לְאוֹר שֶׁל נֵר שֶׁמֶן קָטָן, וְנִסָּה לְשַׁנֵּן אֶת הַמִּשְׁנָה.”?
ג. כיצד הגיב ר' יהושע בן לוי למצב המתואר: “פִּתְאוֹם, אֶלְעָזָר הִרְגִּישׁ יָד חַמָּה וַעֲדִינָה מֻנַּחַת לוֹ עַל הַכָּתֵף.”?
ד. מה השתנה בין “הָעוֹלָם הַזֶּה, אֶלְעָזָר, הוּא גָּדוֹל וְיָפֶה," אָמַר רַבִּי יְהוֹשֻׁעַ בֶּן לֵוִי.” לבין “הוּא מֵכִין לְכָל מִי שֶׁלּוֹמֵד תּוֹרָה וּמִתְאַמֵּץ – שְׁלֹשׁ מֵאוֹת וַעֲשָׂרָה עוֹלָמוֹת קְדוֹשִׁים, טְהוֹרִים…”?
ה. כיצד הגיע הסיפור אל האירוע: “הוּא הֵבִין: הַתּוֹרָה שֶׁהוּא לוֹמֵד יְקָרָה יוֹתֵר מִכָּל הַכֶּסֶף שֶׁיֵּשׁ בָּעוֹלָם, מִכָּל הַיַּהֲלוֹמִים וְהַזָּהָב שֶׁבַּיָּם וּבַיַּבָּשָׁה!”?

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “גדולתו — "ש"י עולמות לעולם הבא"” על ר' יהושע בן לוי? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “הַצַּדִּיק הַגָּדוֹל לָקַח אֶת הַתַּלְמִיד הַקָּטָן בְּיָדוֹ, וְהֵם יָצְאוּ מִבֵּית הַמִּדְרָשׁ אֶל הָאֲוִיר הַפָּתוּחַ שֶׁל הַלַּיְלָה.”.
ג. אילו ר' יהושע בן לוי לא היה פועל כפי שמתואר ב־“הוּא מֵכִין לְכָל מִי שֶׁלּוֹמֵד תּוֹרָה וּמִתְאַמֵּץ – שְׁלֹשׁ מֵאוֹת וַעֲשָׂרָה עוֹלָמוֹת קְדוֹשִׁים, טְהוֹרִים וּמְלֵאִים…” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “הוּא הֵבִין: הַתּוֹרָה שֶׁהוּא לוֹמֵד יְקָרָה יוֹתֵר מִכָּל הַכֶּסֶף שֶׁיֵּשׁ בָּעוֹלָם, מִכָּל הַיַּהֲלוֹמִים וְהַזָּהָב…” מחזק את המסר של הסיפור?

1. יְלָדִים, אִם הֱיִיתֶם נִכְנָסִים לְבֵית הַמִּדְרָשׁ שֶׁל רַבִּי…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “פִּתְאוֹם, אֶלְעָזָר הִרְגִּישׁ יָד חַמָּה וַעֲדִינָה מֻנַּחַת לוֹ עַל הַכָּתֵף.”.
מכתב: כתבו לר' יהושע בן לוי בעקבות “אֲנִי כָּתַבְתִּי אֶת הַסּוֹד הַזֶּה בַּמִּשְׁנָה, וְכָךְ הוּא אוֹמֵר: 'עָתִיד הַקָּדוֹשׁ בָּרוּךְ הוּא לְהַנְחִיל לְכָל צַדִּיק…”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “רַבִּי יְהוֹשֻׁעַ הֵנִיחַ אֶת שְׁתֵּי יָדָיו עַל רֹאשׁוֹ שֶׁל הַיֶּלֶד וּבֵרְכוֹ.”.
דיווח: כתבו ידיעה קצרה שכותרתה “גדולתו — "ש"י עולמות לעולם הבא"”.

המחזה

רגע מן הסיפורהדמויותמה יאמרו או יעשו?
אֶלְעָזָר הִשְׁעִין אֶת רֹאשׁוֹ הַכָּבֵד עַל שְׁתֵּי יָדָיו וְנֶאֱנַח אֲנָחָה קְטַנָּה שֶׁל יֵאוּשׁ.
