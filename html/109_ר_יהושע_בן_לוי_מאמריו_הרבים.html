מאמריו הרבים

דף עבודה בעקבות ר' יהושע בן לוי
שם: כיתה:

המעשה — הסיפור המלא

פְּתִיחָה: רֵיחַ הַקְּלָפִים וְקוֹל הַתּוֹרָה

יְלָדִים, נַסּוּ לְדַמְיֵן אֶת עַצְמְכֶם הוֹלְכִים בִּרְחוֹבוֹת הָעִיר לוֹד לִפְנֵי כְּאַלְפַּיִם שָׁנָה. הָרְחוֹב הָמֶה מֵאֲנָשִׁים, סוֹחֲרִים צוֹעֲקִים בַּשּׁוּק, עֲגָלוֹת עֵץ חוֹרְקוֹת עַל אַבְנֵי הַדֶּרֶךְ. אַךְ מֵעַל כָּל הָרַעַשׁ הַזֶּה, אִם תַּעַצְמוּ רֶגַע אֶת הָעֵינַיִם וְתַקְשִׁיבוּ הֵיטֵב, תִּשְׁמְעוּ קוֹל מָתוֹק מִדְּבַשׁ. קוֹל שֶׁל מֵאוֹת תַּלְמִידִים הַלּוֹמְדִים תּוֹרָה בְּנִגּוּן נִפְלָא.

אִם תִּכָּנְסוּ לְתוֹךְ בֵּית הַמִּדְרָשׁ, יַכֶּה בְּאַפְּכֶם רֵיחַ נָעִים וְעַתִּיק שֶׁל גְּוִילֵי קְלָף אֲרֻכִּים וּסְפָרִים יְשָׁנִים. בַּמֶּרְכָּז, עַל כִּסֵּא מְכֻבָּד, יוֹשֵׁב צַדִּיק עֲנָק שֶׁפָּנָיו מְאִירוֹת כְּמַלְאַךְ ה'. זֶהוּ רַבִּי יְהוֹשֻׁעַ בֶּן לֵוִי !

רַבִּי יְהוֹשֻׁעַ לֹא הָיָה מְלַמֵּד אֶת תַּלְמִידָיו רַק הֲלָכוֹת. הוּא לִמֵּד אוֹתָם אֵיךְ לִהְיוֹת אֲנָשִׁים טוֹבִים יוֹתֵר, אֵיךְ לִשְׁמֹר עַל הַמִּדּוֹת שֶׁלָּהֶם.

"יְלָדִים", קָרָא רַבִּי יְהוֹשֻׁעַ אֶל הַתַּלְמִידִים הַצְּעִירִים, "הַאִם פַּעַם כָּאַב לָכֶם הָרֹאשׁ הַרְבֵּה זְמַן?" הַתַּלְמִידִים הִנְהֲנוּ. "אִם כָּאַב לָכֶם, הַאִם רְצִיתֶם לַעֲזֹב הַכֹּל וְלָלֶכֶת לִישֹׁן? דְּעוּ לָכֶם אֶת הַסּוֹד הַגָּדוֹל: חָשׁ בְּרֹאשׁוֹ – יַעֲסֹק בַּתּוֹרָה! כִּי דִּבְרֵי הַתּוֹרָה הֵם הַתְּרוּפָה הֲכִי חֲזָקָה וּמְתוּקָה בָּעוֹלָם!".

וְאָז, הֵרִים רַבִּי יְהוֹשֻׁעַ אֶת אֶצְבָּעוֹ וְלִמֵּד אוֹתָם עוֹד כְּלָל חָשׁוּב, כְּלָל עַל הַפֶּה שֶׁלָּנוּ: " לְעוֹלָם אַל יוֹצִיא אָדָם דָּבָר מְגֻנֶּה מִפִּיו! (מִלָּה לֹא נְקִיָּה אוֹ מְכֹעֶרֶת). שֶׁהֲרֵי הַתּוֹרָה שִׁנְּתָה וְהֶאֱרִיכָה בְּמִלִּים, וּבִלְבַד שֶׁלֹּא לִכְתֹּב מִלָּה מְגֻנָּה בַּתּוֹרָה הַקְּדוֹשָׁה!". הַתַּלְמִידִים רָאוּ אֵיךְ שְׂפָתָיו שֶׁל רַבָּם זָזוֹת בִּקְדֻשָּׁה עֲצוּמָה. הֵם יָדְעוּ שֶׁמִּפִּיו שֶׁל רַבִּי יְהוֹשֻׁעַ בֶּן לֵוִי לֹא יָצְאָה מֵעוֹלָם, מִיּוֹם שֶׁנּוֹלַד, שׁוּם מִלָּה רָעָה אוֹ לֹא נְקִיָּה. רַק תּוֹרָה, רַק טֹהַר וְרַק בְּרָכוֹת.

הַמֶּרְחָץ וְהַבְּרִיחָה אֶל הַנֶּכֶד

יוֹם אֶחָד, הָלַךְ רַבִּי יְהוֹשֻׁעַ בֶּן לֵוִי לְבֵית הַמֶּרְחָץ כְּדֵי לִרְחֹץ אֶת גּוּפוֹ הַקָּדוֹשׁ כְּדֵי שֶׁיִּהְיֶה נָקִי לְלִמּוּד הַתּוֹרָה. מֵי הַמֶּרְחָץ הָיוּ חַמִּים וּנְעִימִים. אֵדִים עָבִים שֶׁל חֹם וְלַחוּת מִלְּאוּ אֶת הַחֶדֶר. זֶה הָיָה הָרֶגַע הַיָּחִיד בַּיּוֹם שֶׁבּוֹ הָרַב הַגָּדוֹל יָכוֹל הָיָה קְצָת לָנוּחַ מֵעֲמַל הַלִּמּוּד הָרַב. הוּא נִכְנַס לַמַּיִם הַחַמִּים, אַךְ לְפֶתַע, הוּא עָצַר...

עֵינָיו נִפְקְחוּ לִרְוָחָה.

"אוֹי וַאֲבוֹי!" קָרָא רַבִּי יְהוֹשֻׁעַ וְזִנֵּק מֵהַמַּיִם הַחַמִּים, "הַזְּמַן! מָה הַשָּׁעָה עַכְשָׁו?!".

אֲנָשִׁים בַּמֶּרְחָץ הִבִּיטוּ בּוֹ בִּפְלִיאָה: "רַבֵּנוּ, מָה קָרָה? לָמָּה כְּבוֹדוֹ מְמַהֵר כָּל כָּךְ? הַמַּיִם מַמָּשׁ נְעִימִים הַיּוֹם!".

אַךְ רַבִּי יְהוֹשֻׁעַ בֶּן לֵוִי לֹא נִשְׁאַר שָׁם אֲפִלּוּ דַּקָּה אַחַת נוֹסֶפֶת. הוּא מִהֵר לְנַגֵּב אֶת עַצְמוֹ, לָבַשׁ אֶת בְּגָדָיו, רָץ אֶל הָרְחוֹב וּמִהֵר יְשִׁירוֹת אֶל... בֵּית הַמִּדְרָשׁ!

וּמָה הָיָה כָּל כָּךְ דָּחוּף?

כְּשֶׁהִגִּיעַ לְבֵית הַמִּדְרָשׁ, מָצָא שָׁם אֶת נֶכְדּוֹ הַקָּטָן מַמְתִּין לוֹ. רַבִּי יְהוֹשֻׁעַ הִתְיַשֵּׁב לְיַד הַנֶּכֶד בְּשִׂמְחָה עֲצוּמָה, וְהִתְחִיל לִלְמֹד אִתּוֹ אֶת פָּרָשַׁת הַשָּׁבוּעַ!

תַּלְמִידָיו נִגְּשׁוּ אֵלָיו וְשָׁאֲלוּ בְּתַדְהֵמָה: "רַבֵּנוּ, לָמָּה יָצָאתָ בְּאֶמְצַע הַמֶּרְחָץ? יָכוֹלְתָּ לְסַיֵּם וְאָז לָבוֹא!".

עָנָה לָהֶם רַבִּי יְהוֹשֻׁעַ תְּשׁוּבָה שֶׁנֶּחְקְבָה לָנֶצַח: "הַאִם כָּל כָּךְ דָּבָר קָטָן הוּא בְּעֵינֵיכֶם, לְלַמֵּד תּוֹרָה אֶת הַנֶּכֶד? מִי שֶׁמְּלַמֵּד אֶת נֶכְדּוֹ תּוֹרָה, נֶחְשָׁב לוֹ כְּאִלּוּ קִבֵּל בְּאוֹתוֹ הָרֶגַע אֶת הַתּוֹרָה עַצְמָהּ מֵהַר סִינַי! אֵיךְ אוּכַל לְעַכֵּב מַעֲמַד הַר סִינַי בִּשְׁבִיל מַיִם חַמִּים?!".

הַפֶּה שֶׁל רַבִּי יְהוֹשֻׁעַ בֶּן לֵוִי הָיָה דָּבוּק לַתּוֹרָה בְּכָל כֹּחוֹ. אִי אֶפְשָׁר הָיָה לְהַפְרִיד בֵּינֵיהֶם.

אֲבָנִים זוֹהֲרוֹת בַּמְּעָרָה

יְלָדִים, אָדָם שֶׁשּׁוֹמֵר עַל הַפֶּה שֶׁלּוֹ כָּל כָּךְ יָפֶה, וְלוֹמֵד תּוֹרָה מִתּוֹךְ כְּאֵב וְקֹשִׁי – הַקָּדוֹשׁ בָּרוּךְ הוּא עוֹשֶׂה עִמּוֹ חֶסֶד שֶׁאִי אֶפְשָׁר לְתָאֵר.

בְּאוֹתָם יָמִים, הָיָה נָבִיא קָדוֹשׁ שֶׁיָּרַד מִן הַשָּׁמַיִם כְּדֵי לִלְמֹד תּוֹרָה אִם רַבִּי יְהוֹשֻׁעַ בֶּן לֵוִי. מִי זֶה הָיָה? אֵלִיָּהוּ הַנָּבִיא!

יוֹם אֶחָד, יָשְׁבוּ הַשְּׁנַיִם בְּבֵית הַמִּדְרָשׁ. רַבִּי יְהוֹשֻׁעַ בֶּן לֵוִי סָגַר אֶת הַגְּמָרָא שֶׁלּוֹ בְּשֶׁקֶט, וְנֶאֱנַח אֲנָחָה עֲמֻקָּה שֶׁיָּצְאָה מֵהַלֵּב. הוּא נִזְכַּר בִּירוּשָׁלַיִם שֶׁחֲרֵבָה, וּבְבֵית הַמִּקְדָּשׁ שֶׁנִּשְׂרַף. הַנְּבִיאִים הִבְטִיחוּ שֶׁלֶּעָתִיד לָבוֹא, הַקָּדוֹשׁ בָּרוּךְ הוּא יִבְנֶה אֶת קִירוֹת יְרוּשָׁלַיִם מֵאֲבָנִים טוֹבוֹת וְיְקָרוֹת.

"אֵלִיָּהוּ", בִּקֵּשׁ רַבִּי יְהוֹשֻׁעַ מִידִידוֹ שֶׁמִּן הַשָּׁמַיִם, "הַנָּבִיא יְשַׁעְיָהוּ אָמַר שֶׁיְּרוּשָׁלַיִם תִּבָּנֶה מֵאַבְנֵי קֹדֶשׁ שֶׁנִּקְרָאוֹת 'אַבְנֵי כַּדְכֹּד'. הַאִם תּוּכַל... הַאִם תּוּכַל לְהַרְאוֹת לִי פַּעַם אַחַת אֵיךְ נִרְאֵית הָאֶבֶן הַזּוֹ? אֵיךְ יֵרָאֶה הָאוֹר שֶׁל יְרוּשָׁלַיִם הַבְּנוּיָה?".

אֵלִיָּהוּ הַנָּבִיא חִיֵּךְ חִיּוּךְ רָחָב וְהֵשִׁיב: "הֵן. בִּזְכוּת תּוֹרָתְךָ, הַקָּדוֹשׁ בָּרוּךְ הוּא הִרְשָׁה לִי לְהַרְאוֹת לְךָ אֶת אֶבֶן הַכַּדְכֹּד!".

בְּאוֹתוֹ זְמַן, הַרְחֵק מִשָּׁם, בְּאֶמְצַע לֵב הַיָּם הַגָּדוֹל, הָיְתָה סְפִינָה אֲצוּמָה שֶׁנִּקְלְעָה לְסַעֲרָה אֲיֻמָּה. הַגַּלִּים טָלְטוּ אֶת הַסְּפִינָה מִצַּד לְצַד, מַיִם עַזִּים נִכְנְסוּ אֵלֶיהָ, וְהִיא עָמְדָה לִטְבֹּעַ! בְּתוֹךְ הַסְּפִינָה יָשַׁב נַעַר יְהוּדִי שֶׁבָּכָה וְהִתְפַּלֵּל לַה'.

פִּתְאוֹם, מוּל עֵינָיו שֶׁל הַנַּעַר, הוֹפִיעָה דְּמוּת מְאִירָה – זֶה הָיָה אֵלִיָּהוּ הַנָּבִיא!

"נַעַר יָקָר", אָמַר לוֹ אֵלִיָּהוּ עַל הַסְּפִינָה, "אִם תַּסְכִּים לַעֲשׂוֹת בִּשְׁבִילִי שְׁלִיחוּת – הַסְּפִינָה שֶׁלָּכֶם תִּנָּצֵל!".

"כֵּן! בְּוַדַּאי!" זָעַק הַנַּעַר מִבַּעַד לַגַּלִּים, "רַק תַּצִּיל אוֹתָנוּ!".

"אֲנִי אֶתֵּן לְךָ אֶבֶן זְהוֹרִית עֲטוּפָה בְּבַד," הִנְחָה אֵלִיָּהוּ, "לֵךְ מִיָּד לָעִיר לוֹד, חַפֵּשׂ שָׁם אֶת הַצַּדִּיק רַבִּי יְהוֹשֻׁעַ בֶּן לֵוִי. אֲבָל זְכֹר! אַל תַּרְאֶה לוֹ אֶת הָאֶבֶן בָּעִיר כִּי הָאוֹר שֶׁלָּהּ חָזָק מִדַּי! קַח אוֹתוֹ לִמְעָרָה חֲשׁוּכָה מִחוּץ לָעִיר, וְרַק שָׁם תִּפְתַּח אוֹתָהּ".

וְכָךְ קָרָה! הַסְּפִינָה נִצְּלָה בְּנֵס, וְהַנַּעַר מִהֵר לְאֶרֶץ יִשְׂרָאֵל. כְּשֶׁהִגִּיעַ לְלוֹד, הוּא מָצָא אֶת רַבִּי יְהוֹשֻׁעַ יְשׁוּב בְּבֵית הַמִּדְרָשׁ כְּשֶׁפָּנָיו מְאִירוֹת בַּתּוֹרָה.

"רַבִּי," קָרָא הַנַּעַר, "אֵלִיָּהוּ הַנָּבִיא שָׁלַח אוֹתִי אֵלֶיךָ!". רַבִּי יְהוֹשֻׁעַ קָם בְּהִתְרַגְּשׁוּת וְהָלַךְ אַחֲרֵי הַנַּעַר עַד לִמְעָרָה חֲשׁוּכָה מְאֹד מִחוּץ לְלוֹד. הַחֹשֶׁךְ הָיָה סָמִיךְ וְקַר.

"מוּכָן כְּבוֹד הָרַב?" שָׁאַל הַנַּעַר בִּרְעָדָה, וּפָתַח אֶת הַבַּד הַשָּׁחֹר.

יְלָדִים... אִי אֶפְשָׁר בִּכְלָל לְדַמְיֵן אֶת מָה שֶׁקָּרָה שָׁם!

מִתּוֹךְ כַּף יָדוֹ שֶׁל הַנַּעַר בָּקַע אוֹר צָהֹב, זָהָב וְתָכֵלֶת. זֶה לֹא הָיָה אוֹר רָגִיל שֶׁל פָּנָס אוֹ שֶׁל אֵשׁ. זֶה הָיָה "אוֹר שֶׁל תּוֹרָה"! אוֹר גַּן עֵדֶן הַגָּנוּז. אַבְנֵי הַכַּדְכֹּד!

הָאוֹר בָּקַע מֵהַמְּעָרָה בְּכֹחַ עָצוּם, חָדַר אֶת קִירוֹת הָאֶבֶן הַכְּבֵדִים, עָבַר דֶּרֶךְ הֶהָרִים... עַד שֶׁכָּל הָעִיר לוֹד וּסְבִיבָתָהּ, עַד לַמֶּרְחַקִּים הֲכִי גְּדוֹלִים, הָפְכוּ לְמוּאָרִים בְּאוֹר מַדְהִים וּמַזְהִיר כְּמוֹ בְּאֶמְצַע הַיּוֹם!

רַבִּי יְהוֹשֻׁעַ בֶּן לֵוִי הִבִּיט בָּאֶבֶן וּבָכָה מֵרֹב שִׂמְחָה. עַכְשָׁו הוּא רָאָה מָה זֶה הָאוֹר שֶׁל יְרוּשָׁלַיִם הַבְּנוּיָה! הָאֶבֶן זָרְחָה בְּדִיּוּק כְּמוֹ שֶׁהַתּוֹרָה זָרְחָה מִפִּיו שֶׁל רַבִּי יְהוֹשֻׁעַ עַצְמוֹ!

סִיּוּם הַמִּשְׁנָה כֻּלָּהּ

בִּזְכוּת מַאֲמָרָיו הָרַבִּים וּדְבָרָיו הַקְּדוֹשִׁים שֶׁל רַבִּי יְהוֹשֻׁעַ בֶּן לֵוִי, חֲכָמֵינוּ זִכְרוֹנָם לִבְרָכָה נָתְנוּ לוֹ אֶת הַכָּבוֹד הֲכִי גָּדוֹל שֶׁאֶפְשָׁר.

הֵם סִיְּמוּ אֶת כָּל שִׁשָּׁה סִדְרֵי מִשְׁנָה (הַסֵּפֶר הֲכִי גָּדוֹל שֶׁל תּוֹרָה שֶׁבְּעַל פֶּה) בְּמִשְׁפָּט אֶחָד יָפֶה שֶׁלּוֹ:

"אָמַר רַבִּי יְהוֹשֻׁעַ בֶּן לֵוִי: עָתִיד הַקָּדוֹשׁ בָּרוּךְ הוּא לְהַנְחִיל לְכָל צַדִּיק וְצַדִּיק, שְׁלֹשׁ מֵאוֹת וַעֲשָׂרָה (ש"י) עוֹלָמוֹת!". זֶהוּ הַשָּׂכָר לְמִי שֶׁלּוֹמֵד תּוֹרָה, שׁוֹמֵר עַל הַפֶּה שֶׁלּוֹ, וּמֵבִיא הַרְבֵּה אוֹר לָעוֹלָם.

מקורות: "לעולם אל יוציא אדם דבר מגונה מפיו" – מסכת פסחים דף ג עמוד א'. "חש בראשו יעסוק בתורה" – מסכת עירובין דף נד עמוד א'. יצא מהמרחץ ללמוד עם נכדו (פרשת השבוע) – תלמוד ירושלמי, תחילת מסכת ברכות (ויש דמיון למסכת קידושין ל ע"א / מגילה כז ע"ב). אבני הכדכוד, הספינה בים וגילוי אליהו הנביא – פסיקתא רבתי (איש שלום), פיסקא לב' "ענייה סוערה" / ילקוט שמעוני ישעיהו נד, רמז תע"ח. חתימת המשנה ("עתיד הקב"ה להנחיל...") – מסכת עוקצין פרק ג', משנה י"ב.

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “יְלָדִים, נַסּוּ לְדַמְיֵן אֶת עַצְמְכֶם הוֹלְכִים בִּרְחוֹבוֹת הָעִיר לוֹד לִפְנֵי כְּאַלְפַּיִם שָׁנָה.” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “כִּי דִּבְרֵי הַתּוֹרָה הֵם הַתְּרוּפָה הֲכִי חֲזָקָה וּמְתוּקָה בָּעוֹלָם!".”?
ג. כיצד הגיב ר' יהושע בן לוי למצב המתואר: “הוּא מִהֵר לְנַגֵּב אֶת עַצְמוֹ, לָבַשׁ אֶת בְּגָדָיו, רָץ אֶל הָרְחוֹב וּמִהֵר יְשִׁירוֹת אֶל...”?
ד. מה השתנה בין “אֵלִיָּהוּ", בִּקֵּשׁ רַבִּי יְהוֹשֻׁעַ מִידִידוֹ שֶׁמִּן הַשָּׁמַיִם, "הַנָּבִיא יְשַׁעְיָהוּ אָמַר שֶׁיְּרוּשָׁלַיִם…” לבין “קַח אוֹתוֹ לִמְעָרָה חֲשׁוּכָה מִחוּץ לָעִיר, וְרַק שָׁם תִּפְתַּח אוֹתָהּ".”?
ה. כיצד הגיע הסיפור אל האירוע: “עַכְשָׁו הוּא רָאָה מָה זֶה הָאוֹר שֶׁל יְרוּשָׁלַיִם הַבְּנוּיָה!”?

3. מה היה ראשון?

כתבו ליד כל אירוע מספר מ־1 עד 5 לפי סדר התרחשותו בסיפור.

□ רַבִּי יְהוֹשֻׁעַ בֶּן לֵוִי סָגַר אֶת הַגְּמָרָא שֶׁלּוֹ בְּשֶׁקֶט, וְנֶאֱנַח אֲנָחָה עֲמֻקָּה שֶׁיָּצְאָה מֵהַלֵּב.
□ יְלָדִים, נַסּוּ לְדַמְיֵן אֶת עַצְמְכֶם הוֹלְכִים בִּרְחוֹבוֹת הָעִיר לוֹד לִפְנֵי כְּאַלְפַּיִם שָׁנָה.
□ זֶהוּ הַשָּׂכָר לְמִי שֶׁלּוֹמֵד תּוֹרָה, שׁוֹמֵר עַל הַפֶּה שֶׁלּוֹ, וּמֵבִיא הַרְבֵּה אוֹר לָעוֹלָם.
□ הֵם יָדְעוּ שֶׁמִּפִּיו שֶׁל רַבִּי יְהוֹשֻׁעַ בֶּן לֵוִי לֹא יָצְאָה מֵעוֹלָם, מִיּוֹם שֶׁנּוֹלַד, שׁוּם מִלָּה רָעָה אוֹ לֹא נְקִיָּה.
□ כְּשֶׁהִגִּיעַ לְלוֹד, הוּא מָצָא אֶת רַבִּי יְהוֹשֻׁעַ יְשׁוּב בְּבֵית הַמִּדְרָשׁ כְּשֶׁפָּנָיו מְאִירוֹת בַּתּוֹרָה.

לפני ואחרי

מה קרה לפני: “רַבִּי יְהוֹשֻׁעַ בֶּן לֵוִי סָגַר אֶת הַגְּמָרָא שֶׁלּוֹ בְּשֶׁקֶט, וְנֶאֱנַח אֲנָחָה עֲמֻקָּה שֶׁיָּצְאָה מֵהַלֵּב.”?
מה קרה אחרי: “כְּשֶׁהִגִּיעַ לְלוֹד, הוּא מָצָא אֶת רַבִּי יְהוֹשֻׁעַ יְשׁוּב בְּבֵית הַמִּדְרָשׁ כְּשֶׁפָּנָיו מְאִירוֹת בַּתּוֹרָה.”?

סיבה ותוצאה מתוך המעשה

האירוע הנתוןמה גרם לו?מה הייתה תוצאתו?
יוֹם אֶחָד, הָלַךְ רַבִּי יְהוֹשֻׁעַ בֶּן לֵוִי לְבֵית הַמֶּרְחָץ כְּדֵי לִרְחֹץ אֶת גּוּפוֹ הַקָּדוֹשׁ כְּדֵי שֶׁיִּהְיֶה נָקִי לְלִמּוּד הַתּוֹרָה.
פִּתְאוֹם, מוּל עֵינָיו שֶׁל הַנַּעַר, הוֹפִיעָה דְּמוּת מְאִירָה – זֶה הָיָה אֵלִיָּהוּ הַנָּבִיא!

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “מאמריו הרבים” על ר' יהושע בן לוי? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “יְלָדִים, אָדָם שֶׁשּׁוֹמֵר עַל הַפֶּה שֶׁלּוֹ כָּל כָּךְ יָפֶה, וְלוֹמֵד תּוֹרָה מִתּוֹךְ כְּאֵב וְקֹשִׁי – הַקָּדוֹשׁ בָּרוּךְ הוּא עוֹשֶׂה עִמּוֹ חֶסֶד…”.
ג. אילו ר' יהושע בן לוי לא היה פועל כפי שמתואר ב־“קַח אוֹתוֹ לִמְעָרָה חֲשׁוּכָה מִחוּץ לָעִיר, וְרַק שָׁם תִּפְתַּח אוֹתָהּ".” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “עַכְשָׁו הוּא רָאָה מָה זֶה הָאוֹר שֶׁל יְרוּשָׁלַיִם הַבְּנוּיָה!” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. יְלָדִים, נַסּוּ לְדַמְיֵן אֶת עַצְמְכֶם הוֹלְכִים בִּרְחוֹבוֹת הָעִיר…
2. וְאָז, הֵרִים רַבִּי יְהוֹשֻׁעַ אֶת אֶצְבָּעוֹ וְלִמֵּד אוֹתָם עוֹד…
3. רַבִּי יְהוֹשֻׁעַ הִתְיַשֵּׁב לְיַד הַנֶּכֶד בְּשִׂמְחָה עֲצוּמָה,…
4. הַגַּלִּים טָלְטוּ אֶת הַסְּפִינָה מִצַּד לְצַד, מַיִם עַזִּים…
5. אִי אֶפְשָׁר בִּכְלָל לְדַמְיֵן אֶת מָה שֶׁקָּרָה שָׁם!
6. זֶהוּ הַשָּׂכָר לְמִי שֶׁלּוֹמֵד תּוֹרָה, שׁוֹמֵר עַל הַפֶּה שֶׁלּוֹ,…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “הוּא מִהֵר לְנַגֵּב אֶת עַצְמוֹ, לָבַשׁ אֶת בְּגָדָיו, רָץ אֶל הָרְחוֹב וּמִהֵר יְשִׁירוֹת אֶל...”.
מכתב: כתבו לר' יהושע בן לוי בעקבות “פִּתְאוֹם, מוּל עֵינָיו שֶׁל הַנַּעַר, הוֹפִיעָה דְּמוּת מְאִירָה – זֶה הָיָה אֵלִיָּהוּ הַנָּבִיא!”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “מִתּוֹךְ כַּף יָדוֹ שֶׁל הַנַּעַר בָּקַע אוֹר צָהֹב, זָהָב וְתָכֵלֶת.”.
דיווח: כתבו ידיעה קצרה שכותרתה “מאמריו הרבים”.

המחזה

רגע מן הסיפורהדמויותמה יאמרו או יעשו?
יוֹם אֶחָד, הָלַךְ רַבִּי יְהוֹשֻׁעַ בֶּן לֵוִי לְבֵית הַמֶּרְחָץ כְּדֵי לִרְחֹץ אֶת גּוּפוֹ…
קַח אוֹתוֹ לִמְעָרָה חֲשׁוּכָה מִחוּץ לָעִיר, וְרַק שָׁם תִּפְתַּח אוֹתָהּ".

7. חיבור לחיים מתוך המעשה

א. מה אפשר ללמוד מן הדרך שבה ר' יהושע בן לוי פעל באירוע “אֵלִיָּהוּ", בִּקֵּשׁ רַבִּי יְהוֹשֻׁעַ מִידִידוֹ שֶׁמִּן הַשָּׁמַיִם, "הַנָּבִיא יְשַׁעְיָהוּ אָמַר שֶׁיְּרוּשָׁלַיִם תִּבָּנֶה…”?
ב. תארו מצב מחיי הכיתה שבו אפשר ליישם את המסר של “מאמריו הרבים”.
