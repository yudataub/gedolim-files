משפטו של עשיר גאה

דף עבודה בעקבות רב
שם: כיתה:

המעשה — הסיפור המלא

להלן עיבוד חינוכי, עשיר ומרתק על הבעל שם טוב הקדוש, תלמידו רבי דוד ממיקולייב, והעשיר הגאה , המעובד במיוחד לבנים בכיתות ב'-ד' (חינוך דתי-לאומי). הסיפור בנוי על פי המקורות שהבאת, ומדגיש את מידת הענווה וההכרה שכל מה שיש לנו – מהקב"ה הוא.

הסיפור מנוקד במלואו כנדרש.

פְּתִיחָה: אַרְמוֹן שֶׁל זָהָב וְלֵב גֵּאֶה

יְלָדִים, בּוֹאוּ נִכָּנֵס יַחַד לְתוֹךְ אֲחֻזָּתוֹ שֶׁל אֶחָד הָאֲנָשִׁים הַעֲשִׁירִים בְּיוֹתֵר בְּמִזְרַח אֵירוֹפָּה, בִּימֵי מָרָן הַבַּעַל שֵׁם טוֹב הַקָּדוֹשׁ.

עִצְמוּ רֶגַע עֵינַיִם. הַאִם אַתֶּם מַרְגִּישִׁים אֶת כֻּרְסְאוֹת הַקְּטִיפָה הָרַכּוֹת? הַאִם אַתֶּם מוּרִיחִים אֶת רֵיחַ הַבְּשָׂמִים הַיְּקָרִים שֶׁהֵבִיאוּ הַסּוֹחֲרִים מֵאֲרָצוֹת רְחוֹקוֹת? הַבִּיטוּ לְמַעְלָה: נִבְרְשׁוֹת קְרִיסְטָל נוֹצְצוֹת מְאִירוֹת אֶת הַחֶדֶר, וּמִבַּחוּץ נִשְׁמָע קוֹל שִׁקְשׁוּק פַּרְסוֹת הַסּוּסִים הָאֲצִילִים שֶׁשַּׁיָּכִים לַגְּבִיר הַגָּדוֹל.

הַגְּבִיר הַזֶּה הָיָה יְהוּדִי שֶׁלָּמַד תּוֹרָה, וְגַם נָתַן הֲמוֹן צְדָקָה. כָּל עָנִי שֶׁהָיָה דּוֹפֵק בְּדַלְתּוֹ, הָיָה שׁוֹמֵעַ אֶת צִלְצוּל מַטְבְּעוֹת הַזָּהָב נוֹפְלוֹת אֶל תּוֹךְ כִּיסוֹ בְּיָד רְחָבָה.

אֲבָל... הָיְתָה בְּלִבּוֹ שֶׁל הַגְּבִיר נְקֻדָּה אַחַת שְׁחוֹרָה וְקָשָׁה. נְקֻדָּה שֶׁל גַּאֲוָה.

כְּשֶׁהָיוּ הָעֲנִיִּים מְבָרְכִים אוֹתוֹ שֶׁיַּמְשִׁיךְ לִהְיוֹת עָשִׁיר, הָיָה הַגְּבִיר זּוֹקֵף אֶת קוֹמָתוֹ, מְסַדֵּר אֶת מְעִילוֹ הַיָּקָר וְאוֹמֵר בְּקוֹל גֵּאֶה מְאֹד:

"תּוֹדָה עַל הַבְּרָכָה, אֲבָל בֵּינֵינוּ... אֵין שׁוּם דֶּרֶךְ בָּעוֹלָם שֶׁאֲנִי אֶהְיֶה עָנִי! זֶה פָּשׁוּט בִּלְתִּי אֶפְשָׁרִי!"

"לָמָּה?" הָיוּ הָעֲנִיִּים שׁוֹאֲלִים בִּפְלִיאָה.

"כִּי יֵשׁ לִי עֲסָקִים בְּכָל מָקוֹם!" הָיָה הַגְּבִיר מִתְפָּאֵר. "יֵשׁ לִי בָּתֵּי חֲרוֹשֶׁת לְעֵצִים מִצַּד אֶחָד שֶׁל הַמְּדִינָה, סְפִינוֹת מַשָּׂא בַּיָּם הַגָּדוֹל, וְשָׂדוֹת עֲנָקִיִּים שֶׁל תְּבוּאָה! אִם עֵסֶק אֶחָד נִפְגַּע, הַשֵּׁנִי מַצְלִיחַ. אֲנִי בָּנִיתִי אֶת חֲבֵרַת הַמִּסְחָר הַזּוֹ בְּחָכְמָתִי הָרַבָּה, וְאַף אֶחָד לֹא יָכוֹל לְהַפִּיל אוֹתִי מֵהָעֹשֶׁר שֶׁלִּי!".

יְלָדִים, הַמִּלִּים הָאֵלֶּה עָשׂוּ רַעַשׁ גָּדוֹל! אֲבָל לֹא בָּאָרֶץ, אֶלָּא – בַּשָּׁמַיִם.

בְּבֵית דִּין שֶׁל מַעְלָה הִזְדַּעְזְעוּ: הֵיכָן הָאֱמוּנָה?! הַאִם הָאִישׁ הַזֶּה שָׁכַח אֶת הַפָּסוּק בַּתּוֹרָה שֶׁאוֹמֵר: "ה' מוֹרִישׁ וּמַעֲשִׁיר"?! מִי נוֹתֵן לָאָדָם אֶת כֹּחוֹ אִם לֹא בּוֹרֵא עוֹלָם?!

מִיָּד נֶחְתַּם גְּזַר דִּין נוֹרָא בַּשָּׁמַיִם: הַגְּבִיר הַזֶּה, הַגַּאַוְתָן, יְאַבֵּד אֶת כָּל כַּסְפּוֹ וְיֵרֵד מִנְּכָסָיו.

הַשָּׁלִיחַ שֶׁל הַבַּעַל שֵׁם טוֹב

בְּאוֹתָהּ עֵת, בְּעֵירוֹ שֶׁל רַבֵּנוּ הַבַּעַל שֵׁם טוֹב הַקָּדוֹשׁ, הָיָה אוֹר גָּדוֹל. הַבַּעַל שֵׁם טוֹב, שֶׁעֵינָיו הַקְּדוֹשׁוֹת רָאוּ מִסּוֹף הָעוֹלָם וְעַד סוֹפוֹ, רָאָה מַה שֶׁנִּגְזַר עַל אוֹתוֹ גְּבִיר.

הַצַּדִּיק שֶׁאָהַב כָּל יְהוּדִי אֲהָבָה עֲצוּמָה, עָמַד בִּתְפִלָּה עֲמֻקָּה מִקִּירוֹת לִבּוֹ: "רִבּוֹנוֹ שֶׁל עוֹלָם! רַחֵם עָלָיו! תֵּן לוֹ עוֹד הִזְדַּמְּנוּת אַחַת לַחְזֹר בִּתְשׁוּבָה, לְהָבִין שֶׁאַתָּה הוּא מְקוֹר הַכֹּל!".

בַּשָּׁמַיִם קִבְּלוּ אֶת תְּפִלַּת הַצַּדִּיק, וְהִסְכִּימוּ לָתֵת לַגְּבִיר מִבְחָן אַחֲרוֹן בְּהֶחְלֵט.

קָרָא הַבַּעַל שֵׁם טוֹב לְתַלְמִידוֹ, הָרַב הַקָּדוֹשׁ רַבִּי דָּוִד מִמִּיקוֹלַיֶיב זִיעֲ"א, וּבִקֵּשׁ מִמֶּנּוּ לָצֵאת מִיָּד לְבֵיתוֹ שֶׁל הַגְּבִיר. "סַע אֵלָיו," צִוָּה הַבַּעַל שֵׁם טוֹב, "וּפְקַח אֶת עֵינָיו בְּטֶרֶם יִהְיֶה מְאֻחָר מִדַּי!".

הַמִּפְגָּשׁ בָּאַרְמוֹן

רַבִּי דָּוִד מִמִּיקוֹלַיֶיב יָצָא לַדֶּרֶךְ וְהִגִּיעַ לָאַרְמוֹנוֹ שֶׁל הַגְּבִיר. עֵינָיו הַטְּהוֹרוֹת שֶׁל רַבִּי דָּוִד בָּחֲנוּ אֶת הַפְּאֵר שֶׁמִּסָּבִיב, אַךְ הוּא לֹא הִתְרַגֵּשׁ לִרְגַע, כִּי יָדַע שֶׁאֵין עֹשֶׁר אֲמִתִּי מִלְּבַד תּוֹרָה וּמַעֲשִׂים טוֹבִים.

הַגְּבִיר קִבֵּל אֶת כְּבוֹד הָרַב בִּכְבוֹד מְלָכִים! הוּא כִּבֵּד אוֹתוֹ לָשֶׁבֶת עַל כֻּרְסַת עֵץ מְגֻלֶּפֶת, וְהִגִּישׁ לוֹ תְּרוּמָה עֲנָקִית – צְרוֹר מָלֵא בְּמַטְבְּעוֹת נוֹצְצִים עֲבוּר תַּלְמִידֵי הַחֲכָמִים.

וְאָז, נִשְׁעַן הַגְּבִיר לְאָחוֹר, חִיֵּךְ חִיּוּךְ מָלֵא בְּגַאֲוָה וְאָמַר לָרַב אֶת הַמִּשְׁפָּט הַקָּבוּעַ שֶׁלּוֹ:

"כְּבוֹד הָרַב רוֹאֶה אֶת כָּל מַה שֶׁיֵּשׁ שֶׁיֵּשׁ לִי? הַאִם יִתָּכֵן בַּמְּצִיאוּת, שֶׁגְּבִיר עָצוּם כָּמוֹנִי, שֶׁהַכֶּסֶף שֶׁלּוֹ נִמְצָא בְּמֵאָה מְקוֹמוֹת שׁוֹנִים, יַגִּיעַ לְפַת לֶחֶם? הֲרֵי גַּם אִם יִשָּׂרֵף לִי מִפְעָל אֶחָד, הַשֵּׁנִי יַמְשִׁיךְ לְהַרְוִיחַ!"

רַבִּי דָּוִד הִבִּיט יָשָׁר אֶל תּוֹךְ עֵינָיו שֶׁל הֶעָשִׁיר, וּבְקוֹל רוֹעֵד אַךְ יַצִּיב עָנָה לוֹ:

"וַדַּאי שֶׁכֵּן! בְּלִי כָּל סָפֵק!"

הַגְּבִיר פָּעַר אֶת עֵינָיו בְּהֶלֶם. הוּא הִתְנַשֵּׁף בְּכַעַס. מֵעוֹלָם לֹא דִּבְּרוּ אֵלָיו כָּךְ! מֵעוֹלָם לֹא סָתְרוּ אֶת דְּבָרָיו!

"אֵיךְ כְּבוֹד הָרַב מֵעֵז לוֹמַר זֹאת?!" שָׁאַל בְּכַעַס הַסַּכָּנָה צְפוּנָה.

רַבִּי דָּוִד לֹא נִרְתַּע וְתֵאֵר לוֹ סְצֵנָה מְדֻיֶּקֶת, צְמַרְמֹרֶת עָבְרָה בְּגוּפוֹ שֶׁל כָּל מִי שֶׁיָּשַׁב בַּחֶדֶר: "שְׁמַע גְּבִיר יָקָר. הַשֵּׁם יִתְבָּרַךְ מַנְהִיג אֶת הַבְּרִיאָה בַּהַשְׁגָּחָה פְּרָטִית מְדֻיֶּקֶת. מָה יִקְרֶה אִם בְּרֶגַע אֶחָד, הַמַּלָּח הָרָאשִׁי שֶׁל אוֹנִיּוֹתֶיךָ יִרְדַּם וְכָל הַסְּפִינוֹת תִּטְבַּעְנָה בַּיָּם הַסּוֹעֵר? וּבְדִיּוּק בְּאוֹתוֹ הָרֶגַע, בַּמְּדִינָה שֶׁל מִפְעֲלֵי הָעֵצִים שֶׁלְּךָ, תִּפְרֹץ שְׂרֵפַת יַעַר עֲנָקִית שֶׁתִּשְׂרֹף הַכֹּל הַכֹּל הַכֹּל? וּבְדִיּוּק כְּשֶׁהַמִּכְתָּב עַל הָאָסוֹן יַגִּיעַ אֵלֶיךָ, יִפְרְצוּ שׁוֹדְדִים אֶל בֵּיתְךָ וְיִשְׁדְּדוּ אֶת הַכַּסֶּפֶת הָרָאשִׁית שֶׁלְּךָ? הַאִם גַּם אָז תֹּאמַר 'כֹּחִי וְעֹצֶם יָדִי'?".

פָּנָיו שֶׁל הַגְּבִיר אָמְדּוּ כְּלַפִּיד כּוֹעֵס. "דַּי לִי בְּדִבְרֵי הַהֶבֶל הָאֵלֶּה!" קָרָא הַגְּבִיר מִתּוֹךְ גַּאֲוָתוֹ שֶׁעִוְּרָה אֶת עֵינָיו, "צֵא מִבֵּיתִי עַכְשָׁו! שׁוּם שְׂרֵפָה וְשׁוּם סְעָרָה לֹא תַּפִּיל אוֹתִי מִנְּכָסַי!".

רַבִּי דָּוִד עָמַד בְּשֶׁקֶט, נֶאֱנַח אֲנָחָה עֲמֻקָּה שֶׁנָּגְעָה עַד הַשָּׁמַיִם, וְיָצָא אֶל הָרְחוֹב הַקַּר.

הַהִתְמוֹטְטוּת וְהַבֶּכִי הַגָּדוֹל

לֹא עָבְרוּ כַּמָּה דַּקּוֹת. הַגְּבִיר יָשַׁב בְּאַרְמוֹנוֹ הֶחָמִים, שׁוֹתֶה כּוֹס יַיִן יָקָר. וּפִתְאוֹם... טְרָאח!!!

דַּלְתוֹת הָעֵץ הַכְּבֵדוֹת נִפְתְּחוּ בִּסְעָרָה. אֶחָד מִמְּשָׁרְתָיו הַנֶּאֱמָנִים רָץ פְּנִימָה, כֻּלּוֹ רוֹעֵד מִפַּחַד, וּמִכְתָּב עִם שַׁעֲוָה אֲדֻמָּה בְּיָדוֹ. "אֲדוֹנִי! אֲדוֹנִי! בְּשׂוֹרַת אִיּוֹב!" שָׁאַג הַמְּשָׁרֵת. "קַבְּלָן הָעֵצִים שֶׁלָּנוּ בְּצַד הַשֵּׁנִי שֶׁל הַמְּדִינָה מְדַוֵּחַ עַל... שְׂרֵפַת חֲזִית עֲנָקִית! כָּל מִפְעֲלֵי הָעֵצִים נִשְׂרְפוּ כָּלִיל בְּרֶגַע זֶה ממש!".

הַגְּבִיר הֵחֵל מַרְגִּישׁ רְעִידָה קַלָּה בְּבִרְכָּיו. "לֹא נוֹרָא," מִלְמֵל לְעַצְמוֹ בִּשְׂפָתַיִם חִוְּרוֹת, "עוֹד יֵשׁ לִי סְפִינוֹת בַּיָּם...".

אַךְ בְּטֶרֶם סִיֵּם אֶת מִשְׁפָּטוֹ, שָׁלִיחַ שֵׁנִי נִכְנַס בְּרִיצָה, בְּגָדָיו רְטוּבִים מִגֶּשֶׁם מַיִם מְלוּחִים: "אֲדוֹנִי הַנִּכְבָּד! סְעָרָה עֲצוּמָה וּבִלְתִּי תִּבְעִית בַּיָּם הַגָּדוֹל! כָּל סְפִינוֹת מַשָּׂא שֶׁלְּךָ בָּלוּעוֹת בְּתוֹךְ הַמַּיִם הַזּוֹעֲמִים!"

לְפֶתַע, קוֹל פִּצּוּץ עַז נִשְׁמַע מֵחֶדֶר הַכַּסֶּפֶת. שׁוֹדְדִים חֲמוּשִׁים פָּרְצוּ לַבַּיִת, רַצּוּ לַכַּסֶּפֶת הָרָאשִׁית, רוֹקְנוּ אוֹתָהּ בִּשְׁנִיּוֹת, וְנֶעֶלְמוּ בַּחֲשֵׁכָה כְּאִלּוּ בָּלְעָה אוֹתָם הָאֲדָמָה!

הַכֹּל קָרָה בְּבַת אַחַת. בְּדִיּוּק כְּפִי שֶׁאָמַר הָרַב!

הַגְּבִיר נָפַל עַל בִּרְכָּיו עַל הַשָּׁטִיחַ הַיָּקָר. הַגַּאֲוָה שֶׁלּוֹ נִשְׁבְּרָה לִרְסִיסִים. הַנִּבְרֶשֶׁת נִרְאֲתָה לוֹ פִּתְאוֹם אֲפֵלָה וַחֲשׁוּכָה. דְּמָעוֹת גְּדוֹלוֹת, דְּמָעוֹת וְרוֹתְחוֹת שֶׁל תְּשׁוּבָה וַחֲרָטָה, חָנְקוּ אֶת גְּרוֹנוֹ.

"חָטָאתִי!" זָעַק מִקִּירוֹת לִבּוֹ לְאַבָּא שֶׁבַּשָּׁמַיִם. "חָטָאתִי שֶׁחָשַׁבְתִּי שֶׁאֲנִי וְרַק אֲנִי הֵבֵאתִי אֶת הַכֶּסֶף! רִבּוֹנוֹ שֶׁל עוֹלָם, מֵעַתָּה אֲנִי מֵבִין: אַתָּה זֶה שֶׁנּוֹתֵן, וְאַתָּה זֶה שֶׁיָּכוֹל לָקַחַת. הַכֹּל מִמְּךָ, הַכֹּל שֶׁלְּךָ, וַאֲנִי רַק בָּשָׂר וָדָם תָּלוּי בְּרַחֲמֶיךָ!"

הַיְשׁוּעָה וְהַחֲסָדִים

בְּאוֹתוֹ הָרֶגַע שֶׁבּוֹ בָּכָה הַגְּבִיר עִם דִּמְעוֹת הָאֱמֶת, נִפְרְצוּ שַׁעֲרֵי שָׁמַיִם. תְּפִלָּתוֹ שֶׁל הַבַּעַל שֵׁם טוֹב עָשְׂתָה אֶת פְּעֻלָּתָהּ. קִבְּלוּ בַּשָּׁמַיִם אֶת הַתְּשׁוּבָה שֶׁל הַגְּבִיר!

שָׁלִיחַ נוֹסָף, מְזִיעַ אַךְ מְחַיֵּךְ, הִגִּיעַ בְּרִיצָה לָאַרְמוֹן: "אֲדוֹנִי! נֵס וּפֶלֶא! רַק מִפְעָל אֶחָד קָטָן נִשְׂרַף, מַזֶּג הָאֲוִיר שִׁנָּה כִּווּן וְכִבָּה אֶת הָאֵשׁ מִיָּד! וְגַם בַּיָּם... סְפִינוֹתֶיךָ רַק נִסְחֲפוּ מְעַט לַחוֹף, וְהֵן נִצְּלוּ בְּנֵס!" וְלֹא מָחַר אֶלָּא דַּקּוֹת נוֹסָפוֹת עַד שֶׁשּׁוֹמְרֵי הָעִיר הִצְלִיחוּ לִתְפֹּס אֶת הַשּׁוֹדְדִים וְלַהֲשִׁיב אֶת מְלוֹא כַּסְפּוֹ הַשָּׁדוּד שֶׁל הַגְּבִיר.

הַגְּבִיר הֵבִין אֶת גֹּדֶל הַהַשְׁגָּחָה. מֵאוֹתוֹ הַיּוֹם וְהָלְאָה, הוּא חָזַר לִהְיוֹת אִישׁ שֶׁל חֶסֶד וּנְתִינָה, אֲבָל בְּשִׁנּוּי עָצוּם. אֶת צִלְצוּל מַטְבְּעוֹתָיו שׁוּב לֹא הִלְּוְתָה גַּאֲוָה, אֶלָּא שִׁפְלוּת רוּחַ, אֱמוּנָה טְהוֹרָה, וּתְפִלַּת הוֹדָיָה שְׁקֵטָה לְבוֹרֵא עוֹלָם.

א. שְׁאֵלַת פְּתִיחָה:

"יְלָדִים, הַאִם פַּעַם בְּנִיתֶם אַרְמוֹן עֲנָק בְּחוֹף הַיָּם, אוֹ מִגְדַּל קַפְּלָה עֲנָקִי, וִהְיִיתֶם בְּטוּחִים שֶׁשּׁוּם דָּבָר בָּעוֹלָם לֹא יָכוֹל לְהַפִּיל אוֹתוֹ... עַד שֶׁבָּאָה רוּחַ אוֹ גַּל קָטָן וְהַכֹּל נָפַל?

הַיּוֹם נִשְׁמַע עַל יְהוּדִי שֶׁבָּנָה אֶת 'מִגְדַּל הַכֶּסֶף' שֶׁלּוֹ, וְהָיָה בָּטוּחַ שֶׁבְּכֹחוֹ וּבְעֹצֶם יָדוֹ אַף אֶחָד מֵעוֹלָם לֹא יַפִּיל אוֹתוֹ. מָה קָרָה לוֹ? בּוֹאוּ נִשְׁמַע."

ב. הֶעָרוֹת בִּצּוּעַ וְסְטוֹרִיטֵלִינְג לַמְסַפֵּר:

ג. הַמְלָצוֹת לְמַצֶּגֶת (4 שְׁקוּפִיּוֹת צְנוּעוֹת):

ד. פְּעִילוּת מְסַכֶּמֶת ("הַכֹּל מַתָּנָה!"):

"ילדים, בואו נעמוד במעגל ונשחק במשחק 'הכל מתנה'. אני אעביר אליכם כדור. מי שתופס את הכדור, צריך להגיד דבר אחד שהוא טוב בו, או משהו טוב שיש לו, ולהוסיף 'וזה מתנה מהשם'.

למשל: 'אני רץ מהר – וזה מתנה מהשם!'. 'יש לי זיכרון טוב – וזה מתנה מהשם!'.

כך נתרגל להכיר תמיד במי שנותן לנו באמת את כל הטוב!"

מקורות: עלון "שבת טיש - סיפורי צדיקים", פרשת אחרי-קדושים, גיליון קנ"א (הובא במקורות כדפים [1]-[2]). מבוסס על הסיפור הידוע בשם מורנו הבעל שם טוב הקדוש זיע"א, דרך שליחו הרה"ק רבי דוד ממיקולייב זיע"א, שנועד לעורר אדם לאמונה ב"ה' מוריש ומעשיר" ולמנוע את גזירת איבוד העושר בגלל גאווה.

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “להלן עיבוד חינוכי, עשיר ומרתק על הבעל שם טוב הקדוש, תלמידו רבי דוד ממיקולייב, והעשיר הגאה , המעובד במיוחד לבנים בכיתות ב'-ד' (חינוך דתי-לאומי).” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “יֵשׁ לִי בָּתֵּי חֲרוֹשֶׁת לְעֵצִים מִצַּד אֶחָד שֶׁל הַמְּדִינָה, סְפִינוֹת מַשָּׂא בַּיָּם הַגָּדוֹל, וְשָׂדוֹת עֲנָקִיִּים שֶׁל תְּבוּאָה!”?
ג. כיצד הגיב רב למצב המתואר: “רַבִּי דָּוִד מִמִּיקוֹלַיֶיב יָצָא לַדֶּרֶךְ וְהִגִּיעַ לָאַרְמוֹנוֹ שֶׁל הַגְּבִיר.”?
ד. מה השתנה בין “שׁוּם שְׂרֵפָה וְשׁוּם סְעָרָה לֹא תַּפִּיל אוֹתִי מִנְּכָסַי!".” לבין “שׁוֹדְדִים חֲמוּשִׁים פָּרְצוּ לַבַּיִת, רַצּוּ לַכַּסֶּפֶת הָרָאשִׁית, רוֹקְנוּ אוֹתָהּ בִּשְׁנִיּוֹת, וְנֶעֶלְמוּ…”?
ה. כיצד הגיע הסיפור אל האירוע: “סְפִינוֹתֶיךָ רַק נִסְחֲפוּ מְעַט לַחוֹף, וְהֵן נִצְּלוּ בְּנֵס!" וְלֹא מָחַר אֶלָּא דַּקּוֹת נוֹסָפוֹת עַד שֶׁשּׁוֹמְרֵי הָעִיר הִצְלִיחוּ לִתְפֹּס אֶת…”?

3. מה היה ראשון?

כתבו ליד כל אירוע מספר מ־1 עד 5 לפי סדר התרחשותו בסיפור.

□ הַשֵּׁם יִתְבָּרַךְ מַנְהִיג אֶת הַבְּרִיאָה בַּהַשְׁגָּחָה פְּרָטִית מְדֻיֶּקֶת.
□ להלן עיבוד חינוכי, עשיר ומרתק על הבעל שם טוב הקדוש, תלמידו רבי דוד ממיקולייב, והעשיר הגאה , המעובד במיוחד לבנים בכיתות ב'-ד' (חינוך דתי-לאומי).

□ הַבַּעַל שֵׁם טוֹב, שֶׁעֵינָיו הַקְּדוֹשׁוֹת רָאוּ מִסּוֹף הָעוֹלָם וְעַד סוֹפוֹ, רָאָה מַה שֶׁנִּגְזַר עַל אוֹתוֹ גְּבִיר.
□ דְּמָעוֹת גְּדוֹלוֹת, דְּמָעוֹת וְרוֹתְחוֹת שֶׁל תְּשׁוּבָה וַחֲרָטָה, חָנְקוּ אֶת גְּרוֹנוֹ.

לפני ואחרי

מה קרה לפני: “הַשֵּׁם יִתְבָּרַךְ מַנְהִיג אֶת הַבְּרִיאָה בַּהַשְׁגָּחָה פְּרָטִית מְדֻיֶּקֶת.”?
מה קרה אחרי: “דְּמָעוֹת גְּדוֹלוֹת, דְּמָעוֹת וְרוֹתְחוֹת שֶׁל תְּשׁוּבָה וַחֲרָטָה, חָנְקוּ אֶת גְּרוֹנוֹ.”?

סיבה ותוצאה מתוך המעשה

האירוע הנתוןמה גרם לו?מה הייתה תוצאתו?
הַצַּדִּיק שֶׁאָהַב כָּל יְהוּדִי אֲהָבָה עֲצוּמָה, עָמַד בִּתְפִלָּה עֲמֻקָּה מִקִּירוֹת לִבּוֹ: "רִבּוֹנוֹ שֶׁל עוֹלָם!
קַבְּלָן הָעֵצִים שֶׁלָּנוּ בְּצַד הַשֵּׁנִי שֶׁל הַמְּדִינָה מְדַוֵּחַ עַל...

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “משפטו של עשיר גאה” על רב? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “הֲרֵי גַּם אִם יִשָּׂרֵף לִי מִפְעָל אֶחָד, הַשֵּׁנִי יַמְשִׁיךְ לְהַרְוִיחַ!”.
ג. אילו רב לא היה פועל כפי שמתואר ב־“שׁוֹדְדִים חֲמוּשִׁים פָּרְצוּ לַבַּיִת, רַצּוּ לַכַּסֶּפֶת הָרָאשִׁית, רוֹקְנוּ אוֹתָהּ בִּשְׁנִיּוֹת, וְנֶעֶלְמוּ בַּחֲשֵׁכָה…” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “סְפִינוֹתֶיךָ רַק נִסְחֲפוּ מְעַט לַחוֹף, וְהֵן נִצְּלוּ בְּנֵס!" וְלֹא מָחַר אֶלָּא דַּקּוֹת נוֹסָפוֹת עַד שֶׁשּׁוֹמְרֵי הָעִיר…” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. להלן עיבוד חינוכי, עשיר ומרתק על הבעל שם טוב הקדוש, תלמידו רבי דוד…
2. אֲנִי בָּנִיתִי אֶת חֲבֵרַת הַמִּסְחָר הַזּוֹ בְּחָכְמָתִי הָרַבָּה,…
3. עֵינָיו הַטְּהוֹרוֹת שֶׁל רַבִּי דָּוִד בָּחֲנוּ אֶת הַפְּאֵר…
4. אֶחָד מִמְּשָׁרְתָיו הַנֶּאֱמָנִים רָץ פְּנִימָה, כֻּלּוֹ רוֹעֵד…
5. הַכֹּל מִמְּךָ, הַכֹּל שֶׁלְּךָ, וַאֲנִי רַק בָּשָׂר וָדָם תָּלוּי…
6. מי שתופס את הכדור, צריך להגיד דבר אחד שהוא טוב בו, או משהו טוב שיש לו,…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “רַבִּי דָּוִד מִמִּיקוֹלַיֶיב יָצָא לַדֶּרֶךְ וְהִגִּיעַ לָאַרְמוֹנוֹ שֶׁל הַגְּבִיר.”.
מכתב: כתבו לרב בעקבות “קַבְּלָן הָעֵצִים שֶׁלָּנוּ בְּצַד הַשֵּׁנִי שֶׁל הַמְּדִינָה מְדַוֵּחַ עַל...”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “בְּאוֹתוֹ הָרֶגַע שֶׁבּוֹ בָּכָה הַגְּבִיר עִם דִּמְעוֹת הָאֱמֶת, נִפְרְצוּ שַׁעֲרֵי שָׁמַיִם.”.
דיווח: כתבו ידיעה קצרה שכותרתה “משפטו של עשיר גאה”.

המחזה

רגע מן הסיפורהדמויותמה יאמרו או יעשו?
הַצַּדִּיק שֶׁאָהַב כָּל יְהוּדִי אֲהָבָה עֲצוּמָה, עָמַד בִּתְפִלָּה עֲמֻקָּה מִקִּירוֹת לִבּוֹ:…
שׁוֹדְדִים חֲמוּשִׁים פָּרְצוּ לַבַּיִת, רַצּוּ לַכַּסֶּפֶת הָרָאשִׁית, רוֹקְנוּ אוֹתָהּ…

7. חיבור לחיים מתוך המעשה

א. מה אפשר ללמוד מן הדרך שבה רב פעל באירוע “שׁוּם שְׂרֵפָה וְשׁוּם סְעָרָה לֹא תַּפִּיל אוֹתִי מִנְּכָסַי!".”?
ב. תארו מצב מחיי הכיתה שבו אפשר ליישם את המסר של “משפטו של עשיר גאה”.
