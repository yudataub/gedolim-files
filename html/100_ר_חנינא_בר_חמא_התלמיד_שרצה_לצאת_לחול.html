התלמיד שרצה לצאת לחו"ל

דף עבודה בעקבות ר' חנינא בר חמא
שם: כיתה:

המעשה — הסיפור המלא

פְּתִיחָה: אֲבָנִים שֶׁל אַהֲבָה

יְלָדִים, אִם תֵּלְכוּ הַיּוֹם בִּשְׁבִילֵי אֶרֶץ יִשְׂרָאֵל, תַּרְגִּישׁוּ אֶת רֵיחַ הָאֲדָמָה הַטּוֹב, אֶת קַרְנֵי הַשֶּׁמֶשׁ הַמְחַמְּמוֹת, וְתִרְאוּ אֶת הַכְּבִישִׁים הַסְּלוּלִים. אֲבָל פַּעַם, לִפְנֵי שָׁנִים רַבּוֹת מְאֹד, בִּתְקוּפַת חֲכָמֵינוּ זִכְרוֹנָם לִבְרָכָה, הַדְּרָכִים הָיוּ מְלֵאוֹת בְּסְלָעִים כְּבֵדִים, בְּבוֹץ וּבְמִכְשׁוֹלִים.

בְּאוֹתָם יָמִים חַי בְּאֶרֶץ יִשְׂרָאֵל חָכָם עָצוּם וְקָדוֹשׁ מְאֹד. שְׁמוֹ הָיָה רַבִּי חֲנִינָא בַּר חָמָא .

רַבִּי חֲנִינָא הָיָה צַדִּיק מְפֻרְסָם. הוּא הָיָה מְרַפֵּא חוֹלִים, פּוֹסֵק הֲלָכוֹת לְכָל יִשְׂרָאֵל, וְאַלְפֵי תַּלְמִידִים רָעֲדוּ מִפָּנָיו בְּיִרְאַת כָּבוֹד. אֲבָל מִי שֶׁהָיָה מִסְתּוֹבֵב בִּדְרָכֶיהָ שֶׁל הָעִיר צִפּוֹרִי בִּשְׁעַת צָהֳרַיִם, הָיָה יָכוֹל לִרְאוֹת מַחֲזֶה פֶּלֶא: רַבִּי חֲנִינָא הַגָּדוֹל, שֶׁזְּקָנוֹ הַלָּבָן יוֹרֵד עַל חָזֵהוּ, עוֹמֵד תַּחַת הַשֶּׁמֶשׁ הַקּוֹפַחַת. טִפּוֹת שֶׁל זֵעָה נִגָּרוֹת עַל מִצְחוֹ. בְּיָדָיו הַקְּדוֹשׁוֹת הוּא תּוֹפֵס סְלָעִים כְּבֵדִים וַחֲדָדִים שֶׁהָיוּ זְרוּקִים בְּאֶמְצַע הַדֶּרֶךְ, וְסוֹחֵב אוֹתָם לַצַּד. הוּא הָיָה "מְתַקֵּן אֶת שְׁבִילֵי אַרְצֵנוּ".

מַדּוּעַ צַדִּיק כָּל כָּךְ גָּדוֹל עוֹבֵד כְּמוֹ פּוֹעֵל פָּשׁוּט? מִשּׁוּם שֶׁרַבִּי חֲנִינָא כָּל כָּךְ אָהַב אֶת הָאָרֶץ! הוּא לֹא רָצָה שֶׁשּׁוּם אָדָם יִתְקַל בְּאֶבֶן וְיֹאמַר חָלִילָה: "אוּף! אֵיזוֹ אֶרֶץ קָשָׁה מְלֵאָה אֲבָנִים...". הוּא רָצָה שֶׁכֻּלָּם יַרְגִּישׁוּ שֶׁאֶרֶץ יִשְׂרָאֵל הִיא תַּכְשִׁיט נוֹצֵץ מִיָּדָיו שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא.

הַמִּזְוָדָה בְּפֶתַח בֵּית הַמִּדְרָשׁ

יוֹם אֶחָד, הָיָה רַבִּי חֲנִינָא יוֹשֵׁב בְּבֵית הַמִּדְרָשׁ וְלוֹמֵד. קוֹל לִמּוּד הַתּוֹרָה הִדְהֵד בֵּין הַקִּירוֹת. פִּתְאוֹם, נִפְתְּחָה דֶּלֶת הָעֵץ הַכְּבֵדָה בַּחֲרִיקָה חַלָּשָׁה. בַּפֶּתַח עָמַד אֶחָד מִתַּלְמִידָיו הָאֲהוּבִים.

עַל גַּבּוֹ שֶׁל הַתַּלְמִיד הָיָה מֻנָּח תַּרְמִיל מָלֵא בִּבְגָדִים, בְּיָדוֹ מַקֵּל הֲלִיכָה, וְעַל רַגְלָיו נַעֲלַיִם מְכֻסּוֹת בְּאָבָק. זֶה הָיָה בָּרוּר לְכֻלָּם – הַתַּלְמִיד הַזֶּה יוֹצֵא לְמַסָּע רָחוֹק.

לְיָדוֹ עָמַד עוֹד אָדָם, יְהוּדִי יְקַר מִמִּשְׁפַּחַת כֹּהֲנִים, שֶׁגַּם הוּא אָרַז אֶת חֲפָצָיו וְהָיָה מוּכָן לָצֵאת לַדֶּרֶךְ.

הַתַּלְמִיד נִגַּשׁ אֶל שֻׁלְחָנוֹ שֶׁל רַבִּי חֲנִינָא, הִרְכִּין אֶת רֹאשׁוֹ וְלָחַשׁ הַדֶּרֶךְ אֶרֶץ: "שָׁלוֹם עָלֶיךָ כְּבוֹד רַבִּי. בָּאתִי לְבַקֵּשׁ אֶת בִּרְכַּת הַדֶּרֶךְ".

רַבִּי חֲנִינָא הֵרִים אֶת עֵינָיו הַטְּהוֹרוֹת. עֵינַיִם שֶׁרָאוּ כָּל כָּךְ הַרְבֵּה קְדֻשָּׁה בְּחַיֵּיהֶן. הוּא הִבִּיט בַּתַּרְמִיל שֶׁל הַתַּלְמִיד וְשָׁאַל בְּקוֹל עָמֹק: "לְאָן פָּנֶיךָ מְעוֹדוֹת, בְּנִי? וְלָמָּה תַּרְמִילְךָ עַל כְּתֵפֶיךָ?".

עָנָה הַתַּלְמִיד, וְלִבּוֹ מָלֵא בְּכַוָּנוֹת טוֹבוֹת: "רַבִּי, צָמֵא אֲנִי לְלִמּוּד הַתּוֹרָה! אֲנִי מַרְגִּישׁ שֶׁכָּאן בָּאָרֶץ חָסֵר לִי קְצָת לִמּוּד. שָׁמַעְתִּי שֶׁבְּאֶרֶץ בָּבֶל, בְּחוּץ לָאָרֶץ, יֵשׁ יְשִׁיבוֹת עֲנָקִיּוֹת. הֶחְלַטְתִּי לַעֲזֹב אֶת אֶרֶץ יִשְׂרָאֵל, לִנְסֹעַ לְבָבֶל, לִלְמֹד שָׁם תּוֹרָה בְּמֶשֶׁךְ זְמַן רַב, כְּדֵי שֶׁאֶגְדַּל וְאֶהְיֶה תַּלְמִיד חָכָם עָצוּם. רַבִּי... הַאִם תְּבָרֵךְ אוֹתִי שֶׁאֶצְלַח בְּלִמּוּדִי בְּבָבֶל?".

יְלָדִים, שִׂימוּ לֵב – הַתַּלְמִיד הַזֶּה לֹא רוֹצֶה לְטַיֵּל כְּדֵי לֵהָנוֹת, לְצַלֵּם אוֹ לֶאֱכֹל בְּמִסְעָדוֹת. הַתַּלְמִיד הַזֶּה רוֹצֶה לִלְמֹד תּוֹרָה! לִכְאוֹרָה, אֵין דָּבָר טוֹב מִזֶּה. דָּבָר שֶׁאֲפִלּוּ מֻתָּר לַעֲשׂוֹת לְפִי הַהֲלָכָה הַיְבֵשָׁה.

וּמָה רוֹצֶה הָאִישׁ הַשֵּׁנִי, הַכֹּהֵן?

גַּם הַכֹּהֵן נִגַּשׁ וְאָמַר: "רַבִּי, אָחִי נִפְטַר וּקְבוּר בְּחוּץ לָאָרֶץ. אַלְמְנָתוֹ מְחַכָּה שָׁם שֶׁאֶעֱשֶׂה לָהּ מִצְוַת 'חֲלִיצָה' כְּדֵי שֶׁתּוּכַל לְהִתְחַתֵּן מֵחָדָשׁ. זוֹ מִצְוָה שֶׁלֹּא אֶפְשָׁר לַעֲשׂוֹת פֹּה! אֲנִי חַיָּב לַעֲזֹב אֶת אֶרֶץ יִשְׂרָאֵל שֶׁכָּל כָּךְ יְקָרָה לִי, כְּדֵי לְקַיֵּם אֶת מִצְוַת הַתּוֹרָה בְּחוּץ לָאָרֶץ".

גַּם מַטְּרָתוֹ שֶׁל הַכֹּהֵן הִיא לְקַיֵּם מִצְוָה, דָּבָר קָדוֹשׁ וְחָשׁוּב.

שְּׁנֵיהֶם עָמְדוּ וְצִפּוּ לַחִיּוּךְ וְלִבְרָכָה מִפִּיו שֶׁל נְשִׂיא דּוֹרָם, רַבִּי חֲנִינָא בַּר חָמָא. אֲבָל בְּבֵית הַמִּדְרָשׁ הִשְׁתָּרְרָה פִּתְאוֹם דְּמָמָה מֻחְלֶטֶת. לֹא נִשְׁמַעֲתָה אַף מִלָּה.

סוֹד הַחִבּוּק הַגָּדוֹל

רַבִּי חֲנִינָא לֹא חִיֵּךְ. עֵינָיו נֶעֶצְמוּ לְרֶגַע, וּכְשֶׁפָּקַח אוֹתָן, דְּמָעוֹת צְלוּלוֹת זָלְגוּ עַל לְחָיָיו וְנָשְׁרוּ אֶל זְקָנוֹ הַלָּבָן. לִבּוֹ נִשְׁבַּר בְּקִרְבּוֹ. כָּל תַּלְמִידֵי הַיְשִׁיבָה הִבִּיטוּ פְּלִיאִים. הַאִם רַבִּי חֲנִינָא כּוֹעֵס? לָמָּה הוּא כָּל כָּךְ עָצוּב?

רַבִּי חֲנִינָא נָשַׁם נְשִׁימָה עֲמֻקָּה, וְקּוֹלוֹ רָעַד כְּשֶׁדִּבֵּר לַכֹּהֵן.

הוּא אָמַר לוֹ מִשְׁפָּט אֶחָד קָצָר, אֲבָל חָזָק כְּמוֹ רַעַם, מִשְׁפָּט שֶׁמַּסְבִּיר מָה הִיא בֶּאֱמֶת אֶרֶץ יִשְׂרָאֵל. כָּךְ הוּא אָמַר לוֹ:

"אָחִיךָ... עָזַב אֶת חֵיקוֹ שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא, וְחִבֵּק חֵיק שֶׁל גּוֹיִם בְּחוּץ לָאָרֶץ... וָמֵת. וְאַתָּה אָחִיו, מְבַקֵּשׁ לַעֲשׂוֹת כְּמוֹתוֹ?!"

הַדְּבָרִים הִכּוּ בַּתַּלְמִידִים וּבַכֹּהֵן כְּמוֹ הֲלֶם.

"חֵיקוֹ שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא!" יְלָדִים, "חֵיק" פֵּרוּשׁוֹ חִבּוּק!

רַבִּי חֲנִינָא לִמֵּד אוֹתָם אֶת הַסּוֹד הֲכִי עָמֹק שֶׁל אַרְצֵנוּ: אֶרֶץ יִשְׂרָאֵל הִיא הַחִבּוּק שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא עַצְמוֹ! כְּשֶׁיְּהוּדִי נִמְצָא בְּאֶרֶץ יִשְׂרָאֵל, הוּא לֹא סְתָם עוֹמֵד עַל חוֹל וַאֲבָנִים. הוּא עוֹמֵד בְּתוֹךְ הַיָּדַיִם שֶׁל אַבָּא שֶׁבַּשָּׁמַיִם. מִי שֶׁעוֹזֵב אֶת אֶרֶץ יִשְׂרָאֵל – גַּם אִם זֶה כְּדֵי לִלְמֹד הַרְבֵּה תּוֹרָה בְּחוּץ לָאָרֶץ, וְגַם אִם זֶה לְקַיֵּם מִצְוָה, הוּא עוֹזֵב אֶת הַחִבּוּק שֶׁל אֱלֹקִים! הַאִם מִישֶׁהוּ מֵהַיְלָדִים כָּאן הָיָה רוֹצֶה לִבְרֹחַ מֵהַחִבּוּק שֶׁל אַבָּא אוֹ אִמָּא שֶׁלּוֹ כְּדֵי לְשַׂחֵק בְּמַשֶּׁהוּ אַחֵר אֵצֶל שְׁכֵנִים זָרִים?

הַהַחְלָטָה הַגְּדוֹלָה

הַתַּלְמִיד עָמַד שָׁם, וְדִמְעָה זָלְגָה מֵעֵינוֹ. עַכְשָׁו הוּא הֵבִין אֶת מָה שֶׁרַבִּי חֲנִינָא עָשָׂה הַיּוֹם תַּחַת הַשֶּׁמֶשׁ. בִּגְלַל זֶה רַבִּי חֲנִינָא מֵרִים אֲבָנִים כְּבֵדוֹת וּמְסַדֵּר אֶת הַדֶּרֶךְ בְּמוֹ יָדָיו! כִּי אֵלּוּ לֹא סְתָם אֲבָנִים, הֵן הָאַרְמוֹן שֶׁל מֶלֶךְ מַלְכֵי הַמְּלָכִים.

הַתַּלְמִיד לָקַח אֶת תַּרְמִילוֹ הַכָּבֵד, אֲבָל... לֹא כְּדֵי לָצֵאת לַדֶּרֶךְ.

לְקּוֹל שִׂמְחָתָם שֶׁל כָּל תַּלְמִידֵי בֵּית הַמִּדְרָשׁ, הִנִּיחַ הַתַּלְמִיד אֶת תַּרְמִילוֹ עַל הָרִצְפָּה, הֵנִיחַ אֶת מַקְּלוֹ, הִתְיַשֵּׁב עַל סַפְסָל הָעֵץ תַּחַת מַבָּטוֹ הָאוֹהֵב וְהַמֵּאִיר שֶׁל רַבִּי חֲנִינָא, וְאָמַר:

"לֹא אוֹתִיר אֶת חֵיקוֹ שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא בְּעַד שׁוּם הֶשֵׁג בָּעוֹלָם. תּוֹרָה – אֶלְמַד רַק כָּאן, בְּאֶרֶץ הַקֹּדֶשׁ".

גַּם הַכֹּהֵן סָבַב עַל עֲקֵבָיו וְנִשְׁאַר בְּאֶרֶץ יִשְׂרָאֵל.

אוֹתוֹ יוֹם הֵבִינוּ תַּלְמִידֵי יִשְׂרָאֵל: לִקְרֹא עַל ה' סְפָרִים זֶה חָשׁוּב מְאֹד, אַךְ לָשֶׁבֶת בְּאֶרֶץ יִשְׂרָאֵל – זֶה מַמָּשׁ לָשֶׁבֶת בְּתוֹךְ זְרוֹעוֹתָיו.

מקורות: הסיפור המרכזי (היציאה לחו"ל של התלמיד והכהן): תלמוד ירושלמי, מסכת מועד קטן, פרק ג' הלכה א' ("התלמיד שרצה לצאת לחו"ל והכהן שרצה לצאת לחלוץ" - מופיע גם ב'ספר האגדה' שס"ו, כפי שמוזכר במקורות שסופקו `[1], [2]`). רקע האווירה - פינוי האבנים וחיבת הארץ של רבי חנינא: תלמוד ירושלמי, מסכת שביעית פרק ד' הלכה ז', וכן תלמוד בבלי, מסכת כתובות דף קי"ב עמוד א': "רבי חנינא מתקן מתקליה" (היה מתקן ומיישר את מכשולי ואבני הדרך בארץ ישראל מתוך חיבתה). מופיע במקורות שסופקו `[3]`.

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “יְלָדִים, אִם תֵּלְכוּ הַיּוֹם בִּשְׁבִילֵי אֶרֶץ יִשְׂרָאֵל, תַּרְגִּישׁוּ אֶת רֵיחַ הָאֲדָמָה הַטּוֹב, אֶת קַרְנֵי הַשֶּׁמֶשׁ הַמְחַמְּמוֹת, וְתִרְאוּ…” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “הוּא לֹא רָצָה שֶׁשּׁוּם אָדָם יִתְקַל בְּאֶבֶן וְיֹאמַר חָלִילָה: "אוּף!”?
ג. כיצד הגיב ר' חנינא בר חמא למצב המתואר: “עָנָה הַתַּלְמִיד, וְלִבּוֹ מָלֵא בְּכַוָּנוֹת טוֹבוֹת: "רַבִּי, צָמֵא אֲנִי לְלִמּוּד הַתּוֹרָה!”?
ד. מה השתנה בין “אֲנִי חַיָּב לַעֲזֹב אֶת אֶרֶץ יִשְׂרָאֵל שֶׁכָּל כָּךְ יְקָרָה לִי, כְּדֵי לְקַיֵּם אֶת מִצְוַת הַתּוֹרָה בְּחוּץ…” לבין “חֵיקוֹ שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא!" יְלָדִים, "חֵיק" פֵּרוּשׁוֹ חִבּוּק!”?
ה. כיצד הגיע הסיפור אל האירוע: “לְקּוֹל שִׂמְחָתָם שֶׁל כָּל תַּלְמִידֵי בֵּית הַמִּדְרָשׁ, הִנִּיחַ הַתַּלְמִיד אֶת תַּרְמִילוֹ עַל הָרִצְפָּה, הֵנִיחַ אֶת מַקְּלוֹ, הִתְיַשֵּׁב עַל…”?

סיבה ותוצאה מתוך המעשה

האירוע הנתוןמה גרם לו?מה הייתה תוצאתו?
עַל גַּבּוֹ שֶׁל הַתַּלְמִיד הָיָה מֻנָּח תַּרְמִיל מָלֵא בִּבְגָדִים, בְּיָדוֹ מַקֵּל הֲלִיכָה, וְעַל רַגְלָיו נַעֲלַיִם מְכֻסּוֹת בְּאָבָק.
עֵינָיו נֶעֶצְמוּ לְרֶגַע, וּכְשֶׁפָּקַח אוֹתָן, דְּמָעוֹת צְלוּלוֹת זָלְגוּ עַל לְחָיָיו וְנָשְׁרוּ אֶל זְקָנוֹ הַלָּבָן.

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “התלמיד שרצה לצאת לחו"ל” על ר' חנינא בר חמא? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “הֶחְלַטְתִּי לַעֲזֹב אֶת אֶרֶץ יִשְׂרָאֵל, לִנְסֹעַ לְבָבֶל, לִלְמֹד שָׁם תּוֹרָה בְּמֶשֶׁךְ זְמַן רַב, כְּדֵי שֶׁאֶגְדַּל וְאֶהְיֶה תַּלְמִיד חָכָם…”.
ג. אילו ר' חנינא בר חמא לא היה פועל כפי שמתואר ב־“חֵיקוֹ שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא!" יְלָדִים, "חֵיק" פֵּרוּשׁוֹ חִבּוּק!” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “לְקּוֹל שִׂמְחָתָם שֶׁל כָּל תַּלְמִידֵי בֵּית הַמִּדְרָשׁ, הִנִּיחַ הַתַּלְמִיד אֶת תַּרְמִילוֹ עַל הָרִצְפָּה, הֵנִיחַ אֶת…” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. יְלָדִים, אִם תֵּלְכוּ הַיּוֹם בִּשְׁבִילֵי אֶרֶץ יִשְׂרָאֵל,…
2. הוּא רָצָה שֶׁכֻּלָּם יַרְגִּישׁוּ שֶׁאֶרֶץ יִשְׂרָאֵל הִיא תַּכְשִׁיט…
3. אֲנִי מַרְגִּישׁ שֶׁכָּאן בָּאָרֶץ חָסֵר לִי קְצָת לִמּוּד.
4. שְּׁנֵיהֶם עָמְדוּ וְצִפּוּ לַחִיּוּךְ וְלִבְרָכָה מִפִּיו שֶׁל נְשִׂיא…
5. כְּשֶׁיְּהוּדִי נִמְצָא בְּאֶרֶץ יִשְׂרָאֵל, הוּא לֹא סְתָם עוֹמֵד עַל…
6. אוֹתוֹ יוֹם הֵבִינוּ תַּלְמִידֵי יִשְׂרָאֵל: לִקְרֹא עַל ה' סְפָרִים…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “עָנָה הַתַּלְמִיד, וְלִבּוֹ מָלֵא בְּכַוָּנוֹת טוֹבוֹת: "רַבִּי, צָמֵא אֲנִי לְלִמּוּד הַתּוֹרָה!”.
מכתב: כתבו לר' חנינא בר חמא בעקבות “עֵינָיו נֶעֶצְמוּ לְרֶגַע, וּכְשֶׁפָּקַח אוֹתָן, דְּמָעוֹת צְלוּלוֹת זָלְגוּ עַל לְחָיָיו וְנָשְׁרוּ אֶל זְקָנוֹ הַלָּבָן.”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “מִי שֶׁעוֹזֵב אֶת אֶרֶץ יִשְׂרָאֵל – גַּם אִם זֶה כְּדֵי לִלְמֹד הַרְבֵּה תּוֹרָה בְּחוּץ לָאָרֶץ, וְגַם אִם זֶה לְקַיֵּם…”.
דיווח: כתבו ידיעה קצרה שכותרתה “התלמיד שרצה לצאת לחו"ל”.
