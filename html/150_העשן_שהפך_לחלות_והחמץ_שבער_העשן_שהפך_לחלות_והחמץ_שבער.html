העשן שהפך לחלות והחמץ שבער

דף עבודה בעקבות העשן שהפך לחלות והחמץ שבער
שם: כיתה:

המעשה — הסיפור המלא

הסיפור מנוקד במלואו כנדרש.

🔹 שָׁלָב 1: בֵּרוּר מַקְדִּים וְנִתּוּחַ (לַמּוֹרֶה/לַמְסַפֵּר)

1. הָעֹנִי מֵבִיא לְכָךְ שֶׁאֵין קֶמַח לֶאֱפוֹת חַלּוֹת לְשַׁבָּת. הַשְּׁכֵנָה רוֹצָה לְבַיֵּשׁ וְלִלְעֹג, אַךְ אֵשֶׁת רַבִּי חֲנִינָא בּוֹחֶרֶת לֹא לְבַקֵּשׁ טוֹבוֹת וְרַחֲמִים מֵאַף אֶחָד, וּמַסִּיקָה עֵצִים שֶׁיַּעֲלוּ עָשָׁן וְיִשְׁמְרוּ עַל כְּבוֹדָהּ.

2. הַבַּת מִתְבַּלְבֶּלֶת וְשׁוֹפֶכֶת חֹמֶץ בִּמְקוֹם שֶׁמֶן לְנֵרוֹת הַשַּׁבָּת. עֶרֶב שַׁבָּת יורד, וְאֵין אוֹר.

1. הַשְּׁכֵנָה פּוֹתַחַת אֶת הַתַּנּוּר – וְגַל שֶׁל נִיחוֹחַ חַלּוֹת טְרִיּוֹת מַכֶּה בָּהּ! הַתַּנּוּר מָלֵא בְּנֵס.

2. רַבִּי חֲנִינָא מַרְגִּיעַ אֶת בִּתּוֹ בְּמִשְׁפַּט אֱמוּנָה מַדְהִים: "מִי שֶׁאָמַר לַשֶּׁמֶן וְיִדְלַק, יֹאמַר לַחֹמֶץ וְיִדְלַק!".

🔹 שָׁלָב 2: הַסִּפּוּר הַמָּלֵא – "הֶעָשָׁן שֶׁהָפַךְ לְחַלּוֹת וְהַחֹמֶץ שֶׁבָּעַר"

פְּתִיחָה: תַּנּוּר רֵיק מִקֶּמַח, מָלֵא בְּכָבוֹד יְלָדִים, אֲנַחְנוּ כְּבָר יוֹדְעִים שֶׁבַּבַּיִת שֶׁל רַבִּי חֲנִינָא בֶּן דּוֹסָא לֹא הָיָה שֶׁפַע שֶׁל מַאֲכָלִים וּמַמְתַּקִּים. זָכוּר לָנוּ שֶׁרַבִּי חֲנִינָא הָיָה "מִסְתַּפֵּק בְּקַב שֶׁל חֲרוּבִים". אַךְ יֵשׁ דָּבָר אֶחָד שֶׁהָיָה הָכִי חָשׁוּב לְרַבִּי חֲנִינָא וּלְאִשְׁתּוֹ הַצַּדֶּקֶת, אַחֲרֵי אַהֲבַת ה': הַכָּבוֹד הָעַצְמִי שֶׁלָּהֶם. הֵם סָלְדוּ מִמַּתְּנוֹת בָּשָׂר וָדָם. "אֲנַחְנוּ בּוֹטְחִים רַק בַּה'!" אָמְרָה הָרַבָּנִית, אֵשֶׁת רַבִּי חֲנִינָא. "אֲנַחְנוּ לְעוֹלָם לֹא נִתְמַסְכֵּן, לֹא נִבְכֶּה לַשְּׁכֵנִים, וְלֹא נְבַקֵּשׁ רַחֲמִים אוֹ שְׁאֵרִיּוֹת שֶׁל אֹכֶל מֵאַף אֶחָד!".

כָּל יוֹם שִׁשִּׁי, לִקְרַאת שַׁבָּת, כָּל הַנָּשִׁים בַּשְּׁכוּנָה הָיוּ לָשׁוֹת בָּצֵק. רֵיחַ שֶׁל שְׁמָרִים וְחַלּוֹת מְתוּקוֹת נָדַף מִכָּל תַּנּוּרֵי הָאֶבֶן. רַק הַתַּנּוּר שֶׁל מִשְׁפַּחַת בֶּן דּוֹסָא הָיָה קַר וְרֵיק. קֶמַח – לֹא הָיָה לָהֶם. אֲבָל כְּדֵי שֶׁהַשְּׁכֵנוֹת לֹא יַבְחִינוּ בָּעֹנִי הַנּוֹרָא וְלֹא יָבוֹאוּ לְהַצִּיעַ נְדָבוֹת שֶׁל רַחֲמִים, מָה עָשְׂתָה אֵשֶׁת רַבִּי חֲנִינָא? הִיא הִכְנִיסָה לְתוֹךְ הַתַּנּוּר עֵצִים קְטַנִּים וַחֲתִיכוֹת עֵשֶׂב יָבֵשׁ, וְהִדְלִיקָה אֶת הָאֵשׁ! עָשָׁן אֲפַרְפַּר יָצָא מֵהָאֲרֻבָּה שֶׁל הַבַּיִת, וְהִתְעוֹפֵף כְּלַפֵּי הַשָּׁמַיִם. אָמְרוּ הַשְּׁכֵנִים: "אָה! תִּרְאוּ! גַּם אֵשֶׁת רַבִּי חֲנִינָא אוֹפָה לִכְבוֹד שַׁבָּת הַמַּלְכָּה! אֵין עָשָׁן בְּלִי אַפִּיָּה!". הָרַבָּנִית חִיְּכָה לְעַצְמָהּ. כְּבוֹדָהּ נִשְׁמַר.

הַשְּׁכֵנָה הָרָעָה וְהַפֶּלֶא הַמָּתוֹק אַךְ כְּמוֹ תָּמִיד, הָיְתָה בַּשְּׁכוּנָה אִשָּׁה אַחַת שֶׁעֵינָהּ צָרָה. שְׁכֵנָה שֶׁלֹּא אָהֲבָה אֶת הַשֶּׁקֶט שֶׁל הַצַּדִּיקִים. "אֲנִי יוֹדַעַת שֶׁאֵין לָהֶם כְּלוּם!" לָחֲשָׁה הַשְּׁכֵנָה בְּרֹעַ לַחֲבֵרוֹתֶיהָ, "אֵין לָהֶם אֲפִלּוּ קֶמַח שֶׁל נְמָלִים בַּבַּיִת! אֲנִי הוֹלֶכֶת לַעֲשׂוֹת לָהּ בּוּשׁוֹת. אֲנִי אַדְפִּיק בַּדֶּלֶת, אֶפְתַּח לָהּ אֶת הַתַּנּוּר פִּתְאוֹם, וְכֻלָּם יִרְאוּ שֶׁאֵין שָׁם כְּלוּם, סְתָם עָשָׁן שֶׁל שֶׁקֶר!".

אוֹתָהּ שְׁכֵנָה רָצָה, דָּפְקָה בְּחָזְקָה עַל דַּלְתָּהּ שֶׁל אֵשֶׁת רַבִּי חֲנִינָא וְנִכְנְסָה מַהֵר הַפְּנִימָה בְּלִי לְחַכּוֹת. הָרַבָּנִית הַצַּדֶּקֶת, שֶׁלֹּא רָצְתָה לַחֲווֹת אֶת הַבּוּשָׁה עַל כָּךְ שֶׁהִיא מְזַיֶּפֶת אַפִּיָּה, רָצָה וְהִתְחַבְּאָה הֵיטֵב בְּתוֹךְ הַחֶדֶר הַפְּנִימִי, לִבָּהּ דּוֹפֵק בְּיִרְאָה. "הַשֵּׁם יִתְבָּרַךְ," לָחֲשָׁה, "אֲנִי לֹא בִּקַּשְׁתִּי מֵאַף אֶחָד כְּלוּם. אָנָּא, אַל תִּתֵּן לִי לְהִתְבַּיֵּשׁ!".

וְהַקָּדוֹשׁ בָּרוּךְ הוּא, יְלָדִים... אוֹהֵב אֶת הַצַּדִּיקִים! הַשְּׁכֵנָה נִגְּשָׁה לַתַּנּוּר, פָּתְחָה אֶת דֶּלֶת הַבַּרְזֶל שֶׁלּוֹ לִרְוָחָה... וּפִתְאוֹם!!! קָפְצָה צַעַד אֶחָד אָחוֹרָה בְּהֶלֶם. בַּחוּץ לֹא יָצָא רַק עָשָׁן. יָצָא גַּל מְטֹרָף שֶׁל חֹם וְנִיחוֹחַ שֶׁל הַחַלּוֹת הֲכִי טוֹבוֹת שֶׁהֵרִיחָה בִּימֵי חַיֶּיהָ! הַתַּנּוּר הָיָה מָלֵא עַד אֶפֶס מָקוֹם בְּחַלּוֹת תּוֹפְחוֹת, זְהוּבוֹת וְרֵיחָנִיּוֹת, שֶׁנֶּאֱפוּ מַמָּשׁ בְּנֵס גָּלוּי מִן הַשָּׁמַיִם! וּבַקְּעָרָה שֶׁלְּיַד הַתַּנּוּר הָיָה בָּצֵק מָלֵא שֶׁכִּמְעַט גָּלַשׁ לָרִצְפָּה!

"הָבִיאִי מַהֵר רָדַת!" (כַּף אַפִּיָּה מעץ) זָעֲקָה הַשְּׁכֵנָה הַנִּדְהֶמֶת אֶל אֵשֶׁת רַבִּי חֲנִינָא, "הַחַלּוֹת שֶׁלָּךְ נִשְׂרָפוֹת מֵרֹב שֶׁהֵן מְלֵאוֹת כָּאן!". אֵשֶׁת רַבִּי חֲנִינָא חִיְּכָה, יָצְאָה בְּנַחַת עִם כַּף הָעֵץ וְהוֹצִיאָה אֶת הַחַלּוֹת לִכְבוֹד שַׁבָּת קֹדֶשׁ. הִיא לֹא הֻפְתְּעָה כְּמוֹ הַשְּׁכֵנָה, כִּי יָדְעָה מִיָּד – כְּשֶׁיְּהוּדִי שׁוֹמֵר עַל הַכָּבוֹד וְלֹא מִתְחַנֵּן לְמַתְּנוֹת אֲנָשִׁים, בּוֹרֵא עוֹלָם שׁוֹלֵחַ לוֹ מַתָּנָה יְשִׁירוֹת מִכִּסֵּא הַכָּבוֹד!

הַטָּעוּת שֶׁל הַבַּת: מַיִם רָעִים מֵרֵיחַ? הַנִּסִּים בְּבֵיתוֹ שֶׁל רַבִּי חֲנִינָא הָיוּ מַדְהִימִים, וְרַבִּי חֲנִינָא הָיָה חַי בֶּאֱמוּנָה כָּל כָּךְ טִבְעִית, שֶׁגַּם כְּשֶׁהָיָה נִדְמֶה שֶׁקּוֹרֶה אָסוֹן קָטָן, הַחִיּוּךְ לֹא עָזַב אֶת שְׂפָתָיו. כָּךְ קָרָה עֶרֶב שַׁבָּת אֶחָד, כַּאֲשֶׁר רַבִּי חֲנִינָא חָזַר מִבֵּית הַמִּדְרָשׁ וְרָאָה אֶת בִּתּוֹ עֲצוּבָה. הַיַּלְדָּה עָמְדָה בַּפִּנָּה וְהַדְּמָעוֹת זוֹלְגוֹת מֵעֵינֶיהָ. הַשַּׁבָּת עוֹמֶדֶת לְהִכָּנֵס, אַךְ פְּנֵי הַיַּלְדָּה חֲשׁוּכוֹת.

"בִּתִּי הַיְּקָרָה, מָה לָךְ שֶׁאַתְּ עֲצוּבָה בְּיוֹם קָדוֹשׁ זֶה?" לִטֵּף רַבִּי חֲנִינָא אֶת רֹאשָׁהּ בְּרַכּוּת. "אוֹי, אַבָּא!" יִבְּבָה הַבַּת. "אֵיזוֹ טָעוּת עָשִׂיתִי! בָּאתִי לְהָכִין אֶת נֵרוֹת הַשַּׁבָּת שֶׁלָּנוּ בַּקְּעָרוֹת (כְּדֵי שֶׁיָּאִירוּ כָּל הַלַּיְלָה). בַּחֹשֶׁךְ שֶׁל הַמְּזָוֶה הִתְבַּלְבַּלְתִּי בֵּין הַכַּדִּים... בִּמְקוֹם לִמְזֹג לַקְּעָרוֹת שֶׁמֶן-זַיִת מֵאִיר – מָזַגְתִּי תּוֹכָן חֹמֶץ חָמוּץ! עַכְשָׁו נִכְנְסָה שַׁבָּת, אָסוּר לְתַקֵּן, וְהַפְּתִילוֹת בְּתוֹךְ הַחֹמֶץ יִכְבּוּ מִיָּד! יִהְיֶה לָנוּ חֹשֶׁךְ כָּל שַׁבָּת הַמַּלְכָּה!".

הַבַּת בָּכְתָה כִּי יָדְעָה שֶׁחֹמֶץ (שֶׁהוּא כְּמוֹ יַיִן אוֹ מַיִם חֲמוּצִים) מְכַבֶּה אֵשׁ וְלֹא מְדַלֵּק אוֹתָהּ. חֹמֶץ אֵינוֹ נַפְט וְלֹא שֶׁמֶן. זֶה אֲסוֹן מֻחְלָט לַשַּׁבָּת.

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. הָעֹנִי מֵבִיא לְכָךְ שֶׁאֵין קֶמַח לֶאֱפוֹת חַלּוֹת לְשַׁבָּת.
2. אֲנַחְנוּ לְעוֹלָם לֹא נִתְמַסְכֵּן, לֹא נִבְכֶּה לַשְּׁכֵנִים, וְלֹא…
3. אוֹתָהּ שְׁכֵנָה רָצָה, דָּפְקָה בְּחָזְקָה עַל דַּלְתָּהּ שֶׁל אֵשֶׁת…
4. הִיא לֹא הֻפְתְּעָה כְּמוֹ הַשְּׁכֵנָה, כִּי יָדְעָה מִיָּד –…
5. רַבִּי חֲנִינָא הֵבִין שֶׁכָּל חֻקֵּי הַטֶּבַע שַׁיָּכִים לַה', וְאִם…
6. 📚 מְקוֹרוֹת (מִתּוֹךְ מַחְבֶּרֶת גְּדוֹלֵי יִשְׂרָאֵל שֶׁל…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “אֲנִי יוֹדַעַת שֶׁאֵין לָהֶם כְּלוּם!" לָחֲשָׁה הַשְּׁכֵנָה בְּרֹעַ לַחֲבֵרוֹתֶיהָ, "אֵין לָהֶם אֲפִלּוּ קֶמַח שֶׁל נְמָלִים…”.
מכתב: כתבו להעשן שהפך לחלות והחמץ שבער בעקבות “כָּךְ קָרָה עֶרֶב שַׁבָּת אֶחָד, כַּאֲשֶׁר רַבִּי חֲנִינָא חָזַר מִבֵּית הַמִּדְרָשׁ וְרָאָה אֶת בִּתּוֹ עֲצוּבָה.”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “הַבַּת הִדְלִיקָה בְּזְהִירוּת אֶת קְצֵה הַפְּתִילָה הַסְּפוּגָה בַּחֹמֶץ הֶחָמוּץ...”.
דיווח: כתבו ידיעה קצרה שכותרתה “העשן שהפך לחלות והחמץ שבער”.

המחזה

רגע מן הסיפורהדמויותמה יאמרו או יעשו?
רַק הַתַּנּוּר שֶׁל מִשְׁפַּחַת בֶּן דּוֹסָא הָיָה קַר וְרֵיק.
מִי שֶׁאָמַר לַשֶּׁמֶן וְיִדְלַק הִבִּיט רַבִּי חֲנִינָא בְּבִתּוֹ, הַנָּשִׂיא שֶׁל הָאֱמוּנָה.

7. חיבור לחיים מתוך המעשה

א. מה אפשר ללמוד מן הדרך שבה העשן שהפך לחלות והחמץ שבער פעל באירוע “הָבִיאִי מַהֵר רָדַת!" (כַּף אַפִּיָּה מעץ) זָעֲקָה הַשְּׁכֵנָה הַנִּדְהֶמֶת אֶל אֵשֶׁת רַבִּי חֲנִינָא, "הַחַלּוֹת שֶׁלָּךְ…”?
ב. תארו מצב מחיי הכיתה שבו אפשר ליישם את המסר של “העשן שהפך לחלות והחמץ שבער”.
