יסוריו של רבי — צער בעלי חיים

דף עבודה בעקבות רבי יהודה הנשיא (רבי)
שם: כיתה:

המעשה — הסיפור המלא

להלן עיבוד חינוכי, עשיר, מרתק ומרגש לסיפורו של רבי יהודה הנשיא (רבינו הקדוש) – יסוריו ורחמנותו על בעלי החיים . הסיפור נכתב בדיוק לפי ההנחיות עבור בנים בכיתות ב'-ד' (חינוך דתי-לאומי), תוך שימוש בעקרונות סטוריטלינג, תיאורים חושיים (מראה, שמיעה, ריח, תחושה), הקפדה עליונה על כבוד התורה, והדגשת ההשגחה הפרטית.

הסיפור מנוקד במלואו כנדרש.

פְּתִיחָה: רֵיחָהּ שֶׁל הַתּוֹרָה וּגְדֻלַּת הַנָּשִׂיא

יְלָדִים, אִם הֱיִיתֶם נִכְנָסִים לְבֵיתוֹ שֶׁל רַבִּי יְהוּדָה הַנָּשִׂיא בָּעִיר צִפּוֹרִי אוֹ בְּבֵית שְׁעָרִים לִפְנֵי כְּאַלְפַּיִם שָׁנָה, הֱיִיתֶם רוֹאִים מַרְאֶה נִפְלָא שֶׁמְּשַׁלֵּב קֹדֶשׁ וּמַלְכוּת.

מִבַּחוּץ, הַבַּיִת נִרְאָה כְּמוֹ אַרְמוֹן. שׁוֹמְרִים חֲסוֹנִים עָמְדוּ בַּפֶּתַח, שֻׁלְחָנוֹת אֲרֻכִּים הָיוּ מְלֵאִים בִּירָקוֹת אֲפִלּוּ בְּשִׂיא הַחֹרֶף, וְהַקֵּיסָר הָרוֹמָאִי בְּעַצְמוֹ הָיָה מַגִּיעַ לְבַקֵּר.

אֲבָל מִבִּפְנִים? מִבִּפְנִים עָמַד בָּאֲוִיר רֵיחַ עָמֹק וּמָתוֹק שֶׁל קְלָפִים יְשָׁנִים וּדְיוֹ טָרִי. קוֹלוֹת נְעִימִים שֶׁל מֵאוֹת תַּלְמִידֵי חֲכָמִים הִתְנַגְּנוּ בַּחֲדָרִים: "אָמַר רַבִּי יְהוּדָה... אָמַר רַבִּי מֵאִיר...". בָּאֶמְצַע יָשַׁב רַבִּי יְהוּדָה הַנָּשִׂיא, שֶׁכֻּלָּם קָרְאוּ לוֹ פָּשׁוּט "רַבִּי" אוֹ "רַבֵּנוּ הַקָּדוֹשׁ". הוּא יָשַׁב וְכָתַב אֶת "הַמִּשְׁנָה" – הַסֵּפֶר שֶׁהִצִּיל אֶת הַתּוֹרָה שֶׁבְּעַל-פֶּה לְנֶצַח.

רַבִּי הָיָה צַדִּיק יְסוֹד עוֹלָם. הַקָּדוֹשׁ בָּרוּךְ הוּא אָהַב אוֹתוֹ אַהֲבָה עֲצוּמָה. וּבְדִיּוּק בִּגְלַל שֶׁה' כָּל כָּךְ אוֹהֵב צַדִּיקִים, הוּא מְדַקְדֵּק אִתָּם עַל כָּל תְּנוּעָה קְטַנָּה, מַמָּשׁ "כְּחוּט הַשַּׂעֲרָה".

הַהִמָּלְטוּת בַּשּׁוּק וְהַבַּקָּשָׁה הָאִלֶּמֶת

יוֹם אֶחָד, הָלַךְ רַבִּי יְהוּדָה הַנָּשִׂיא בְּרַחֲבוֹת הַשּׁוּק. קוֹל הֲמֻלַּת הַשּׁוּק הָיָה חָזָק: צַעֲקוֹת הַסּוֹחֲרִים, נַהֲמַת עֲגָלוֹת הָעֵץ עַל שְׁבִילֵי הָאֶבֶן, וְרֵיחַ עַז שֶׁל תַּבְלִינִים וְתַבּוּאָה מִלֵּא אֶת הָאֲוִיר.

בְּאוֹתָהּ שָׁעָה, הוֹבִיל שׁוֹחֵט אֶחָד עֵגֶל צָעִיר בַּחֶבֶל. הָעֵגֶל הָיָה מְבֹהָל. הוּא פָּעָה קוֹלוֹת שֶׁל בֶּכִי: "מּוּ... מוּוּוּ...", רַגְלָיו בָּעֲטּוּ בָּאֲדָמָה, וְהוּא נִסָּה לְהִתְנַגֵּד וְלִמְשֹׁךְ אֶת עַצְמוֹ לְאָחוֹר.

וְהִנֵּה, מֵרֹב מַאֲמָץ, חָמַק הָעֵגֶל מִיָּדוֹ שֶׁל הַקַּצָּב. הוּא רָץ בְּבֶהָלָה לְכָל כִּוּוּן, עַד שֶׁרָאָה אֶת רַבִּי יְהוּדָה הַנָּשִׂיא, מַנְהִיג יִשְׂרָאֵל הָרַחוּם, עָטוּף בִּגְלִימָה אֲרֻכָּה וּמְכֻבֶּדֶת.

הָעֵגֶל הַמִּסְכֵּן רָץ יָשָׁר אֶל רַבִּי! הוּא תָּחַב אֶת רֹאשׁוֹ מִתַּחַת לַגְּלִימָה הַגְּדוֹלָה שֶׁל רַבִּי, נִצְמַד אֶל רַגְלָיו שֶׁל הַצַּדִּיק, וְהִתְחִיל לִבְכּוֹת. דְּמָעוֹת גְּדוֹלוֹת וּרְטֻבּוֹת זָלְגוּ מֵעֵינָיו שֶׁל בַּעַל הַחַיִּים. זֶה הָיָה כְּאִלּוּ הוּא אוֹמֵר: "הַצֵּל אוֹתִי! רַחֵם עָלַי! אֲנִי מְפַחֵד!". רַבִּי מַמָּשׁ הִרְגִּישׁ אֶת מַגַּע הַפַּרְוָה הַחַמָּה וְהָרוֹעֶדֶת שֶׁל הָעֵגֶל נִצְמֶדֶת אֵלָיו.

בְּאוֹתָם יָמִים, הָיָה נָהוּג שֶׁעֲגָלִים נוֹעֲדוּ בְּעִקָּר לַחֲרִישָׁה בַּשָּׂדוֹת כְּדֵי לְהָבִיא תְּבוּאָה. אַךְ בְּסוֹף יְמֵיהֶם, מֻתָּר הָיָה עַל פִּי דִּין תּוֹרָה לִשְׁחֹט אוֹתָם כַּהֲלָכָה כְּדֵי לְהַאֲכִיל אֶת הָאָדָם. רַבִּי הִסְתַּכֵּל עַל הָעֵגֶל. הוּא יָדַע שֶׁהַקַּצָּב אֵינוֹ עוֹשֶׂה עֲבֵרָה, אֶלָּא מְכִין בָּשָׂר לִסְעוּדוֹת מִצְוָה שֶׁל עַם יִשְׂרָאֵל. לָכֵן, רַבִּי לֹא עָטַף אֶת הָעֵגֶל בְּיָדָיו. תַּחַת זֹאת, הוּא חָלַץ אֶת הָעֵגֶל מִתּוֹךְ גְּלִימָתוֹ, הִבִּיט בּוֹ וְאָמַר בְּקוֹל שָׁקֵט:

"לֵךְ! לְכָךְ נוֹצַרְתָּ!" (כְּלוֹמַר – לְמַטָּרָה זוֹ נִבְרֵאתָ בָּעוֹלָם, כְּדֵי לְשַׁמֵּשׁ אֶת הָאָדָם לְמָזוֹן אוֹ לַקָּרְבָּנוֹת).

הַגְּזֵרָה הַשְּׁמֵימִית וְיִסּוּרֵי הַנָּשִׂיא

הָעֵגֶל נִלְקַח בַּחֲזָרָה לְיָדָיו שֶׁל הַקַּצָּב.

אֲבָל בַּשָּׁמַיִם? בְּבֵית דִּין שֶׁל מַעְלָה עָמְדוּ וְהִסְתַּכְּלוּ בַּמַּחֲזֶה. מַלְאֲכֵי הַשָּׁרֵת אָמְרוּ: אָמְנָם רַבִּי יְהוּדָה הַנָּשִׂיא אָמַר הֲלָכָה פְּשׁוּטָה וּנְכוֹנָה, הוּא לֹא גָּזַל וְלֹא מָעַל. מֻתָּר לֶאֱכֹל בָּשָׂר. אֲבָל... הֵיכָן הָרַחֲמִים הַגְּדוֹלִים שֶׁל נְשִׂיא יִשְׂרָאֵל?! יְצוּר חַי רָץ אֵלֶיךָ, בּוֹכֶה וּמִתְחַנֵּן לְעֶזְרָה – הַאִם לֹא יָכֹלְתָּ לְלַטֵּף אוֹתוֹ? לַעֲזֹר לוֹ? לְבַקֵּשׁ שֶׁיָּנִיחוּ לוֹ וְלִפְדּוֹת אוֹתוֹ מֵהַקַּצָּב?! מֵאָדָם עֲנָק כָּל כָּךְ ה' דּוֹרֵשׁ רַחֲמָנוּת וְהַקְפָּדָה עַל צַעַר בַּעֲלֵי חַיִּים לִפְנִים מִשּׁוּרַת הַדִּין!

מִיָּד יָצְאָה בַּת-קוֹל (קוֹל שְׁמֵימִי) וְהִכְרִיזָה:

"הוֹאִיל וְלֹא רִחֵם – שֶׁיָּבוֹאוּ עָלָיו יִסּוּרִים!"

וּמֵאוֹתוֹ יוֹם, חַיָּיו שֶׁל רַבִּי הִשְׁתַּנּוּ. כְּאֵבִים אֲיֻמִּים תָּקְפוּ אֶת גּוּפוֹ הַקָּדוֹשׁ. שִׁנָּיו הֵחֵלּוּ לִכְאֹב צְמַרְמֹרֶת מְיַסֶּרֶת. שָׁנָה אַחַר שָׁנָה הָיָה רַבִּי מִתְפַּתֵּל מִכְּאֵב, עַד שֶׁקּוֹלוֹת הָאֲנָחָה שֶׁלּוֹ נִשְׁמְעוּ מֵרָחוֹק. תַּלְמִידָיו בָּכוּ עִמּוֹ, כָּל עַם יִשְׂרָאֵל הִתְפַּלֵּל לַהֲשָׁבַת בְּרִיאוּתוֹ, אַךְ הַשָּׁמַיִם הָיוּ נְעוּלִים. רַבִּי לָמַד תּוֹרָה מִתּוֹךְ סֵבֶל נוֹרָא, מַמָּשׁ שְׁלוֹשׁ עֶשְׂרֵה שָׁנִים רְצוּפוֹת! הַגּוּף כָּאַב, אַךְ הַנְּשָׁמָה נִשְׁאֲרָה טְהוֹרָה וּזְכוּכָה, מְמַתֶּנֶת לְרַחֲמֵי שָׁמַיִם.

הַמַּטְאֲטֵא, הַגּוּרִים וְתִקּוּן הַלֵּב

עָבְרוּ שְׁלוֹשׁ עֶשְׂרֵה שָׁנָה. בֹּקֶר אֶחָד, יָשַׁב רַבִּי יְהוּדָה הַנָּשִׂיא עַל כִּסְאוֹ בַּבַּיִת הַגָּדוֹל. הַשֶּׁמֶשׁ שָׁלְחָה קַרְנַיִם מוּזְהָבוֹת דֶּרֶךְ הַחַלּוֹנוֹת, וּמְשָׁרֶתֶת צְעִירָה נִקְּתָה אֶת הָרִצְפָּה הַמְרֻצֶּפֶת אַבְנֵי שַׁיִשׁ.

הַמְּשָׁרֶתֶת הֶחְזִיקָה מַטְאֲטֵא עָשׂוּי מֵעַנְפֵי קוֹצִים קָשִׁים, וְגֵרְדָה אֶת הָאֲבָק בְּרַעַשׁ גָּדוֹל: קְרְרְט! קְרְרְט! רֵיחַ הָאָבָק עָלָה בָּאֲוִיר. לְפֶתַע, צָעֲקָה הַמְּשָׁרֶתֶת בִּגְעָלָה וְהֵרִימָה אֶת הַמַּטְאֲטֵא.

בַּפִּנָּה, צְמוּדִים לַקִּיר, הָיוּ מֻנָּחִים גּוּרִים קְטַנְטַנִּים שֶׁל חֻלְדָּה (וְיֵשׁ אוֹמְרִים שֶׁל עַכְבָּר). הֵם הָיוּ וְרֻדִּים, קֵרְחִים, וְעֵינֵיהֶם עֲדַיִן לֹא נִפְקְחוּ לָעוֹלָם. הֵם צִיְּצוּ בְּקוֹלוֹת דַּקִּים וַחֲלָשִׁים: "צִיק... צִיק...".

הַמְּשָׁרֶתֶת לֹא חָשְׁבָה פַּעֲמַיִם. הִיא הֵנִיפָה אֶת הַמַּטְאֲטֵא וְהִתְכּוֹנְנָה לִגְרֹף אֶת כָּל הַגּוּרִים הַקְּטַנִּים מִבַּעַד לַדֶּלֶת הַחוּצָה לָאֲשְׁפָּה, שָׁם בְּוַדַּאי יָמוּתוּ מִקֹּר אוֹ שֶׁחַיּוֹת רָעוֹת יִטְרְפוּ אוֹתָם.

"עִצְרִי!"

קָרָא קוֹל אוֹהֵב וְרַחוּם. זֶה הָיָה קוֹלוֹ שֶׁל רַבִּי יְהוּדָה הַנָּשִׂיא. הַמְּשָׁרֶתֶת עָמְדָה קְפוּאָה.

רַבִּי הִבִּיט בַּיְּצוּרִים הַחַלָּשִׁים. הוּא לֹא אָמַר 'לְכָךְ נוֹצְרוּ'. הַפַּעַם, לִבּוֹ נִפְתַּח מַמָּשׁ כְּמוֹ שֶׁל אַבָּא אֶל יְלָדָיו. הוּא הִצְבִּיעַ עַל הַגּוּרִים וְאָמַר לַמְּשָׁרֶתֶת בְּרֹךְ:

"הָנִיחִי לָהֶם, שֶׁיִּחְיוּ וְלֹא יָמוּתוּ. הֲרֵי כָּתוּב בְּסֵפֶר תְּהִלִּים (קמ"ה): 'וְרַחֲמָיו עַל כָּל מַעֲשָׂיו!' – הַקָּדוֹשׁ בָּרוּךְ הוּא מוֹרִיד רַחֲמִים אֲפִלּוּ עַל הַיְצוּרִים הַקְּטַנִּים בְּיוֹתֵר, וְגַם מֵאִתָּנוּ מְבֻקָּשׁ לְרַחֵם!".

הַמְּשָׁרֶתֶת הֵנִיחָה בַּעֲדִינוּת אֶת הַמַּטְאֲטֵא, וְעָזְבָה אֶת הַגּוּרִים הַקְּטַנִּים לְנַפְשָׁם, שֶׁיִּגְדְּלוּ בְּבִטְחָה.

וְאָז – קָרָא פֶּלֶא שְׁמֵימִי!

מַמָּשׁ בְּאוֹתוֹ הָרֶגַע, בְּבֵית הַדִּין שֶׁל מַעְלָה הִבִּיטוּ הַמַּלְאָכִים בְּרַבִּי. הֵם רָאוּ אֵיךְ נְשִׂיא יִשְׂרָאֵל דּוֹאֵג וּמְרַחֵם גַּם עַל יְצוּר פעוּט וְזָנִיחַ כְּחֻלְדָּה אוֹ עַכְבָּר. מִיָּד הִדְהֲדָה הַכְרָזָה רַעֲמַתִית מִתַּחַת לְכִסֵּא הַכָּבוֹד:

"הוֹאִיל וְרִחֵם – שֶׁיְּרַחֲמוּ עָלָיו מִן הַשָּׁמַיִם!"

בִּשְׁנִיָּה אַחַת – כָּל הַכְּאֵבִים שֶׁהָיוּ בְּגוּפוֹ שֶׁל רַבִּי מֶשֶׁךְ 13 שָׁנָה – נֶעֶלְמוּ כְּלֹא הָיוּ! הוּא קָם מִכִּסְאוֹ, זוֹקֵף אֶת מָתְנָיו, אוֹר שֶׁל בְּרִיאוּת וְשִׂמְחָה זָרַח עַל פָּנָיו הַקְּדוֹשׁוֹת. הַקָּדוֹשׁ בָּרוּךְ הוּא לִמֵּד אֶת רַבִּי, וְדַרְכּוֹ אֶת עַם יִשְׂרָאֵל כֻּלּוֹ: מִי שֶׁעוֹשֶׂה עַצְמוֹ רַחֲמָן לְמַטָּה, אֲפִלּוּ לַבְּרִיּוֹת הַקְּטַנּוֹת בְּיוֹתֵר, מְעוֹרֵר אֶת רַחֲמֵי הַשָּׁמַיִם עָלָיו!

א. שְׁאֵלַת פְּתִיחָה (לְהַצִּיג לִפְנֵי הַסִּפּוּר):

"יְלָדִים, לְמִי מִכֶּם יֵשׁ חַיַּת מַחְמָד בַּבַּיִת אוֹ בַּחָצֵר שֶׁל סַבָּא וְסָבְתָא? מִישֶׁהוּ פַּעַם רָאָה שְׁיָרָה שֶׁל נְמָלִים אוֹ חָתוּל קָטָן בָּרְחוֹב?

אֵיךְ נָכוֹן לְהִתְנַהֵג אֲלֵיהֶם? הַיּוֹם נִשְׁמַע עַל הַמַּנְהִיג הֲכִי גָּדוֹל שֶׁל עַם יִשְׂרָאֵל בִּתְקוּפַת הַתַּנָּאִים, וְאֵיךְ שְׁנֵי מִפְגָּשִׁים שֶׁלּוֹ עִם בַּעֲלֵי חַיִּים שִׁנּוּ אֶת חַיָּיו!"

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “להלן עיבוד חינוכי, עשיר, מרתק ומרגש לסיפורו של רבי יהודה הנשיא (רבינו הקדוש) – יסוריו ורחמנותו על בעלי החיים .” — מה קרה מיד לאחר מכן?

ה. כיצד הגיע הסיפור אל האירוע: “הֵם רָאוּ אֵיךְ נְשִׂיא יִשְׂרָאֵל דּוֹאֵג וּמְרַחֵם גַּם עַל יְצוּר פעוּט וְזָנִיחַ כְּחֻלְדָּה אוֹ עַכְבָּר.”?

א. מה מלמד המעשה “יסוריו של רבי — צער בעלי חיים” על רבי יהודה הנשיא (רבי)? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “הוּא יָדַע שֶׁהַקַּצָּב אֵינוֹ עוֹשֶׂה עֲבֵרָה, אֶלָּא מְכִין בָּשָׂר לִסְעוּדוֹת מִצְוָה שֶׁל עַם יִשְׂרָאֵל.”.
ג. אילו רבי יהודה הנשיא (רבי) לא היה פועל כפי שמתואר ב־“הַמְּשָׁרֶתֶת הֶחְזִיקָה מַטְאֲטֵא עָשׂוּי מֵעַנְפֵי קוֹצִים קָשִׁים, וְגֵרְדָה אֶת הָאֲבָק בְּרַעַשׁ גָּדוֹל: קְרְרְט!” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “הֵם רָאוּ אֵיךְ נְשִׂיא יִשְׂרָאֵל דּוֹאֵג וּמְרַחֵם גַּם עַל יְצוּר פעוּט וְזָנִיחַ כְּחֻלְדָּה אוֹ עַכְבָּר.” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. להלן עיבוד חינוכי, עשיר, מרתק ומרגש לסיפורו של רבי יהודה הנשיא (רבינו…
2. יוֹם אֶחָד, הָלַךְ רַבִּי יְהוּדָה הַנָּשִׂיא בְּרַחֲבוֹת הַשּׁוּק.
3. רַבִּי מַמָּשׁ הִרְגִּישׁ אֶת מַגַּע הַפַּרְוָה הַחַמָּה וְהָרוֹעֶדֶת…
4. תַּלְמִידָיו בָּכוּ עִמּוֹ, כָּל עַם יִשְׂרָאֵל הִתְפַּלֵּל לַהֲשָׁבַת…
5. הִיא הֵנִיפָה אֶת הַמַּטְאֲטֵא וְהִתְכּוֹנְנָה לִגְרֹף אֶת כָּל…
6. הַיּוֹם נִשְׁמַע עַל הַמַּנְהִיג הֲכִי גָּדוֹל שֶׁל עַם יִשְׂרָאֵל…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “דְּמָעוֹת גְּדוֹלוֹת וּרְטֻבּוֹת זָלְגוּ מֵעֵינָיו שֶׁל בַּעַל הַחַיִּים.”.
מכתב: כתבו לרבי יהודה הנשיא (רבי) בעקבות “רַבִּי לָמַד תּוֹרָה מִתּוֹךְ סֵבֶל נוֹרָא, מַמָּשׁ שְׁלוֹשׁ עֶשְׂרֵה שָׁנִים רְצוּפוֹת!”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “הַפַּעַם, לִבּוֹ נִפְתַּח מַמָּשׁ כְּמוֹ שֶׁל אַבָּא אֶל יְלָדָיו.”.
דיווח: כתבו ידיעה קצרה שכותרתה “יסוריו של רבי — צער בעלי חיים”.
