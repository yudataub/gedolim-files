הילד השבוי — ידע פסוק קשה מישעיהו בעל פה

דף עבודה בעקבות רבי ישמעאל בן אלישע
שם: כיתה:

המעשה — הסיפור המלא

פְּתִיחָה: הַמַּסָּע אֶל עִיר הָרְשָׁעִים

יְלָדִים יְקָרִים, עִצְמוּ אֶת הָעֵינַיִם וְנַסּוּ לְדַמְיֵן עִיר עֲנָקִית וְרוֹעֶשֶׁת. זוֹהִי רוֹמָא הַגְּדוֹלָה, בִּירַת הָאִימְפֶּרְיָה. בָּרְחוֹבוֹת יֵשׁ שְׁאוֹן עֲגָלוֹת, סוּסִים דּוֹהֲרִים, חַיָּלִים נוֹשְׂאֵי חֲרָבוֹת שֶׁשִּׁרְיוֹנֵיהֶם מַבְרִיקִים בַּשֶּׁמֶשׁ, וְרֵיחַ כָּבֵד שֶׁל בָּשָׂר צָלוּי וְעָשָׁן אֲשֶׁר אוֹפֵף אֶת הָאֲוִיר. הַכֹּל כָּאן מְשַׁדֵּר כֹּחַ, עוֹשֶׁר וְגַאֲוָה.

אַךְ בְּתוֹךְ כָּל הָרַעַשׁ הַזֶּה, צוֹעֵד אָדָם אֶחָד, עָנָיו וְקָדוֹשׁ. זְקָנוֹ יָורֵד עַל מִדּוֹתָיו וְעֵינָיו מְלֵאוֹת בְּאַהֲבַת יִשְׂרָאֵל וְדַאֲגָה עֲמֻקָּה. זֶהוּ הַתַּנָּא הַגָּדוֹל, רַבִּי יְהוֹשֻׁעַ בֶּן חֲנַנְיָה .

יְרוּשָׁלַיִם כְּבָר נֶחְרְבָה, וְרַבִּי יְהוֹשֻׁעַ הִגִּיעַ הַרְחֵק לְרוֹמָא כְּדֵי לִמְצֹא שְׁבוּיִים יְהוּדִים וְלִפְדּוֹת אוֹתָם. פִּתְאוֹם, נִגָּשׁ אֵלָיו מַכָּר וְלוֹחֵשׁ עַל אָזְנוֹ סוֹד כּוֹאֵב: "רַבִּי, בְּתוֹךְ בֵּית הָאֲסוּרִים הֶחָשׁוּךְ מִכֹּל, שָׁם, בַּמַּרְתֵּף שֶׁבְּקַצְוֵי הָעִיר, נִמְצָא יֶלֶד יְהוּדִי אֶחָד. הוּא נִלְקַח מִיְּרוּשָׁלַיִם. הוּא יָפֶה עֵינַיִם, וְטוֹב רֹאִי קְוֻצּוֹתָיו תַּלְתַּלִּים... יֶלֶד מְיֻחָד בְּמִינוֹ. אֲבָל רַבִּי... הוּא שָׁם כְּבָר זְמַן רַב בֵּין הַגּוֹיִים, בַּקֹּר וּבַטֻּמְאָה. מִי יוֹדֵעַ אִם עוֹד נִשְׁאַר בּוֹ נִיצוֹץ יְהוּדִי? מִי יוֹדֵעַ אִם הוּא יָכוֹל לַחְזֹר אֵלֵינוּ?"

הַהֲלִיכָה אֶל הַכֶּלֶא – קֹר שֶׁל אֲבָנִים וּדְמָעוֹת

רַבִּי יְהוֹשֻׁעַ לֹא מֵהֵסֵּס. כְּשֶׁמְּדֻבָּר בְּנֶפֶשׁ שֶׁל יֶלֶד יְהוּדִי, עוֹזְבִים הַכֹּל. הוּא מְמַהֵר אֶל עֵבֶר אוֹתוֹ מִבְנֶה אֵימְתָנִי שֶׁל בֵּית הָאֲסוּרִים. כְּכָל שֶׁהוּא מִתְקָרֵב, הָאוֹר בָּרְחוֹבוֹת הוֹלֵךְ וְנֶעֱלָם. רֵיחַ מְחֻנָּק שֶׁל אֲדָמָה טְחוּבָה עָלָה בְּאַפּוֹ, וְקוֹל שִׁקְשׁוּק כָּבֵד שֶׁל שַׁרְשְׁרָאוֹת בַּרְזֶל מִתְנַגֵּשׁ הִדְהֵד סְבִיבָהּ.

הַבִּנְיָן מֻקָּף בְּחוֹמַת אֶבֶן עָבָה. רַבִּי יְהוֹשֻׁעַ מִסְתּוֹבֵב סָבִיב סָבִיב, עַד שֶׁהוּא מוֹצֵא אֶת מְקוֹם הַמַּרְתֵּף. יֵשׁ שָׁם חַלּוֹן קָטָן, קָטָן מְאֹד, חָסוּם בְּסוֹרְגֵי בַּרְזֶל חֲלוּדִים. דֶּרֶךְ הַחַלּוֹן הַזֶּה נִכְנֶסֶת רַק אַלֻּמַּת אוֹר קְטַנְטַנָּה, וְכָל שְׁאָר הַחֶדֶר צָלוּל בַּחֲשֵׁכָה מַפְחִידָה.

"הַיֶּלֶד הַזֶּה נִמְצָא שָׁם לְמַטָּה..." חוֹשֵׁב הַצַּדִּיק הַגָּדוֹל. הַלֵּב שֶׁלּוֹ הוֹמֶה מֵרַחֲמִים.

אֲבָל כֵּיצַד יִפְנֶה לַיֶּלֶד? אִם יִקְרָא לוֹ "יֶלֶד יְהוּדִי", אוּלַי הַשּׁוֹמְרִים יִשְׁמְעוּ וְיַכּוּ אֶת הַיֶּלֶד? וְאוּלַי הַיֶּלֶד עַצְמוֹ מְבֹהָל כָּל כָּךְ שֶׁכְּבָר שָׁכַח שְׂפַת אִמּוֹ? הַהַשְׁגָּחָה הָעֶלְיוֹנָה נוֹתֶנֶת רַעְיוֹן קָדוֹשׁ בְּלִבּוֹ שֶׁל רַבִּי יְהוֹשֻׁעַ. אִם הַיֶּלֶד הַזֶּה לָמַד תּוֹרָה בִּירוּשָׁלַיִם, אִם לִבּוֹ עֲדַיִן מְחֻבָּר לְאָבִיו וּלְאִמּוֹ הַיְּהוּדִים – יֵשׁ מַפְתֵּחַ אֶחָד שֶׁיָּכוֹל לִפְתֹּחַ לְבָבוֹת. פְּסוּקֵי תּוֹרָה!

הַשְּׁאֵלָה שֶׁל הַצַּדִּיק וּתְשׁוּבַת הַיֶּלֶד

רַבִּי יְהוֹשֻׁעַ נֶעֱמָד לְיַד הַסּוֹרְגִים, עוֹצֵם אֶת עֵינָיו, וּבְקוֹל רוֹעֵד מֵהִתְרַגְּשׁוּת וָצַעַר עַל יְרוּשָׁלַיִם שֶׁחָרְבָה, הוּא קוֹרֵא חֲצִי פָּסוּק מִתּוֹךְ סֵפֶר יְשַׁעְיָהוּ הַנָּבִיא:

"מִי נָתַן לִמְשִׁסָּה יַעֲקֹב... וְיִשְׂרָאֵל לְבֹזְזִים?!"

הַפָּסוּק מְהַדְהֵד בַּקִּירוֹת הָאֲפֵלִים. רַבִּי יְהוֹשֻׁעַ מַמְתִּין. הַשֶּׁקֶט מֻחְלָט. אַף אֶחָד לֹא עוֹנֶה. לִבּוֹ כִּמְעַט נִשְׁבָּר מִיֵּאוּשׁ. יִתָּכֵן שֶׁהַיֶּלֶד לֹא זוֹכֵר... הַפָּסוּק הַזֶּה הוּא פָּסוּק קָשֶׁה מְאֹד וְנָדִיר!

אַךְ אָז... מִתּוֹךְ הָעֲלָטָה הַכְּבֵדָה, מִבֵּין הַשַּׁרְשְׁרָאוֹת וְהָרִצְפָּה הַקָּרָה, עוֹלֶה פִּתְאוֹם קוֹל. זֶהוּ קוֹל שֶׁל יֶלֶד קָטָן, אֲבָל הוּא צָלוּל, זַךְ, וּמָלֵא בֶּאֱמוּנָה טְהוֹרָה שֶׁלֹּא נִכְבְּתָה! הַיֶּלֶד מְסַיֵּם אֶת הַפָּסוּק בְּדִיּוּק מֻשְׁלָם:

"הֲלֹא ה' – זוּ חָטָאנוּ לוֹ!! וְלֹא אָבוּ בִדְרָכָיו הָלוֹךְ, וְלֹא שָׁמְעוּ בְּתוֹרָתוֹ!!"

רַבִּי יְהוֹשֻׁעַ עוֹצֵם אֶת עֵינָיו בְּחָזְקָה וּדְמָעוֹת חַמּוֹת וַגְּדוֹלוֹת מַתְחִילוֹת לִזְלֹג עַל לְחָיָיו וְנוֹטְפוֹת עַל זְקָנוֹ. אֵלּוּ אֵינָן דְּמָעוֹת שֶׁל עֶצֶב... אֵלּוּ דְּמָעוֹת שֶׁל שִׂמְחָה! שִׂמְחָה עֲצוּמָה!

יֶלֶד כָּל כָּךְ קָטָן, בְּמָקוֹם כָּל כָּךְ טָמֵא... וְהוּא זוֹכֵר פָּסוּק נָדִיר מִנָּבִיא! הוּא לוֹקֵחַ אֶת הַתּוֹרָה אִתּוֹ לְכָל מָקוֹם!

הַהַבְטָחָה וְהַפִּדְיוֹן

"מֻבְטְחַנִי בּוֹ שֶׁמּוֹרֶה הוֹרָאָה בְּיִשְׂרָאֵל!" קוֹרֵא רַבִּי יְהוֹשֻׁעַ בְּקוֹל נִרְגָּשׁ אֶל עֵבֶר סוֹרְגֵי הַתָּא הֶחָשׁוּךְ. "הַיֶּלֶד הַזֶּה – יִגְדַּל לִהְיוֹת צַדִּיק וְגָדוֹל בַּתּוֹרָה! הָעוֹלָם כֻּלּוֹ יָאִיר מִתּוֹרָתוֹ!". הוּא נִשְׁבַּע בְּאוֹתוֹ רֶגַע שְׁבוּעָה קְדוֹשָׁה: "הָעֲבוֹדָה שֶׁמַּעֲבִידִים אוֹתִי! לֹא אָזוּז מִכָּאן עַד שֶׁאֶפְדֶּה אוֹתוֹ בְּכָל סְכוּם שֶׁיִּפְסְקוּ עָלָיו הָרוֹמָאִים!".

וְכָךְ הָיָה. רַבִּי יְהוֹשֻׁעַ בֶּן חֲנַנְיָה לֹא חָסַךְ בְּמַאֲמַצִּים. הוּא אָסַף אֶת כָּל הַמָּמוֹן וְהַזָּהָב שֶׁהָיָה יָכוֹל לְהַשִּׂיג, הָלַךְ אֶל הַמְפַקְּדִים הָרוֹמָאִים הָאַכְזָרִיִּים, וְשִׁלֵּם תּוֹעֲפוֹת זָהָב (מָמוֹן רַב מְאֹד) כְּדֵי לִפְדּוֹת אֶת הַיֶּלֶד מִבּוֹר הַשְּׁבִי.

כַּאֲשֶׁר הוֹצִיאוּ אֶת הַיֶּלֶד לָאוֹר, חִבֵּק אוֹתוֹ רַבִּי יְהוֹשֻׁעַ כְּמוֹ אַבָּא הַמְחַבֵּק אֶת בְּנוֹ יַקִּירוֹ, וְהֵבִיא אוֹתוֹ יַחַד אִתּוֹ בַּחֲזָרָה לְאֶרֶץ יִשְׂרָאֵל, יָשָׁר אֶל בֵּית הַמִּדְרָשׁ הַטָּהוֹר.

יְלָדִים, אַתֶּם וַדַּאי שׁוֹאֲלִים מִי הָיָה הַיֶּלֶד הַזֶּה? אוֹתוֹ יֶלֶד שֶׁהָיָה בַּכֶּלֶא הָרוֹמָאִי, גָּדַל וְלָמַד אֵצֶל עֲנָקֵי הַתַּנָּאִים. הַהַבְטָחָה שֶׁל רַבִּי יְהוֹשֻׁעַ הִתְקַיְּמָה בִּמְלוֹאָהּ! הוּא הָפַךְ לִהְיוֹת אֶחָד מִגְּדוֹלֵי עַם יִשְׂרָאֵל בְּכָל הַדּוֹרוֹת. זֶהוּ רַבִּי יִשְׁמָעֵאל בֶּן אֱלִישָׁע (הַכֹּהֵן הַגָּדוֹל), תַּנָּא קָדוֹשׁ שֶׁתּוֹרָתוֹ מְאִירָה אֶת עֵינֵינוּ כָּל יוֹם, אֲשֶׁר גִּלָּה לָנוּ אֶת "שְׁלוֹשׁ עֶשְׂרֵה מִדּוֹת שֶׁהַתּוֹרָה נִדְרֶשֶׁת בָּהֶן", מַמָּשׁ אוֹתָהּ תְּפִלָּה שֶׁאָנוּ קוֹרְאִים בְּכָל בֹּקֶר בַּסִּדּוּר!

הַתּוֹרָה שֶׁשִּׁנֵּן כְּיֶלֶד הִצִּילָה אוֹתוֹ, וְאָז הוּא הִצִּיל אֶת הַתּוֹרָה שֶׁל עַם יִשְׂרָאֵל לָנֶצַח נְצָחִים.

מקורות: תלמוד בבלי , מסכת גיטין, דף נ"ח, עמוד א' (סיפור חטיפתו ופדיונו של ר' ישמעאל בן אלישע ברומא ע"י ר' יהושע בן חנניה, והפסוקים מספר ישעיהו מ"ב, כ"ד). מדרש איכה רבה , פרשה ד', פסקה ד'. תוספתא , מסכת הוריות, פרק ב'. סידור התפילה – ברייתא דרבי ישמעאל ("שלוש עשרה מידות שהתורה נדרשת בהן").

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “יְלָדִים יְקָרִים, עִצְמוּ אֶת הָעֵינַיִם וְנַסּוּ לְדַמְיֵן עִיר עֲנָקִית וְרוֹעֶשֶׁת.” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “יְרוּשָׁלַיִם כְּבָר נֶחְרְבָה, וְרַבִּי יְהוֹשֻׁעַ הִגִּיעַ הַרְחֵק לְרוֹמָא כְּדֵי לִמְצֹא שְׁבוּיִים יְהוּדִים וְלִפְדּוֹת אוֹתָם.”?
ג. כיצד הגיב רבי ישמעאל בן אלישע למצב המתואר: “רַבִּי יְהוֹשֻׁעַ מִסְתּוֹבֵב סָבִיב סָבִיב, עַד שֶׁהוּא מוֹצֵא אֶת מְקוֹם הַמַּרְתֵּף.”?
ד. מה השתנה בין “אִם הַיֶּלֶד הַזֶּה לָמַד תּוֹרָה בִּירוּשָׁלַיִם, אִם לִבּוֹ עֲדַיִן מְחֻבָּר לְאָבִיו וּלְאִמּוֹ הַיְּהוּדִים – יֵשׁ…” לבין “מֻבְטְחַנִי בּוֹ שֶׁמּוֹרֶה הוֹרָאָה בְּיִשְׂרָאֵל!" קוֹרֵא רַבִּי יְהוֹשֻׁעַ בְּקוֹל נִרְגָּשׁ אֶל עֵבֶר סוֹרְגֵי…”?
ה. כיצד הגיע הסיפור אל האירוע: “כַּאֲשֶׁר הוֹצִיאוּ אֶת הַיֶּלֶד לָאוֹר, חִבֵּק אוֹתוֹ רַבִּי יְהוֹשֻׁעַ כְּמוֹ אַבָּא הַמְחַבֵּק אֶת בְּנוֹ יַקִּירוֹ, וְהֵבִיא אוֹתוֹ יַחַד אִתּוֹ…”?

3. מה היה ראשון?

כתבו ליד כל אירוע מספר מ־1 עד 5 לפי סדר התרחשותו בסיפור.

□ הַהַשְׁגָּחָה הָעֶלְיוֹנָה נוֹתֶנֶת רַעְיוֹן קָדוֹשׁ בְּלִבּוֹ שֶׁל רַבִּי יְהוֹשֻׁעַ.
□ יְלָדִים יְקָרִים, עִצְמוּ אֶת הָעֵינַיִם וְנַסּוּ לְדַמְיֵן עִיר עֲנָקִית וְרוֹעֶשֶׁת.
□ הַתּוֹרָה שֶׁשִּׁנֵּן כְּיֶלֶד הִצִּילָה אוֹתוֹ, וְאָז הוּא הִצִּיל אֶת הַתּוֹרָה שֶׁל עַם יִשְׂרָאֵל לָנֶצַח נְצָחִים.
□ הוּא שָׁם כְּבָר זְמַן רַב בֵּין הַגּוֹיִים, בַּקֹּר וּבַטֻּמְאָה.
□ הַיֶּלֶד הַזֶּה – יִגְדַּל לִהְיוֹת צַדִּיק וְגָדוֹל בַּתּוֹרָה!

לפני ואחרי

מה קרה לפני: “הַהַשְׁגָּחָה הָעֶלְיוֹנָה נוֹתֶנֶת רַעְיוֹן קָדוֹשׁ בְּלִבּוֹ שֶׁל רַבִּי יְהוֹשֻׁעַ.”?
מה קרה אחרי: “הַיֶּלֶד הַזֶּה – יִגְדַּל לִהְיוֹת צַדִּיק וְגָדוֹל בַּתּוֹרָה!”?

סיבה ותוצאה מתוך המעשה

האירוע הנתוןמה גרם לו?מה הייתה תוצאתו?
הוּא מְמַהֵר אֶל עֵבֶר אוֹתוֹ מִבְנֶה אֵימְתָנִי שֶׁל בֵּית הָאֲסוּרִים.
זֶהוּ קוֹל שֶׁל יֶלֶד קָטָן, אֲבָל הוּא צָלוּל, זַךְ, וּמָלֵא בֶּאֱמוּנָה טְהוֹרָה שֶׁלֹּא נִכְבְּתָה!

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “הילד השבוי — ידע פסוק קשה מישעיהו בעל פה” על רבי ישמעאל בן אלישע? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “הַיֶּלֶד הַזֶּה נִמְצָא שָׁם לְמַטָּה..." חוֹשֵׁב הַצַּדִּיק הַגָּדוֹל.”.
ג. אילו רבי ישמעאל בן אלישע לא היה פועל כפי שמתואר ב־“מֻבְטְחַנִי בּוֹ שֶׁמּוֹרֶה הוֹרָאָה בְּיִשְׂרָאֵל!" קוֹרֵא רַבִּי יְהוֹשֻׁעַ בְּקוֹל נִרְגָּשׁ אֶל עֵבֶר סוֹרְגֵי הַתָּא…” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “כַּאֲשֶׁר הוֹצִיאוּ אֶת הַיֶּלֶד לָאוֹר, חִבֵּק אוֹתוֹ רַבִּי יְהוֹשֻׁעַ כְּמוֹ אַבָּא הַמְחַבֵּק אֶת בְּנוֹ יַקִּירוֹ, וְהֵבִיא…” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. יְלָדִים יְקָרִים, עִצְמוּ אֶת הָעֵינַיִם וְנַסּוּ לְדַמְיֵן עִיר…
2. פִּתְאוֹם, נִגָּשׁ אֵלָיו מַכָּר וְלוֹחֵשׁ עַל אָזְנוֹ סוֹד כּוֹאֵב:…
3. יֵשׁ שָׁם חַלּוֹן קָטָן, קָטָן מְאֹד, חָסוּם בְּסוֹרְגֵי בַּרְזֶל…
4. מִתּוֹךְ הָעֲלָטָה הַכְּבֵדָה, מִבֵּין הַשַּׁרְשְׁרָאוֹת וְהָרִצְפָּה…
5. הוּא נִשְׁבַּע בְּאוֹתוֹ רֶגַע שְׁבוּעָה קְדוֹשָׁה: "הָעֲבוֹדָה…
6. הַתּוֹרָה שֶׁשִּׁנֵּן כְּיֶלֶד הִצִּילָה אוֹתוֹ, וְאָז הוּא הִצִּיל אֶת…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “רַבִּי יְהוֹשֻׁעַ מִסְתּוֹבֵב סָבִיב סָבִיב, עַד שֶׁהוּא מוֹצֵא אֶת מְקוֹם הַמַּרְתֵּף.”.
מכתב: כתבו לרבי ישמעאל בן אלישע בעקבות “זֶהוּ קוֹל שֶׁל יֶלֶד קָטָן, אֲבָל הוּא צָלוּל, זַךְ, וּמָלֵא בֶּאֱמוּנָה טְהוֹרָה שֶׁלֹּא נִכְבְּתָה!”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “לֹא אָזוּז מִכָּאן עַד שֶׁאֶפְדֶּה אוֹתוֹ בְּכָל סְכוּם שֶׁיִּפְסְקוּ עָלָיו הָרוֹמָאִים!".”.
דיווח: כתבו ידיעה קצרה שכותרתה “הילד השבוי — ידע פסוק קשה מישעיהו בעל פה”.
