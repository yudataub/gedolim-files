מי אכל שום? — שלא לבייש

דף עבודה בעקבות ר' חייא
שם: כיתה:

המעשה — הסיפור המלא

הנה מערך חינוכי, רגשי ועשיר על פי בקשתך, המותאם במיוחד לתלמידי כיתות ב'-ד' בחינוך הדתי-לאומי, מבוסס על הסיפור המפורסם של רבי חייא ואכילת השום (סנהדרין י"א ע"א).

פְּתִיחָה: רֵיחָהּ שֶׁל תּוֹרָה בִּישִׁיבַת רַבֵּנוּ הַקָּדוֹשׁ

יְלָדִים, אִם הֱיִיתֶם יְכוֹלִים לְעַצֵּם עֵינַיִם וְלָעוּף בְּמִנְהֶרֶת הַזְּמַן אַלְפַּיִם שָׁנָה אָחוֹרָה, אֶל בֵּית הַמִּדְרָשׁ הַגָּדוֹל בְּאֶרֶץ יִשְׂרָאֵל, מָה הֱיִיתֶם רוֹאִים וְשׁוֹמְעִים?

הֱיִיתֶם וַדַּאי רוֹאִים מֵאוֹת רַבּוֹת שֶׁל תַּלְמִידֵי חֲכָמִים צְפוּפִים. שׁוּרוֹת-שׁוּרוֹת שֶׁל סַפְסָלִים עֲשׂוּיִים עֵץ כָּבֵד. הֱיִיתֶם רוֹאִים אֶת קַרְנֵי הַשֶּׁמֶשׁ שֶׁל הַבֹּקֶר חֲודְרוֹת מִבַּעַד לַחַלּוֹנוֹת וּמְאִירוֹת אֶת חֶדֶר הַלִּמּוּד. הֱיִיתֶם מְרִיחִים אֶת רֵיחַ הַקְּלָף הַיָּשָׁן שֶׁל סִפְרֵי הַתּוֹרָה, וּשְׁוֹמְעִים – אוֹהוֹ, מָה הֱיִיתֶם שׁוֹמְעִים! – קוֹל אַדִּיר, מָתוֹק וּמְרַגֵּשׁ, שֶׁל מֵאוֹת אֲנָשִׁים שֶׁלּוֹמְדִים וְשָׁרִים דִּבְרֵי תּוֹרָה יַחַד.

בְּמִזְרַח בֵּית הַמִּדְרָשׁ יָשַׁב רַבִּי יְהוּדָה הַנָּשִׂיא. כֻּלָּם קָרְאוּ לוֹ בְּיִרְאַת כָּבוֹד: "רַבֵּנוּ הַקָּדוֹשׁ". כַּאֲשֶׁר הוּא הָיָה פּוֹתֵחַ אֶת פִּיו, הַשֶּׁקֶט הָיָה מֻחְלָט. אֲפִלּוּ הַצִּפּוֹרִים בַּחוּץ כְּאִלּוּ הִפְסִיקוּ לְצַיֵּץ כְּדֵי לְהַקְשִׁיב לוֹ.

מוּלוֹ יָשַׁב הַתַּלְמִיד הַבָּכִיר שֶׁלּוֹ, אָדָם שֶׁהָיָה גָּדוֹל כָּל כָּךְ שֶׁרַבִּי יְהוּדָה אַהֲבָה אוֹתוֹ כְּמוֹ אָח – רַבִּי חִיָּא . רַבִּי חִיָּא הָיָה עָנָיו עָצוּם. אִישׁ חֶסֶד שֶׁבְּכָל כּוֹחוֹ דָּאַג שֶׁתּוֹרַת יִשְׂרָאֵל לֹא תִּשָּׁכַח לְעוֹלָם.

הַהַפְרָעָה בְּאֶמְצַע הַשִּׁיעוּר

וְהִנֵּה, יוֹם אֶחָד, בְּאֶמְצַע שִׁיעוּר עָמֹק וּמְרוֹמֵם שֶׁל רַבֵּנוּ הַקָּדוֹשׁ, הִתְחִיל לְהִתְפַּשֵּׁט בָּאֲוִיר רֵיחַ שׁוֹנֶה לְגַמְרֵי מֵרֵיחַ הַקְּלָף. זֶה הָיָה רֵיחַ חָזָק, צְוֹרֵב וַחֲרִיף.

מִישֶׁהוּ הִכְנִיס אֶת הָרֵיחַ הַזֶּה לְבֵית הַמִּדְרָשׁ.

זֶה הָיָה רֵיחַ שֶׁל שׁוּם .

רַבִּי יְהוּדָה הַנָּשִׂיא הָיָה אָדָם עָדִין מְאֹד וְאִסְטְנִיס (אָדָם שֶׁרָגִישׁ מְאֹד לְרֵיחוֹת וּלְלִכְלוּךְ). הָרֵיחַ הֶחָרִיף שֶׁל הַשּׁוּם גָּרַם לוֹ לְכְאֵב רֹאשׁ עַד שֶׁלֹּא יָכוֹל הָיָה לְהַמְשִׁיךְ לְהִתְרַכֵּז בַּלִּמּוּד וּלְהָבִין אֶת סוֹדוֹת הַתּוֹרָה. דִּבְרֵי הַתּוֹרָה קְדוֹשִׁים, וְהַמָּקוֹם צָרִיךְ לִהְיוֹת נָקִי מֵרֵיחוֹת רָעִים.

לְפֶתַע, רַבֵּנוּ הַקָּדוֹשׁ שָׁתַק. הוּא הִפְסִיק לְדַבֵּר.

שֶקֶט מָתוּחַ וְכָבֵד הִשְׁתָּרֵר בְּבֵית הַמִּדְרָשׁ. הַתַּלְמִידִים הִבִּיטוּ זֶה בָּזֶה, תְּמֵהִים לָמָּה הָרַב עָצַר.

רַבִּי יְהוּדָה עָמַד, הִבִּיט בַּקָּהָל הָעָצוּם, וְאָמַר בְּקוֹל בָּרוּר:

"מִי שֶׁאָכַל שׁוּם – יֵצֵא."

דְּפִיקוֹת הַלֵּב שֶׁל הַיְהוּדִי הָאַלְמוֹנִי

יְלָדִים, בּוֹאוּ נְדַמְיֵן עַכְשָׁו אֶת אוֹתוֹ יְהוּדִי מִסְכֵּן שֶׁאָכַל אֶת הַשּׁוּם לִפְנֵי הַשִּׁיעוּר. הוּא בְּוַדַּאי לֹא עָשָׂה זֹאת כְּדֵי לְהַפְרִיעַ, אוּלַי הָיָה זֶה הָאֹכֶל הַיָּחִיד שֶׁהָיָה לוֹ, אוּלַי הָיָה זֶה מִטַּעֲמֵי בְּרִיאוּת (שׁוּם הָיָה כְּלִי רְפוּאִי בְּאוֹתָם יָמִים).

כְּשֶׁהוּא שָׁמַע אֶת הַמִּלִּים "מִי שֶׁאָכַל שׁוּם יֵצֵא", הַלֵּב שֶׁלּוֹ דָּפַק בְּכֹחַ בְּתוֹךְ הֶחָזֶה כְּמוֹ פַּטִּישׁ: בּוּם-בּוּם... בּוּם-בּוּם . זֵעָה קָרָה כִּסְּתָה אֶת הַמֵּצַח שֶׁלּוֹ.

הוּא אָמַר לְעַצְמוֹ בַּלֵּב: "אִם אֲנִי אָקוּם עַכְשָׁו, מוּל רַבֵּנוּ הַקָּדוֹשׁ, מוּל מֵאוֹת חֲבֵרַי לַסַּפְסָל... כֻּלָּם יִסְתַּכְּלוּ עָלַי! כֻּלָּם יַגִּידוּ: הִנֵּה זֶה שֶׁהִפְרִיעַ לְשַׂר הַתּוֹרָה הַגָּדוֹל! אֲנִי אֶתְבַּיֵּשׁ לָנֶצַח. אֵיךְ אוּכַל לְהַרְאוֹת אֶת הַפָּנִים שֶׁלִּי מָחָר? פְנַי יֶאֱדִימוּ, וְאָז יַלְבִּינוּ... שָׁעָה שֶׁל מָוֶת מִבּוּשָׁה!"

הָאִישׁ קָפָא עַל מְקוֹמוֹ. רַגְלָיו כְּאִלּוּ נֻדְבְּקוּ לְרִצְפַּת הָאֶבֶן שֶׁל בֵּית הַמִּדְרָשׁ. הוּא לֹא קָם.

וְרַבִּי יְהוּדָה – מַמְתִּין. הַשֶּׁקֶט רוֹעֵם.

הַהַחְלָטָה שֶׁל רַבִּי חִיָּא

בַּשּׁוּרָה הָרִאשׁוֹנָה, יוֹשֵׁב רַבִּי חִיָּא. רַבִּי חִיָּא לֹא אָכַל שׁוּם. פִּיו נָקִי כְּמַיִם צְלוּלִים, וְהוּא מוּכָן כֻּלּוֹ לְהַמְשִׁיךְ לִלְמֹד אֶת תּוֹרַת ה'.

אֲבָל רַבִּי חִיָּא, בְּחָכְמָתוֹ הַקְּדוֹשָׁה וּבְעֵינָיו הָרַחוּמוֹת, קוֹלֵט מָה קוֹרֶה. הוּא מוֹדֵד בְּרֹאשׁוֹ אֶת הַשְּׁנִיּוֹת. הַזְּמַן עוֹבֵר, אִישׁ לֹא קָם. הַזּוּלַת שָׁם מֵאָחוֹר נִמְצָא בְּבֶהָלָה נוֹרָאָה. עוֹד שְׁנִיָּה אַחַת שֶׁל שֶׁקֶט, וְאוּלַי מִישֶׁהוּ יַצְבִּיעַ עָלָיו? אוּלַי יִגְעֲרוּ בּוֹ? הַבּוּשָׁה כְּבָר בְּדֶרֶךְ אֵלָיו!

לְהַלְבִּין פָּנִים שֶׁל יְהוּדִי? מַמָּשׁ לֹא בְּמִשְׁמַרְתִּי! - אוֹמֵר רַבִּי חִיָּא בְּלִבּוֹ.

בְּלִי לוֹמַר מִלָּה, קָם רַבִּי חִיָּא מִמְּקוֹמוֹ. הָרַב הַגָּדוֹל קָם לִמְלוֹא קוֹמָתוֹ. צְלִיל עֵץ הַסַּפְסָל הַנֶּהֱדַף לְאָחוֹר נִשְׁמָע בֶּחָלָל.

כֻּלָּם שׁוֹאֲפִים אֲוִיר בְּהַפְתָּעָה – מָה? יִתָּכֵן שֶׁרַבִּי חִיָּא הַגָּדוֹל הוּא זֶה שֶׁהִפְרִיעַ לָרַב? הַאִם רַבִּי חִיָּא חָטָא בְּחֹסֶר כָּבוֹד לְרַבֵּנוּ הַקָּדוֹשׁ?

אֲבָל רַבִּי חִיָּא לֹא חָשַׁב עַל כְּבוֹדוֹ שֶׁלּוֹ. הוּא לָקַח אֶת חֲפָצָיו, הִרְכִּין רֹאשׁוֹ, וְצָעַד הַחוּצָה בִּצְעָדִים מְהַדְהֲדִים. צְעָדָיו נִשְׁמְעוּ עַל מִרְצְפוֹת הָאֶבֶן: טַק... טַק... טַק . וְהוּא יָצָא אֶל מִחוּץ לְבֵית הַמִּדְרָשׁ.

הֶעָנָן מִתְפַּזֵּר

כַּאֲשֶׁר שְׁאָר הַתַּלְמִידִים הַחֲשׁוּבִים רָאוּ אֶת רַבִּי חִיָּא קָם וְיוֹצֵא, הֵם אָמְרוּ לְעַצְמָם: "מָה?! כְּבוֹד רַבִּי חִיָּא עָזַב? אֵיךְ נוּכַל אֲנַחְנוּ לְהִשָּׁאֵר לָשֶׁבֶת הַיּוֹם בְּשִׁיעוּר כְּשֶׁהוּא בַּחוּץ?"

בְּזֶה אַחַר זֶה, קָמוּ כָּל תַּלְמִידֵי הַיְשִׁיבָה וְיָצְאוּ גַּם הֵם הַחוּצָה לִכְבוֹדוֹ שֶׁל רַבִּי חִיָּא!

בֵּית הַמִּדְרָשׁ הִתְרוֹקֵן.

וּבְתוֹךְ הֶהָמוֹן הַיּוֹצֵא, מְגַמֵּד וְנֶחְבָּא אֶל הַכֵּלִים, יָצָא גַּם הָאִישׁ הַהוּא. הָאִישׁ שֶׁאָכַל בֶּאֱמֶת אֶת הַשּׁוּם. אִישׁ לֹא הִבְחִין בּוֹ. אִישׁ לֹא יָדַע מִי הוּא מִבֵּין הַיּוֹצְאִים.

הַסַּכָּנָה חָלְפָה. הוּא נִצַּל מִבּוּשָׁה מַחְרִידָה, כִּי אָדָם גָּדוֹל, צַדִּיק יְסוֹד עוֹלָם, פָּרַשׂ עָלָיו מָגֵן כְּשֶׁלָּקַח אֶת הָאַשְׁמָה עַל כְּתֵפָיו!

שִׂיחַת הַבֹּקֶר וְהַסּוֹד הַמִּתְגַּלֶּה

לְמָחֳרַת בַּבֹּקֶר, רַבִּי שִׁמְעוֹן, בְּנוֹ הַיַּקִּיר שֶׁל רַבִּי יְהוּדָה הַנָּשִׂיא, פָּגַשׁ אֶת רַבִּי חִיָּא בְּסִמְטְאוֹת הָעִיר. הַשֶּׁמֶשׁ שֶׁל בֹּקֶר חִמְּמָה אֶת אֲבָנֵי הָעִיר, וְצִפּוֹרִים חָזְרוּ לְצַיֵּץ.

רַבִּי שִׁמְעוֹן פָּנָה לְרַבִּי חִיָּא וְאָמַר בִּפְלִיאָה וּבִתְמִיהָה, "רַבִּי חִיָּא הַיָּקָר... אַתָּה הוּא זֶה שֶׁצִּעַר אֶת אָבִי בָּרַבִּים הֶחָשׁוּב? הַאִם אַתָּה בֶּאֱמֶת אוֹתוֹ אֶחָד שֶׁאָכַל שׁוּם קֹדֶם שֶׁנִּכְנַס לִמְקוֹם הַקֹּדֶשׁ?"

הִבִּיט רַבִּי חִיָּא בְּרַבִּי שִׁמְעוֹן. עֵינָיו בָּרְקוּ מֵאַהֲבַת יִשְׂרָאֵל, עֵינַיִם שֶׁל מַלְאָךְ מַמָּשׁ. וְהוּא עָנָה אֶת מִשְׁפַּט הַמַּפְתֵּחַ, מִשְׁפָּט שֶׁמְּהַדְהֵד עַד יָמֵינוּ אָנוּ:

"לֹא תְּהֵא כָּזֹאת בְּיִשְׂרָאֵל!"

(כְּלוֹמַר – 'חַס וְחָלִילָה שֶׁאֲנִי אֶעֱשֶׂה דָּבָר כָּזֶה! אֲבָל חַס וְחָלִילָה גַּם, שֶׁאֲנִי אֵשֵׁב מִנֶּגֶד, וְאֶתֵּן לִיהוּדִי אֶחָד, אֲפִלּוּ אָדָם פָּשׁוּט בְּיוֹתֵר, לַעֲבֹר כָּזוֹ בוּשָׁה וְהַלְבָּנַת פָּנִים אֲיֻמָּה בִּגְלַל טָעוּת! הָיִיתִי מוּכָן שֶׁתַּחְשְׁבוּ שֶׁזֶּה אֲנִי, רַק שֶׁהוּא יִשָּׁאֵר שָׁלֵם וְלֹא יִשָּׁבֵר לִבּוֹ!').

א. שְׁאֵלַת פְּתִיחָה (לְעוֹרֵר זִהוּי וּסְקַרְנוּת אֵצֶל הַיְלָדִים):

"יְלָדִים, סַפְרוּ לִי: הַאִם הָיָה פַּעַם מִקְרֶה שֶׁלְּקַחְתֶּם עַל עַצְמְכֶם 'אַשְׁמָה' שֶׁל אָח קָטָן אוֹ שֶׁל חָבֵר דָּחוּק, רַק כְּדֵי שֶׁהוּא לֹא יְקַבֵּל כַּעַס גָּדוֹל אוֹ יִתְבַּיֵּשׁ? הַיּוֹם נִלְמַד עַל הַנֶּפֶשׁ הַגְּדוֹלָה בְּיוֹתֵר - עַל גִּבּוֹר שֶׁל אָהֲבָה וְוִתּוּר עַל כָּבוֹד, גִּבּוֹר מִיְּמֵי חֲזַ"ל!"

ב. הֶעָרוֹת בִּצּוּעַ וְסְטוֹרִיטֶלִינְג שֶׁל דֶּרֶךְ סִפּוּר:

ג. הַמְלָצוֹת לְאַרְבַּע שְׁקוּפִיּוֹת בַּמַּצֶּגֶת:

ד. פְּעִילוּת מְסַכֶּמֶת ("מָגֵן הַכָּבוֹד"):

(מוּכָן כָּעֵת לְהַעֲבָרָה לַמּוֹרִים לְלִימּוּד אִיכּוּתִי וּמְעוֹרֵר אֶצֶל הַיְּלָדִים!)

מקורות: תַּלְמוּד בַּבְלִי, מַסֶּכֶת סַנְהֶדְרִין, דַּף י"א עַמּוּד א' : "כִּי הָא דְּיָתִיב רַבִּי וְקָא דָּרֵישׁ, וְהֵרִיחַ רֵיחַ שׁוּם. אָמַר: מִי שֶׁאָכַל שׁוּם - יֵצֵא! עָמַד רַבִּי חִיָּא וְיָצָא. עָמְדוּ כּוּלָּן וְיָצְאוּ. בְּשַׁחַר, מְצָאוֹ רַבִּי שִׁמְעוֹן בְּרַבִּי לְרַבִּי חִיָּא. אָמַר לֵיהּ: אַתָּה הוּא שֶׁצִּעַרְתָּ לְאַבָּא?! אָמַר לֵיהּ: לֹא תְּהֵא כָּזֹאת בְּיִשְׂרָאֵל!" (מְקוֹר הַמַּאֲמָר מוֹפִיעַ בַּמַּחְבֶּרֶת גְּדוֹלֵי יִשְׂרָאֵל, תַּחַת נְקוּדַת "רַבִּי חִיָּא – מִי אָכַל שׁוּם"). הַקְשֵׁר רַעְיוֹנִי (תַּלְמוּד בַּבְלִי, בָּבָא מְצִיעָא נ"ט עַמּוּד א'): "נוֹחַ לוֹ לְאָדָם שֶׁיַּפִּיל עַצְמוֹ לְתוֹךְ כִּבְשַׁן הָאֵשׁ וְאַל יַלְבִּין פְּנֵי חֲבֵרוֹ בָּרַבִּים". אִמְרָה זוֹ יְדוּעָה וּמַמְחִישָׁה אֶת מְסִירוּת-נַפְשׁוֹ שֶׁל רַבִּי חִיָּא בַּמַּעֲשֶׂה. (מוּכָן כָּעֵת לְהַעֲבָרָה לַמּוֹרִים לְלִימּוּד אִיכּוּתִי וּמְעוֹרֵר אֶצֶל הַיְּלָדִים!)

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “הנה מערך חינוכי, רגשי ועשיר על פי בקשתך, המותאם במיוחד לתלמידי כיתות ב'-ד' בחינוך הדתי-לאומי, מבוסס על הסיפור המפורסם של רבי חייא ואכילת השום (סנהדרין י"א…” — מה קרה מיד לאחר מכן?

ה. כיצד הגיע הסיפור אל האירוע: “אֲבָל חַס וְחָלִילָה גַּם, שֶׁאֲנִי אֵשֵׁב מִנֶּגֶד, וְאֶתֵּן לִיהוּדִי אֶחָד, אֲפִלּוּ אָדָם פָּשׁוּט בְּיוֹתֵר, לַעֲבֹר כָּזוֹ בוּשָׁה וְהַלְבָּנַת…”?

סיבה ותוצאה מתוך המעשה

האירוע הנתוןמה גרם לו?מה הייתה תוצאתו?
רַבִּי יְהוּדָה הַנָּשִׂיא הָיָה אָדָם עָדִין מְאֹד וְאִסְטְנִיס (אָדָם שֶׁרָגִישׁ מְאֹד לְרֵיחוֹת וּלְלִכְלוּךְ).

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “מי אכל שום? — שלא לבייש” על ר' חייא? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “הוּא בְּוַדַּאי לֹא עָשָׂה זֹאת כְּדֵי לְהַפְרִיעַ, אוּלַי הָיָה זֶה הָאֹכֶל הַיָּחִיד שֶׁהָיָה לוֹ, אוּלַי הָיָה זֶה מִטַּעֲמֵי בְּרִיאוּת (שׁוּם הָיָה…”.
ג. אילו ר' חייא לא היה פועל כפי שמתואר ב־“הוּא נִצַּל מִבּוּשָׁה מַחְרִידָה, כִּי אָדָם גָּדוֹל, צַדִּיק יְסוֹד עוֹלָם, פָּרַשׂ עָלָיו מָגֵן כְּשֶׁלָּקַח אֶת הָאַשְׁמָה…” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “אֲבָל חַס וְחָלִילָה גַּם, שֶׁאֲנִי אֵשֵׁב מִנֶּגֶד, וְאֶתֵּן לִיהוּדִי אֶחָד, אֲפִלּוּ אָדָם פָּשׁוּט בְּיוֹתֵר, לַעֲבֹר כָּזוֹ…” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. הנה מערך חינוכי, רגשי ועשיר על פי בקשתך, המותאם במיוחד לתלמידי כיתות…
2. מוּלוֹ יָשַׁב הַתַּלְמִיד הַבָּכִיר שֶׁלּוֹ, אָדָם שֶׁהָיָה גָּדוֹל…
3. רַבִּי יְהוּדָה עָמַד, הִבִּיט בַּקָּהָל הָעָצוּם, וְאָמַר בְּקוֹל…
4. אֲבָל רַבִּי חִיָּא, בְּחָכְמָתוֹ הַקְּדוֹשָׁה וּבְעֵינָיו הָרַחוּמוֹת,…
5. רַבִּי שִׁמְעוֹן פָּנָה לְרַבִּי חִיָּא וְאָמַר בִּפְלִיאָה…
6. (מוּכָן כָּעֵת לְהַעֲבָרָה לַמּוֹרִים לְלִימּוּד אִיכּוּתִי וּמְעוֹרֵר…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “הַתַּלְמִידִים הִבִּיטוּ זֶה בָּזֶה, תְּמֵהִים לָמָּה הָרַב עָצַר.”.
מכתב: כתבו לר' חייא בעקבות “הוּא לָקַח אֶת חֲפָצָיו, הִרְכִּין רֹאשׁוֹ, וְצָעַד הַחוּצָה בִּצְעָדִים מְהַדְהֲדִים.”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “רַבִּי שִׁמְעוֹן פָּנָה לְרַבִּי חִיָּא וְאָמַר בִּפְלִיאָה וּבִתְמִיהָה, "רַבִּי חִיָּא הַיָּקָר...”.
דיווח: כתבו ידיעה קצרה שכותרתה “מי אכל שום? — שלא לבייש”.
