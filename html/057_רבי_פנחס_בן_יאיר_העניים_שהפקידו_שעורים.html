העניים שהפקידו שעורים

דף עבודה בעקבות רבי פנחס בן יאיר
שם: כיתה:

המעשה — הסיפור המלא

להלן עיבוד חינוכי, רגשי ומרתק לסיפור המופלא על רבי פנחס בן יאיר והעניים שהפקידו שעורים , מעובד במיוחד לבנים בכיתות ב'-ד' (חינוך דתי-לאומי). הסיפור מדגיש את החשיבות העצומה של שמירה על רכוש הזולת, עד כדי מסירות נפש של ממש, ומראה כיצד עמל כפיים ואמונה מביאים לברכה והשגחה פרטית.

הסיפור מנוקד במלואו כנדרש.

פְּתִיחָה: סְעָרָה בַּלַּיְלָה וְשַׂקִּית שֶׁל מַחְסוֹר

יְלָדִים, סִגְרוּ לְרֶגַע אֶת הָעֵינַיִם. דַּמְיְנוּ עֶרֶב חֹרְפִּי וְקַר מְאֹד. רוּחַ חֲזָקָה שׁוֹרֶקֶת בַּחוּץ, מְעִיפָה עָנָפִים וְעָלִים. טִפּוֹת גֶּשֶׁם כְּבֵדוֹת נוֹפְלוֹת עַל גַּגּוֹת הַבָּתִּים וְרֵיחַ שֶׁל אֲדָמָה רְטֻבָּה וּבֹץ מְמַלֵּא אֶת הָאֲוִיר. בְּתוֹךְ הַסְּעָרָה הַזּוֹ, בַּדֶּרֶךְ לָעִיר, צוֹעֲדִים בְּקֹשִׁי שְׁנֵי אֲנָשִׁים עֲנִיִּים. בִּגְדֵיהֶם רְטֻבִּים וְסְפוּגִים בְּמַיִם, וּנְעָלֵיהֶם בּוֹסְסוֹת בַּבֹּץ.

הֵם נוֹדְדִים כְּבָר יָמִים אֲרֻכִּים לְחַפֵּשׂ עֲבוֹדָה אוֹ מְעַט כֶּסֶף לְמִשְׁפְּחוֹתֵיהֶם. בִּידֵיהֶם שֶׁל הָעֲנִיִּים יֵשׁ רַק דָּבָר אֶחָד יָקָר – שַׂקִּית קְטַנָּה עֲשׂוּיָה מִבַּד גַּס, וּבְתוֹכָהּ מְעַט גַּרְגִּירֵי שְׂעוֹרִים (סוּג שֶׁל תְּבוּאָה). זֶה מָה שֶׁנִּשְׁאַר לָהֶם בָּעוֹלָם כֻּלּוֹ.

הָעֲנִיִּים הִגִּיעוּ לְעִירוֹ שֶׁל הַתַּנָּא הַקָּדוֹשׁ, רַבִּי פִּנְחָס בֶּן יָאִיר.

טּוּק, טּוּק, טּוּק. הֵם דָּפְקוּ בַּיָּדַיִם רוֹעֲדוֹת מִקֹּר עַל דֶּלֶת הָעֵץ הַכְּבֵדָה שֶׁל בֵּית הַצַּדִּיק.

הַדֶּלֶת נִפְתְּחָה, וּמוּלָם עָמַד רַבִּי פִּנְחָס בֶּן יָאִיר. פָּנָיו הֵאִירוּ כְּאוֹר הַשֶּׁמֶשׁ. הוּא הִכְנִיס אוֹתָם מִיָּד פְּנִימָה, הוֹשִׁיב אוֹתָם לְיַד תַּנּוּר חַם, וְהִגִּישׁ לָהֶם תֵּה מַהְבִּיל שֶׁחִמֵּם אֶת הַגּוּף וְאֶת הַנְּשָׁמָה.

לְמָחֳרָת בַּבֹּקֶר, הָעֲנִיִּים יָצאוּ מֵהַבַּיִת כְּדֵי לְחַפֵּשׂ פַּרְנָסָה בַּשּׁוּק שֶׁל הָעִיר. אַךְ מֵרֹב עֲיֵפוּת וּמִהוּר – הֵם שָׁכְחוּ אֶת שַׂקִּית הַשְּׂעוֹרִים בַּפִּנָּה שֶׁל בֵּית הָרַב!

הַהַחְלָטָה שֶׁל הַצַּדִּיק: חֹרֶף, אָבִיב, וַעֲבוֹדָה קָשָׁה

עָבַר יוֹם, עָבְרוּ יוֹמַיִם, וְהָעֲנִיִּים לֹא חָזְרוּ.

רַבִּי פִּנְחָס מָצָא אֶת הַשַּׂקִּית בַּפִּנָּה. הוּא נָגַע בַּגַּרְגִּירִים הַקָּשִׁים. מָה אָדָם רָגִיל הָיָה עוֹשֶׂה? הָיָה אוֹמֵר: "נוּ, זֶה קְצָת שְׂעוֹרִים. אָשִׂים אֶת זֶה בָּאָרוֹן. אִם זֶה יִתְקַלְקֵל אוֹ יֵרָקֵב – מָה לַעֲשׂוֹת? זֶה לֹא אַשְׁמָתִי! הֵם שָׁכְחוּ!".

אֲבָל רַבִּי פִּנְחָס בֶּן יָאִיר יָדַע שֶׁאֲבֵדָה וּפִקָּדוֹן שֶׁל יְהוּדִי זֶה דָּבָר קָדוֹשׁ. שֶׁלֹּא נִתָּן לְזַלְזֵל אֲפִלּוּ בִּפְרוּטָה שֶׁל הַזּוּלַת!

כְּשֶׁהִגִּיעַ הָאָבִיב, שָׂם לֵב רַבִּי פִּנְחָס שֶׁהַשְּׂעוֹרִים בַּשַּׂקִּית מַתְחִילוֹת לְהֵרָאוֹת יְשָׁנוֹת, וְיֵשׁ סַכָּנָה שֶׁתּוֹלָעִים יֹאכְלוּ אוֹתָן.

מָה הוּא עָשָׂה? רַבִּי פִּנְחָס הַגָּדוֹל, נְשִׂיא הַדּוֹר, לָקַח מַעְדֵּר קָטָן, יָצָא אֶל הַשָּׂדֶה שֶׁלּוֹ, וְהִתְחִיל לְנַכֵּשׁ אֶת הָאֲדָמָה הַקָּשָׁה!

שְׁלִיק... שְׁלָאק... קוֹל הַמַּעְדֵּר מַכֶּה בָּאֲדָמָה נִשְׁמַע הֵיטֵב. זֵעָה יָרְדָה עַל מִצְחוֹ הַקָּדוֹשׁ, אַךְ הוּא זָרַע אֶת כָּל הַשְּׂעוֹרִים שֶׁל הָעֲנִיִּים טִפָּה אַחַר טִפָּה, שׁוּרָה אַחַר שׁוּרָה.

"הַקָּדוֹשׁ בָּרוּךְ הוּא," לָחַשׁ הָרַב, "שְׁמֹר עַל הַזְּרָעִים הָאֵלֶּה. הֵם שֶׁל אֲנָשִׁים מִסְכֵּנִים."

שֶׁבַע שָׁנִים שֶׁל מְסִירוּת וּפֶלֶא

הַהַשְׁגָּחָה הָעֶלְיוֹנָה רָאֲתָה אֶת לִבּוֹ הַטָּהוֹר שֶׁל הַצַּדִּיק. הַשֶּׁמֶשׁ זָרְחָה, מַיִם מְתוּקִים יָרְדוּ מִן הַשָּׁמַיִם, וְהַשְּׂעוֹרִים הַקְּטַנּוֹת גָּדְלוּ וְהָפְכוּ לְשִׁבֳּלִים גְּבוֹהוֹת וְזֹהֲבוֹת שֶׁהִתְנוֹעֲעוּ בָּרוּחַ כְּיָם שֶׁל זָהָב.

בַּקַּיִץ, יָצָא רַבִּי פִּנְחָס לִקְצֹר אֶת הַשָּׂדֶה. הוּא אָסַף אֶת כָּל הַתְּבוּאָה. מִשַּׂקִּית קְטַנָּה עֲשׂוּיָה בַּד – צָמְחוּ עַכְשָׁו כַּמָּה וְכַמָּה שַׂקִּים עֲנָקִיִּים שֶׁל תְּבוּאָה חֲדָשָׁה וְרֵיחָנִית!

אֲבָל הָעֲנִיִּים? הֵם עֲדַיִן לֹא חָזְרוּ.

בַּשָּׁנָה הַבָּאָה, שׁוּב עָמְדָה הַתְּבוּאָה לְהִתְקַלְקֵל. וְשׁוּב, יָצָא רַבִּי פִּנְחָס אֶל הַשָּׂדֶה, זָרַע אֶת כָּל הַשַּׂקִּים בַּאֲדָמָה גְּדוֹלָה יוֹתֵר!

וּבַקַּיִץ שֶׁלְּאַחַר מִכֵּן, קָצַר תְּבוּאָה כְּפוּלָה וּמְכֻפֶּלֶת! וְכָךְ – שָׁנָה אַחַר שָׁנָה. הוּא נֶאֱלַץ לִבְנוֹת מַחְסָנִים שְׁלֵמִים. שָׁנָה שְׁלִישִׁית, שְׁבִיעִית, שִׁשִּׁית...

שֶׁבַע שָׁנִים שֶׁרַבִּי פִּנְחָס בֶּן יָאִיר מַקְדִּישׁ אֶת כֹּחוֹ, זְמַנּוֹ וַאֲדָמָתוֹ – הַכֹּל כְּדֵי שֶׁהָרְכוּשׁ שֶׁל יְהוּדִי אַחֵר לֹא יֵהָרֵס, אֶלָּא יִצְמַח וְיִגְדַּל! וְהוּא שׁוֹמֵר עַל זֶה בְּשֶׁקֶט בְּשֶׁקֶט, כְּסוֹד קָדוֹשׁ בֵּינוֹ לְבֵין הַקָּדוֹשׁ בָּרוּךְ הוּא.

הַפְּגִישָׁה בַּשּׁוּק וְדֶלֶת הָאָסָם הַכְּבֵדָה

יוֹם אֶחָד, כַּעֲבֹר שֶׁבַע שָׁנִים (!), הוֹלֵךְ רַבִּי פִּנְחָס מִבֵּית הַמִּדְרָשׁ וְשׁוֹמֵעַ שְׁנֵי אֲנָשִׁים מְדַבְּרִים בַּשּׁוּק. קוֹלָם הָיָה שָׁבוּר, חָלוּשׁ וְעָצוּב. הֵם הָיוּ מְבֻגָּרִים יוֹתֵר, פְּנֵיהֶם מְקֻמָּטוֹת, וּבִגְדֵיהֶם קְרוּעִים אַף יוֹתֵר מִבֶּעָבָר.

אָמַר עָנִי אֶחָד לַחֲבֵרוֹ בַּאֲנָחָה כְּבֵדָה: "אַתָּה זוֹכֵר... לִפְנֵי שֶׁבַע שָׁנִים שָׁכַחְנוּ שַׂקִּית שֶׁל שְׂעוֹרִים אֵצֶל רַבִּי פִּנְחָס? אוי... אִלּוּ רַק הָיוּ לָנוּ הַגַּרְגִּירִים הָאֵלֶּה הַיּוֹם, הָיִינוּ מְכִינִים דַּיְסָה וְיָשְׁנִים עִם בֶּטֶן מְלֵאָה."

עָנָה לוֹ חֲבֵרוֹ: "הַלְוַאי. אֲבָל אֵיךְ נֵלֵךְ אֵלָיו עַכְשָׁו? אַחֲרֵי שֶׁבַע שָׁנִים, הַשְּׂעוֹרִים כְּבָר הָפְכוּ לְעָפָר וַאֲבַק רִקָּבוֹן. בֶּטַח זָרְקוּ אוֹתָם מִזְּמַן".

פִּתְאוֹם, הֵם הִרְגִּישׁוּ יָד חַמָּה מֻנַּחַת עַל כִּתְפֵיהֶם. הֵם הִסְתּוֹבְבוּ וְרָאוּ אֶת פָּנָיו הַמְּאִירוֹת שֶׁל רַבִּי פִּנְחָס.

"בָּנַי הַיְּקָרִים!" חִיֵּךְ הָרַב, "שָׁמַעְתִּי אֶת קוֹלְכֶם. אַתֶּם הוּא שֶׁשְּׁכַחְתֶּם אֶת הַשְּׂעוֹרִים לִפְנֵי שֶׁבַע שָׁנִים?"

"כֵּן, כְּבוֹד הָרַב," הִשְׁפִּילוּ הָעֲנִיִּים אֶת מַבָּטָם בְּבוּשָׁה. "אֲנַחְנוּ מִצְטַעֲרִים שֶׁטָּרַדְנוּ אֶת הָרַב, הֵם בְּוַדַּאי נִרְקְבוּ..."

"בּוֹאוּ אַחֲרַי!" חִיֵּךְ הָרַב.

הוּא הוֹבִיל אוֹתָם אֶל חֲצֵרוֹ הָאֲחוֹרִית. הָיוּ שָׁם אֲסָמִים גְּדוֹלִים (מַחְסְנֵי תְּבוּאָה). הָרַב הוֹצִיאוֹ מַפְתֵּחַ גָּדוֹל, הִכְנִיס לַמַּנְעוּל וְסוֹבֵב. חֲרִיקקקק – הַדֶּלֶת הַכְּבֵדָה נִפְתְּחָה לִרְוָחָה.

הָעֲנִיִּים עָמְדוּ וְלֹא הֶאֱמִינוּ לְמַרְאֵה עֵינֵיהֶם. הָרִיחוּ אֶת רֵיחַ הַשָּׂדֶה הַנִּפְלָא. הָאָסָם הָיָה מָלֵא עַד לַתִּקְרָה בְּהַר גָּדוֹל שֶׁל גַּרְגִּירֵי שְׂעוֹרָה מְשֻׁבָּחִים, טְרִיִּים וּמְלֵאִים בִּבְרָכָה! שָׁוְיָם שֶׁל הַשְּׂעוֹרִים הָיָה כֶּסֶף רַב מְאֹד – עֹשֶׁר שֶׁיָּכוֹל לְפַרְנֵס אוֹתָם שָׁנִים אֲרֻכּוֹת.

רַבִּי פִּנְחָס נֶעֱמַד מִן הַצַּד, הִצְבִּיעַ בְּיָדוֹ וְאָמַר בְּאַהֲבָה:

"שֶׁלָּכֶם הוּא. קְחוּ אוֹתוֹ. אֶת כָּל מָה שֶׁנָּתַתֶּם לִי, זָרַעְתִּי וְקָצַרְתִּי וְשָׁמַרְתִּי בַּעֲבוּרְכֶם. קְחוּ אֶת מָה שֶׁשֶּׁלָּכֶם וּלְכוּ לְשָׁלוֹם."

משחק "מוצא האבדה ואיש הפקדון":

נחלק את הילדים לזוגות. נותנים לכל זוג פריט קטן (מחק, כדור קטן, עיפרון). ילד א' נותן לילד ב' לשמור עליו למשך ה"שבוע" (שתי דקות של מוזיקה בכיתה שבהם עושים פעילות אחרת). ילד ב' צריך לחשוב על דרך מקורית כיצד שמר ודאג לחפץ עד שהחזיר אותו לילד א' (למשל: "הכנסתי לקלמר מיוחד", "ניקיתי לך אותו"). לאחר מכן שואלים את הילדים איך ההרגשה לקבל משהו חזרה בצורה מכובדת ושמורה, וכך מחברים זאת לרעיון של רבי פנחס שעמל שבע שנים!

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “להלן עיבוד חינוכי, רגשי ומרתק לסיפור המופלא על רבי פנחס בן יאיר והעניים שהפקידו שעורים , מעובד במיוחד לבנים בכיתות ב'-ד' (חינוך דתי-לאומי).” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “הֵם דָּפְקוּ בַּיָּדַיִם רוֹעֲדוֹת מִקֹּר עַל דֶּלֶת הָעֵץ הַכְּבֵדָה שֶׁל בֵּית הַצַּדִּיק.”?
ג. כיצד הגיב רבי פנחס בן יאיר למצב המתואר: “זֵעָה יָרְדָה עַל מִצְחוֹ הַקָּדוֹשׁ, אַךְ הוּא זָרַע אֶת כָּל הַשְּׂעוֹרִים שֶׁל הָעֲנִיִּים טִפָּה אַחַר טִפָּה, שׁוּרָה אַחַר שׁוּרָה.”?
ד. מה השתנה בין “וְהוּא שׁוֹמֵר עַל זֶה בְּשֶׁקֶט בְּשֶׁקֶט, כְּסוֹד קָדוֹשׁ בֵּינוֹ לְבֵין הַקָּדוֹשׁ בָּרוּךְ הוּא.” לבין “כֵּן, כְּבוֹד הָרַב," הִשְׁפִּילוּ הָעֲנִיִּים אֶת מַבָּטָם בְּבוּשָׁה.”?
ה. כיצד הגיע הסיפור אל האירוע: “הֵם הֵבִינוּ אֶת הַנֵּס הָעָצוּם וְאֶת הַקְּדֻשָּׁה שֶׁל רַבִּי פִּנְחָס – שֶׁמָּמוֹן יִשְׂרָאֵל הָיָה יָקָר בְּעֵינָיו יוֹתֵר מִכָּל דָּבָר אַחֵר…”?

האירוע הנתוןמה גרם לו?מה הייתה תוצאתו?

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “העניים שהפקידו שעורים” על רבי פנחס בן יאיר? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “מִשַּׂקִּית קְטַנָּה עֲשׂוּיָה בַּד – צָמְחוּ עַכְשָׁו כַּמָּה וְכַמָּה שַׂקִּים עֲנָקִיִּים שֶׁל תְּבוּאָה חֲדָשָׁה וְרֵיחָנִית!”.
ג. אילו רבי פנחס בן יאיר לא היה פועל כפי שמתואר ב־“כֵּן, כְּבוֹד הָרַב," הִשְׁפִּילוּ הָעֲנִיִּים אֶת מַבָּטָם בְּבוּשָׁה.” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “הֵם הֵבִינוּ אֶת הַנֵּס הָעָצוּם וְאֶת הַקְּדֻשָּׁה שֶׁל רַבִּי פִּנְחָס – שֶׁמָּמוֹן יִשְׂרָאֵל הָיָה יָקָר בְּעֵינָיו יוֹתֵר…” מחזק את המסר של הסיפור?
