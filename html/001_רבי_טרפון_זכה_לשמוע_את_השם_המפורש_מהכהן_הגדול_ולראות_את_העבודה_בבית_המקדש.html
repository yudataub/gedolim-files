זכה לשמוע את השם המפורש מהכהן הגדול ולראות את העבודה בבית המקדש

דף עבודה בעקבות רבי טרפון
שם: כיתה:

המעשה — הסיפור המלא

הַבֹּקֶר הַמֵּאִיר בִּירוּשָׁלַיִם

יְלָדִים, עִצְמוּ אֶת הָעֵינַיִם. הַאִם אַתֶּם יְכוֹלִים לְהָרִיחַ אֶת הָרֵיחַ הַמָּתוֹק וְהַמְּיֻחָד שֶׁנִּשָּׂא עַכְשָׁו בָּאֲוִיר? זֶהוּ לֹא רֵיחַ שֶׁל פְּרָחִים רְגִילִים, וְלֹא רֵיחַ שֶׁל מַאֲפִיָּה. זֶהוּ רֵיחַ קְטֹרֶת הַסַּמִּים שֶׁמַּגִּיעַ יָשָׁר מֵהֵיכַל אֲדֹנָי, וּמְמַלֵּא אֶת כָּל רְחוֹבוֹת יְרוּשָׁלַיִם. בַּבֹּקֶר הַזֶּה, אֲבָנֶיהָ שֶׁל יְרוּשָׁלַיִם מַבְרִיקוֹת תַּחַת קַרְנֵי הַשֶּׁמֶשׁ, וְהֲמוֹנֵי יְהוּדִים צוֹעֲדִים מִכָּל קַצְוֵי הָאָרֶץ לְכִוּוּן בֵּית הַמִּקְדָּשׁ.

בְּתוֹךְ הַהָמוֹן הַגָּדוֹל שֶׁפּוֹסֵעַ בְּסִמְטְאוֹת הָאֶבֶן, הוֹלֵךְ יֶלֶד אֶחָד קָטָן שֶׁקּוֹרְאִים לוֹ טַרְפוֹן. יֵשׁ לוֹ עֵינַיִם בּוֹרְקוֹת, צְמֵאוֹת, שֶׁמְּחַפְּשׂוֹת לִבְלֹעַ כָּל דָּבָר שֶׁקָּשׁוּר בְּתוֹרָה. טַרְפוֹן הַקָּטָן אֵינוֹ יֶלֶד רָגִיל מִיִּשְׂרָאֵל, טַרְפוֹן הוּא כֹּהֵן! אֶת יָדוֹ הַקְּטַנָּה הוּא אוֹחֵז בְּיָדוֹ הַגְּדוֹלָה וְהַחֲזָקָה שֶׁל דּוֹדוֹ, שִׁמְשׁוֹן.

"דוֹדִי שִׁמְשׁוֹן," לוֹחֵשׁ טַרְפוֹן הַקָּטָן בְּהִתְרַגְּשׁוּת, "הַאִם הַיּוֹם נַעֲלֶה יַחַד לַדּוּכָן? הַאִם אֶזְכֶּה לַעֲמֹד יַחַד עִם כָּל הַכֹּהֲנִים לְבָרֵךְ אֶת עַם יִשְׂרָאֵל?"

שִׁמְשׁוֹן מַחְלִיק בְּרַכּוּת עַל רֹאשׁוֹ שֶׁל בֶּן אֲחוֹתוֹ וּמְחַיֵּךְ. "כֵּן, טַרְפוֹן, יַלְדִּי הַקָּדוֹשׁ. הַיּוֹם תַּעֲמֹד אִתִּי עַל מַדְרֵגוֹת הַדּוּכָן הַקָּרוֹת. אַךְ זְכֹר – בְּבֵית הַמִּקְדָּשׁ, כְּשֶׁבָּאִים לְבָרֵךְ, צָרִיךְ לְנַקּוֹת אֶת הַלֵּב מִכָּל מַחֲשָׁבָה שֶׁל חֹל."

הַצַּעַר שֶׁל הַכֹּהֲנִים וְהַשֵּׁם שֶׁהֻסְתַּר

כַּאֲשֶׁר הֵם נִכְנָסִים דֶּרֶךְ שַׁעֲרֵי הָעֲזָרָה, קוֹל נְגִינַת נִבְלֵי הַלְוִיִּים מַרְעִיד אֶת הָאֲוִיר. תְּקִיעָה... תְּרוּעָה... תְּקִיעָה! צְלִיל חֲצוֹצְרוֹת הַכֶּסֶף מְפַלֵּחַ אֶת הַשָּׁמַיִם.

אַךְ בְּלִבּוֹ שֶׁל טַרְפוֹן מְנַקֶּרֶת צְבִיטָה קְטַנָּה. דּוֹדוֹ שִׁמְשׁוֹן לִמֵּד אוֹתוֹ סוֹד כּוֹאֵב שֶׁעָבַר בֵּין הַכֹּהֲנִים הַצְּנוּעִים וְהַקְּדוֹשִׁים. בַּיָּמִים שֶׁלִּפְנֵי כֵּן, כַּאֲשֶׁר כָּל עַם יִשְׂרָאֵל הָיָה שָׁלֵם וְטָהוֹר, אֲנָשִׁים הָיוּ מְבָרְכִים אֶת בִּרְכַּת הַכֹּהֲנִים תוֹךְ כְּדֵי לְבַטֵּא אֶת "הַשֵּׁם הַמְפֹרָשׁ" שֶׁל בּוֹרֵא עוֹלָם כְּמוֹ שֶׁהוּא נִכְתָּב. זֶהוּ שֵׁם כָּל כָּךְ קָדוֹשׁ, שֶׁמִּי שֶׁשּׁוֹמֵעַ אוֹתוֹ מַרְגִּישׁ שֶׁאֵשׁ טְהוֹרָה נִכְנֶסֶת אֶל תּוֹךְ נִשְׁמָתוֹ. אַךְ בַּדּוֹר הַזֶּה, דּוֹר חֻרְבַּן הַבַּיִת הַשֵּׁנִי, רַבּוּ פּוֹשְׁעִים וַאֲנָשִׁים שֶׁאֵינָם רְאוּיִים לִשְׁמֹעַ אֶת הַסּוֹד הָעֶלְיוֹן וּלְהִשְׁתַּמֵּשׁ בּוֹ.

עַל כֵּן, תִּקְּנוּ חֲכָמִים תַּקָּנָה חֲדָשָׁה. הַכֹּהֲנִים הַקְּדוֹשִׁים וְהַצְּנוּעִים – בְּשָׁעָה שֶׁבֵּרְכוּ אֶת הָעָם – הָיוּ מְנַגְּנִים בְּקוֹל עָבֶה וְחָזָק אֶת שְׁאָר הַמִּלִּים, וּבְתוֹךְ הַשִּׁירָה הַחֲזָקָה, בְּתוֹךְ הַמַּנְגִּינָה שֶׁהִתְגַּלְגְּלָה מִפִּי עַשְׂרוֹת כֹּהֲנִים שֶׁאָמְרוּ שֵׁמוֹת רְגִילִים שֶׁל קְדֻשָּׁה, הָיוּ אֶחָד אוֹ שְׁנַיִם מְבַלְעִים ("מַבְלִיעִים") בְּשֶׁקֶט בְּשֶׁקֶט אֶת הַשֵּׁם הַמְפֹרָשׁ, כְּדֵי שֶׁרַק מִי שֶׁרָאוּי וּטְהוֹר לֵב אָכֵן יִקְלֹט אוֹתוֹ.

טַרְפוֹן רָצָה, מֵעֹמֶק הַנְּשָׁמָה שֶׁלּוֹ, לִשְׁמֹעַ אֶת הַשֵּׁם שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא! "אֲנִי רוֹצֶה לְהִתְחַבֵּר אֶל מֶלֶךְ מַלְכֵי הַמְּלָכִים," חָשַׁב בְּלִבּוֹ הַקָּטָן וְהַפּוֹעֵם, "אֲבָל אֵיךְ אוּכַל לִשְׁמֹעַ אוֹתוֹ בְּתוֹךְ כָּל הָרַעַשׁ וְהַמַּנְגִּינָה?"

עַל מַדְרֵגוֹת הַדּוּכָן הַקָּרוֹת

הִגִּיעַ הָרֶגַע.

"כֹּהֲנִים!" קָרָא הַמְמֻנֶּה.

הַכֹּהֲנִים חָלְצוּ אֶת נַעֲלֵיהֶם, נָטְלוּ אֶת יְדֵיהֶם מִידֵי אֲחֵיהֶם הַלְוִיִּים, וְצָעֲדוּ מַעְלָה אֶל עֵבֶר הַדּוּכָן. מַדְרֵגוֹת הַשַּׁיִשׁ בָּעֲזָרָה הָיוּ קָרוֹת כְּקֶרַח לְמַגַּע כַּפּוֹת הָרַגְלַיִם הַיְּחֵפוֹת, אַךְ לִבּוֹ שֶׁל טַרְפוֹן בָּעַר כְּתַנּוּר לוֹהֵט. דּוֹדוֹ שִׁמְשׁוֹן הֶעֱמִיד אֶת הַיֶּלֶד לְצִדּוֹ.

וְהִנֵּה – הַשְׁגָּחָה פְּרָטִית מֻפְלָאָה מִן הַשָּׁמַיִם! לְצִדּוֹ שֶׁל שִׁמְשׁוֹן הַכֹּהֵן נֶעֱמָד לֹא אַחֵר מֵאֲשֶׁר... הַכֹּהֵן הַגָּדוֹל בִּכְבוֹדוֹ וּבְעַצְמוֹ! עָטוּף בְּבִגְדֵי לָבָן זַכִּים הַמַּבְהִיקִים תַּחַת אוֹר הַשֶּׁמֶשׁ הַיְרוּשַׁלְמִית, וּפָנָיו קוֹרְנוֹת אוֹר שֶׁל עֲנָוָה מַלְכוּתִית.

הַכֹּהֲנִים הֵרִימוּ אֶת יְדֵיהֶם כְּנֶגֶד גּוֹבַהּ כִּתְפֵיהֶם. מֵאוֹת כֹּהֲנִים פָּרְשׂוּ אֶת הַטַּלִּיתוֹת הַצְּחוֹרוֹת עַל רָאשֵׁיהֶם וְהִסְתִּירוּ אֶת הַפָּנִים. עַכְשָׁו אִישׁ לֹא רוֹאֶה דָּבָר, רַק שׁוֹמְעִים.

"יְבָרֶכְךָ..." הִתְחִילָה הַמַּנְגִּינָה הָעֲצוּמָה לְהִתְגַּלְגֵּל בֶּחָלָל כְּקוֹל שֶׁל מַיִם רַבִּים, אֲדִירִים. אֶת קוֹלָם שֶׁל מֵאוֹת הַכֹּהֲנִים לִוּוּ הַלְוִיִּים שֶׁפָּרְטוּ עַל מֵיתְרֵי הַנְּבָלִים. הָעֲזָרָה רָעֲדָה מִקְּדֻשָּׁה וּמֵרַעַשׁ תְּפִלָּה אָרֹךְ וְנִפְלָא.

הַהַטָּיָה וְהַנְּפִילָה עַל הַפָּנִים

וְאָז הִגִּיעוּ לְתֵבַת הַשֵּׁם מֵהַפָּסוּק הָרִאשׁוֹן.

הַמַּנְגִּינָה עָלְתָה וְגָבְרָה, הַכֹּהֲנִים מָשְׁכוּ אֶת קוֹלָם. טַרְפוֹן עָצַם אֶת עֵינָיו בְּחָזְקָה אֶל תּוֹךְ מַחְשַׁכֵּי הַטַּלִּית שֶׁל דּוֹדוֹ. אִם הוּא רַק יַקְשִׁיב לָרַעַשׁ שֶׁבַּחוּץ – הוּא יַפְסִיד אֶת הַסּוֹד. הוּא הֵבִין שֶׁכְּדֵי לִשְׁמֹעַ אֶת ה', הוּא חַיָּב לְנַתֵּק אֶת עַצְמוֹ מִכָּל קוֹלוֹת הַסְּבִיבָה!

הַיֶּלֶד עָשָׂה תְּנוּעָה קְטַנָּה: מִבְּלִי שֶׁאִישׁ יִרְאֶה, הוּא הִטָּה אֶת אָזְנוֹ וְקֵרֵב אֶת רֹאשׁוֹ הַקָּטָן אֶל מֵאֲחוֹרֵי גַּבּוֹ שֶׁל הַכֹּהֵן הַגָּדוֹל .

רַעַשׁ כָּבֵד עָטַף אֶת אָזְנוֹ הַיְמָנִית, אַךְ אָזְנוֹ הַשְּׂמָאלִית הָיְתָה מְכֻוֶּנֶת רַק אֶל שִׂפְתֵי הַכֹּהֵן הַגָּדוֹל.

וְאָז, מִתּוֹךְ הַשִּׁירָה, זָרַם קוֹל עָדִין, שָׁקֵט וְטָהוֹר לְהַפְלִיא. קוֹל שֶׁהָיָה בּוֹ אֶת כָּל אַהֲבַת הַשָּׁמַיִם בָּעוֹלָם, שֶׁנִּגֵּן אֶת אוֹתִיּוֹת "הַשֵּׁם הַמְפֹרָשׁ" כְּמַלְאָךְ הַמַּקְרִיא טַל שֶׁל בְּרָכָה. הַכֹּהֵן הַגָּדוֹל "הִבְלִיעַ" אֶת הַשֵּׁם, וְטַרְפוֹן... טַרְפוֹן שָׁמַע!

בְּאוֹתוֹ רֶגַע שֶׁהָאוֹתִיּוֹת הַקְּדוֹשׁוֹת נִכְנְסוּ בְּאָזְנוֹ וְחָדְרוּ אֶל תּוֹךְ נִשְׁמָתוֹ – הַגּוּף הַקָּטָן שֶׁלּוֹ כְּבָר לֹא הָיָה יָכוֹל לַעֲמֹד יַצִּיב עַל שְׁתֵּי רַגְלַיִם. אֵשׁ קְדוֹשָׁה וְיִרְאַת שָׁמַיִם מָלְאָה אֶת לִבּוֹ עַד אֲשֶׁר תָּפְסָה קְרִיסָה אֶת רַגְלָיו.

בְּלִי לַחְשֹׁב, לְלֹא שׁוּם שְׁלִיטָה, הַיֶּלֶד טַרְפוֹן כָּרַע קָדִימָה וְ... נָפַל עַל פָּנָיו בְּיִרְאָה! הַלֶּחִי שֶׁלּוֹ דָּבְקָה בְּרִצְפַּת הַשַּׁיִשׁ שֶׁל הַעֲזָרָה, וְהוּא קָרָא מֵעֹמֶק נִשְׁמָתוֹ עִם שְׁאָר הַכֹּהֲנִים הַקְּרוֹבִים: "בָּרוּךְ שֵׁם כְּבוֹד מַלְכוּתוֹ לְעוֹלָם וָעֶד!" .

הוּא לֹא זָז מִמְּקוֹמוֹ וְלֹא פָּקַח אֶת עֵינָיו עַד שֶׁהַקְּדֻשָּׁה חִלְחֲלָה בְּכָל רְמַ"ח אֵבָרָיו, עַד אֲשֶׁר הַשֵּׁם "נֶעֱלַם" שׁוּב וְהַתְּפִלָּה הִסְתַּיְּמָה.

הָאוֹר שֶׁמַּמְשִׁיךְ לִדְלֹק בַּחֲשֵׁכָה

יְלָדִים מְתוּקִים, שָׁנִים רַבּוֹת לְאַחַר מִכֵּן, בֵּית הַמִּקְדָּשׁ, לְמַרְבֵּה הַצַּעַר, נֶחְרַב. הַחֲצוֹצְרוֹת שָׁתְקוּ, בִּגְדֵי הַלָּבָן קֻפְּלוּ, וְרֵיחַ הַקְּטֹרֶת נֶעֱלַם מֵרְחוֹבוֹת יְרוּשָׁלַיִם.

אֲבָל אוֹתוֹ יֶלֶד כֹּהֵן, אוֹתוֹ יֶלֶד שֶׁהִתְאַמֵּץ וְהִטָּה אֶת אָזְנוֹ מֵאֲחוֹרֵי גַּב הַכֹּהֵן הַגָּדוֹל, גָּדַל וְהָפַךְ לְאֶחָד מִגְּדוֹלֵי עַנְקֵי הָרוּחַ שֶׁל דּוֹרוֹ – הַתַּנָּא רַבִּי טַרְפוֹן ! חַכְמֵי יִשְׂרָאֵל הָיוּ קוֹרְאִים לוֹ בְּאַהֲבָה: "אַבָּא שֶׁל כָּל יִשְׂרָאֵל".

הַלֵּב שֶׁלּוֹ, שֶׁשָּׁמַע בְּיַלְדוּתוֹ אֶת הַשֵּׁם הַמְפֹרָשׁ בְּאֹזֶן וּבִקְדֻשָּׁה, סָפַג אֶת הָאוֹר שֶׁל בֵּית הַמִּקְדָּשׁ, וְרַבִּי טַרְפוֹן הֵבִיא אֶת הָאוֹר הַזֶּה אֶל תּוֹךְ בָּתֵּי הַמִּדְרָשׁ בְּיַבְנֶה וּבְלוֹד. הוּא מָסַר אֶת תּוֹרַת הַמִּקְדָּשׁ לְתַלְמִידוֹ הָעֲנָק רַבִּי עֲקִיבָא, וְלִמֵּד אוֹתָנוּ שֶׁגַּם כְּשֶׁבַּחוּץ חָשׁוּךְ – אִם רַק נָטֶה אוֹזֶן, תָּמִיד נוּכַל לִשְׁמֹעַ אֶת הַקְּדֻשָּׁה.

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “זכה לשמוע את השם המפורש מהכהן הגדול ולראות את העבודה בבית המקדש” על רבי טרפון? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “עָטוּף בְּבִגְדֵי לָבָן זַכִּים הַמַּבְהִיקִים תַּחַת אוֹר הַשֶּׁמֶשׁ הַיְרוּשַׁלְמִית, וּפָנָיו קוֹרְנוֹת אוֹר שֶׁל עֲנָוָה מַלְכוּתִית.”.
ג. אילו רבי טרפון לא היה פועל כפי שמתואר ב־“קוֹל שֶׁהָיָה בּוֹ אֶת כָּל אַהֲבַת הַשָּׁמַיִם בָּעוֹלָם, שֶׁנִּגֵּן אֶת אוֹתִיּוֹת "הַשֵּׁם הַמְפֹרָשׁ" כְּמַלְאָךְ הַמַּקְרִיא…” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “הַחֲצוֹצְרוֹת שָׁתְקוּ, בִּגְדֵי הַלָּבָן קֻפְּלוּ, וְרֵיחַ הַקְּטֹרֶת נֶעֱלַם מֵרְחוֹבוֹת יְרוּשָׁלַיִם.” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. זֶהוּ לֹא רֵיחַ שֶׁל פְּרָחִים רְגִילִים, וְלֹא רֵיחַ שֶׁל מַאֲפִיָּה.
2. אַךְ זְכֹר – בְּבֵית הַמִּקְדָּשׁ, כְּשֶׁבָּאִים לְבָרֵךְ, צָרִיךְ…
3. מַדְרֵגוֹת הַשַּׁיִשׁ בָּעֲזָרָה הָיוּ קָרוֹת כְּקֶרַח לְמַגַּע…
4. אִם הוּא רַק יַקְשִׁיב לָרַעַשׁ שֶׁבַּחוּץ – הוּא יַפְסִיד אֶת הַסּוֹד.
5. בְּלִי לַחְשֹׁב, לְלֹא שׁוּם שְׁלִיטָה, הַיֶּלֶד טַרְפוֹן כָּרַע…
6. הוּא מָסַר אֶת תּוֹרַת הַמִּקְדָּשׁ לְתַלְמִידוֹ הָעֲנָק רַבִּי…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “טַרְפוֹן רָצָה, מֵעֹמֶק הַנְּשָׁמָה שֶׁלּוֹ, לִשְׁמֹעַ אֶת הַשֵּׁם שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא!”.
מכתב: כתבו לרבי טרפון בעקבות “הַיֶּלֶד עָשָׂה תְּנוּעָה קְטַנָּה: מִבְּלִי שֶׁאִישׁ יִרְאֶה, הוּא הִטָּה אֶת אָזְנוֹ וְקֵרֵב אֶת רֹאשׁוֹ הַקָּטָן אֶל…”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “הַלֶּחִי שֶׁלּוֹ דָּבְקָה בְּרִצְפַּת הַשַּׁיִשׁ שֶׁל הַעֲזָרָה, וְהוּא קָרָא מֵעֹמֶק נִשְׁמָתוֹ עִם שְׁאָר הַכֹּהֲנִים…”.
דיווח: כתבו ידיעה קצרה שכותרתה “זכה לשמוע את השם המפורש מהכהן הגדול ולראות את העבודה בבית המקדש”.

המחזה

7. חיבור לחיים מתוך המעשה

א. מה אפשר ללמוד מן הדרך שבה רבי טרפון פעל באירוע “אֶת קוֹלָם שֶׁל מֵאוֹת הַכֹּהֲנִים לִוּוּ הַלְוִיִּים שֶׁפָּרְטוּ עַל מֵיתְרֵי הַנְּבָלִים.”?
