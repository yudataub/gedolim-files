יריקה בפני הדרשן — שלום בית

דף עבודה בעקבות רבי מאיר
שם: כיתה:

המעשה — הסיפור המלא

פְּתִיחָה: אוֹר הַתּוֹרָה בְּלֵיל שַׁבָּת

יְלָדִים, עִצְמוּ רֶגַע אֶת הָעֵינַיִם. דַּמְיְנוּ אֶת לֵיל שַׁבָּת בְּעִירַת חַמַּת אֲשֶׁר סְמוּכָה לִטְבֶרְיָה. בַּחוּץ נוֹשֶׁבֶת רוּחַ קְרִירָה, הָרְחוֹבוֹת חֲשׁוּכִים, אֲבָל מִתּוֹךְ בֵּית הַכְּנֶסֶת הַגָּדוֹל בּוֹקֵעַ אוֹר צָהֹב וְחַם שֶׁל נֵרוֹת שֶׁמֶן, וְנִשְׁמָע קוֹל עָמֹק וּמָתוֹק. קוֹל שֶׁנִּכְנַס יָשָׁר לַלֵּב.

זֶהוּ קוֹלוֹ שֶׁל גְּדוֹל הַדּוֹר, הַתַּנָּא הַקָּדוֹשׁ רַבִּי מֵאִיר .

בְּכָל לֵיל שַׁבָּת, לְאַחַר הַסְּעוּדָה, כְּשֶׁהַבֶּטֶן מְלֵאָה בְּדָגִים וּבְמָרָק חַם, הָיוּ אֲנָשִׁים וְנָשִׁים מִכָּל רַחֲבֵי הָעִיר מִתְאַסְּפִים כְּדֵי לִשְׁמֹעַ אֶת הַדְּרָשָׁה שֶׁלּוֹ. רַבִּי מֵאִיר לֹא רַק דִּבֵּר, הוּא צִיֵּר מִלִּים! מֵאוֹת אֲנָשִׁים יָשְׁבוּ צְפוּפִים, מַחְזִיקִים אֶת נְשִׁימָתָם, וְהִקְשִׁיבוּ לְדִבְרֵי הַתּוֹרָה שֶׁנָּטְפוּ מִפִּיו כִּדְבַשׁ.

בְּעֶזְרַת הַנָּשִׁים, יָשְׁבָה אִשָּׁה אַחַת צַדֶּקֶת. הִיא כָּל כָּךְ אָהֲבָה אֶת לִמּוּד הַתּוֹרָה, הִיא בָּלְעָה בְּצִמָּאוֹן כָּל פָּסוּק, כָּל הֲלָכָה. מֵרֹב שֶׁהַדְּרָשָׁה הָיְתָה מְרַתֶּקֶת, הִיא שָׁכְחָה מֵהַזְּמַן. הִיא לֹא שָׂמָה לֵב שֶׁהַלַּיְלָה מִתְאָרֵךְ, שֶׁנֵּרוֹת בֵּית הַכְּנֶסֶת כְּבָר מַתְחִילִים לְהַבְהֵב וְלִדְעֹךְ.

הַבַּיִת הַחָשׁוּךְ וְהַשְּׁבוּעָה הַנּוֹרָאָה

כְּשֶׁסִּיֵּם רַבִּי מֵאִיר אֶת דְּרָשָׁתוֹ, הִזְדָּרְזָה הָאִשָּׁה וְיָצְאָה אֶל קֹר הַלַּיְלָה. הִיא הָלְכָה בִּמְהִירוּת בַּסִּמְטְאוֹת הַחֲשׁוּכוֹת. לִבָּהּ פָּעַם בְּשִׂמְחָה שֶׁל מִצְוָה, אֲבָל גַּם בִּדְאָגָה קְטַנָּה.

כְּשֶׁהִגִּיעָה לְבֵיתָהּ, פָּתְחָה אֶת הַדֶּלֶת בַּחֲשָׁשׁ... חֹשֶׁךְ. נֵרוֹת הַשַּׁבָּת כְּבָר כָּבוּ מִזְּמַן.

וּבַחֹשֶׁךְ הַזֶּה, עָמַד בַּעֲלָהּ. יֵצֶר הָרַע שֶׁל הַכַּעַס הִשְׁתַּלֵּט עָלָיו וְסִנְוֵר אֶת עֵינָיו. פָּנָיו בָּעֲרוּ מִזַּעַם.

"הֵיכָן הָיִית עַד עַכְשָׁו?!" קוֹלוֹ חָתַךְ אֶת הַשֶּׁקֶט שֶׁל שַׁבָּת כְּמוֹ רַעַם דּוֹרְסָנִי.

"בַּעֲלִי הַיָּקָר," הִיא לָחֲשָׁה בְּקוֹל רוֹעֵד, מַשְׁפִּילָה אֶת עֵינֶיהָ, "הָיִיתִי בְּבֵית הַמִּדְרָשׁ. הַדְּרָשָׁה שֶׁל רַבִּי מֵאִיר הָיְתָה כָּל כָּךְ קְדוֹשָׁה וְאֲרֻכָּה... לֹא שַׂמְתִּי לֵב לַשָּׁעָה."

בִּמְקוֹם לְהִתְרַגֵּשׁ שֶׁאִשְׁתּוֹ אוֹהֶבֶת תּוֹרָה, הַכַּעַס שֶׁל הַבַּעַל הִתְגַּבֵּר עוֹד יוֹתֵר. הוּא סָגַר אֶת הַדֶּלֶת וְהִשְׁאִיר אוֹתָהּ בַּחוּץ. וְאָז, מִתּוֹךְ רֹגֶז מֻחְלָט וְטִפְּשׁוּת, הוּא קָרָא מִבַּעַד לַדֶּלֶת אֶת הַמִּשְׁפָּט הֶחָמוּר בְּיוֹתֵר שֶׁיָּכוֹל הָיָה לוֹמַר:

"אֲנִי נִשְׁבָּע! אַתְּ לֹא תִּכָּנְסִי לַבַּיִת הַזֶּה לְעוֹלָם... עַד שֶׁתֵּלְכִי אֶל הַדַּרְשָׁן שֶׁלָּךְ, וְתִירְקִי לוֹ מַמָּשׁ בְּתוֹךְ הַפַּרְצוּף!"

יְלָדִים, שֶׁקֶט נָפַל בָּרְחוֹב. הָאִשָּׁה עָמְדָה שָׁם, קוֹפֵאת מִקֹּר, וּדְמָעוֹת חַמּוֹת זָלְגוּ עַל לְחָיֶיהָ. לִירוֹק בְּפָנָיו שֶׁל רַבִּי מֵאִיר?! בַּפָּנִים שֶׁל צַדִּיק יְסוֹד עוֹלָם?! גְּדוֹל תַּלְמִידָיו שֶׁל רַבִּי עֲקִיבָא?! הִיא יָדְעָה שֶׁהִיא תַּעֲדִיף לִישֹׁן בָּרְחוֹב כָּל חַיֶּיהָ, וְלֹא לַעֲשׂוֹת מַעֲשֶׂה נוֹרָא כָּזֶה שֶׁל בִּזּוּי כְּבוֹד תּוֹרָה.

וְכָךְ חָלְפוּ שְׁבוּעוֹת! שָׁבוּעַ אַחַר שָׁבוּעַ שֶׁהָאִשָּׁה גָּרָה אֵצֶל שְׁכֵנוֹתֶיהָ, מְבַקֶּשֶׁת רַחֲמִים, וְהַבַּעַל בְּכַעֲסוֹ – עוֹמֵד בְּשֶׁלּוֹ.

עֵינוֹ שֶׁל הַצַּדִּיק בּוֹעֶרֶת

אֲבָל מִשָּׁמַיִם רָאוּ אֶת כְּאֵבָהּ. וְלֹא רַק מִשָּׁמַיִם.

הַקָּדוֹשׁ בָּרוּךְ הוּא גִּלָּה אֶת סוֹד הַמְּרִיבָה הַזּוֹ לְרַבִּי מֵאִיר! רַבִּי מֵאִיר יָדַע בְּרוּחַ הַקֹּדֶשׁ שֶׁיֵּשׁ בָּעִיר אִשָּׁה שֶׁבֵּיתָהּ נֶחֱרַב בִּגְלָלוֹ, גַּם אִם הוּא לֹא אָשֵׁם בִּכְלָל.

הוּא יָכוֹל הָיָה לוֹמַר: "שֶׁהַבַּעַל הַטִּפֵּשׁ הַזֶּה יָבוֹא לְבַקֵּשׁ סְלִיחָה!". אֲבָל זֶה לֹא מָה שֶׁעָשָׂה הַצַּדִּיק. רַבִּי מֵאִיר חִפֵּשׂ דֶּרֶךְ לְהַחְזִיר אֶת הַשָּׁלוֹם, אֲפִלּוּ אִם זֶה יַעֲלֶה לוֹ בְּבִזְיוֹן נוֹרָא.

וִילָדִים, שִׂימוּ לֵב לַחָכְמָה. בְּאַחַד הַיָּמִים, יָשַׁב רַבִּי מֵאִיר בְּבֵית הַמִּדְרָשׁ הַגָּדוֹל. הַמָּקוֹם הָיָה הוֹמֶה מֵאוֹת תַּלְמִידִים מְבֻגָּרִים וַחֲשׁוּבִים. לְפֶתַע, עָצַם רַבִּי מֵאִיר אֶת עֵינָיו, תָּפַס אֶת עֵינוֹ הַיְמָנִית בְּיָדוֹ, וְעָשָׂה עַצְמוֹ כְּמִי שֶׁסּוֹבֵל מִכְּאֵבִים עַזִּים!

"אֲחָי וְתַלְמִידַי," קָרָא רַבִּי מֵאִיר בְּקוֹל נֶאֱנָח, "עֵינִי כּוֹאֶבֶת עַד מְאֹד. הַאִם יֵשׁ מִי שֶׁיּוֹדַעַת לִלְחֹשׁ לַחַשׁ וּתְפִלָּה עַל הָעַיִן כְּדֵי שֶׁתִּתְרַפֵּא?"

הַשְּׁכֵנוֹת שֶׁל אוֹתָהּ אִשָּׁה עֲגוּנָה וַעֲצוּבָה שָׁמְעוּ זֹאת, וּמִיָּד רָצוּ אֵלֶיהָ.

"זוֹ הַהִזְדַּמְּנוּת שֶׁלָּךְ!" קָרְאוּ הַשְּׁכֵנוֹת, לְקוֹחוֹת אוֹתָהּ בַּיָּד. "תַּעֲשִׂי אֶת עַצְמֵךְ כְּאִלּוּ אַתְּ מְרַפְּאָה אוֹתוֹ, וְתִירְקִי בְּעֵינוֹ מַיִם, וְכָךְ יִפְתַּר הַתְּנַאי שֶׁל בַּעֲלֵךְ וְתַחְזְרִי הַבַּיְתָה!".

שֶׁבַע שֶׁל בּוּשָׁה עֲבוּר הַשָּׁלוֹם

הָאִשָּׁה נִדְחֲפָה לְתוֹךְ בֵּית הַמִּדְרָשׁ הֶעָנָק. מֵאוֹת זוּגוֹת עֵינַיִם שֶׁל רַבָּנִים שָׁתְקוּ וְהִבִּיטוּ בָּהּ. רַגְלֶיהָ רָעֲדוּ כְּמוֹ עֲלֵה טֶרֶף בָּרוּחַ. הִיא נִגְּשָׁה לַשֻּׁלְחָן שֶׁל נְשִׂיא הַדּוֹר, שֶׁעָמַד שָׁם מַחְזִיק אֶת עֵינוֹ.

"בִּתִּי," שָׁאַל בְּרֹךְ, "הַאִם אַתְּ יוֹדַעַת לִלְחֹשׁ לַחַשׁ שֶׁמְּרַפֵּא אֶת הָעַיִן?"

הָאִשָּׁה רָאֲתָה אֶת הָאוֹר הַקָּדוֹשׁ שֶׁהֵאִיר מִפָּנָיו שֶׁל רַבִּי מֵאִיר, וְלֹא יָכְלָה לְשַׁקֵּר. הִיא פָּרְצָה בִּבְכִי וְאָמְרָה: "רַבִּי... אֲנִי לֹא מְבִינָה כְּלוּם בִּרְפוּאָה. הַשְּׁכֵנוֹת סְתָם הֵבִיאוּ אוֹתִי הֵנָּה... אֵינֶנִּי יוֹדַעַת כְּלוּם!"

רַבִּי מֵאִיר לֹא הִתְעַצְבֵּן. הוּא הִבִּיט בָּהּ מִתּוֹךְ אַהֲבָה עֲצוּמָה וְאָמַר לָהּ מִשְׁפָּט מַדְהִים:

"לֹא מְשַׁנֶּה שֶׁאֵינֵךְ יוֹדַעַת. תִּתְקָרְבִי קָרוֹב אֵלַי... וְתִירְקִי מַיִם בְּתוֹךְ הָעַיִן שֶׁלִּי. שֶׁבַע פְּעָמִים! "

"מָה?!" נָשְׁמוּ הַתַּלְמִידִים הֶהֲמוּמִים סָבִיב.

אֲבָל רַבִּי מֵאִיר הִתְעַקֵּשׁ. הָאִשָּׁה, בְּיָדַיִם רוֹעֲדוֹת וּמִתּוֹךְ מִשְׁמַעַת לַצַּדִּיק, הִתְקָרְבָה, וְיָרְקָה אֶל עֵבֶר עֵינוֹ, מַמָּשׁ כְּמוֹ תְּרוּפַת עַמְמִית, וְלֹא פַּעַם אַחַת – אֶלָּא שֶׁבַע פְּעָמִים רְצוּפוֹת! כָּל תַּלְמִידֵי הַיְשִׁיבָה הורידו מבט בְּבּוּשָׁה אֲיֻמָּה עַל כְּבוֹד רַבָּם הַנִּרְמָס.

כְּשֶׁסִּיְּמָה, חִיֵּךְ אֵלֶיהָ רַבִּי מֵאִיר פָּנִים מְאִירוֹת. "תּוֹדָה רַבָּה לָךְ בִּתִּי. עַכְשָׁו, רְצִי מַהֵר לְבַעֲלֵךְ וְתַגִּידִי לוֹ אֶת הַמִּלִּים הָאֵלֶּה: 'אַתָּה, בַּעֲלִי, דָּרַשְׁתָּ שֶׁאִירוֹק בְּעֵינוֹ שֶׁל רַבִּי מֵאִיר פַּעַם אַחַת? תֵּדַע לְךָ שֶׁאֲנִי יָרַקְתִּי לֹא פַּעַם אַחַת... אֶלָּא שֶׁבַע פְּעָמִים!'. לְכִי לְשָׁלוֹם לְבֵיתֵךְ."

כָּבוֹד אוֹ שָׁלוֹם?

כְּשֶׁהָאִשָּׁה יָצְאָה, מְאֻשֶּׁרֶת, בַּדֶּרֶךְ לַחְזֹר לְבַעֲלָהּ, בֵּית הַמִּדְרָשׁ כִּמְעַט הִתְפּוֹצֵץ!

הַתַּלְמִידִים הַגְּדוֹלִים קָמוּ מִמְּקוֹמוֹתֵיהֶם, כּוֹעֲסִים מְאֹד. "רַבֵּנוּ!!!" זָעֲקוּ הַתַּלְמִידִים, "אֵיךְ נָתַנְתָּ לָזֶה לִקְרוֹת?! זֶה בִּזּוּי כְּבוֹד תּוֹרָה לַעֲשׂוֹת צְחוֹק כָּזֶה מִכְּבוֹדְךָ! אִם הָיִיתָ אוֹמֵר לָנוּ, הָיִינוּ מְבִיאִים אֶת הַבַּעַל הֶחָצוּף הַזֶּה לְכָאן, מַכִּים אוֹתוֹ בְּמַקְּלוֹת, וּמְכְרִיחִים אוֹתוֹ לְבַקֵּשׁ מִמֶּךָּ סְלִיחָה וּלְהַחְזִיר אֶת אִשְׁתּוֹ לַבַּיִת!"

עָמַד רַבִּי מֵאִיר, מָחָה אֶת פָּנָיו, שֶׁקֶט וְשָׁלֵו. הוּא פָּתַח אֶת יָדָיו וְאָמַר מִשְׁפָּט אֶחָד, שֶׁמְּהַדְהֵד עַד הַיּוֹם וּמְלַמֵּד אוֹתָנוּ מָהוּ סוֹד הַשָּׁלוֹם:

"תַּלְמִידַי הַיְקָרִים... וְכִי הַכָּבוֹד שֶׁל מֵאִיר גָּדוֹל יוֹתֵר מֵהַכָּבוֹד שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא בִּכְבוֹדוֹ וּבְעַצְמוֹ?"

הַתַּלְמִידִים הִשְׁתַּתְּקוּ.

"הֲרֵי צִוָּה הַקָּדוֹשׁ בָּרוּךְ הוּא בַּתּוֹרָה," הִמְשִׁיךְ רַבִּי מֵאִיר בְּקוֹל נֶאֱצָל, "שֶׁכַּאֲשֶׁר יֵשׁ מְרִיבָה קָשָׁה וַחֲשָׁדוֹת בֵּין בַּעַל לְאִשְׁתּוֹ (בְּפָרָשַׁת סוֹטָה), יִקְּחוּ אֶת הַשֵּׁם הַקָּדוֹשׁ שֶׁל ה', שֶׁנִּכְתַּב עַל מְגִלַּת הַקְּלָף בִּקְדֻשָּׁה, וְיִמְחֲקוּ אוֹתוֹ בְּתוֹךְ כּוֹס מַיִם רַק כְּדֵי שֶׁהַבַּעַל וְהָאִשָּׁה יַשְׁלִימוּ וְיַחְזְרוּ לֶאֱהֹב אֶחָד אֶת הַשְּׁנִיָּה מֵחָדָשׁ! אִם שֵׁם ה' יִתְבָּרַךְ נִמְחָק וּמִתְבַּזֶּה בַּמַּיִם עֲבוּר שְׁלוֹם בַּיִת – מִי אֲנִי הַקָּטָן שֶׁאֶשְׁמֹר עַל הַכָּבוֹד שֶׁלִּי וְאֶתֵּן לַבַּיִת הַזֶּה לְהֵהָרֵס?!"

אוֹתוֹ בַּעַל כַּעֲסָן, כְּשֶׁשָּׁמַע מֵאִשְׁתּוֹ מָה שֶׁעָשָׂה עֲבוּרָם גְּדוֹל הַדּוֹר רַבִּי מֵאִיר, נִשְׁבַּר לִבּוֹ. הוּא הֵבִין שֶׁעָשָׂה טָעוּת נוֹרָאָה. הוּא כִּבֵּד אֶת אִשְׁתּוֹ מֵאוֹתוֹ יוֹם תָּמִיד, וְהַשָּׁלוֹם, הַשָּׁלוֹם הָאֲמִתִּי, חָזַר לְאוֹתוֹ הַבַּיִת – וְהַכֹּל בִּזְכוּת דִּמְעָה, יְרִיקָה, וְצַדִּיק אֶחָד גָּדוֹל שֶׁיָּדַע לְוַתֵּר.

מקורות: [1] התלמוד הירושלמי , מסכת סוטה, פרק א', הלכה ד': הסיפור המקורי על דרשתו של רבי מאיר בבית הכנסת בחמת, האישה שאיחרה, שבועת הבעל, וחכמתו של רבי מאיר שוויתר על כבודו. 1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “דַּמְיְנוּ אֶת לֵיל שַׁבָּת בְּעִירַת חַמַּת אֲשֶׁר סְמוּכָה לִטְבֶרְיָה.” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “כְּשֶׁסִּיֵּם רַבִּי מֵאִיר אֶת דְּרָשָׁתוֹ, הִזְדָּרְזָה הָאִשָּׁה וְיָצְאָה אֶל קֹר הַלַּיְלָה.”?
ג. כיצד הגיב רבי מאיר למצב המתואר: “הָאִשָּׁה עָמְדָה שָׁם, קוֹפֵאת מִקֹּר, וּדְמָעוֹת חַמּוֹת זָלְגוּ עַל לְחָיֶיהָ.”?
ד. מה השתנה בין “בְּאַחַד הַיָּמִים, יָשַׁב רַבִּי מֵאִיר בְּבֵית הַמִּדְרָשׁ הַגָּדוֹל.” לבין “הִיא נִגְּשָׁה לַשֻּׁלְחָן שֶׁל נְשִׂיא הַדּוֹר, שֶׁעָמַד שָׁם מַחְזִיק אֶת עֵינוֹ.”?
ה. כיצד הגיע הסיפור אל האירוע: “אִם הָיִיתָ אוֹמֵר לָנוּ, הָיִינוּ מְבִיאִים אֶת הַבַּעַל הֶחָצוּף הַזֶּה לְכָאן, מַכִּים אוֹתוֹ בְּמַקְּלוֹת, וּמְכְרִיחִים אוֹתוֹ לְבַקֵּשׁ מִמֶּךָּ…”?

לפני ואחרי

מה קרה לפני: “הוּא יָכוֹל הָיָה לוֹמַר: "שֶׁהַבַּעַל הַטִּפֵּשׁ הַזֶּה יָבוֹא לְבַקֵּשׁ סְלִיחָה!".”?

סיבה ותוצאה מתוך המעשה

האירוע הנתוןמה גרם לו?מה הייתה תוצאתו?
הַדְּרָשָׁה שֶׁל רַבִּי מֵאִיר הָיְתָה כָּל כָּךְ קְדוֹשָׁה וְאֲרֻכָּה...
הַשְּׁכֵנוֹת שֶׁל אוֹתָהּ אִשָּׁה עֲגוּנָה וַעֲצוּבָה שָׁמְעוּ זֹאת, וּמִיָּד רָצוּ אֵלֶיהָ.

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “יריקה בפני הדרשן — שלום בית” על רבי מאיר? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “הַקָּדוֹשׁ בָּרוּךְ הוּא גִּלָּה אֶת סוֹד הַמְּרִיבָה הַזּוֹ לְרַבִּי מֵאִיר!”.
ג. אילו רבי מאיר לא היה פועל כפי שמתואר ב־“הִיא נִגְּשָׁה לַשֻּׁלְחָן שֶׁל נְשִׂיא הַדּוֹר, שֶׁעָמַד שָׁם מַחְזִיק אֶת עֵינוֹ.” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “אִם הָיִיתָ אוֹמֵר לָנוּ, הָיִינוּ מְבִיאִים אֶת הַבַּעַל הֶחָצוּף הַזֶּה לְכָאן, מַכִּים אוֹתוֹ בְּמַקְּלוֹת, וּמְכְרִיחִים…” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. דַּמְיְנוּ אֶת לֵיל שַׁבָּת בְּעִירַת חַמַּת אֲשֶׁר סְמוּכָה…
2. לִבָּהּ פָּעַם בְּשִׂמְחָה שֶׁל מִצְוָה, אֲבָל גַּם בִּדְאָגָה קְטַנָּה.
3. הִיא יָדְעָה שֶׁהִיא תַּעֲדִיף לִישֹׁן בָּרְחוֹב כָּל חַיֶּיהָ, וְלֹא…
4. אֲחָי וְתַלְמִידַי," קָרָא רַבִּי מֵאִיר בְּקוֹל נֶאֱנָח, "עֵינִי…
5. הָאִשָּׁה, בְּיָדַיִם רוֹעֲדוֹת וּמִתּוֹךְ מִשְׁמַעַת לַצַּדִּיק,…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “הָאִשָּׁה עָמְדָה שָׁם, קוֹפֵאת מִקֹּר, וּדְמָעוֹת חַמּוֹת זָלְגוּ עַל לְחָיֶיהָ.”.
מכתב: כתבו לרבי מאיר בעקבות “הַשְּׁכֵנוֹת שֶׁל אוֹתָהּ אִשָּׁה עֲגוּנָה וַעֲצוּבָה שָׁמְעוּ זֹאת, וּמִיָּד רָצוּ אֵלֶיהָ.”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “כָּל תַּלְמִידֵי הַיְשִׁיבָה הורידו מבט בְּבּוּשָׁה אֲיֻמָּה עַל כְּבוֹד רַבָּם הַנִּרְמָס.”.
דיווח: כתבו ידיעה קצרה שכותרתה “יריקה בפני הדרשן — שלום בית”.

רגע מן הסיפורהדמויותמה יאמרו או יעשו?

הִיא נִגְּשָׁה לַשֻּׁלְחָן שֶׁל נְשִׂיא הַדּוֹר, שֶׁעָמַד שָׁם מַחְזִיק אֶת עֵינוֹ.
