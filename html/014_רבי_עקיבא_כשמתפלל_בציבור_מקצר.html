סיפור 14 • רבי עקיבא • כשמתפלל בציבור מקצר

כשמתפלל בציבור מקצר

דף עבודה בעקבות רבי עקיבא
שם: כיתה:

המעשה — הסיפור המלא

פְּתִיחָה: שְׁקִיעָה בְּעִיר הַתּוֹרָה

יְלָדִים, עִצְמוּ אֶת הָעֵינַיִם וְנַסּוּ לְדַמְיֵן שֶׁאֲנַחְנוּ צוֹעֲדִים עַכְשָׁו בִּשְׁבִילֵי הֶעָפָר שֶׁל הָעִיר בְּנֵי בְּרַק, לִפְנֵי כִּמְעַט אַלְפַּיִם שָׁנָה. הַשֶּׁמֶשׁ מַתְחִילָה לָרֶדֶת וְצוֹבַעַת אֶת הַשָּׁמַיִם בִּצְבָעִים שֶׁל כָּתֹם, אָדֹם וְזָהָב. בָּאֲוִיר אֶפְשָׁר לְהָרִיחַ אֶת הָרֵיחַ הַמָּתוֹק שֶׁל עֲצֵי הַתְּאֵנָה וְהַזַּיִת.

הַיּוֹם מִסְתַּיֵּם. מִן הַשָּׂדוֹת הַגְּדוֹלִים פּוֹסְעִים חֲזָרָה הַחַקְלָאִים הַיְּהוּדִים. צַעֲדֵיהֶם אִטִּיִּים, וְקַבַּיִם שֶׁל עֵץ לְרַגְלֵיהֶם. הֵם עָבְדוּ קָשֶׁה מְאֹד מֵאָז עֲלוֹת הַשַּׁחַר – חָרְשׁוּ, זָרְעוּ, נָטְעוּ וְקָצְרוּ. הַיָּדַיִם שֶׁלָּהֶם מְלֵאוֹת בְּיַבָּלוֹת, הַפָּנִים שֶׁלָּהֶם מְכֻסּוֹת בְּזֵעָה וּבְאָבָק, וְהַבֶּטֶן שֶׁלָּהֶם כְּבָר מְקַרְקֶרֶת מֵרָעָב. הֵם רוֹצִים רַק דָּבָר אֶחָד: לְהַגִּיעַ לַבַּיִת, לִשְׁתּוֹת מַיִם קָרִים, לֶאֱכֹל פְּרוּסַת לֶחֶם חַמָּה, וְלָנוּחַ מֵהֶעָמָל הַקָּשֶׁה.

אַךְ לִפְנֵי שֶׁהֵם פּוֹנִים לַבָּתִּים שֶׁלָּהֶם, יֵשׁ לָהֶם מָקוֹם חָשׁוּב יוֹתֵר לָלֶכֶת אֵלָיו. הֵם מְנַעֲרִים אֶת הָאָבָק מִבִּגְדֵיהֶם, חוֹבְשִׁים מַהֵר אֶת הַכִּפּוֹת שֶׁלָּהֶם, וְנִכְנָסִים בְּיִרְאַת שָׁמַיִם אֶל בֵּית הַמִּדְרָשׁ הַגָּדוֹל, כְּדֵי לְהִתְפַּלֵּל תְּפִלַּת עַרְבִית שֶׁל חֹל אֶל הַקָּדוֹשׁ בָּרוּךְ הוּא, אֲשֶׁר נָתַן לָהֶם כֹּחַ לַעֲבֹד בְּמֶשֶׁךְ כָּל הַיּוֹם.

הַהַמְתָּנָה לַתַּנָּא הַקָּדוֹשׁ

בֵּית הַמִּדְרָשׁ כְּבָר מָלֵא בַּאֲנָשִׁים. בַּסַּפְסָלִים הָרִאשׁוֹנִים יוֹשְׁבִים תַּלְמִידֵי חֲכָמִים הַלּוֹמְדִים גְּמָרָא מִתּוֹךְ סְפָרִים וּגְוִילִים. בַּסַּפְסָלִים הָאֲחוֹרִיִּים יוֹשְׁבִים הַפּוֹעֲלִים הַפְּשׁוּטִים, הַסַּנְדְּלָרִים וְהַחַקְלָאִים. כֻּלָּם מְחַכִּים.

לְפֶתַע, מִשְׁתָּרֵר שֶׁקֶט מֻחְלָט בָּאוּלָם. אֲפִלּוּ צִיּוּץ שֶׁל צִפּוֹר בַּחוּץ נִשְׁמַע פִּתְאוֹם חָזָק. אָדָם בַּעַל הַדְרַת פָּנִים וְזָקָן לָבָן אָרֹךְ, נִכְנַס בַּדֶּלֶת. מִפָּנָיו קוֹרֵן אוֹר מְיֻחָד שֶׁל קְדֻשָּׁה שֶׁקָּשֶׁה לְתָאֵר בְּמִלִּים. זֶהוּ גְּדוֹל הַדּוֹר, רַבִּי עֲקִיבָא . כֻּלָּם קָמִים עַל רַגְלֵיהֶם מִתּוֹךְ כָּבוֹד עָצוּם לַתּוֹרָה הַקְּדוֹשָׁה שֶׁהוּא שׁוֹנֶה לָהֶם מִדֵּי יוֹם.

הַגַּבַּאי פוֹנֶה בַּעֲדִינוּת אֶל רַבִּי עֲקִיבָא וּמְבַקֵּשׁ: "רַבֵּנוּ, הַאִם תּוֹאִיל לַעֲבֹר לִפְנֵי הַתֵּבָה? הַאִם תּוֹאִיל לִהְיוֹת הַחַזָּן שֶׁלָּנוּ בִּתְפִלַּת עַרְבִית?".

רַבִּי עֲקִיבָא מְהַנְהֵן בְּרֹאשׁוֹ וְנִגָּשׁ בַּעֲנָוָה אֶל מִזְרַח בֵּית הַכְּנֶסֶת.

יְלָדִים, אִם גְּדוֹל הַדּוֹר הוּא הַחַזָּן שֶׁלָּכֶם עַכְשָׁו – לְמָה אַתֶּם מְצַפִּים?

בְּוַדַּאי אַתֶּם מְדַמְיְנִים שֶׁהַתְּפִלָּה שֶׁל רַבִּי עֲקִיבָא תִּקַּח שָׁעוֹת אֲרֻכּוֹת! הֲרֵי הוּא מִתְפַּלֵּל וּמְכַוֵּן סוֹדוֹת עֲצוּמִים לְרִבּוֹנוֹ שֶׁל עוֹלָם! הַחַקְלָאִים הֶעָיֵפִים לָקְחוּ נְשִׁימָה אֲרֻכָּה, וְהֵכִינוּ אֶת עַצְמָם לַעֲמֹד עַל הָרַגְלַיִם הַכּוֹאֲבוֹת שֶׁלָּהֶם זְמַן רַב, כִּי הֲרֵי רַבִּי עֲקִיבָא בְּוַדַּאי יִבְכֶּה וְיַאֲרִיךְ בְּכָל מִלָּה וּמִלָּה.

הַהַפְתָּעָה הַגְּדוֹלָה שֶׁל הַמִּתְפַּלְּלִים

וְאָז, הִתְרַחֵשׁ דָּבָר שֶׁהִפְתִּיעַ אֶת כֻּלָּם!

רַבִּי עֲקִיבָא סִיֵּם אֶת הַתְּפִלָּה בִּמְהִירוּת שֶׁאִי אֶפְשָׁר לְתָאֵר! הוּא לֹא אָמַר אֶת הַמִּלִּים בְּצורה מְזֻלְזֶלֶת חָלִילָה, אֶלָּא קָרָא מַהֵר, בְּכַוָּנָה בְּרוּרָה אַךְ כְּמוֹ קְפִיץ חָזָק וְזָרִיז. תּוֹךְ מִסְפַּר דַּקּוֹת – "עוֹשֶׂה שָׁלוֹם בִּמְרוֹמָיו... וְאִמְרוּ אָמֵן!" [1].

הוּא פָּסַע שָׁלוֹשׁ פְּסִיעוֹת אָחוֹרָה, וְהַתְּפִלָּה נִגְמְרָה!

הַפּוֹעֲלִים הַפְּשׁוּטִים נָשְׁמוּ לִרְוָחָה. אֵיזֶה חֶסֶד! אֵיזוֹ הַקָּלָה! הֵם חִיְּכוּ זֶה לָזֶה, לָקְחוּ אֶת כְּלֵי הָעֲבוֹדָה שֶׁלָּהֶם, וּמִהֲרוּ הַבַּיְתָה לֶאֱכֹל וְלָנוּחַ.

אַךְ בֵּין הַסַּפְסָלִים עָמַד צָעִיר חָכָם וְחָרִיף, תַּלְמִידוֹ הֶחָשׁוּב שֶׁל רַבִּי עֲקִיבָא – קָרְאוּ לוֹ רַבִּי יְהוּדָה .

רַבִּי יְהוּדָה לֹא הֵבִין. הוּא יָדַע מִי זֶה רַבִּי עֲקִיבָא! הוּא יָדַע עַד כַּמָּה רַבּוֹ צוֹלֵל לְתוֹךְ יָם שֶׁל קְדֻשָּׁה כְּשֶׁהוּא עוֹמֵד לִפְנֵי הַמֶּלֶךְ, וְכָאן – הַתְּפִלָּה הָיְתָה כָּל כָּךְ קְצָרָה? אֵיךְ יִתָּכֵן? מַהוּ הַסּוֹד כָּאן?

הוּא הֶחְלִיט בְּלִבּוֹ לִבְדֹּק אֶת הַדָּבָר.

תְּפִלַּת הַלַּיְלָה הַנִּסְתֶּרֶת

עָבְרוּ כַּמָּה יָמִים. יְרוּשָׁלַיִם הָיְתָה כְּבָר רְחוֹקָה, וּבְנֵי בְּרַק הָיְתָה שְׁקֵטָה. בְּאַחַד הַלֵּילוֹת הַמְאֻחָרִים, נִשְׁאַר רַבִּי עֲקִיבָא בְּבֵית הַמִּדְרָשׁ. כֻּלָּם כְּבָר הָלְכוּ. הַתַּלְמִידִים יָשְׁנוּ. רַק נֵר קָטָן שֶׁל שֶׁמֶן זַיִת רִצֵּד עַל אֶבֶן בַּפִּנָּה, וּמִלֵּא אֶת הַחֶדֶר בִּצְלָלִים עֲנָקִיִּים שֶׁרָקְדוּ עַל הַקִּירוֹת.

אֲבָל רַבִּי עֲקִיבָא לֹא הָיָה לְבַדּוֹ. בְּאַחַת הַפִּנּוֹת, מִסְתַּתֵּר מֵאֲחוֹרֵי סַפְסַל עֵץ אָרֹךְ, יָשַׁב רַבִּי יְהוּדָה הַתַּלְמִיד, עָצַר אֶת נְשִׁימָתוֹ, וְצָפָה בְּרַבּוֹ.

רַבִּי עֲקִיבָא נָטַל אֶת יָדָיו בְּמַיִם קָרִים, עָמַד בַּזָּוִית הַמִּזְרָחִית שֶׁל בֵּית הַמִּדְרָשׁ, סָמוּךְ לַקִּיר, שִׁלֵּב אֶת יָדָיו עַל לִבּוֹ, וְהִתְחִיל לְהִתְפַּלֵּל תְּפִלַּת עַמִידָה אֶל רִבּוֹנוֹ שֶׁל עוֹלָם.

רַבִּי יְהוּדָה הִבִּיט וְנִדְהַם [1]. מַה שֶּׁרָאוּ עֵינָיו בְּאוֹתוֹ לַיְלָה – לֹא רָאָה מֵעוֹלָם!

רַבִּי עֲקִיבָא הִתְחִיל אֶת הַתְּפִלָּה בַּפִּנָּה הַיְמָנִית. אֲבָל פִּתְאוֹם... אֲנָחָה עֲמֻקָּה יָצְאָה מִלִּבּוֹ. הַגּוּף הַקָּדוֹשׁ שֶׁלּוֹ כָּרַע קָדִימָה. הוּא הִשְׁתַּחֲוָה כָּל כָּךְ לְמַטָּה, מַמָּשׁ עַד לָרִצְפָּה, וּמֵרֹב הִתְלַהֲבוּת וּדְבֵקוּת, לְלֹא מֵשִׂים לֵב בִּכְלָל, רַגְלָיו דָּחֲפוּ אוֹתוֹ קָדִימָה, וְהוּא מָצָא אֶת עַצְמוֹ בְּזָוִית אַחֶרֶת!

דְּמָעוֹת שֶׁל אֶשֶׁד גָּדוֹל נָשְׁרוּ מֵעֵינָיו וְנָפְלוּ עַל אַבְנֵי הָרִצְפָּה הַקָּרוֹת. קוֹלוֹ רָעַד. הוּא שׁוּב כָּרַע עַל בִּרְכָּיו, וְשׁוּב הִשְׁתַּחֲוָה שָׁעוֹת אֲרֻכּוֹת. הַנְּשָׁמָה שֶׁלּוֹ כְּאִלּוּ בִּקְּשָׁה לְהִתְנַתֵּק מֵהַגּוּף וְלָעוּף יְשִׁירוֹת עַד לְכִסֵּא הַכָּבוֹד! מֵרֹב הַהִשְׁתַּחֲוָיוֹת, מֵרֹב הַתְּנוּעוֹת הָעַזּוֹת שֶׁל אָדָם הָעוֹמֵד מוּל מֶלֶךְ מַלְכֵי הַמְּלָכִים, הוּא חָצָה אֶת כָּל בֵּית הַמִּדְרָשׁ – הוּא הִתְחִיל בַּפִּנָּה שֶׁלְּיַד הַדֶּלֶת, וְסִיֵּם אֶת הַתְּפִלָּה אֵי שָׁם בַּפִּנָּה שֶׁלְּיַד אֲרוֹן הַקֹּדֶשׁ! "אָדָם מַנִּיחוֹ בְּזָוִית זוֹ, וּמוֹצְאוֹ בְּזָוִית אַחֶרֶת", נִזְכַּר רַבִּי יְהוּדָה [1].

הַהֶסְבֵּר מֵאִיר הָעֵינַיִם: לִרְאוֹת אֶת הַזּוּלַת

בַּבֹּקֶר הַמָּחֳרָת, לְאַחַר תְּפִלַּת שַׁחֲרִית, אָזַר רַבִּי יְהוּדָה אֹמֶץ, נִגַּשׁ לְרַבּוֹ הַגָּדוֹל, וְשָׁאַל בְּדֶרֶךְ אֶרֶץ: "רַבִּי, יִסְלַח לִי מוֹרִי וְרַבִּי שֶׁאֲנִי שׁוֹאֵל. לֹא הֵבַנְתִּי. כַּאֲשֶׁר הָרַב מִתְפַּלֵּל בְּצִבּוּר – הוּא מְקַצֵּר מְאֹד. אֲבָל אֶתְמוֹל בַּלַּיְלָה רָאִיתִי אֶת תְּפִלָּתוֹ הַפְּרָטִית שֶׁל כְּבוֹד הָרַב... זֶה לָקַח שָׁעוֹת עַל גַּבֵּי שָׁעוֹת מֵרֹב בְּכִי, וּכְרִיעוֹת וְהִשְׁתַּחֲוָיוֹת! מַדּוּעַ אֵין הָרַב מִתְפַּלֵּל כָּךְ גַּם כְּשֶׁהוּא שְׁלִיחַ צִבּוּר לְפָנֵינוּ?"

חִיֵּךְ רַבִּי עֲקִיבָא חִיּוּךְ גָּדוֹל, עֵינָיו בָּרְקוּ וְהוּא הִנִּיחַ אֶת יָדוֹ הַחַמָּה עַל כְּתֵפוֹ שֶׁל הַתַּלְמִיד.

"יְהוּדָה בְּנִי, עֲלֵינוּ לָדַעַת לְהַבְדִּיל! כַּאֲשֶׁר אֲנִי עוֹמֵד לְבַדִּי מִלִּפְנֵי הַקָּדוֹשׁ בָּרוּךְ הוּא, זֶה רַק אֲנִי וְהַבּוֹרֵא. אֲנִי יָכוֹל לִבְכּוֹת לְפָנָיו עַד אֲשֶׁר אֶפֹּל עַל פָּנַי, וְאִישׁ אֵינוֹ נִפְגָּע. אֲבָל כַּאֲשֶׁר אֲנִי נוֹשֵׂא אֶת תְּפִלַּת הַצִּבּוּר... אָסוּר לִי לַחְשֹׁב רַק עַל הַהִתְלַהֲבוּת שֶׁלִּי! שָׁם, מֵאֲחוֹרַי, עוֹמְדִים פּוֹעֲלִים שֶׁחָזְרוּ עֲיֵפִים מִן הַשָּׂדֶה. קָשֶׁה לָהֶם לַעֲמֹד עַל רַגְלֵיהֶם. יֵשׁ שָׁם זְקֵנִים שֶׁגַּבָּם כּוֹאֵב, וִילָדִים קְטַנִּים שֶׁרְעֵבִים לַאֲרוּחַת הָעֶרֶב. וְכִי מֻתָּר לִי לְצַעֵר אוֹתָם עַל חֶשְׁבּוֹן הַתְּפִלָּה שֶׁלִּי? תְּפִלָּה הִיא דָּבָר עָצוּם, אַךְ קוֹדֶמֶת לָהּ – הַהִתְחַשְּׁבוּת בַּזּוּלַת!" [1].

וּבַשָּׁעָה הַהִיא, הֵבִין רַבִּי יְהוּדָה שֶׁהַגְּדֻלָּה שֶׁל רַבִּי עֲקִיבָא אֵינָהּ רַק בְּסוֹדוֹת הַתּוֹרָה שֶׁהוּא יוֹדֵעַ, אֶלָּא בַּלֵּב הַגָּדוֹל שֶׁלּוֹ, שֶׁיּוֹדֵעַ לְוַתֵּר עַל תְּפִלָּה שֶׁל שָׁעוֹת, כְּדֵי לֹא לִגְרֹם כְּאֵב אוֹ עֲיֵפוּת לִיהוּדִי אֶחָד פָּשׁוּט.

מקורות: תלמוד בבלי, מסכת ברכות, דף ל"א עמוד א': "אמר רב יהודה אמר רב, ואיתימא רבי אלעזר בר רבי שמעון: כשהיה רבי עקיבא מתפלל עם הצבור - היה מקצר ועולה מפני טורח צבור. וכשהיה מתפלל בינו לבין עצמו - אדם מניחו בזוית זו ומוצאו בזוית אחרת; וכל כך למה? מפני כריעות והשתחויות" [1].

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “יְלָדִים, עִצְמוּ אֶת הָעֵינַיִם וְנַסּוּ לְדַמְיֵן שֶׁאֲנַחְנוּ צוֹעֲדִים עַכְשָׁו בִּשְׁבִילֵי הֶעָפָר שֶׁל הָעִיר בְּנֵי בְּרַק, לִפְנֵי כִּמְעַט…” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “בַּסַּפְסָלִים הָרִאשׁוֹנִים יוֹשְׁבִים תַּלְמִידֵי חֲכָמִים הַלּוֹמְדִים גְּמָרָא מִתּוֹךְ סְפָרִים וּגְוִילִים.”?
ג. כיצד הגיב רבי עקיבא למצב המתואר: “הַחַקְלָאִים הֶעָיֵפִים לָקְחוּ נְשִׁימָה אֲרֻכָּה, וְהֵכִינוּ אֶת עַצְמָם לַעֲמֹד עַל הָרַגְלַיִם הַכּוֹאֲבוֹת שֶׁלָּהֶם זְמַן רַב, כִּי הֲרֵי רַבִּי…”?
ד. מה השתנה בין “רַק נֵר קָטָן שֶׁל שֶׁמֶן זַיִת רִצֵּד עַל אֶבֶן בַּפִּנָּה, וּמִלֵּא אֶת הַחֶדֶר בִּצְלָלִים עֲנָקִיִּים שֶׁרָקְדוּ…” לבין “הַנְּשָׁמָה שֶׁלּוֹ כְּאִלּוּ בִּקְּשָׁה לְהִתְנַתֵּק מֵהַגּוּף וְלָעוּף יְשִׁירוֹת עַד לְכִסֵּא הַכָּבוֹד!”?
ה. כיצד הגיע הסיפור אל האירוע: “כַּאֲשֶׁר אֲנִי עוֹמֵד לְבַדִּי מִלִּפְנֵי הַקָּדוֹשׁ בָּרוּךְ הוּא, זֶה רַק אֲנִי וְהַבּוֹרֵא.”?

לפני ואחרי

מה קרה לפני: “יְרוּשָׁלַיִם הָיְתָה כְּבָר רְחוֹקָה, וּבְנֵי בְּרַק הָיְתָה שְׁקֵטָה.”?
מה קרה אחרי: “אָדָם מַנִּיחוֹ בְּזָוִית זוֹ, וּמוֹצְאוֹ בְּזָוִית אַחֶרֶת", נִזְכַּר רַבִּי יְהוּדָה [1].”?

סיבה ותוצאה מתוך המעשה

האירוע הנתוןמה גרם לו?מה הייתה תוצאתו?
כֻּלָּם קָמִים עַל רַגְלֵיהֶם מִתּוֹךְ כָּבוֹד עָצוּם לַתּוֹרָה הַקְּדוֹשָׁה שֶׁהוּא שׁוֹנֶה לָהֶם מִדֵּי יוֹם.
מַה שֶּׁרָאוּ עֵינָיו בְּאוֹתוֹ לַיְלָה – לֹא רָאָה מֵעוֹלָם!

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “כשמתפלל בציבור מקצר” על רבי עקיבא? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “הֵם חִיְּכוּ זֶה לָזֶה, לָקְחוּ אֶת כְּלֵי הָעֲבוֹדָה שֶׁלָּהֶם, וּמִהֲרוּ הַבַּיְתָה לֶאֱכֹל וְלָנוּחַ.”.
ג. אילו רבי עקיבא לא היה פועל כפי שמתואר ב־“הַנְּשָׁמָה שֶׁלּוֹ כְּאִלּוּ בִּקְּשָׁה לְהִתְנַתֵּק מֵהַגּוּף וְלָעוּף יְשִׁירוֹת עַד לְכִסֵּא הַכָּבוֹד!” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “כַּאֲשֶׁר אֲנִי עוֹמֵד לְבַדִּי מִלִּפְנֵי הַקָּדוֹשׁ בָּרוּךְ הוּא, זֶה רַק אֲנִי וְהַבּוֹרֵא.” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. יְלָדִים, עִצְמוּ אֶת הָעֵינַיִם וְנַסּוּ לְדַמְיֵן שֶׁאֲנַחְנוּ…
2. אֲפִלּוּ צִיּוּץ שֶׁל צִפּוֹר בַּחוּץ נִשְׁמַע פִּתְאוֹם חָזָק.
3. רַבִּי עֲקִיבָא סִיֵּם אֶת הַתְּפִלָּה בִּמְהִירוּת שֶׁאִי אֶפְשָׁר…
4. רַבִּי עֲקִיבָא נָטַל אֶת יָדָיו בְּמַיִם קָרִים, עָמַד בַּזָּוִית…
5. כַּאֲשֶׁר הָרַב מִתְפַּלֵּל בְּצִבּוּר – הוּא מְקַצֵּר מְאֹד.
6. תְּפִלָּה הִיא דָּבָר עָצוּם, אַךְ קוֹדֶמֶת לָהּ – הַהִתְחַשְּׁבוּת…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “הַחַקְלָאִים הֶעָיֵפִים לָקְחוּ נְשִׁימָה אֲרֻכָּה, וְהֵכִינוּ אֶת עַצְמָם לַעֲמֹד עַל הָרַגְלַיִם הַכּוֹאֲבוֹת שֶׁלָּהֶם זְמַן…”.
מכתב: כתבו לרבי עקיבא בעקבות “מַה שֶּׁרָאוּ עֵינָיו בְּאוֹתוֹ לַיְלָה – לֹא רָאָה מֵעוֹלָם!”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “אֲבָל אֶתְמוֹל בַּלַּיְלָה רָאִיתִי אֶת תְּפִלָּתוֹ הַפְּרָטִית שֶׁל כְּבוֹד הָרַב...”.
דיווח: כתבו ידיעה קצרה שכותרתה “כשמתפלל בציבור מקצר”.

המחזה

רגע מן הסיפורהדמויותמה יאמרו או יעשו?
כֻּלָּם קָמִים עַל רַגְלֵיהֶם מִתּוֹךְ כָּבוֹד עָצוּם לַתּוֹרָה הַקְּדוֹשָׁה שֶׁהוּא שׁוֹנֶה לָהֶם…
הַנְּשָׁמָה שֶׁלּוֹ כְּאִלּוּ בִּקְּשָׁה לְהִתְנַתֵּק מֵהַגּוּף וְלָעוּף יְשִׁירוֹת עַד לְכִסֵּא…

7. חיבור לחיים מתוך המעשה

א. מה אפשר ללמוד מן הדרך שבה רבי עקיבא פעל באירוע “רַק נֵר קָטָן שֶׁל שֶׁמֶן זַיִת רִצֵּד עַל אֶבֶן בַּפִּנָּה, וּמִלֵּא אֶת הַחֶדֶר בִּצְלָלִים עֲנָקִיִּים שֶׁרָקְדוּ עַל…”?
