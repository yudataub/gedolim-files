קשריו עם המלך הערבי ארטבן — מזוזה מול מרגלית

דף עבודה בעקבות רבי יהודה הנשיא (רבי)
שם: כיתה:

המעשה — הסיפור המלא

להלן מערך חינוכי, רגשי ומרתק על רבי יהודה הנשיא וארטבן מלך פרס (המרגלית והמזוזה) , מעובד במיוחד לבנים בכיתות ב'-ד' בחינוך הדתי-לאומי. המערך נבנה בקפידה לפי עקרונות הסטוריטלינג, תוך שימוש בתיאורים חושיים, העצמת המתח הרגשי, והקפדה מוחלטת על כבוד התורה והחכמים.

הסיפור מנוקד במלואו כנדרש.

פְּתִיחָה: אַרְמוֹן הַזָּהָב וְהַמֶּלֶךְ הַגֵּאֶה

יְלָדִים, בּוֹאוּ נַפְלִיג בַּדִּמְיוֹן הַרְחֵק הַרְחֵק מִכָּאן. שְׁמָעוּ אֶת נְקִישׁוֹת הַפַּרְסוֹת שֶׁל אַלְפֵי סוּסִים אֲצִילִיִּים. הָרִיחוּ אֶת רֵיחַ הַבְּשָׂמִים, הַקִּנָּמוֹן וְהַוֶּרֶד שֶׁנִּשָּׂא בָּאֲוִיר. אֲנַחְנוּ נִמְצָאִים בְּאַרְמוֹנוֹ הָעֲנָק שֶׁל אַרְטַבָּן, מֶלֶךְ פָּרַס וּמָדַי.

אַרְטַבָּן הָיָה אֶחָד הַמְּלָכִים הָעֲשִׁירִים בְּיוֹתֵר בָּעוֹלָם כֻּלּוֹ. בְּאַרְמוֹנוֹ הָיוּ עַמּוּדֵי שַׁיִשׁ לְבָנִים, כִּסְאוֹת מְצֻפִּים בְּזָהָב טָהוֹר, וַעֲבָדִים שֶׁהִסְתּוֹבְבוּ עִם מַגְּשֵׁי כֶּסֶף עֲמוּסִים בְּכָל טוּב.

בְּאוֹתָם יָמִים, בְּאֶרֶץ יִשְׂרָאֵל, חַי מַנְהִיגָם שֶׁל יִשְׂרָאֵל, גְּדוֹל הַדּוֹר וּנְשִׂיא הַסַּנְהֶדְרִין – רַבִּי יְהוּדָה הַנָּשִׂיא , שֶׁכֻּלָּם קָרְאוּ לוֹ פָּשׁוּט "רַבִּי". גַּם רַבִּי יְהוּדָה הָיָה עָשִׁיר מְאֹד בְּגַשְׁמִיּוּת. עַל שֻׁלְחָנוֹ תָּמִיד הָיָה שֶׁפַע לְאָרֵחַ תַּלְמִידֵי חֲכָמִים וַעֲנִיִּים, אַךְ הוּא עַצְמוֹ הָיָה קָדוֹשׁ וְטָהוֹר, שֶׁלֹּא נֶהֱנָה מֵהָעוֹלָם הַזֶּה אֲפִלּוּ בְּאֶצְבַּע קְטַנָּה! כָּל מַחֲשָׁבוֹתָיו הָיוּ רַק בַּתּוֹרָה.

הַמֶּלֶךְ אַרְטַבָּן שָׁמַע מֵרָחוֹק עַל עָשְׁרוֹ וְחָכְמָתוֹ שֶׁל רַבִּי יְהוּדָה הַנָּשִׂיא. יוֹם אֶחָד יָשַׁב אַרְטַבָּן עַל כִּסְאוֹ, גֵּאֶה בְּעַצְמוֹ, וְאָמַר לְשָׂרָיו: "רַבִּי שֶׁל הַיְּהוּדִים עָשִׁיר? נִרְאֶה! אֲנִי אֶשְׁלַח לוֹ אֶת הָאוֹצָר הֲכִי יָקָר שֶׁיֵּשׁ לִי בְּכָל הַמַּמְלָכָה. נִרְאֶה אִם יֵשׁ לוֹ מַשֶּׁהוּ יָקָר מִזֶּה כְּדֵי לִשְׁלֹחַ לִי בַּחֲזָרָה!".

הַמַּתָּנָה הַנּוֹצֶצֶת שֶׁל אַרְטַבָּן

אַרְטַבָּן יָרַד אֶל מַרְתְּפֵי הָאַרְמוֹן, פָּתַח כַּסָּפוֹת כְּבֵדוֹת מִבַּרְזֶל שֶׁחָרְקוּ בְּקוֹל עָמֹק, וְהוֹצִיא מִשָּׁם מַרְגָּלִית קְטַנָּה אַךְ עֲצוּמָה וּמַדְהִימָה. זֶה לֹא הָיָה סְתָם יַהֲלוֹם; הָיְתָה זוֹ מַרְגָּלִית (אֶבֶן טוֹבָה) נְדִירָה כָּל כָּךְ, שֶׁכְּשֶׁפָּתְחוּ אֶת הַקֻּפְסָא שֶׁלָּהּ, הִיא הֵאִירָה אֶת הַחֶדֶר הַחָשׁוּךְ כְּמוֹ שֶׁמֶשׁ לְבָנָה וְטְהוֹרָה!

הַשָּׂרִים עָמְדוּ מִסָּבִיב וְהַשְּׁתָּאוּ מֵהַיֹּפִי וּמֵהַשֹּׁוִי שֶׁל הָאֶבֶן – מִי לֹא הָיָה רוֹצֶה בָּהּ? יָכְלוּ לִקְנוֹת בָּהּ עָרִים שְׁלֵמוֹת!

קָרָא הַמֶּלֶךְ לְשָׁלִיחַ, שָׂם אֶת הַמַּרְגָּלִית בְּתוֹךְ קֻפְסַת עֵץ נָדִיר, וְשָׁלַח פְּלֻגַּת חַיָּלִים שְׁלֵמָה חֲמוּשָׁה בַּחֲרָבוֹת וְרֹמְחִים לִשְׁמֹר עַל הָאֶבֶן בַּדֶּרֶךְ. "סְעוּ לְאֶרֶץ יִשְׂרָאֵל. תְּנוּ לְרַבִּי אֶת זֶה, וְנִרְאֶה אִם יוּכַל לְהַחְזִיר לִי אוֹצָר יָקָר מִמֶּנּוּ!".

הָאוֹצָר מַגִּיעַ לְבֵית הַנָּשִׂיא

לְאַחַר מַסָּע אָרֹךְ וּמְסֻכָּן בַּמִּדְבָּר וּבֶהָרִים, הִגִּיעוּ הַשְּׁלִיחִים לְבֵית הַמִּדְרָשׁ שֶׁל רַבִּי יְהוּדָה הַנָּשִׂיא.

הֵם נִכְנְסוּ אֶל הֶחָצֵר. קוֹל לִמּוּד הַתּוֹרָה בָּקַע מִן הַחַלּוֹנוֹת, קוֹל מָתוֹק מִדְּבַשׁ. מִפְּנֵי כְּבוֹד הַמַּלְכוּת, קִבֵּל אוֹתָם רַבִּינוּ הַקָּדוֹשׁ בְּסֵבֶר פָּנִים יָפוֹת. הַשָּׁלִיחַ כָּרַע בֶּרֶךְ וְהִגִּישׁ אֶת קֻפְסַת הָעֵץ הַמְּפֹאֶרֶת.

"אַרְטַבָּן מֶלֶךְ פָּרַס שׁוֹלֵחַ לְךָ מִנְחָה!" אָמַר הַשָּׁלִיחַ

כַּאֲשֶׁר נִפְתְּחָה הַקֻּפְסָה, נֶעֶצְרָה נְשִׁימָתָם שֶׁל תַּלְמִידֵי הַחֲכָמִים. מִימֵיהֶם לֹא רָאוּ יֹפִי גַּשְׁמִי וְהָדָר כָּזֶה! עֵינֵיהֶם הֻסְנְוְרוּ. אַךְ רַבִּי יְהוּדָה הַנָּשִׂיא סָגַר בְּרֹגַע אֶת הַקֻּפְסָה, כְּמִי שֶׁרָאָה עֲרֵמַת אֲבָנִים פְּשׁוּטוֹת בָּרְחוֹב, וְאָמַר לַשָּׁלִיחַ: "מְסֹר לְמַלְכְּךָ תּוֹדָה רַבָּה".

עַכְשָׁו עָמְדָה בָּאֲוִיר שְׁאֵלָה גְּדוֹלָה: מָה רַבִּי יִשְׁלַח מִנְגֶּד? תַּלְמִידָיו חָשְׁבוּ: 'רַבִּי בְּוַדַּאי יִשְׁלַח לַמֶּלֶךְ מַרְתֵּף שָׁלֵם מָלֵא בְּתַכְשִׁיטִים, שְׁקָקִים שֶׁל מַטְבְּעוֹת תַּרְשִׁישׁ וְזָהָב!'

הַהַפְתָּעָה שֶׁל רַבִּינוּ הַקָּדוֹשׁ

אַךְ רַבִּי לֹא הָלַךְ לְחַדְרֵי הָאוֹצָרוֹת. הוּא הָלַךְ אֶל הַסּוֹפֵר סְתָ"ם. הוּא מָצַא קְלָף אֶחָד כָּשֵׁר וְטָהוֹר, שֶׁעָלָיו כְּתוּבוֹת שְׁתֵּי פָּרָשִׁיּוֹת בַּאֲשׁוּרִית, מְגֻלְגָּלוֹת לְגָלִיל קָטָן.

כֵּן, יְלָדִים! קְלָף פָּשׁוּט שֶׁל מְזוּזָה!

רַבִּי לָקַח אֶת הַמְּזוּזָה, עָטַף אוֹתָהּ, וְהִגִּישׁ לַשָּׁלִיחַ. "זֶה הָאוֹצָר שֶׁלִּי לַמֶּלֶךְ אַרְטַבָּן".

הַשָּׁלִיחַ הָיָה הָמוּם. זֶה הַכֹּל? קְלָף קָטָן? אֲפִלּוּ לֹא מְשֻׁבָּץ בְּיַהֲלוֹמִים? אַךְ הוּא לֹא הֵעֵז לְהִתְוַכֵּחַ עִם נְשִׂיא יִשְׂרָאֵל, לָקַח אֶת הַחֲבִילָה הַקְּטַנָּה, קָרָא לַחַיָּלִים, וְרָכַב בַּחֲזָרָה לְפָרַס.

הַזַּעַם שֶׁל הַמֶּלֶךְ

כַּאֲשֶׁר הִגִּיעַ הַשָּׁלִיחַ לְאַרְמוֹן פָּרַס, אַרְטַבָּן יָשַׁב בְּמֶתַח. "מָה שָׁלַח לִי הֶחָכָם הַיְּהוּדִי? מַסֶּכֶת שֶׁל שַׁיִשׁ? כֶּתֶר מְשֻׁבָּץ סַפִּירִים?". הוּא חָטַף בִּמְהִירוּת אֶת הַחֲבִילָה וּפָתַח אוֹתָהּ מִיָּד בִּידָיו!

כְּשֶׁעֵינָיו נָפְלוּ עַל הַקְּלָף הַמְּגֻלְגַּל, פָּנָיו הֶאֱדִימוּ. קַמְטֵי כַּעַס הֵחֵלּוּ לְמַלֵּא אֶת מִצְחוֹ. נְשִׁימוֹתָיו גָּבְרוּ.

"מָה זֶה?!" צָעַק הַמֶּלֶךְ בְּקוֹל רוֹעֵם שֶׁהִרְעִיד אֶת הַשַּׁנְדֵּלִירִים מִזְּכוּכִית. "אֲנִי... אַרְטַבָּן, מֶלֶךְ הַמְּלָכִים... שָׁלַחְתִּי לוֹ מַרְגָּלִית שֶׁאֵין בְּכָל הָעוֹלָם עוֹד אַחַת כְּמוֹתָהּ מִלְּבַדָּהּ... וְהוּא שׁוֹלֵחַ לִי מַשֶּׁהוּ שֶׁשָּׁוֶה פְּרוּטָה אַחַת שֶׁל כֶּסֶף?! זוֹ חוּצְפָּה! זֶה זִלְזוּל!".

הוּא קָרָא לַשָּׁלִיחַ וְצִוָּה עָלָיו: "דְּהַר חֲזָרָה לְרַבִּי שֶׁל הַיְּהוּדִים, וֶאֱמֹר לוֹ דָּבָר אֶחָד: 'אֲנִי שָׁלַחְתִּי לְךָ אוֹצָר שֶׁאֵין לוֹ שִׁעוּר! וְאַתָּה שָׁלַחְתָּ לִי מַשֶּׁהוּ שֶׁשָּׁוֶה רַק מַטְבֵּעַ אֶחָד שֶׁל כֶּסֶף (פוֹלָר)?!' עַכְשָׁו, לֵךְ!".

הָאוֹצָר שֶׁשּׁוֹמֵר עָלֶיךָ

הַשָּׁלִיחַ חָזַר רָץ לְאֶרֶץ יִשְׂרָאֵל, מַגִּיעַ נוֹשֵׁם וְנוֹשֵׁף לְבֵיתוֹ שֶׁל רַבִּינוּ הַקָּדוֹשׁ. הוּא הֶעֱבִיר אֶת כַּעֲסוֹ שֶׁל מַלְכּוֹ.

רַבִּי יְהוּדָה הַנָּשִׂיא הִבִּיט בַּשָּׁלִיחַ. הוּא לֹא נִבְהַל. עֵינָיו שָׁפְעוּ קְדֻשָּׁה, וְחִיּוּךְ שָׁלֵו וְחָכָם נָסוּךְ עַל שְׂפָתָיו.

וְאָז, הִשְׁמִיעַ רַבִּי אֶת הַתְּשׁוּבָה שֶׁחָרוּטָה לְעוֹלְמֵי עַד, תְּשׁוּבָה שֶׁהֵבִיכָה וְהֵאִירָה אֶת הַמֶּלֶךְ אַרְטַבָּן כְּאֶחָד:

אֲבָל אֲנִי... אֲנִי, אַרְטַבָּן יַקִּירִי, שָׁלַחְתִּי לְךָ אוֹצָר אֲמִתִּי, מַתָּנָה כָּזוֹ – שֶׁתִּשְׁמֹר הִיא עָלֶיךָ אֲפִלּוּ כְּשֶׁאַתָּה יָשֵׁן! " כְּמוֹ שֶׁאוֹמֵר הַפָּסוּק (בְּמִשְׁלֵי): 'בְּשָׁכְבְּךָ - תִּשְׁמֹר עָלֶיךָ!'."

דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “אֲנִי צָרִיךְ חַיָּלִים, וְכַסָּפוֹת וְשׁוֹמְרִים בַּלַּיְלָה כְּדֵי שֶׁלֹּא יִגְנְבוּ אוֹתָהּ מִמֶּנִּי.”.
דיווח: כתבו ידיעה קצרה שכותרתה “קשריו עם המלך הערבי ארטבן — מזוזה מול מרגלית”.

המחזה

רגע מן הסיפורהדמויותמה יאמרו או יעשו?
