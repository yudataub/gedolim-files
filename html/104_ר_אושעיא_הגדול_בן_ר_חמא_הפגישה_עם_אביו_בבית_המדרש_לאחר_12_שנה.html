הפגישה עם אביו בבית המדרש לאחר 12 שנה

דף עבודה בעקבות ר' אושעיא הגדול בן ר' חמא
שם: כיתה:

המעשה — הסיפור המלא

להלן מערך סיפורי מקיף, רגשי ומרתק על פגישתם המופלאה של רבי חָמָא וּבְנוֹ רַבִּי אוֹשַׁעְיָה הַגָּדוֹל , לאחר שתים עשרה שנות לימוד תורה. הסיפור ערוך בדיוק לפי ההנחיות שלך עבור בנים בחינוך הדתי-לאומי, מנוקד במלואו, עשיר בתיאורים חושיים, ומשלב כבוד עצום לתורה לצד השגחה פרטית ברורה.

פְּתִיחָה: הַפְּרֵדָה בְּדֶמַע

יְלָדִים, אֲנַחְנוּ חוֹזְרִים בַּזְּמַן אֶל יְמֵי הָאָמוֹרָאִים. דַּמְיְנוּ אֶת אַרְצֵנוּ הַקְּדוֹשָׁה שֶׁל פַּעַם, עֲיָרוֹת קְטַנּוֹת מֻקָּפוֹת כְּרָמִים, שְׁבִילֵי עָפָר רַכִּים, וְרֵיחַ שֶׁל עֲצֵי זַיִת וּתְאֵנִים שֶׁמְּמַלֵּא אֶת הָאֲוִיר.

בְּאַחַת הֶעָרִים לְיַד צִפּוֹרִי, גָּרָה מִשְׁפָּחָה מְיֻחֶדֶת בְּמִינָהּ. סַבָּא שֶׁל הַמִּשְׁפָּחָה, רַבִּי בִּיסָא, הָיָה צַדִּיק יְסוֹד עוֹלָם וְתַלְמִיד חָכָם שֶׁכֻּלָּם הִכִּירוּ. בְּנוֹ, רַבִּי חָמָא, רָצָה גַּם הוּא לִגְדֹּל וְלִהְיוֹת גָּדוֹל בַּתּוֹרָה כְּמוֹ אָבִיו. אֲבָל כְּדֵי לִגְדֹּל בַּתּוֹרָה בֶּאֱמֶת, יָדַע רַבִּי חָמָא שֶׁעָלָיו לִנְסֹעַ לִישִׁיבָה מְרֻחֶקֶת, לְהִנָּתֵק מִכָּל טִרְדוֹת הָעוֹלָם, וְלִשְׁקֹעַ אַךְ וְרַק בְּדַפֵּי הַגְּמָרָא.

"אֲנִי מֻכְרָח לָלֶכֶת לִלְמֹד תּוֹרָה כְּדֵי לְהָאִיר אֶת עַם יִשְׂרָאֵל," אָמַר רַבִּי חָמָא לְאִשְׁתּוֹ הַצַּדֶּקֶת בְּקוֹל דּוֹמֵעַ. הִיא הִנְהֲנָה בְּהַסְכָּמָה, לַמְרוֹת שֶׁלִּבָּהּ הָיָה שָׁבוּר. בְּאוֹתוֹ זְמַן, הָיָה לָהֶם בַּבַּיִת תִּינוֹק קָטָן, רַךְ וּמָתוֹק, שֶׁשְּׁמוֹ אוֹשַׁעְיָה. רַבִּי חָמָא לִטֵּף אֶת רֹאשׁוֹ שֶׁל בְּנוֹ הַקָּטָן, נָשַׁק לוֹ עַל מִצְחוֹ הַחַם, נָטַל אֶת צְרוֹרוֹ הַדַּל, וְיָצָא לַדֶּרֶךְ.

שְׁתֵּים עֶשְׂרֵה שְׁנוֹת עָמָל

עָבְרָה שָׁנָה, וְעוֹד שָׁנָה, וְעוֹד שָׁנָה. רַבִּי חָמָא יָשַׁב בְּבֵית הַמִּדְרָשׁ וְלָמַד בִּמְסִירוּת נֶפֶשׁ שֶׁאִי אֶפְשָׁר לְתָאֵר. הַלַּיְלָה הָפַךְ לְיוֹם וְהַיּוֹם לְלַיְלָה. הוּא לֹא אָכַל מַעֲדַנִּים וְלֹא יָשַׁן עַל מִטּוֹת רַכּוֹת; רֵיחַ קְלָפֵי עֵץ וְדִיוֹ עַתִּיק הָיָה הַבֹּסֶם שֶׁלּוֹ. קוֹלוֹ הִדְהֵד בֵּין קִירוֹת הָאֶבֶן כְּשֶׁלָּמַד שׁוּב וָשׁוּב: "אָמַר רַבִּי... תָּנוּ רַבָּנַן...". הוּא לֹא חָזַר הַבַּיְתָה אֲפִלּוּ פַּעַם אַחַת! שְׁתֵּים עֶשְׂרֵה שָׁנִים תְּמִימוֹת עָמַל רַבִּי חָמָא, עַד שֶׁהִגִּיעַ לְמַדְרֵגָה כּוֹנְנָה שֶׁל צַדִּיק יְסוֹד עוֹלָם, בָּקִי בְּכָל רָזֵי תּוֹרָתֵנוּ הַקְּדוֹשָׁה.

וּמָה עִם מִשְׁפַּחְתּוֹ? בַּבַּיִת הַקָּטָן, אִשְׁתּוֹ שֶׁל רַבִּי חָמָא דָּאֲגָה לַיֶּלֶד. הִיא לֹא נָתְנָה לְעוֹנִי לְשַׁבֵּר אוֹתָהּ. כְּשֶׁאוֹשַׁעְיָה מְעַט גָּדַל, הִיא לָקְחָה אוֹתוֹ יָד בְּיָד, עָטְפָה אוֹתוֹ בְּמַעֲטֶפֶת שֶׁל אַהֲבָה וּתְפִלּוֹת, וְהוֹשִׁיבָה אוֹתוֹ עַל סַפְסַל הַלִּמּוּדִים מוּל מְלַמֵּד הַתִּינוֹקוֹת. הַקָּדוֹשׁ בָּרוּךְ הוּא אָהַב אֶת מְסִירוּת הַנֶּפֶשׁ הַזּוֹ. מִשָּׁמַיִם הִשְׁפִּיעוּ עַל אוֹשַׁעְיָה שֵׂכֶל חָרִיף וּמָהִיר כְּבָרָק. תּוֹךְ כַּמָּה שָׁנִים, הַיֶּלֶד הַקָּטָן צָמַח וְהָפַךְ לְעִלּוּי שֶׁכָּל זִקְנֵי הָעִיר נֶהֱנוּ מֵחָכְמָתוֹ.

הַחֲזָרָה וּבֵית הַמִּדְרָשׁ

בְּתֹם שְׁתֵּים עֶשְׂרֵה הַשָּׁנִים, יָצָא רַבִּי חָמָא לַדֶּרֶךְ חֲזָרָה. דַּמְיְנוּ אֶת הַהִתְרַגְּשׁוּת שֶׁלּוֹ. כַּפּוֹת רַגְלָיו נָגְעוּ בַּאֲדָמָה הַמֻּכֶּרֶת, וּמֵרָחוֹק כְּבָר רָאָה אֶת גַּגּוֹת הָעִיר שֶׁלּוֹ. אֲבָל רַבִּי חָמָא הָיָה אִישׁ קָדוֹשׁ וְחָכָם. הוּא אָמַר לְעַצְמוֹ: "אִם אִכָּנֵס פִּתְאוֹם הַבַּיְתָה כָּךְ יְשִׁירוֹת בְּלִי הוֹדָעָה, מֵרֹב בֶּהָלָה וְהִתְרַגְּשׁוּת עֲלוּלָה אִשְׁתִּי לְקַבֵּל שָׁבָץ וּלְהִפָּגַע מִבְּחִינָה בְּרִיאוּתִית." (חֲזַ"ל מְלַמְּדִים אוֹתָנוּ – לְעוֹלָם אַל יִכָּנֵס אָדָם לְבֵיתוֹ בְּפִתְאוֹם).

מָה עָשָׂה? הָלַךְ קֹדֶם כֹּל לְבֵית הַמִּדְרָשׁ שֶׁל הָעִיר, שָׁלַח שָׁלִיחַ לְבֵיתוֹ שֶׁיּוֹדִיעַ לְאִשְׁתּוֹ בַּעֲדִינוּת שֶׁהוּא חָזַר וְנִמְצָא בְּבֵית הַמִּדְרָשׁ, וְהִמְתִּין שָׁם.

בֵּית הַמִּדְרָשׁ הָיָה מָלֵא. קוֹל תּוֹרָה מָתוֹק כִּדְבַשׁ נִשְׁמַע בַּחֲלַל הָאֲוִיר. רַבִּי חָמָא פָּסַע פְּנִימָה, הִתְיַשֵּׁב בַּפִּנָּה וּפָתַח סֵפֶר.

פִּתְאוֹם, קוֹל צָעִיר וְצָלוּל הִדְהֵד בַּחֶדֶר. נַעַר צָעִיר, אוּלַי בֶּן שְׁתֵּים עֶשְׂרֵה אוֹ שְׁלֹשׁ עֶשְׂרֵה, עָמַד שָׁם וְשָׁאַל קֻשְׁיָה תַּלְמוּדִית כָּל כָּךְ חֲזָקָה שֶׁכָּל הָרַבָּנִים הַמְּבֻגָּרִים שָׁתְקוּ.

רַבִּי חָמָא הֵרִים אֶת הָעֵינַיִם וְהִסְתַּכֵּל בַּנַּעַר. פָּנָיו הָיוּ מוּאָרוֹת בְּאוֹר שֶׁל צַדִּיקוּת. רַבִּי חָמָא נִכְנַס עִמּוֹ לְדִיּוּן בַּהֲלָכָה. הוּא הִקְשָׁה וְהַנַּעַר תֵּרֵץ; הַנַּעַר הִקְשָׁה וְרַבִּי חָמָא עָנָה. שְׁלֹשֶׁת הַשָּׁעוֹת הַבָּאוֹת הָיוּ מְלֵאוֹת בְּאֵשׁ שֶׁל שִׂמְחַת תּוֹרָה. רַבִּי חָמָא הָיָה הָמוּם. הוּא, עֲנַק הָרוּחַ שֶׁלָּמַד 12 שָׁנִים שְׁלֵמוֹת, מוֹצֵא בַּנַּעַר הַזֶּה חָבֵר שָׁוֶה שֶׁמֵּבִין לְעֹמֶק הַסְּבָרוֹת הֲכִי קָשׁוֹת בַּתּוֹרָה!

דִּמְעוֹתֵיהֶן שֶׁל צַדִּיקִים

אַךְ יְלָדִים, בְּתוֹךְ תּוֹכוֹ, תּוֹךְ כְּדֵי שֶׁהוּא שָׂמֵחַ עַל דִּבְרֵי הַתּוֹרָה שֶׁל הַנַּעַר, הֵחֵל לְחַלְחֵל כְּאֵב נוֹרָא אֶל לִבּוֹ שֶׁל רַבִּי חָמָא. תּוּגָה עֲמֻקָּה עָטְפָה אוֹתוֹ.

הוּא הִבִּיט בַּנַּעַר הַצָּעִיר, שָׁמַע אֶת חָכְמָתוֹ, וְאָמַר לְעַצְמוֹ בְּלֵב נִשְׁבָּר וּבְעֵינַיִם דּוֹמְעוֹת:

"אוֹי... לוּ הָיִיתִי נִשְׁאָר בַּבַּיִת בְּיַחַד עִם אִשְׁתִּי... בְּנִי אוֹשַׁעְיָה כְּבָר בְּדִיּוּק בַּגִּיל הַזֶּה, כְּמוֹ הַנַּעַר שֶׁיּוֹשֵׁב כָּאן. אִלּוּ לֹא עָזַבְתִּי, אִלּוּ הָיִיתִי אִתּוֹ עַד עַכְשָׁו, הָיִיתִי יָכוֹל לְלַמֵּד אוֹתוֹ תּוֹרָה, וַדַּאי גַּם הוּא הָיָה נַעֲשֶׂה לַמְדָן עָצוּם וְחָכָם כְּמוֹ הַנַּעַר הַקָּדוֹשׁ הַזֶּה! מִי יוֹדֵעַ מָה קוֹרֶה הַיּוֹם עִם אוֹשַׁעְיָה שֶׁלִּי... יַלְדִּי הַקָּטָן, שֶׁגָּדַל בְּלִי אָב שֶׁיִּלְמַד אִתּוֹ... מִסְתָּמָא הוּא נִשְׁאַר בּוּר וְעַם הָאָרֶץ...". דִּמְעָה חַמָּה גָּלְשָׁה עַל לֶחְיוֹ שֶׁל רַבִּי חָמָא, אַךְ הוּא בָּלַע אוֹתָהּ לַחֲשַׁאי, שֶׁלֹּא יַבְחִינוּ בָּהּ הָאֲנָשִׁים סְבִיבוֹ. הוּא לֹא יָדַע שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא צָפָה מִלְמַעְלָה וְהֵכִין לוֹ הַפְתָּעָה שֶׁלֹּא חָלַם עָלֶיהָ.

הַהַפְתָּעָה בַּבַּיִת: "וְכִי קָם אָב בִּפְנֵי בְּנוֹ?!"

הַשָּׁעָה הִגִּיעָה. הַשָּׁלִיחַ שָׁב לְבֵית הַמִּדְרָשׁ וְאָמַר לְרַבִּי חָמָא: "הָרַבָּנִית יוֹדַעַת. הַבַּיִת מוּכָן."

רַבִּי חָמָא סָגַר אֶת הַגְּמָרָא, בֵּרֵךְ בְּהַדְרַת כָּבוֹד אֶת הַנַּעַר הַמַּפְלִיא בַּחָכְמָתוֹ שֶׁיָּשַׁב לְיָדוֹ, וְיָצָא לְכִוּוּן בֵּיתוֹ.

הוּא הִגִּיעַ אֶל דֶּלֶת הָעֵץ הַיְּשָׁנָה, פָּתַח אוֹתָהּ, וְרָאָה אֶת אִשְׁתּוֹ. דִּמְעוֹת אוֹשֶׁר שֶׁל תּוֹרָה עָמְדוּ בְּעֵינֵיהֶם. הַבַּיִת לָבַשׁ חַג. עֲשָׂרוֹת פְּסוּקִים וּמִשְׁנָיוֹת אָמְרוּ זֶה לָזוֹ, לְסַפֵּר עַל הֶעָמָל הַגָּדוֹל שֶׁל שְׁנֵיהֶם בְּמֶשֶׁךְ שְׁנֵים עָשָׂר שָׁנִים אֲרֻכּוֹת.

וְהִנֵּה, תּוֹךְ כְּדֵי שֶׁהֵם עֲדַיִן מְדַבְּרִים, הַדֶּלֶת נִפְתֶּחֶת שׁוּב.

יְלָדִים, אֶת מִי רוֹאֶה רַבִּי חָמָא נִכְנָס בַּפֶּתַח? נָכוֹן! מַמָּשׁ אֶת אוֹתוֹ נַעַר אָהוּב וְגָאוֹן עִמּוֹ לָמַד עַד לִפְנֵי רֶגַע בְּבֵית הַמִּדְרָשׁ!

רַבִּי חָמָא נִבְהַל. "הוּא בֶּטַח בָּא לִשְׁאֹל אוֹתִי שְׁאֵלָה בַּהֲלָכָה," חָשַׁב רַבִּי חָמָא בַּעֲנָוָה עֲצוּמָה. בְּלִי לַחְשֹׁב פַּעֲמַיִם, זִנֵּק הָרַב מִכִּסְאוֹ, וְעָמַד יָשָׁר עַל רַגְלָיו כְּדֵי לִחְלוֹק כָּבוֹד שֶׁל גְּדוֹלֵי תּוֹרָה לַנַּעַר הַגָּאוֹן. זֶה הָיָה מַחֲזֶה שֶׁל עֲנָוָה אֲמִתִּית – תַּלְמִיד חָכָם זָקֵן שֶׁקָּם לִכְבוֹד נַעַר, כִּי הוּא מַעֲרִיךְ תּוֹרָה אֲמִתִּית הֵיכָן שֶׁהִיא נִמְצֵאת.

אַךְ אָז, הִבִּיטָה אִשְׁתּוֹ שֶׁל רַבִּי חָמָא בְּבַעֲלָהּ. חִיּוּךְ גָּדוֹל, קָדוֹשׁ וְטָהוֹר זָרַח מֵעַל שְׂפָתֶיהָ. עֵינֶיהָ נָצְצוּ מִשִּׂמְחָה וְהִיא צָחֲקָה קַלּוֹת.

"אִישִׁי הַיָּקָר," הִיא אָמְרָה בְּקוֹל רוֹעֵד מֵאֹשֶׁר, "וְכִי יֵשׁ בָּעוֹלָם אָב שֶׁקָּם בִּפְנֵי בְּנוֹ?!"

רַבִּי חָמָא קָפָא בִּמְקוֹמוֹ. נְשִׁימָתוֹ נֶעֶתְקָה.

"בְּנִי?" הוּא לָחַשׁ, עֵינָיו רְחָבוֹת. "זֶה הַנַּעַר הַזֶּה... זֶה בְּנִי אוֹשַׁעְיָה?!"

"כֵּן, אִישִׁי הַגָּדוֹל!" בָּכְתָה הָאֵם שֶׁלֹּא יָכְלָה לְהַחֲזִיק יוֹתֵר אֶת הַסּוֹד. "הַקָּדוֹשׁ בָּרוּךְ הוּא בֵּרֵךְ אוֹתָנוּ. אַתָּה מָסַרְתָּ אֶת נַפְשְׁךָ לְלִמּוּד תּוֹרָה, וַאֲנִי עָשִׂיתִי כָּל מַאֲמָץ בַּבַּיִת, וְהַשֵּׁם דָּאֲג שֶׁהַבֵּן שֶׁלָּנוּ יִצְמַח לִהְיוֹת רַבִּי אוֹשַׁעְיָה, מִגְּדוֹלֵי הַדּוֹר, שֶׁעוֹד יַעֲמִיד תַּלְמִידִים רַבִּים!"

רַבִּי חָמָא פָּרַשׂ אֶת יָדָיו, דְּמָעוֹת שֶׁל תּוֹדָה עֲצוּמָה זָלְגוּ עַל זְקָנוֹ הַלָּבָן. רַבִּי אוֹשַׁעְיָה רָץ אֶל זְרוֹעוֹתָיו שֶׁל אָבִיו וּשְׁנֵיהֶם הִתְחַבְּקוּ חִבּוּק כָּל כָּךְ אָמִיץ וְחַם, חִבּוּק שֶׁל מִשְׁפָּחָה וְשֶׁל תּוֹרַת אֱמֶת. הֵם הוֹדוּ מִקִּירוֹת לִבָּם לְבוֹרֵא עוֹלָם שֶׁלֹּא עָזַב אוֹתָם לְרֶגַע, וְזָכוּ שֶׁבְּאוֹתוֹ דּוֹר יָשְׁבוּ בְּבֵית הַמִּדְרָשׁ – הַסַּבָּא רַבִּי בִּיסָא, הָאָב רַבִּי חָמָא, וְהַבֵּן רַבִּי אוֹשַׁעְיָה הַגָּדוֹל, כֻּלָּם גְּדוֹלֵי יִשְׂרָאֵל אֲהוּבִים.

הם (הדמויות): רַבִּי חָמָא עָשָׂה הַכֹּל כְּדֵי לִלְמֹד, וּבְנוֹ רַבִּי אוֹשַׁעְיָה יָשַׁב בְּבֵית הַמִּדְרָשׁ, הִתְאַמֵּץ וְהָפַךְ לְגָאוֹן מִתּוֹךְ אַהֲבָה לַשֵּׁם יִתְבָּרַךְ.

אנחנו (היום-יום שלנו): גַּם לָנוּ קָשֶׁה לִפְעָמִים. לִפְעָמִים הַלִּמּוּד נִרְאֶה קָשֶׁה מִדַּי, חָסֵר לָנוּ צִיּוּן אוֹ עֶזְרָה, וַאֲנַחְנוּ עֲלוּלִים לְוַתֵּר אוֹ לְהִסְתַּכֵּל הַצִּדָּה וּלְהַרְגִּישׁ מִסְכֵּנִים.

אתם (הייישום הבוקר): אֲבָל מָחָר בַּבֹּקֶר, כְּשֶׁנִּפְתַּח אֶת סֵפֶר הַחֻמָּשׁ אוֹ הַגְּמָרָא, נִזְכֹּר אֶת רַבִּי אוֹשַׁעְיָה: אִם נַשְׁקִיעַ, נֶאֱהַב וְלֹא נְוַתֵּר – הַשֵּׁם יִתְבָּרַךְ יִשְׁלַח לָנוּ שֵׂכֶל טוֹב וּבְרָכָה, וְנַעֲשֶׂה גַּאֲוָה גְּדוֹלָה גַּם לְהַקָּדוֹשׁ בָּרוּךְ הוּא וְגַם לַהוֹרִים שֶׁלָּנוּ!

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “להלן מערך סיפורי מקיף, רגשי ומרתק על פגישתם המופלאה של רבי חָמָא וּבְנוֹ רַבִּי אוֹשַׁעְיָה הַגָּדוֹל , לאחר שתים עשרה שנות לימוד תורה.” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “רַבִּי חָמָא לִטֵּף אֶת רֹאשׁוֹ שֶׁל בְּנוֹ הַקָּטָן, נָשַׁק לוֹ עַל מִצְחוֹ הַחַם, נָטַל אֶת צְרוֹרוֹ הַדַּל, וְיָצָא לַדֶּרֶךְ.”?
ג. כיצד הגיב ר' אושעיא הגדול בן ר' חמא למצב המתואר: “בְּתֹם שְׁתֵּים עֶשְׂרֵה הַשָּׁנִים, יָצָא רַבִּי חָמָא לַדֶּרֶךְ חֲזָרָה.”?
ד. מה השתנה בין “הוּא הִבִּיט בַּנַּעַר הַצָּעִיר, שָׁמַע אֶת חָכְמָתוֹ, וְאָמַר לְעַצְמוֹ בְּלֵב נִשְׁבָּר וּבְעֵינַיִם דּוֹמְעוֹת:” לבין “וְהִנֵּה, תּוֹךְ כְּדֵי שֶׁהֵם עֲדַיִן מְדַבְּרִים, הַדֶּלֶת נִפְתֶּחֶת שׁוּב.”?
ה. כיצד הגיע הסיפור אל האירוע: “רַבִּי אוֹשַׁעְיָה רָץ אֶל זְרוֹעוֹתָיו שֶׁל אָבִיו וּשְׁנֵיהֶם הִתְחַבְּקוּ חִבּוּק כָּל כָּךְ אָמִיץ וְחַם, חִבּוּק שֶׁל מִשְׁפָּחָה וְשֶׁל תּוֹרַת…”?

מה קרה לפני: “הוּא, עֲנַק הָרוּחַ שֶׁלָּמַד 12 שָׁנִים שְׁלֵמוֹת, מוֹצֵא בַּנַּעַר הַזֶּה חָבֵר שָׁוֶה שֶׁמֵּבִין לְעֹמֶק הַסְּבָרוֹת הֲכִי קָשׁוֹת בַּתּוֹרָה!”?
מה קרה אחרי: “מַמָּשׁ אֶת אוֹתוֹ נַעַר אָהוּב וְגָאוֹן עִמּוֹ לָמַד עַד לִפְנֵי רֶגַע בְּבֵית הַמִּדְרָשׁ!”?

סיבה ותוצאה מתוך המעשה

האירוע הנתוןמה גרם לו?מה הייתה תוצאתו?
בַּבַּיִת הַקָּטָן, אִשְׁתּוֹ שֶׁל רַבִּי חָמָא דָּאֲגָה לַיֶּלֶד.
הוּא לֹא יָדַע שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא צָפָה מִלְמַעְלָה וְהֵכִין לוֹ הַפְתָּעָה שֶׁלֹּא חָלַם עָלֶיהָ.

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “הפגישה עם אביו בבית המדרש לאחר 12 שנה” על ר' אושעיא הגדול בן ר' חמא? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “הוּא הִקְשָׁה וְהַנַּעַר תֵּרֵץ; הַנַּעַר הִקְשָׁה וְרַבִּי חָמָא עָנָה.”.
ג. אילו ר' אושעיא הגדול בן ר' חמא לא היה פועל כפי שמתואר ב־“וְהִנֵּה, תּוֹךְ כְּדֵי שֶׁהֵם עֲדַיִן מְדַבְּרִים, הַדֶּלֶת נִפְתֶּחֶת שׁוּב.” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “רַבִּי אוֹשַׁעְיָה רָץ אֶל זְרוֹעוֹתָיו שֶׁל אָבִיו וּשְׁנֵיהֶם הִתְחַבְּקוּ חִבּוּק כָּל כָּךְ אָמִיץ וְחַם, חִבּוּק שֶׁל…” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. להלן מערך סיפורי מקיף, רגשי ומרתק על פגישתם המופלאה של רבי חָמָא…
2. רַבִּי חָמָא יָשַׁב בְּבֵית הַמִּדְרָשׁ וְלָמַד בִּמְסִירוּת נֶפֶשׁ…
3. הָלַךְ קֹדֶם כֹּל לְבֵית הַמִּדְרָשׁ שֶׁל הָעִיר, שָׁלַח שָׁלִיחַ…
4. מִי יוֹדֵעַ מָה קוֹרֶה הַיּוֹם עִם אוֹשַׁעְיָה שֶׁלִּי...
5. זֶה הָיָה מַחֲזֶה שֶׁל עֲנָוָה אֲמִתִּית – תַּלְמִיד חָכָם זָקֵן…
6. תראו לנו כמה תדהמה ושמחה אפשר להכניס בחיבוק הזה של התורה!

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “בְּתֹם שְׁתֵּים עֶשְׂרֵה הַשָּׁנִים, יָצָא רַבִּי חָמָא לַדֶּרֶךְ חֲזָרָה.”.
מכתב: כתבו לר' אושעיא הגדול בן ר' חמא בעקבות “הוּא לֹא יָדַע שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא צָפָה מִלְמַעְלָה וְהֵכִין לוֹ הַפְתָּעָה שֶׁלֹּא חָלַם עָלֶיהָ.”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “אַךְ אָז, הִבִּיטָה אִשְׁתּוֹ שֶׁל רַבִּי חָמָא בְּבַעֲלָהּ.”.
דיווח: כתבו ידיעה קצרה שכותרתה “הפגישה עם אביו בבית המדרש לאחר 12 שנה”.

המחזה

רגע מן הסיפורהדמויותמה יאמרו או יעשו?
בַּבַּיִת הַקָּטָן, אִשְׁתּוֹ שֶׁל רַבִּי חָמָא דָּאֲגָה לַיֶּלֶד.
וְהִנֵּה, תּוֹךְ כְּדֵי שֶׁהֵם עֲדַיִן מְדַבְּרִים, הַדֶּלֶת נִפְתֶּחֶת שׁוּב.

7. חיבור לחיים מתוך המעשה

א. מה אפשר ללמוד מן הדרך שבה ר' אושעיא הגדול בן ר' חמא פעל באירוע “הוּא הִבִּיט בַּנַּעַר הַצָּעִיר, שָׁמַע אֶת חָכְמָתוֹ, וְאָמַר לְעַצְמוֹ בְּלֵב נִשְׁבָּר וּבְעֵינַיִם דּוֹמְעוֹת:”?
