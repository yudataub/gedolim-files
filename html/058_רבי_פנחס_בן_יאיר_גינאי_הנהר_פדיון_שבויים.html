גינאי הנהר — פדיון שבויים

דף עבודה בעקבות רבי פנחס בן יאיר
שם: כיתה:

המעשה — הסיפור המלא

פְּתִיחָה: זְעָקָה מִתּוֹךְ הַבּוֹר

יְלָדִים, עִצְמוּ אֶת הָעֵינַיִם לְרֶגַע וְנַסּוּ לְדַמְיֵן מָקוֹם חָשׁוּךְ מְאֹד. אָנוּ מַרְגִּישִׁים קֹר שֶׁחוֹדֵר לָעֲצָמוֹת, וְרֵיחַ שֶׁל טְחָב וַאֲדָמָה רְטֻבָּה עוֹלֶה בָּאַפַּיִם. זֶהוּ בּוֹר הַכֶּלֶא הַנּוֹרָא שֶׁל הַשּׁוֹדְדִים. בְּתוֹכוֹ, קְשׁוּרִים בְּשַׁרְשְׁרָאוֹת בַּרְזֶל כְּבֵדוֹת שֶׁמְּצַלְצְלוֹת בְּכָל תְּנוּעָה קַלָּה, יוֹשְׁבִים כַּמָּה יְהוּדִים עֲנִיִּים שֶׁנִּלְקְחוּ בַּשֶּׁבִי. הַלֵּב שֶׁלָּהֶם בּוֹכֶה, וְהֵם מִתְפַּלְּלִים לְהַקָּדוֹשׁ בָּרוּךְ הוּא: "אַבָּא, הוֹשִׁיעָה נָּא! מִי יִפְדֶּה אוֹתָנוּ?".

בְּאוֹתָהּ שָׁעָה מַמָּשׁ, בְּמֶרְחָק גָּדוֹל מִשָּׁם, פָּסַע עַל שְׁבִיל הֶעָפָר צַדִּיק יְסוֹד עוֹלָם. הָיָה זֶה רַבִּי פִּינְחָס בֶּן יָאִיר , חֲתָנוֹ הַגָּדוֹל שֶׁל רַבִּי שִׁמְעוֹן בַּר יוֹחַאי. פָּנָיו שֶׁל רַבִּי פִּינְחָס הָיוּ מְאִירוֹת כְּמַלְאַךְ ה', וְלוֹ בִּטָּחוֹן עָצוּם בַּבּוֹרֵא. בְּיָדוֹ אָחַז שַׂקִּיק עִם מַטְבְּעוֹת כֶּסֶף. אֵלּוּ הָיוּ "דְּמֵי הַפִּדְיוֹן" – הַכֶּסֶף שֶׁהוּא אָסַף בִּמְסִירוּת כְּדֵי לְשַׁחְרֵר אֶת אוֹתָם יְהוּדִים מִשֶּׁבִיָּם. מִצְוַת "פִּדְיוֹן שְׁבוּיִים" הִיא אַחַת הַמִּצְווֹת הַגְּדוֹלוֹת בַּתּוֹרָה, וְרַבִּי פִּינְחָס מִהֵר כְּדֵי לְהַצִּיל אֶת חַיֵּיהֶם פְּשׁוּטוֹ כְּמַשְׁמָעוֹ.

הַנָּהָר הַחֹסֵם אֶת הַדֶּרֶךְ

הַדֶּרֶךְ לְאוֹתוֹ כֶּלֶא מְרֻחָק הָיְתָה רְצוּפָה קְשָׁיִים, אַךְ כְּשֶׁהִגִּיעַ רַבִּי פִּינְחָס אֶל פַּאֲתֵי עֵמֶק עָמֹק, הוּא נֶעֱצַר. תָּאֲרוּ לָכֶם רַעַשׁ שֶׁל שְׁאִיגָה אַדִּירָה – "ששששוווו!" מַיִם רַבִּים קוֹצְפִים וְשׁוֹצְפִים, מַכִּים בַּסְּלָעִים מִכָּל עֵבֶר. זֶה הָיָה נְהַר גִּינַאי . בְּיָמִים רְגִילִים אֶפְשָׁר הָיָה לַחֲצוֹת אוֹתוֹ בְּקַלּוּת, אַךְ עַכְשָׁו, לְאַחַר גְּשָׁמִים רַבִּים, הַנָּהָר עָלָה עַל גְּדוֹתָיו. מִי גַּלִּים כַּבִּירִים סָחֲפוּ אִתָּם עֲנָפִים וּבֹץ. אִי אֶפְשָׁר הָיָה לַעֲבֹר בְּלִי לִטְבֹּעַ חָלִילָה! גֶּשֶׁר – לֹא הָיָה. סִירָה – גַּם כֵּן לֹא הָיְתָה בַּנִּמְצָא.

אָדָם רָגִיל הָיָה אָמַר: "רִבּוֹנוֹ שֶׁל עוֹלָם, רָצִיתִי לְקַיֵּם מִצְוָה, אֲבָל הַטֶּבַע עָצַר אוֹתִי. אֲנִי אָנוּס. אֶחְזֹר הַבַּיְתָה". אֲבָל רַבִּי פִּינְחָס? בִּשְׁבִיל לְהַצִּיל נְפָשׁוֹת שֶׁל עַם יִשְׂרָאֵל, לֹא מוְותְּרִים כָּל כָּךְ מַהֵר!

הוּא נֶעֱמַד עַל הַשָּׂפָה הַבֻּצִּית שֶׁל הַנָּהָר, הִבִּיט יָשָׁר אֶל עֵבֶר הַמַּיִם הַגּוֹעֲשִׁים, וְאָמַר בְּקוֹל שָׁקֵט אַךְ תַּקִּיף וְסַמְכוּתִי:

"גִּינַאי הַנָּהָר! חֲלֹק לִי אֶת מֵימֶיךָ (פְּתַח לִי שְׁבִיל) וְאֶעֱבֹר בְּךָ!".

הַוִּכּוּחַ מִתַּחַת לַגַּלִּים

יְלָדִים, וְכִי מַיִם יְכוֹלִים לַעֲנוֹת? הַקָּדוֹשׁ בָּרוּךְ הוּא, שֶׁמַּשְׁגִּיחַ עַל הַכֹּל, נָתַן בְּרֶגַע זֶה קוֹל לְשַׂר הַמְּמֻנֶּה עַל הַנָּהָר בַּשָּׁמַיִם. רַעַשׁ הַמַּיִם כְּאִלּוּ יָצַר מִלִּים:

"רַבִּי פִּינְחָס," אָמַר הַנָּהָר דֶּרֶךְ גַּלָּיו הַנִּשְׁבָּרִים, "אַתָּה הָלוֹךְ לַעֲשׂוֹת רְצוֹן קוֹנְךָ – כְּדֵי לִפְדּוֹת שְׁבוּיִים. אֲבָל גַּם אֲנִי הוֹלֵךְ לַעֲשׂוֹת רְצוֹן קוֹנִי – כִּי ה' צִוָּה עָלַי לִזְרֹם לַיָּם! אֶלָּא שֶׁאַתָּה, סָפֵק תַּצְלִיחַ לַעֲשׂוֹת מִצְוָה זוֹ וְסָפֵק לֹא (אוּלַי הַשּׁוֹבִים לֹא יַסְכִּימוּ לְקַבֵּל אֶת הַכֶּסֶף), וְאִלּוּ אֲנִי וַדַּאי שֶׁאֲנִי עוֹשֶׂה רְצוֹן בּוֹרְאִי עַכְשָׁו כְּשֶׁאֲנִי זוֹרֵם. עַל כֵּן, אֵינֶנִּי צָרִיךְ לְהִבָּקַע מִפָּנֶיךָ!".

וְהַנָּהָר הִמְשִׁיךְ לִשְׁעֹג.

הַאִם רַבִּי פִּינְחָס נִרְתַּע? לֹא. כִּי הוּא יָדַע סוֹד גָּדוֹל: הַתּוֹרָה גְּדוֹלָה מֵהַטֶּבַע. יְהוּדִי שֶׁמּוֹסֵר נֶפֶשׁ עַל מִצְוָה, מוֹשֵׁל עַל הַבְּרִיאָה כֻּלָּהּ. רַבִּי פִּינְחָס הֵרִים אֶת מַטֵּהוּ וְקָרָא בְּקוֹל שֶׁחָתַךְ אֶת שְׁאוֹן הַגַּלִּים:

"אִם אִי אַתָּה חוֹלֵק לִי מֵימֶיךָ – אֲנִי גּוֹזֵר עָלֶיךָ שֶׁלֹּא יַעַבְרוּ בְּךָ מַיִם לְעוֹלָם וָעֶד!".

בְּרֶגַע אֶחָד, קָרָה דָּבָר נִפְלָא. קוֹל סְעָרַת הַמַּיִם הָפַךְ לְדְמָמָה. שְׁשְׁשְׁ...

הַמַּיִם הַגּוֹעֲשִׁים נֶעֶצְרוּ. הֵם נֶעֶמְדוּ כְּמוֹ שְׁנֵי קִירוֹת שְׁקוּפִים שֶׁל אֶבֶן סַפִּיר, הַבֹּץ שֶׁעַל הַקַּרְקָעִית הִתְיַבֵּשׁ, וּשְׁבִיל רָחָב וְנוֹחַ נִפְתַּח בְּלֵב נְהַר הַגִּינַאי! מַמָּשׁ כְּמוֹ קְרִיעַת יַם סוּף!

רַבִּי פִּינְחָס בֶּן יָאִיר הִתְקַדֵּם לְתוֹךְ הַנָּהָר בְּלֵב בּוֹטֵחַ.

הָעַיִן שֶׁלֹּא פִּסְפְסָה אַף אֶחָד

עַכְשָׁו, יְלָדִים, שִׂימוּ לֵב טוֹב לְמָה שֶׁקָּרָה. רַבִּי פִּינְחָס כְּבָר הָיָה יָכוֹל לָרוּץ לַמַּטָּרָה שֶׁלּוֹ. אֲבָל בְּדִיּוּק אָז, הוּא שָׁמַע קוֹל אֲנָחָה מֵאָחוֹר.

הוּא הִסְתּוֹבֵב, וְרָאָה עַל שְׂפַת הַנָּהָר אִישׁ יְהוּדִי פָּשׁוּט וְעָיֵף. הָאִישׁ הַזֶּה לֹא הָלַךְ לַעֲשׂוֹת פִּדְיוֹן שְׁבוּיִים אוֹ דָּבָר גָּדוֹל שֶׁכָּזֶה. הוּא פָּשׁוּט סָחַב עַל כְּתֵפָיו הַכּוֹאֲבוֹת שַׂק גָּדוֹל שֶׁל חִטִּים. "אֲנִי חַיָּב לְהַגִּיעַ לַטַּחֲנָה," מִלְמֵל הַיְּהוּדִי בְּעֶצֶב, "כְּדֵי לִטְחֹן אֶת הַחִטִּים הָאֵלֶּה כְּדֵי שֶׁיִּהְיֶה לִי קֶמַח לֶאֱפוֹת מַצּוֹת לְחַג הַפֶּסַח. אֲבָל אֵיךְ אֶעֱבֹר? הַמַּיִם בְּוַדַּאי יַחְזְרוּ מִיָּד כְּשֶׁהַצַּדִּיק יַעֲבֹר, וַאֲנִי אֶטְבַּע!".

רַבִּי פִּינְחָס רָאָה אֶת הַדְּאָגָה בְּעֵינָיו שֶׁל הַיְּהוּדִי. הוּא יָכוֹל הָיָה לַחְשֹׁב: "אֲנִי עָסוּק עַכְשָׁו בַּדְּבָרִים שֶׁל עַם יִשְׂרָאֵל! אֵין לִי זְמַן לְשַׂק חִטִּים!". אַךְ לֹא! אָמַר הַצַּדִּיק לַנָּהָר: "חֲלֹק לְךָ מֵימֶיךָ גַּם לַיְּהוּדִי הַזֶּה שֶׁנּוֹשֵׂא חִטִּים לְמַצּוֹת מִצְוָה, כִּי גַּם הוּא עוֹשֶׂה רְצוֹן ה'!". וְהַמַּיִם הִמְשִׁיכוּ לַעֲמֹד כַּחוֹמָה, עַד שֶׁהַיְּהוּדִי הַפָּשׁוּט חָצָה אֶת הַנָּהָר בְּשָׁלוֹם עִם הַשַּׂק שֶׁלּוֹ.

וְאָז... רַבִּי פִּינְחָס רָאָה מַשֶּׁהוּ נוֹסָף. מֵאָחוֹרָיו, נִשְׁאַר בּוֹדֵד עוֹד אָדָם אֶחָד. זֶה אֲפִלּוּ לֹא הָיָה אִישׁ יְהוּדִי, אֶלָּא סוֹחֵר נָכְרִי עֲרָבִי (טַיְּיעָא) שֶׁהִתְלַוָּה לְרַבִּי פִּינְחָס בַּדֶּרֶךְ כְּדֵי לִשְׁמֹר עַל עַצְמוֹ מִסַּכָּנוֹת הַדְּרָכִים. הַנָּכְרִי הַזֶּה רָעַד מִפַּחַד עַל הַגָּדָה הַשְּׁנִיָּה.

חָשַׁב רַבִּי פִּינְחָס הַקָּדוֹשׁ: "אִם אֶעֱזֹב אוֹתוֹ שָׁם לְבַדּוֹ, הוּא יַרְגִּישׁ שֶׁזִּלְזַלְתִּי בּוֹ. מָה יֹאמְרוּ הַבְּרִיּוֹת? 'כָּךְ יְהוּדִים עָשׂוּ לְבֶן הַלְּוָיָה שֶׁלָּהֶם? הִשְׁאִירוּ אוֹתוֹ מֵאָחוֹר וְדָאֲגוּ רַק לְעַצְמָם?'. זֶה עָלוּל חָלִילָה לִגְרֹם לְחִלּוּל ה'!".

בְּלִי לְהַסֵּס, פָּנָה הַצַּדִּיק אֶל הַנָּהָר בַּפַּעַם הַשְּׁלִישִׁית: "חֲלֹק עַצְמְךָ גַּם לַנָּכְרִי הַזֶּה! שֶׁלֹּא יֹאמַר – 'כָּךְ יְהוּדִים מְבַזִּים אֶת מִי שֶׁמִּתְלַוֶּה אֲלֵיהֶם'!".

וְשׁוּב, הַמַּיִם הָעַזִּים מָשְׁעוּ בִּקְפִיאָתָם, וְהַסּוֹחֵר הַזָּר חָצָה אֶת הַנָּהָר בְּשָׁלוֹם.

רַק אַחֲרֵי שֶׁכֻּלָּם הָיוּ בְּטוּחִים מֵעֵבֶר לַנָּהָר, רַק אָז הִרְשָׁה רַבִּי פִּינְחָס לַנָּהָר לַחְזֹר וְלִזְרֹם. בְּקוֹל שְׁאָגָה חָזְרוּ הַמַּיִם לִגְעֹשׁ, וְהַצַּדִּיק מִהֵר עִם כַּסְפּוֹ לַבּוֹר, שִׁלֵּם אֶת הַכֹּפֶר, וְהוֹצִיא אֶת הַיְּהוּדִים הַקְּדוֹשִׁים לְאוֹר וּלְחֵרוּת.

וּמָה אָמְרוּ חֲכָמֵינוּ כְּשֶׁשָּׁמְעוּ עַל כָּךְ? הָאֲמוֹרָא הַגָּדוֹל, רַב יוֹסֵף, שָׁמַע מָה שֶׁקָּרָה וְאָמַר בְּהִתְפַּעֲלוּת: "כַּמָּה גָּדוֹל אָדָם זֶה מִמֹּשֶׁה רַבֵּנוּ! שֶׁלְּמֹשֶׁה רַבֵּנוּ נִקְרַע הַיָּם פַּעַם אַחַת, וּלְרַבִּי פִּינְחָס... שָׁלוֹשׁ פְּעָמִים בְּאוֹתוֹ הַיּוֹם!".

מקורות: תלמוד בבלי, מסכת חולין, דף ז' עמוד א': "רבי פנחס בן יאיר הוה קאזיל לפדיון שבויים, פגע ביה בגינאי נהרא... אמר ליה: חלוק לי מימיך ואעבור בך... אמר ליה: אם אי אתה חולק, גוזרני עליך שלא יעברו בך מים לעולם... חלק לההוא גברא דהוה דרי חיטי לפסחא... חלק לההוא טייעא דהוה לוה בהדייהו..." תלמוד ירושלמי, מסכת דמאי, פרק א' הלכה ג': מובא הסיפור בגרסה דומה, שמדגישה גם היא את קריעת הנהר ומדרגתו הרוחנית העליונה. מדרש דברים רבה, פרשה ג': אזכור מופתיו של ר' פנחס ושמירתו המוחלטת על ממון וחפצי אחרים. (כפי שצוין במקורות: ספר האגדה רפ"ג).

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “יְלָדִים, עִצְמוּ אֶת הָעֵינַיִם לְרֶגַע וְנַסּוּ לְדַמְיֵן מָקוֹם חָשׁוּךְ מְאֹד.” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “מִצְוַת "פִּדְיוֹן שְׁבוּיִים" הִיא אַחַת הַמִּצְווֹת הַגְּדוֹלוֹת בַּתּוֹרָה, וְרַבִּי פִּינְחָס מִהֵר כְּדֵי לְהַצִּיל אֶת חַיֵּיהֶם פְּשׁוּטוֹ…”?
ג. כיצד הגיב רבי פנחס בן יאיר למצב המתואר: “הַקָּדוֹשׁ בָּרוּךְ הוּא, שֶׁמַּשְׁגִּיחַ עַל הַכֹּל, נָתַן בְּרֶגַע זֶה קוֹל לְשַׂר הַמְּמֻנֶּה עַל הַנָּהָר בַּשָּׁמַיִם.”?
ד. מה השתנה בין “רַבִּי פִּינְחָס בֶּן יָאִיר הִתְקַדֵּם לְתוֹךְ הַנָּהָר בְּלֵב בּוֹטֵחַ.” לבין “רַבִּי פִּינְחָס רָאָה אֶת הַדְּאָגָה בְּעֵינָיו שֶׁל הַיְּהוּדִי.”?
ה. כיצד הגיע הסיפור אל האירוע: “וְשׁוּב, הַמַּיִם הָעַזִּים מָשְׁעוּ בִּקְפִיאָתָם, וְהַסּוֹחֵר הַזָּר חָצָה אֶת הַנָּהָר בְּשָׁלוֹם.”?

משפט מן המעשה או שינוי שלונכוןלא נכוןתקנו או הביאו הוכחה
בְּאוֹתָהּ שָׁעָה מַמָּשׁ, בְּמֶרְחָק גָּדוֹל מִשָּׁם, פָּסַע עַל שְׁבִיל הֶעָפָר צַדִּיק יְסוֹד עוֹלָם.□□
בסיפור לא התרחש הדבר הבא: אָדָם רָגִיל הָיָה אָמַר: "רִבּוֹנוֹ שֶׁל עוֹלָם, רָצִיתִי לְקַיֵּם מִצְוָה, אֲבָל הַטֶּבַע עָצַר אוֹתִי.□□
כִּי הוּא יָדַע סוֹד גָּדוֹל: הַתּוֹרָה גְּדוֹלָה מֵהַטֶּבַע.□□
בסיפור לא התרחש הדבר הבא: הָאִישׁ הַזֶּה לֹא הָלַךְ לַעֲשׂוֹת פִּדְיוֹן שְׁבוּיִים אוֹ דָּבָר גָּדוֹל שֶׁכָּזֶה.□□
זֶה אֲפִלּוּ לֹא הָיָה אִישׁ יְהוּדִי, אֶלָּא סוֹחֵר נָכְרִי עֲרָבִי (טַיְּיעָא) שֶׁהִתְלַוָּה לְרַבִּי פִּינְחָס בַּדֶּרֶךְ כְּדֵי לִשְׁמֹר עַל עַצְמוֹ מִסַּכָּנוֹת הַדְּרָכִים.□□

3. מה היה ראשון?

כתבו ליד כל אירוע מספר מ־1 עד 5 לפי סדר התרחשותו בסיפור.

□ רַבִּי פִּינְחָס הֵרִים אֶת מַטֵּהוּ וְקָרָא בְּקוֹל שֶׁחָתַךְ אֶת שְׁאוֹן הַגַּלִּים:
□ יְלָדִים, עִצְמוּ אֶת הָעֵינַיִם לְרֶגַע וְנַסּוּ לְדַמְיֵן מָקוֹם חָשׁוּךְ מְאֹד.
□ שֶׁלְּמֹשֶׁה רַבֵּנוּ נִקְרַע הַיָּם פַּעַם אַחַת, וּלְרַבִּי פִּינְחָס...
□ בְּיָמִים רְגִילִים אֶפְשָׁר הָיָה לַחֲצוֹת אוֹתוֹ בְּקַלּוּת, אַךְ עַכְשָׁו, לְאַחַר גְּשָׁמִים רַבִּים, הַנָּהָר עָלָה עַל גְּדוֹתָיו.
□ הוּא יָכוֹל הָיָה לַחְשֹׁב: "אֲנִי עָסוּק עַכְשָׁו בַּדְּבָרִים שֶׁל עַם יִשְׂרָאֵל!

לפני ואחרי

מה קרה לפני: “רַבִּי פִּינְחָס הֵרִים אֶת מַטֵּהוּ וְקָרָא בְּקוֹל שֶׁחָתַךְ אֶת שְׁאוֹן הַגַּלִּים:”?
מה קרה אחרי: “הוּא יָכוֹל הָיָה לַחְשֹׁב: "אֲנִי עָסוּק עַכְשָׁו בַּדְּבָרִים שֶׁל עַם יִשְׂרָאֵל!”?

סיבה ותוצאה מתוך המעשה

האירוע הנתוןמה גרם לו?מה הייתה תוצאתו?
אָדָם רָגִיל הָיָה אָמַר: "רִבּוֹנוֹ שֶׁל עוֹלָם, רָצִיתִי לְקַיֵּם מִצְוָה, אֲבָל הַטֶּבַע עָצַר אוֹתִי.
הָאִישׁ הַזֶּה לֹא הָלַךְ לַעֲשׂוֹת פִּדְיוֹן שְׁבוּיִים אוֹ דָּבָר גָּדוֹל שֶׁכָּזֶה.

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “גינאי הנהר — פדיון שבויים” על רבי פנחס בן יאיר? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “כִּי הוּא יָדַע סוֹד גָּדוֹל: הַתּוֹרָה גְּדוֹלָה מֵהַטֶּבַע.”.
ג. אילו רבי פנחס בן יאיר לא היה פועל כפי שמתואר ב־“רַבִּי פִּינְחָס רָאָה אֶת הַדְּאָגָה בְּעֵינָיו שֶׁל הַיְּהוּדִי.” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “וְשׁוּב, הַמַּיִם הָעַזִּים מָשְׁעוּ בִּקְפִיאָתָם, וְהַסּוֹחֵר הַזָּר חָצָה אֶת הַנָּהָר בְּשָׁלוֹם.” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. יְלָדִים, עִצְמוּ אֶת הָעֵינַיִם לְרֶגַע וְנַסּוּ לְדַמְיֵן מָקוֹם…
2. הַדֶּרֶךְ לְאוֹתוֹ כֶּלֶא מְרֻחָק הָיְתָה רְצוּפָה קְשָׁיִים, אַךְ…
3. אֲבָל גַּם אֲנִי הוֹלֵךְ לַעֲשׂוֹת רְצוֹן קוֹנִי – כִּי ה' צִוָּה עָלַי…
4. אֲבָל בְּדִיּוּק אָז, הוּא שָׁמַע קוֹל אֲנָחָה מֵאָחוֹר.
5. וְהַמַּיִם הִמְשִׁיכוּ לַעֲמֹד כַּחוֹמָה, עַד שֶׁהַיְּהוּדִי הַפָּשׁוּט…
6. שֶׁלְּמֹשֶׁה רַבֵּנוּ נִקְרַע הַיָּם פַּעַם אַחַת, וּלְרַבִּי…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “הַקָּדוֹשׁ בָּרוּךְ הוּא, שֶׁמַּשְׁגִּיחַ עַל הַכֹּל, נָתַן בְּרֶגַע זֶה קוֹל לְשַׂר הַמְּמֻנֶּה עַל הַנָּהָר בַּשָּׁמַיִם.”.
מכתב: כתבו לרבי פנחס בן יאיר בעקבות “הָאִישׁ הַזֶּה לֹא הָלַךְ לַעֲשׂוֹת פִּדְיוֹן שְׁבוּיִים אוֹ דָּבָר גָּדוֹל שֶׁכָּזֶה.”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “זֶה אֲפִלּוּ לֹא הָיָה אִישׁ יְהוּדִי, אֶלָּא סוֹחֵר נָכְרִי עֲרָבִי (טַיְּיעָא) שֶׁהִתְלַוָּה לְרַבִּי פִּינְחָס בַּדֶּרֶךְ…”.
דיווח: כתבו ידיעה קצרה שכותרתה “גינאי הנהר — פדיון שבויים”.

המחזה

רגע מן הסיפורהדמויותמה יאמרו או יעשו?
אָדָם רָגִיל הָיָה אָמַר: "רִבּוֹנוֹ שֶׁל עוֹלָם, רָצִיתִי לְקַיֵּם מִצְוָה, אֲבָל הַטֶּבַע עָצַר…
רַבִּי פִּינְחָס רָאָה אֶת הַדְּאָגָה בְּעֵינָיו שֶׁל הַיְּהוּדִי.
