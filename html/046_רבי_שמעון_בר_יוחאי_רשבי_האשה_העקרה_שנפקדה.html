האשה העקרה שנפקדה

דף עבודה בעקבות רבי שמעון בר יוחאי (רשב"י)
שם: כיתה:

המעשה — הסיפור המלא

להלן עיבוד חינוכי, רגשי ומרתק לסיפור המופלא על רבי שמעון בר יוחי והאישה מצידון , המעובד במיוחד לבנים בכיתות ב'-ד' (חינוך דתי-לאומי). הסיפור מדגיש את קדושתו של רשב"י שראה למרחוק, את ערכם של שלום בית ואהבה, ואת ההבנה שהאוצר האמיתי בחיים אינו כסף או זהב – אלא האנשים שאנחנו אוהבים.

הסיפור מנוקד במלואו כנדרש.

פְּתִיחָה: הָאַרְמוֹן הַשָּׁקֵט מִדַּי

יְלָדִים, עִצְמוּ אֶת הָעֵינַיִם וְנַסּוּ לְדַמְיֵן בַּיִת שֶׁל אֲנָשִׁים מְאֹד מְאֹד עֲשִׁירִים. בַּיִת עָנָק, מָלֵא בִּשְׁטִיחִים מִמֶּשִׁי רַךְ שֶׁנָּעִים לִדְרֹךְ עֲלֵיהֶם. רֵיחַ נִפְלָא שֶׁל בְּשָׂמִים יְקָרִים מְמַלֵּא אֶת הָאֲוִיר, וְעַל הַשֻּׁלְחָן כְּלֵי כֶּסֶף וְזָהָב נוֹצְצִים מֵאוֹר הַשֶּׁמֶשׁ.

בְּבַיִת כָּזֶה בְּדִיּוּק, בָּעִיר צִידוֹן, גָּרוּ אִישׁ וְאִשָּׁה, יְהוּדִים יִרְאֵי שָׁמַיִם וְטוֹבֵי לֵב.

אֲבָל אִם הֱיִיתֶם נִכְנָסִים לַבַּיִת הַזֶּה, הֱיִיתֶם מַרְגִּישִׁים מִיָּד שֶׁמַּשֶּׁהוּ חָסֵר. הָיָה שָׁם... שָׁקֵט מִדַּי.

לֹא שָׁמְעוּ שָׁם קוֹלוֹת שֶׁל יְלָדִים מְשַׂחֲקִים. לֹא הָיוּ שָׁם קֻבִּיּוֹת מְפֻזָּרוֹת עַל הָרִצְפָּה, וְלֹא קוֹל שֶׁל תִּינוֹק הַבּוֹכֶה בַּלַּיְלָה.

כְּבָר עֶשֶׂר שָׁנִים שֶׁהָאִישׁ וְהָאִשָּׁה הַלָּלוּ נְשׂוּאִים, הֵם אָהֲבוּ זֶה אֶת זוֹ מְאֹד, תָּמִיד דִּבְּרוּ בְּכָבוֹד, עָזְרוּ לָעֲנִיִּים בַּכֶּסֶף שֶׁלָּהֶם, אֲבָל הַקָּדוֹשׁ בָּרוּךְ הוּא לֹא נָתַן לָהֶם יְלָדִים. כָּל לַיְלָה, כְּשֶׁהַבַּיִת הֶחְשִׁיךְ, הָיְתָה הָאִשָּׁה מוֹרִידָה דִּמְעָה חַמָּה עַל כָּרִיתָהּ, וְגַם לִבּוֹ שֶׁל הַבַּעַל קָרַע מִצַּעַר. הֵם כָּל כָּךְ רָצוּ לְהָקִים מִשְׁפָּחָה!

הַהַחְלָטָה הַקָּשָׁה וְהַבִּקּוּר בְּמֵרוֹן

עַל פִּי הַהֲלָכָה שֶׁל הַתּוֹרָה, אִם אִישׁ וְאִשָּׁה נְשׂוּאִים עֶשֶׂר שָׁנִים וְאֵין לָהֶם יְלָדִים, הֵם צְרִיכִים לְהִתְגָּרֵשׁ. הַבַּעַל חַיָּב לְקַיֵּם אֶת מִצְוַת "פְּרִיָּה וּרְבִיָּה", לְהָבִיא יְלָדִים לָעוֹלָם שֶׁיִּלְמְדוּ תּוֹרָה.

זֶה הָיָה הַיּוֹם הֶעָצוּב בְּיוֹתֵר בְּחַיֵּיהֶם. הֵם לֹא כָּעֲסוּ אֶחָד עַל הַשְּׁנִיָּה, חָלִילָה, הֵם פָּשׁוּט בָּכוּ. הֵם הֶחְלִיטוּ שֶׁלִּפְנֵי שֶׁהֵם נִפְרָדִים, הֵם יִסְעוּ לְאֶרֶץ יִשְׂרָאֵל, אֶל הָהָר הַגָּבוֹהַּ שֶׁל מֵרוֹן, כְּדֵי לְקַבֵּל הַדְרָכָה מִגְּדוֹל הַדּוֹר, הַתַּנָּא הַקָּדוֹשׁ – רַבִּי שִׁמְעוֹן בַּר יוֹחַאי .

הֵם נִכְנְסוּ לְבֵית מִדְרָשׁוֹ שֶׁל רַשְׁבִּ"י. מִי שֶׁרָאָה אֶת רַבִּי שִׁמְעוֹן, הִרְגִּישׁ שֶׁקַּרְנֵי שֶׁמֶשׁ שֶׁל קְדֻשָּׁה מְאִירוֹת מִפָּנָיו. הָאִישׁ וְהָאִשָּׁה עָמְדוּ מוּלוֹ, הִרְכִּינוּ אֶת רֹאשָׁם, וְהַבַּעַל אָמַר בְּקוֹל שָׁבוּר:

"רַבִּי, אֲנַחְנוּ נְשׂוּאִים עֶשֶׂר שָׁנִים. הַקָּדוֹשׁ בָּרוּךְ הוּא לֹא זִכָּה אוֹתָנוּ בְּפְרִי בֶּטֶן. אָנוּ רוֹצִים לְהִפָּרֵד כַּדָּת וְכַדִּין".

רַבִּי שִׁמְעוֹן שָׁתַק מִסְפַּר שְׁנִיּוֹת. עֵינָיו הַקְּדוֹשׁוֹת יָרְדוּ עַד לְעִמְקֵי הַנְּשָׁמָה שֶׁל הַזּוּג שֶׁלְּפָנָיו. הוּא רָאָה אֶת הָאַהֲבָה הַגְּדוֹלָה שֶׁיֵּשׁ בֵּינֵיהֶם. הוּא יָדַע בְּרוּחַ הַקֹּדֶשׁ שֶׁאָסוּר לָהֶם לְהִפָּרֵד!

"בָּנַי הַיְקָרִים," אָמַר לָהֶם רַבִּי שִׁמְעוֹן בְּקוֹל חַם וְעָמוֹק, "מִכֵּיוָן שֶׁאַתֶּם אֲנָשִׁים מְכֻבָּדִים, אֲנִי מְבַקֵּשׁ מִכֶּם בַּקָּשָׁה מְיֻחֶדֶת. כְּשֶׁהִתְחַתַּנְתֶּם, הֲרֵי עֲשִׂיתֶם מְסִבָּה גְּדוֹלָה עִם מַאֲכָל וּמִשְׁתֶּה? כָּךְ אֲנִי גּוֹזֵר עֲלֵיכֶם לַעֲשׂוֹת כָּעֵת. שׁוּבוּ לְבֵיתְכֶם, עִרְכוּ סְעוּדָה גְּדוֹלָה שֶׁל פְּרֵדָה – עִם אוֹכֶל שֶׁל מְלָכִים וְיַיִן מְשֻׁבָּח. רַק לְאַחַר מִכֵּן, תִּפָּרְדוּ!".

הַסְּעוּדָה הָאַחֲרוֹנָה וְהַבַּקָּשָׁה

הַזּוּג לֹא הֵבִין, אֲבָל מִי מֵעֵז לְהִתְוַכֵּחַ עִם רַבִּי שִׁמְעוֹן בַּר יוֹחַאי? הֵם חָזְרוּ לְצִידוֹן וְהֵכִינוּ סְעוּדַת מְלָכִים.

אֵיזֶה שֻׁלְחָן זֶה הָיָה! רֵיחוֹת שֶׁל בָּשָׂר צָלוּי, חַלּוֹת טְרִיּוֹת וּפֵרוֹת נִפְלָאִים מִלְּאוּ אֶת הַבַּיִת. עַל הַשֻּׁלְחָן עָמְדוּ גְּבִיעֵי יַיִן יָקָר.

בְּמַהֲלַךְ הַסְּעוּדָה, הַבַּעַל שָׁתָה עוֹד כּוֹס וְעוֹד כּוֹס. לִבּוֹ הִתְמַלֵּא בְּחוֹם, וְהוּא הִרְגִּישׁ קִרְבָה עֲצוּמָה לְאִשְׁתּוֹ. הוּא הִסְתַּכֵּל עָלֶיהָ, עַל הָאִשָּׁה הַטּוֹבָה שֶׁדָּאֲגָה לוֹ כָּל כָּךְ הַרְבֵּה שָׁנִים, וְאָמַר לָהּ בְּקוֹל אוֹהֵב וְנִרְגָּשׁ:

"אִשְׁתִּי הַיְקָרָה. בֹּקֶר יַגִּיעַ, וַאֲנַחְנוּ נִפָּרֵד וְתַחְזְרִי לְבֵית אָבִיךְ. אֲבָל דְּעִי לָךְ, שֶׁאֲנִי רוֹצֶה שֶׁתִּקְחִי מֵהַבַּיִת שֶׁלִּי אֶת חֵלֶקְךָ. מָחָר, תִּקְחִי לָךְ כָּל חֵפֶץ שֶׁאַתְּ אוֹהֶבֶת . הַדָּבָר הֲכִי יָקָר שֶׁיֵּשׁ לִי בַּבַּיִת, תַּכְשִׁיט, כֶּסֶף, אֶבֶן חֵן – קְחִי אוֹתוֹ שֶׁיִּהְיֶה שֶׁלָּךְ בְּבֵית אָבִיךְ!".

הָאִשָּׁה חִיְּכָה וְהִנְהֲנָה לְאַט.

אַחֲרֵי הַסְּעוּדָה, מֵרֹב אֹכֶל וְיַיִן שֶׁשָּׁתָה, חָשׁ הַבַּעַל עֲיֵפוּת עֲצוּמָה. הוּא עָצַם אֶת עֵינָיו עַל כּוּרְסָתוֹ וְנִרְדַּם שֵׁנָה עֲמֻקָּה. רְחִימוּת (נְחִירוֹת) קַלּוֹת נִשְׁמְעוּ מִפִּיו.

הַהַפְתָּעָה בְּאֶמְצַע הַלַּיְלָה

הָאִשָּׁה קָמָה מִיָּד. הִיא לֹא נִגְּשָׁה לְכוֹסֶפֶת הַתַּכְשִׁיטִים, הִיא לֹא נִגְּשָׁה לַשְּׁטִיחִים. הִיא קָרְאָה בְּשֶׁקֶט לַמְּשָׁרְתִים שֶׁלָּהּ.

"קְחוּ אֶת הַמִּטָּה שֶׁל בַּעֲלִי," הִיא לָחֲשָׁה וְהִצְבִּיעָה עַל הַבַּעַל הַיָּשֵׁן. "הָרִימוּ אוֹתוֹ בַּעֲדִינוּת, וְהַעֲבִירוּ אוֹתוֹ כְּמוֹ שֶׁהוּא – אֶל הַבַּיִת שֶׁל אַבָּא שֶׁלִּי!".

הַמְּשָׁרְתִים פָּעֲרוּ עֵינַיִם הֲמוּמוֹת, אֲבָל עָשׂוּ כְּמִצְוָתָהּ. בְּחֶשְׁכַת הַלַּיְלָה, בְּשֶׁקֶט בְּשֶׁקֶט, נָשְׂאוּ אֶת הַבַּעַל הַיָּשֵׁן וְהִנִּיחוּ אוֹתוֹ בְּחַדְרָהּ בְּבֵית הוֹרֶיהָ.

הַיְּקִיצָה וְהַנֵּס

בַּחֲצוֹת הַלַּיְלָה, פָּקַח הַבַּעַל אֶת עֵינָיו. הוּא הִבִּיט לְמַעְלָה... הַתִּקְרָה שׁוֹנָה! הוּא הִבִּיט לַצְּדָדִים... הַקִּירוֹת שׁוֹנִים! רֵיחַ אַחֵר בָּאֲוִיר!

"הֵיכָן אֲנִי?!" קָרָא הַבַּעַל בְּבֶהָלָה. "הַאִם נֶחְטַפְתִּי?".

הָאִשָּׁה שֶׁיָּשְׁבָה כָּל הַזְּמַן לְיַד מִטָּתוֹ, אָחֲזָה בְּיָדוֹ וְחִיְּכָה:

"לֹא נֶחְטַפְתָּ, בַּעֲלִי הַיָּקָר. אַתָּה נִמְצָא אִתִּי, בְּבֵית אָבִי".

"אֲבָל מַה לִּי וּלְבֵית אָבִיךְ?!" תָּמַהּ הַבַּעַל. "הֲרֵי הָיִינוּ אֲמוּרִים לְהִפָּרֵד!".

דִּמְעָה זוֹלֶגֶת עַל לֶחְיָהּ שֶׁל הָאִשָּׁה, וְהִיא עוֹנָה לוֹ מִלִּים שֶׁיְּהַדְהֲדוּ לָנֶצַח:

"בַּעֲלִי, הֲרֵי בַּסְּעוּדָה אָמַרְתָּ לִי מְפֹרָשׁ 'קְחִי לָךְ אֶת הַחֵפֶץ הֲכִי יָקָר וְהֲכִי אָהוּב עָלַיִךְ שֶׁיֵּשׁ לִי בַּבַּיִת'. חִפַּשְׂתִּי וְחִפַּשְׂתִּי... וְלֹא מָצָאתִי בָּעוֹלָם כֻּלּוֹ דָּבָר יָקָר לִי יוֹתֵר מִמְּךָ! לֹא זָהָב, לֹא כֶּסֶף וְלֹא יַהֲלוֹמִים. אַתָּה הוּא הַדָּבָר שֶׁאֲנִי הֲכִי רוֹצָה בָּעוֹלָם!".

כְּשֶׁשָּׁמַע זֹאת הַבַּעַל, זָלְגוּ דְּמָעוֹת גַּם מֵעֵינָיו. שְׁנֵיהֶם הֵבִינוּ שֶׁהֵם לֹא יְכוֹלִים לְהִפָּרֵד. הַקֶּשֶׁר בֵּינֵיהֶם כָּל כָּךְ קָדוֹשׁ וַאֲמִתִּי.

כְּשֶׁעָלָה הַבֹּקֶר, חָזְרוּ שְׁנֵיהֶם לְמֵרוֹן, אֶל רַבִּי שִׁמְעוֹן בַּר יוֹחַאי. מִיָּד כְּשֶׁנִּכְנְסוּ, הֵבִין רַשְׁבִּ"י בְּחָכְמָתוֹ אֶת אֲשֶׁר אֵרַע וְחִיֵּךְ חִיּוּךְ רָחָב וְקָדוֹשׁ.

רַשְׁבִּ"י עָמַד בִּתְפִלָּה עֲמוּקָה לִפְנֵי רִבּוֹנוֹ שֶׁל עוֹלָם: "אָנָּא ה', רְאֵה אֶת הָאַהֲבָה וְהַשָּׁלוֹם שֶׁיֵּשׁ בֵּין בָּנֶיךָ הַצַּדִּיקִים הָאֵלֶּה. זַכֵּה אוֹתָם בְּפְרִי בֶּטֶן!".

וְהַקָּדוֹשׁ בָּרוּךְ הוּא, שֶׁאוֹהֵב בָּתִּים שֶׁיֵּשׁ בָּהֶם שָׁלוֹם, שָׁמַע אֶת תְּפִלַּת הַתַּנָּא הַקָּדוֹשׁ. כַּעֲבֹר תִּשְׁעָה חֳדָשִׁים, הִתְמַלֵּא הָאַרְמוֹן הַשָּׁקֵט בְּצִידוֹן בְּקוֹלוֹת בֶּכִי מְתוּקִים שֶׁל תִּינוֹק קָטָן וְטָהוֹר, וְהַשִּׂמְחָה שָׁלְמָה!

א. שְׁאֵלַת פְּתִיחָה:

"יְלָדִים, אִם הָיוּ אוֹמְרִים לָכֶם לְהִכָּנֵס לְחֶדֶר מָלֵא בְּכָל הַדְּבָרִים שֶׁבָּעוֹלָם, וְלוֹקְחִים מִשָּׁם רַק דָּבָר אֶחָד יָקָר בְּיוֹתֵר וְיוֹצְאִים – מָה הֱיִיתֶם בּוֹחֲרִים? (תְּנוּ לַיְלָדִים לְהָשִׁיב – הֵם וַדַּאי יַגִּידוּ זָהָב, אֵקְסְבּוֹקְס, כֶּסֶף כְּדֵי לִקְנוֹת עוֹד דְּבָרִים וכו').

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “להלן עיבוד חינוכי, רגשי ומרתק לסיפור המופלא על רבי שמעון בר יוחי והאישה מצידון , המעובד במיוחד לבנים בכיתות ב'-ד' (חינוך דתי-לאומי).” — מה קרה מיד לאחר מכן?

ה. כיצד הגיע הסיפור אל האירוע: “(תְּנוּ לַיְלָדִים לְהָשִׁיב – הֵם וַדַּאי יַגִּידוּ זָהָב, אֵקְסְבּוֹקְס, כֶּסֶף כְּדֵי לִקְנוֹת עוֹד דְּבָרִים וכו').”?

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “האשה העקרה שנפקדה” על רבי שמעון בר יוחאי (רשב"י)? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “רֵיחוֹת שֶׁל בָּשָׂר צָלוּי, חַלּוֹת טְרִיּוֹת וּפֵרוֹת נִפְלָאִים מִלְּאוּ אֶת הַבַּיִת.”.
ג. אילו רבי שמעון בר יוחאי (רשב"י) לא היה פועל כפי שמתואר ב־“דִּמְעָה זוֹלֶגֶת עַל לֶחְיָהּ שֶׁל הָאִשָּׁה, וְהִיא עוֹנָה לוֹ מִלִּים שֶׁיְּהַדְהֲדוּ לָנֶצַח:” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “(תְּנוּ לַיְלָדִים לְהָשִׁיב – הֵם וַדַּאי יַגִּידוּ זָהָב, אֵקְסְבּוֹקְס, כֶּסֶף כְּדֵי לִקְנוֹת עוֹד דְּבָרִים וכו').” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. להלן עיבוד חינוכי, רגשי ומרתק לסיפור המופלא על רבי שמעון בר יוחי והאישה…
2. כָּל לַיְלָה, כְּשֶׁהַבַּיִת הֶחְשִׁיךְ, הָיְתָה הָאִשָּׁה מוֹרִידָה…
3. בָּנַי הַיְקָרִים," אָמַר לָהֶם רַבִּי שִׁמְעוֹן בְּקוֹל חַם וְעָמוֹק,…
4. הוּא עָצַם אֶת עֵינָיו עַל כּוּרְסָתוֹ וְנִרְדַּם שֵׁנָה עֲמֻקָּה.
5. כְּשֶׁעָלָה הַבֹּקֶר, חָזְרוּ שְׁנֵיהֶם לְמֵרוֹן, אֶל רַבִּי שִׁמְעוֹן…
6. (כְּפִי שֶׁמוֹפִיעַ בַּסִּפְרוּת וּבַמָּקוֹר הֶחָשׁוּב שֶׁסֻּפַּק:…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “הַקָּדוֹשׁ בָּרוּךְ הוּא לֹא זִכָּה אוֹתָנוּ בְּפְרִי בֶּטֶן.”.
מכתב: כתבו לרבי שמעון בר יוחאי (רשב"י) בעקבות “קְחוּ אֶת הַמִּטָּה שֶׁל בַּעֲלִי," הִיא לָחֲשָׁה וְהִצְבִּיעָה עַל הַבַּעַל הַיָּשֵׁן.”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “מִיָּד כְּשֶׁנִּכְנְסוּ, הֵבִין רַשְׁבִּ"י בְּחָכְמָתוֹ אֶת אֲשֶׁר אֵרַע וְחִיֵּךְ חִיּוּךְ רָחָב וְקָדוֹשׁ.”.
דיווח: כתבו ידיעה קצרה שכותרתה “האשה העקרה שנפקדה”.

המחזה

רגע מן הסיפורהדמויותמה יאמרו או יעשו?
הֵם לֹא כָּעֲסוּ אֶחָד עַל הַשְּׁנִיָּה, חָלִילָה, הֵם פָּשׁוּט בָּכוּ.
דִּמְעָה זוֹלֶגֶת עַל לֶחְיָהּ שֶׁל הָאִשָּׁה, וְהִיא עוֹנָה לוֹ מִלִּים שֶׁיְּהַדְהֲדוּ לָנֶצַח:

7. חיבור לחיים מתוך המעשה

א. מה אפשר ללמוד מן הדרך שבה רבי שמעון בר יוחאי (רשב"י) פעל באירוע “הַדָּבָר הֲכִי יָקָר שֶׁיֵּשׁ לִי בַּבַּיִת, תַּכְשִׁיט, כֶּסֶף, אֶבֶן חֵן – קְחִי אוֹתוֹ שֶׁיִּהְיֶה שֶׁלָּךְ בְּבֵית אָבִיךְ!".”?
