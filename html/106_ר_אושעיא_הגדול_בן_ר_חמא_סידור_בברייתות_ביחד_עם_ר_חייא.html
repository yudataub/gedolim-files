סידור בברייתות ביחד עם ר' חייא

דף עבודה בעקבות ר' אושעיא הגדול בן ר' חמא
שם: כיתה:

המעשה — הסיפור המלא

פְּתִיחָה: הָאוֹצָר שֶׁנִּשְׁאַר בַּחוּץ

יְלָדִים יְקָרִים, דַּמְיְנוּ לָעַצְמְכֶם חֶדֶר עֲנָק, מָלֵא בְּהַרְבֵּה מְאֹד יַהֲלוֹמִים נוֹצְצִים. יַהֲלוֹמִים שֶׁשָּׁוִים מִילְיַארְדֵי שְׁקָלִים. הַמֶּלֶךְ בִּקֵּשׁ מֵהָאֲנָשִׁים שֶׁלּוֹ לַאֲסֹף אֶת כָּל הַיַּהֲלוֹמִים וּלְהַכְנִיס אוֹתָם לְתוֹךְ תֵּבַת זָהָב שְׁמוּרָה, מִפַּחַד שֶׁלֹּא יֵלְכוּ לְאִבּוּד. הָאֲנָשִׁים אָסְפוּ אֶת כָּל הַיַּהֲלוֹמִים שֶׁהֵם מָצְאוּ, נָעֲלוּ אֶת הַתֵּבָה בְּמַנְעוּל כָּבֵד, וְנָשְׁמוּ לִרְוָחָה. "בָּרוּךְ ה', סִיַּמְנוּ!", הֵם שָׂמְחוּ.

אֲבָל... בַּחוּץ, מִתַּחַת לַשֻּׁלְחָנוֹת, בְּתוֹךְ כִּיסִים נִסְתָּרִים וּבַפִּנּוֹת שֶׁל הַחֶדֶר, עֲדַיִן נִשְׁאֲרוּ פְּזוּרִים אַלְפֵי יַהֲלוֹמִים יְקָרִים מְאֹד, יַהֲלוֹמִים נְדִירִים שֶׁלֹּא נִכְנְסוּ לַתֵּבָה הַגְּדוֹלָה! וְהַיַּהֲלוֹמִים הָאֵלֶּה הָיוּ בְּסַכָּנָה עֲצוּמָה.

יְלָדִים, הַיַּהֲלוֹמִים הָאֵלֶּה אֵינָם אֲבָנִים שֶׁל כֶּסֶף. הֵם הָיוּ יַהֲלוֹמִים שֶׁל הֲלָכוֹת! כַּאֲשֶׁר רַבִּי יְהוּדָה הַנָּשִׂיא – "רַבֵּנוּ הַקָּדוֹשׁ" – חָתַם אֶת הַמִּשְׁנָה, הוּא בָּחַר אֶת הַהֲלָכוֹת הֲכִי חֲשׁוּבוֹת וְסִדֵּר אוֹתָן בְּשִׁשָּׁה סִדְרֵי מִשְׁנָה. זֶה הָיָה הָאוֹצָר הַגָּדוֹל, "תֵּבַת הַזָּהָב" שֶׁל עַם יִשְׂרָאֵל.

אַךְ מָה קָרָה לְכָל שְׁאָר הַהֲלָכוֹת וְהַמָּסוֹרוֹת הַקְּדוֹשׁוֹת שֶׁנִּמְסְרוּ לְמֹשֶׁה בְּסִינַי וְלֹא נִכְנְסוּ לַמִּשְׁנָה? מָה עִם כָּל מָה שֶׁלִּמְּדוּ רַבִּי עֲקִיבָא, רַבִּי מֵאִיר וְרַבִּי שִׁמְעוֹן בַּר יוֹחַאי בְּבָתֵּי הַמִּדְרָשׁ שֶׁלָּהֶם? הַהֲלָכוֹת הָאֵלֶּה נִקְרְאוּ " בָּרַיְיתוֹת " (בָּרַיְיתָא בַּאֲרָמִית פֵּרוּשָׁהּ – חִיצוֹנִית, הֲלָכָה שֶׁנִּשְׁאֲרָה "בַּחוּץ").

וְכָאן לְפֶתַע, הִתְחִילָה בְּעָיָה גְּדוֹלָה מְאֹד.

הַשְּׁגִיאוֹת הַמְּסֻכָּנוֹת בְּבֵית הַמִּדְרָשׁ

הַשָּׁנִים מִהֲרוּ לַחְלֹף. בְּבָתֵּי הַמִּדְרָשׁ בְּאֶרֶץ יִשְׂרָאֵל יָשְׁבוּ אֲלָפִים שֶׁל תַּלְמִידֵי חֲכָמִים. הַקּוֹלוֹת שֶׁל לִמּוּד הַתּוֹרָה מִלְּאוּ אֶת הָאֲוִיר, זִמְזוּם מָתוֹק וְקָדוֹשׁ כְּמוֹ שֶׁל דְּבוֹרִים הָאוֹסְפוֹת צִיּוּף. "אָמַר רַבִּי פְּלוֹנִי... הֲלָכָה כָּךְ וְכָךְ... מֻתָּר! אָסוּר!".

אֲבָל פִּתְאוֹם, תַּלְמִידִים הִתְחִילוּ לְהִתְבַּלְבֵּל. יְלָדִים, אַתֶּם מַכִּירִים אֶת הַמִּשְׂחָק "טֵלֵפוֹן שָׁבוּר"? כְּשֶׁמִּישֶׁהוּ לוֹחֵשׁ מִלָּה, וְהַשֵּׁנִי לוֹחֵשׁ לַשְּׁלִישִׁי, וּבַסּוֹף הַמִּלָּה מִשְׁתַּנָּה לְגַמְרֵי?

בַּתּוֹרָה, אָסוּר שֶׁיִּהְיֶה "טֵלֵפוֹן שָׁבוּר"! אִם מִישֶׁהוּ מִתְבַּלְבֵּל בְּמִלָּה אַחַת, בִּמְקוֹם "מֻתָּר" הוּא יַגִּיד "אָסוּר". בִּמְקוֹם לְטַהֵר, הוּא יְטַמֵּא!

אֲנָשִׁים קָמוּ בְּבֵית הַמִּדְרָשׁ וְאָמְרוּ: "שָׁמַעְתִּי בָּרַיְיתָא הֲלָכָה שֶׁאוֹמֶרֶת כָּךְ!", אֲבָל חֲבֵרוֹ צָעַק לְעֻמָּתוֹ: "מָה פִּתְאוֹם! אֲנִי שָׁמַעְתִּי אֶת אוֹתָהּ הֲלָכָה בְּדִיּוּק הָפוּךְ!".

סַכָּנָה אֲיֻמָּה רִחֲפָה עַל תּוֹרַת יִשְׂרָאֵל. הַ"יַּהֲלוֹמִים" הַקְּדוֹשִׁים הִתְחִילוּ לְהִתְלַכְלֵךְ בְּבֹץ שֶׁל שְׁגִיאוֹת, וַאֲנָשִׁים כְּבָר לֹא יָדְעוּ מָה הָאֱמֶת.

שְׁנֵי שׁוֹמְרֵי הָאֱמֶת וּמְסִירוּת נַפְשָׁם

בַּדּוֹר הַהוּא חָיוּ שְׁנֵי צַדִּיקִים עֲנָקִים, חֲכָמִים גְּאוֹנִים שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא שָׁלַח בְּהַשְׁגָּחָה פְּרָטִית מְדֻיֶּקֶת כְּדֵי לְהַצִּיל אֶת הַתּוֹרָה. לָאֶחָד קָרְאוּ רַבִּי חִיָּיא , וְלַשֵּׁנִי קָרְאוּ רַבִּי אוֹשַׁעְיָה הַגָּדוֹל (בְּנוֹ שֶׁל רַבִּי חָמָא).

רַבִּי אוֹשַׁעְיָה הָיָה אָדָם שֶׁתּוֹרָתוֹ בָּעֲרָה בּוֹ כְּמוֹ אֵשׁ תָּמִיד. הוּא וְרַבִּי חִיָּיא יָדְעוּ בְּעַל-פֶּה אֶת כָּל מָה שֶׁנֶּאֱמַר לְמֹשֶׁה בְּסִינַי. הֵם הֶחְלִיטוּ בְּנַפְשָׁם: "אֲנַחְנוּ לֹא נִתֵּן שֶׁאַף הֲלָכָה תִּשְׁתַּבֵּשׁ! אֲנַחְנוּ נַאַסְפָה אֶת כָּל הַבָּרַיְיתוֹת, נִבְדֹּק אוֹתָן מִלָּה אַחַר מִלָּה, אוֹת אַחַר אוֹת, עַד שֶׁהֵן יִהְיוּ נְקִיּוֹת מִכָּל טָעוּת!".

וְכָךְ יְלָדִים, שָׁנִים עַל גַּבֵּי שָׁנִים, הֵם מָסְרוּ אֶת נַפְשָׁם.

דַּמְיְנוּ אֶת הַלַּיְלָה יּוֹרֵד עַל יְרוּשָׁלַיִם וְטְבֶרְיָה. אֲנָשִׁים כְּבָר הוֹלְכִים לִישֹׁן, מִתְכַּסִּים בִּשְׂמִיכוֹת חַמּוֹת כִּי בַּחוּץ רוּחַ קָרָה שׁוֹרֶקֶת וְטִפּוֹת שֶׁל גֶּשֶׁם דּוֹפְקוֹת עַל הַחַלּוֹן.

אֲבָל בְּבֵית הַמִּדְרָשׁ, דּוֹלֵק כְּלִי קָטָן שֶׁל חֶרֶס עִם שֶׁמֶן זַיִת וּפְתִילָה. רֵיחַ הַשֶּׁמֶן הַשָּׂרוּף נִמְהָל בְּרֵיחַ קְלָפִים יְשָׁנִים מֵעוֹר שֶׁל בְּהֵמוֹת, רֵיחַ שֶׁל קְדֻשָּׁה וְתּוֹרָה. רַבִּי חִיָּיא וְרַבִּי אוֹשַׁעְיָה רוֹכְנִים עַל הַסְּפָרִים. הַעֵינַיִם שֶׁלָּהֶם אֲדֻמּוֹת מֵעֲיֵפוּת, אֲבָל הַפָּנִים שֶׁלָּהֶם מְאִירוֹת בְּאוֹר שֶׁל מַלְאָכִים!

"הַאִם הָרַב הַזֶּה בֶּאֱמֶת שָׁמַע אֶת הַהֲלָכָה מֵרַבִּי עֲקִיבָא?" שׁוֹאֵל רַבִּי חִיָּיא.

"לֹא," עָנָה רַבִּי אוֹשַׁעְיָה, "בָּדַקְתִּי וּמָצָאתִי שֶׁתַּלְמִידוֹ שֶׁל רַבִּי עֲקִיבָא אָמַר אַחֶרֶת! יֵשׁ טָעוּת בַּמִּלָּה הַזּוֹ! חַיָּבִים לְתַקֵּן!".

קְרִיק... קְרִיק... קְרִיק... נִשְׁמַע קוֹל עֵט הַנּוֹצָה וְהַדְּיוֹ הַשָּׁחוֹר, כַּאֲשֶׁר רַבִּי אוֹשַׁעְיָה כָּתַב אֶת הַהֲלָכָה הַמְדֻיֶּקֶת, בְּלִי לְזַיֵּף בְּקוֹצוֹ שֶׁל יוֹד!

כְּמוֹ צוֹרֵף זָהָב שֶׁמְּנַקֶּה אֶת הַתַּכְשִׁיטִים, כָּךְ נִקּוּ רַבִּי חִיָּיא וְרַבִּי אוֹשַׁעְיָה מֵאוֹת וְאַלְפֵי הֲלָכוֹת! הֵם סִדְּרוּ אוֹתָן שׁוּרָה אַחַר שׁוּרָה, עַד שֶׁעָמַד לִפְנֵיהֶם אֹסֶף קָדוֹשׁ וְטָהוֹר, נָקִי מִכָּל פְּגָם. מַמָּשׁ תּוֹרָה מִסִּינַי!

מִבְחַן הָאֱמֶת וְנִצְּחוֹן הַתּוֹרָה

עָבְרוּ שָׁנִים, וְרַבִּי חִיָּיא וְרַבִּי אוֹשַׁעְיָה כְּבָר נִפְטְרוּ מִן הָעוֹלָם וְעָלוּ לִישִׁיבָה שֶׁל מַעְלָה.

בְּבֵית הַמִּדְרָשׁ יָשְׁבוּ חֲכָמִים חֲדָשִׁים (הָאָמוֹרָאִים). יוֹם אֶחָד, הָיָה וִכּוּחַ סוֹעֵר עַל הֲלָכָה שֶׁל כַּשְׁרוּת. אַחַד הַתַּלְמִידִים קָם וְאָמַר: "אֲנִי יוֹדֵעַ! מָצָאתִי בָּרַיְיתָא יְשָׁנָה שֶׁנִּכְתְּבָה עַל יְדֵי חָכָם לִפְנֵי שָׁנִים מְרוּבּוֹת. הִנֵּה, כָּתוּב שֶׁהַדָּבָר הַזֶּה מֻתָּר לַאֲכִילָה!".

רֶגַע שֶׁל שֶׁקֶט עָמַד בַּחֶדֶר. כֻּלָּם כִּמְעַט סָמְכוּ עַל הַהֲלָכָה הַזּוֹ וְרָצוּ לְהַתִּיר אֶת הַמַּאֲכָל.

אֲבָל אָז קָם קוֹל חָזָק וְצָלוּל מִסּוֹף הַשּׁוּרָה. זֶה הָיָה רַבִּי זֵירָא! אֶחָד מִגְּדוֹלֵי אֶרֶץ יִשְׂרָאֵל. הוּא דָּפַק עַל הַשֻּׁלְחָן, סִדֵּר אֶת גְּלִימָתוֹ, וְאָמַר בְּטּוֹן מְלֵא יִרְאַת כָּבוֹד:

"עִצְרוּ מִיָּד! כָּל מִשְׁנָה אוֹ בָּרַיְיתָא שֶׁלֹּא נִשְׁנְתָה בְּבֵית מִדְרָשָׁם שֶׁל רַבִּי חִיָּיא וְרַבִּי אוֹשַׁעְיָה – מְשֻׁבֶּשֶׁת הִיא! וְלֹא תַּקְשׁוּ מִמֶּנָּה בְּבֵית הַמִּדְרָשׁ!"

יְלָדִים, אֵיזֶה רֶגַע אַדִּיר זֶה הָיָה! רַבִּי זֵירָא קָבַע כְּלַל בַּרְזֶל שֶׁנִּשְׁאַר עַד יָמֵינוּ. בִּזְכוּת הָעֲמָל הָעָצוּם שֶׁל הַלַּיְלוֹת, הַדִּיּוּק הַמֻּחְלָט וִירְאַת הַשָּׁמַיִם, רַק הַבָּרַיְיתוֹת שֶׁל רַבִּי חִיָּיא וְרַבִּי אוֹשַׁעְיָה הַגָּדוֹל נֶחְשְׁבוּ לָאֱמֶת שֶׁאֵין בָּהּ סָפֵק! כָּל שְׁאָר הַמְּקוֹרוֹת הָיוּ מְשֻׁבָּשִׁים אוֹ חֲסֵרִים, אֲבָל מָה שֶׁעָבַר תַּחַת יְדֵיהֶם שֶׁל שְׁנֵי הַצַּדִּיקִים הָאֵלֶּה – זָרַח כְּיַהֲלוֹם שֶׁל אֱמֶת לָנֶצַח נְצָחִים.

בִּזְכוּתָם, אֲנַחְנוּ יוֹדְעִים הַיּוֹם בְּדִיּוּק מָה מֻתָּר וּמָה אָסוּר, וּמַמְשִׁיכִים לִלְמוֹד תּוֹרָה שֶׁלֹּא נִשְׁתַּכְּחָה מִיִּשְׂרָאֵל!

מקורות: תלמוד בבלי, מסכת חולין, דף קמ"א עמוד א': "אָמַר רַבִּי זֵירָא: כָּל מִשְׁנָה [ברייתא] שֶׁלֹּא נִשְׁנֵית בְּבֵית מִדְרָשָׁם שֶׁל רַבִּי חִיָּיא וְרַבִּי אוֹשַׁעְיָה - מְשֻׁבֶּשֶׁת הִיא, וְלֹא תַּקְשׁוּ מִמֶּנָּה בְּבֵית הַמִּדְרָשׁ" (וכן בגרסה דומה בירושלמי). תלמוד ירושלמי, מסכת הוריות, פרק ג', הלכה ה': (על גדלותם, סמכותם וייחודיותם של הברייתות של רבי חייא ורבי אושעיא לעומת שאר הברייתות). הערה להקשר: הבאת הדוגמה של דיוק בלימוד כסינון יהלומים (מתכתבת עם חז"ל שמשלו את התורה לפנינים ויהלומים, ומסדר המשניות והברייתות נקרא כמי שמלקט פנינים ומרגליות, למשל בתיאור על מידות חכמים כ"שולחני מלקט"). הגרעין המרכזי של אמינות הברייתות וגבורת השימור שלהן הובא כאן ע"פ הוראות הספרות מהחומר.

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “יְלָדִים יְקָרִים, דַּמְיְנוּ לָעַצְמְכֶם חֶדֶר עֲנָק, מָלֵא בְּהַרְבֵּה מְאֹד יַהֲלוֹמִים נוֹצְצִים.” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “בְּבָתֵּי הַמִּדְרָשׁ בְּאֶרֶץ יִשְׂרָאֵל יָשְׁבוּ אֲלָפִים שֶׁל תַּלְמִידֵי חֲכָמִים.”?
ג. כיצד הגיב ר' אושעיא הגדול בן ר' חמא למצב המתואר: “לָאֶחָד קָרְאוּ רַבִּי חִיָּיא , וְלַשֵּׁנִי קָרְאוּ רַבִּי אוֹשַׁעְיָה הַגָּדוֹל (בְּנוֹ שֶׁל רַבִּי חָמָא).”?
ד. מה השתנה בין “אֲבָל בְּבֵית הַמִּדְרָשׁ, דּוֹלֵק כְּלִי קָטָן שֶׁל חֶרֶס עִם שֶׁמֶן זַיִת וּפְתִילָה.” לבין “הֵם סִדְּרוּ אוֹתָן שׁוּרָה אַחַר שׁוּרָה, עַד שֶׁעָמַד לִפְנֵיהֶם אֹסֶף קָדוֹשׁ וְטָהוֹר, נָקִי מִכָּל פְּגָם.”?
ה. כיצד הגיע הסיפור אל האירוע: “רַבִּי זֵירָא קָבַע כְּלַל בַּרְזֶל שֶׁנִּשְׁאַר עַד יָמֵינוּ.”?

3. מה היה ראשון?

כתבו ליד כל אירוע מספר מ־1 עד 5 לפי סדר התרחשותו בסיפור.

□ וְכָךְ יְלָדִים, שָׁנִים עַל גַּבֵּי שָׁנִים, הֵם מָסְרוּ אֶת נַפְשָׁם.
□ יְלָדִים יְקָרִים, דַּמְיְנוּ לָעַצְמְכֶם חֶדֶר עֲנָק, מָלֵא בְּהַרְבֵּה מְאֹד יַהֲלוֹמִים נוֹצְצִים.
□ בִּזְכוּתָם, אֲנַחְנוּ יוֹדְעִים הַיּוֹם בְּדִיּוּק מָה מֻתָּר וּמָה אָסוּר, וּמַמְשִׁיכִים לִלְמוֹד תּוֹרָה שֶׁלֹּא נִשְׁתַּכְּחָה מִיִּשְׂרָאֵל!
□ אִם מִישֶׁהוּ מִתְבַּלְבֵּל בְּמִלָּה אַחַת, בִּמְקוֹם "מֻתָּר" הוּא יַגִּיד "אָסוּר".
□ עָבְרוּ שָׁנִים, וְרַבִּי חִיָּיא וְרַבִּי אוֹשַׁעְיָה כְּבָר נִפְטְרוּ מִן הָעוֹלָם וְעָלוּ לִישִׁיבָה שֶׁל מַעְלָה.

לפני ואחרי

מה קרה לפני: “וְכָךְ יְלָדִים, שָׁנִים עַל גַּבֵּי שָׁנִים, הֵם מָסְרוּ אֶת נַפְשָׁם.”?
מה קרה אחרי: “עָבְרוּ שָׁנִים, וְרַבִּי חִיָּיא וְרַבִּי אוֹשַׁעְיָה כְּבָר נִפְטְרוּ מִן הָעוֹלָם וְעָלוּ לִישִׁיבָה שֶׁל מַעְלָה.”?

סיבה ותוצאה מתוך המעשה

האירוע הנתוןמה גרם לו?מה הייתה תוצאתו?
אֲנָשִׁים קָמוּ בְּבֵית הַמִּדְרָשׁ וְאָמְרוּ: "שָׁמַעְתִּי בָּרַיְיתָא הֲלָכָה שֶׁאוֹמֶרֶת כָּךְ!", אֲבָל חֲבֵרוֹ צָעַק לְעֻמָּתוֹ: "מָה פִּתְאוֹם!
לֹא," עָנָה רַבִּי אוֹשַׁעְיָה, "בָּדַקְתִּי וּמָצָאתִי שֶׁתַּלְמִידוֹ שֶׁל רַבִּי עֲקִיבָא אָמַר אַחֶרֶת!

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “סידור בברייתות ביחד עם ר' חייא” על ר' אושעיא הגדול בן ר' חמא? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “הֵם הֶחְלִיטוּ בְּנַפְשָׁם: "אֲנַחְנוּ לֹא נִתֵּן שֶׁאַף הֲלָכָה תִּשְׁתַּבֵּשׁ!”.
ג. אילו ר' אושעיא הגדול בן ר' חמא לא היה פועל כפי שמתואר ב־“הֵם סִדְּרוּ אוֹתָן שׁוּרָה אַחַר שׁוּרָה, עַד שֶׁעָמַד לִפְנֵיהֶם אֹסֶף קָדוֹשׁ וְטָהוֹר, נָקִי מִכָּל פְּגָם.” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “רַבִּי זֵירָא קָבַע כְּלַל בַּרְזֶל שֶׁנִּשְׁאַר עַד יָמֵינוּ.” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. יְלָדִים יְקָרִים, דַּמְיְנוּ לָעַצְמְכֶם חֶדֶר עֲנָק, מָלֵא…
2. בְּבָתֵּי הַמִּדְרָשׁ בְּאֶרֶץ יִשְׂרָאֵל יָשְׁבוּ אֲלָפִים שֶׁל…
3. רַבִּי אוֹשַׁעְיָה הָיָה אָדָם שֶׁתּוֹרָתוֹ בָּעֲרָה בּוֹ כְּמוֹ אֵשׁ…
4. הַעֵינַיִם שֶׁלָּהֶם אֲדֻמּוֹת מֵעֲיֵפוּת, אֲבָל הַפָּנִים שֶׁלָּהֶם…
5. מָצָאתִי בָּרַיְיתָא יְשָׁנָה שֶׁנִּכְתְּבָה עַל יְדֵי חָכָם לִפְנֵי…
6. בִּזְכוּתָם, אֲנַחְנוּ יוֹדְעִים הַיּוֹם בְּדִיּוּק מָה מֻתָּר וּמָה…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “לָאֶחָד קָרְאוּ רַבִּי חִיָּיא , וְלַשֵּׁנִי קָרְאוּ רַבִּי אוֹשַׁעְיָה הַגָּדוֹל (בְּנוֹ שֶׁל רַבִּי חָמָא).”.
מכתב: כתבו לר' אושעיא הגדול בן ר' חמא בעקבות “לֹא," עָנָה רַבִּי אוֹשַׁעְיָה, "בָּדַקְתִּי וּמָצָאתִי שֶׁתַּלְמִידוֹ שֶׁל רַבִּי עֲקִיבָא אָמַר אַחֶרֶת!”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “מָצָאתִי בָּרַיְיתָא יְשָׁנָה שֶׁנִּכְתְּבָה עַל יְדֵי חָכָם לִפְנֵי שָׁנִים מְרוּבּוֹת.”.
דיווח: כתבו ידיעה קצרה שכותרתה “סידור בברייתות ביחד עם ר' חייא”.
