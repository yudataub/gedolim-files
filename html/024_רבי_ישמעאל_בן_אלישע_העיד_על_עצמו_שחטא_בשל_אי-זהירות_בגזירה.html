העיד על עצמו שחטא בשל אי-זהירות בגזירה

דף עבודה בעקבות רבי ישמעאל בן אלישע
שם: כיתה:

המעשה — הסיפור המלא

פְּתִיחָה: רֵיחַ שֶׁמֶן הַזַּיִת וְאוֹרָהּ שֶׁל שַׁבָּת

יְלָדִים, נַסּוּ לְדַמְיֵן שֶׁאַתֶּם יוֹשְׁבִים בְּחֶדֶר חָשׁוּךְ. בַּחוּץ נוֹשֶׁבֶת רוּחַ קָרָה שֶׁל לֵיל חֹרֶף. אֵין חַשְׁמַל, אֵין מְנוֹרוֹת פְלוֹרֵסֶנְט, אֵין פָּנָסִים. רַק דָּבָר אֶחָד מֵאִיר אֶת הַחֶדֶר: עַל הַשֻּׁלְחָן מֻנַּחַת מְנוֹרַת חֶרֶס קְטַנָּה. בְּתוֹךְ הַחֶרֶס יֵשׁ שֶׁמֶן זַיִת זַךְ הַמֵּפִיץ רֵיחַ נָעִים שֶׁל בֹּקֶר בַּכֶּרֶם, וּבְקָצֵהוּ מְבַצְבֶּצֶת פְּתִילָה שֶׁעָלֶיהָ מְרַקֶּדֶת שַׁלְהֶבֶת קְטַנָּה וּצְהֻבָּה שֶׁל אֵשׁ.

הַשֶּׁקֶט עָמֹק. אֶפְשָׁר מַמָּשׁ לִשְׁמֹעַ אֶת קוֹל הַפְּצִפּוּץ הַחֲרִישִׁי שֶׁל הָאֵשׁ אוֹכֶלֶת אֶת הַשֶּׁמֶן.

בַּלַּיְלָה הַקָּדוֹשׁ הַזֶּה, לֵיל שַׁבָּת קֹדֶשׁ, יָשַׁב אֶחָד מִגְּדוֹלֵי עַם יִשְׂרָאֵל. קָרְאוּ לוֹ הַתַּנָּא רַבִּי יִשְׁמָעֵאל בֶּן אֱלִישָׁע .

רַבִּי יִשְׁמָעֵאל אָהַב אֶת הַתּוֹרָה יוֹתֵר מִכָּל דָּבָר בָּעוֹלָם. עוֹד כְּשֶׁהָיָה יֶלֶד קָטָן, כְּשֶׁנִּלְקַח לַשֶּׁבִי הָאַכְזָרִי, הַתּוֹרָה הִיא שֶׁהֶחְזִיקָה אוֹתוֹ. הַמְּתִיקוּת שֶׁל דִּבְרֵי הַתּוֹרָה הָיְתָה בִּשְׁבִילוֹ כְּמוֹ מַיִם קָרִים וּמְשִׁיבֵי נֶפֶשׁ לְאָדָם צָמֵא בַּמִּדְבָּר. גַּם עַכְשָׁו, כְּשֶׁהוּא תַּלְמִיד חָכָם עָצוּם, נְשִׂיא אֱלֹקִים, כְּשֶׁפָּתַח אֶת סִפְרֵי הַקֹּדֶשׁ – הוּא שָׁכַח אֶת כָּל הָעוֹלָם שֶׁבַּחוּץ.

הַגְּזֵרָה שֶׁל חֲכָמִים וְהַבִּטָּחוֹן הַגָּדוֹל

יְלָדִים, חֲכָמֵינוּ זִכְרוֹנָם לִבְרָכָה, שֶׁיָּדְעוּ לִשְׁמֹר אֶת עַם יִשְׂרָאֵל שֶׁלֹּא יִכָּשְׁלוּ בַּעֲבֵרוֹת, גָּזְרוּ גְּזֵרָה שֶׁקְּשׁוּרָה בְּדִיּוּק לַמְּנוֹרָה הַזּוֹ.

הֵם אָמְרוּ: "לֹא יִקְרָא אָדָם בְּלֵיל שַׁבָּת לְאוֹר הַנֵּר!". לָמָּה? מִפְּנֵי שֶׁכַּאֲשֶׁר אָדָם קוֹרֵא בְּסֵפֶר, וְהַשֶּׁמֶן שֶׁבַּמְּנוֹרָה נִגְמָר וְהוֹלֵךְ, הָאוֹר הוֹפֵךְ לִהְיוֹת חַלָּשׁ וְעָמוּם מְאֹד. מָה אָדָם רָגִיל עוֹשֶׂה מִיָּד בְּיוֹם חוֹל? הוּא בְּלִי לַחְשֹׁב שׁוֹלֵחַ אֶת הַיָּד, נוֹגֵעַ בִּמְנוֹרַת הַחֶרֶס, וּמַטֶּה אוֹתָהּ קָדִימָה! כָּךְ הַשֶּׁמֶן שֶׁנִּשְׁאַר בַּצַּד, נוֹזֵל יָשָׁר לְתוֹךְ הַפְּתִילָה, וְהָאֵשׁ שׁוּב נִדְלֶקֶת וּמְאִירָה כְּלַפִּי מַעְלָה.

אֲבָל... בְּשַׁבָּת, מִי שֶׁעוֹשֶׂה אֶת זֶה, מִי שֶׁ"מַּטֶּה אֶת הַנֵּר", עוֹבֵר עַל עֲבֵרָה נוֹרָאָה מִן הַתּוֹרָה שֶׁל הַבְעָרַת אֵשׁ בְּשַׁבָּת! (מְלֶאכֶת מַבְעִיר).

לָכֵן, כְּדֵי שֶׁאַף יְהוּדִי לֹא יִטְעֶה – אָסְרוּ לִקְרֹא לְאוֹר הַנֵּר אֲפִלּוּ אִם אַתָּה מַבְטִיחַ שֶׁתִּזָּהֵר.

אֲבָל רַבִּי יִשְׁמָעֵאל בֶּן אֱלִישָׁע הָיָה אָדָם קָדוֹשׁ. הָרֹאשׁ שֶׁלּוֹ לֹא הָיָה תָּפוּס בִּשְׁטֻיּוֹת. הַקֶּשֶׁר שֶׁלּוֹ לַקָּדוֹשׁ בָּרוּךְ הוּא הָיָה כָּל כָּךְ חָזָק!

אָמַר רַבִּי יִשְׁמָעֵאל בְּלִבּוֹ: "הַתַּקָּנָה הַזּוֹ הִיא לַאֲנָשִׁים רְגִילִים שֶׁהַדַּעַת שֶׁלָּהֶם מֻסַּחַת. אֲבָל אֲנִי? אֲנִי זָהִיר! אֲנִי זָכוּר כָּל רֶגַע שֶׁשַּׁבָּת הַיּוֹם! וְהַתּוֹרָה כָּל כָּךְ חֲשׁוּבָה לִי!".

וְכָךְ קָבַע הַצַּדִּיק: "אֲנִי אֶקְרָא – וְלֹא אַטֶּה" . "אֲנִי אֶקְרָא בַּסֵּפֶר, אֲבָל אַף פַּעַם לֹא אֶשְׁכַּח וְלֹא אַטֶּה אֶת הַמְּנוֹרָה, הַגְּזֵרָה לֹא תִּקְרֶה לִי".

הָאוֹר נֶחֱלָשׁ – הַלֵּב מִשְׁתּוֹקֵק

לֵיל שַׁבָּת יָרַד עַל אֶרֶץ יִשְׂרָאֵל. כֻּלָּם כְּבָר יְשֵׁנִים. הַחֶדֶר שֶׁל רַבִּי יִשְׁמָעֵאל מָלֵא בְּצֵלְלִיּוֹת (צְלָלִים שֶׁל הָאֵשׁ) עַל הַקִּירוֹת.

רַבִּי יִשְׁמָעֵאל פּוֹתֵחַ סֵפֶר שֶׁל דִּבְרֵי קֹדֶשׁ. הוּא מִסְתַּכֵּל בָּאוֹתִיּוֹת וְשׁוֹקֵעַ בְּתוֹךְ סוֹדוֹת הַתּוֹרָה. רֶגַע אֶחָד הוּא לוֹמֵד כֵּיצַד אָבוֹתֵינוּ שָׁמְרוּ עַל הַמִּצְווֹת, וְרֶגַע שֵׁנִי הוּא לוֹמֵד אֶת הֲלָכוֹת קָרְבְּנוֹת בֵּית הַמִּקְדָּשׁ.

הַזְּמַן עָבַר. וְעָבַר עוֹד.

וּבְלִי שֶׁרַבִּי יִשְׁמָעֵאל יָשִׂים לֵב, מִפְּנֵי שֶׁהָיָה שָׁקוּעַ כֻּלּוֹ בְּתוֹךְ תּוֹרַת יִשְׂרָאֵל... הַשֶּׁמֶן מַתְחִיל לְהִגָּמֵר.

פְסְס... טְסְסְ... נִשְׁמַע קוֹל חַלָּשׁ מִן הַפְּתִילָה שֶׁמִּתְחִילָה לְהִתְיַבֵּשׁ מִשֶּׁמֶן.

הַשַּׁלְהֶבֶת שֶׁקֹּדֶם הֵאִירָה הֵיטֵב, מִתְכַּוֶּצֶת וְיוֹרֶדֶת. הָאוֹתִיּוֹת בַּסֵּפֶר שֶׁמּוּל רַבִּי יִשְׁמָעֵאל נִהְיוֹת חֲשׁוּכוֹת. קָשֶׁה לוֹ לִקְרֹא. קָשֶׁה לִרְאוֹת מָה הַמִּלָּה הַבָּאָה!

בּוֹאוּ נַעֲצֹר רֶגַע, יְלָדִים. מָה קוֹרֶה לְאָדָם שֶׁאוֹהֵב לַעֲשׂוֹת מַשֶּׁהוּ וּפִתְאוֹם זֶה נֶעֱצָר לוֹ מִבְּלִי לָשִׂים לֵב? הַיָּד שֶׁלּוֹ זָזָה מֵעַצְמָהּ כְּדֵי לְתַקֵּן אֶת הַמַּצָּב!

מִתּוֹךְ כָּךְ שֶׁרַבִּי יִשְׁמָעֵאל מִתְאַמֵּץ לִקְרֹא עוֹד אוֹת וְעוֹד אוֹת, הוּא מְכַוֵּץ אֶת עֵינָיו... הוּא רוֹכֵן (מִתְכּוֹפֵף) אֶל הַסֵּפֶר... הַחֹשֶׁךְ מַפְרִיעַ לוֹ! הָרֹאשׁ שֶׁלּוֹ בַּשָּׁמַיִם, הוּא כֻּלּוֹ בְּלִמּוּד...

וְאָז, הַיָּד שֶׁלּוֹ נִשְׁלַחַת קָדִימָה. בְּלִי לַחְשֹׁב. עֲשִׂיָּה מֵרֹב צִמָּאוֹן לַתּוֹרָה.

אֶצְבָּעוֹת שֶׁל נְשִׂיא אֱלֹקִים נוֹגְעוֹת בַּחֶרֶס הַקָּר, וְ...

מַטּוֹת אֶת הַנֵּר.

הַהֶלֶם וְהַזְּעָקָה בַּחֹשֶׁךְ

אוֹר גָּדוֹל קָפַץ פִּתְאוֹם בַּחֶדֶר. הַשֶּׁמֶן הִגִּיעַ לַפְּתִילָה, מִלֵּא אוֹתָהּ בְּכֹחַ, וְהַשַּׁלְהֶבֶת זָרְחָה עַד הַתִּקְרָה! חָם הָיָה בַּחֶדֶר פִּתְאוֹם.

בְּאוֹתוֹ הָרֶגַע שֶׁהָאוֹר נִדְלַק – נִדְלַק אוֹר גַּם בְּתוֹךְ רֹאשׁוֹ שֶׁל רַבִּי יִשְׁמָעֵאל.

הוּא בָּלַע אֶת רֻקּוֹ. הוּא הִבִּיט עַל יָדוֹ, שֶׁעוֹד לָפְתָה אֶת צַד הַמְּנוֹרָה. הוּא נָשַׁם נְשִׁימָה עֲמֻקָּה וְחֶרְדָּה כִּסְּתָה אֶת פָּנָיו הַטְּהוֹרוֹת.

"מָה עָשִׂיתִי? הֵטֵיתִי אֶת הַנֵּר בְּשַׁבָּת הַקְּדוֹשָׁה?!"

אָדָם אַחֵר יָכוֹל הָיָה לוֹמַר עַכְשָׁו תֵּרוּצִים בְּעַל פֶּה: "זֶה בִּשְׁגָגָה! לֹא שַׂמְתִּי לֵב! עָשִׂיתִי אֶת זֶה מִתּוֹךְ כָּךְ שֶׁלָּמַדְתִּי תּוֹרָה, זֶה בֶּטַח לֹא נוֹרָא!".

אֲבָל רַבִּי יִשְׁמָעֵאל הָיָה צַדִּיק אֱמֶת. וְצַדִּיק אֲמִתִּי לֹא מְחַפֵּשׂ תֵּרוּצִים בְּשֶׁקֶר.

הוּא תָּפַס בִּשְׁתֵּי יָדָיו אֶת רֹאשׁוֹ, שִׂפְתוֹתָיו רָעֲדוּ, וּדְמָעוֹת שֶׁל אֶמֶת עָמְדוּ בְּעֵינָיו. הוּא קָרָא מֵעֻמְקָא דְּלִבָּא בְּקוֹל רוֹעֵד:

"כַּמָּה גְּדוֹלִים דִּבְרֵי חֲכָמִים שֶׁהָיוּ אוֹמְרִים: 'לֹא יִקְרָא לְאוֹר הַנֵּר'!" .

הוּא הֵבִין שֶׁהַחֲכָמִים יָדְעוּ בְּדִיּוּק מָה הֵם אוֹמְרִים. הַחֲכָמִים מַכִּירִים אֶת בְּנֵי הָאָדָם, אֲפִלּוּ אֶת הַגְּדוֹלִים שֶׁבָּהֶם. כְּשֶׁאָדָם בָּטוּחַ בְּעַצְמוֹ מִדַּי – בְּדִיּוּק שָׁם הוּא יָכוֹל לִפֹּל.

הַהוֹדָאָה שֶׁבַּפִּנְקָס

מוֹצָאֵי שַׁבָּת הִגִּיעַ. הַשָּׁמַיִם הֶחְשִׁיכוּ בַּחוּץ וּשְׁלוֹשָׁה כּוֹכָבִים הֵצִיצוּ לְתוֹךְ הַחֶדֶר. רַבִּי יִשְׁמָעֵאל עָשָׂה הַבְדָּלָה, הוֹצִיא אֶת שַׁבָּת הַמַּלְכָּה.

מִיָּד לְאַחַר מִכֵּן, בְּלִי לְבַזְבֵּז רֶגַע, הוּא נִגַּשׁ לַאֲרוֹנוֹ. הוּא הוֹצִיא מִשָּׁם פִּנְקָס קָטָן שֶׁל קְלָף אֲשֶׁר עָלָיו הָיָה כּוֹתֵב דְּבָרִים שֶׁאָסוּר לְשַׁכֵּחַ. הוּא טָבַל אֶת נוֹצַת הַקֻּלְמוֹס הַשְּׁחוֹרָה בְּתוֹךְ קֶסֶת הַדְּיוֹ הָרֵיחָנִית.

אִם מִישֶׁהוּ קוֹרֵא פִּנְקָס שֶׁל רַב דָּגוּל – הוּא בֶּטַח מְצַפֶּה לִרְאוֹת שָׁם כָּתוּב חִדּוּשֵׁי תּוֹרָה עֲמֻקִּים, אוֹ סוֹדוֹת קַבָּלָה עֶלְיוֹנִים.

אֲבָל רַבִּי יִשְׁמָעֵאל, הָעֲנָק בַּעֲנָקִים, כָּתַב אוֹתִיּוֹת בְּרוּרוֹת שֶׁל עֲנָוָה, כְּדֵי שֶׁכָּל דּוֹרוֹת יִשְׂרָאֵל יִרְאוּ לָנֶצַח. קוֹל חֲרִיקַת הַנּוֹצָה בַּדַּף עָשָׂה קְרַאצְ'... קְרַאצְ'...

וְכָךְ הוּא כָּתַב בְּלֹא מָסָךְ מַסְתִּיר:

"אֲנִי, יִשְׁמָעֵאל בֶּן אֱלִישָׁע, קָרִיתִי וְהִטֵּיתִי אֶת הַנֵּר בְּשַׁבָּת! וְלִכְשֶׁיִּבָּנֶה בֵּית הַמִּקְדָּשׁ, אָבִיא חַטָּאת שְׁמֵנָה!"

הוּא קִבֵּל עַל עַצְמוֹ! הוּא לֹא הִסְתִּיר. הוּא הִכְרִיז לִפְנֵי עַצְמוֹ וְלִפְנֵי הַקָּדוֹשׁ בָּרוּךְ הוּא: אֲנִי טָעִיתִי, נִכְשַׁלְתִּי בַּגְּזֵרָה שֶׁל חֲכָמִים כִּי חָשַׁבְתִּי שֶׁאֲנִי יוֹתֵר חָזָק מֵהַטֶּבַע שֶׁל הָאָדָם. אֲנִי אֶקַּח אַחְרָיוּת אֲמִתִּית! בִּשְׁבִיל לְכַפֵּר עַל הַטָּעוּת הַזּוֹ, כְּשֶׁיִּבָּנֶה בֵּית הַמִּקְדָּשׁ תִּכֵּף וּמִיָּד (הֲרֵי זֶה הָיָה מְעַט אַחֲרֵי שֶׁנֶּחֱרַב), אֲנִי בְּעַצְמִי אֵלֵךְ, אֶקְנֶה בְּכַסְפִּי קָרְבַּן חַטָּאת גָּדוֹל וְשָׁמֵן, וְאַקְרִיב אוֹתוֹ בְּבֵית הַמִּקְדָּשׁ!

כָּךְ לִמֵּד אוֹתָנוּ הַתַּנָּא הַקָּדוֹשׁ – לֹא לְזַלְזֵל אַף פַּעַם בְּשׁוּם הוֹרָאָה וּגְזֵרָה, אֲפִלּוּ אִם בָּטוּחַ לָנוּ "שֶׁלָּנוּ זֶה מַמָּשׁ לֹא יִקְרֶה!".

מקורות: תלמוד בבלי, מסכת שבת, דף י"ב עמוד ב': "לֹא יִקְרָא לְאוֹר הַנֵּר... אָמַר רַבִּי יִשְׁמָעֵאל בֶּן אֱלִישָׁע: אֲנִי אֶקְרָא וְלֹא אַטֶּה. פַּעַם אַחַת קָרָא וּבִקֵּשׁ לְהַטּוֹת. אָמַר: כַּמָּה גְּדוֹלִים דִּבְרֵי חֲכָמִים שֶׁהָיוּ אוֹמְרִים 'לֹא יִקְרָא לְאוֹר הַנֵּר'. רַבִּי נָתָן אוֹמֵר: קָרָא וְהִטָּה, וְכָתַב עַל פִּנְקָסוֹ: 'אֲנִי יִשְׁמָעֵאל בֶּן אֱלִישָׁע קָרִיתִי וְהִטֵּיתִי נֵר בְּשַׁבָּת, לִכְשֶׁיִּבָּנֶה בֵּית הַמִּקְדָּשׁ אָבִיא חַטָּאת שְׁמֵנָה'". הערה למורה: הסיפור אומץ לפי גרסתו המחמירה והנוקבת יותר של רבי נתן, שבה אכן הטה בפועל וכתב בפנקסו – מה שמדגיש ביתר שאת את קבלת האחריות.

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “רַק דָּבָר אֶחָד מֵאִיר אֶת הַחֶדֶר: עַל הַשֻּׁלְחָן מֻנַּחַת מְנוֹרַת חֶרֶס קְטַנָּה.” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “יְלָדִים, חֲכָמֵינוּ זִכְרוֹנָם לִבְרָכָה, שֶׁיָּדְעוּ לִשְׁמֹר אֶת עַם יִשְׂרָאֵל שֶׁלֹּא יִכָּשְׁלוּ בַּעֲבֵרוֹת, גָּזְרוּ גְּזֵרָה שֶׁקְּשׁוּרָה…”?
ג. כיצד הגיב רבי ישמעאל בן אלישע למצב המתואר: “הַקֶּשֶׁר שֶׁלּוֹ לַקָּדוֹשׁ בָּרוּךְ הוּא הָיָה כָּל כָּךְ חָזָק!”?
ד. מה השתנה בין “מִתּוֹךְ כָּךְ שֶׁרַבִּי יִשְׁמָעֵאל מִתְאַמֵּץ לִקְרֹא עוֹד אוֹת וְעוֹד אוֹת, הוּא מְכַוֵּץ אֶת עֵינָיו...” לבין “כַּמָּה גְּדוֹלִים דִּבְרֵי חֲכָמִים שֶׁהָיוּ אוֹמְרִים: 'לֹא יִקְרָא לְאוֹר הַנֵּר'!" .”?
ה. כיצד הגיע הסיפור אל האירוע: “אִם מִישֶׁהוּ קוֹרֵא פִּנְקָס שֶׁל רַב דָּגוּל – הוּא בֶּטַח מְצַפֶּה לִרְאוֹת שָׁם כָּתוּב חִדּוּשֵׁי תּוֹרָה עֲמֻקִּים, אוֹ סוֹדוֹת קַבָּלָה עֶלְיוֹנִים.”?

3. מה היה ראשון?

□ נִשְׁמַע קוֹל חַלָּשׁ מִן הַפְּתִילָה שֶׁמִּתְחִילָה לְהִתְיַבֵּשׁ מִשֶּׁמֶן.
□ רַק דָּבָר אֶחָד מֵאִיר אֶת הַחֶדֶר: עַל הַשֻּׁלְחָן מֻנַּחַת מְנוֹרַת חֶרֶס קְטַנָּה.

□ הוּא בְּלִי לַחְשֹׁב שׁוֹלֵחַ אֶת הַיָּד, נוֹגֵעַ בִּמְנוֹרַת הַחֶרֶס, וּמַטֶּה אוֹתָהּ קָדִימָה!
□ הוּא הֵבִין שֶׁהַחֲכָמִים יָדְעוּ בְּדִיּוּק מָה הֵם אוֹמְרִים.

לפני ואחרי

מה קרה לפני: “נִשְׁמַע קוֹל חַלָּשׁ מִן הַפְּתִילָה שֶׁמִּתְחִילָה לְהִתְיַבֵּשׁ מִשֶּׁמֶן.”?
מה קרה אחרי: “הוּא הֵבִין שֶׁהַחֲכָמִים יָדְעוּ בְּדִיּוּק מָה הֵם אוֹמְרִים.”?

סיבה ותוצאה מתוך המעשה

האירוע הנתוןמה גרם לו?מה הייתה תוצאתו?
כָּךְ הַשֶּׁמֶן שֶׁנִּשְׁאַר בַּצַּד, נוֹזֵל יָשָׁר לְתוֹךְ הַפְּתִילָה, וְהָאֵשׁ שׁוּב נִדְלֶקֶת וּמְאִירָה כְּלַפִּי מַעְלָה.
הוּא נָשַׁם נְשִׁימָה עֲמֻקָּה וְחֶרְדָּה כִּסְּתָה אֶת פָּנָיו הַטְּהוֹרוֹת.

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “העיד על עצמו שחטא בשל אי-זהירות בגזירה” על רבי ישמעאל בן אלישע? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “רֶגַע אֶחָד הוּא לוֹמֵד כֵּיצַד אָבוֹתֵינוּ שָׁמְרוּ עַל הַמִּצְווֹת, וְרֶגַע שֵׁנִי הוּא לוֹמֵד אֶת הֲלָכוֹת קָרְבְּנוֹת בֵּית הַמִּקְדָּשׁ.”.
ג. אילו רבי ישמעאל בן אלישע לא היה פועל כפי שמתואר ב־“כַּמָּה גְּדוֹלִים דִּבְרֵי חֲכָמִים שֶׁהָיוּ אוֹמְרִים: 'לֹא יִקְרָא לְאוֹר הַנֵּר'!" .” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “אִם מִישֶׁהוּ קוֹרֵא פִּנְקָס שֶׁל רַב דָּגוּל – הוּא בֶּטַח מְצַפֶּה לִרְאוֹת שָׁם כָּתוּב חִדּוּשֵׁי תּוֹרָה עֲמֻקִּים, אוֹ…” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. רַק דָּבָר אֶחָד מֵאִיר אֶת הַחֶדֶר: עַל הַשֻּׁלְחָן מֻנַּחַת מְנוֹרַת…
2. הֵם אָמְרוּ: "לֹא יִקְרָא אָדָם בְּלֵיל שַׁבָּת לְאוֹר הַנֵּר!".
3. וְכָךְ קָבַע הַצַּדִּיק: "אֲנִי אֶקְרָא – וְלֹא אַטֶּה" .
4. בְּאוֹתוֹ הָרֶגַע שֶׁהָאוֹר נִדְלַק – נִדְלַק אוֹר גַּם בְּתוֹךְ…
5. כְּשֶׁאָדָם בָּטוּחַ בְּעַצְמוֹ מִדַּי – בְּדִיּוּק שָׁם הוּא יָכוֹל…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “הַקֶּשֶׁר שֶׁלּוֹ לַקָּדוֹשׁ בָּרוּךְ הוּא הָיָה כָּל כָּךְ חָזָק!”.
מכתב: כתבו לרבי ישמעאל בן אלישע בעקבות “הוּא נָשַׁם נְשִׁימָה עֲמֻקָּה וְחֶרְדָּה כִּסְּתָה אֶת פָּנָיו הַטְּהוֹרוֹת.”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “הַשָּׁמַיִם הֶחְשִׁיכוּ בַּחוּץ וּשְׁלוֹשָׁה כּוֹכָבִים הֵצִיצוּ לְתוֹךְ הַחֶדֶר.”.
דיווח: כתבו ידיעה קצרה שכותרתה “העיד על עצמו שחטא בשל אי-זהירות בגזירה”.

המחזה

רגע מן הסיפורהדמויותמה יאמרו או יעשו?
כָּךְ הַשֶּׁמֶן שֶׁנִּשְׁאַר בַּצַּד, נוֹזֵל יָשָׁר לְתוֹךְ הַפְּתִילָה, וְהָאֵשׁ שׁוּב נִדְלֶקֶת…
כַּמָּה גְּדוֹלִים דִּבְרֵי חֲכָמִים שֶׁהָיוּ אוֹמְרִים: 'לֹא יִקְרָא לְאוֹר הַנֵּר'!" .
