המעשה — הסיפור המלא

להלן מערך שיעור וסיפור חינוכי, רגשי ומרתק, המבוסס על אמרותיו הטהורות של רבי חנינא בר חמא (מגדולי האמוראים, תלמידו של רבי יהודה הנשיא). הסיפור נבנה במיוחד עבור בנים בכיתות ב'-ד' בחינוך הדתי-לאומי, תוך שימוש בעקרונות הסטוריטלינג וההמחשה החושית, והקפדה על כבוד התורה.

הסיפור מנוקד במלואו כנדרש.

פְּתִיחָה: רְחוֹבוֹתֶיהָ שֶׁל צִפּוֹרִי

יְלָדִים, אֲנַחְנוּ סַבִּים עַתָּה לְאָחוֹר בְּמִנְהֶרֶת הַזְּמַן, כִּמְעַט אַלְפַּיִם שָׁנָה אָחוֹרָה, אֶל הָעִיר הַגְּדוֹלָה צִפּוֹרִי שֶׁבַּגָּלִיל.

הַשֶּׁמֶשׁ שֶׁל שְׁעַת הַצָּהֳרַיִם קָפְחָה מֵעַל, מְחַמֶּמֶת אֶת אַבְנֵי הָרְחוֹב הַלְּבָנוֹת. הַשּׁוּק שֶׁל צִפּוֹרִי הָיָה מָלֵא בַּאֲנָשִׁים. אִם הֱיִיתֶם עוֹמְדִים שָׁם, הֱיִיתֶם יְכוֹלִים לְהָרִיחַ אֶת הָרֵיחַ הַמָּתוֹק שֶׁל תְּאֵנִים טְרִיּוֹת שֶׁזֶּה עַתָּה נִקְטְפוּ, אֶת רֵיחַ הַתַּבְלִינִים הֶחָרִיף, וְאֶת נִיחוֹחַ הַלֶּחֶם הַחַם שֶׁיָּצָא מֵהַתַּנּוּר. בָּאָזְנַיִם הָיָה אֶפְשָׁר לִשְׁמֹעַ הֲמֻלָּה גְּדוֹלָה: סוֹחֲרִים הִכְרִיזוּ בְּקוֹל רָם עַל סְחוֹרָתָם, חֲמוֹרִים נָעֲרוּ, וּפַעֲמוֹנֵי כְּבָשִׂים צִלְצְלוּ.

בְּאוֹתָהּ עִיר קְדוֹשָׁה וְתוֹסֶסֶת, חַי מַנְהִיג עַם יִשְׂרָאֵל, הַצַּדִּיק הַגָּדוֹל, רַבִּי חֲנִינָא בַּר חָמָא . רַבִּי חֲנִינָא הָיָה אָדָם גְּבַהּ קוֹמָה, עִם זָקָן לָבָן כַּשֶּׁלֶג, וְעֵינַיִם שֶׁהֵאִירוּ כְּמוֹ כּוֹכָבִים בַּלַּיְלָה. כַּאֲשֶׁר הָיָה הוֹלֵךְ בָּרְחוֹב, הָיָה מֻרְגָּשׁ בָּאֲוִיר רֵיחַ עָדִין שֶׁל קְדֻשָּׁה, וְכֻלָּם הָיוּ עוֹמְדִים עַל רַגְלֵיהֶם מִתּוֹךְ יִרְאַת כָּבוֹד גְּדוֹלָה.

הַהִתְנַגְּשׁוּת וְהַכַּעַס הַמִּתְלַקֵּחַ

בְּאַחַד הַיָּמִים, קָרְתָה בַּשּׁוּק תְּאוּנָה קְטַנָּה, אַךְ מְעַצְבֶּנֶת מְאֹד. נִקְרָא לַסּוֹחֲרִים יְהוּדָה וְשִׁמְעוֹן. יְהוּדָה צָעַד בִּמְהִירוּת כְּשֶׁהוּא נוֹשֵׂא עַל כְּתֵפוֹ כַּד עָנָק שֶׁל שֶׁמֶן זַיִת יָקָר. מִמּוּלוֹ הִגִּיעַ שִׁמְעוֹן, סוֹחֵב עֲרֵמָה גְּדוֹלָה שֶׁל בַּדִּים יָפִים. רֶגַע אֶחָד שֶׁל חֹסֶר תְּשׂוּמַת לֵב, וְ... "טְרָאח!". הֵם הִתְנַגְּשׁוּ זֶה בָּזֶה!

כַּד הַשֶּׁמֶן שֶׁל יְהוּדָה נִשְׁמַט, הִתְרַסֵּק עַל מַרְצְפוֹת הָאֶבֶן בְּקוֹל נִפּוּץ צָרְמָנִי, וְשֶׁמֶן יְרַקְרַק יָקָר מְאֹד הִתְחִיל לִנְזֹל עַל הָרִצְפָּה. כַּמָּה טִפּוֹת שֶׁמֶן הִשְׁפְּרִיצוּ יָשָׁר עַל הַבַּדִּים הַיְקָרִים שֶׁל שִׁמְעוֹן וְהִכְתִּימוּ אוֹתָם.

יְלָדִים, אֲנַחְנוּ שׁוֹאֲלִים מָה עוֹשִׂים אֲנָשִׁים בְּרֶגַע כָּזֶה? הַאִם אֶחָד בִּקֵּשׁ סְלִיחָה? מַמָּשׁ לֹא. הַיֵּצֶר הָרַע, שֶׁמְּחַפֵּשׂ תָּמִיד מְרִיבוֹת, קָפַץ מִיָּד בֵּינֵיהֶם.

"הַיָּמִים הַלָּלוּ אַתָּה מְשֻׁגָּע?!" צָעַק יְהוּדָה כְּשֶׁפָּנָיו מַאֲדִימוֹת כְּעַגְבָנִיָּה, "שָׁבַרְתָּ לִי שֶׁמֶן שֶׁשָּׁוֶה חֲמִשִּׁים סְלָעִים! תְּשַׁלֵּם לִי מִיָּד!".

"אֲנִי אֲשַׁלֵּם לְךָ?!" הֵשִׁיב שִׁמְעוֹן בִּשְׁאָגָה וֶהֱרִים אֶת קוֹלוֹ כָּךְ שֶׁכָּל הַשּׁוּק שָׁמַע, "אַתָּה נִכְנַסְתָּ בִּי! תִּרְאֶה אֶת הַבַּדִּים הַמֻּכְתָּמִים שֶׁלִּי! בִּגְלָלְךָ הִפְסַדְתִּי מֵאָה סְלָעִים! שׁוֹטֶה שֶׁכְּמוֹתְךָ!".

הַקּוֹלוֹת שֶׁלָּהֶם עָלוּ וְגָבְרוּ. הַכַּעַס בָּעַר כְּמוֹ אֵשׁ. הַיָּדַיִם שֶׁל שִׁמְעוֹן רָעֲדוּ מֵרֹב עֲצַבִּים. הוּא לֹא שָׁלַט בְּעַצְמוֹ. הָעֵינַיִם שֶׁלּוֹ זָעֲמוּ, הוּא קָמַץ אֶת כַּף יָדוֹ לְאֶגְרוֹף קָשֶׁה... וֶהֱרִים אֶת הַיָּד כְּלַפֵּי יְהוּדָה כְּדֵי לְהַנְחִית עָלָיו מַכָּה נִמְרֶצֶת!

כֻּלָּם עָצְרוּ אֶת נְשִׁימָתָם בַּשּׁוּק. הַמַּכָּה עָמְדָה לָרֶדֶת.

"כָּל הַמַּגְבִּיהַּ יָדוֹ..."

"עֲצֹר מִיָּד, בְּנִי!" נִשְׁמַע פִּתְאוֹם קוֹל שָׁקֵט אֲבָל חָזָק וַחֲדוּר סַמְכוּת מֵאֲחוֹרֵיהֶם. זֶה הָיָה קוֹל שֶׁחָדַר יָשָׁר לַלֵּב.

הָאֶגְרוֹף שֶׁל שִׁמְעוֹן קָפָא בָּאֲוִיר. הוּא וִיהוּדָה סוֹבְבוּ אֶת רָאשֵׁיהֶם, וְהַבּוּשָׁה כִּסְּתָה אֶת פְּנֵיהֶם. מוּלָם לֹא עָמַד שׁוֹטֵר. מוּלָם עָמַד רַבִּי חֲנִינָא בַּר חָמָא בִּכְבוֹדוֹ וּבְעַצְמוֹ, כְּשֶׁהוּא אוֹחֵז בְּסֵפֶר קֹדֶשׁ תַּחַת זְרוֹעוֹ.

רַבִּי חֲנִינָא לֹא צָעַק, אֲבָל מַבָּטוֹ הָיָה נוֹקֵב וּכְאוּב.

"שִׁמְעוֹן," אָמַר הַצַּדִּיק בְּקוֹל רוֹטֵט מִכְּאֵב, "הוֹרֵד אֶת יָדְךָ בְּבַקָּשָׁה. הַאִם אַתָּה יוֹדֵעַ מָה אוֹמֶרֶת תּוֹרָתֵנוּ הַקְּדוֹשָׁה? 'כָּל הַמַּגְבִּיהַּ יָדוֹ עַל חֲבֵרוֹ – אַף עַל פִּי שֶׁלֹּא הִכָּהוּ, נִקְרָא חוֹטֵא!' ".

שִׁמְעוֹן הֵרִיד אֶת יָדוֹ לְאַט לְאַט. "אֲבָל... אֲבָל הַצַּדִּיק!" גִּמְגֵּם בְּבִלְבּוּל, "רַק הֲרַמְתִּי אֶת הַיָּד כְּדֵי לְהַפְחִיד אוֹתוֹ! אֲפִלּוּ לֹא נָגַעְתִּי בּוֹ! עֲדַיִן לֹא הִכֵּיתִי, לָמָּה אֲנִי חוֹטֵא?".

הִתְקָרֵב אֵלָיו רַבִּי חֲנִינָא, הִנִּיחַ יָד אַמִּיצָה וְרַכָּה עַל כְּתֵפוֹ שֶׁל שִׁמְעוֹן וְהִסְבִּיר לוֹ וּלְכָל סוֹחֲרֵי הַשּׁוּק שֶׁהִתְאַסְּפוּ: "יְלָדָיו שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא קְדוֹשִׁים הֵם. כְּשֶׁאַתָּה מֵרִים יָד, אַתָּה מַכְנִיס בַּחֲבֵרֲךָ פַּחַד, אַתָּה בּוֹחֵר בְּדֶרֶךְ שֶׁל כֹּחַ וְאַלִּימוּת מִשֶּׁל אוּמוֹת הָעוֹלָם. בְּעַם יִשְׂרָאֵל הַיָּד נוֹעֲדָה כְּדֵי לָתֵת צְדָקָה! הַיָּד נוֹעֲדָה כְּדֵי לְהָנִיחַ תְּפִלִּין וְלַעֲזֹר אִישׁ לְאָחִיו. כַּאֲשֶׁר אַתָּה מֵרִים אוֹתָהּ בְּכַעַס, כְּבָר בָּרֶגַע הַזֶּה קִלְקַלְתָּ אֶת מַטְּרַתְךָ כִּיהוּדִי, וְלָכֵן נִקְרֵאתָ חוֹטֵא!".

סוֹד הַבְּחִירָה: הַכֹּל בִּידֵי שָׁמַיִם

יְהוּדָה הַסּוֹחֵר נִגַּשׁ גַּם הוּא וְאָמַר לְרַבִּי חֲנִינָא: "אֲבָל רַבִּי! תִּרְאֶה אֶת הַשֶּׁמֶן הַיָּקָר שֶׁלִּי עַל הָרִצְפָּה. אֵיךְ אֶפְשָׁר לֹא לִכְעֹס? אֵיךְ אֶפְשָׁר שֶׁלֹּא לְקַלֵּל וְלָרִיב?".

רַבִּי חֲנִינָא חִיֵּךְ חִיּוּךְ מָלֵא בַּאֲצִילוּת וְהַבָּעָה שֶׁל נֹעַם. הוּא בִּקֵּשׁ מִשְּׁנֵיהֶם לָגֶשֶׁת אִתּוֹ שְׁנֵי מֶטְרִים הַצִּדָּה, אֶל מִתַּחַת לָעֵץ, בַּצֵּל הַקָּרִיר. אוֹר הַשֶּׁמֶשׁ סִנֵּן קַרְנַיִם רַכּוֹת דֶּרֶךְ הֶעָלִים.

"בָּנַי הַיְקָרִים," לָחַשׁ הַחָכָם, "מַדּוּעַ אֲנַחְנוּ כּוֹעֲסִים בַּחַיִּים? כִּי מַשֶּׁהוּ קוֹרֶה לֹא כְּפִי שֶׁתִּכְנַנּוּ. תִּכְנַנְתֶּם לַעֲשׂוֹת כֶּסֶף, וְהִנֵּה הַשֶּׁמֶן וְהַבַּדִּים נֶהֶרְסוּ. אֲבָל אָנָּא מִכֶּם, הַקְשִׁיבוּ לִכְלָל גָּדוֹל שֶׁמָּסַרְתִּי לְתַלְמִידַי בְּבֵית הַמִּדְרָשׁ: 'הַכֹּל בִּידֵי שָׁמַיִם – חוּץ מִיִּרְאַת שָׁמַיִם!' ".

שִׁמְעוֹן וִיהוּדָה הִבִּיטוּ זֶה בָּזֶה, סַקְרָנִים לְהָבִין עַל מָה סָב הַדָּבָר.

"הַקָּדוֹשׁ בָּרוּךְ הוּא קוֹבֵעַ מִי יִהְיֶה עָשִׁיר וּמִי יִהְיֶה עָנִי, מָתַי יֵרֵד גֶּשֶׁם וּמָתַי לֹא," הִסְבִּיר הַצַּדִּיק. "גַּם הָעֻבְדָּה שֶׁנִּתְקַלְתֶּם זֶה בָּזֶה כָּאן בַּשּׁוּק וְהַכַּד נִשְׁבַּר – גַּם זֶה הָיָה בִּידֵי שָׁמַיִם! אֵין מִקְרִים בָּעוֹלָם. הַקָּדוֹשׁ בָּרוּךְ הוּא גָּזַר שֶׁהַכֶּסֶף הַזֶּה יֵאָבֵד הַיּוֹם. אֲבָל... מָה לֹא נִמְצָא בִּידֵי שָׁמַיִם? דָּבָר אֶחָד יָחִיד הַשֵּׁם יִתְבָּרַךְ מָסַר לַשְּׁלִיטָה שֶׁלָּכֶם: 'יִרְאַת שָׁמַיִם'! שֶׁזֶּה אֹמֶר – הַבְּחִירָה הַאִם לִכְעֹס אוֹ לְהִשָּׁאֵר רְגוּעִים, הַאִם לְשַׁלֵּחַ אֶגְרוֹף אוֹ לְבַקֵּשׁ שָׁלוֹם – זֶה נִמְצָא רַק בַּיָּדַיִם שֶׁלָּכֶם! הָאֵירוּעַ קָרָה. אֲבָל הַתְּגוּבָה? הִיא בַּבְּחִירָה שֶׁלָּכֶם. וְכַאֲשֶׁר אַתֶּם נִכְנָעִים לַכַּעַס, אַתֶּם מַפְסִידִים בְּמִבְחַן הַבְּחִירָה הַיְחִידָה שֶׁנִּתְּנָה לָכֶם עַל יְדֵי בּוֹרֵא עוֹלָם!".

תַּלְמִידֵי חֲכָמִים מַרְבִּים שָׁלוֹם

הַדְּבָרִים הַלָּלוּ יָרְדוּ כְּטַל קָרִיר עַל לִבּוֹתֵיהֶם הַבּוֹעֲרִים שֶׁל הַסּוֹחֲרִים. הַכַּעַס הֵחֵל לְהִתְפּוֹגֵג. הֵם הֵבִינוּ שֶׁבִּגְלַל כַּד וְכַמָּה בַּדִּים, הֵם הָיוּ מוּכָנִים לִפְגֹּעַ בִּיהוּדִי אֶחָד, וְלֶחֱטֹא בְּחוֹטֵא לִפְנֵי הַשֵּׁם!.

"אָז מָה עָלֵינוּ לַעֲשׂוֹת כָּעֵת, רַבִּי?" לָחַשׁ שִׁמְעוֹן בְּקּוֹל מָלֵא כָּבוֹד, וְעֵינָיו בָּרִצְפָּה.

רַבִּי חֲנִינָא הִבִּיט בָּהֶם וְאָמַר מִשְׁפָּט אֲשֶׁר נוֹתָר חָקוּק בַּתַּלְמוּד עַד יָמֵינוּ אָנוּ:

"אָמַר רַבִּי חֲנִינָא בַּר חָמָא: תַּלְמִידֵי חֲכָמִים מַרְבִּים שָׁלוֹם בָּעוֹלָם!"

וְהִסְבִּיר הַצַּדִּיק: "לִלְמֹד תּוֹרָה זֶה דָּבָר עָצוּם, אֲבָל מִי שֶׁהוּא תַּלְמִיד חָכָם אֲמִתִּי, מִי שֶׁלּוֹמֵד אֶת תּוֹרַת ד', חַיָּב שֶׁהַתּוֹרָה הַזּוֹ תּוֹבִיל אוֹתוֹ לְדָבָר אֶחָד כְּלַפֵּי אַחֵינוּ בְּנֵי יִשְׂרָאֵל: לְשָׁלוֹם! קְחוּ נְשִׁימָה. בּוֹאוּ תֵּשְׁבוּ עִמִּי. נַעֲשֶׂה חֶשְׁבּוֹן צוֹדֵק בֵּינֵיכֶם לְפִי דִּינֵי הַתּוֹרָה שֶׁל מַסֶּכֶת 'בָּבָא קַמָּא' בְּצוּרָה שְׁקֵטָה וַהֲגוּנָה. הַכֶּסֶף יַחֲזֹר אוֹ יִשָּׁלֵם, אֲבָל הַלֵּב? הַלֵּב יִשָּׁאֵר קָדוֹשׁ מִשִּׂנְאָה וְכַעַס".

יְהוּדָה וְשִׁמְעוֹן הִבִּיטוּ אִישׁ בְּרֵעֵהוּ. יְהוּדָה פָּתַח בְּחִיּוּךְ קָטָן, וְשִׁמְעוֹן מִיָּד הֵשִׁיב לוֹ. "סְלִיחָה, שִׁמְעוֹן. לֹא רָאִיתִי לְאָן אֲנִי הוֹלֵךְ," אָמַר יְהוּדָה. "וְאַתָּה תִּסְלַח לִי," הֵשִׁיב שִׁמְעוֹן, "שֶׁכִּמְעַט הֵרַמְתִּי עָלֶיךָ יָד. הַחֵטְא הַזֶּה נוֹרָא יוֹתֵר מִכָּל בַּד מוּכְתָּם!".

הֵם צָעֲדוּ שְׁלָשְׁתָּם אַחֲרֵי רַבִּי חֲנִינָא אֶל עֲבַר הַצֵּל שֶׁל בֵּית הַמִּדְרָשׁ, מַשְׁאִירִים אֶת כַּעֲסָם כַּאֲבַק דְּרָכִים מֵאָחוֹר, וּמְקַבְּלִים עֲלֵיהֶם אֶת הַכֹּתֶרֶת הָאֲמִתִּית שֶׁל הַיְּהוּדִי: נוֹשְׂאֵי הַשָּׁלוֹם.

א. שְׁאֵלַת פְּתִיחָה (לְעוֹרֵר סַקְרָנוּת):

"יְלָדִים, הַאִם פַּעַם מִישֶׁהוּ קִלְקֵל לָכֶם מִשְׂחָק, וְהִרְגַּשְׁתֶּם כַּעַס גָּדוֹל שֶׁמַּמָּשׁ עוֹלֶה לָכֶם בְּכָל הַגּוּף? הַאִם הָיָה קָשֶׁה לָכֶם לַעֲצֹר מִלִּדְחֹף او לְהָרִים יָד?

הַיּוֹם נִשְׁמַע עַל צַדִּיק עָצוּם שֶׁלִּמֵּד אֲנָשִׁים מָה לַעֲשׂוֹת בְּדִיּוּק בָּרֶגַע שֶׁהַכַּעַס מִתְפּוֹצֵץ..."

ב. הֶעָרוֹת בִּצּוּעַ וּסְטוֹרִיטֶלִינְג (לַמּוֹרֶה):

ג. הַמְלָצוֹת לְמַצֶּגֶת (4 שְׁקוּפִיּוֹת מוּרְאוֹת-רוּחַ):

ד. פְּעִילוּת מְסַכֶּמֶת (מִשְׂחַק תַּפְקִידִים וְעֲצִירָה):

מִשְׂחַק "הַיָּד שֶׁבּוֹחֶרֶת"

מקורות: "כָּל הַמַּגְבִּיהַּ יָדוֹ עַל חֲבֵרוֹ... נִקְרָא חוֹטֵא" - מַסֶּכֶת סַנְהֶדְרִין דַּף נ"ח, עַמּוּד ב' (מוֹפִיעַ כְּמַאֲמָרוֹ שֶׁל רַבִּי חֲנִינָא). "הַכֹּל בִּידֵי שָׁמַיִם חוּץ מִיִּרְאַת שָׁמַיִם" - מַסֶּכֶת מְגִלָּה דַּף כ"ה, עַמּוּד א' וּמַסֶּכֶת בְּרָכוֹת דַּף ל"ג, עַמּוּד ב' (מַאֲמָר מְפֻרְסָם שֶׁל רַבִּי חֲנִינָא). "תַּלְמִידֵי חֲכָמִים מַרְבִּים שָׁלוֹם בָּעוֹלָם" - מַסֶּכֶת בְּרָכוֹת דַּף ס"ד, עַמּוּד א', וִיבָמוֹת דַּף קכ"ב, עַמּוּד ב' (סִיּוּם הַמַּסֶּכֶת שֶׁנֶּאֱמַר עַל יְדֵי רַבִּי חֲנִינָא). מָקוֹם מוֹשָׁבוֹ שֶׁל רַבִּי חֲנִינָא בַּר חָמָא - בָּעִיר צִפּוֹרִי שֶׁבַּגָּלִיל (לְאַחַר רַבִּי יְהוּדָה הַנָּשִׂיא), גְּדוּלָתוֹ בָּרְפוּאָה, בַּחַקְלָאוּת, בְּהַנְהָגַת תּוֹרָה וּבְמִדּוֹת שֶׁל שָׁלוֹם וֶעֱנָוָה מוֹפִיעוֹת בְּרַחֲבֵי הַתַּלְמוּד הַבַּבְלִי וְהַיְרוּשַׁלְמִי.

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “להלן מערך שיעור וסיפור חינוכי, רגשי ומרתק, המבוסס על אמרותיו הטהורות של רבי חנינא בר חמא (מגדולי האמוראים, תלמידו של רבי יהודה הנשיא).” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “בְּאַחַד הַיָּמִים, קָרְתָה בַּשּׁוּק תְּאוּנָה קְטַנָּה, אַךְ מְעַצְבֶּנֶת מְאֹד.”?
ג. כיצד הגיב ר' חנינא בר חמא למצב המתואר: “עֲצֹר מִיָּד, בְּנִי!" נִשְׁמַע פִּתְאוֹם קוֹל שָׁקֵט אֲבָל חָזָק וַחֲדוּר סַמְכוּת מֵאֲחוֹרֵיהֶם.”?
ד. מה השתנה בין “הַיָּד נוֹעֲדָה כְּדֵי לְהָנִיחַ תְּפִלִּין וְלַעֲזֹר אִישׁ לְאָחִיו.” לבין “הַקָּדוֹשׁ בָּרוּךְ הוּא גָּזַר שֶׁהַכֶּסֶף הַזֶּה יֵאָבֵד הַיּוֹם.”?
ה. כיצד הגיע הסיפור אל האירוע: “נַעֲשֶׂה חֶשְׁבּוֹן צוֹדֵק בֵּינֵיכֶם לְפִי דִּינֵי הַתּוֹרָה שֶׁל מַסֶּכֶת 'בָּבָא קַמָּא' בְּצוּרָה שְׁקֵטָה וַהֲגוּנָה.”?

□ הִתְקָרֵב אֵלָיו רַבִּי חֲנִינָא, הִנִּיחַ יָד אַמִּיצָה וְרַכָּה עַל כְּתֵפוֹ שֶׁל שִׁמְעוֹן וְהִסְבִּיר לוֹ וּלְכָל סוֹחֲרֵי הַשּׁוּק שֶׁהִתְאַסְּפוּ:…
□ להלן מערך שיעור וסיפור חינוכי, רגשי ומרתק, המבוסס על אמרותיו הטהורות של רבי חנינא בר חמא (מגדולי האמוראים, תלמידו של רבי יהודה הנשיא).
□ הַיּוֹם נִשְׁמַע עַל צַדִּיק עָצוּם שֶׁלִּמֵּד אֲנָשִׁים מָה לַעֲשׂוֹת בְּדִיּוּק בָּרֶגַע שֶׁהַכַּעַס מִתְפּוֹצֵץ...
□ כַּד הַשֶּׁמֶן שֶׁל יְהוּדָה נִשְׁמַט, הִתְרַסֵּק עַל מַרְצְפוֹת הָאֶבֶן בְּקוֹל נִפּוּץ צָרְמָנִי, וְשֶׁמֶן יְרַקְרַק יָקָר מְאֹד הִתְחִיל לִנְזֹל עַל…
□ דָּבָר אֶחָד יָחִיד הַשֵּׁם יִתְבָּרַךְ מָסַר לַשְּׁלִיטָה שֶׁלָּכֶם: 'יִרְאַת שָׁמַיִם'!

לפני ואחרי

מה קרה לפני: “הִתְקָרֵב אֵלָיו רַבִּי חֲנִינָא, הִנִּיחַ יָד אַמִּיצָה וְרַכָּה עַל כְּתֵפוֹ שֶׁל שִׁמְעוֹן וְהִסְבִּיר לוֹ וּלְכָל סוֹחֲרֵי הַשּׁוּק שֶׁהִתְאַסְּפוּ:…”?
מה קרה אחרי: “דָּבָר אֶחָד יָחִיד הַשֵּׁם יִתְבָּרַךְ מָסַר לַשְּׁלִיטָה שֶׁלָּכֶם: 'יִרְאַת שָׁמַיִם'!”?

סיבה ותוצאה מתוך המעשה

האירוע הנתוןמה גרם לו?מה הייתה תוצאתו?
כַּמָּה טִפּוֹת שֶׁמֶן הִשְׁפְּרִיצוּ יָשָׁר עַל הַבַּדִּים הַיְקָרִים שֶׁל שִׁמְעוֹן וְהִכְתִּימוּ אוֹתָם.
אֲבָל אָנָּא מִכֶּם, הַקְשִׁיבוּ לִכְלָל גָּדוֹל שֶׁמָּסַרְתִּי לְתַלְמִידַי בְּבֵית הַמִּדְרָשׁ: 'הַכֹּל בִּידֵי שָׁמַיִם – חוּץ מִיִּרְאַת שָׁמַיִם!' ".

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “מאמריו: "תלמידי חכמים מרבים שלום...", "כל המגביה ידו...", "הכל בידי שמים חוץ מיראת שמים"” על ר' חנינא בר חמא? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “'כָּל הַמַּגְבִּיהַּ יָדוֹ עַל חֲבֵרוֹ – אַף עַל פִּי שֶׁלֹּא הִכָּהוּ, נִקְרָא חוֹטֵא!' ".”.
ג. אילו ר' חנינא בר חמא לא היה פועל כפי שמתואר ב־“הַקָּדוֹשׁ בָּרוּךְ הוּא גָּזַר שֶׁהַכֶּסֶף הַזֶּה יֵאָבֵד הַיּוֹם.” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “נַעֲשֶׂה חֶשְׁבּוֹן צוֹדֵק בֵּינֵיכֶם לְפִי דִּינֵי הַתּוֹרָה שֶׁל מַסֶּכֶת 'בָּבָא קַמָּא' בְּצוּרָה שְׁקֵטָה וַהֲגוּנָה.” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. להלן מערך שיעור וסיפור חינוכי, רגשי ומרתק, המבוסס על אמרותיו הטהורות של…
2. יְהוּדָה צָעַד בִּמְהִירוּת כְּשֶׁהוּא נוֹשֵׂא עַל כְּתֵפוֹ כַּד עָנָק…
3. מוּלָם עָמַד רַבִּי חֲנִינָא בַּר חָמָא בִּכְבוֹדוֹ וּבְעַצְמוֹ,…
4. רַבִּי חֲנִינָא חִיֵּךְ חִיּוּךְ מָלֵא בַּאֲצִילוּת וְהַבָּעָה שֶׁל…
5. וְכַאֲשֶׁר אַתֶּם נִכְנָעִים לַכַּעַס, אַתֶּם מַפְסִידִים בְּמִבְחַן…
6. הַיּוֹם נִשְׁמַע עַל צַדִּיק עָצוּם שֶׁלִּמֵּד אֲנָשִׁים מָה לַעֲשׂוֹת…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “עֲצֹר מִיָּד, בְּנִי!" נִשְׁמַע פִּתְאוֹם קוֹל שָׁקֵט אֲבָל חָזָק וַחֲדוּר סַמְכוּת מֵאֲחוֹרֵיהֶם.”.
מכתב: כתבו לר' חנינא בר חמא בעקבות “אֲבָל אָנָּא מִכֶּם, הַקְשִׁיבוּ לִכְלָל גָּדוֹל שֶׁמָּסַרְתִּי לְתַלְמִידַי בְּבֵית הַמִּדְרָשׁ: 'הַכֹּל בִּידֵי שָׁמַיִם – חוּץ…”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “הַדְּבָרִים הַלָּלוּ יָרְדוּ כְּטַל קָרִיר עַל לִבּוֹתֵיהֶם הַבּוֹעֲרִים שֶׁל הַסּוֹחֲרִים.”.
דיווח: כתבו ידיעה קצרה שכותרתה “מאמריו: "תלמידי חכמים מרבים שלום...", "כל המגביה ידו...", "הכל בידי שמים חוץ מיראת שמים"”.

המחזה

רגע מן הסיפורהדמויותמה יאמרו או יעשו?
כַּמָּה טִפּוֹת שֶׁמֶן הִשְׁפְּרִיצוּ יָשָׁר עַל הַבַּדִּים הַיְקָרִים שֶׁל שִׁמְעוֹן וְהִכְתִּימוּ…
הַקָּדוֹשׁ בָּרוּךְ הוּא גָּזַר שֶׁהַכֶּסֶף הַזֶּה יֵאָבֵד הַיּוֹם.

7. חיבור לחיים מתוך המעשה

א. מה אפשר ללמוד מן הדרך שבה ר' חנינא בר חמא פעל באירוע “הַיָּד נוֹעֲדָה כְּדֵי לְהָנִיחַ תְּפִלִּין וְלַעֲזֹר אִישׁ לְאָחִיו.”?
ב. תארו מצב מחיי הכיתה שבו אפשר ליישם את המסר של “מאמריו: "תלמידי חכמים מרבים שלום...", "כל המגביה ידו...", "הכל בידי שמים חוץ מיראת שמים"”.
