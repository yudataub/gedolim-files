יצא מהמרחץ כדי ללמוד עם נכדו

דף עבודה בעקבות ר' יהושע בן לוי
שם: כיתה:

המעשה — הסיפור המלא

פְּתִיחָה: רֵיחוֹת שֶׁל עֶרֶב שַׁבָּת וְקוֹלוֹת שֶׁל תּוֹרָה

יְלָדִים אֲהוּבִים, דַּמְיְנוּ עַצְמְכֶם בָּעִיר לוֹד שֶׁבְּאֶרֶץ יִשְׂרָאֵל, לִפְנֵי שָׁנִים רַבּוֹת. הַיּוֹם הוּא יוֹם שִׁשִּׁי, עֶרֶב שַׁבָּת קֹדֶשׁ. בָּרְחוֹבוֹת אֶפְשָׁר לְהָרִיחַ אֶת הָרֵיחַ הַמָּתוֹק וְהַחַם שֶׁל חַלּוֹת טְרִיּוֹת שֶׁזֶּה עַתָּה יָצאוּ מִתַּנּוּרֵי הָאֶבֶן, וְאֶת הָרֵיחַ הֶעָמֹק שֶׁל תַּבְשִׁילֵי הַשַּׁבָּת הַמִּתְבַּשְּׁלִים בַּסִּירִים. כֻּלָּם מְמַהֲרִים. קוֹלוֹת שֶׁל צַעֲדֵי הֲלִיכָה מְהִירָה, שֶׁל סוֹחֲרִים שֶׁסּוֹגְרִים אֶת חֲנוּיוֹתֵיהֶם, וְשֶׁל אִמָּהוֹת הַקּוֹרְאוֹת לַיְּלָדִים לְהִתְכּוֹנֵן לַשַּׁבָּת.

בְּמֶרְכַּז הָעִיר הָיָה הַבַּיִת הַחָשׁוּב מִכֻּלָּם – בֵּיתוֹ שֶׁל הַצַּדִּיק, שַׂר הַתּוֹרָה, רַבִּי יְהוֹשֻׁעַ בֶּן לֵוִי . רַבִּי יְהוֹשֻׁעַ הָיָה אָדָם גָּדוֹל עַד כְּדֵי כָּךְ, שֶׁהָיָה זוֹכֶה לִרְאוֹת אֶת אֵלִיָּהוּ הַנָּבִיא בָּא לִלְמֹד אִתּוֹ תּוֹרָה!

אַךְ הָיָה לְרַבִּי יְהוֹשֻׁעַ חֹק קָבוּעַ, חֹק שֶׁעָלָיו לֹא וִתֵּר מֵעוֹלָם: בְּכָל יוֹם שִׁשִּׁי, רֶגַע לִפְנֵי קַבָּלַת שַׁבָּת, הָיָה מוֹשִׁיב עַל בִּרְכָּיו אֶת נֶכְדּוֹ הַקָּטָן. הַסַּבָּא הַגָּדוֹל, שֶׁכָּל הָרַבָּנִים רָעֲדוּ מִפָּנָיו, הָיָה מַבִּיט בְּעֵינַיִם בּוֹרְקוֹת מֵאַהֲבָה בְּנֶכְדּוֹ הַקָּטָן, וְהַנֶּכֶד הָיָה קוֹרֵא בְּקוֹל צָלוּל וּמָתוֹק אֶת פָּרָשַׁת הַשָּׁבוּעַ שֶׁלָּמַד בַּ"חֶדֶר". בִּשְׁבִיל רַבִּי יְהוֹשֻׁעַ, הַקּוֹל הַזֶּה הָיָה שָׁוֶה יוֹתֵר מִכָּל הַזָּהָב שֶׁבָּעוֹלָם!

הַחֹם, הַשֶּׁקֶט, וְהַבֶּהָלָה

וְהִנֵּה, בְּיוֹם שִׁשִּׁי אֶחָד, הָיָה רַבִּי יְהוֹשֻׁעַ עָיֵף בִּמְיֻחָד. כָּל הַשָּׁבוּעַ לָמַד מֵהַבֹּקֶר עַד אִישׁוֹן הַלַּיְלָה. הוּא הָלַךְ אֶל בֵּית הַמֶּרְחָץ כְּדֵי לִרְחֹץ אֶת גּוּפוֹ וְלִכְבוֹד שַׁבָּת מַלְכְּתָא.

הוּא נִכְנַס פְּנִימָה. רֵיחַ נָעִים שֶׁל שְׁמָנִים נָשָׂא בָּאֲוִיר, וְאֵדִים עָבִים וּלְבָנִים כִּסּוּ אֶת הַחֶדֶר. מַיִם רְתוּחִים נִשְׁפְּכוּ בְּקוֹל אִוְשָׁה וְהִרְפּוּ אֶת כָּל הַשְּׁרִירִים הַתְּפוּסִים. כַּמָּה נָעִים, כַּמָּה מַרְגִּיעַ... סוֹף סוֹף, רֶגַע קָטָן שֶׁל מְנוּחָה.

לְפֶתַע, תּוֹךְ כְּדֵי שֶׁהוּא טוֹבֵל בַּמַּיִם הַחַמִּים... רַבִּי יְהוֹשֻׁעַ קָפָא בִּמְקוֹמוֹ.

הוּא טָפַח עַל מִצְחוֹ. עֵינָיו נִפְעֲרוּ לִרְוָחָה.

"אוֹי וַאֲבוֹי!" לָחַשׁ הַצַּדִּיק לְעַצְמוֹ, "נֶכְדִּי! עוֹד לֹא שָׁמַעְתִּי הַיּוֹם אֶת פָּרָשַׁת הַשָּׁבוּעַ מִפִּיו שֶׁל נֶכְדִּי הַיָּקָר!"

יְלָדִים, בּוֹאוּ נַחְשֹׁב יַחַד – מָה עָשָׂה רַבִּי יְהוֹשֻׁעַ? הוּא הֲרֵי צַדִּיק גָּדוֹל! אָדָם רָגִיל שֶׁנִּמְצָא בַּמֶּרְחָץ הֶחָם כְּשֶׁבַּחוּץ נוֹשֶׁבֶת רוּחַ קָרָה, הָיָה אוֹמֵר בְּלִבּוֹ: "הַכֹּל בְּסֵדֶר. אֲנִי אֶסְתַּבֵּן, אֶתְנַגֵּב, אֶתְלַבֵּשׁ בִּבְגָדַי הַמְּכֻבָּדִים, אֶצְעַד הַבַּיְתָה בְּנַחַת, וְאָז אֶשְׁמַע אֶת הַפָּרָשָׁה. הַנֶּכֶד יוּכַל לְחַכּוֹת. סַבָּא צָרִיךְ רֶגַע מְנוּחָה".

אֲבָל לֹא רַבִּי יְהוֹשֻׁעַ בֶּן לֵוִי. אֶצְלוֹ תּוֹרָה הִיא תּוֹרַת אֱמֶת, תּוֹרַת אֵשׁ הַבּוֹעֶרֶת בָּעֲצָמוֹת!

בִּשְׁנִיָּה אַחַת, הוּא זִנֵּק מִתּוֹךְ בְּרֵכַת הַמַּיִם הַחַמִּים! הוּא לֹא חִכָּה אֲפִלּוּ כְּדֵי לְהִתְלַבֵּשׁ בִּבְגָדָיו הַמְּכֻבָּדִים וּבִגְלִימָתוֹ אֲרוּכַת הַשּׁוּלַיִם. הוּא פָּשׁוּט לָקַח בַּד מַגֶּבֶת גָּדוֹל, עָטַף אֶת עַצְמוֹ בִּמְהִירוּת שֶׁיִּהְיֶה מְכֻסֶּה וְצָנוּעַ, וְרָץ... פָּשׁוּט רָץ מִתּוֹךְ הַמֶּרְחָץ הַחוּצָה אֶל הָרְחוֹב!

הַמְּרוּצָה בָּרְחוֹב וְהַפְּלִיאָה שֶׁל הַתַּלְמִידִים

הָרוּחַ הַקָּרָה שֶׁל יוֹם הַשִּׁשִּׁי הִכְּתָה עַל גּוּפוֹ שֶׁנָּטַף עֲדַיִן מַיִם, אֲבָל רַבִּי יְהוֹשֻׁעַ לֹא הִרְגִּישׁ קֹר, הוּא הִרְגִּישׁ רַק חֹם שֶׁל מִצְוָה. גְּדוֹל הַדּוֹר הוֹלֵךְ בָּרְחוֹב, לְלֹא מַלְבּוּשֵׁי מַלְכוּתוֹ, רַק בְּסָדִין וּמַגֶּבֶת! אֲנָשִׁים בַּשּׁוּק עָצְרוּ וְהִבִּיטוּ.

"מָה קָרָה?!" שָׁאֲלוּ אֶחָד אֶת הַשֵּׁנִי, "הַאִם פָּרְצָה דְּלֵקָה? הַאִם רַבִּי יְהוֹשֻׁעַ בּוֹרֵחַ מֵאוֹיֵב?!"

רַב יְהוּדָה וְרַבִּי חִיָּיא בַּר אַבָּא, שְׁנַיִם מִתַּלְמִידָיו הַגְּדוֹלִים, רָאוּ אֶת רַבָּם בָּרְחוֹב וְלֹא הֶאֱמִינוּ לְמַרְאֵה עֵינֵיהֶם. הֵם רָצוּ מִיָּד אֵלָיו וּפָנוּ אֵלָיו בְּיִרְאַת הַכָּבוֹד:

"רַבֵּנוּ... הַצַּדִּיק! רַבִּי!" פָּתְחוּ בְּקוֹל רוֹעֵד, "מָה מִהֲרָה יֵשׁ בַּדָּבָר? לָמָּה כְּבוֹדוֹ יוֹצֵא בְּלִי הַבְּגָדִים הַמְּכֻבָּדִים שֶׁלּוֹ? הֲרֵי אַתָּה נְשִׂיא לָנוּ וּגְדוֹל לָנוּ! הֲרֵי זֶה בִּזָּיוֹן לִכְבוֹד הַתּוֹרָה שֶׁהָרַב נִרְאֶה כָּךְ בָּרְחוֹב! הָרַב יָכוֹל הָיָה לְהִתְלַבֵּשׁ, וַאֲנַחְנוּ הָיִינוּ רָצִים לְהָבִיא לוֹ אֶת מָה שֶׁהוּא צָרִיךְ!".

רַבִּי יְהוֹשֻׁעַ בֶּן לֵוִי נֶעֱצַר. הוּא נָשַׁם נְשִׁימָה עֲמֻקָּה. הוּא הִבִּיט בְּתַלְמִידָיו, וְהַפָּנִים שֶׁלּוֹ אָרְוּ בְּאוֹר קָדוֹשׁ.

"תַּלְמִידַי הַיְקָרִים," לָחַשׁ בְּקוֹל אֵיתָן, "אַתֶּם חוֹשְׁבִים שֶׁבִּזִּיתִי אֶת כְּבוֹדִי? דְּעוּ לָכֶם שֶׁלֹּא! שָׁכַחְתִּי לִשְׁמֹעַ הַיּוֹם אֶת הַפָּרָשָׁה מִפִּי נֶכְדִּי. הַאִם אַתֶּם יוֹדְעִים מָה זֶה אוֹמֵר סַבָּא שֶׁשּׁוֹמֵעַ אֶת נֶכְדּוֹ לוֹמֵד תּוֹרָה? הֲלֹא דָּוִד הַמֶּלֶךְ אָמַר: 'וְהוֹדַעְתָּם לְבָנֶיךָ וְלִבְנֵי בָנֶיךָ', וּמִיָּד שָׁם כָּתוּב הַפָּסוּק הַבָּא: 'יוֹם אֲשֶׁר עָמַדְתָּ לִפְנֵי ה' אֱלֹקֶיךָ בְּחֹרֵב'!".

רַבִּי יְהוֹשֻׁעַ הֵרִים אֶת יָדוֹ כְּלַפֵּי הַשָּׁמַיִם וּזְקָנוֹ הַלָּבָן זָע בָּרוּחַ:

"כָּל מִי שֶׁמְּלַמֵּד תּוֹרָה אֶת הַנֶּכֶד שֶׁלּוֹ, אוֹ שׁוֹמֵעַ תּוֹרָה מֵהַנֶּכֶד שֶׁלּוֹ – בַּשָּׁמַיִם מַחְשִׁיבִים לוֹ אֶת זֶה כְּאִלּוּ הוּא עוֹמֵד עַכְשָׁו תַּחַת הַר סִינַי! כְּאִלּוּ מֹשֶׁה רַבֵּנוּ עַכְשָׁו מוֹרִיד אֶת הַלּוּחוֹת! כְּאִלּוּ יֵשׁ עַכְשָׁו רְעָמִים וּבְרָקִים וְקוֹל שׁוֹפָר! הַאִם כְּשֶׁה' נוֹתֵן תּוֹרָה בְּהַר סִינַי, אֲנִי אֵשֵׁב בְּבֵית הַמֶּרְחָץ וְאֶתְלַבֵּשׁ לְאַט לְאַט?! כְּשֶׁמְּקַבְּלִים תּוֹרָה – רָצִים!".

בְּמַעֲמַד הַר סִינַי שֶׁל הַדּוֹר הַבָּא

רַבִּי יְהוֹשֻׁעַ בֶּן לֵוִי סִיֵּם אֶת דְּבָרָיו, עָזַב אֶת מְלַוָּיו, וְהִמְשִׁיךְ לָרוּץ אֶל בֵּית הַמִּדְרָשׁ שֶׁל הַיְּלָדִים הַקְּטַנִּים.

כְּשֶׁנִּכְנַס סוֹף סוֹף פְּנִימָה, נֶעֶצְרָה נְשִׁימָתוֹ שֶׁל הַנֶּכֶד, וְשֶׁל כָּל יַלְדֵי הַכִּתָּה. סַבָּא כָּאן!

רַבִּי יְהוֹשֻׁעַ הִתְיַשֵּׁב עַל הַסַּפְסָל שֶׁלְּיַד נֶכְדּוֹ, לֹא מַרְגִּישׁ כְּלָל אֶת הָרְטִיבוּת בְּשַׂעֲרוֹ אוֹ אֶת קֹר הַקִּירוֹת. הוּא הוֹשִׁיב אֶת הַנֶּכֶד קָרוֹב-קָרוֹב אֵלָיו, עָצַם אֶת עֵינָיו, נָשַׁם עָמֹק, וְלָחַשׁ בְּקוֹל רוֹטֵט מֵהִתְרַגְּשׁוּת:

"קְרָא בְּנִי... אֲנִי שׁוֹמֵעַ... הַר סִינַי שֶׁלִּי נִמְצָא מַמָּשׁ כָּאן."

וְהַנֶּכֶד, בְּקוֹל דַּק וּמָתוֹק כִּדְבַשׁ, פָּתַח בְּנִגּוּן שֶׁל תּוֹרָה. רַבִּי יְהוֹשֻׁעַ בָּכָה דִּמְעוֹת גִּיל. הַשַּׁבָּת הִגִּיעָה לִירוּשָׁלַיִם.

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “פְּתִיחָה: רֵיחוֹת שֶׁל עֶרֶב שַׁבָּת וְקוֹלוֹת שֶׁל תּוֹרָה” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “אַךְ הָיָה לְרַבִּי יְהוֹשֻׁעַ חֹק קָבוּעַ, חֹק שֶׁעָלָיו לֹא וִתֵּר מֵעוֹלָם: בְּכָל יוֹם שִׁשִּׁי, רֶגַע לִפְנֵי קַבָּלַת שַׁבָּת, הָיָה מוֹשִׁיב עַל…”?
ג. כיצד הגיב ר' יהושע בן לוי למצב המתואר: “מַיִם רְתוּחִים נִשְׁפְּכוּ בְּקוֹל אִוְשָׁה וְהִרְפּוּ אֶת כָּל הַשְּׁרִירִים הַתְּפוּסִים.”?
ד. מה השתנה בין “הוּא פָּשׁוּט לָקַח בַּד מַגֶּבֶת גָּדוֹל, עָטַף אֶת עַצְמוֹ בִּמְהִירוּת שֶׁיִּהְיֶה מְכֻסֶּה וְצָנוּעַ, וְרָץ...” לבין “הָרַב יָכוֹל הָיָה לְהִתְלַבֵּשׁ, וַאֲנַחְנוּ הָיִינוּ רָצִים לְהָבִיא לוֹ אֶת מָה שֶׁהוּא צָרִיךְ!".”?
ה. כיצד הגיע הסיפור אל האירוע: “כְּשֶׁנִּכְנַס סוֹף סוֹף פְּנִימָה, נֶעֶצְרָה נְשִׁימָתוֹ שֶׁל הַנֶּכֶד, וְשֶׁל כָּל יַלְדֵי הַכִּתָּה.”?

3. מה היה ראשון?

כתבו ליד כל אירוע מספר מ־1 עד 5 לפי סדר התרחשותו בסיפור.

□ בִּשְׁנִיָּה אַחַת, הוּא זִנֵּק מִתּוֹךְ בְּרֵכַת הַמַּיִם הַחַמִּים!
□ פְּתִיחָה: רֵיחוֹת שֶׁל עֶרֶב שַׁבָּת וְקוֹלוֹת שֶׁל תּוֹרָה

לפני ואחרי

מה קרה לפני: “בִּשְׁנִיָּה אַחַת, הוּא זִנֵּק מִתּוֹךְ בְּרֵכַת הַמַּיִם הַחַמִּים!”?
מה קרה אחרי: “הוּא הִבִּיט בְּתַלְמִידָיו, וְהַפָּנִים שֶׁלּוֹ אָרְוּ בְּאוֹר קָדוֹשׁ.”?

סיבה ותוצאה מתוך המעשה

האירוע הנתוןמה גרם לו?מה הייתה תוצאתו?
וְהִנֵּה, בְּיוֹם שִׁשִּׁי אֶחָד, הָיָה רַבִּי יְהוֹשֻׁעַ עָיֵף בִּמְיֻחָד.
רַב יְהוּדָה וְרַבִּי חִיָּיא בַּר אַבָּא, שְׁנַיִם מִתַּלְמִידָיו הַגְּדוֹלִים, רָאוּ אֶת רַבָּם בָּרְחוֹב וְלֹא הֶאֱמִינוּ לְמַרְאֵה עֵינֵיהֶם.

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “יצא מהמרחץ כדי ללמוד עם נכדו” על ר' יהושע בן לוי? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “אֲנִי אֶסְתַּבֵּן, אֶתְנַגֵּב, אֶתְלַבֵּשׁ בִּבְגָדַי הַמְּכֻבָּדִים, אֶצְעַד הַבַּיְתָה בְּנַחַת, וְאָז אֶשְׁמַע אֶת הַפָּרָשָׁה.”.
ג. אילו ר' יהושע בן לוי לא היה פועל כפי שמתואר ב־“הָרַב יָכוֹל הָיָה לְהִתְלַבֵּשׁ, וַאֲנַחְנוּ הָיִינוּ רָצִים לְהָבִיא לוֹ אֶת מָה שֶׁהוּא צָרִיךְ!".” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “כְּשֶׁנִּכְנַס סוֹף סוֹף פְּנִימָה, נֶעֶצְרָה נְשִׁימָתוֹ שֶׁל הַנֶּכֶד, וְשֶׁל כָּל יַלְדֵי הַכִּתָּה.” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. פְּתִיחָה: רֵיחוֹת שֶׁל עֶרֶב שַׁבָּת וְקוֹלוֹת שֶׁל תּוֹרָה
2. הַסַּבָּא הַגָּדוֹל, שֶׁכָּל הָרַבָּנִים רָעֲדוּ מִפָּנָיו, הָיָה…
3. אָדָם רָגִיל שֶׁנִּמְצָא בַּמֶּרְחָץ הֶחָם כְּשֶׁבַּחוּץ נוֹשֶׁבֶת…
4. הָרוּחַ הַקָּרָה שֶׁל יוֹם הַשִּׁשִּׁי הִכְּתָה עַל גּוּפוֹ שֶׁנָּטַף…
5. הֲלֹא דָּוִד הַמֶּלֶךְ אָמַר: 'וְהוֹדַעְתָּם לְבָנֶיךָ וְלִבְנֵי…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “מַיִם רְתוּחִים נִשְׁפְּכוּ בְּקוֹל אִוְשָׁה וְהִרְפּוּ אֶת כָּל הַשְּׁרִירִים הַתְּפוּסִים.”.

דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “רַבִּי יְהוֹשֻׁעַ הֵרִים אֶת יָדוֹ כְּלַפֵּי הַשָּׁמַיִם וּזְקָנוֹ הַלָּבָן זָע בָּרוּחַ:”.
דיווח: כתבו ידיעה קצרה שכותרתה “יצא מהמרחץ כדי ללמוד עם נכדו”.
