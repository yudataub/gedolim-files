סיפור 87 • ר' שמעון בן חלפתא • השדה שנתן לו ר' חייא

השדה שנתן לו ר' חייא

דף עבודה בעקבות ר' שמעון בן חלפתא
שם: כיתה:

המעשה — הסיפור המלא

פְּתִיחָה: קוֹלוֹת שֶׁל תּוֹרָה מוּל הַכִּנֶּרֶת

יְלָדִים, עִצְמוּ רֶגַע אֶת הָעֵינַיִם. דַּמְיְנוּ שֶׁאַתֶּם עוֹמְדִים עַל שְׂפַת יָם הַכִּנֶּרֶת, לִפְנֵי כְּאַלְפַּיִם שָׁנָה. הַגַּלִּים הַכְּחֻלִּים מִתְנַפְּצִים בְּרַכּוּת אַל הַחוֹף, רֵיחַ מָתוֹק שֶׁל מַיִם מְתוּקִים וְאֲדָמָה רְטֻבָּה נִשָּׂא בָּאֲוִיר. שָׁם, בְּעִיר טְבֶרְיָה הַקְּדוֹשָׁה, יָשְׁבָה חֲבוּרָה שֶׁל תַּלְמִידֵי חֲכָמִים עֲנָקִיִּים.

בְּתוֹךְ בֵּית הַמִּדְרָשׁ הַמּוּאָר, יָשְׁבוּ שְׁנֵי חֲבֵרִים טוֹבִים, עֲנָקֵי הָרוּחַ – רַבִּי חִיָּיא הַגָּדוֹל, וְרַבִּי שִׁמְעוֹן בֶּן חֲלַפְתָּא. קוֹלָם הִתְנַגֵּן בְּשִׁירָה עֲרֵבָה שֶׁל תַּלְמוּד, קוֹל שֶׁחָדַר יָשָׁר אֶל הַלֵּב.

רַבִּי חִיָּיא הָיָה אָדָם שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא בֵּרַךְ אוֹתוֹ בְּפַרְנָסָה טוֹבָה. הָיוּ לוֹ שָׂדוֹת רְחָבִים מְלֵאִים בִּתְבוּאָה. מִנֶּגֶד, חֲבֵרוֹ הַטּוֹב רַבִּי שִׁמְעוֹן בֶּן חֲלַפְתָּא, הָיָה עָנִי מָרוּד. מַמָּשׁ כָּךְ. לְעִתִּים קְרוֹבוֹת הָיָה מַגִּיעַ לְבֵית הַמִּדְרָשׁ כַּאֲשֶׁר בִּגְדּוֹ קָרוּעַ וּמְתֻקָּן בְּחוּטִים פְּשׁוּטִים, וּפָנָיו הָיוּ חִוְּרוֹת מֵרָעָב. אַךְ מֵעוֹלָם, מֵעוֹלָם לֹא שָׁמַע אִישׁ אֶת רַבִּי שִׁמְעוֹן מִתְלוֹנֵן! הוּא תָּמִיד חִיֵּךְ, לָמַד תּוֹרָה בְּשִׂמְחָה, וְהִרְגִּישׁ שֶׁהוּא הָאָדָם הֶעָשִׁיר בָּעוֹלָם כִּי יֵשׁ לוֹ אֶת תּוֹרַת ה'.

הַהַצָּעָה הַחֲכָמָה שֶׁל רַבִּי חִיָּיא

אַךְ רַבִּי חִיָּיא הָיָה חָבֵר אֲמִתִּי. חָבֵר אֲמִתִּי רוֹאֶה גַּם מָה שֶׁלֹּא אוֹמְרִים לוֹ. הוּא רָאָה אֶת קְצָוֹת אֶצְבְּעוֹתָיו הָרוֹעֲדוֹת שֶׁל רַבִּי שִׁמְעוֹן שֶׁלֹּא אָכַל לֶחֶם חַם כְּבָר יָמִים.

"אֲנִי חַיָּב לַעֲזֹר לְאָחִי הַיָּקָר," חָשַׁב רַבִּי חִיָּיא בְּלִבּוֹ. "אֲבָל אִם סְתָם אֶתֵּן לוֹ צְדָקָה בַּיָּד, מַטְבְּעוֹת שֶׁל כֶּסֶף – הוּא יִתְבַּיֵּשׁ מְאֹד. צַדִּיק כָּמוֹהוּ לֹא יִרְצֶה לְקַבֵּל מַתְּנוֹת חִנָּם מִבְּלִי לַעֲבֹד."

מָה עָשָׂה חָכָם גָּדוֹל כְּרַבִּי חִיָּיא? הוּא נִגַּשׁ לְרַבִּי שִׁמְעוֹן וְאָמַר לוֹ כָּךְ:

"רַבִּי שִׁמְעוֹן, יֵשׁ לִי חֶלְקַת שָׂדֶה גְּדוֹלָה וְטוֹבָה כָּאן קָרוֹב, בְּבִקְעַת גִּנּוֹסַר. תִּרְאֶה, לִי אֵין הַרְבֵּה זְמַן לְטַפֵּל בָּהּ. מָה דַּעְתְּךָ שֶׁתִּהְיֶה 'אָרִיס' שֶׁלִּי? אַתָּה תַּעֲבֹד בַּשָּׂדֶה, תַּחֲרֹשׁ וְתִקְצֹר, וּכְשֶׁתִּגְדַּל הַחִטָּה – נִתְחַלֵּק בָּרֶוַח!"

יְלָדִים, אֵיזוֹ דֶּרֶךְ נִפְלָאָה לָתֵת! כָּךְ רַבִּי שִׁמְעוֹן בֶּן חֲלַפְתָּא הִרְגִּישׁ מְכֻבָּד. הוּא לֹא בַּקְּשָׁן. הוּא פּוֹעֵל! עֵינָיו אוֹרוּ, וְהוּא הִסְכִּים מִיָּד. "יְגִיעַ כַּפֶּיךָ כִּי תֹאכֵל אַשְׁרֶיךָ וְטוֹב לָךְ," לָחַשׁ.

עֲבוֹדַת הַשָּׂדֶה וְהַשֶּׁמֶשׁ הַקּוֹפַחַת

מֵאוֹתוֹ יוֹם, חַיָּיו שֶׁל רַבִּי שִׁמְעוֹן הִשְׁתַּנּוּ. בַּבֹּקֶר הָיָה לוֹמֵד בְּהִתְמָדָה גְּדוֹלָה עִם תַּלְמִידָיו, וּבְשָׁעוֹת הָאַחַר-הַצָּהֳרַיִם הָיָה יוֹרֵד לַשָּׂדֶה.

דַּמְיְנוּ אֶת הַתַּנָּא הַגָּדוֹל, קוֹשֵׁר אֶת גְּלִימָתוֹ עַל מָתְנָיו, לוֹקֵחַ מַעֲדֵר וּמַחְרֵשָׁה, וְעוֹבֵד אֶת הָאֲדָמָה! אֲגַלֵּי זֵעָה חַמִּים נָטְפוּ עַל מִצְחוֹ בַּשֶּׁמֶשׁ הַקּוֹפַחַת שֶׁל טְבֶרְיָה. הוּא חָרַשׁ כָּל שׁוּרָה, פִּזֵּר אֶת הַזְּרָעִים, וְנָשַׁם אֶת הָרֵיחַ שֶׁל אֲדָמָה וְטַל. "רִבּוֹנוֹ שֶׁל עוֹלָם," מִלְמֵל אֶל הַשָּׁמַיִם, "עֲשֵׂה שֶׁתִּגְדַּל הַחִטָּה, כְּדֵי שֶׁאוּכַל לְהָבִיא לֶחֶם קָדוֹשׁ לִבְנֵי מִשְׁפַּחְתִּי לִכְבוֹד הַשַּׁבָּת."

וְהַשָּׂדֶה... גָּדַל! מִזְּרָעִים קְטַנִּים צָמְחוּ שִׁבֳּלִים דַּקּוֹת שֶׁהָפְכוּ לְיָם זָהוּב וּמַבְרִיק שֶׁל תְּבוּאָה הַמִּתְנוֹדֶדֶת בָּרוּחַ.

הִגִּיעַ יְמֵי הַקָּצִיר. רַבִּי שִׁמְעוֹן קָצַר בְּמַגָּל אֶת הַחִטָּה בְּשִׂמְחָה רַבָּה. הוּא אָסַף אֶת כָּל הַשִּׁבּוֹלִים לַ'גֹּרֶן' – מִשְׁטָח עָגֹל וְגָדוֹל בְּקָצֶה הַשָּׂדֶה. הוּא דָּשׁ אֶת הַחִטָּה וּמָעַךְ אוֹתָהּ עַד שֶׁנּוֹצְרָה עֲרֵמָה גְּדוֹלָה.

אֲבָל עַכְשָׁו, הָיְתָה לוֹ בְּעָיָה חֲדָשָׁה.

הַהַמְתָּנָה לָרוּחַ

יְלָדִים, אַתֶּם יוֹדְעִים שֶׁגַּרְעִין הַחִטָּה בָּא עִם "קְלִפָּה" קְטַנָּה וִיבֵשָׁה? לַקְּלִפָּה הַזּוֹ קוֹרְאִים מוֹץ . אִי אֶפְשָׁר לֶאֱכֹל מוֹץ, זֶה כְּמוֹ חול! חַיָּבִים לְהַפְרִיד בֵּין קֶמַח לַמּוֹץ.

אֵיךְ מַפְרִידִים עֲרֵמָה כָּל כָּךְ גְּדוֹלָה לִפְנֵי שְׁנוֹת אַלְפַּיִם בְּלִי מְכוֹנוֹת עֲנָקִיּוֹת?

קָרְאוּ לְזֶה "לִזְרוֹת בָּרוּחַ". הַחַקְלַאי הָיָה לוֹקֵחַ מַזְלֵג עֵץ עֲנָק ('רַחַת'), מֵרִים הַרְבֵּה חִטָּה לָאֲוִיר, וְזוֹרֵק לְמַעְלָה! גַּרְעִינֵי הַחִטָּה הֵם כְּבֵדִים – שׁוֹקְעִים יָשָׁר לְמַטָּה, אֲבָל הַמּוֹץ (הַקְּלִפּוֹת) הוּא מַמָּשׁ קַל – הָרוּחַ תּוֹפֶסֶת אוֹתוֹ וּמְעִיפָה אוֹתוֹ הַרְחֵק-הַרְחֵק.

רַבִּי שִׁמְעוֹן לָקַח אֶת כְּלִי הָעֵץ הֶעָבֶה בְּיָדָיו, הֵרִים עֲרֵמָה שֶׁל חִטָּה, וְחִכָּה לָרוּחַ.

אַךְ מָה קָרָה? דְּמָמָה.

שֶׁקֶט מֻחְלָט. אוֹתוֹ יוֹם הָיָה חַם מְאֹד. אֲפִלּוּ עָלֶה אֶחָד בָּעֵצִים לֹא זָז. יַתּוּשׁ יָכוֹל הָיָה לָעוּף בְּלִי קֹשִׁי. לֹא הָיְתָה טִפָּה שֶׁל אֲוִיר קַר!

הוּא חִכָּה שָׁעָה. חִכָּה שְׁעָתַיִם. יוֹם תָּמִים! הוּא לֹא יָכוֹל לָקַחַת חִטָּה הַבַּיְתָה כָּךְ, הַכֹּל מָלֵא בְּפָסוֹלֶת. מָה יֹאכְלוּ יְלָדָיו בְּשַׁבָּת?!

הֵרִים רַבִּי שִׁמְעוֹן אֶת עֵינָיו לַשָּׁמַיִם, וְדִבֵּר מִתּוֹךְ הַלֵּב, מַמָּשׁ כְּמוֹ בֵּן מְדַבֵּר לְאָבִיו:

"רִבּוֹנוֹ שֶׁל עוֹלָם! אַתָּה בּוֹרֵא הָרוּחַ. תִּרְאֶה אֵיךְ עָמַלְתִּי, תִּרְאֶה אֵיךְ שָׁמַרְתִּי שֶׁכֹּל הָעֲבוֹדָה תִּהְיֶה יְשָׁרָה וּכְשֵׁרָה. אָנָּא, שְׁלַח לִי בְּרָכָה, שְׁלַח רוּחַ טוֹבָה כְּדֵי שֶׁאוּכַל לַהֲכִין לֶחֶם לַיְּלָדִים!"

הַנֵּס... וְהַנֶּזֶק!

וְהִנֵּה... כְּמוֹ קֶסֶם – הַשָּׁמַיִם הִתְמַלְּאוּ רוּחַ פִּתְאוֹמִית וַחֲזָקָה! עַנְפֵי הָעֵצִים הֵחֵלּוּ לְהִשְׁתַּחֲווֹת. מַיִם הֵחֵלּוּ לִגְעֹשׁ בַּכִּנֶּרֶת. שָׂמַח רַבִּי שִׁמְעוֹן שִׂמְחָה עֲצוּמָה! הַקָּדוֹשׁ בָּרוּךְ הוּא שָׁמַע אֶת תְּפִלָּתוֹ!

הוּא הִכְנִיס אֶת מַזְלֵג הָעֵץ הָרָחָב לְתוֹךְ עֲרֵמַת הַחִטָּה הַקָּשָׁה, זָרַק אוֹתָהּ בְּכֹחַ הַרְבֵּה לְמַעְלָה... "וּוּשְּׁשְּׁשְּׁ!" הָרוּחַ נָשְׁבָה, גַּרְעִינֵי הַחִטָּה הַנְּקִיִּים צָנְחוּ לַאֲדָמָה כְּמוֹ אוֹצָר מְלֹא כֶּסֶף, וְהַמּוֹץ הַקַּל עָף לַשָּׁמַיִם וְהִתְרַחֵק...

אֲבָל... לְפֶתַע עָצַר רַבִּי שִׁמְעוֹן.

הַחִיּוּךְ נִמְחַק מִפָּנָיו.

הוּא עָמַד עַל קְצוֹת אַצְבְּעוֹתָיו וְהִבִּיט לְאָן בְּדִיּוּק הָרוּחַ מְעִיפָה אֶת הַמּוֹץ? לְאָן טָסוֹת הַקְּלִפּוֹת וְכָל הָעָבָק?

וְאָז זִנֵּק לִבּוֹ מִפַּחַד. הָרוּחַ הָיְתָה כָּל כָּךְ חֲזָקָה, עַד שֶׁהִיא לָקְחָה אֶת כָּל הַפְּסֹלֶת מֵהַשָּׂדֶה שֶׁל רַבִּי שִׁמְעוֹן... וְהִשְׁלִיכָה אוֹתָהּ יָשָׁר לְתוֹךְ הַשָּׂדֶה שֶׁל הַשָּׁכֵן שֶׁלּוֹ! הַשָּׁכֵן בְּדִיּוּק סִיֵּם לְנַקּוֹת אֶת הַשָּׂדֶה שֶׁלּוֹ מִפְּסוֹלֶת, וְעַכְשָׁו, בִּגְלַל הָרוּחַ, הַשָּׂדֶה שֶׁלּוֹ שׁוּב מִתְמַלֵּא בְּקוֹצִים וּמוֹץ!

יְלָדִים, אָנָּא חִשְׁבוּ – מָה אָדָם רָגִיל הָיָה עוֹשֶׂה עַכְשָׁו? הָיָה אָדָם רָגִיל מַמְשִׁיךְ וְאוֹמֵר: "מָה אִכְפַּת לִי! הַשֵּׁם עָשָׂה נֵס! עַכְשָׁו יֵשׁ לִי רוּחַ! אִם הָרוּחַ הוֹלֶכֶת לַשָּׁכֵן, זֶה בְּעָיָה שֶׁל הַשֵּׁם, לֹא שֶׁלִּי! אֲנִי חַיָּב לְסַיֵּם כְּדֵי לְהָבִיא אוֹכֶל לַיְלָדִים הָרְעֵבִים שֶׁלִּי בַּבַּיִת!"

אֲבָל רַבִּי שִׁמְעוֹן לֹא עָשָׂה כָּךְ.

צַדִּיק שֶׁל אֱמֶת, יְלָדִים, אַף פַּעַם לֹא יַסְכִּים לְהַרְוִיחַ מַשֶּׁהוּ שֶׁיִּגְרֹם כְּאֵב אוֹ נֶזֶק לְיְהוּדִי אַחֵר!

מִיָּד שָׁמַט רַבִּי שִׁמְעוֹן אֶת הַכְּלִי מִיָּדָיו. "לֹא יָקוּם וְלֹא יִהְיֶה!" צָעַק לְעַצְמוֹ בַּשָּׂדֶה. "אָסוּר לִי לְהַזִּיק לַשָּׁכֵן שֶׁלִּי! אִם הַפְּסֹלֶת שֶׁלִּי כּוֹסְתָה שָׂדֶה שֶׁל מִישֶׁהוּ אַחֵר וְהוֹרֶסֶת אֶת עֲבוֹדָתוֹ – אֲנִי מַפְסִיק לַעֲבֹד. אֲנִי מַעֲדִיף לִהְיוֹת עָנִי עוֹד יוֹם, רַק שֶׁלֹּא נְגַלְגֵּל נֶזֶק לְאָדָם אַחֵר."

הַתְּפִלָּה הַשְּׁנִיָּה מֵהַלֵּב הַטָּהוֹר

עָצַם רַבִּי שִׁמְעוֹן עֵינַיִם קָשׁוֹת כְּשֶׁלִּבּוֹ נִשְׁבָּר. עֲבוֹדָה קָשָׁה כָּל כָּךְ, נֵס שֶׁהִגִּיעַ, וְעַכְשָׁו אִי אֶפְשָׁר לְהִשְׁתַּמֵּשׁ בּוֹ.

הוּא נָשָׂא כַּפָּיו הַלְּמַעְלָה וְאָמַר כְּשֶׁהוּא בּוֹכֶה בְּשֶׁקֶט:

"רִבּוֹנוֹ שֶׁל עוֹלָם! אַתָּה נָתַתָּ לִי בְּמַתָּנָה אֶת הָרוּחַ הַזֹּאת. אֲנִי יוֹדֵעַ שֶׁשָּׁמַעְתָּ לִתְפִלָּתִי. אֲבָל... אִם הָרוּחַ הַזּוֹ עָפָה לַשָּׂדֶה שֶׁל חֲבֵרִי, וּמְקַלְקֶלֶת אוֹתָהּ, בְּבַקָּשָׁה – קַח אֶת הָרוּחַ בַּחֲזָרָה! לֹא רָאוּי שֶׁבְּנֵי מִשְׁפַּחְתִּי יֹאכְלוּ לֶחֶם בְּעֵת שֶׁחֲבֵרִי בּוֹכֶה עַל שָׂדֵהוּ!"

דְּמָמָה נָפְלָה בַּשָּׂדֶה. הָרוּחַ פָּסְקָה לְגַמְרֵי. רַבִּי שִׁמְעוֹן עָמַד שָׁם, עוֹצֵם מַבָּטוֹ, רָגוּעַ בְּהַחְלָטָתוֹ הַקְּדוֹשָׁה.

אַךְ יְלָדִים, הַקָּדוֹשׁ בָּרוּךְ הוּא אֵינוֹ אוֹהֵב לִרְאוֹת אֶת צַדִּיקָיו עֲצוּבִים, עַל אַחַת כַּמָּה וְכַמָּה צַדִּיק כָּל כָּךְ יָשָׁר שֶׁנִּזְהָר בִּרְכוּשָׁם שֶׁל אֲחֵרִים!

פִּתְאוֹם... שָׁרְקָה רוּחַ קַלָּה.

אַךְ הַפַּעַם, הִיא לֹא נָשְׁבָה מִן הַמַּעֲרָב בְּכִוּוּן שְׂדֵה הַשָּׁכֵן. מַמָּשׁ בְּאֹרַח פֶּלֶא מֵעַל הַטֶּבַע, הִשְׁתַּנָּה כִּוּוּן הָרוּחַ לְגַמְרֵי! הָרוּחַ מִן הַדָּרוֹם הֵחֵלָּה לְנַשֵּׁב בְּעָצְמָה קָרָה וְרַעֲנַנָּה... וְהִיא כִּוְּנָה דֶּרֶךְ שְׂדֵהוּ שֶׁל רַבִּי שִׁמְעוֹן, יָשָׁר-יָשָׁר... הַרְחֵק לְתוֹךְ מֵי הַכִּנֶּרֶת! מָקוֹם שֶׁאֵין בּוֹ אַף אָדָם וְלֹא נִגְרָם בּוֹ נֶזֶק לְאִישׁ!

חִיּוּךְ רָחָב וְזוֹהֵר נִפְרַשׂ עַל פָּנָיו שֶׁל רַבִּי שִׁמְעוֹן. "אָה... אָבִי שֶׁבַּשָּׁמַיִם," לָחַשׁ בְּהוֹדָיָה. הָרוּחַ הָיְתָה מֻשְׁלֶמֶת! הוּא לָקַח מַהֵר אֶת הָרַחַת, הֵנִיף אֶת עֲרֵמוֹת הַתְּבוּאָה, וְכָל הַמּוֹץ וְהַחול הָעוּ כְּעָנָן יָשָׁר לְתוֹךְ הַיָּם.

בְּאוֹתוֹ עֶרֶב, נִכְנַס רַבִּי שִׁמְעוֹן לְבֵיתוֹ כְּשֶׁבְּיָדוֹ קֶמַח לָבָן וְטָהוֹר, שֶׁהֻשַּׂג בְּלִי סַפָק שֶׁל נֶזֶק, מִתּוֹךְ כָּבוֹד, יֹשֶׁר – וְנֵס עָצוּם שֶׁל אֱמוּנָה!

מקורות: מדרש רות רבה (פרשה ה', ז'): "רבי שמעון בן חלפתא ורבי חייא... נתן לו רבי חייא שדה לחכור... לא היתה לו רוח לזרות... התפלל...". (ויש מדרשים דומים בקהלת רבה ושיר השירים רבה). ספר "כה עשו חכמינו" , חלק ה', עמ' 103 ("השדה שנתן לו ר' חייא"). העיקרון מבוסס על נאמנותו, הרחקתו מגזל ואהבת התורה שהתבטא במאמריו של ר' שמעון בן חלפתא (כולל היותו עני גדול שה' מפרנסו בנסים, כמו באגדת "המרגלית מן השמיים").

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “דַּמְיְנוּ שֶׁאַתֶּם עוֹמְדִים עַל שְׂפַת יָם הַכִּנֶּרֶת, לִפְנֵי כְּאַלְפַּיִם שָׁנָה.” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “אֲנִי חַיָּב לַעֲזֹר לְאָחִי הַיָּקָר," חָשַׁב רַבִּי חִיָּיא בְּלִבּוֹ.”?
ג. כיצד הגיב ר' שמעון בן חלפתא למצב המתואר: “רִבּוֹנוֹ שֶׁל עוֹלָם," מִלְמֵל אֶל הַשָּׁמַיִם, "עֲשֵׂה שֶׁתִּגְדַּל הַחִטָּה, כְּדֵי שֶׁאוּכַל לְהָבִיא לֶחֶם קָדוֹשׁ לִבְנֵי מִשְׁפַּחְתִּי לִכְבוֹד…”?
ד. מה השתנה בין “אָנָּא, שְׁלַח לִי בְּרָכָה, שְׁלַח רוּחַ טוֹבָה כְּדֵי שֶׁאוּכַל לַהֲכִין לֶחֶם לַיְּלָדִים!” לבין “צַדִּיק שֶׁל אֱמֶת, יְלָדִים, אַף פַּעַם לֹא יַסְכִּים לְהַרְוִיחַ מַשֶּׁהוּ שֶׁיִּגְרֹם כְּאֵב אוֹ נֶזֶק לְיְהוּדִי…”?
ה. כיצד הגיע הסיפור אל האירוע: “הָרוּחַ מִן הַדָּרוֹם הֵחֵלָּה לְנַשֵּׁב בְּעָצְמָה קָרָה וְרַעֲנַנָּה...”?

לפני ואחרי

מה קרה לפני: “הוּא לֹא יָכוֹל לָקַחַת חִטָּה הַבַּיְתָה כָּךְ, הַכֹּל מָלֵא בְּפָסוֹלֶת.”?
מה קרה אחרי: “אֲנִי מַעֲדִיף לִהְיוֹת עָנִי עוֹד יוֹם, רַק שֶׁלֹּא נְגַלְגֵּל נֶזֶק לְאָדָם אַחֵר.”?

האירוע הנתוןמה גרם לו?מה הייתה תוצאתו?
יְגִיעַ כַּפֶּיךָ כִּי תֹאכֵל אַשְׁרֶיךָ וְטוֹב לָךְ," לָחַשׁ.
וְהִשְׁלִיכָה אוֹתָהּ יָשָׁר לְתוֹךְ הַשָּׂדֶה שֶׁל הַשָּׁכֵן שֶׁלּוֹ!

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “השדה שנתן לו ר' חייא” על ר' שמעון בן חלפתא? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “הַחַקְלַאי הָיָה לוֹקֵחַ מַזְלֵג עֵץ עֲנָק ('רַחַת'), מֵרִים הַרְבֵּה חִטָּה לָאֲוִיר, וְזוֹרֵק לְמַעְלָה!”.
ג. אילו ר' שמעון בן חלפתא לא היה פועל כפי שמתואר ב־“צַדִּיק שֶׁל אֱמֶת, יְלָדִים, אַף פַּעַם לֹא יַסְכִּים לְהַרְוִיחַ מַשֶּׁהוּ שֶׁיִּגְרֹם כְּאֵב אוֹ נֶזֶק לְיְהוּדִי אַחֵר!” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “הָרוּחַ מִן הַדָּרוֹם הֵחֵלָּה לְנַשֵּׁב בְּעָצְמָה קָרָה וְרַעֲנַנָּה...” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. דַּמְיְנוּ שֶׁאַתֶּם עוֹמְדִים עַל שְׂפַת יָם הַכִּנֶּרֶת, לִפְנֵי…
2. אֲבָל אִם סְתָם אֶתֵּן לוֹ צְדָקָה בַּיָּד, מַטְבְּעוֹת שֶׁל כֶּסֶף –…
3. רַבִּי שִׁמְעוֹן קָצַר בְּמַגָּל אֶת הַחִטָּה בְּשִׂמְחָה רַבָּה.
4. וּוּשְּׁשְּׁשְּׁ!" הָרוּחַ נָשְׁבָה, גַּרְעִינֵי הַחִטָּה הַנְּקִיִּים…
5. הוּא נָשָׂא כַּפָּיו הַלְּמַעְלָה וְאָמַר כְּשֶׁהוּא בּוֹכֶה בְּשֶׁקֶט:
6. בְּאוֹתוֹ עֶרֶב, נִכְנַס רַבִּי שִׁמְעוֹן לְבֵיתוֹ כְּשֶׁבְּיָדוֹ קֶמַח…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “רִבּוֹנוֹ שֶׁל עוֹלָם," מִלְמֵל אֶל הַשָּׁמַיִם, "עֲשֵׂה שֶׁתִּגְדַּל הַחִטָּה, כְּדֵי שֶׁאוּכַל לְהָבִיא לֶחֶם קָדוֹשׁ לִבְנֵי…”.
מכתב: כתבו לר' שמעון בן חלפתא בעקבות “וְהִשְׁלִיכָה אוֹתָהּ יָשָׁר לְתוֹךְ הַשָּׂדֶה שֶׁל הַשָּׁכֵן שֶׁלּוֹ!”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “אִם הָרוּחַ הַזּוֹ עָפָה לַשָּׂדֶה שֶׁל חֲבֵרִי, וּמְקַלְקֶלֶת אוֹתָהּ, בְּבַקָּשָׁה – קַח אֶת הָרוּחַ בַּחֲזָרָה!”.
דיווח: כתבו ידיעה קצרה שכותרתה “השדה שנתן לו ר' חייא”.

המחזה

רגע מן הסיפורהדמויותמה יאמרו או יעשו?
יְגִיעַ כַּפֶּיךָ כִּי תֹאכֵל אַשְׁרֶיךָ וְטוֹב לָךְ," לָחַשׁ.
צַדִּיק שֶׁל אֱמֶת, יְלָדִים, אַף פַּעַם לֹא יַסְכִּים לְהַרְוִיחַ מַשֶּׁהוּ שֶׁיִּגְרֹם כְּאֵב אוֹ…

7. חיבור לחיים מתוך המעשה

א. מה אפשר ללמוד מן הדרך שבה ר' שמעון בן חלפתא פעל באירוע “אָנָּא, שְׁלַח לִי בְּרָכָה, שְׁלַח רוּחַ טוֹבָה כְּדֵי שֶׁאוּכַל לַהֲכִין לֶחֶם לַיְּלָדִים!”?
ב. תארו מצב מחיי הכיתה שבו אפשר ליישם את המסר של “השדה שנתן לו ר' חייא”.
