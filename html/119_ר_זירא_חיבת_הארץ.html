חיבת הארץ

דף עבודה בעקבות ר' זירא
שם: כיתה:

המעשה — הסיפור המלא

פְּתִיחָה: הַקּוֹלוֹת שֶׁבְּבָבֶל וְהַלֵּב שֶׁבִּירוּשָׁלַיִם

יְלָדִים, אִם הֱיִיתֶם נִכְנָסִים לְבֵית הַמִּדְרָשׁ הַגָּדוֹל בְּבָבֶל לִפְנֵי כְּאַלְפַּיִם שָׁנָה, הֱיִיתֶם עוֹמְדִים בְּהִשְׁתָּאוּת. מֵאוֹת תַּלְמִידֵי חֲכָמִים יָשְׁבוּ שָׁם, וְקוֹל לִמּוּד הַתּוֹרָה הִדְהֵד כְּמוֹ רַעַם אַדִּיר שֶׁל מַיִם רַבִּים. בָּבֶל הָיְתָה עִיר עֲשִׁירָה מְאֹד. הָיוּ בָּהּ עֵצִים יְרֻקִּים רַבִּים, שְׁוָקִים עֲמוּסִים בְּתַבְלִינִים שֶׁרֵיחָם הַמָּתוֹק מִלֵּא אֶת הָאֲוִיר, וִישִׁיבוֹת שֶׁבָּהֶן לָמְדוּ תּוֹרָה יוֹמָם וָלַיְלָה.

בְּשׁוּרָה הָרִאשׁוֹנָה שֶׁל בֵּית הַמִּדְרָשׁ יָשַׁב אֶחָד מִגְּדוֹלֵי הַדּוֹר – הָאָמוֹרָא הַקָּדוֹשׁ רַבִּי זֵירָא .

רַבִּי זֵירָא הָיָה עֲנָק בַּתּוֹרָה. בְּכָל פַּעַם שֶׁפָּתַח אֶת פִּיו, הָיָה מֵאִיר אֶת הַהֲלָכָה, וְכֻלָּם הִקְשִׁיבוּ לוֹ בְּיִרְאַת שָׁמַיִם. אַךְ מִי שֶׁהִבִּיט הֵיטֵב בְּעֵינָיו שֶׁל רַבִּי זֵירָא, הָיָה יָכוֹל לִרְאוֹת שֶׁמַּשֶּׁהוּ חָסֵר לוֹ. לְעִתִּים, בְּאֶמְצַע שִׁעוּר, הָיָה רַבִּי זֵירָא עוֹצֵם אֶת עֵינָיו, מַפְנֶה אֶת פָּנָיו שְׂמֹאלָה – לְכִוּוּן מַעֲרָב, וּדְמָעוֹת שֶׁקּוֹפוֹת הָיוּ זולְגוֹת מֵעֵינָיו אֶל זְקָנוֹ הַטָּהוֹר.

"רַבֵּנוּ," שָׁאֲלוּ אוֹתוֹ הַתַּלְמִידִים בְּלַחַשׁ, "מַדּוּעַ אַתָּה בּוֹכֶה? הֲרֵי יֵשׁ לָנוּ כָּאן יְשִׁיבָה גְּדוֹלָה, תּוֹרָה וְחַיִּים טוֹבִים!".

רַבִּי זֵירָא נֶאֱנַח אֲנָחָה אֲרֻכָּה שֶׁבָּאָה מֵעֹמֶק נִשְׁמָתוֹ: "גּוּפִי נִמְצָא כָּאן בְּבָבֶל," הוּא לָחַשׁ, "אֲבָל נִשְׁמָתִי – נִשְׁמָתִי נִמְצֵאת שָׁם. בְּאֶרֶץ יִשְׂרָאֵל! אֲנִי מַרְגִּישׁ אֶת אֲוִיר הָאָרֶץ הַקְּדוֹשָׁה קוֹרֵא לִי. אֲנִי רוֹצֶה לָלֶכֶת עַל הָאֲדָמָה שֶׁבָּהּ צָעֲדוּ הַנְּבִיאִים! אֲנִי רוֹצֶה שֶׁתּוֹרָתִי תִּהְיֶה תּוֹרַת אֶרֶץ יִשְׂרָאֵל!".

אהֲבָתוֹ שֶׁל רַבִּי זֵירָא לְאֶרֶץ יִשְׂרָאֵל בָּעֲרָה כְּלַפִּיד שֶׁל אֵשׁ. הוּא יָדַע שֶׁהַדֶּרֶךְ קָשָׁה וּמְסֻכֶּנֶת, שֶׁיֵּשׁ שׁוֹדְדִים, מִדְבָּרִיּוֹת צְמֵאִים לְמַיִם וְהָרִים גְּבוֹהִים לַחֲצוֹת, אַךְ הַהַשְׁגָּחָה הַפְּרָטִית שֶׁל אוֹהֵב הָאָרֶץ שָׁמְרָה עָלָיו, וְהוּא הֶחְלִיט – הוּא יוֹצֵא לַמַּסָּע!

הַמַּסָּע וְהַנָּהָר הַגּוֹעֵשׁ

חֳדָשִׁים אֲרֻכִּים פָּסַע רַבִּי זֵירָא בַּדְּרָכִים. הַשֶּׁמֶשׁ קָפְחָה עַל רֹאשׁוֹ, רוּחוֹת שֶׁל חוֹל צָרְבוּ אֶת פָּנָיו, רַגְלָיו נִמְלְאוּ בַּאֲבַק מִדְבָּר – אַךְ לִבּוֹ הָיָה שָׂמֵחַ וְשָׁר! "עוֹד מְעַט..." הוּא חָשַׁב, "עוֹד צַעַד אֶחָד, וְאֶהְיֶה בְּאֶרֶץ שֶׁעֵינֵי ה' אֱלֹקֶיךָ בָּהּ!".

כָּל צַעַד שֶׁצָּעַד הָיָה מִצְוָה, כָּל טִפַּת זֵעָה הָיְתָה תְּפִלָּה.

וְהִנֵּה, לְאַחַר תְּלָאוֹת רַבּוֹת, הוּא הִגִּיעַ אֶל הַגְּבוּל. מוּל עֵינָיו נִגְלָה נָהָר אַדִּיר, כַּנִּרְאֶה נְהַר הַיַּרְדֵּן אוֹ אֶחָד מִיְּבָלָיו. הַמַּיִם זָרְמוּ בְּשֶׁצֶף-קֶצֶף, מַכִּים בַּסְּלָעִים וּמַעֲלִים קֶצֶף לָבָן. רַעַשׁ הַמַּיִם הָיָה מַחֲרִישׁ אָזְנַיִם. אֶרֶץ יִשְׂרָאֵל הָיְתָה מַמָּשׁ שָׁם – בַּעֲבָרוֹ הַשֵּׁנִי שֶׁל הַנָּהָר! רַבִּי זֵירָא יָכוֹל הָיָה מַמָּשׁ לְהָרִיחַ אֶת רֵיחַ הֶעָפָר וְהַטַּל שֶׁל אֶרֶץ הַקֹּדֶשׁ.

אֲבָל הָיְתָה בְּעָיָה קָשָׁה. מַעְבּוֹרֶת הָעֵץ הַגְּדוֹלָה, שֶׁעָלֶיהָ הָיוּ עוֹבְרִים אֶת הַנָּהָר בְּבִטְחָה, לֹא הָיְתָה שָׁם! הִיא חָנְתָה עַל הַגָּדָה מִמּוּל, מַמְתִּינָה לַנּוֹסְעִים הַבָּאִים.

עַל שְׂפַת הַנָּהָר עָמְדוּ סוֹחֲרִים וַאֲנָשִׁים רַבִּים. "נִצְטָרֵךְ לְהַמְתִּין מִסְפַּר שָׁעוֹת," שׂוֹחֲחוּ בֵּינֵיהֶם. הֵם הִנִּיחוּ אֶת הַשַּׂקִּים שֶׁלָּהֶם עַל הָאֲדָמָה הַלַּחָה, הִתְיַשְּׁבוּ בְּצֵל הָעֵצִים וְחִכּוּ בְּסַבְלָנוּת.

אַךְ עֲבוּר רַבִּי זֵירָא, לְהַמְתִּין עַכְשָׁו נִרְאָה כְּמוֹ נֶצַח! "אֵיךְ אוּכַל לַעֲמֹד כָּאן וּלְחַכּוֹת?!" סָעֲרָה נַפְשׁוֹ. "אַרְמוֹנוֹ שֶׁל הַמֶּלֶךְ, אֶרֶץ חֶמְדָּה, נִמְצֵאת מֶרְחָק מִסְפַּר מֶטְרִים מִמֶּנִּי – וַאֲנִי אֵשֵׁב וְאָנוּחַ עַל אֲדָמָה שֶׁל חוּץ לָאָרֶץ?!"

הַלַּעַג וְהַחֶבֶל שֶׁל מְסִירוּת הַנֶּפֶשׁ

הַדָּבָר הַיָּחִיד שֶׁחִבֵּר בֵּין שְׁתֵּי הַגָּדוֹת, מִלְּבַד הַמַּעְבּוֹרֶת, הָיָה 'מִצְרָא' – חֶבֶל עָבֶה וּמְחֻסְפָּס מְאֹד, שֶׁהָיָה תָּלוּי גָּבוֹהַּ מֵעַל לַמַּיִם. יְלָדִים, זֶה לֹא הָיָה גֶּשֶׁר בָּטוּחַ שֶׁל קְרָשִׁים, אֶלָּא רַק חֶבֶל, אוֹ מְשׂוּכָה דַּקָּה וְרוֹעֶדֶת מֵעַל לַנָּהָר הָאֵימְתָנִי. מִי שֶׁמִּסְתַּכֵּן לַעֲבֹר עָלָיו וּמַחֲלִיק – נוֹפֵל יְשִׁירוֹת לְמַעֲמַקֵּי הַמַּיִם הַקָּרִים!

לְלֹא הִסּוּס שֶׁל רֶגַע, רַבִּי זֵירָא צָעַד קָדִימָה. הָאָמוֹרָא הַקָּדוֹשׁ, הָרַב הַגָּדוֹל, קָפַץ וְנֶאֱחַז בַּחֶבֶל הֶעָבֶה בִּשְׁתֵּי יָדָיו!

הוּא הִתְחִיל לַחֲצוֹת אֶת הַנָּהָר, תָּלוּי בֵּין שָׁמַיִם וָאָרֶץ. הַחֶבֶל הַמְּחֻסְפָּס שָׂרַט אֶת יָדָיו. הַמַּיִם הַגּוֹעֲשִׁים לְמַטָּה הִשְׁפְּרִיצוּ טִפּוֹת קָרוֹת שֶׁנָּגְעוּ בְּפָנָיו. הָרוּחַ טִלְטְלָה אֶת גּוּפוֹ – אַךְ עֵינָיו נָצְצוּ מֵאֹשֶׁר! "אֶרֶץ יִשְׂרָאֵל!" הָלַם לִבּוֹ עִם כָּל תְּנוּעָה.

לְפֶתַע, מִגְּדַת הַנָּהָר, נִשְׁמַע קוֹל צְחוֹק גָּדוֹל מְאֹד. צְחוֹק רַע וּמְזַלְזֵל.

עָמַד שָׁם מִין, אָדָם רָשָׁע וְצוֹרֵר יִשְׂרָאֵל, שֶׁשָּׂנֵא אֶת לִמּוּד הַתּוֹרָה וְאֶת הַתַּלְמִידֵי חֲכָמִים.

הַמִּין הַזֶּה הִצְבִּיעַ עַל הָרַב הַגָּדוֹל הַתָּלוּי עַל הַחֶבֶל וְצָעַק לְעֶבְרוֹ:

"הָה! עַם פְּזִיז וּמְמַהֵר עַם יִשְׂרָאֵל! עַמָּא פְּזִיזָא!" זָעַק הָרָשָׁע בְּאֲרָמִית. "מֵעוֹלָם הֱיִיתֶם פְּזִיזִים, וַעֲדַיִן פְּזִיזִים אַתֶּם! כְּשֶׁעֲמַדְתֶּם בְּהַר סִינַי, מִהַרְתֶּם בְּלִי לַחְשֹׁב וְאָמַרְתֶּם מִיָּד עַל הַתּוֹרָה 'נַעֲשֶׂה וְנִשְׁמַע', לִפְנֵי שֶׁבִּכְלָל שְׁמַעְתֶּם מָה מְבַקְּשִׁים מִכֶּם! וְגַם עַכְשָׁו! לָמָּה אַתָּה מְמַהֵר כָּל כָּךְ לְהִסְתַּכֵּן עַל הַחֶבֶל הַזֶּה בִּמְקוֹם לַחֲכּוֹת לַסְּפִינָה?! אֵיזֶה עַם נִמְהָר!".

תְּשׁוּבָה מְנַצַּחַת

רַבִּי זֵירָא לֹא נִבְהַל. גּוּפוֹ הָיָה תָּלוּי מֵעַל הַנָּהָר הָאֵימְתָנִי, שְׁרִירָיו כָּאֲבוּ, אַךְ לִבּוֹ הָיָה מָלֵא אֱמוּנָה וַהֲדַר קֹדֶשׁ. הוּא סוֹבֵב אֶת רֹאשׁוֹ הַטָּהוֹר אֶל הָרָשָׁע, הִבִּיט בּוֹ בְּעֵינַיִם מְלֵאוֹת עֹז וֶאֱמוּנָה בַּה', וְעָנָה לוֹ תְּשׁוּבָה שֶׁנֶּחְקְקָה לָנֶצַח בַּתַּלְמוּד בִּמְסֶכֶת כְּתֻבּוֹת:

"רָשָׁע שֶׁכְּמוֹתְךָ!" זָעַק רַבִּי זֵירָא בְּקוֹל שֶׁגָּבַר עַל רַעַשׁ הַגַּלִּים, "מָקוֹם שֶׁמֹּשֶׁה רַבֵּנוּ וְאַהֲרֹן הַכֹּהֵן לֹא זָכוּ לְהִכָּנֵס אֵלָיו! מֹשֶׁה רַבֵּנוּ גְּדוֹל הַנְּבִיאִים הִתְחַנֵּן וְהִתְפַּלֵּל חֲמֵשׁ מֵאוֹת וַחֲמֵשׁ עֶשְׂרֵה תְּפִלּוֹת רַק כְּדֵי לַעֲבֹר אֶת הַיַּרְדֵּן וְלֹא זָכָה. אִם הַקָּדוֹשׁ בָּרוּךְ הוּא נָתַן לִי אֶת הַזְּכוּת הָעֲצוּמָה וְהַנְּדִירָה לִדְרֹךְ בְּאֶרֶץ הַקֹּדֶשׁ – מִי אָמַר שֶׁאִם אַמְתִּין עוֹד שָׁעָה לַסְּפִינָה אֶזְכֶּה לְכָךְ?! אוּלַי אָמוּת אוֹ יִקְרֶה לִי מַשֶּׁהוּ? לְאַרְמוֹנוֹ שֶׁל הַמֶּלֶךְ רָצִים, רָשָׁע, לֹא מַמְתִּינִים!".

וּבְכֹחַ אַחֲרוֹן, מָשַׁךְ אֶת עַצְמוֹ רַבִּי זֵירָא עַל הַחֶבֶל הַדַּק, וְצָנַח בְּשָׁלוֹם עַל גְּדַת הַנָּהָר – בְּתוֹךְ אֶרֶץ יִשְׂרָאֵל הַקְּדוֹשָׁה.

הַמִּין שָׁתַק, הָמוּם מִקְּדֻשָּׁתוֹ וּמִגְּבוּרָתוֹ שֶׁל הַתַּלְמִיד חָכָם.

וְרַבִּי זֵירָא? הוּא לֹא קָם עַל רַגְלָיו בְּגַאֲוָה כְּמִי שֶׁנִּצַּח בְּוִכּוּחַ. לֹא. הָאָמוֹרָא הַגָּדוֹל וְהַקָּדוֹשׁ, הָרַב שֶׁאַלְפֵי תַּלְמִידִים רָעֲדוּ מִפָּנָיו, נָפַל אֶל הָאֲדָמָה וְהִתְחִיל לְנַשֵּׁק בְּשִׂמְחָה אֶת גּוּשֵׁי הֶעָפָר וְאֶת סַלְעֵי אֶרֶץ יִשְׂרָאֵל, כְּשֶׁדִּמְעוֹת תּוֹדָה חוֹמוֹת זולְגוֹת מֵעֵינָיו וּמַרְטִיבוֹת אֶת רִגְבֵי קֹדֶשׁ נַחֲלַת אֲבוֹתֵינוּ.

מקורות: תלמוד בבלי, מסכת כתובות, דף קי"ב, עמוד א' (סיפור חציית הנהר – "נקט במצרא ועבר", והתשובה לצדוקי). תלמוד בבלי, מסכת שבת, דף מ"א, עמוד א' / מסכת בבא מציעא דף פ"ה עמוד א' (רקע על געגועיו העזים של רבי זירא ועלייתו לארץ הקודש).

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “יְלָדִים, אִם הֱיִיתֶם נִכְנָסִים לְבֵית הַמִּדְרָשׁ הַגָּדוֹל בְּבָבֶל לִפְנֵי כְּאַלְפַּיִם שָׁנָה, הֱיִיתֶם עוֹמְדִים בְּהִשְׁתָּאוּת.” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “רַבִּי זֵירָא נֶאֱנַח אֲנָחָה אֲרֻכָּה שֶׁבָּאָה מֵעֹמֶק נִשְׁמָתוֹ: "גּוּפִי נִמְצָא כָּאן בְּבָבֶל," הוּא לָחַשׁ, "אֲבָל נִשְׁמָתִי – נִשְׁמָתִי…”?
ג. כיצד הגיב ר' זירא למצב המתואר: “וְהִנֵּה, לְאַחַר תְּלָאוֹת רַבּוֹת, הוּא הִגִּיעַ אֶל הַגְּבוּל.”?
ד. מה השתנה בין “הֵם הִנִּיחוּ אֶת הַשַּׂקִּים שֶׁלָּהֶם עַל הָאֲדָמָה הַלַּחָה, הִתְיַשְּׁבוּ בְּצֵל הָעֵצִים וְחִכּוּ בְּסַבְלָנוּת.” לבין “הַמַּיִם הַגּוֹעֲשִׁים לְמַטָּה הִשְׁפְּרִיצוּ טִפּוֹת קָרוֹת שֶׁנָּגְעוּ בְּפָנָיו.”?
ה. כיצד הגיע הסיפור אל האירוע: “רָשָׁע שֶׁכְּמוֹתְךָ!" זָעַק רַבִּי זֵירָא בְּקוֹל שֶׁגָּבַר עַל רַעַשׁ הַגַּלִּים, "מָקוֹם שֶׁמֹּשֶׁה רַבֵּנוּ וְאַהֲרֹן הַכֹּהֵן לֹא זָכוּ לְהִכָּנֵס…”?

לפני ואחרי

מה קרה לפני: “הִיא חָנְתָה עַל הַגָּדָה מִמּוּל, מַמְתִּינָה לַנּוֹסְעִים הַבָּאִים.”?
מה קרה אחרי: “הָרוּחַ טִלְטְלָה אֶת גּוּפוֹ – אַךְ עֵינָיו נָצְצוּ מֵאֹשֶׁר!”?

סיבה ותוצאה מתוך המעשה

האירוע הנתוןמה גרם לו?מה הייתה תוצאתו?
הוּא יָדַע שֶׁהַדֶּרֶךְ קָשָׁה וּמְסֻכֶּנֶת, שֶׁיֵּשׁ שׁוֹדְדִים, מִדְבָּרִיּוֹת צְמֵאִים לְמַיִם וְהָרִים גְּבוֹהִים לַחֲצוֹת, אַךְ הַהַשְׁגָּחָה…
מִי שֶׁמִּסְתַּכֵּן לַעֲבֹר עָלָיו וּמַחֲלִיק – נוֹפֵל יְשִׁירוֹת לְמַעֲמַקֵּי הַמַּיִם הַקָּרִים!

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “חיבת הארץ” על ר' זירא? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “רַבִּי זֵירָא יָכוֹל הָיָה מַמָּשׁ לְהָרִיחַ אֶת רֵיחַ הֶעָפָר וְהַטַּל שֶׁל אֶרֶץ הַקֹּדֶשׁ.”.
ג. אילו ר' זירא לא היה פועל כפי שמתואר ב־“הַמַּיִם הַגּוֹעֲשִׁים לְמַטָּה הִשְׁפְּרִיצוּ טִפּוֹת קָרוֹת שֶׁנָּגְעוּ בְּפָנָיו.” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “רָשָׁע שֶׁכְּמוֹתְךָ!" זָעַק רַבִּי זֵירָא בְּקוֹל שֶׁגָּבַר עַל רַעַשׁ הַגַּלִּים, "מָקוֹם שֶׁמֹּשֶׁה רַבֵּנוּ וְאַהֲרֹן…” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. יְלָדִים, אִם הֱיִיתֶם נִכְנָסִים לְבֵית הַמִּדְרָשׁ הַגָּדוֹל בְּבָבֶל…
2. אֲנִי מַרְגִּישׁ אֶת אֲוִיר הָאָרֶץ הַקְּדוֹשָׁה קוֹרֵא לִי.
3. מוּל עֵינָיו נִגְלָה נָהָר אַדִּיר, כַּנִּרְאֶה נְהַר הַיַּרְדֵּן אוֹ…
4. יְלָדִים, זֶה לֹא הָיָה גֶּשֶׁר בָּטוּחַ שֶׁל קְרָשִׁים, אֶלָּא רַק…
5. עָמַד שָׁם מִין, אָדָם רָשָׁע וְצוֹרֵר יִשְׂרָאֵל, שֶׁשָּׂנֵא אֶת…
6. הוּא לֹא קָם עַל רַגְלָיו בְּגַאֲוָה כְּמִי שֶׁנִּצַּח בְּוִכּוּחַ.

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “וְהִנֵּה, לְאַחַר תְּלָאוֹת רַבּוֹת, הוּא הִגִּיעַ אֶל הַגְּבוּל.”.
מכתב: כתבו לר' זירא בעקבות “מִי שֶׁמִּסְתַּכֵּן לַעֲבֹר עָלָיו וּמַחֲלִיק – נוֹפֵל יְשִׁירוֹת לְמַעֲמַקֵּי הַמַּיִם הַקָּרִים!”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “הַמִּין הַזֶּה הִצְבִּיעַ עַל הָרַב הַגָּדוֹל הַתָּלוּי עַל הַחֶבֶל וְצָעַק לְעֶבְרוֹ:”.
דיווח: כתבו ידיעה קצרה שכותרתה “חיבת הארץ”.
