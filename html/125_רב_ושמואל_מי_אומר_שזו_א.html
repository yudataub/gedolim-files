"מי אומר שזו א'"

דף עבודה בעקבות רב ושמואל
שם: כיתה:

המעשה — הסיפור המלא

להלן מערך שיעור וסיפור חינוכי, עשיר ומרתק על רב ושמואל וסוד מסורת התורה (מי אומר שזו א') . הסיפור מעובד בקפידה עבור בנים בכיתות ב'-ד' (חינוך דתי-לאומי), תוך שימוש בעקרונות סטוריטלינג, בניית מתח, ותיאורים חושיים, על מנת להעביר את המסר העצום של האמונה במסורת אבותינו ורבותינו .

הסיפור מנוקד במלואו כנדרש.

פְּתִיחָה: רְחוֹבוֹת בָּבֶל וְקוֹל הַתּוֹרָה

יְלָדִים, אֲנַחְנוּ עוֹצְמִים אֶת הָעֵינַיִם וְחוֹזְרִים כִּמְעַט אַלְפַּיִם שָׁנָה אָחוֹרָה, אֶל אֶרֶץ בָּבֶל.

רֵיחַ שֶׁל נְהָרוֹת גְּדוֹלִים נִשָּׂא בָּאֲוִיר, עֲצֵי דֶּקֶל רָמִים עוֹמְדִים בְּצִדֵּי הַדְּרָכִים, וְהַשֶּׁמֶשׁ הַחַמָּה שׁוֹלַחַת קַרְנַיִם שֶׁל זָהָב עַל הַבָּתִּים שֶׁל נְהַרְדְּעָא וְסוּרָא.

בָּרְחוֹבוֹת הַצָּרִים הוֹלְכִים שְׁנַיִם מִגְּדוֹלֵי יִשְׂרָאֵל. כָּל מִי שֶׁעוֹבֵר עַל פְּנֵיהֶם עוֹצֵר, מַרְכִּין אֶת רֹאשׁוֹ בְּיִרְאַת כָּבוֹד, וּמְפַנֶּה אֶת הַדֶּרֶךְ.

הָאֶחָד הוּא גְּבַהּ קוֹמָה, מַרְאֵהוּ כְּמַלְאַךְ ה' צְבָאוֹת, וְשְׁמוֹ "אַבָּא אֲרִיכָא", אַךְ כֻּלָּם קוֹרְאִים לוֹ בְּשֵׁם הַכָּבוֹד: "רַב" . הַשֵּׁנִי שֶׁלְּצִדּוֹ מְלֻמָּד עָצוּם, שֶׁיּוֹדֵעַ אֶת כָּל סוֹדוֹת מַסְלוּלֵי הַכּוֹכָבִים בַּשָּׁמַיִם כְּמוֹ אֶת הָרְחוֹבוֹת שֶׁל עִירוֹ, וְרוֹפֵא דָּגוּל – שְׁמוּאֵל .

הֵם צוֹעֲדִים יַחַד, מְשׂוֹחֲחִים בְּדִבְרֵי תּוֹרָה. קוֹלָם הֶעָמֹק מִשְׁתַּלֵּב בִּצְלִילֵי הַשּׁוּק הֶהוֹמֶה.

וְהִנֵּה, כְּשֶׁהֵם עָבְרוּ סָמוּךְ לְאַחַד הַבָּתִּים, שָׁמְעוּ קוֹלוֹת מְתוּקִים מִבִּפְנִים. קָמַץ אַלֶ"ף – אָ... קָמַץ בֵּי"ת – בָּ... זֶה הָיָה קוֹלָם שֶׁל תִּינוֹקוֹת שֶׁל בֵּית רַבָּן, יְלָדִים קְטַנִּים שֶׁמַּתְחִילִים לִלְמֹד אֶת הָאוֹתִיּוֹת הַקְּדוֹשׁוֹת שֶׁל הַתּוֹרָה. רַב וּשְׁמוּאֵל חִיְּכוּ. אֵין בָּעוֹלָם דָּבָר אָהוּב עַל הַקָּדוֹשׁ בָּרוּךְ הוּא כְּמוֹ קוֹלָם שֶׁל יְלָדִים שֶׁלּוֹמְדִים תּוֹרָה בְּטָהֳרָה. אֲבָל פִּתְאוֹם... הַקּוֹלוֹת פָּסְקוּ.

מִתּוֹךְ הַ"חֵדֶר" בָּקַע קוֹל כּוֹעֵס שֶׁל אוֹתוֹ הַמְלַמֵּד: "תַּגִּיד אָלֶ"ף! לָמָּה אַתָּה עַקְשָׁן כָּל כָּךְ?! תַּגִּיד אָלֶ"ף!".

הַתַּלְמִיד הָעַקְשָׁן

רַב וּשְׁמוּאֵל הִבִּיטוּ זֶה בָּזֶה, וְהֶחְלִיטוּ לְהִכָּנֵס בְּשֶׁקֶט לְתוֹךְ הַכִּתָּה.

בְּתוֹךְ הַחֶדֶר, קַרְנֵי הַשֶּׁמֶשׁ נִכְנְסוּ מִבַּעַד לַחַלּוֹן, וְהֵאִירוּ חֶלְקִיקֵי אָבָק שֶׁרִקְּדוּ בָּאֲוִיר. הָרֵיחַ בַּכִּתָּה הָיָה רֵיחַ עַתִּיק וְנָעִים שֶׁל קְלָפִים שֶׁל תּוֹרָה וּדְיוֹ. עַל הַסַּפְסָלִים הַקָּשִׁים וְהַפְּשׁוּטִים מֵעֵץ, יָשְׁבוּ עֲשָׂרוֹת יְלָדִים. וּמוּל הַמְלַמֵּד, עָמַד יֶלֶד אֶחָד קָטָן, אוּלַי בֶּן שֶׁבַע, שֶׁפָּנָיו הָיוּ אֲדֻמּוֹת. הוּא שִׁלֵּב אֶת יָדָיו עַל הֶחָזֶה שֶׁלּוֹ, שְׂפָתָיו הָיוּ קְפוּצוֹת, וְהוּא שָׁתַק.

הַמְלַמֵּד הָיָה חֲסַר אוֹנִים. הַזֵּעָה צָצָה מִמִּצְחוֹ וְהוּא הִקִּישׁ בְּאֶצְבָּעוֹ שׁוּב וָשׁוּב עַל מְגִלַּת הַקְּלָף הַפְּתוּחָה שֶׁהָיְתָה מֻנַּחַת עַל הַשֻּׁלְחָן.

"יַלְדִּי," אָמַר הַמְלַמֵּד בַּחֲצִי יֵאוּשׁ, "תִּסְתַּכֵּל עַל הָאוֹת הַזּוֹ, עִם הַקַּו בָּאֶמְצַע וּשְׁתֵּי הַנְּקֻדּוֹת (יוּד וְיוּד הֲפוּכָה) בַּצְּדָדִים. קוֹרְאִים לָהּ אָלֶ"ף. בּוֹא תַּגִּיד אַחֲרַי: אָלֶ"ף!".

אֲבָל הַיֶּלֶד מָשַׁךְ בִּכְתֵפָיו. "אֲנִי לֹא מַסְכִּים," עָנָה בְּקוֹל בָּטוּחַ.

"מָה זֹאת אוֹמֶרֶת לֹא מַסְכִּים?!" נִדְהַם הַמְלַמֵּד. "לָמָּה?!".

"כִּי מִי אוֹמֵר שֶׁזּוֹ בֶּאֱמֶת אוֹת אָלֶ"ף?!" צָעַק הַיֶּלֶד הַחָכָם וְהָעַקְשָׁן. "אוּלַי זֶה סְתָם צִיּוּר? אוּלַי קוֹרְאִים לָזֶה 'גִּימֶל'? אוּלַי 'שִׁין'? אֲנִי לֹא מוּכָן לוֹמַר דָּבָר שֶׁאֲנִי לֹא מֵבִין בַּשֵּׂכֶל שֶׁלִּי. תּוֹכִיחַ לִי שֶׁכָּכָה נִרְאֵית הָאוֹת אָלֶ"ף, וְרַק אָז אַסְכִּים לִקְרֹא אוֹתָהּ!".

הַמְלַמֵּד תָּפַס אֶת רֹאשׁוֹ בִּשְׁתֵּי יָדָיו. מֵעוֹלָם לֹא נִתְקַל בְּיֶלֶד כָּזֶה. אֵיךְ מוֹכִיחִים שֶׁאָלֶ"ף הִיא אָלֶ"ף? זֶה חֻקִּי הַשָּׂפָה! כָּכָה זֶה וְזֶהוּ! הוּא הָיָה בְּטוּחַ שֶׁהַיֶּלֶד סְתָם מִתְחַצֵּף כְּדֵי לְהַפְרִיעַ לַשִּׁעוּר, וּכְבָר הֵרִים אֶת קוֹלוֹ כְּדֵי לִנְזֹף בּוֹ בְּחֻמְרָה.

אֵשׁ שֶׁל אֱמוּנָה

לְפֶתַע, צֵל כִּסָּה אֶת פֶּתַח הַכִּתָּה.

שְׁנֵי צַדִּיקִים גְּדוֹלִים, עֲטוּפִים בְּטַלִּיתוֹת וְעֵינֵיהֶם מְאִירוֹת כְּכוֹכָבִים, עָמְדוּ בַּפֶּתַח. הַסַּפְסָלִים שָׁתְקוּ. הַמְלַמֵּד הִשְׁתַּחֲוָה מִיָּד, שָׁכַח אֶת כַּעֲסוֹ: "שָׁלוֹם לְרַבּוֹתַי, רַב וּשְׁמוּאֵל. עַל מָה זָכִיתִי לְבִקּוּר הֶחָשׁוּב הַזֶּה?".

"שָׁמַעְנוּ שֶׁיֵּשׁ לְךָ פֹּה יֶלֶד שֶׁשּׁוֹאֵל שְׁאֵלוֹת קָשׁוֹת," אָמַר 'רַב' בְּקוֹלוֹ הֶעָמֹק וְהַחַם. "תַּרְשֶׁה לָנוּ לְדַבֵּר אִתּוֹ?".

הַמְלַמֵּד זָז מְעַט הַצִּדָּה, כְּלוֹמַר – בְּבַקָּשָׁה.

רַב וּשְׁמוּאֵל הִתְקָרְבוּ לַיֶּלֶד הַקָּטָן. הוּא הִבִּיט בָּהֶם, וְלִבּוֹ הַקָּטָן דָּפַק בְּחָזְקָה, אַךְ עֲדַיִן עָמַד בְּשֶׁלּוֹ.

שְׁמוּאֵל הִנִּיחַ יָד חַמָּה וְרַכָּה עַל כְּתֵפוֹ שֶׁל הַיֶּלֶד. "מָה שִׁמְךָ, צַדִּיק הַרְבֵּה?"

"יִצְחָק," לָחַשׁ הַיֶּלֶד.

"וּמַדּוּעַ אֵינְךָ רוֹצֶה לִקְרֹא מֵהַסֵּפֶר אֶת הָאוֹת הָרִאשׁוֹנָה?"

יִצְחָק אָזַר אֹמֶץ. "כִּי... כִּי אֲנִי מְבַקֵּשׁ הוֹכָחָה! מִי אָמַר לַמּוֹרֶה שֶׁזּוֹ הָאוֹת אָלֶ"ף? אוּלַי טָעוּ פַּעַם? אוּלַי מִישֶׁהוּ הִמְצִיא אֶת זֶה? אֲנִי לֹא מַאֲמִין בִּכְלוּם עַד שֶׁלֹּא יוֹכִיחוּ לִי!".

רַב וּשְׁמוּאֵל לֹא כָּעֲסוּ עָלָיו אֲפִלּוּ טִפָּה. בְּעֵינֵיהֶם דָּלַק נִיצוֹץ שֶׁל גַּאֲוָה עַל הַחָכְמָה שֶׁל יַלְדֵי יִשְׂרָאֵל.

אָז פָּנָה אֵלָיו 'רַב' בְּעֶדְנָה: "תֹּאמַר לִי, יִצְחָק, מִי הֵם אֲנָשִׁים שֶׁיּוֹשְׁבִים עַכְשָׁו בַּבַּיִת שֶׁלְּךָ?".

"הַהוֹרִים שֶׁלִּי," עָנָה הַיֶּלֶד בִּפְלִיאָה עַל הַשְּׁאֵלָה שֶׁלֹּא קְשׁוּרָה.

"הַאִם אַתָּה אוֹהֵב אֶת אַבָּא שֶׁלְּךָ?" שָׁאַל שְׁמוּאֵל.

"בְּוַדַּאי שֶׁאֲנִי אוֹהֵב אוֹתוֹ!" קָפַץ יִצְחָק.

"וְאֵיךְ אַתָּה בְּמֵאָה אֲחוּזִים בָּטוּחַ שֶׁהוּא בֶּאֱמֶת אַבָּא שֶׁלְּךָ? יֵשׁ לְךָ הוֹכָחָה לָזֶה בַּשֵּׂכֶל?"

הַיֶּלֶד הִשְׁתַּתֵּק. הוּא קִמֵּט אֶת מִצְחוֹ בְּמַחֲשָׁבָה. "לֹא..." חָשַׁב בְּקוֹל. "אֲנִי יוֹדֵעַ שֶׁהוּא אַבָּא שֶׁלִּי, כִּי... כִּי אִמָּא שֶׁלִּי אָמְרָה לִי! וְכִי הוּא תָּמִיד הָיָה אַבָּא שֶׁלִּי מֵאָז שֶׁאֲנִי תִּינוֹק!".

"יָפֶה מְאֹד," חִיֵּךְ 'רַב' חִיּוּךְ מֵאִיר כַּשֶּׁמֶשׁ. "אַתָּה אֲפִלּוּ לֹא זָקוּק לְהוֹכָחָה בַּשֵּׂכֶל. לָמָּה? כִּי יֵשׁ לְךָ אֵמוּן בְּהוֹרֶיךָ. אַתָּה בּוֹטֵחַ בָּהֶם וּמַאֲמִין שֶׁהֵם אוֹמְרִים לְךָ אֶת הָאֱמֶת שֶׁל הַחַיִּים. בְּלִי הָאֵמוּן הַזֶּה, אִי אֶפְשָׁר לִחְיוֹת!".

שַׁרְשֶׁרֶת הַזָּהָב

שְׁמוּאֵל נָגַע בַּקְּלָף, יָדוֹ לִטְּפָה אֶת הָאוֹת אָלֶ"ף הַשְּׁחוֹרָה וְהַיָּפָה בַּעֲדִינוּת שֶׁל אוֹמָן. "יִצְחָק בְּנִי, בַּתּוֹרָה שֶׁלָּנוּ שׁוּם דָּבָר אֵינוֹ סְתָם הַמְצָאָה שֶׁל אָדָם בָּרְחוֹב. אַתָּה שׁוֹאֵל מִי אָמַר שֶׁזּוֹ הָאוֹת אָלֶ"ף? הַמּוֹרֶה שֶׁלְּךָ אוֹמֵר לְךָ שֶׁזּוֹ הָאוֹת אָלֶ"ף. הַאִם אַתָּה סוֹמֵךְ עַל הַמּוֹרֶה שֶׁלְּךָ?"

יִצְחָק הִנְהֵן. עֵינָיו נִפְתְּחוּ לִרְוָחָה.

"וּמִי אָמַר לַמּוֹרֶה שֶׁלְּךָ?" שָׁאַל 'רַב'.

"הַמּוֹרֶה שֶׁלּוֹ כְּשֶׁהוּא הָיָה קָטָן!" צָעֲקוּ כַּמָּה יְלָדִים מֵהַסַּפְסָלִים מֵאָחוֹר.

"נָכוֹן מְאֹד! וְלַמּוֹרֶה שֶׁלּוֹ אָמַר הַמּוֹרֶה שֶׁלּוֹ. אֲנַחְנוּ יְהוּדִים, יֵשׁ לָנוּ שַׁרְשֶׁרֶת זָהָב גְּדוֹלָה שֶׁאַף פַּעַם לֹא נִקְרְעָה. עֶזְרָא הַסּוֹפֵר לָמַד אֶת כְּתָב הַקֹּדֶשׁ מִיִּרְמְיָהוּ וִישַׁעְיָהוּ, שֶׁלָּמְדוּ מִדָּוִד הַמֶּלֶךְ, שֶׁלָּמַד מִשְּׁמוּאֵל הַנָּבִיא... עַד יְהוֹשֻׁעַ בִּן נוּן! וִיהוֹשֻׁעַ לָמַד מִמֹּשֶׁה רַבֵּנוּ בְּהַר סִינַי! וּמֹשֶׁה שָׁמַע מִפִּי הַקָּדוֹשׁ בָּרוּךְ הוּא בִּכְבוֹדוֹ וּבְעַצְמוֹ שֶׁהַצּוּרָה הַזּוֹ בַּשָּׁמַיִם הִיא 'אָלֶ"ף'!"

דְּמָמָה יָרְדָה עַל הַכִּתָּה.

יִצְחָק תָּפַס אֶת עֹמֶק הַסּוֹד. לֹא כָּל דָּבָר בָּעוֹלָם אֲנַחְנוּ צְרִיכִים לְהוֹכִיחַ שׁוּב וָשׁוּב, כִּי יֵשׁ לָנוּ אוֹצָר יָקָר שֶׁשּׁוּם עַם אַחֵר אֵין לוֹ: אֱמוּנַת חֲכָמִים וּמָסֹרֶת אָבוֹת . לִסְמֹךְ עַל אַבָּא וְאִמָּא, לִסְמֹךְ עַל הַמּוֹרֶה, לִסְמֹךְ עַל גְּדוֹלֵי הַדּוֹר.

"אִם כָּךְ," לָחַשׁ יִצְחָק וְחִיּוּךְ רָחָב עָלָה עַל שְׂפָתָיו, "הָאָלֶ"ף הַזּוֹ הִיא לֹא סְתָם צִיּוּר. הִיא צִיּוּר שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא צִיֵּר!".

"בְּדִיּוּק," אָמְרוּ יַחַד רַב וּשְׁמוּאֵל. "עַכְשָׁו אֱמֹר יַלְדִּי הַיָּקָר, מָה הָאוֹת הַזּוֹ?"

יִצְחָק נָשַׁם נְשִׁימָה עֲמֻקָּה וְזָעַק בְּשִׂמְחָה עֲצוּמָה: "אָלֶ"ף!!!" .

וְכָל הַכִּתָּה חָזְרָה אַחֲרָיו בְּשִׁירָה מְתוּקָה, עַד שֶׁקּוֹלָם שֶׁל הַתִּינוֹקוֹת בָּקַע עַד אֶל מֵעֵבֶר לְעַנְנֵי הַשָּׁמַיִם.

א. שְׁאֵלַת פְּתִיחָה:

"יְלָדִים, אִם עַכְשָׁו אֲנִי אֶצְבַּע עַל הַשֻּׁלְחָן הַזֶּה וְאַגִּיד שֶׁקּוֹרְאִים לָזֶה 'מָטוֹס', תַּאֲמִינוּ לִי? בָּרוּר שֶׁלֹּא! וּמִי אָמַר שֶׁבֶּאֱמֶת קוֹרְאִים לָזֶה 'שֻׁלְחָן'? מֵאֵיפֹה אַתֶּם יוֹדְעִים? [תֵּן לַיְלָדִים לַעֲנוֹת: אַבָּא אָמַר, אִמָּא אָמְרָה]. מְצֻיָּן! הַיּוֹם נִלְמַד שֶׁבַּתּוֹרָה הַקְּדוֹשָׁה – הַסּוֹד הֲכִי גָּדוֹל שֶׁלָּנוּ הוּא... הַמָּסֹרֶת. בּוֹאוּ נִשְׁמַע סִפּוּר מַדְהִים עַל יֶלֶד שֶׁרָצָה הוֹכָחָה!"

ב. הֶעָרוֹת בִּצּוּעַ וְסְטוֹרִיטֶלִינְג:

ג. הַמְלָצוֹת לְאַרְבַּע שְׁקוּפִיּוֹת בַּמַּצֶּגֶת (עִם הַקְפָּדָה עַל צְנִיעוּת):

ד. פְּעִילוּת מְסַכֶּמֶת מֻמְלֶצֶת:

"טֶלֶפוֹן שָׁבוּר שֶׁל תּוֹרָה" אוֹ "מִסְרוֹ לִיהוֹשֻׁעַ"

יַעַמְדוּ כָּל הַתַּלְמִידִים בְּשׁוּרָה לְאֹרֶךְ הַכִּתָּה. הַמּוֹרֶה יִלְחַשׁ לָרִאשׁוֹן פָּסוּק קָצָר (כְּמוֹ: "תּוֹרָה צִוָּה לָנוּ מֹשֶׁה"). הַיֶּלֶד יַעֲבִיר אוֹתוֹ לַיֶּלֶד הַבָּא בְּדִיּוּק מֻחְלָט לְלֹא טָעֻיּוֹת! הַמַּטָּרָה לְהַרְאוֹת שֶׁרַק כְּשֶׁיֵּשׁ אֵמוּן מָלֵא בֵּין הַמּוֹסֵר לַמְקַבֵּל בַּשַּׁרְשֶׁרֶת – הַמִּלָּה מַגִּיעָה שְׁלֵמָה לַסּוֹף (לֹא מְשַׁנִּים כְּלוּם!).

מקורות: קֹהֶלֶת רַבָּה (וִילְנָא) פָּרָשָׁה ז', כ"ג (ד"ה "טוֹב אַחֲרִית דָּבָר"): הַמִּדְרָשׁ מֵבִיא אֶת הָרַעְיוֹן שֶׁל "מִי אוֹמֵר שֶׁזּוֹ אָלֶ"ף", כְּדֻגְמָה מֻבְהֶקֶת לְ"טוֹב אַחֲרִית דָּבָר" – וְשָׁם מֻזְכָּרִים רַב וּשְׁמוּאֵל עַל קַבָּלַת הַקְּרִיאָה בְּמָסֹרֶת (רַב פָּלַח קְרִי, שְׁמוּאֵל פָּלַח קְרִי). "כֹּה עָשׂוּ חֲכָמֵינוּ" חֵלֶק ג', עמ' 73: שָׁם מְבֹאָר הַסִּפּוּר בְּעִבּוּד לִילָדִים עַל רַב וּשְׁמוּאֵל ("מִי אוֹמֵר שֶׁזּוֹ אָלֶ"ף"), כְּשֶׁהֵם נוֹתְנִים לַיֶּלֶד אֶת הַתְּשׁוּבָה שֶׁחַיָּבִים לִסְמֹךְ עַל הַמָּסֹרֶת – "מִי אוֹמֵר שֶׁזּוֹ אִמְּךָ? אֶלָּא שֶׁכָּךְ לִמְּדוּךָ וְנֶאֱמָן הַדָּבָר עָלֶיךָ". (וְכֵן מֻזְכָּר בְּסִפְרִיַת עָבָר/הִלֵּל הַזָּקֵן בְּגִרְסָה מַקְבִּילָה לַגֵּר).

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “להלן מערך שיעור וסיפור חינוכי, עשיר ומרתק על רב ושמואל וסוד מסורת התורה (מי אומר שזו א') .” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “אֵין בָּעוֹלָם דָּבָר אָהוּב עַל הַקָּדוֹשׁ בָּרוּךְ הוּא כְּמוֹ קוֹלָם שֶׁל יְלָדִים שֶׁלּוֹמְדִים תּוֹרָה בְּטָהֳרָה.”?
ג. כיצד הגיב רב ושמואל למצב המתואר: “הַזֵּעָה צָצָה מִמִּצְחוֹ וְהוּא הִקִּישׁ בְּאֶצְבָּעוֹ שׁוּב וָשׁוּב עַל מְגִלַּת הַקְּלָף הַפְּתוּחָה שֶׁהָיְתָה מֻנַּחַת עַל הַשֻּׁלְחָן.”?
ד. מה השתנה בין “שְׁמוּאֵל הִנִּיחַ יָד חַמָּה וְרַכָּה עַל כְּתֵפוֹ שֶׁל הַיֶּלֶד.” לבין “אֲנַחְנוּ יְהוּדִים, יֵשׁ לָנוּ שַׁרְשֶׁרֶת זָהָב גְּדוֹלָה שֶׁאַף פַּעַם לֹא נִקְרְעָה.”?
ה. כיצד הגיע הסיפור אל האירוע: “בּוֹאוּ נִשְׁמַע סִפּוּר מַדְהִים עַל יֶלֶד שֶׁרָצָה הוֹכָחָה!”?

מה קרה אחרי: “עֶזְרָא הַסּוֹפֵר לָמַד אֶת כְּתָב הַקֹּדֶשׁ מִיִּרְמְיָהוּ וִישַׁעְיָהוּ, שֶׁלָּמְדוּ מִדָּוִד הַמֶּלֶךְ, שֶׁלָּמַד מִשְּׁמוּאֵל הַנָּבִיא...”?

סיבה ותוצאה מתוך המעשה

האירוע הנתוןמה גרם לו?מה הייתה תוצאתו?
הָרֵיחַ בַּכִּתָּה הָיָה רֵיחַ עַתִּיק וְנָעִים שֶׁל קְלָפִים שֶׁל תּוֹרָה וּדְיוֹ.
אַתָּה בּוֹטֵחַ בָּהֶם וּמַאֲמִין שֶׁהֵם אוֹמְרִים לְךָ אֶת הָאֱמֶת שֶׁל הַחַיִּים.

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “"מי אומר שזו א'"” על רב ושמואל? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “שְׁנֵי צַדִּיקִים גְּדוֹלִים, עֲטוּפִים בְּטַלִּיתוֹת וְעֵינֵיהֶם מְאִירוֹת כְּכוֹכָבִים, עָמְדוּ בַּפֶּתַח.”.
ג. אילו רב ושמואל לא היה פועל כפי שמתואר ב־“אֲנַחְנוּ יְהוּדִים, יֵשׁ לָנוּ שַׁרְשֶׁרֶת זָהָב גְּדוֹלָה שֶׁאַף פַּעַם לֹא נִקְרְעָה.” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “בּוֹאוּ נִשְׁמַע סִפּוּר מַדְהִים עַל יֶלֶד שֶׁרָצָה הוֹכָחָה!” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. להלן מערך שיעור וסיפור חינוכי, עשיר ומרתק על רב ושמואל וסוד מסורת התורה…
2. מִתּוֹךְ הַ"חֵדֶר" בָּקַע קוֹל כּוֹעֵס שֶׁל אוֹתוֹ הַמְלַמֵּד:…
3. אֲנִי לֹא מוּכָן לוֹמַר דָּבָר שֶׁאֲנִי לֹא מֵבִין בַּשֵּׂכֶל שֶׁלִּי.
4. הַהוֹרִים שֶׁלִּי," עָנָה הַיֶּלֶד בִּפְלִיאָה עַל הַשְּׁאֵלָה שֶׁלֹּא…
5. לֹא כָּל דָּבָר בָּעוֹלָם אֲנַחְנוּ צְרִיכִים לְהוֹכִיחַ שׁוּב וָשׁוּב,…
6. הַמַּטָּרָה לְהַרְאוֹת שֶׁרַק כְּשֶׁיֵּשׁ אֵמוּן מָלֵא בֵּין הַמּוֹסֵר…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “הַזֵּעָה צָצָה מִמִּצְחוֹ וְהוּא הִקִּישׁ בְּאֶצְבָּעוֹ שׁוּב וָשׁוּב עַל מְגִלַּת הַקְּלָף הַפְּתוּחָה שֶׁהָיְתָה מֻנַּחַת עַל…”.
מכתב: כתבו לרב ושמואל בעקבות “אַתָּה בּוֹטֵחַ בָּהֶם וּמַאֲמִין שֶׁהֵם אוֹמְרִים לְךָ אֶת הָאֱמֶת שֶׁל הַחַיִּים.”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “לִסְמֹךְ עַל אַבָּא וְאִמָּא, לִסְמֹךְ עַל הַמּוֹרֶה, לִסְמֹךְ עַל גְּדוֹלֵי הַדּוֹר.”.
דיווח: כתבו ידיעה קצרה שכותרתה “"מי אומר שזו א'"”.

המחזה

רגע מן הסיפורהדמויותמה יאמרו או יעשו?
הָרֵיחַ בַּכִּתָּה הָיָה רֵיחַ עַתִּיק וְנָעִים שֶׁל קְלָפִים שֶׁל תּוֹרָה וּדְיוֹ.
אֲנַחְנוּ יְהוּדִים, יֵשׁ לָנוּ שַׁרְשֶׁרֶת זָהָב גְּדוֹלָה שֶׁאַף פַּעַם לֹא נִקְרְעָה.

7. חיבור לחיים מתוך המעשה

א. מה אפשר ללמוד מן הדרך שבה רב ושמואל פעל באירוע “שְׁמוּאֵל הִנִּיחַ יָד חַמָּה וְרַכָּה עַל כְּתֵפוֹ שֶׁל הַיֶּלֶד.”?
ב. תארו מצב מחיי הכיתה שבו אפשר ליישם את המסר של “"מי אומר שזו א'"”.
