קורא פסוק ראשון של ק"ש תוך כדי הלימוד

דף עבודה בעקבות רבי יהודה הנשיא (רבי)
שם: כיתה:

המעשה — הסיפור המלא

להלן מערך חינוכי, רגשי ומרתק על רבי יהודה הנשיא (רבינו הקדוש) ומסירותו העצומה ללימוד התורה עם תלמידיו , המעובד במיוחד לבנים בכיתות ב'-ד' (חינוך דתי-לאומי).

הסיפור נכתב על פי כללי הסטוריטלינג שביקשת, משלב תיאורים חושיים, מתח רגשי, ומנוקד במלואו.

פְּתִיחָה: הָאַרְמוֹן שֶׁל הַתּוֹרָה

יְלָדִים, נַסּוּ לְדַמְיֵן שֶׁאַתֶּם עוֹמְדִים מוּל שַׁעַר גָּדוֹל, עָשׂוּי מֵעֵץ אַלּוֹן כָּבֵד וְעָבֶה. אַתֶּם פּוֹתְחִים מְעַט אֶת הַדֶּלֶת, וּמִיָּד מַכֶּה בָּכֶם גַּל שֶׁל רֵיחוֹת מְיֻחָדִים – רֵיחַ מָתוֹק שֶׁל שֶׁמֶן זַיִת טָהוֹר שֶׁבּוֹעֵר בַּמְּנוֹרוֹת, יַחַד עִם רֵיחַ עַתִּיק שֶׁל קְלָפִים יְשָׁנִים שֶׁל סִפְרֵי קֹדֶשׁ.

זֶהוּ בֵּית הַמִּדְרָשׁ שֶׁל נְשִׂיא יִשְׂרָאֵל. הָאִישׁ הַגָּדוֹל בְּיוֹתֵר בַּדּוֹר! מַנְהִיגָם שֶׁל הַיְּהוּדִים!

כַּאֲשֶׁר אַתֶּם נִכְנָסִים פְּנִימָה, הָאָזְנַיִם שֶׁלָּכֶם כִּמְעַט נִסְתָּמוֹת מֵהָרַעַשׁ. מֵאוֹת, וְאוּלַי אֲלָפִים שֶׁל תַּלְמִידֵי חֲכָמִים לְבוּשִׁים בִּגְלִימוֹת לְבָנוֹת, יוֹשְׁבִים צְפוּפִים עַל סַפְסָלִים. קוֹלוֹתֵיהֶם מִתְרוֹמְמִים וְיוֹרְדִים כְּמוֹ גַּלִּים בַּיָּם. הֵם דָּנִים, מִתְוַכְּחִים, שׁוֹאֲלִים וְעוֹנִים מִתּוֹךְ אֵשׁ שֶׁל דִּבְרֵי תּוֹרָה! הָאֲוִיר בַּחֶדֶר חַם, וְאֶפְשָׁר לִרְאוֹת אֶת טִפּוֹת הַזֵּעָה עַל הַמִּצְחוֹת שֶׁל הַתַּלְמִידִים הַמִּתְאַמְּצִים. כֻּלָּם מִתְאַמְּצִים לְהָבִין אֶת תּוֹרַת ה'.

וּבַמֶּרְכָּז – יוֹשֵׁב רַבִּי יְהוּדָה הַנָּשִׂיא . זְקָנוֹ הַלָּבָן אָרוֹךְ וְטָהוֹר כְּשֶׁלֶג, עֵינָיו מְאִירוֹת כְּמוֹ פָּנָסִים עוֹצְמָתִיִּים. רַבִּי יְהוּדָה הָיָה אָדָם עָשִׁיר מְאֹד. הָיָה לוֹ כֶּסֶף כְּמוֹ לְקֵיסָר [3, 4]. הָיוּ לוֹ שָׂדוֹת, עֲדָרִים, מְשָׁרְתִים. הוּא יָכוֹל הָיָה לָשֶׁבֶת כָּל הַיּוֹם בְּאַרְמוֹנוֹ הַמְּפֹאָר עַל כֻּרְסוֹת מִקְּטִיפָה וְלָנוּחַ.

אֲבָל רַבִּי לֹא הֶחְשִׁיב אֶת הַכֶּסֶף. בִּשְׁבִילוֹ, הַיַּהֲלוֹם הַיָּקָר בְּיוֹתֵר בָּעוֹלָם הָיָה מַשֶּׁהוּ אַחֵר: הַתַּלְמִידִים שֶׁלּוֹ. הַתּוֹרָה שֶׁהוּא מְלַמֵּד אוֹתָם! [1, 5]. הוּא יָדַע, שֶׁעַם יִשְׂרָאֵל סוֹבֵל מִגְּזֵרוֹת הָרוֹמָאִים, וְאִם הוּא לֹא יְלַמֵּד תּוֹרָה לְלֹא הַפְסָקָה, חָלִילָה תִּשְׁתַּכַּח הַתּוֹרָה! [6, 7].

הַמֶּתַח עַל הַסַּפְסָלִים

יוֹם אֶחָד, יָשַׁב רַבִּי יְהוּדָה הַנָּשִׂיא וְלִמֵּד אֶת הַתַּלְמִידִים הֲלָכוֹת עֲמֻקּוֹת בְּיוֹתֵר וְסוֹדוֹת הַתּוֹרָה. רַבִּי קָרָא מִתּוֹךְ זִכְרוֹנוֹ הַמַּדְהִים, פֵּרֵשׁ, הִסְבִּיר וְתֵרֵץ קֻשְׁיוֹת. כָּל הַתַּלְמִידִים יָשְׁבוּ בְּדִמְמַת אַלְחוּט. אִישׁ לֹא הֵזִיז אֲפִלּוּ אֶצְבַּע. כֻּלָּם הָיוּ מְרֻתָּקִים, שׁוֹתִים בְּצִמָּאוֹן עָצוּם כָּל מִלָּה וּמִלָּה הַיּוֹצֵאת מִפִּיו שֶׁל "רַבֵּנוּ הַקָּדוֹשׁ".

אֲבָל... בַּחוּץ, מִבַּעַד לְחַלּוֹנוֹת בֵּית הַמִּדְרָשׁ, הַבֹּקֶר הֵחֵל לְהָאִיר. קַרְנַיִם זְהוּבוֹת שֶׁל שֶׁמֶשׁ נִכְנְסוּ לְתוֹךְ הָאוּלָם, וּמִסְפַּר תַּלְמִידִים הֵחֵלּוּ לָזוּז בְּאִי נוֹחוּת עַל הַמּוֹשָׁבִים שֶׁלָּהֶם.

הֵם הִבִּיטוּ זֶה בָּזֶה בְּחֲשָׁשׁ. אַחַר כָּךְ הִבִּיטוּ הַחוּצָה.

"מָה נַעֲשֶׂה?" לָחַשׁ תַּלְמִיד אֶחָד לַחֲבֵרוֹ.

"אֲנִי לֹא יוֹדֵעַ," עָנָה הַשֵּׁנִי, כַּפּוֹת יָדָיו מַזִּיעוֹת. "הַזְּמַן שֶׁל 'קְרִיאַת שְׁמַע שֶׁל בֹּקֶר' מַגִּיעַ לְסִיּוּמוֹ! אֲנַחְנוּ חַיָּבִים לִקְרֹא קְרִיאַת שְׁמַע!"

בְּדֶרֶךְ כְּלָל, כְּשֶׁמַּגִּיעַ זְמַן תְּפִלָּה, רַב עָצַר אֶת הַשִּׁעוּר, כֻּלָּם עָמְדוּ, הִתְפַּלְּלוּ בְּקוֹל שֶׁל שִׂמְחָה וְקַבָּלַת עֹל מַלְכוּת שָׁמַיִם, וְרַק אָז חָזְרוּ לִלְמֹד. אֲבָל בְּאוֹתוֹ בֹּקֶר, רַבִּי יְהוּדָה הַנָּשִׂיא הָיָה כָּל כָּךְ עָמוּק בְּתוֹךְ לִמּוּד הַתּוֹרָה! הָיָה לוֹ רֶצֶף חֲשִׁיבָה אַדִּיר, וְהוּא הִסְבִּיר יְסוֹדוֹת שֶׁאִם הוּא יַפְסִיק בָּאֶמְצַע לְכַמָּה דַּקּוֹת, הַתַּלְמִידִים יְאַבְּדוּ אֶת מַהֲלַךְ הַחֲשִׁיבָה! הוּא דָּאַג לְהַבָּנָה שֶׁלָּהֶם. תּוֹרָה הִיא חַיִּים, וְלַעֲצֹר אוֹתָהּ עַכְשָׁו זֶה כְּמוֹ לַעֲצֹר מַיִם שֶׁזּוֹרְמִים לְאָדָם צָמֵא הַשּׁוֹתֶה בַּמִּדְבָּר! [1, 2].

הַשֶּׁמֶשׁ שֶׁבַּחוּץ הִמְשִׁיכָה לַעֲלוֹת. הַזְּמַן דָּחַק.

הַתַּלְמִידִים לֹא יָכְלוּ יוֹתֵר לְהַמְתִּין. עַל פִּי הַהֲלָכָה, אִם יְפַסְפְסוּ, הֵם מַפְסִידִים אֶת מִצְוַת קְרִיאַת שְׁמַע. בְּקוֹל שֶׁקֶט הֵם הִפְסִיקוּ לִשְׁמֹעַ, עָצְמוּ אֶת הָעֵינַיִם, שָׂמוּ אֶת יְמִינָם הַקְּדוֹשָׁה עַל הַפָּנִים וְקָרְאוּ בְּלַחַשׁ: "שְׁמַע יִשְׂרָאֵל... ה' אֱלֹקֵינוּ ה' אֶחָד..." . הֵם קִבְּלוּ עַל עַצְמָם אֶת הַמַּלְכוּת שֶׁל מֶלֶךְ מַלְכֵי הַמְּלָכִים, הַקָּדוֹשׁ בָּרוּךְ הוּא.

וְרַבִּי? הַאִם רַבִּי כָּעַס שֶׁהֵם הִפְסִיקוּ אֶת הַהַקְשָׁבָה לַשִּׁעוּר? הַאִם רַבִּי קָרָא אִתָּם? כֻּלָּם פָּקְחוּ אֶת הָעֵינַיִם וְהִבִּיטוּ יָשָׁר בְּרַבִּי יְהוּדָה הַנָּשִׂיא.

הֵם רָאוּ מַחֲזֶה שֶׁלֹּא יִשְׁכְּחוּ לְעוֹלָם!

קַבָּלַת עֹל מַלְכוּת שָׁמַיִם שֶׁל הַנָּשִׂיא

רַבִּי יְהוּדָה, שֶׁרָאָה שֶׁהַזְּמַן מַגִּיעַ, לֹא סָגַר אֶת הַסֵּפֶר! הוּא הֵבִין שֶׁהַתּוֹרָה הִיא חַיֶּינוּ וְלֹא רָצָה לְבַטֵּל אוֹתָהּ, אַף לֹא לְרֶגַע אֶחָד, כְּדֵי לֹא לְקַטֵּעַ אֶת לִמּוּד תַּלְמִידָיו הָאֲהוּבִים. בִּמְקוֹם לַעֲצֹר וְלַעֲשׂוֹת הַפְסָקָה גְּדוֹלָה, הוּא עָשָׂה דָּבָר עָצוּם וְנִפְלָא: תּוֹךְ כְּדֵי שֶׁהוּא דּוֹרֵשׁ וּמְלַמֵּד דִּבְרֵי קֹדֶשׁ, רַבִּי הֶעֱבִיר אֶת כַּף יָדוֹ בִּמְהִירוּת שֶׁכָּזוֹ עַל פָּנָיו – עֵינָיו נֶעֶצְמוּ, רֹאשׁוֹ רָכַן קָדִימָה. בַּשְּׁנִיָּה הַזּוֹ, בְּהִבְהוּב חֲרִישִׁי, נִשְׁמָתוֹ עָלְתָה לַשָּׁמַיִם. הוּא הִמְלִיךְ אֶת בּוֹרֵא עוֹלָם בְּלִבּוֹ וּבְמַחֲשַׁבְתּוֹ, "שְׁמַע יִשְׂרָאֵל", מִתּוֹךְ אָהֲבַת ה' הֲכִי חֲזָקָה בָּעוֹלָם, בְּרִכּוּז עָמֹק וְאַדִּיר, וְהוּא הִסְפִּיק לַעֲשׂוֹת זֹאת בְּאֶמְצַע לִמּוּד הַתּוֹרָה! [2].

אֲבָל... יְלָדִים, אֲנַחְנוּ שׁוֹאֲלִים מִיָּד שְׁאֵלָה – קְרִיאַת שְׁמַע זֶה לֹא רַק הַפָּסוּק הָרִאשׁוֹן! בִּקְרִיאַת שְׁמַע אֲנַחְנוּ קוֹרְאִים שָׁלוֹשׁ פָּרָשִׁיּוֹת. וּבַפָּרָשָׁה הַשְּׁלִישִׁית הַקָּדוֹשׁ בָּרוּךְ הוּא צִוָּה עָלֵינוּ לְהַזְכִּיר מַשֶּׁהוּ מְאֹד חָשׁוּב. אֶת... זֵכֶר יְצִיאַת מִצְרַיִם! עָלֵינוּ לִזְכֹּר בְּכָל יוֹם, בַּבֹּקֶר וּבָעֶרֶב, שֶׁה' הוֹצִיא אוֹתָנוּ מִמִּצְרַיִם. אִם רַבִּי רַק קָרָא "שְׁמַע יִשְׂרָאֵל", כֵּיצַד הוּא הִזְכִּיר אֶת יְצִיאַת מִצְרַיִם?

וְכָאן הִתְגַּלְּתָה הַגְּדֻלָּה שֶׁל מַנְהִיג הַדּוֹר. מִיָּד אַחֲרֵי שֶׁהוֹרִיד אֶת יָדוֹ מֵעֵינָיו, רַבִּי חִיֵּךְ אֶל מֵאוֹת תַּלְמִידָיו, הִבִּיט עֲלֵיהֶם בְּאַהֲבָה גְּדוֹלָה, וּמִיָּד הִמְשִׁיךְ לְלַמֵּד תּוֹרָה. אֲבָל פִּתְאוֹם... הוּא שִׁנָּה קְצָת אֶת נוֹשֵׂא הַשִּׁעוּר. מִבְּלִי לַעֲצֹר, בְּחָכְמָה עֲצוּמָה וּבִגְאוֹנוּת, הוּא הֵבִיא אֶת דִּבְרֵי הַתּוֹרָה עַל חֲשִׁיבוּת יְצִיאַת עַם יִשְׂרָאֵל מִמִּצְרַיִם! הוּא דָּרַשׁ דְּרָשָׁה מְרַתֶּקֶת, עַל כָּךְ שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא גָּאַל אוֹתָנוּ.

כָּךְ שִׁלֵּב רַבִּי בְּגְאוֹנוּתוֹ אֶת הֲלָכוֹת יְצִיאַת מִצְרַיִם בְּתוֹךְ שִׁעוּר הַתּוֹרָה שֶׁלּוֹ, בְּלִי לְעַכֵּב אֶת הַתַּלְמִידִים [1, 2].

הַתַּלְמִידִים יָשְׁבוּ וְהִקְשִׁיבוּ. לִבָּם פָּעַם בְּחָזְקָה. הֵם הֵבִינוּ שֶׁהָרַב שֶׁלָּהֶם עָשָׂה עַכְשָׁו נֵס גָּלוּי שֶׁל מְסִירוּת נֶפֶשׁ. אָהֲבָתוֹ שֶׁל רַבִּי אֲלֵיהֶם, וְאַהֲבָתוֹ לַתּוֹרָה – הָיוּ כָּל כָּךְ חֲזָקוֹת! הוּא לֹא הָיָה מוּכָן שֶׁאַף אֶחָד מֵהֶם יַפְסִיק לִלְמֹד, לֹא הָיָה מוּכָן לְבַטֵּל רֶגַע אֶחָד שֶׁל קְדֻשַּׁת הַלִּמּוּד שֶׁלָּהֶם יַחַד [1]. וְאַף עַל פִּי כֵן, קִיֵּם אֶת צִוּוּי בּוֹרֵא עוֹלָם וְקָרָא קְרִיאַת שְׁמַע.

בְּאוֹתוֹ יוֹם, כְּשֶׁיָּצְאוּ הַתַּלְמִידִים מִבֵּית הַמִּדְרָשׁ, הֵם הָיוּ אֲנָשִׁים אַחֵרִים. הֵם לָמְדוּ הֲלָכוֹת וּמִשְׁנָיוֹת, אֲבָל יוֹתֵר מִכָּל – הֵם לָמְדוּ דֻּגְמָא אִישִׁית. הֵם לָמְדוּ כַּמָּה יָקָר כָּל שְׁנִיָּה שֶׁל תּוֹרָה! זֶהוּ מַנְהִיג אֲמִתִּי. זֶהוּ "רַבֵּנוּ הַקָּדוֹשׁ"!

א. שְׁאֵלַת פְּתִיחָה (לִיצִירַת סַקְרָנוּת וְהִזְדַּהוּת קֹדֶם לַסִּפּוּר):

"יְלָדִים עֲצְמוּ רֶגַע עֵינַיִם. אַתֶּם בְּאֶמְצַע לְשַׂחֵק בַּמִּשְׂחָק הֲכִי שָׁוֶה עִם חֲבֵרִים שֶׁלָּכֶם, פִּתְאוֹם - אִמָּא קוֹרֵאת לָכֶם לַאֲרוּחַת עֶרֶב אוֹ לַעֲזֹר לָהּ. מָה אַתֶּם רְגִילִים לַעֲשׂוֹת? קוֹפְצִים וְאוֹמְרִים 'הַפּוּז, הַפְסָקָה!' נָכוֹן?

הַיּוֹם נִשְׁמַע עַל נְשִׂיא יִשְׂרָאֵל, גְּדוֹל הַדּוֹר שֶׁיָּשַׁב כָּל כָּךְ שָׁקוּעַ בְּדָבָר עָצוּם לְמַעַן הַתַּלְמִידִים שֶׁלּוֹ, שֶׁהוּא הָיָה צָרִיךְ לִמְצֹא פִּתְרוֹן סוֹדִי אֵיךְ לֹא לַעֲצֹר אֲפִלּוּ לִשְׁנִיָּה אַחַת!"

ב. הֶעָרוֹת בִּצּוּעַ וְסְטוֹרִיטֵלִינְג (דַּגֵּשׁ קוֹלִי וְתְנוּעָה):

ג. הַמְלָצוֹת לְ-4 שְׁקוּפִיּוֹת בַּמַּצֶּגֶת (אִיּוּרִים עִם צְנִיעוּת וּקְדֻשָּׁה):

ד. פְּעִילוּת מְסַכֶּמֶת (פְּעִילוּת "קִפָּאוֹן מֻחְלָט שֶׁל קְדֻשָּׁה"):

"בּוֹאוּ נְשַׂחֵק בְּ'מֶלֶךְ מַלְכֵי הַמְּלָכִים מְצַוֶּה'. אֲנִי אֶתֵּן לָכֶם לַעֲשׂוֹת פְּעֻלָּה (לְמָשָׁל: לְשַׂחֵק בִּכְאִלּוּ כַּדּוּרַגְל, לִכְתֹּב בַּמַּחְבֶּרֶת, לְשׂוֹחֵחַ עִם הֶחָבֵר).

כְּשֶׁאֲנִי אוֹמֵר 'וְאָהַבְתָּ!' - אַתֶּם חַיָּבִים לְהַקְפִּיא אֶת הַמִּשְׂחָק הַקּוֹדֵם בְּדִיּוּק בַּשְּׁנִיָּה (בְּלִי לִפֹּל וּבְלִי לָזוּז!), לָשִׂים יָד עַל הָעֵינַיִם, וְלִלְחֹשׁ בְּיִרְאַת שָׁמַיִם אֲמִתִּית: 'ה' אֱלֹקֵינוּ ה' אֶחָד'. רַק מִי שֶׁעוֹשֶׂה אֶת זֶה מִתּוֹךְ רִכּוּז וְאַהֲבָה לַה' - מְנַצֵּחַ אֶת הַיֵּצֶר הָרַע וּמַצְלִיחַ לַעֲצֹר הַכֹּל מִיָּד!"

מקורות: גמרא, מסכת ברכות, דף י"ג עמוד ב': "רבי יהודה הנשיא - היה קורא פסוק ראשון של קריאת שמע תוך כדי הלימוד ועוסק בשמועה שתזכיר את יציאת מצרים... [כדי לא לבטל את] מסירותו ללמד תלמידיו בלי הפסקה". (הוזכר במסמכי העזר - קטעים 35, 150). תלמוד בבלי, מסכת עבודה זרה (על חשיבותו ועושרו הרב של רבי, אך הקדשתו המוחלטת לתורה והענווה). משנה מסכת סוטה ט', ט"ו: "משמת רבי בטלה ענווה ויראת חטא".

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “להלן מערך חינוכי, רגשי ומרתק על רבי יהודה הנשיא (רבינו הקדוש) ומסירותו העצומה ללימוד התורה עם תלמידיו , המעובד במיוחד לבנים בכיתות ב'-ד' (חינוך דתי-לאומי).” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “הוּא יָכוֹל הָיָה לָשֶׁבֶת כָּל הַיּוֹם בְּאַרְמוֹנוֹ הַמְּפֹאָר עַל כֻּרְסוֹת מִקְּטִיפָה וְלָנוּחַ.”?
ג. כיצד הגיב רבי יהודה הנשיא (רבי) למצב המתואר: “אֲנִי לֹא יוֹדֵעַ," עָנָה הַשֵּׁנִי, כַּפּוֹת יָדָיו מַזִּיעוֹת.”?
ד. מה השתנה בין “כֻּלָּם פָּקְחוּ אֶת הָעֵינַיִם וְהִבִּיטוּ יָשָׁר בְּרַבִּי יְהוּדָה הַנָּשִׂיא.” לבין “מִבְּלִי לַעֲצֹר, בְּחָכְמָה עֲצוּמָה וּבִגְאוֹנוּת, הוּא הֵבִיא אֶת דִּבְרֵי הַתּוֹרָה עַל חֲשִׁיבוּת יְצִיאַת עַם…”?
ה. כיצד הגיע הסיפור אל האירוע: “אַתֶּם בְּאֶמְצַע לְשַׂחֵק בַּמִּשְׂחָק הֲכִי שָׁוֶה עִם חֲבֵרִים שֶׁלָּכֶם, פִּתְאוֹם - אִמָּא קוֹרֵאת לָכֶם לַאֲרוּחַת עֶרֶב אוֹ לַעֲזֹר לָהּ.”?

3. מה היה ראשון?

כתבו ליד כל אירוע מספר מ־1 עד 5 לפי סדר התרחשותו בסיפור.

□ בְּקוֹל שֶׁקֶט הֵם הִפְסִיקוּ לִשְׁמֹעַ, עָצְמוּ אֶת הָעֵינַיִם, שָׂמוּ אֶת יְמִינָם הַקְּדוֹשָׁה עַל הַפָּנִים וְקָרְאוּ בְּלַחַשׁ: "שְׁמַע יִשְׂרָאֵל...
□ להלן מערך חינוכי, רגשי ומרתק על רבי יהודה הנשיא (רבינו הקדוש) ומסירותו העצומה ללימוד התורה עם תלמידיו , המעובד במיוחד לבנים בכיתות ב'-ד' (חינוך דתי-לאומי).
□ רַק מִי שֶׁעוֹשֶׂה אֶת זֶה מִתּוֹךְ רִכּוּז וְאַהֲבָה לַה' - מְנַצֵּחַ אֶת הַיֵּצֶר הָרַע וּמַצְלִיחַ לַעֲצֹר הַכֹּל מִיָּד!
□ יוֹם אֶחָד, יָשַׁב רַבִּי יְהוּדָה הַנָּשִׂיא וְלִמֵּד אֶת הַתַּלְמִידִים הֲלָכוֹת עֲמֻקּוֹת בְּיוֹתֵר וְסוֹדוֹת הַתּוֹרָה.
□ הוּא דָּרַשׁ דְּרָשָׁה מְרַתֶּקֶת, עַל כָּךְ שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא גָּאַל אוֹתָנוּ.

לפני ואחרי

מה קרה לפני: “בְּקוֹל שֶׁקֶט הֵם הִפְסִיקוּ לִשְׁמֹעַ, עָצְמוּ אֶת הָעֵינַיִם, שָׂמוּ אֶת יְמִינָם הַקְּדוֹשָׁה עַל הַפָּנִים וְקָרְאוּ בְּלַחַשׁ: "שְׁמַע יִשְׂרָאֵל...”?
מה קרה אחרי: “הוּא דָּרַשׁ דְּרָשָׁה מְרַתֶּקֶת, עַל כָּךְ שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא גָּאַל אוֹתָנוּ.”?

סיבה ותוצאה מתוך המעשה

האירוע הנתוןמה גרם לו?מה הייתה תוצאתו?
רַבִּי קָרָא מִתּוֹךְ זִכְרוֹנוֹ הַמַּדְהִים, פֵּרֵשׁ, הִסְבִּיר וְתֵרֵץ קֻשְׁיוֹת.
יְלָדִים, אֲנַחְנוּ שׁוֹאֲלִים מִיָּד שְׁאֵלָה – קְרִיאַת שְׁמַע זֶה לֹא רַק הַפָּסוּק הָרִאשׁוֹן!

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “קורא פסוק ראשון של ק"ש תוך כדי הלימוד” על רבי יהודה הנשיא (רבי)? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “הָיָה לוֹ רֶצֶף חֲשִׁיבָה אַדִּיר, וְהוּא הִסְבִּיר יְסוֹדוֹת שֶׁאִם הוּא יַפְסִיק בָּאֶמְצַע לְכַמָּה דַּקּוֹת, הַתַּלְמִידִים יְאַבְּדוּ אֶת מַהֲלַךְ…”.
ג. אילו רבי יהודה הנשיא (רבי) לא היה פועל כפי שמתואר ב־“מִבְּלִי לַעֲצֹר, בְּחָכְמָה עֲצוּמָה וּבִגְאוֹנוּת, הוּא הֵבִיא אֶת דִּבְרֵי הַתּוֹרָה עַל חֲשִׁיבוּת יְצִיאַת עַם יִשְׂרָאֵל…” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “אַתֶּם בְּאֶמְצַע לְשַׂחֵק בַּמִּשְׂחָק הֲכִי שָׁוֶה עִם חֲבֵרִים שֶׁלָּכֶם, פִּתְאוֹם - אִמָּא קוֹרֵאת לָכֶם לַאֲרוּחַת עֶרֶב…” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. להלן מערך חינוכי, רגשי ומרתק על רבי יהודה הנשיא (רבינו הקדוש) ומסירותו…
2. הוּא יָכוֹל הָיָה לָשֶׁבֶת כָּל הַיּוֹם בְּאַרְמוֹנוֹ הַמְּפֹאָר עַל…
3. בְּדֶרֶךְ כְּלָל, כְּשֶׁמַּגִּיעַ זְמַן תְּפִלָּה, רַב עָצַר אֶת…
4. הוּא הֵבִין שֶׁהַתּוֹרָה הִיא חַיֶּינוּ וְלֹא רָצָה לְבַטֵּל אוֹתָהּ,…
5. אָהֲבָתוֹ שֶׁל רַבִּי אֲלֵיהֶם, וְאַהֲבָתוֹ לַתּוֹרָה – הָיוּ כָּל…
6. רַק מִי שֶׁעוֹשֶׂה אֶת זֶה מִתּוֹךְ רִכּוּז וְאַהֲבָה לַה' - מְנַצֵּחַ…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “אֲנִי לֹא יוֹדֵעַ," עָנָה הַשֵּׁנִי, כַּפּוֹת יָדָיו מַזִּיעוֹת.”.
מכתב: כתבו לרבי יהודה הנשיא (רבי) בעקבות “יְלָדִים, אֲנַחְנוּ שׁוֹאֲלִים מִיָּד שְׁאֵלָה – קְרִיאַת שְׁמַע זֶה לֹא רַק הַפָּסוּק הָרִאשׁוֹן!”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “אָהֲבָתוֹ שֶׁל רַבִּי אֲלֵיהֶם, וְאַהֲבָתוֹ לַתּוֹרָה – הָיוּ כָּל כָּךְ חֲזָקוֹת!”.
דיווח: כתבו ידיעה קצרה שכותרתה “קורא פסוק ראשון של ק"ש תוך כדי הלימוד”.

רגע מן הסיפורהדמויותמה יאמרו או יעשו?
רַבִּי קָרָא מִתּוֹךְ זִכְרוֹנוֹ הַמַּדְהִים, פֵּרֵשׁ, הִסְבִּיר וְתֵרֵץ קֻשְׁיוֹת.
