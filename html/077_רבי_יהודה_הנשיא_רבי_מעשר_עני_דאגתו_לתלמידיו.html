מעשר עני — דאגתו לתלמידיו

דף עבודה בעקבות רבי יהודה הנשיא (רבי)
שם: כיתה:

המעשה — הסיפור המלא

להלן עיבוד חינוכי, עשיר, מרגש ומרתק על דאגתו העצומה של רבי יהודה הנשיא לתלמידיו וחלוקת מעשר עני לחבריא . הסיפור בנוי במיוחד עבור בנים בכיתות ב'-ד' (חינוך דתי-לאומי), תוך שימוש בכללי סטוריטלינג, תיאורים חושיים, הקפדה על שפה מכבדת ואווירה של קדושה.

הסיפור מנוקד במלואו כנדרש.

פְּתִיחָה: אַרְמוֹנוֹ שֶׁל נְשִׂיא יִשְׂרָאֵל

יְלָדִים, דַּמְיְנוּ עַצְמְכֶם מִתְהַלְּכִים בִּרְחוֹבוֹתֶיהָ שֶׁל הָעִיר צִפּוֹרִי, בִּימֵי תְּקוּפַת הַתַּנָּאִים.

אִם תִּשְׁאֲפוּ אֲוִיר מְלֹא הָרֵאוֹת, תָּרִיחוּ לְפֶתַע רֵיחוֹת מְשַׁכְּרִים שֶׁאֵין כְּמוֹתֶם בְּשׁוּם מָקוֹם אַחֵר: רֵיחַ שֶׁל יְרָקוֹת טְרִיִּים, צְנוֹנִיּוֹת פְּרִיכוֹת, חֲזֶרֶת יְרֻקָּה וְקִשּׁוּאִים מְתוּקִים שֶׁנִּקְטְפוּ זֶה עַתָּה. רֵיחַ שֶׁל תַּבְשִׁילִים הַמַּעֲלִים אֵדִים חַמִּים בְּסִירוֹת נְחֹשֶׁת עֲנָקִיּוֹת.

הַאִם לְשׁוּק הִגַּעְנוּ? לֹא. הִגַּעְנוּ אֶל הֶחָצֵר שֶׁל רַבִּי יְהוּדָה הַנָּשִׂיא, שֶׁכֻּלָּם קָרְאוּ לוֹ פָּשׁוּט "רַבֵּנוּ הַקָּדוֹשׁ".

חֲכָמֵינוּ זִכְרוֹנָם לִבְרָכָה מְסַפְּרִים לָנוּ, שֶׁעַל שֻׁלְחָנוֹ שֶׁל רַבִּי לְעוֹלָם לֹא חָסְרוּ מַאֲכָלִים שֶׁהָיוּ יְקָרִים וּנְדִירִים – אֲפִלּוּ לֹא בְּשִׂיא הַקַּיִץ וַאֲפִלּוּ לֹא בְּעֹמֶק הַחֹרֶף. הַקָּדוֹשׁ בָּרוּךְ הוּא בֵּרֵךְ אֶת רַבִּי בְּעֹשֶׁר שֶׁלֹּא יְאֻמַּן. שְׂדוֹתָיו שְׂתָרְעוּ עַל מֶרְחֲבֵי עֲנָק, מֵאוֹת פּוֹעֲלִים עָבְדוּ בְּכַרְמֵי הַזֵּיתִים וְהָעֲנָבִים שֶׁלּוֹ, וַאֲפִלּוּ הַקֵּיסָר הָרוֹמָאִי, אַנְטוֹנִינוּס, הָיָה שׁוֹלֵחַ לוֹ בְּסֵתֶר שַׂקִּים מְלֵאִים בְּמַטְבְּעוֹת זָהָב אֲדַמְדַּם. הָיָה נִדְמֶה שֶׁמִּי שֶׁגָּר בְּבַיִת כָּזֶה – אֵין לוֹ שׁוּם דְּאָגוֹת בָּעוֹלָם.

אַךְ רַבִּי יְהוּדָה הַנָּשִׂיא הָיָה קָדוֹשׁ מֵרֶחֶם אִמּוֹ. אֶת כָּל כַּסְפּוֹ אָחַז, אַךְ לִבּוֹ לֹא הָיָה בַּזָּהָב וְלֹא בַּשָּׂדוֹת. לִבּוֹ הָיָה נָתוּן רַק בְּמָקוֹם אֶחָד – בְּבֵית הַמִּדְרָשׁ, בַּמָּקוֹם שֶׁבּוֹ יָשְׁבוּ מֵאוֹת תַּלְמִידָיו, הַ"חֶבְרַיָּא", וְלָמְדוּ תּוֹרָה מִבֹּקֶר עַד לַיְלָה בְּקוֹל שֶׁהִרְעִיד אֶת הַקִּירוֹת.

הַצְּלָלִים בְּבֵית הַמִּדְרָשׁ

יוֹם אֶחָד, יָשַׁב רַבֵּנוּ הַקָּדוֹשׁ וְלִמֵּד תּוֹרָה לְכָל עֲדַת הַתַּלְמִידִים. קוֹלוֹ הַנָּעִים וְהֶעָמֹק נִשְׁמַע בֶּחָלָל, כָּל קֻשְׁיָה שֶׁשָּׁאֲלוּ – נַעֲנֵית בְּחִיּוּךְ וּבְחָכְמָה עֲצוּמָה.

אַךְ תּוֹךְ כְּדֵי שֶׁהוּא מְדַבֵּר, עֵינָיו הַבּוֹחֲנוֹת וְהָאוֹהֲבוֹת שֶׁל רַבִּי – סָרְקוּ אֶת הַסַּפְסָלִים.

הוּא שָׂם לֵב לְאֶחָד הַתַּלְמִידִים הַצְּעִירִים שֶׁיָּשַׁב בַּשּׁוּרָה הַשְּׁלִישִׁית. פָּנָיו שֶׁל הַבָּחוּר הָיוּ חִוְּרוֹת מְאֹד. עֵינָיו הָיוּ אֲדֻמּוֹת מֵחֹסֶר שֵׁנָה, וּמִדֵּי פַּעַם, רַבִּי הִבְחִין כִּיצַד הַתַּלְמִיד אוֹחֵז בְּבִטְנוֹ, מִתְכַּוֵּץ מְעַט מִכְּאֵב הָרָעָב, וּמִיָּד מִזְדַּקֵּף וּמַמְשִׁיךְ לְנַגֵּן אֶת הַלִּמּוּד כְּאִלּוּ כְּלוּם לֹא קָרָה.

רַבִּי שָׁתַק, אֲבָל בְּלִבּוֹ – בָּכָה.

כְּשֶׁהִסְתַּיֵּם הַשִּׁיעוּר וְהַתַּלְמִידִים יָצאוּ לְהִתְרַעֲנֵן, קָרָא רַבִּי לַגַּבַּאי שֶׁלּוֹ הַצִּדָּה וְשָׁאַל בְּלַחַשׁ: "אֱמֹר לִי, מַה מַּצָּבוֹ בַּבַּיִת שֶׁל הַתַּלְמִיד הַהוּא? שֶׁל פְּלוֹנִי?".

הַגַּבַּאי אָנַח וְאָמַר: "יִסְלַח לִי רַבֵּנוּ, אַךְ עֹנִי נורָא שׂוֹרֵר בְּבֵיתוֹ. הוּא וּמִשְׁפַּחְתּוֹ לֹא טָעֲמוּ לֶחֶם חַם כְּבָר שָׁבוּעוֹת. אֲנִי חֲשַׁשְׁתִּי לְסַפֵּר לְךָ, כִּי הַתַּלְמִיד אָמַר לִי בְּפֵרוּשׁ: 'אֲנִי מַעֲדִיף לָצוּם, וְלֹא לְבַקֵּשׁ צְדָקָה וּלְבַיֵּשׁ אֶת עַצְמִי עַל לִמּוּד הַתּוֹרָה'. הוּא אִישׁ שֶׁל כָּבוֹד, יְגִיעַ כַּפַּיִם, וְהוּא לֹא יוֹשִׁיט יָד לְנְדָבָה מִשּׁוּם אָדָם בָּעוֹלָם!".

סוֹד הַמַּעֲשֵׂר

רַבִּי יְהוּדָה הַנָּשִׂיא חָזַר לְחַדְרוֹ, הִתְעַטֵּף בְּטַלִּיתוֹ וְחָשַׁב.

"אֵיךְ אוּכַל לַעֲזֹר לוֹ, וְלִשְׁאָר תַּלְמִידַי הָעֲנִיִּים, בְּלִי לִפְגֹּעַ בִּכְבוֹדָם? אִם אֶשְׁלַח אֲלֵיהֶם אֶת עֲבָדַי עִם אַרְגָּזֵי בָּשָׂר וְלֶחֶם – הֵם יִתְבַּיְּשׁוּ לָקַחַת. הֵם לֹא יִרְצוּ לִהְיוֹת לְעָמָל עָלַי. וַאֲנִי... אֲנִי הֲרֵי אוֹהֵב אוֹתָם כְּמוֹ אֶת יְלָדַי שֶׁלִּי! בִּזְכוּת הַתּוֹרָה שֶׁלָּהֶם קַיָּם כָּל הָעוֹלָם!".

וְאָז, הִבְזִיק בְּרֹאשׁוֹ שֶׁל רַבִּי רַעְיוֹן פִּלְאִי, שֶׁרַק אָדָם מָלֵא בְּיִרְאַת שָׁמַיִם יָכוֹל לַחְשֹׁב עָלָיו.

הַתּוֹרָה מְצַוָּה שֶׁבְּשָׁנִים מְסֻיָּמוֹת מִשְּׁנוֹת שְׁמִטָּה, כָּל אָדָם שֶׁיֵּשׁ לוֹ שָׂדֶה חַיָּב לְהַפְרִישׁ "מַעֲשַׂר עָנִי" . כְּלוֹמַר, חֵלֶק מֵהַתְּבוּאָה שֶׁלּוֹ שַׁיָּךְ עַל פִּי חֹק מֵהַשָּׁמַיִם יְשִׁירוֹת לָעֲנִיִּים. זֶה לֹא חֶסֶד אוֹ טוֹבָה שֶׁעוֹשֶׂה בַּעַל הַשָּׂדֶה – הַכֶּסֶף אוֹ הַחִטָּה הַזּוֹ שַׁיָּכִים מַמָּשׁ לֶעָנִי! הַקָּדוֹשׁ בָּרוּךְ הוּא צִוָּה כָּךְ.

רַבִּי קָרָא מִיָּד לִמְנַהֵל שְׂדוֹתָיו. "תִּשְׁמַע לִי הֵיטֵב," אָמַר בְּקוֹל רוֹעֵד מֵהִתְרַגְּשׁוּת, "מָחָר בַּבֹּקֶר, כְּשֶׁתְּקַצְּרוּ אֶת הַחִטָּה הַטּוֹבָה בְּיוֹתֵר וְתִבְצְרוּ אֶת הָעֲנָבִים הֲכִי מְתוּקִים – אֲנִי רוֹצֶה שֶׁתַּפְרִישׁ אֶת 'מַעֲשַׂר הֶעָנִי' בְּצַד. אַל תְּחַלֵּק אוֹתוֹ בַּשּׁוּק לְכָל עוֹבֵר וָשָׁב."

"אֲבָל כְּבוֹד הַנָּשִׂיא," תָּמַהּ מְנַהֵל הָעֲבוֹדָה, "לְמִי לְהַעֲבִיר אֶת הַמַּעֲשֵׂר הָעָצוּם הַזֶּה? יֵשׁ פֹּה טוֹנוֹת שֶׁל חִטָּה וְשֶׁמֶן!"

רַבִּי חִיֵּךְ. חִיּוּךְ רָחָב וְזוֹהֵר. "תַּעֲמִיס אֶת הַכֹּל עַל עֲגָלוֹת יָפוֹת, וְתָבִיא הַכֹּל... אֵלַי. לְחָצֵר שֶׁלְּיַד בֵּית הַמִּדְרָשׁ!".

הָעֲגָלוֹת מַגִּיעוֹת

לְמָחֳרַת, הַשֶּׁמֶשׁ עָלְתָה עַל הָעִיר. תַּלְמִידֵי הַ"חֶבְרַיָּא" יָשְׁבוּ בְּבֵית הַמִּדְרָשׁ. אוֹתוֹ תַּלְמִיד עָנִי עָצַם עֵינַיִם לְרֶגַע, מְנַסֶּה לְהִתְגַּבֵּר עַל תְּחוּשַׁת הַסְּחַרְחֹרֶת שֶׁל הָרָעָב. הוּא אָמַר לְעַצְמוֹ בְּלֵב: "אוּלַי... אוּלַי אֶצְטָרֵךְ לַעֲזֹב אֶת הַלִּמּוּד. אוּלַי אֵין לִי בְּרֵרָה אֶלָּא לָצֵאת לִשְׁבֹּר אֲבָנִים בַּקֹּר, רַק שֶׁיִּהְיֶה מַה לְּהָבִיא לַיְלָדִים בַּבַּיִת...". דִּמְעָה כִּמְעַט זָלְגָה מֵעֵינָיו.

פִּתְאוֹם – נִשְׁמַע רַעַשׁ שֶׁל אֲבָנִים דּוֹרְסוֹת מִבַּחוּץ. חֲרִיקוֹת אוֹפַנֵּי עֵץ עֲנָקִיִּים. קוֹלוֹת שֶׁל צוֹהֲלִים. רֵיחַ מְטֹרָף חָדַר דֶּרֶךְ הַחַלּוֹנוֹת: רֵיחַ שֶׁל קֶמַח טָרִי הַמּוּכָן לַאֲפִיָּה, רֵיחַ שֶׁל קַנְקַנֵּי יַיִן שֶׁזֶּה עַתָּה נִדְרְכוּ.

הַתַּלְמִידִים יָצְאוּ הַחוּצָה בִּפְלִיאָה. מוּלָם עָמְדוּ עֲשָׂרוֹת עֲגָלוֹת מְלֵאוֹת בְּכָל טוּב! וּמִי שֶׁעָמַד שָׁם, בְּאֶמְצַע הֶחָצֵר, עִם שַׁרְווּלִים מֻפְשָׁלִים מְעַט, הָיָה לֹא אַחֵר מֵאֲשֶׁר רַבֵּנוּ הַקָּדוֹשׁ בִּכְבוֹדוֹ וּבְעַצְמוֹ!

"חֶבְרַיָּא יְקָרִים וַאֲהוּבִים שֶׁלִּי!" קָרָא רַבִּי יְהוּדָה הַנָּשִׂיא. "הַשָּׁנָה אָסַפְתִּי יְבוּל מְבֹרָךְ שֶׁל 'מַעֲשַׂר עָנִי'. הַתּוֹרָה מְצַוָּה עָלַי לָתֵת אֶת זֶה לְמִי שֶׁשַּׁיָּךְ לוֹ. לְמִי שֶׁמַּגִּיעַ לוֹ בְּדִין. אֲבָל... מִי קוֹבֵעַ מִי יְקַבֵּל? אֲנִי קוֹבֵעַ! וְהֶחְלַטְתִּי שֶׁאֵין עֲנִיִּים חֲשׁוּבִים יוֹתֵר, קְדוֹשִׁים יוֹתֵר, שֶׁמּוֹסְרִים אֶת נַפְשָׁם, יוֹתֵר מִכֶּם! הַחִטָּה הַזּוֹ הִיא שֶׁלָּכֶם, מֵהַקָּדוֹשׁ בָּרוּךְ הוּא יְשִׁירוֹת. קְחוּ!"

רַבִּי נִגַּשׁ לְאוֹתוֹ תַּלְמִיד חִוֵּר וְרָעֵב. הוּא לֹא הִשְׁפִּיל אֶת מַבָּטוֹ, אֶלָּא הִסְתַּכֵּל יָשָׁר לְעֵינָיו. הוּא הִנִּיחַ בְּיָדָיו שַׂק כָּבֵד שֶׁל קֶמַח, וְאָמַר לוֹ בְּחִיּוּךְ גָּדוֹל וּמְכַבֵּד: "אָחִי הָאֲהוּב, תּוֹדָה שֶׁאַתָּה לוֹמֵד אִתִּי תּוֹרָה. קַח מַה שֶּׁשַּׁיָּךְ לְךָ, כְּדֵי שֶׁתּוּכַל לַחְזֹר לִלְמֹד בְּשִׂמְחָה לְלֹא מַחְשָׁבוֹת שֶׁל צַעַר".

הַתַּלְמִיד חִבֵּק אֶת הַשַּׂק, וּדְמָעוֹת שֶׁל הִתְרַגְּשׁוּת זָלְגוּ מֵעֵינָיו. לֹא רַק שֶׁבִּטְנוֹ שָׂבְעָה בְּאוֹתוֹ יוֹם, אֶלָּא לִבּוֹ הִתְמַלֵּא בְּאוֹר וּבְגַאֲוָה קְדוֹשָׁה. הַנָּשִׂיא בְּעַצְמוֹ הִרְגִּישׁ בִּמְצוּקָתוֹ וּפָתַר אוֹתָהּ בְּצוּרָה שֶׁלֹּא עָשְׂתָה לוֹ שׁוּם בּוּשָׁה, אֶלָּא נָתְנָה לוֹ כָּבוֹד תּוֹרְנִי אֲמִתִּי.

כָּךְ לִמֵּד אוֹתָנוּ רַבִּי יְהוּדָה הַנָּשִׂיא אֶת דַּאֲגָתוֹ הָעֲצוּמָה לְתַלְמִידָיו – לֹא רַק כְּרַב הַמּוֹסֵר שִׁיעוּר, אֶלָּא כְּאַבָּא הַדּוֹאֵג לִילָדָיו וּמַצִּיל אֶת הַתּוֹרָה שֶׁל עַם יִשְׂרָאֵל.

א. שְׁאֵלַת פְּתִיחָה:

"יְלָדִים, מָה עֲדִיפ? שֶׁמִּישֶׁהוּ יִתֵּן לָכֶם מַתָּנָה עֲנָקִית וְיָקָרָה אֲבָל יִזְרוֹק לָכֶם אוֹתָהּ עִם פַּרְצוּף כּוֹעֵס וּמְזַלְזֵל, אוֹ שֶׁיִּתֵּן לָכֶם סֻכָּרִיָּה קְטַנָּה עִם חִיּוּךְ גָּדוֹל וְחִבּוּק?...

הַיּוֹם נִשְׁמַע עַל רַבֵּנוּ הַקָּדוֹשׁ, שֶׁיָּדַע לָתֵת אֶת הַמַּתָּנָה הֲכִי גְּדוֹלָה, בַּחִיּוּךְ וּבַכָּבוֹד הֲכִי גָּדוֹל בָּעוֹלָם."

ב. שְׁאֵלַת סִיּוּם:

"לָמָּה רַבִּי קָרָא לַחִלּוּק אֹכֶל 'מַעֲשַׂר עָנִי' וְלֹא סְתָם 'מַתָּנָה'?

[תֵּן לַיְלָדִים לַעֲנוֹת]

נָכוֹן מְאֹד – כְּדֵי שֶׁהַתַּלְמִידִים יַרְגִּישׁוּ שֶׁזֶּה שֶׁלָּהֶם וְלֹא יִתְבַּיְּשׁוּ לָקַחַת!"

ג. הֶעָרוֹת לְבִצּוּעַ וּסְטוֹרִיטֶלִינְג (Storytelling):

ד. הַמְלָצוֹת לַמַּצֶּגֶת (4 שְׁקוּפִיּוֹת לְלִוּוּי הַסִּפּוּר - הוֹרָאוֹת לְתְמוּנוֹת צוֹנְעוֹת וּקְדוֹשׁוֹת):

ה. פְּעִילוּת מְסַכֶּמֶת - מִשְׂחַק תַּפְקִידִים:

נַעֲמִיד שְׁנֵי יְלָדִים. אֶחָד שָׁכַח צִיּוּד (עִפָּרוֹן/מַחַק) וְהַשֵּׁנִי רוֹצֶה לָתֵת לוֹ אֶת שֶׁלּוֹ. נְבַקֵּשׁ מֵהַזּוּג לְשַׂחֵק זֹאת פַּעַם אַחַת בְּאֹפֶן מַעֲלִיב ("נוּ תִּקַּח אֶת שֶׁלִּי וְזֶהוּ..."), וּפַעַם אַחַת בְּאֹפֶן מְכַבֵּד ("אֲנִי מַמָּשׁ אֶשְׂמַח לַחְלֹק אִתְּךָ, הֵבֵאתִי קַלְמָר כָּפוּל!"). הַכִּתָּה תְּנַתֵּחַ אֶת הַהַרְגָּשָׁה מִשְּׁתֵּי הַבְּחִינוֹת בַּיַּחַס לְמַה שֶּׁעָשָׂה רַבִּי יְהוּדָה הַנָּשִׂיא.

מקורות: דַּאֲגָתוֹ לַתַּלְמִידִים וַחֲלֻקַּת מַעֲשַׂר עָנִי: תַּלְמוּד יְרוּשַׁלְמִי, מַסֶּכֶת סוֹטָה, פֶּרֶק גּ' הֲלָכָה ד' ("רִבִּי הֲוָה יָהֵיב מַעֲשַׂר עָנִי דִּילֵיהּ לַחֶבְרַיָּא..." – חֲלֻקַּת מַעֲשַׂר לְתַלְמִידֵי הַחֲכָמִים כְּדֵי שֶׁיּוּכְלוּ לִלְמֹד בְּשַׁלְוָה, כְּפִי הַמּוּבָא בְּסֵפֶר 'כֹּה עָשׂוּ חֲכָמֵינוּ' וּבְ'סֵפֶר הָאַגָּדָה' סִימָן שט / כע"ח ד' 83). עָשְׁרוֹ הַמֻּפְלָג שֶׁל רַבִּי: תַּלְמוּד בַּבְלִי, מַסֶּכֶת עֲבוֹדָה זָרָה י"א ע"א ("לֹא פָּסַק מֵעַל שֻׁלְחָנוֹ צְנוֹן חֲזֶרֶת וְקִשּׁוּא בֵּין בִּימוֹת הַחַמָּה לְבֵין בִּימוֹת הַגְּשָׁמִים..."). קְשָׁרָיו הַכַּלְכָּלִיִּים עִם אַנְטוֹנִינוּס (מַטְבְּעוֹת זָהָב בְּסֵתֶר): תַּלְמוּד בַּבְלִי, עֲבוֹדָה זָרָה י' ע"ב.

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “להלן עיבוד חינוכי, עשיר, מרגש ומרתק על דאגתו העצומה של רבי יהודה הנשיא לתלמידיו וחלוקת מעשר עני לחבריא .” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “אַךְ רַבִּי יְהוּדָה הַנָּשִׂיא הָיָה קָדוֹשׁ מֵרֶחֶם אִמּוֹ.”?
ג. כיצד הגיב רבי יהודה הנשיא (רבי) למצב המתואר: “הוּא וּמִשְׁפַּחְתּוֹ לֹא טָעֲמוּ לֶחֶם חַם כְּבָר שָׁבוּעוֹת.”?
ד. מה השתנה בין “תִּשְׁמַע לִי הֵיטֵב," אָמַר בְּקוֹל רוֹעֵד מֵהִתְרַגְּשׁוּת, "מָחָר בַּבֹּקֶר, כְּשֶׁתְּקַצְּרוּ אֶת הַחִטָּה…” לבין “וְהֶחְלַטְתִּי שֶׁאֵין עֲנִיִּים חֲשׁוּבִים יוֹתֵר, קְדוֹשִׁים יוֹתֵר, שֶׁמּוֹסְרִים אֶת נַפְשָׁם, יוֹתֵר מִכֶּם!”?
ה. כיצד הגיע הסיפור אל האירוע: “הַיּוֹם נִשְׁמַע עַל רַבֵּנוּ הַקָּדוֹשׁ, שֶׁיָּדַע לָתֵת אֶת הַמַּתָּנָה הֲכִי גְּדוֹלָה, בַּחִיּוּךְ וּבַכָּבוֹד הֲכִי גָּדוֹל בָּעוֹלָם.”?

לפני ואחרי

מה קרה לפני: “כְּלוֹמַר, חֵלֶק מֵהַתְּבוּאָה שֶׁלּוֹ שַׁיָּךְ עַל פִּי חֹק מֵהַשָּׁמַיִם יְשִׁירוֹת לָעֲנִיִּים.”?
מה קרה אחרי: “הוּא לֹא הִשְׁפִּיל אֶת מַבָּטוֹ, אֶלָּא הִסְתַּכֵּל יָשָׁר לְעֵינָיו.”?

סיבה ותוצאה מתוך המעשה

האירוע הנתוןמה גרם לו?מה הייתה תוצאתו?
אַךְ תּוֹךְ כְּדֵי שֶׁהוּא מְדַבֵּר, עֵינָיו הַבּוֹחֲנוֹת וְהָאוֹהֲבוֹת שֶׁל רַבִּי – סָרְקוּ אֶת הַסַּפְסָלִים.
רֵיחַ מְטֹרָף חָדַר דֶּרֶךְ הַחַלּוֹנוֹת: רֵיחַ שֶׁל קֶמַח טָרִי הַמּוּכָן לַאֲפִיָּה, רֵיחַ שֶׁל קַנְקַנֵּי יַיִן שֶׁזֶּה עַתָּה נִדְרְכוּ.

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “מעשר עני — דאגתו לתלמידיו” על רבי יהודה הנשיא (רבי)? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “וְאָז, הִבְזִיק בְּרֹאשׁוֹ שֶׁל רַבִּי רַעְיוֹן פִּלְאִי, שֶׁרַק אָדָם מָלֵא בְּיִרְאַת שָׁמַיִם יָכוֹל לַחְשֹׁב עָלָיו.”.
ג. אילו רבי יהודה הנשיא (רבי) לא היה פועל כפי שמתואר ב־“וְהֶחְלַטְתִּי שֶׁאֵין עֲנִיִּים חֲשׁוּבִים יוֹתֵר, קְדוֹשִׁים יוֹתֵר, שֶׁמּוֹסְרִים אֶת נַפְשָׁם, יוֹתֵר מִכֶּם!” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “הַיּוֹם נִשְׁמַע עַל רַבֵּנוּ הַקָּדוֹשׁ, שֶׁיָּדַע לָתֵת אֶת הַמַּתָּנָה הֲכִי גְּדוֹלָה, בַּחִיּוּךְ וּבַכָּבוֹד הֲכִי גָּדוֹל…” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. להלן עיבוד חינוכי, עשיר, מרגש ומרתק על דאגתו העצומה של רבי יהודה הנשיא…
2. אֶת כָּל כַּסְפּוֹ אָחַז, אַךְ לִבּוֹ לֹא הָיָה בַּזָּהָב וְלֹא…
3. הוּא אִישׁ שֶׁל כָּבוֹד, יְגִיעַ כַּפַּיִם, וְהוּא לֹא יוֹשִׁיט יָד…
4. אוּלַי אֵין לִי בְּרֵרָה אֶלָּא לָצֵאת לִשְׁבֹּר אֲבָנִים בַּקֹּר, רַק…
5. קַח מַה שֶּׁשַּׁיָּךְ לְךָ, כְּדֵי שֶׁתּוּכַל לַחְזֹר לִלְמֹד…
6. הַכִּתָּה תְּנַתֵּחַ אֶת הַהַרְגָּשָׁה מִשְּׁתֵּי הַבְּחִינוֹת…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “הוּא וּמִשְׁפַּחְתּוֹ לֹא טָעֲמוּ לֶחֶם חַם כְּבָר שָׁבוּעוֹת.”.
מכתב: כתבו לרבי יהודה הנשיא (רבי) בעקבות “רֵיחַ מְטֹרָף חָדַר דֶּרֶךְ הַחַלּוֹנוֹת: רֵיחַ שֶׁל קֶמַח טָרִי הַמּוּכָן לַאֲפִיָּה, רֵיחַ שֶׁל קַנְקַנֵּי יַיִן שֶׁזֶּה עַתָּה…”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “הַתַּלְמִיד חִבֵּק אֶת הַשַּׂק, וּדְמָעוֹת שֶׁל הִתְרַגְּשׁוּת זָלְגוּ מֵעֵינָיו.”.
דיווח: כתבו ידיעה קצרה שכותרתה “מעשר עני — דאגתו לתלמידיו”.

המחזה

רגע מן הסיפורהדמויותמה יאמרו או יעשו?
אַךְ תּוֹךְ כְּדֵי שֶׁהוּא מְדַבֵּר, עֵינָיו הַבּוֹחֲנוֹת וְהָאוֹהֲבוֹת שֶׁל רַבִּי – סָרְקוּ אֶת…
וְהֶחְלַטְתִּי שֶׁאֵין עֲנִיִּים חֲשׁוּבִים יוֹתֵר, קְדוֹשִׁים יוֹתֵר, שֶׁמּוֹסְרִים אֶת נַפְשָׁם,…

7. חיבור לחיים מתוך המעשה

א. מה אפשר ללמוד מן הדרך שבה רבי יהודה הנשיא (רבי) פעל באירוע “תִּשְׁמַע לִי הֵיטֵב," אָמַר בְּקוֹל רוֹעֵד מֵהִתְרַגְּשׁוּת, "מָחָר בַּבֹּקֶר, כְּשֶׁתְּקַצְּרוּ אֶת הַחִטָּה הַטּוֹבָה…”?
ב. תארו מצב מחיי הכיתה שבו אפשר ליישם את המסר של “מעשר עני — דאגתו לתלמידיו”.
