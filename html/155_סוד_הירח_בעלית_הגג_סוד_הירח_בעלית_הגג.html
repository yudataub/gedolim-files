🔹 שָׁלָב 1: בֵּרוּר מַקְדִּים וְנִתּוּחַ (לַמּוֹרֶה/לַמְסַפֵּר)

🔹 שָׁלָב 2: הַסִּפּוּר הַמָּלֵא – "סוֹד הַיָּרֵחַ בַּעֲלִיַּת הַגַּג"

פְּתִיחָה: מִי שׁוֹלֵט בַּזְּמַן? יְלָדִים, אִם אֶשְׁאַל אֶתְכֶם: "מִי מַחְלִיט מָתַי מַתְחִיל יוֹם חָדָשׁ?" אַתֶּם בֶּטַח תַּגִּידוּ – "הַשֶּׁמֶשׁ! כְּשֶׁהִיא זָרְחָה – בֹּקֶר!". וּמִי מַחְלִיט מָתַי חֹרֶף אוֹ קַיִץ? "הַקָּדוֹשׁ בָּרוּךְ הוּא, הוּא מֵבִיא אֶת הָעֲנָנִים!". אֲבָל... יֵשׁ סוֹד אֶחָד שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא מָסַר רַק לְעַם יִשְׂרָאֵל. הוּא אָמַר: "אֲנִי מֶלֶךְ הָעוֹלָם, וְאַתֶּם הַבָּנִים וְהַשֻּׁתָּפִים שֶׁלִּי! לָכֵן, אֲנִי לֹא אַחְלִיט מָתַי יִהְיֶה רֹאשׁ חֹדֶשׁ, וּמָתַי יִהְיוּ הַחַגִּים. אַתֶּם תַּחְלִיטוּ, וַאֲנִי אַסְכִּים אִתְּכֶם!". הַאִם שְׁמַעְתֶּם פַּעַם עַל מֶלֶךְ שֶׁנּוֹתֵן לָעָם שֶׁלּוֹ כָּזֶה כֹּחַ אֲדִירוֹ?! ה' אוֹהֵב אֶת עַמּוֹ עַד כְּדֵי כָּךְ שֶׁהוּא שָׂם אֶת הַזְּמַן בַּיָּדַיִם שֶׁלָּנוּ!

הָעֵדִים הַפְּשׁוּטִים עַל הָהָר וְאֵיךְ קוֹבְעִים מָתַי מַתְחִיל חֹדֶשׁ נִיסָן, כְּדֵי שֶׁנֵּדַע מָתַי לַחְגֹּג אֶת פֶּסַח? הַכֹּל תָּלוּי בַּיָּרֵחַ. כְּשֶׁהַיָּרֵחַ קָטָן-קָטָן, מַמָּשׁ נִרְאֶה כְּמוֹ פַּס כֶּסֶף זָעִיר וּמֵאִיר שֶׁבּוֹקֵעַ בַּשָּׁמַיִם הַשְּׁחֹרִים שֶׁל הַלַּיְלָה – זֶה סִימָן שֶׁהִתְחִיל חֹדֶשׁ חָדָשׁ.

דַּמְיְנוּ אֶת כְּנַעַן וְשִׁמְעוֹן, שְׁנֵי רוֹעֵי צֹאן עֲנִיִּים וּפְשׁוּטִים ("הֶדְיוֹטוֹת"). הֵם יָשְׁבוּ עַל הָהָר, מִתְחַמְּמִים לְיַד הַמְּדוּרָה, וּפִתְאוֹם... אֶחָד מֵהֶם הִצְבִּיעַ לַשָּׁמַיִם. "תִּרְאֶה!" זָעַק שִׁמְעוֹן, "הַמּוֹלָד! הַיָּרֵחַ הֶחָדָשׁ!". שָׂמְחוּ הָרוֹעִים. הֵם לָקְחוּ אֶת מַקְלוֹתֵיהֶם, לֹא וִתְּרוּ עַל אַף דַּקָּה, וְהִתְחִילוּ לָרוּץ בְּכָל כֹּחָם וּמַרְצָם, דֶּרֶךְ סְלָעִים וְשׁוּעָלִים נַהֲמָנִיִּים, יָשָׁר אֶל הָעִיר הַקְּדוֹשָׁה יַבְנֶה! כִּי שָׁם יָשְׁבָה תִּקְוַת יִשְׂרָאֵל, שָׁם יָשְׁבָה סַנְהֶדְרִין, וּבְרֹאשָׁהּ: רַבָּן גַּמְלִיאֵל הַנָּשִׂיא.

עֲלִיַּת הַגַּג הַסּוֹדִית כְּנַעַן וְשִׁמְעוֹן הִגִּיעוּ לְיַבְנֶה עֲיֵפִים וְנוֹשְׁמִים בִּכְבֵדוּת. הֵם הֻכְנְסוּ בְּיִרְאַת שָׁמַיִם לַחֶדֶר הַגָּדוֹל. רַבָּן גַּמְלִיאֵל, שֶׁעֵינָיו חֲכָמוֹת וְזוֹהֲרוֹת כְּאֵשׁ תּוֹרָה, הִבִּיט בָּהֶם. יְלָדִים, זֶה אָחֳרָיוּת עֲצוּמָה! אִם רַבָּן גַּמְלִיאֵל טוֹעֶה בְּהַחְלָטָתוֹ, כָּל עַם יִשְׂרָאֵל יֹאכְלוּ חָמֵץ בְּפֶסַח הַלֹּא נָכוֹן חָלִילָה! אוֹ יָצוּמוּ יוֹם כִּפּוּר בְּיוֹם רָגִיל! אֵיךְ נָשִׂיא יִשְׂרָאֵל יָכוֹל לִסְמֹךְ עַל שְׁנֵי אֲנָשִׁים פְּשׁוּטִים בְּעִנְיָן כָּל כָּךְ חָשׁוּב?

הַמִּשְׁנָה מְגַלָּה לָנוּ אֶת סוֹד חָכְמָתוֹ שֶׁל נְשִׂיא יִשְׂרָאֵל. "עֲלוּ אִתִּי לַעֲלִיַּת הַגַּג" לָחַשׁ לָהֶם רַבָּן גַּמְלִיאֵל. הָרוֹעִים טִפְּסוּ בְּמַדְרֵגוֹת עֵץ שֶׁחָרְקוּ חֶרֶשׁ, הַדֶּלֶת נִפְתְּחָה... וּפִיהֶם נִפְעַר בְּתַדְהֵמָה! עַל קִירוֹת עֲלִיַּת הַגַּג, וְעַל טַבְלַת עֵץ רְחָבָה מְיֻחֶדֶת ("דְּמוּת צוּרוֹת לְבָנוֹת הָיוּ לוֹ לְרַבָּן גַּמְלִיאֵל בַּטַּבְלָא וּבַכֹּתֶל"), הָיוּ חֲקוּקוֹת צוּרוֹת-צוּרוֹת שֶׁל יָרֵחַ מַבְרִיק מִכֶּסֶף טָהוֹר! הָיָה שָׁם יָרֵחַ רָחָב, וְיָרֵחַ שָׁמֵן, יָרֵחַ דַּק מְאֹד נָטוּי שְׂמֹאלָה, וְיָרֵחַ דַּק מְאֹד נָטוּי יָמִינָה, וְיָרֵחַ עִם סִימָנִים וּפְגָמִים. זֶה הָיָה מַמָּשׁ "מִצְפֵּה כּוֹכָבִים תּוֹרָנִי"!

הֲכָזֶה רָאִיתָ אוֹ כָזֶה? הוֹשִׁיב רַבָּן גַּמְלִיאֵל אֶת הָרוֹעֶה הָרִאשׁוֹן לְבַדּוֹ. הוּא הֵרִים אֶת אֶצְבָּעוֹ אֶל הַקִּיר וְשָׁאַל בְּקוֹל עָמֹק וּמַלְכוּתִי: "בְּנִי... הֲכָזֶה רָאִיתָ אוֹ כָזֶה?!" הוּא הִצְבִּיעַ עַל הַיָּרֵחַ הַנָּטוּי שְׂמֹאלָה, וְאָז עַל הַיָּרֵחַ הַנָּטוּי יָמִינָה. הוּא שָׁאַל בְּדִיּוּק הֵיכָן הָיְתָה הַשֶּׁמֶשׁ בְּאוֹתוֹ רֶגַע, הַאִם הַיָּרֵחַ הָיָה גָּבוֹהַּ בַּשָּׁמַיִם, אוֹ נָמוּךְ סָמוּךְ לָהָרִים?

הָעֵד הַפָּשׁוּט הִבִּיט בַּטַּבְלָא, וּמִיָּד אָמַר: "רַבֵּנוּ! אֲנִי בָּטוּחַ! רָאִיתִי בְּדִיּוּק אֶת הַצּוּרָה הַזּוֹ! וְהִיא הָיְתָה נְמוּכָה מְאֹד!". אָז הִכְנִיס רַבָּן גַּמְלִיאֵל אֶת הָעֵד הַשֵּׁנִי וְשָׁאַל אוֹתוֹ שְׁאֵלוֹת דּוֹמוֹת בְּעֶזְרַת תְּמוּנוֹת הַיָּרֵחַ. כַּאֲשֶׁר שְׁנֵיהֶם אָמְרוּ בְּדִיּוּק אֶת אוֹתוֹ הַדָּבָר, וְהַדָּבָר הִתְאִים לַחִשּׁוּבֵי הַמַּתֵמָטִיקָה שֶׁל הַשָּׁמַיִם שֶׁרַבָּן גַּמְלִיאֵל יָדַע כְּלִפְנֵי וְלִפְנִים – אָז הוּא יָדַע: הֵם דּוֹבְרֵי אֱמֶת! הַחֹדֶשׁ הִתְחִיל!

"מְקֻדָּשׁ, מְקֻדָּשׁ!" יְלָדִים, אָז הִגִּיעַ הָרֶגַע הַגָּדוֹל. רַבָּן גַּמְלִיאֵל נַעֲמַד עַל רַגְלָיו. שִׁבְעִים וְאֶחָד רַבָּנֵי הַסַּנְהֶדְרִין קָמוּ כְּאִישׁ אֶחָד. רַבָּן גַּמְלִיאֵל זָקַף רֹאשׁוֹ וְהִכְרִיז לְכָל עַם יִשְׂרָאֵל: "מְקֻדָּשׁ!" וְכָל הָעָם הֶעָצוּם, עֲשָׂרוֹת תַּלְמִידִים וְרוֹעִים וִיהוּדִים קְדוֹשִׁים צָעֲקוּ בַּחֲזָרָה: "מְקֻדָּשׁ! מְקֻדָּשׁ!".

וְאַתֶּם יוֹדְעִים מָה קָרָה בְּאוֹתָהּ דַּקָּה מַמָּשׁ לְמַעְלָה בַּשָּׁמַיִם? הַמַּלְאָכִים שָׁאֲלוּ אֶת הַקָּדוֹשׁ בָּרוּךְ הוּא: "רִבּוֹנוֹ שֶׁל עוֹלָם, מָתַי יִהְיֶה רֹאשׁ הַשָּׁנָה הַשָּׁנָה?". וְהַקָּדוֹשׁ בָּרוּךְ הוּא חִיֵּךְ לַמַּלְאָכִים וְאָמַר לָהֶם: "אַל תִּשְׁאֲלוּ אוֹתִי! רְדוּ לְמַטָּה לִיְשִׁיבַת יַבְנֶה, וְתִשְׁמְעוּ מָה רַבָּן גַּמְלִיאֵל וְהָעֵדִים פָּסְקוּ! אֲנִי נוֹתֵן אֶת הַכֹּחַ לַבָּנִים הָאֲהוּבִים שֶׁלִּי! בִּזְכוּת מַה שֶּׁהֵם פָּסְקוּ לְמַטָּה, כָּךְ אֲנִי מְסַדֵּר אֶת הַזְּמַן לְמַעְלָה!".

אֵיזוֹ אַהֲבָה מַדְהִימָה ה' רָחַשׁ לְעַמּוֹ, "ה' אוֹהֵב אֶת עַמּוֹ – וְנָתַן לָנוּ אֶת הַזְּכוּת לִקְבֹּעַ אֶת הַזְּמַנִּים!".

🔹 שָׁלָב 3: חִבּוּר עֶרְכִּי – ME-WE-YOU

🔹 שָׁלָב 4: מַדְרִיךְ לַמּוֹרֶה / לַמְסַפֵּר (לְשִׁלּוּב בַּמַּצֶּגֶת וּבַשִּׁעוּר)

א. שְׁאֵלַת פְּתִיחָה (לְעוֹרֵר זִהוּי וְהִשְׁתַּתְּפוּת): "יְלָדִים, נָנִיחַ שֶׁאִמָּא יָצְאָה לְסִדּוּרִים וְהִשְׁאִירָה אֶתְכֶם לְבַד בַּבַּיִת, וְהִיא הִשְׁאִירָה לָכֶם לְהַחְלִיט: 'יְלָדִים, אַתֶּם קוֹבְעִים הַיּוֹם מָתַי אוֹכְלִים צָהֳרַיִם וּמָתַי אֲנִי חוֹזֶרֶת הַבַּיְתָה!'. מָה זֶה אוֹמֵר? זֶה אוֹמֵר שֶׁהִיא סוֹמֶכֶת עֲלֵיכֶם מְאֹד, נָכוֹן? הַיּוֹם נִלְמַד שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא עָשָׂה מַשֶּׁהוּ שֶׁאַף מֶלֶךְ לֹא עוֹשֶׂה – הוּא נָתַן לָנוּ לְהַחְלִיט אֶת לוּחַ הַשָּׁנָה שֶׁלּוֹ! בּוֹאוּ נִרְאֶה אֵיךְ גְּדוֹלֵי יִשְׂרָאֵל, רַבָּן גַּמְלִיאֵל, עָשׂוּ אֶת זֶה לְלֹא טָעוּת!".

ב. הֶעָרוֹת בִּצּוּעַ וְהַמְחָשָׁה כִּתָּתִית:

ג. הַמְלָצוֹת שִׁלּוּב בְּמַצֶּגֶת (תְּמוּנוֹת צְנוּעוֹת וְרוּחָנִיּוֹת שֶׁמַּרְאוֹת שְׁלִיטָה וַהֲדָרָה):

ד. פְּעִילוּת מְסַכֶּמֶת "בְּדִיקַת עֵדִים בַּכִּתָּה":

📚 מְקוֹרוֹת (מִתּוֹךְ הַקְּבָצִים וּמַחְבֶּרֶת הַמּוֹרֶה):

מקורות: המקורות מופיעים בקובץ המקורי.

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “🔹 שָׁלָב 2: הַסִּפּוּר הַמָּלֵא – "סוֹד הַיָּרֵחַ בַּעֲלִיַּת הַגַּג” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “כְּשֶׁהַיָּרֵחַ קָטָן-קָטָן, מַמָּשׁ נִרְאֶה כְּמוֹ פַּס כֶּסֶף זָעִיר וּמֵאִיר שֶׁבּוֹקֵעַ בַּשָּׁמַיִם הַשְּׁחֹרִים שֶׁל הַלַּיְלָה – זֶה סִימָן…”?
ג. כיצד הגיב סוד הירח בעלית הגג למצב המתואר: “רַבָּן גַּמְלִיאֵל, שֶׁעֵינָיו חֲכָמוֹת וְזוֹהֲרוֹת כְּאֵשׁ תּוֹרָה, הִבִּיט בָּהֶם.”?
ד. מה השתנה בין “עַל קִירוֹת עֲלִיַּת הַגַּג, וְעַל טַבְלַת עֵץ רְחָבָה מְיֻחֶדֶת ("דְּמוּת צוּרוֹת לְבָנוֹת הָיוּ לוֹ לְרַבָּן…” לבין “רַבָּן גַּמְלִיאֵל זָקַף רֹאשׁוֹ וְהִכְרִיז לְכָל עַם יִשְׂרָאֵל: "מְקֻדָּשׁ!" וְכָל הָעָם הֶעָצוּם, עֲשָׂרוֹת…”?
ה. כיצד הגיע הסיפור אל האירוע: “🔹 שָׁלָב 4: מַדְרִיךְ לַמּוֹרֶה / לַמְסַפֵּר (לְשִׁלּוּב בַּמַּצֶּגֶת וּבַשִּׁעוּר)”?

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “סוד הירח בעלית הגג” על סוד הירח בעלית הגג? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “הַמִּשְׁנָה מְגַלָּה לָנוּ אֶת סוֹד חָכְמָתוֹ שֶׁל נְשִׂיא יִשְׂרָאֵל.”.
ג. אילו סוד הירח בעלית הגג לא היה פועל כפי שמתואר ב־“רַבָּן גַּמְלִיאֵל זָקַף רֹאשׁוֹ וְהִכְרִיז לְכָל עַם יִשְׂרָאֵל: "מְקֻדָּשׁ!" וְכָל הָעָם הֶעָצוּם, עֲשָׂרוֹת תַּלְמִידִים…” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “🔹 שָׁלָב 4: מַדְרִיךְ לַמּוֹרֶה / לַמְסַפֵּר (לְשִׁלּוּב בַּמַּצֶּגֶת וּבַשִּׁעוּר)” מחזק את המסר של הסיפור?

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “רַבָּן גַּמְלִיאֵל, שֶׁעֵינָיו חֲכָמוֹת וְזוֹהֲרוֹת כְּאֵשׁ תּוֹרָה, הִבִּיט בָּהֶם.”.
מכתב: כתבו לסוד הירח בעלית הגג בעקבות “הוּא הֵרִים אֶת אֶצְבָּעוֹ אֶל הַקִּיר וְשָׁאַל בְּקוֹל עָמֹק וּמַלְכוּתִי: "בְּנִי...”.
