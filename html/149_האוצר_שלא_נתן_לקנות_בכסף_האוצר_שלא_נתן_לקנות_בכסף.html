🔹 שָׁלָב 1: בֵּרוּר מַקְדִּים וְנִתּוּחַ (לַמּוֹרֶה/לַמְסַפֵּר)

🔹 שָׁלָב 2: הַסִּפּוּר הַמָּלֵא – "הָאוֹצָר שֶׁלֹּא נִתָּן לִקְנוֹת בְּכֶסֶף"

פְּתִיחָה: הָעָשִׁיר הַסּוֹדִי יְלָדִים, אִם הָיִיתִי מְסַפֵּר לָכֶם עַל אָדָם שֶׁהוּא מְפַרְנֵס אֶת כָּל הָעוֹלָם כֻּלּוֹ... אָדָם שֶׁבִּזְכוּתוֹ יוֹרֵד גֶּשֶׁם, בִּזְכוּתוֹ שְׂדוֹת הַחִטָּה צוֹמְחִים מְלֵאִים בְּתַבּוּאָה, בִּזְכוּתוֹ צֶמֶר הַכְּבָשִׂים גָּדֵל וְעֵצִים מְלֵאִים בְּפֵרוֹת מְתוּקִים – אֵיךְ הֱיִיתֶם מְדַמְיְנִים אוֹתוֹ? בְּוַדַּאי הֱיִיתֶם צוֹיְרִים בְּדִמְיוֹנְכֶם מֶלֶךְ עָשִׁיר, שֶׁיּוֹשֵׁב בְּאַרְמוֹן עֲנָק בָּנוּי מִשַּׁיִשׁ וְזָהָב, עִם עֲשָׂרוֹת מְשָׁרְתִים, שֻׁלְחָנוֹת אֲרֻכִּים הָעֲמוּסִים בְּמַאַפִים טְרִיִּים, עוּגוֹת קַצֶּפֶת, מְרָקִים מַהְבִּילִים וּפֵרוֹת נְדִירִים. אָדָם שֶׁמַּאֲכִיל אֶת כָּל הָעוֹלָם, צָרִיךְ לִהְיוֹת הָאָדָם הַשָּׂבֵעַ בְּיוֹתֵר בָּעוֹלָם, נָכוֹן?

אֲבָל בְּאֶרֶץ יִשְׂרָאֵל, בִּימֵי בֵּית הַמִּקְדָּשׁ הַשֵּׁנִי, חַי הַתַּנָּא הַקָּדוֹשׁ רַבִּי חֲנִינָא בֶּן דּוֹסָא. וְרַבִּי חֲנִינָא הָיָה הָאָדָם הֲכִי שׁוֹנֶה מֵהַדִּמְיוֹן הַזֶּה. רַבִּי חֲנִינָא וְאִשְׁתּוֹ חָיוּ בְּבַיִת קָטָן וּפָשׁוּט עָשׂוּי מֵאֲבָנִים וְטִיט. לֹא הָיוּ בּוֹ כִּסְאוֹת מְרֻפָּדִים אוֹ וִילוֹנוֹת מִמֶּשִׁי. וּמָה גַּם לֹא הָיָה בּוֹ? לֹא הָיָה בּוֹ אֹכֶל.

קַב שֶׁל חֲרוּבִים יְבֵשִׁים דַּמְיְנוּ בֹּקֶר קַר שֶׁל יּוֹם חוֹל. רַבִּי חֲנִינָא קָם מִשְּׁנָתוֹ לִפְנֵי עֲלוֹת הַשַּׁחַר. הַבֶּטֶן מְקַרְקֶרֶת מְעַט מֵרָעָב. הוּא נִגַּשׁ לָאָרוֹן הַקָּטָן בַּמִּטְבָּח. אֵין שָׁם כִּכַּר לֶחֶם רַכָּה, גַּם לֹא יְרָקוֹת. יֵשׁ שָׁם קְעָרָה קְטַנָּה אַחַת, וּבְתוֹכָהּ חֲרוּבִים. פֵּרוֹת חוּמִים, נוֹקְשִׁים וִיבֵשִׁים, שֶׁקָּשֶׁה מְאֹד לִלְעֹס אוֹתָם.

חֲכָמֵינוּ זִכְרוֹנָם לִבְרָכָה מְגַלִּים לָנוּ סוֹד עָצוּם עַל הַצַּדִּיק הַזֶּה: הַקָּדוֹשׁ בָּרוּךְ הוּא אָהַב כָּל כָּךְ אֶת הַתְּפִלּוֹת וְאֶת הַתּוֹרָה שֶׁל רַבִּי חֲנִינָא, עַד שֶׁבְּכָל יוֹם וָיוֹם הָיְתָה יוֹצֵאת "בַּת קוֹל" (קוֹל שֶׁמַּגִּיעַ מִן הַשָּׁמַיִם), וּמַכְרִיזָה אֶת הַדְּבָרִים הַבָּאִים מִכִּסֵּא הַכָּבוֹד: "כָּל הָעוֹלָם כֻּלּוֹ נִזּוֹן (מְקַבֵּל שֶׁפַע וּמָזוֹן) בִּשְׁבִיל חֲנִינָא בְּנִי! וַחֲנִינָא בְּנִי – דַּיּוֹ בְּקַב חֲרוּבִים מֵעֶרֶב שַׁבָּת לְעֶרֶב שַׁבָּת!". כָּל הַשָּׁבוּעַ כֻּלּוֹ, מִיּוֹם רִאשׁוֹן וְעַד יוֹם שִׁשִּׁי, רַבִּי חֲנִינָא אָכַל רַק "קַב חֲרוּבִים". קַב זוֹ מִדָּה מַמָּשׁ קְטַנָּה (כְּמוֹ קֻפְסַת אוֹכֶל קְטַנָּה מְאוֹד שֶׁלְּוֹקְחִים לְבֵית הַסֵּפֶר). כַּמָּה חֲרוּבִים נוֹקְשִׁים וּמְעַט מַיִם.

וְעַכְשָׁו, יְלָדִים, אֲנִי שׁוֹאֵל אֶתְכֶם: אִם מִישֶׁהוּ מֵאִתָּנוּ הָיָה צָרִיךְ לֶאֱכֹל רַק חֲרוּבִים קָשִׁים מֶשֶׁךְ שָׁבוּעַ שָׁלֵם, וְלֹא הָיָה לוֹ גַּם לֶחֶם אוֹ חֲבִיתָה, אֵיךְ הוּא הָיָה נִרְאֶה? וַדַּאי הוּא הָיָה הוֹלֵךְ שָׁפוּף, מִסְכֵּן, בּוֹכֶה כָּל הַזְּמַן, כּוֹעֵס וּמְרוֹמְרָם. "לָמָּה אֵין לִי?! לָמָּה לַשָּׁכֵן כֵּן יֵשׁ?!".

הָאוֹצָר שֶׁבְּתוֹךְ הַבַּיִת אֲבָל בּוֹאוּ נָצִיץ יַחַד לְתוֹךְ בֵּיתוֹ שֶׁל רַבִּי חֲנִינָא בֶּן דּוֹסָא כְּדֵי לִרְאוֹת מָה קוֹרֶה שָׁם בֶּאֱמֶת. הָרֶקַע מֵאֲחוֹרֵי דַּלְתוֹת הָעֵץ הַפְּשׁוּטוֹת אֵינוֹ קוֹלוֹת שֶׁל בֶּכִי, וְלֹא קוֹלוֹת שֶׁל קְטָטָה. מִתּוֹךְ הַבַּיִת בּוֹקְעִים סִלְסוּלִים שֶׁל נִגּוּן מָתוֹק וְנָעִים! קוֹל שֶׁמְּחַמֵּם אֶת הַלֵּב מִיָּד. רַבִּי חֲנִינָא יוֹשֵׁב לְיַד שֻׁלְחָן הָעֵץ. סֵפֶר קָדוֹשׁ וְעַתִּיק מֻנָּח לְפָנָיו, וְהוּא לוֹמֵד בּוֹ תּוֹרָה. אִשְׁתּוֹ יוֹשֶׁבֶת בַּחֶדֶר, מַקְשִׁיבָה לְדִבְרֵי הַתּוֹרָה, וּפָנֶיהָ – הֵן מְאִירוֹת כְּמוֹ פָּנִים שֶׁל מַלְכָּה. גַּם הַיְלָדִים סְבִיבָם, יוֹשְׁבִים וּמַקְשִׁיבִים.

אֵין לָהֶם כֶּסֶף! אֵין לָהֶם עֹשֶׁר! הַתַּנּוּר רֵיק! אֲבָל... יֵשׁ לָהֶם תּוֹרָה! וְהַתּוֹרָה, אָמַר רַבִּי חֲנִינָא לִבְנֵי מִשְׁפַּחְתּוֹ, הִיא מְתוּקָה יוֹתֵר מִכָּל דְּבַשׁ שֶׁבָּעוֹלָם. כְּשֶׁלּוֹמְדִים תּוֹרָה שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא, הַנְּשָׁמָה כָּל כָּךְ אוֹהֶבֶת אֶת זֶה, וְכָל כָּךְ נֶהֱנֵית מִזֶּה, שֶׁפָּשׁוּט מַרְגִּישִׁים שְׂבֵעִים! הַמִּלִּים שֶׁל הַתּוֹרָה מִלְּאוּ לָהֶם אֶת הַלֵּב בְּשִׂמְחָה עֲצוּמָה, עַד שֶׁהֵם בִּכְלָל לֹא הִרְגִּישׁוּ אֶת הָרָעָב וְאֶת הָעֹנִי הַנּוֹרָא. "הַקָּדוֹשׁ בָּרוּךְ הוּא נָתַן לָנוּ אֶת מַתְּנַת הַתּוֹרָה", חִיֵּךְ רַבִּי חֲנִינָא לְאִשְׁתּוֹ בְּאַהֲבָה רַבָּה, "מִי שֶׁיֵּשׁ לוֹ תּוֹרָה בַּבַּיִת – יֵשׁ לוֹ אֶת הַכֹּל!". הוּא שָׂמַח וְהוֹדָה לַה' עַל קַב הַחֲרוּבִים הַקָּטָן, וְלֹא בִּקֵּשׁ לְעַצְמוֹ שׁוּם עֹשֶׁר.

הַהִסְתַּפְּקוּת שֶׁמְּאִירָה אֶת הָעוֹלָם בַּחוּץ, בָּרְחוֹבוֹת יְרוּשָׁלַיִם אוֹ בַּכְּפָרִים שֶׁמִּסָּבִיב, אֲנָשִׁים אָכְלוּ לֶחֶם חַם וְצָלִי בָּשָׂר. הֵם לֹא יָדְעוּ שֶׁהַגֶּשֶׁם שֶׁמַּשְׁקֶה אֶת שְׂדוֹתֵיהֶם הִגִּיעַ בִּזְכוּת אוֹתוֹ חָכָם צָנוּעַ שֶׁיּוֹשֵׁב בַּבַּיִת הֶעָנִי, לוֹעֵס חָרוּב וְאוֹמֵר עָלָיו בְּרָכָה בְּכַוָּנָה עֲצוּמָה וּבְשִׂמְחָה עַד הַשָּׁמַיִם.

בְּזָכוּת הַהִסְתַּפְּקוּת בַּמּוּעָט שֶׁלּוֹ (שֶׁהוּא יָדַע שֶׁמָּה שֶׁיֵּשׁ זֶה בְּדִיּוּק מָה שֶׁה' רָצָה בִּשְׁבִילוֹ), וּבִזְכוּת הַשִּׂמְחָה הַטְּהוֹרָה בְּמִלִּים שֶׁל תּוֹרָה – עָמַד כָּל הָעוֹלָם כֻּלּוֹ. כִּי הַתּוֹרָה, יְלָדִים, הִיא לֹא רַק סֵפֶר – הִיא הָאוֹר הֲכִי מְשַׂמֵּחַ בַּיְקוּם, אוֹר שֶׁאֲפִלּוּ מַטְבְּעוֹת שֶׁל זָהָב וּכְלֵי קְרִיסְטָל נוֹצְצִים לֹא יְכוֹלִים לִקְנוֹת!

🔹 שָׁלָב 3: חִבּוּר עֶרְכִּי – ME-WE-YOU

🔹 שָׁלָב 4: מַדְרִיךְ לַמּוֹרֶה / לַמְסַפֵּר (לְשִׁלּוּב בַּמַּצֶּגֶת וּבַשִּׁעוּר)

א. שְׁאֵלַת פְּתִיחָה (לְעוֹרֵר זִהוּי וְהִשְׁתַּתְּפוּת): "יְלָדִים, אִם עַכְשָׁו הָיָה נִכְנַס הֵנָּה לַכִּתָּה הָאִישׁ הֲכִי עָשִׁיר בָּעוֹלָם, בְּדִיּוּק בָּרֶגַע שֶׁהוּא אִבֵּד אֶת הַכֶּסֶף שֶׁלּוֹ. אֵיךְ נִרְאֶה הַפַּרְצוּף שֶׁלּוֹ? (עָצוּב, כּוֹעֵס). וְאִם יִכָּנֵס הֵנָּה אָדָם שֶׁאֵין לוֹ שׁוּם צַעֲצוּעַ וְשׁוּם מַמְתָּק, אֲבָל... הוּא הֲכִי אָהוּב עַל הַקָּדוֹשׁ בָּרוּךְ הוּא, וְהוּא לוֹמֵד תּוֹרָה כָּל הַיּוֹם – אֵיךְ יֵרָאֶה הַפַּרְצוּף שֶׁלּוֹ? הַיּוֹם נַכִּיר תַּנָּא כָּזֶה מְדֻיָּק!"

ב. הֶעָרוֹת בִּצּוּעַ (לַמְסַפֵּר):

ג. הַמְלָצוֹת שִׁלּוּב בְּמַצֶּגֶת (15-20 שְׁקוּפִיּוֹת עִם טֶקְסְט מְנֻקָּד):

ד. פְּעִילוּת מְסַכֶּמֶת מֻמְלֶצֶת:

מקורות: המקורות מופיעים בקובץ המקורי.

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “🔹 שָׁלָב 2: הַסִּפּוּר הַמָּלֵא – "הָאוֹצָר שֶׁלֹּא נִתָּן לִקְנוֹת בְּכֶסֶף” — מה קרה מיד לאחר מכן?
