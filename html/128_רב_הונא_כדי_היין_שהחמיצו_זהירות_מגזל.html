כדי היין שהחמיצו — זהירות מגזל

דף עבודה בעקבות רב הונא
שם: כיתה:

המעשה — הסיפור המלא

להלן מערך שיעור וסיפור חינוכי, רגשי ומרתק על רב הונא וכדי היין שהחמיצו . הסיפור מעובד בדיוק רב על פי המקורות, מותאם לבנים בכיתות ב'-ד' (חינוך דתי-לאומי), ומעביר מסר עצום של יושר מוחלט, לקיחת אחריות, וההבנה שלעולם אין להצדיק מעשה פסול בטענה ש"השני התחיל" או "הוא עשה לי קודם".

הסיפור מנוקד במלואו כנדרש, שזור בתיאורים חושיים ומבנה עלילתי סוחף.

פְּתִיחָה: רֵיחַ הַיַּיִן הַמָּתוֹק

יְלָדִים, אִם הֱיִיתֶם עוֹבְרִים לִפְנֵי שָׁנִים רַבּוֹת בְּרַחֲבוֹת הָעִיר סוּרָא שֶׁבְּבָבֶל, הֱיִיתֶם וַדַּאי מַרְגִּישִׁים בָּאַפְּכֶם רֵיחַ מְתוּק וְנִפְלָא שֶׁל עֲנָבִים טְרִיִּים וְיַיִן מְשֻׁבָּח, שֶׁנִּשָּׂא בָּאֲוִיר. הָרֵיחַ הַזֶּה הִגִּיעַ מִמַּרְתְּפֵי הַיַּיִן הָעֲנָקִיִּים שֶׁל גְּדוֹל הַדּוֹר, רַב הוּנָא .

רַב הוּנָא הָיָה רֹאשׁ הַיְשִׁיבָה הַגָּדוֹל בְּיוֹתֵר, צַדִּיק עָצוּם, וְגַם אָדָם עָשִׁיר מְאֹד. אֲבָל הָעֹשֶׁר לֹא הָפַךְ אוֹתוֹ לְגַאַוְתָן, חָלִילָה! הָיָה לוֹ מִנְהָג מְיֻחָד: כַּאֲשֶׁר הָיָה מִתְיַשֵּׁב לֶאֱכֹל אֶת סְעוּדָתוֹ, הָיָה פּוֹתֵחַ לִרְוָחָה אֶת דַּלְתוֹת בֵּיתוֹ וְקוֹרֵא בְּקוֹל לְרְחוֹב: "כָּל מִי שֶׁרָעֵב – יָבוֹא וְיֹאכַל!". אֲלָפִים שֶׁל תַּלְמִידִים וַעֲנִיִּים הָיוּ חַיִּים מֵהַחֶסֶד שֶׁלּוֹ.

וּמֵהֵיכָן הִגִּיעַ כַּסְפּוֹ הָרַב? מֵהַכְּרָמִים. מִשְּׂדוֹת עֲנָבִים עֲנָקִיִּים, מֵהֶם הָיוּ מְיַצְּרִים אֶת הַיַּיִן הֲכִי טוֹב בְּבָבֶל, שֶׁהָיָה נִשְׁמָר בְּמַרְתְּפֵי קְרִירִים וַאֲפֵלִים עַד שֶׁבָּשַׁל לְקְרַאת שִׁוּוּק.

הָאָסוֹן בַּמַּרְתֵּף הַקָּר

וְהִנֵּה, בְּיוֹם אֶחָד יָרְדָה צָרָה גְּדוֹלָה עַל בֵּיתוֹ שֶׁל רַב הוּנָא.

דַּמְיְנוּ אֶת מְנַהֵל הָעֲבוֹדָה שֶׁלּוֹ יוֹרֵד אֶל מַרְתֵּף הַיַּיִן הַקָּר עִם לַפִּיד דּוֹלֵק כְּדֵי לִבְדֹּק אֶת הֶחָבִיּוֹת. הַקִּירוֹת שָׁם לַחִים וְקָרִים מְאֹד. הוּא פּוֹתֵחַ חָבִית אַחַת... וּלְפֶתַע נוֹפֵל לְאָחוֹר! דְּמָעוֹת שֶׁל עֲקִיצָה מְמַלְאוֹת אֶת עֵינָיו.

אֵין כָּאן רֵיחַ שֶׁל יַיִן מָתוֹק. יֵשׁ כָּאן רֵיחַ חָרִיף, דּוֹקֵר, עַז מְאֹד וְחוֹמְצִי! רֵיחַ שֶׁל חֹמֶץ!

הַפּוֹעֵל רָץ לֶחָבִית הַשְּׁנִיָּה – חֹמֶץ! לַשְּׁלִישִׁית – חֹמֶץ!

מַהֵר מַהֵר רָצִים הַפּוֹעֲלִים וּבוֹדְקִים אֶת כָּל הֶחָבִיּוֹת שֶׁבַּמַּרְתֵּף. הַלֵּב שֶׁלָּהֶם דּוֹפֵק בְּחָזְקָה, הֵם לֹא מַאֲמִינִים לְמַרְאֵה עֵינֵיהֶם – אַרְבַּע מֵאוֹת חָבִיּוֹת שֶׁל יַיִן יָקָר – הֶחְמִיצוּ לְגַמְרֵי! מֵאוֹת אֲלָפִים שֶׁל מַטְבְּעוֹת זָהָב יָרְדוּ לְטִמְיוֹן בְּרֶגַע אֶחָד. הַכֹּל נֶהֱרַס!

הֵם בָּאוּ לְרַב הוּנָא בְּפָנִים נְפוּלוֹת, רָעֲדוּ כֻּלָּם וְסִפְּרוּ לוֹ אֶת הַבְּשׂוֹרָה הַקָּשָׁה.

יְלָדִים, מָה אָדָם רָגִיל הָיָה עוֹשֶׂה? וַדַּאי צּוֹעֵק, בּוֹכֶה, כּוֹעֵס, אוֹ מְחַפֵּשׂ אֶת הַפּוֹעֵל הָאָשֵׁם כְּדֵי לְפַטֵּר אוֹתוֹ! אֲבָל רַב הוּנָא שָׁתַק. הַקָּדוֹשׁ בָּרוּךְ הוּא נָתַן, וְהַקָּדוֹשׁ בָּרוּךְ הוּא לָקַח. הוּא קִבֵּל אֶת הַדִּין בְּאַהֲבָה.

בִּקּוּר הַחֲכָמִים: הַשְּׁאֵלָה הַקָּשָׁה מִכֹּל

הַשְּׁמוּעָה עַל הָאָסוֹן הַכַּלְכָּלִי שֶׁל רַב הוּנָא הִתְפַּשְּׁטָה מַהֵר בְּכָל בָּבֶל.

בְּאוֹתוֹ עֶרֶב, נִשְׁמְעוּ קוֹלוֹת שֶׁל צַעֲדִים כְּבֵדִים בְּפֶתַח בֵּיתוֹ שֶׁל רַב הוּנָא. גְּדוֹלֵי חַכְמֵי יִשְׂרָאֵל, וּבָרֹאשׁ שֶׁלָּהֶם הָרַב הוּנָא רַבָּה, רַב יְהוּדָה וְרַבָּה, בָּאוּ לְבַקֵּר אֶת חֲבֵרָם בִּשְׁעַת צַעֲרוֹ.

הֵם יָשְׁבוּ מוּלוֹ. עֵינֵיהֶם הָיוּ עֲצוּבוֹת, אַךְ הֵם הָיוּ אַנְשֵׁי אֱמֶת. בִּמְקוֹם לַגִּיד "אוֹי, מִסְכֵּן שֶׁלָּנוּ, אֵיזֶה חֹסֶר מַזָּל יֵשׁ לְךָ", הֵם אָמְרוּ מַשֶּׁהוּ אַחֵר לְגַמְרֵי.

"יְעַיֵּן כְּבוֹדוֹ בְּמַעֲשָׂיו," הֵם לָחֲשׁוּ בְּקוֹל שָׁקֵט אֲבָל חוֹדֵר וְעָמֹק. כְּלוֹמַר: רַב הוּנָא יַקִּירֵנוּ, אַתָּה תַּלְמִיד חָכָם עָצוּם, אֲבָל מַכָּה כָּזוֹ גְּדוֹלָה לֹא קוֹרֵית סְתָם בְּלִי סִבָּה. הַקָּדוֹשׁ בָּרוּךְ הוּא שׁוֹלֵחַ רֶמֶז! אוּלַי, תַּחְשֹׁב טוֹב טוֹב, אוּלַי עָשִׂיתָ עֲבֵרָה קְטַנָּה, טָעוּת לְלֹא כַּוָּנָה בֵּין אָדָם לַחֲבֵרוֹ שֶׁבִּגְלָלָהּ זֶה קָרָה?

רַב הוּנָא הָיָה הָמוּם. "מָה?!" שָׁאַל בְּכְאֵב. "הַאִם אֲנִי חָשׁוּד בְּעֵינֵיכֶם?! אַתֶּם חוֹשְׁבִים שֶׁחָטָאתִי?!"

עָנוּ לוֹ הַחֲכָמִים: "חָלִילָה לָנוּ מִלַּחְשֹׁד בְּךָ שֶׁאַתָּה אִישׁ רַע. אַךְ תַּגִּיד אַתָּה: הַאִם הַקָּדוֹשׁ בָּרוּךְ הוּא חָשׁוּד בְּעֵינֶיךָ שֶׁהוּא מַעֲנִישׁ אָדָם בְּלִי סִבָּה?! הֲרֵי אֲנַחְנוּ יוֹדְעִים שֶׁ'צַדִּיק ה' בְּכָל דְּרָכָיו'! אִם קוֹרֶה מַשֶּׁהוּ רַע, זֶה כִּי יֵשׁ סִבָּה שֶׁצָּרִיךְ לְתַקֵּן!".

"אִם כָּךְ," הִרְכִּין רַב הוּנָא אֶת רֹאשׁוֹ בַּעֲנָוָה, אֲפִלּוּ שֶׁהָיָה מְבֻגָּר וְגָדוֹל מֵהֶם בַּתּוֹרָה, "אִם מִישֶׁהוּ מִכֶּם שָׁמַע שֶׁעָשִׂיתִי מַשֶּׁהוּ לֹא נָכוֹן – שֶׁיַּגִּיד לִי מִיָּד, וַאֲנִי אֲתַקֵּן!".

וִכּוּחַ "הָאֲרִיס" וַעֲנָפֵי הַגֶּפֶן

אֶחָד הַחֲכָמִים פִּנָּה אֶת גְּרוֹנוֹ, הִסְתַּכֵּל יָשָׁר לְעֵינָיו הַטְּהוֹרוֹת שֶׁל רַב הוּנָא, וְאָמַר:

"אֲנַחְנוּ שָׁמַעְנוּ, שֶׁכְּבוֹדוֹ לֹא נוֹתֵן מַסְפִּיק כֶּסֶף, אוֹ יוֹתֵר נָכוֹן, לֹא נוֹתֵן אֶת הָ'זְמוֹרוֹת' לְ'אָרִיס' שֶׁלּוֹ."

בּוֹאוּ נָבִין רֶגַע יְלָדִים, מָה זֶה?

בְּאוֹתָם יָמִים, הָיָה אָדָם שֶׁקּוֹרְאִים לוֹ 'אָרִיס' – פּוֹעֵל שֶׁעוֹבֵד בַּכֶּרֶם תְּמוּרַת חֵלֶק מֵהַיְבוּל. וְהָיָה הֶסְכֵּם: כְּשֶׁגּוֹזְמִים אֶת עַנְפֵי הַגֶּפֶן בַּכֶּרֶם (שֶׁנִּקְרָאִים 'זְמוֹרוֹת', וְהֵם הָיוּ מְשַׁמְּשִׁים כְּעֵצִים מְצֻיָּנִים לַהֲסָקָה וְלַתַּנּוּרִים), הָאָרִיס לָקַח חֵלֶק מֵהָעֲנָפִים לְבֵיתוֹ כְּחֵלֶק מֵהַשָּׂכָר. אֲבָל רַב הוּנָא מָנַע מֵהָאָרִיס שֶׁלּוֹ לָקַחַת אֶת הָעֲנָפִים הַלָּלוּ! מִדִּין תּוֹרָה, זֶהוּ כְּבָר חֲשָׁשׁ גֶּזֶל!

רַב הוּנָא פָּתַח אֶת עֵינָיו בְּהַפְתָּעָה: "עַל זֶה אַתֶּם מְדַבְּרִים?! הֲרֵי תֵּדְעוּ לָכֶם, שֶׁהָאָרִיס הַזֶּה הוּא גַּנָּב שֶׁקְרָן בְּעַצְמוֹ! הוּא כְּבָר בְּמֵילָא גּוֹנֵב מִמֶּנִּי אֶת כָּל עַנְפֵי הַגֶּפֶן הַטּוֹבִים עוֹד לִפְנֵי שֶׁאֲנִי מַסְפִּיק לָבוֹא לַשָּׂדֶה! הוּא כְּבָר לָקַח יוֹתֵר מִמָּה שֶׁמַּגִּיעַ לוֹ! מָה אַתֶּם רוֹצִים, שֶׁאֶתֵּן לַגַּנָּב הַזֶּה עוֹד כֶּסֶף אוֹ עוֹד עֲנָפִים מֵהַכֶּרֶם? מַגִּיעַ לוֹ שֶׁלֹּא אֶתֵּן לוֹ!".

רַב הוּנָא הָיָה בָּטוּחַ בְּצִדְקָתוֹ. "אִם הוּא גּוֹנֵב מִמֶּנִּי, זְכוּתִי לֹא לָתֵת לוֹ מָה שֶׁסִּכַּמְנוּ בַּהַתְחָלָה."

אֲבָל הַחֲכָמִים הַקְּדוֹשִׁים סֵרְבוּ לְהַסְכִּים אִתּוֹ. אֶחָד מֵהֶם הִתְקָרֵב אֵלָיו, עָנָה בְּקוֹל צָלוּל שֶׁהִדְהֵד בַּחֶדֶר, וְאָמַר אֶת אֶחָד הַמִּשְׁפָּטִים הַחֲשׁוּבִים בַּתּוֹרָה:

"בָּתַר גַּנָּבָא גְּנוֹב – וְטַעֲמָא טְעִים!" .

מָה פֵּרוּשׁ הַמִּשְׁפָּט בַּאֲרָמִית? "מִי שֶׁגּוֹנֵב מִן הַגַּנָּב – אַף עַל פִּי כֵן בְּפִיו נִשְׁאָר טַעַם שֶׁל גְּנֵבָה!".

הַחֲכָמִים אָמְרוּ לוֹ: רַב הוּנָא, זֶה בִּכְלָל לֹא מְשַׁנֶּה שֶׁהָאָרִיס הוּא גַּנָּב. אִם תִּקַּח אֶת תַּפְקִיד הַשּׁוֹטֵר וְתַחְלִיט "לְקַזֵּז" וְלֹא לָתֵת לוֹ אֶת מַה שֶׁהֻסְכַּם בַּחוֹזֶה מִבְּלִי לָלֶכֶת לְבֵית מִשְׁפָּט כְּדָת וְכַדִּין – בַּשָּׁמַיִם זֶה נִרְשָׁם לְךָ כְּחֶסְרוֹן שֶׁל גֶּזֶל!

הַיֹּשֶׁר הַיְּהוּדִי הוּא מֻחְלָט. אַתָּה לֹא יָכוֹל לַעֲשׂוֹת מַשֶּׁהוּ שֶׁאֵינוֹ הוֹגֵן וּלְהִצְטַדֵּק וְלַגִּיד "אֲבָל הוּא הִתְחִיל!". גְּנֵבָה זוֹ גְּנֵבָה, אֲפִלּוּ מִגַּנָּב!

הַנִּצָּחוֹן וְהַנֵּס הַגָּדוֹל

יְלָדִים, בְּדֶרֶךְ כְּלָל, כְּשֶׁמְּעִירִים לְאָדָם כָּל כָּךְ גָּדוֹל, אָדָם שֶׁהוּא רֹאשׁ יְשִׁיבָה, מָה הוּא מַרְגִּישׁ? בּוּשָׁה. הָאֵגוֹ שֶׁלּוֹ נִפְגָּע. הוּא יָכוֹל לִכְעֹס וּלְגָרֵשׁ אוֹתָם.

אֲבָל רַב הוּנָא... רַב הוּנָא הָיָה קְדוֹשׁ עֶלְיוֹן. בְּאוֹתוֹ הָרֶגַע, הַפָּנִים שֶׁלּוֹ הוּאֲרוּ בֶּאֱמֶת. הוּא סָגַר אֶת עֵינָיו לְרֶגַע, וּבְעוֹד דִּמְעָה שֶׁל אֱמֶת נוֹצֶצֶת בְּזָוִית עֵינוֹ, הוּא אָמַר שְׁתֵּי מִלִּים שֶׁדּוֹרְשׁוֹת כֹּחַ מְטֹרָף:

"קִבַּלְתִּי עָלַי."

אֲנִי מְקַבֵּל בְּאַהֲבָה שֶׁטָּעִיתִי. אֲנִי חֲזַרְתִּי בִּתְשׁוּבָה. אֲנִי אֶלֵּךְ עַכְשָׁו, לַמְרוֹת הַהֶפְסֵד שֶׁל 400 חָבִיּוֹת הַיַּיִן שֶׁלִּי, וֶאֱשַׁלֵּם לָאָרִיס הַגַּנָּב אֶת כָּל הַכֶּסֶף שֶׁל הָעֲנָפִים שֶׁלֹּא נָתַתִּי לוֹ!

וְאָז, מַשֶּׁהוּ פִּלְאִי, מַשֶּׁהוּ שֶׁקָּשֶׁה לְדַמְיֵן, קָרָה בְּאוֹתוֹ הַלַּיְלָה. הַקָּדוֹשׁ בָּרוּךְ הוּא, שֶׁרָאָה אֶת חֲזָרָתוֹ בִּתְשׁוּבָה שֶׁל רַב הוּנָא, מִהֵר לִשְׁלֹחַ לוֹ בְּרָכָה. הֲרֵי "צַדִּיק גּוֹזֵר וְהַקָּדוֹשׁ בָּרוּךְ הוּא מְקַיֵּם". הַגְּמָרָא מְסַפֶּרֶת לָנוּ שֶׁבְּאוֹתוֹ לַיְלָה מַמָּשׁ, הַנֵּס קָרָה.

יֵשׁ חֲכָמִים בַּגְּמָרָא הָאוֹמְרִים, שֶׁרַב הוּנָא יָרַד בַּבֹּקֶר לַמַּרְתֵּף, וְהִנֵּה – רֵיחַ הַחֹמֶץ הַחָרִיף נֶעֱלַם! אַרְבַּע מֵאוֹת הֶחָבִיּוֹת חָזְרוּ לִהְיוֹת יַיִן מָתוֹק וּמְשֻׁבָּח!

וְיֵשׁ חֲכָמִים הָאוֹמְרִים דָּבָר שֶׁמַּרְאֶה הַשְׁגָּחָה פְּרָטִית מֻפְלָאָה אֲפִלּוּ עוֹד יוֹתֵר: הַחֹמֶץ לֹא חָזַר לִהְיוֹת יַיִן. הוּא נִשְׁאַר חֹמֶץ. אֲבָל בְּדִיּוּק בְּאוֹתָהּ הַשָּׁנָה, פָּרְצָה מַחֲלָה בְּבָבֶל (אוֹ מַחְסוֹר אַדִּיר), וְהָרוֹפְאִים מָצְאוּ שֶׁהַתְּרוּפָה הַיְחִידָה שֶׁמַּעֲבִירָה אֶת הַמַּחֲלָה הִיא תַּרְכִּיז שֶׁל חֹמֶץ! בְּבַת אַחַת, כָּל אַנְשֵׁי הָעִיר חִפְּשׂוּ נוֹאָשׁוֹת מְעַט חֹמֶץ, וְהַמְּחִיר שֶׁל חֹמֶץ בַּשּׁוּק עָלָה מַהֵר מַהֵר עַד שֶׁבְּדִיּוּק שָׁוָה לִמְחִיר שֶׁל יַיִן טוֹב! רַב הוּנָא לֹא הִפְסִיד אֲפִלּוּ פְּרוּטָה אַחַת!

כְּשֶׁלּוֹקְחִים אַחֲרָיוּת עַל הַמַּעֲשִׂים, הַקָּדוֹשׁ בָּרוּךְ הוּא הוֹפֵךְ אֶת כָּל הַחֹמֶץ הַחָרִיף שֶׁל הַחַיִּים לְשֶׁפַע גָּדוֹל!

א. שְׁאֵלַת פְּתִיחָה (לִפְנֵי הַסִּפּוּר):

"יְלָדִים, הַאִם פַּעַם קָרָה שֶׁאִבַּדְתֶּם מַשֶּׁהוּ מְאֹד חָשׁוּב – כְּמוֹ קְלָף נָדִיר אוֹ כֶּסֶף שֶׁחֲסַכְתֶּם, וּבִמְקוֹם לִכְעֹס עַל הַחֲבֵרִים אוֹ לִבְכּוֹת, עֲצַרְתֶּם לְרֶגַע וְחֲשַׁבְתֶּם: 'אוּלַי ה' מְנַסֶּה לִרְמֹז לִי לִבְדֹּק כֵּיצַד הִתְנַהַגְתִּי'? הַיּוֹם נִפְגֹּשׁ עֲנַק-רוּחַ, גָּדוֹל בְּיִשְׂרָאֵל, שֶׁהִפְסִיד בְּשְׁנִיָּה אַחַת מִילְיוֹנִים וְשָׁאַל אֶת עַצְמוֹ בְּדִיּוּק אֶת הַשְּׁאֵלָה הַזּוֹ."

ב. הֶעָרוֹת בִּצּוּעַ וּסְטוֹרִיטֶלִינְג (Storytelling):

ג. הַמְלָצוֹת לַמַּצֶּגֶת (4 שְׁקוּפִיּוֹת מֻקְפָּדוֹת):

(זְכֹר: אֵין לְצַיֵּר אֶת הַשֵּׁם יִתְבָּרַךְ, כָּל דְּמוּת צְרִיכָה לִהְיוֹת בְּלְבוּשׁ צָנוּעַ – שַׁרְווּלִים, כִּסּוּי רֹאשׁ קָבוּעַ וּמְכֻבָּד).

ד. פְּעִילוּת מְסַכֶּמֶת (הַמְלָצָה):

מקורות: המקורות מופיעים בקובץ המקורי.

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “להלן מערך שיעור וסיפור חינוכי, רגשי ומרתק על רב הונא וכדי היין שהחמיצו .” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “דַּמְיְנוּ אֶת מְנַהֵל הָעֲבוֹדָה שֶׁלּוֹ יוֹרֵד אֶל מַרְתֵּף הַיַּיִן הַקָּר עִם לַפִּיד דּוֹלֵק כְּדֵי לִבְדֹּק אֶת הֶחָבִיּוֹת.”?
ג. כיצד הגיב רב הונא למצב המתואר: “גְּדוֹלֵי חַכְמֵי יִשְׂרָאֵל, וּבָרֹאשׁ שֶׁלָּהֶם הָרַב הוּנָא רַבָּה, רַב יְהוּדָה וְרַבָּה, בָּאוּ לְבַקֵּר אֶת חֲבֵרָם בִּשְׁעַת צַעֲרוֹ.”?
ד. מה השתנה בין “בְּאוֹתָם יָמִים, הָיָה אָדָם שֶׁקּוֹרְאִים לוֹ 'אָרִיס' – פּוֹעֵל שֶׁעוֹבֵד בַּכֶּרֶם תְּמוּרַת חֵלֶק מֵהַיְבוּל.” לבין “אַתָּה לֹא יָכוֹל לַעֲשׂוֹת מַשֶּׁהוּ שֶׁאֵינוֹ הוֹגֵן וּלְהִצְטַדֵּק וְלַגִּיד "אֲבָל הוּא הִתְחִיל!".”?
ה. כיצד הגיע הסיפור אל האירוע: “בְּבַת אַחַת, כָּל אַנְשֵׁי הָעִיר חִפְּשׂוּ נוֹאָשׁוֹת מְעַט חֹמֶץ, וְהַמְּחִיר שֶׁל חֹמֶץ בַּשּׁוּק עָלָה מַהֵר מַהֵר עַד שֶׁבְּדִיּוּק שָׁוָה לִמְחִיר…”?

3. מה היה ראשון?

כתבו ליד כל אירוע מספר מ־1 עד 5 לפי סדר התרחשותו בסיפור.

□ אֶחָד הַחֲכָמִים פִּנָּה אֶת גְּרוֹנוֹ, הִסְתַּכֵּל יָשָׁר לְעֵינָיו הַטְּהוֹרוֹת שֶׁל רַב הוּנָא, וְאָמַר:
□ להלן מערך שיעור וסיפור חינוכי, רגשי ומרתק על רב הונא וכדי היין שהחמיצו .
□ (הוּבָא בַּמְּקוֹרוֹת שֶׁל "כְּדֵי הַיַּיִן שֶׁהֶחְמִיצוּ - זְהִירוּת מִגָּזֵל").
□ מֵאוֹת אֲלָפִים שֶׁל מַטְבְּעוֹת זָהָב יָרְדוּ לְטִמְיוֹן בְּרֶגַע אֶחָד.
□ הוּא סָגַר אֶת עֵינָיו לְרֶגַע, וּבְעוֹד דִּמְעָה שֶׁל אֱמֶת נוֹצֶצֶת בְּזָוִית עֵינוֹ, הוּא אָמַר שְׁתֵּי מִלִּים שֶׁדּוֹרְשׁוֹת כֹּחַ מְטֹרָף:

לפני ואחרי

מה קרה לפני: “אֶחָד הַחֲכָמִים פִּנָּה אֶת גְּרוֹנוֹ, הִסְתַּכֵּל יָשָׁר לְעֵינָיו הַטְּהוֹרוֹת שֶׁל רַב הוּנָא, וְאָמַר:”?
מה קרה אחרי: “הוּא סָגַר אֶת עֵינָיו לְרֶגַע, וּבְעוֹד דִּמְעָה שֶׁל אֱמֶת נוֹצֶצֶת בְּזָוִית עֵינוֹ, הוּא אָמַר שְׁתֵּי מִלִּים שֶׁדּוֹרְשׁוֹת כֹּחַ מְטֹרָף:”?

סיבה ותוצאה מתוך המעשה

האירוע הנתוןמה גרם לו?מה הייתה תוצאתו?
הֵם בָּאוּ לְרַב הוּנָא בְּפָנִים נְפוּלוֹת, רָעֲדוּ כֻּלָּם וְסִפְּרוּ לוֹ אֶת הַבְּשׂוֹרָה הַקָּשָׁה.
אִם הוּא גּוֹנֵב מִמֶּנִּי, זְכוּתִי לֹא לָתֵת לוֹ מָה שֶׁסִּכַּמְנוּ בַּהַתְחָלָה.

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “כדי היין שהחמיצו — זהירות מגזל” על רב הונא? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “עָנוּ לוֹ הַחֲכָמִים: "חָלִילָה לָנוּ מִלַּחְשֹׁד בְּךָ שֶׁאַתָּה אִישׁ רַע.”.
ג. אילו רב הונא לא היה פועל כפי שמתואר ב־“אַתָּה לֹא יָכוֹל לַעֲשׂוֹת מַשֶּׁהוּ שֶׁאֵינוֹ הוֹגֵן וּלְהִצְטַדֵּק וְלַגִּיד "אֲבָל הוּא הִתְחִיל!".” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “בְּבַת אַחַת, כָּל אַנְשֵׁי הָעִיר חִפְּשׂוּ נוֹאָשׁוֹת מְעַט חֹמֶץ, וְהַמְּחִיר שֶׁל חֹמֶץ בַּשּׁוּק עָלָה מַהֵר מַהֵר עַד…” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. להלן מערך שיעור וסיפור חינוכי, רגשי ומרתק על רב הונא וכדי היין שהחמיצו .
2. יֵשׁ כָּאן רֵיחַ חָרִיף, דּוֹקֵר, עַז מְאֹד וְחוֹמְצִי!
3. בִּמְקוֹם לַגִּיד "אוֹי, מִסְכֵּן שֶׁלָּנוּ, אֵיזֶה חֹסֶר מַזָּל יֵשׁ…
4. הֲרֵי תֵּדְעוּ לָכֶם, שֶׁהָאָרִיס הַזֶּה הוּא גַּנָּב שֶׁקְרָן…
5. הַקָּדוֹשׁ בָּרוּךְ הוּא, שֶׁרָאָה אֶת חֲזָרָתוֹ בִּתְשׁוּבָה שֶׁל רַב…
6. (הוּבָא בַּמְּקוֹרוֹת שֶׁל "כְּדֵי הַיַּיִן שֶׁהֶחְמִיצוּ - זְהִירוּת…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “גְּדוֹלֵי חַכְמֵי יִשְׂרָאֵל, וּבָרֹאשׁ שֶׁלָּהֶם הָרַב הוּנָא רַבָּה, רַב יְהוּדָה וְרַבָּה, בָּאוּ לְבַקֵּר אֶת חֲבֵרָם…”.
מכתב: כתבו לרב הונא בעקבות “אִם הוּא גּוֹנֵב מִמֶּנִּי, זְכוּתִי לֹא לָתֵת לוֹ מָה שֶׁסִּכַּמְנוּ בַּהַתְחָלָה.”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “הַגְּמָרָא מְסַפֶּרֶת לָנוּ שֶׁבְּאוֹתוֹ לַיְלָה מַמָּשׁ, הַנֵּס קָרָה.”.
דיווח: כתבו ידיעה קצרה שכותרתה “כדי היין שהחמיצו — זהירות מגזל”.

המחזה

רגע מן הסיפורהדמויותמה יאמרו או יעשו?
הֵם בָּאוּ לְרַב הוּנָא בְּפָנִים נְפוּלוֹת, רָעֲדוּ כֻּלָּם וְסִפְּרוּ לוֹ אֶת הַבְּשׂוֹרָה…
אַתָּה לֹא יָכוֹל לַעֲשׂוֹת מַשֶּׁהוּ שֶׁאֵינוֹ הוֹגֵן וּלְהִצְטַדֵּק וְלַגִּיד "אֲבָל הוּא…

7. חיבור לחיים מתוך המעשה

א. מה אפשר ללמוד מן הדרך שבה רב הונא פעל באירוע “בְּאוֹתָם יָמִים, הָיָה אָדָם שֶׁקּוֹרְאִים לוֹ 'אָרִיס' – פּוֹעֵל שֶׁעוֹבֵד בַּכֶּרֶם תְּמוּרַת חֵלֶק מֵהַיְבוּל.”?
ב. תארו מצב מחיי הכיתה שבו אפשר ליישם את המסר של “כדי היין שהחמיצו — זהירות מגזל”.
