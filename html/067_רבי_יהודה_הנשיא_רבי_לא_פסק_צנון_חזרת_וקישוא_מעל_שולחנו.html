לא פסק צנון, חזרת וקישוא מעל שולחנו

דף עבודה בעקבות רבי יהודה הנשיא (רבי)
שם: כיתה:

המעשה — הסיפור המלא

פְּתִיחָה: חֹרֶף עַל הָהָר

יְלָדִים, נַסּוּ לְדַמְיֵן לַיְלָה קַר וְקָפוּא בֶּהָרִים שֶׁל אֶרֶץ יִשְׂרָאֵל. רוּחַ חֲזָקָה שׁוֹרֶקֶת בַּחוּץ – וּוּוּ... שְׁשְׁשְׁ... – וּמְעִיפָה טִפּוֹת שֶׁל גֶּשֶׁם עַל הַחַלּוֹנוֹת. הַקֹּר כּוֹנֵס לְתוֹךְ הָעֲצָמוֹת. אִישׁ לֹא מֵעֵז לָצֵאת מִבֵּיתוֹ. בַּיָּמִים הָהֵם, לִפְנֵי כִּמְעַט אַלְפַּיִם שָׁנָה, לֹא הָיוּ חֲמָמוֹת מְחֻמָּמוֹת, וְהַשָּׂדוֹת כֻּלָּם עָמְדוּ רֵיקִים מִפֵּרוֹת וִירָקוֹת קַיִץ. הַכֹּל הָיָה בּוֹץ וַאֲדָמָה קָרָה.

אֲבָל בְּתוֹךְ אַרְמוֹנוֹ הַגָּדוֹל שֶׁל "רַבִּי" – הוּא רַבֵּנוּ הַקָּדוֹשׁ, רַבִּי יְהוּדָה הַנָּשִׂיא – הָיָה חַם, מוּאָר וְנָעִים.

הַקָּדוֹשׁ בָּרוּךְ הוּא בֵּרֵךְ אֶת רַבִּי יְהוּדָה הַנָּשִׂיא בְּעֹשֶׁר מְפֹאָר וְעָצוּם. אֲפִלּוּ מַלְכֵי אוּמוֹת הָעוֹלָם וְקֵיסָר רוֹמָא, אַנְטוֹנִינוּס, רָחֲשׁוּ לוֹ כָּבוֹד וְהֶעֱרִיצוּ אֶת חָכְמָתוֹ. אֲבָל הַדָּבָר הַמַּדְהִים בְּיוֹתֵר בְּרַבִּי הָיָה הַלֵּב שֶׁלּוֹ. לַמְרוֹת שֶׁהָיָה מֻקָּף בְּכֵלִים שֶׁל מֶשִׁי וְזָהָב, הוּא לֹא נֶהֱנָה מִן הָעוֹלָם הַזֶּה אֲפִלּוּ בְּאֶצְבַּע קְטַנָּה בִּשְׁבִיל עַצְמוֹ! כָּל הָעֹשֶׁר שֶׁנָּתַן לוֹ הַשֵּׁם יִתְבָּרַךְ, נוֹעַד רַק לְדָבָר אֶחָד: לִכְבוֹד תּוֹרָה וּלְאַהֲבַת יִשְׂרָאֵל .

הָאוֹרֵחַ הָרָעֵב מִן הַגָּלִיל

בְּאוֹתוֹ לַיְלָה סוֹעֵר, הִגִּיעַ לְבֵיתוֹ שֶׁל רַבִּי תַּלְמִיד חָכָם עָנִי מִכְּפָר רָחוֹק בַּגָּלִיל. נִקְרָא לוֹ שִׁמְעוֹן. שִׁמְעוֹן צָעַד יָמִים אֲרֻכִּים בַּקֹּר רַק כְּדֵי לִזְכּוֹת לִלְמֹד תּוֹרָה מִפִּיו שֶׁל רַבֵּנוּ הַקָּדוֹשׁ. בְּגָדָיו הָיוּ רְטוּבִים, וּבִטְנוֹ קִרְקְרָה מֵרָעָב שֶׁל יָמִים רַבִּים שֶׁבָּהֶם אָכַל רַק פִּסּוֹת לֶחֶם יָבֵשׁ.

כְּשֶׁנִּכְנַס שִׁמְעוֹן לְטְרַקְלִין סְעוּדוֹתָיו שֶׁל הַנָּשִׂיא, הִסְתַּחְרֵר רֹאשׁוֹ. אוֹרָן הַזּוֹהֵר שֶׁל עֲשָׂרוֹת מְנוֹרוֹת הִשְׁתַּקֵּף עַל קְעָרוֹת כֶּסֶף כְּבֵדוֹת. וְהָרֵיחַ... אוֹי, הָרֵיחַ! מִן הַמִּטְבָּח עָלָה רֵיחַ כָּבֵד וּמְשַׁכֵּר שֶׁל בְּשַׂר בָּקָר עָסִיסִי הַנִּצְלֶה עַל גֶּחָלִים, רֵיחַ שֶׁל מְרָקִים עֲשִׁירִים הַמַּעֲלִים אָדִים שֶׁל חֹם, וְרֵיחַ שֶׁל לֶחֶם רַךְ שֶׁנֶּאֱפָה זֶה עַתָּה!

"שֵׁב, בְּנִי, וֶאֱכֹל עִמָּנוּ," חִיֵּךְ אֵלָיו רַבִּי יְהוּדָה הַנָּשִׂיא בְּעֵינַיִם מְלֵאוֹת אַהֲבָה וְקָדוֹשָׁה.

שִׁמְעוֹן הִתְיַשֵּׁב וְהֵחֵל לֶאֱכֹל מֵהַשֶּׁפַע שֶׁהֻגַּשׁ לוֹ. הוּא אָכַל מֵהַבָּשָׂר הַשָּׁמֵן, טָעַם מֵהַתַּבְשִׁילִים, וְשָׂבַע הֵיטֵב.

הַחֲשָׁשׁ וְהַבֶּטֶן הַכּוֹאֶבֶת

אֲבָל, לְאַט לְאַט, כְּשֶׁסִּיֵּם שִׁמְעוֹן לֶאֱכֹל, הֵחֵל הַלֵּב שֶׁלּוֹ לַחְשֹׁשׁ, וּבִטְנוֹ הִתְחִילָה לֶאֱכֹאב הַרְבֵּה יוֹתֵר!

מַדּוּעַ?

יְלָדִים, אָדָם עָנִי שֶׁרָגִיל לֶאֱכֹל רַק לֶחֶם וּמַיִם, וּפִתְאוֹם גּוּפוֹ מְקַבֵּל כַּמּוּיוֹת עֲצוּמוֹת שֶׁל תַּבְשִׁילֵי מְלָכִים, בָּשָׂר כָּבֵד וְתַבְלִינִים – הַגּוּף שֶׁלּוֹ מִתְקַשֶּׁה לְעַכֵּל אֶת הַמָּזוֹן. זֶה עוֹשֶׂה בָּלָגָן גָּדוֹל בַּבֶּטֶן.

שִׁמְעוֹן חָשַׁב בְּלִבּוֹ עִם תּוּגָה וָצַעַר: "רַבִּי בֶּאֱמֶת רָצָה לַעֲשׂוֹת לִי טוֹבָה וּלְהַשְׂבִּיעַ אוֹתִי, אַךְ עַכְשָׁו תִּכְאַב לִי הַבֶּטֶן כָּל הַלַּיְלָה! לֹא אוּכַל לִישֹׁן מֵרֹב מַכְאוֹבִים (צָרֶבֶת וּכְבֵדוּת), וּמָחָר בַּבֹּקֶר הָרֹאשׁ שֶׁלִּי לֹא יִהְיֶה פָּנוּי לִלְמֹד תּוֹרָה! לוּ רַק הָיָה לִי עַכְשָׁו אֵיזֶה יָרָק טָרִי שֶׁיַּעֲזֹר לְעַכֵּל אֶת כָּל הַבָּשָׂר הַזֶּה... אֲבָל חֹרֶף עַכְשָׁו! שֶׁלֶג יוֹרֵד בֶּהָרִים! אֵין שׁוּם יְרָקוֹת טְרִיִּים בָּעוֹלָם!".

שִׁמְעוֹן הֵנִיחַ אֶת יָדוֹ בַּסֵּתֶר עַל בִּטְנוֹ, וְעָצַם אֶת עֵינָיו בְּהַשְׁלָמָה. "זֶה הַמַּצָּב. אֶצְטָרֵךְ לִסְבֹּל בְּשֶׁקֶט," מִלְמֵל.

הַהַפְתָּעָה הַיְרֻקָּה

אֲבָל הוּא לֹא הִכִּיר אֶת רַבֵּנוּ הַקָּדוֹשׁ. רַבִּי יְהוּדָה הַנָּשִׂיא רָאָה הַכֹּל, הֵבִין הַכֹּל, וְהֵכִין אֶת הַכֹּל מֵרֹאשׁ!

רַבִּי יְהוּדָה סִימֵן בְּאֶצְבָּעוֹ, וּלְפֶתַע, דַּלְתוֹת הָעֵץ הַגְּדוֹלוֹת שֶׁל טְרַקְלִין סְעוּדוֹת הַמֶּלֶךְ נִפְתְּחוּ לִרְוָחָה. עֲשָׂרָה מְשָׁרְתִים צָעֲדוּ פְּנִימָה, וּבִידֵיהֶם מַגְּשֵׁי כֶּסֶף נוֹצְצִים.

שִׁמְעוֹן פָּקַח אֶת עֵינָיו וְכִמְעַט נָפַל מִן הַכִּסֵּא מֵרֹב תַּדְהֵמָה!

מָה הָיָה מֻנָּח עַל הַמַּגָּשִׁים? בְּאֶמְצַע עֲנָנֵי הַחֹרֶף, בְּתוֹךְ קְפִיאַת הַשְּׁלָגִים, הֻנְּחוּ עַל הַשֻּׁלְחָן גִּבְעוֹת יְרֻקּוֹת וְלְבָנוֹת שֶׁל יְרָקוֹת טְרִיִּים!

הָיוּ שָׁם פְלָחִים שֶׁל קִשּׁוּאִים עֲסִיסִיִּים הַמַּבְרִיקִים מֵרֹב טְרִיּוּת. הָיוּ שָׁם חֲתִיכוֹת עָבוֹת שֶׁל צְנוֹן חָרִיף שֶׁהָרֵיחַ הַמְּרַעֲנֵן שֶׁלּוֹ כְּבָר פָּתַח אֶת הַנְּשִׁימָה, וְעָלִים יְרֻקִּים וְשָׁלֵמִים שֶׁל חֲזֶרֶת (חַסָּה מְרִירוֹת)!

מִיָּד חָתַךְ שִׁמְעוֹן חֲתִיכָה מֵהַקִּשּׁוּא וּנְגָסָהּ. קְרָאנְצ'...

טַעַם הַמַּיִם הַטְּרִיִּים שֶׁל הַקִּשּׁוּא, וְהַחֲרִיפוּת הַנְּעִימָה שֶׁל הַצְּנוֹן וְהַחֲזֶרֶת, מִלְּאוּ אֶת גּוּפוֹ בְּתַחוּשַׁת קְרִירוּת שֶׁשָּׁטְפָה אֶת הַכְּבֵדוּת. בְּתוֹךְ רְגָעִים מִסְפָּר, כְּאֵב הַבֶּטֶן מַמָּשׁ נֶעֱלַם כְּלֹא הָיָה! הַגּוּף עִכֵּל אֶת הָאֹכֶל בְּקַלּוּת וּבְשִׂמְחָה.

אֲבָל... מֵאֵיפֹה?!

שִׁמְעוֹן חָשׁ הֲקָלָה עֲצוּמָה, אֲבָל גַּם פְּלִיאָה שֶׁאֵין כָּמוֹהָ. הוּא הִבִּיט בְּרַבֵּנוּ הַקָּדוֹשׁ, שֶׁיָּשַׁב בְּרֹאשׁ הַשֻּׁלְחָן מֻקָּף מַלְאָכִים שֶׁל תּוֹרָה, וְלֹא יָכוֹל הָיָה לְהַסְתִּיר אֶת שְׁאֵלָתוֹ:

"רַבֵּנוּ... תּוֹרָתְךָ וַחֲסָדֶיךָ מְאִירִים אֶת כָּל הָעוֹלָם... אֲבָל... הֵיכָן מָצָא כְּבוֹדוֹ צְנוֹן וְקִשּׁוּא בְּעִיצּוּמוֹ שֶׁל הַחֹרֶף?! מֵאֵיפֹה הֵבִיאוּ אֶת זֶה?!"

רַבִּי יְהוּדָה הַנָּשִׂיא חִיֵּךְ חִיּוּךְ מָלֵא עֲנָוָה, אֶת אוֹתוֹ חִיּוּךְ קָדוֹשׁ שֶׁכָּבַשׁ אֶת לֵב כָּל הָעָם.

"שִׁמְעוֹן בְּנִי, הַקָּדוֹשׁ בָּרוּךְ הוּא נָתַן לִי אֶת אוֹצְרוֹתָיו לֹא כְּדֵי שֶׁאֶשְׁמֹר אוֹתָם בְּתוֹךְ תֵּבוֹת שֶׁל כֶּסֶף, אֶלָּא כְּדֵי שֶׁאֶשְׁתַּמֵּשׁ בָּהֶם לְמַעַן לִמּוּד הַתּוֹרָה שֶׁל עַם יִשְׂרָאֵל. יָדַעְתִּי שֶׁתַּלְמִידֵי חֲכָמִים, הָאוֹרְחִים הַבָּאִים מִמֶּרְחַקִּים, עֲלוּלִים לִסְבֹּל מִכְּאֵבֵי מֵעַיִם אִם יֹאכְלוּ אֶת בְּשַׂר סְעוּדוֹתַי בְּלִי לְאַזֵּן אוֹתוֹ מִיָּד בִּירָקוֹת! הֲרֵי צְנוֹן, חֲזֶרֶת וְקִשּׁוּא עוֹזְרִים מְאֹד לְעִכּוּל הַמָּזוֹן בַּגּוּף. אָז שָׁלַחְתִּי אֶת אֲנָשַׁי לַעֲמֹל קָשֶׁה!"

וְאָז הִתְגַּלָּה הַסּוֹד הַגָּדוֹל.

רַבִּי שִׁלֵּם סְכוּמֵי כֶּסֶף אַדִּירִים – הַרְבֵּה יוֹתֵר מִמְּחִיר הַבָּשָׂר עַצְמוֹ! – לְפָעֲלִים מְיֻחָדִים שֶׁיִּנְסְעוּ לִמְקוֹמוֹת רְחוֹקִים וְחַמִּים יוֹתֵר, אוֹ יְגַדְּלוּ בִּשְׁבִילוֹ צְנוֹנִיּוֹת וְקִשּׁוּאִים בְּתַנָּאִים מְיֻחָדִים, שֶׁשּׁוֹמְרִים אֶת חֹם הַקַּיִץ גַּם כְּשֶׁבַּחוּץ סוּפוֹת! כָּךְ הִשְׁגִּיחַ הַנָּשִׂיא עַל גּוּפָם שֶׁל כָּל תַּלְמִידָיו וְאוֹרְחָיו!

כְּפִי שֶׁכָּתוּב בַּגְּמָרָא, מֵעוֹלָם " לֹא פָסַק צְנוֹן וַחֲזֶרֶת וְקִשּׁוּא מֵעַל שֻׁלְחָנוֹ שֶׁל רַבִּי " – תָּמִיד, גַּם בַּקַּיִץ וְגַם בַּחֹרֶף!

סִיּוּם: שִׁעוּר קָטָן בְּאַהֲבָה גְּדוֹלָה

דְּמָעוֹת שֶׁל רֶגֶשׁ עָמְדוּ בְּעֵינָיו שֶׁל שִׁמְעוֹן. "זֶה חֶסֶד מֻשְׁלָם," לָחַשׁ. "לֹא רַק שֶׁאַתָּה מַאֲכִיל אוֹתִי מַאֲכָלִים שֶׁלְּעוֹלָם לֹא יָכֹלְתִּי לִקְנוֹת, אַתָּה אֲפִלּוּ דּוֹאֵג לַבֶּטֶן שֶׁלִּי, כְּדֵי שֶׁלֹּא אֶסְבֹּל כְּאֵב שֶׁאַף אֶחָד לֹא רוֹאֶה!"

בְּאוֹתוֹ לַיְלָה, יָשַׁב שִׁמְעוֹן בְּבֵית הַמִּדְרָשׁ כְּשֶׁגּוּפוֹ קַל וּמְרַעֲנָן. הוּא לָמַד אֶת תּוֹרָתוֹ שֶׁל רַבֵּנוּ הַקָּדוֹשׁ – וְהֵבִין, שֶׁלִּפְנֵי שֶׁלּוֹמְדִים אֶת חָכְמַת הַמִּשְׁנָה, חַיָּבִים לִלְמֹד אֶת הָאַהֲבָה הָעֲצוּמָה וְהַדְּאָגָה הַסּוֹדִית, אֲפִלּוּ לַקִּשּׁוּאִים וְלַצְּנוֹנִיּוֹת שֶׁהָיוּ תָּמִיד מוּכָנִים רַק כְּדֵי שֶׁלִּיהוּדִי אַחֵר לֹא יִכְאַב.

מקורות: תלמוד בבלי, מסכת עבודה זרה, דף י"א, עמוד א': "אמרו עליו על רבנו הקדוש שלא פסק מעל שולחנו לא חזרת ולא קישואים ולא צנון, לא בימות החמה ולא בימות הגשמים." רש"י (שם): "שבריאים הם לאכילה." חוברת "רבי יהודה הנשיא" להרב יואל שוורץ (מצוטט במקורות להוראה - מקורות מספר 35, 89, 124, 150 באוסף דפי הלימוד): ציון עובדת היות הירקות קשים להשגה שלא בעונתם (היעדר חממות), והקשר לכך שרצה רבי שאורחיו הרבים לא יינזקו בבטנם מריבוי הסעודה.

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “יְלָדִים, נַסּוּ לְדַמְיֵן לַיְלָה קַר וְקָפוּא בֶּהָרִים שֶׁל אֶרֶץ יִשְׂרָאֵל.” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “בְּאוֹתוֹ לַיְלָה סוֹעֵר, הִגִּיעַ לְבֵיתוֹ שֶׁל רַבִּי תַּלְמִיד חָכָם עָנִי מִכְּפָר רָחוֹק בַּגָּלִיל.”?
ג. כיצד הגיב רבי יהודה הנשיא (רבי) למצב המתואר: “הוּא אָכַל מֵהַבָּשָׂר הַשָּׁמֵן, טָעַם מֵהַתַּבְשִׁילִים, וְשָׂבַע הֵיטֵב.”?
ד. מה השתנה בין “רַבִּי יְהוּדָה סִימֵן בְּאֶצְבָּעוֹ, וּלְפֶתַע, דַּלְתוֹת הָעֵץ הַגְּדוֹלוֹת שֶׁל טְרַקְלִין סְעוּדוֹת הַמֶּלֶךְ…” לבין “שִׁמְעוֹן חָשׁ הֲקָלָה עֲצוּמָה, אֲבָל גַּם פְּלִיאָה שֶׁאֵין כָּמוֹהָ.”?
ה. כיצד הגיע הסיפור אל האירוע: “– לְפָעֲלִים מְיֻחָדִים שֶׁיִּנְסְעוּ לִמְקוֹמוֹת רְחוֹקִים וְחַמִּים יוֹתֵר, אוֹ יְגַדְּלוּ בִּשְׁבִילוֹ צְנוֹנִיּוֹת וְקִשּׁוּאִים בְּתַנָּאִים…”?

3. מה היה ראשון?

כתבו ליד כל אירוע מספר מ־1 עד 5 לפי סדר התרחשותו בסיפור.

□ שִׁמְעוֹן הֵנִיחַ אֶת יָדוֹ בַּסֵּתֶר עַל בִּטְנוֹ, וְעָצַם אֶת עֵינָיו בְּהַשְׁלָמָה.
□ יְלָדִים, נַסּוּ לְדַמְיֵן לַיְלָה קַר וְקָפוּא בֶּהָרִים שֶׁל אֶרֶץ יִשְׂרָאֵל.

לפני ואחרי

מה קרה לפני: “שִׁמְעוֹן הֵנִיחַ אֶת יָדוֹ בַּסֵּתֶר עַל בִּטְנוֹ, וְעָצַם אֶת עֵינָיו בְּהַשְׁלָמָה.”?
מה קרה אחרי: “הוּא הִבִּיט בְּרַבֵּנוּ הַקָּדוֹשׁ, שֶׁיָּשַׁב בְּרֹאשׁ הַשֻּׁלְחָן מֻקָּף מַלְאָכִים שֶׁל תּוֹרָה, וְלֹא יָכוֹל הָיָה לְהַסְתִּיר אֶת שְׁאֵלָתוֹ:”?

סיבה ותוצאה מתוך המעשה

האירוע הנתוןמה גרם לו?מה הייתה תוצאתו?
כְּשֶׁנִּכְנַס שִׁמְעוֹן לְטְרַקְלִין סְעוּדוֹתָיו שֶׁל הַנָּשִׂיא, הִסְתַּחְרֵר רֹאשׁוֹ.
הָיוּ שָׁם פְלָחִים שֶׁל קִשּׁוּאִים עֲסִיסִיִּים הַמַּבְרִיקִים מֵרֹב טְרִיּוּת.

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “לא פסק צנון, חזרת וקישוא מעל שולחנו” על רבי יהודה הנשיא (רבי)? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “לֹא אוּכַל לִישֹׁן מֵרֹב מַכְאוֹבִים (צָרֶבֶת וּכְבֵדוּת), וּמָחָר בַּבֹּקֶר הָרֹאשׁ שֶׁלִּי לֹא יִהְיֶה פָּנוּי לִלְמֹד תּוֹרָה!”.
ג. אילו רבי יהודה הנשיא (רבי) לא היה פועל כפי שמתואר ב־“שִׁמְעוֹן חָשׁ הֲקָלָה עֲצוּמָה, אֲבָל גַּם פְּלִיאָה שֶׁאֵין כָּמוֹהָ.” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “– לְפָעֲלִים מְיֻחָדִים שֶׁיִּנְסְעוּ לִמְקוֹמוֹת רְחוֹקִים וְחַמִּים יוֹתֵר, אוֹ יְגַדְּלוּ בִּשְׁבִילוֹ צְנוֹנִיּוֹת…” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. יְלָדִים, נַסּוּ לְדַמְיֵן לַיְלָה קַר וְקָפוּא בֶּהָרִים שֶׁל אֶרֶץ…
2. בְּאוֹתוֹ לַיְלָה סוֹעֵר, הִגִּיעַ לְבֵיתוֹ שֶׁל רַבִּי תַּלְמִיד חָכָם…
3. יְלָדִים, אָדָם עָנִי שֶׁרָגִיל לֶאֱכֹל רַק לֶחֶם וּמַיִם, וּפִתְאוֹם…
4. שִׁמְעוֹן פָּקַח אֶת עֵינָיו וְכִמְעַט נָפַל מִן הַכִּסֵּא מֵרֹב…
5. שִׁמְעוֹן בְּנִי, הַקָּדוֹשׁ בָּרוּךְ הוּא נָתַן לִי אֶת אוֹצְרוֹתָיו…
6. בְּאוֹתוֹ לַיְלָה, יָשַׁב שִׁמְעוֹן בְּבֵית הַמִּדְרָשׁ כְּשֶׁגּוּפוֹ…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “הוּא אָכַל מֵהַבָּשָׂר הַשָּׁמֵן, טָעַם מֵהַתַּבְשִׁילִים, וְשָׂבַע הֵיטֵב.”.
מכתב: כתבו לרבי יהודה הנשיא (רבי) בעקבות “הָיוּ שָׁם פְלָחִים שֶׁל קִשּׁוּאִים עֲסִיסִיִּים הַמַּבְרִיקִים מֵרֹב טְרִיּוּת.”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “שִׁמְעוֹן בְּנִי, הַקָּדוֹשׁ בָּרוּךְ הוּא נָתַן לִי אֶת אוֹצְרוֹתָיו לֹא כְּדֵי שֶׁאֶשְׁמֹר אוֹתָם בְּתוֹךְ תֵּבוֹת שֶׁל כֶּסֶף,…”.
דיווח: כתבו ידיעה קצרה שכותרתה “לא פסק צנון, חזרת וקישוא מעל שולחנו”.

המחזה

רגע מן הסיפורהדמויותמה יאמרו או יעשו?
