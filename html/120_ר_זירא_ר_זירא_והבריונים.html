ר' זירא והבריונים

דף עבודה בעקבות ר' זירא
שם: כיתה:

המעשה — הסיפור המלא

פְּתִיחָה: רְחוֹב הַפַּחַד

יְלָדִים, דַּמְיְנוּ עִיר גְּדוֹלָה בְּבָבֶל שֶׁל יְמֵי קֶדֶם. בְּמֶרְכַּז הָעִיר, שׁוֹקֵק בֵּית הַמִּדְרָשׁ שֶׁל הָאֲמוֹרָאִים. קוֹל תּוֹרָה מָתוֹק מִתְנַגֵּן כָּל הַיּוֹם וְכָל הַלַּיְלָה.

אַךְ לֹא הַרְחֵק מִשָּׁם, הָיְתָה שְׁכוּנָה אַחֶרֶת. שְׁכוּנָה שֶׁאֲנָשִׁים פָּחֲדוּ לְהִכָּנֵס אֵלֶיהָ. בָּרְחוֹב הַזֶּה הִתְגּוֹרְרוּ קְבוּצָה שֶׁל "בִּרְיוֹנִים".

הֵם הָיוּ אֲנָשִׁים מַפְחִידִים מְאֹד. כְּשֶׁהֵם הָיוּ צוֹעֲדִים בַּשּׁוּק, כֻּלָּם הָיוּ זָזִים הַצִּדָּה. רֵיחַ שֶׁל אָבָק, אֵפֶר וְזֵעָה נָדַף מִבִּגְדֵיהֶם הַקְּרוּעִים. קוֹלָם הָיָה עָבֶה וְרוֹעֵם, וּכְשֶׁצָּחֲקוּ, צְחוֹקָם צָרַם בָּאָזְנַיִם כְּמוֹ חֲרִיקָה שֶׁל אֶבֶן עַל סֶלַע. הֵם לֹא שָׁמְרוּ שַׁבָּת, לֹא כִּבְּדוּ רַבָּנִים, הִצִּיקוּ לָעוֹבְרִים וְהַשָּׁבִים, וְרַק הַמַּבָּט בְּעֵינֵיהֶם הָרוֹשְׁפוֹת מִלֵּא כָּל יְהוּדִי בְּחֶרְדָּה.

כַּאֲשֶׁר תַּלְמִידֵי הַחֲכָמִים הָיוּ הוֹלְכִים לְבֵית הַמִּדְרָשׁ, הֵם הָיוּ עוֹבְרִים לְצִדּוֹ הַשֵּׁנִי שֶׁל הָרְחוֹב. "אָסוּר לְדַבֵּר אִתָּם," אָמְרוּ הָאֲנָשִׁים זֶה לָזֶה, "הֵם אֲנָשִׁים רָעִים וְחוֹטְאִים. ה' בְּוַדַּאי כּוֹעֵס עֲלֵיהֶם. מוּטָב לְהִתְרַחֵק".

הֶחָכָם הַקָּטָן וְהָעֲנָק

אַךְ אָדָם אֶחָד, גְּדוֹל הַדּוֹר, לֹא עָבַר לַצַּד הַשֵּׁנִי שֶׁל הָרְחוֹב. קָרְאוּ לוֹ רַבִּי זֵירָא .

רַבִּי זֵירָא הָיָה קָטָן בְּקוֹמָתוֹ בְּאֹפֶן מְיֻחָד. גּוּפוֹ הָיָה חַלָּשׁ וְעָדִין, מִשּׁוּם שֶׁצָּם תַּעֲנִיּוֹת רַבּוֹת מְאֹד. אֲבָל בְּתוֹךְ הַגּוּף הַקָּטָן וְהַטָּהוֹר הַזֶּה, פָעַם הַלֵּב הֲכִי גָּדוֹל בְּכָל מְדִינַת בָּבֶל! לֵב שֶׁהָיָה מָלֵא בְּאַהֲבַת יִשְׂרָאֵל אֲמִתִּית הַבּוֹעֶרֶת כְּלַפִּיד שֶׁל אֵשׁ.

בְּכָל פַּעַם שֶׁרַבִּי זֵירָא הָיָה פּוֹגֵשׁ בַּבִּרְיוֹנִים שֶׁל הַשְּׁכוּנָה, הוּא לֹא צָעַק עֲלֵיהֶם. הוּא לֹא אָמַר לָהֶם: "אַתֶּם רְשָׁעִים!".

הוּא הָיָה מִתְקָרֵב אֲלֵיהֶם בִּצְעָדִים חֲרִישִׁיִּים. הוּא הָיָה מֵרִים אֶת עֵינָיו הַמְּאִירוֹת כְּכוֹכָבִים אֶל פַּרְצוּפֵיהֶם הַקְּשׁוּחִים, מְחַיֵּךְ אֲלֵיהֶם מַמָּשׁ, וּמְדַבֵּר אִתָּם בְּקוֹל רַךְ כְּקַטִיפָה וָשֶׁמֶן.

"אַחַי הַיְקָרִים," הָיָה רַבִּי זֵירָא לוֹחֵשׁ לָהֶם בַּחֹם, "יֵשׁ בָּכֶם נְשָׁמָה כָּל כָּךְ קְדוֹשָׁה. אַתֶּם בָּנָיו הָאֲהוּבִים שֶׁל מֶלֶךְ מַלְכֵי הַמְּלָכִים. בּוֹאוּ, כָּנְסוּ אִתִּי לְבֵית הַמִּדְרָשׁ, בּוֹאוּ נִלְמַד דַּף גְּמָרָא. ה' מְחַכֶּה לָכֶם!".

הַבִּרְיוֹנִים שֶׁלֹּא הָיוּ רְגִילִים שֶׁמְּדַבְּרִים אֲלֵיהֶם בְּכָבוֹד, הָיוּ מְסַבְּבִים אֶת הָרֹאשׁ. לִפְעָמִים הָיוּ מַשְׁמִיעִים נַהֲמַת זִלְזוּל מִתּוֹךְ גְּרוֹנָם וּמַמְשִׁיכִים לְשַׂחֵק בְּקוּבִּיּוֹת שֶׁלָּהֶם. אֲבָל בְּמַעֲמַקֵּי הַלֵּב – מַשֶּׁהוּ נָגַע בָּהֶם. אִישׁ מֵעוֹלָם לֹא הִבִּיט בָּהֶם כְּמוֹ רַבִּי זֵירָא.

רַבִּי זֵירָא הָיָה מַזְכִּיר אוֹתָם בִּתְפִלּוֹתָיו. בְּאִישׁוֹן לַיְלָה, כְּשֶׁדִּמְעוֹתָיו הָיוּ שׁוֹטְפוֹת אֶת סֵפֶר הַתּוֹרָה, הָיָה אוֹמֵר: "רִבּוֹנוֹ שֶׁל עוֹלָם, זַכֵּה אֶת הַשְּׁכֵנִים שֶׁלִּי לַחֲזֹר לְמוּטָב. הָסֵר אֶת הַמָּסָךְ שֶׁמַּסְתִּיר אֶת הַטּוֹב שֶׁבָּהֶם!".

תְּרֻעַת הַחֲכָמִים וְעַקְשָׁנוּת שֶׁל חֶסֶד

שְּׁאָר תַּלְמִידֵי הַחֲכָמִים שֶׁל יְשִׁיבַת בָּבֶל – רַבָּנִים קְדוֹשִׁים וִידֵידָיו שֶׁל רַבִּי זֵירָא – לֹא הֵבִינוּ מָה הוּא עוֹשֶׂה! הֵם כָּעֲסוּ מְאֹד.

"רַבִּי זֵירָא!" הֵם אָמְרוּ בְּקוֹל קְפֵּדָן, "לָמָּה אַתָּה מִתְחַבֵּר לָאֲנָשִׁים הַלָּלוּ? הֲרֵי אֵלּוּ בִּרְיוֹנִים רְשָׁעִים הָעוֹבְרִים עַל חֻקֵּי הַתּוֹרָה! הֵם רוֹמְסִים כָּל דָּבָר שֶׁבִּקְדֻשָּׁה! בִּמְקוֹם לְקָרֵב אוֹתָם, צָרִיךְ לְנַדּוֹת וּלְהַרְחִיק אוֹתָם לְמַעַן יִרְאוּ יְלָדֵינוּ וְלֹא יִלְמְדוּ מִדַּרְכָּם!".

אֲבָל רַבִּי זֵירָא אָטַם אֶת אָזְנָיו לְבִקָּרְתָם. הוּא יָדַע שֶׁהַחֲכָמִים צוֹדְקִים מִצַּד שׁוּרַת הַדִּין, וְשֶׁבֶּאֱמֶת צָרִיךְ לְהִזָּהֵר מֵאֲנָשִׁים כָּאֵלּוּ. אֲבָל הוּא אָמַר: "אִם גַּם אֲנִי אַפְנֶה לָהֶם עוֹרֶף, מִי יַעֲזֹר לָהֶם? מִי יִזְכֹּר שֶׁהֵם יְהוּדִים? אֲנִי עָלַי לִמְסֹר נַפְשִׁי בִּתְפִלָּה עֲלֵיהֶם עַד שֶׁיַּחְזְרוּ בִּתְשׁוּבָה!". הוּא הִמְשִׁיךְ לְבַקֵּר אוֹתָם, לִשְׁאֹל בִּשְׁלוֹמָם, וּלְקָרֵב אוֹתָם בְּעָבוֹתוֹת שֶׁל אַהֲבָה.

הַיּוֹם שֶׁבּוֹ חָשְׁכָה הַשֶּׁמֶשׁ

וְאָז, הִגִּיעַ הַיּוֹם הַמַּר וְהַנִּמְהָר.

שְׁמוּעָה קָשָׁה הִכְּתָה אֶת מְדִינַת בָּבֶל כְּמוֹ רַעַם בְּיוֹם בָּהִיר: "הָרַב הַדּוֹמֶה לְמַלְאַךְ ה', מְנוֹרַת הַמָּאוֹר – רַבִּי זֵירָא – נִפְטַר לְעוֹלָמוֹ!".

שַׂק וָאֵפֶר לָבְשׁוּ כֻּלָּם. בְּכִי חֲרִישִׁי נִשְׁמַע מִכָּל עֵבֶר. אַלְפֵי יְהוּדִים לִוּוּ אֶת מִטָּתוֹ שֶׁל רַבִּי זֵירָא, כְּשֶׁהֵם קוֹרְעִים אֶת בִּגְדֵיהֶם מֵרֹב צַעַר וְכְאֵב. תּוֹרָתוֹ הֶעֲצוּמָה שֶׁל הַצַּדִּיק שָׁתְקָה לְעוֹלָמִים.

אַךְ כַּאֲשֶׁר עָלְתָה הַלְּבָנָה בַּלַּיְלָה וְהָרְחוֹבוֹת הִתְרוֹקְנוּ, הִתְרַחֲשָׁה סְצֵנָה קוֹרַעַת לֵב בִּשְׁכוּנַת הַבִּרְיוֹנִים.

בְּאוֹתוֹ לַיְלָה לֹא שָׁמְעוּ שָׁם קוֹלוֹת שֶׁל קֻבִּיּוֹת מִשְׂחָק נִזְרָקוֹת אֶל הַקַּרְקַע. לֹא נִשְׁמַע רַעַשׁ שֶׁל שִׂיחוֹת קוֹלָנִיּוֹת. דְּמָמָה דַּקָּה וּמַקְפִּיאָה כִּסְּתָה אֶת הַשְּׁכוּנָה.

הַבִּרְיוֹנִים יָשְׁבוּ סָבִיב לַמְּדוּרָה הַקְּטַנָּה בַּחֲצֵרָם. פְּנֵיהֶם שֶׁהָיוּ תָּמִיד נוֹקְשׁוֹת וְזָעוּמוֹת, לְפֶתַע הָיוּ חִוְרוֹת וְשָׁמוּטוֹת. עֵינֵיהֶם הָיוּ מוּשְׁפָּלוֹת לָאֲדָמָה.

הַבִּרְיוֹן הַגָּדוֹל מִכֻּלָּם, אִישׁ עִם כְּתֵפַיִם רְחָבוֹת וְקַעֲקוּעִים עַל זְרוֹעוֹתָיו, שָׁבַר אֶת הַשֶּׁקֶט. קוֹלוֹ הַמְחֻסְפָּס רָעַד.

"עַד עַכְשָׁו," הוּא אָמַר כְּשֶׁדִּמְעָה כְּבֵדָה זָלְגָה עַל לֶחְיוֹ מְלֵאַת הַצַּלָּקוֹת, "עַד הַיּוֹם הַזֶּה... הָיָה מִי שֶׁשָּׁמַר עָלֵינוּ. זֶה שֶׁנָּמַךְ הָיָה וְקָטָן (חֲרִיךְ שׁוֹקֵיהּ), רַבִּי זֵירָא... הָיָה מִתְפַּלֵּל עָלֵינוּ כָּל יוֹם, וּמְבַקֵּשׁ רַחֲמִים שֶׁה' לֹא יַעֲנִישׁ אוֹתָנוּ עַל מַעֲשֵׂינוּ הָרָעִים... הוּא הָיָה הַיָּחִיד בָּעוֹלָם שֶׁאָהַב אוֹתָנוּ!".

"נָכוֹן," מִלְמֵל בִּרְיוֹן שֵׁנִי, כּוֹבֵשׁ אֶת פָּנָיו בְּיָדָיו, "וְעַכְשָׁו? עַכְשָׁו הוּא מֵת. מִי יְבַקֵּשׁ רַחֲמִים עֲלֵינוּ מֵהַקָּדוֹשׁ בָּרוּךְ הוּא כָּעֵת? מִי יַעֲצֹר אֶת הַדִּין שֶׁיָּבוֹא עֲלֵינוּ?".

הַפַּחַד וְהַצַּעַר אָחֲזוּ בָּהֶם כְּכֶבֶל שֶׁל בַּרְזֶל. הֵם הֵבִינוּ לְפֶתַע, יוֹתֵר מִכָּל זְמַן אַחֵר, כַּמָּה רָחוֹק הֵם נָפְלוּ. הֵם הֵבִינוּ שֶׁהַמָּגֵן שֶׁלָּהֶם נֶעֱלַם, וְשֶׁאִם הֵם לֹא יַעֲשׂוּ שִׁנּוּי, הֵם אֲבוּדִים לְנֶצַח.

"אָנוּ לֹא יְכוֹלִים לְהַמְשִׁיךְ כָּךְ!" זָעַק הַבִּרְיוֹן הָרִאשׁוֹן, וְקוֹלוֹ הִדְהֵד בַּסִּמְטְאוֹת, "רַבִּי זֵירָא אָמַר שֶׁאֲנַחְנוּ יְלָדִים שֶׁל ה'! הִגִּיעַ הַזְּמַן שֶׁנּוֹכִיחַ אֶת זֶה! עַכְשָׁו עָלֵינוּ לְהִתְפַּלֵּל עַל עַצְמֵנוּ!".

הַנִּצָּחוֹן בְּבֵית הַמִּדְרָשׁ

וִילָדִים, שָׁם, בְּאֶמְצַע הַלַּיְלָה הַקָּר וְהֶחָשׁוּךְ – קָרָה פֶלֶא שֶׁכְּמוֹתוֹ לֹא נִרְאָה! הָאֲנָשִׁים הַקְּשׁוּחִים וְהָרָעִים, שֶׁפָּגְעוּ בַּאֲנָשִׁים בְּלִי לְמַצְמֵץ, הִתְחִילוּ לִבְכּוֹת מַמָּשׁ כְּמוֹ תִּינוֹקוֹת! הֵם קָרְעוּ אֶת לְבָבָם הַחוֹטֵא. הֵם הֶחְלִיטוּ בַּאֲמִתּוּת שְׁלֵמָה: "לֹא עוֹד עֲבֵרוֹת! לֹא עוֹד גֶּזֶל! אָנוּ חוֹזְרִים בִּתְשׁוּבָה שְׁלֵמָה לַה' כְּדֵי שֶׁרַבִּי זֵירָא יִשְׂמַח בָּנוּ לְמַעְלָה בַּשָּׁמַיִם!".

לְמָחֳרַת בַּבֹּקֶר. תַּלְמִידֵי הַחֲכָמִים יָשְׁבוּ בְּבֵית הַמִּדְרָשׁ שְׁקוּעִים בְּאֵבֶל וּבְלִמּוּד.

וְהִנֵּה – הַדְּלָתוֹת נִפְתְּחוּ בַּחֲרִיקָה. צְעָדִים כְּבֵדִים נִשְׁמְעוּ עַל הַמַּדְרֵגוֹת. כָּל תַּלְמִידֵי הַחֲכָמִים עָצְרוּ אֶת נְשִׁימָתָם וְהִסְתּוֹבְבוּ לַפֶּתַח...

מִי עָמַד שָׁם? אֵלּוּ הָיוּ הַבִּרְיוֹנִים הַמַּפְחִידִים בְּיוֹתֵר שֶׁל הָעִיר!

אֲבָל חַכּוּ... מָה הֵם עָשׂוּ?

הֵם לֹא בָּאוּ כְּדֵי לְהַפְרִיעַ אוֹ לַעֲשׂוֹת צָרוֹת. רָאשֵׁיהֶם הָיוּ מְכֻסִּים. צִיצִיּוֹת אֲרֻכּוֹת וִיהוּדִיּוֹת הִשְׁתַּלְשְׁלוּ מֵעַל בִּגְדֵיהֶם. עֵינֵיהֶם הָיוּ נְפוּחוֹת מִבֶּכִי, וּבִידֵיהֶם סִפְרֵי תּוֹרָה צְהֻבִּים וִישָׁנִים שֶׁהֵם מָצְאוּ.

הֵם צָעֲדוּ בִּכְבֵדוּת וּבַעֲנָוָה, הִתְיַשְּׁבוּ עַל סַפְסְלֵי הָעֵץ מֵאָחוֹר, וּבִקְּשׁוּ לִלְמֹד תּוֹרָה.

הַחֲכָמִים שֶׁיָּשְׁבוּ שָׁם לֹא הֶאֱמִינוּ לְמַרְאֵה עֵינֵיהֶם. הַבִּרְיוֹנִים? שׁוֹמְרֵי תּוֹרָה וּמִצְווֹת? חוֹזְרִים בִּתְשׁוּבָה?

וְאָז, הֵבִינוּ שְׁאָר הָרַבָּנִים אֶת כֹּחוֹ הָעָצוּם וְסוֹדוֹ שֶׁל הַצַּדִּיק רַבִּי זֵירָא: הָאַהֲבָה הַטְּהוֹרָה וְהָאֱמוּנָה בַּנְּשָׁמָה שֶׁל הַיְּהוּדִי, הֵן תָּמִיד מְנַצְּחוֹת. הָאוֹר שֶׁל רַבִּי זֵירָא, גַּם לְאַחַר פְּטִירָתוֹ, נִצַּח אֶת הַחֹשֶׁךְ.

מקורות: תלמוד בבלי, מסכת סנהדרין, דף ל"ז עמוד א'. מדרשי חז"ל וספר "כה עשו חכמינו", חלק ד', עמוד 99 (סיפור רבי זירא והבריונים).

לפני ואחרי

מה קרה לפני: “אֲנִי עָלַי לִמְסֹר נַפְשִׁי בִּתְפִלָּה עֲלֵיהֶם עַד שֶׁיַּחְזְרוּ בִּתְשׁוּבָה!".”?
מה קרה אחרי: “הֵם הֵבִינוּ לְפֶתַע, יוֹתֵר מִכָּל זְמַן אַחֵר, כַּמָּה רָחוֹק הֵם נָפְלוּ.”?

סיבה ותוצאה מתוך המעשה

האירוע הנתוןמה גרם לו?מה הייתה תוצאתו?
הוּא הָיָה מֵרִים אֶת עֵינָיו הַמְּאִירוֹת כְּכוֹכָבִים אֶל פַּרְצוּפֵיהֶם הַקְּשׁוּחִים, מְחַיֵּךְ אֲלֵיהֶם מַמָּשׁ, וּמְדַבֵּר אִתָּם בְּקוֹל רַךְ…
פְּנֵיהֶם שֶׁהָיוּ תָּמִיד נוֹקְשׁוֹת וְזָעוּמוֹת, לְפֶתַע הָיוּ חִוְרוֹת וְשָׁמוּטוֹת.

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “ר' זירא והבריונים” על ר' זירא? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “הֲרֵי אֵלּוּ בִּרְיוֹנִים רְשָׁעִים הָעוֹבְרִים עַל חֻקֵּי הַתּוֹרָה!”.
ג. אילו ר' זירא לא היה פועל כפי שמתואר ב־“הָיָה מִתְפַּלֵּל עָלֵינוּ כָּל יוֹם, וּמְבַקֵּשׁ רַחֲמִים שֶׁה' לֹא יַעֲנִישׁ אוֹתָנוּ עַל מַעֲשֵׂינוּ הָרָעִים...” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “הֵם לֹא בָּאוּ כְּדֵי לְהַפְרִיעַ אוֹ לַעֲשׂוֹת צָרוֹת.” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. יְלָדִים, דַּמְיְנוּ עִיר גְּדוֹלָה בְּבָבֶל שֶׁל יְמֵי קֶדֶם.
2. גּוּפוֹ הָיָה חַלָּשׁ וְעָדִין, מִשּׁוּם שֶׁצָּם תַּעֲנִיּוֹת רַבּוֹת…
3. בְּאִישׁוֹן לַיְלָה, כְּשֶׁדִּמְעוֹתָיו הָיוּ שׁוֹטְפוֹת אֶת סֵפֶר…
4. אַךְ כַּאֲשֶׁר עָלְתָה הַלְּבָנָה בַּלַּיְלָה וְהָרְחוֹבוֹת…
5. וִילָדִים, שָׁם, בְּאֶמְצַע הַלַּיְלָה הַקָּר וְהֶחָשׁוּךְ – קָרָה…
6. הָאוֹר שֶׁל רַבִּי זֵירָא, גַּם לְאַחַר פְּטִירָתוֹ, נִצַּח אֶת…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “לִפְעָמִים הָיוּ מַשְׁמִיעִים נַהֲמַת זִלְזוּל מִתּוֹךְ גְּרוֹנָם וּמַמְשִׁיכִים לְשַׂחֵק בְּקוּבִּיּוֹת שֶׁלָּהֶם.”.
מכתב: כתבו לר' זירא בעקבות “פְּנֵיהֶם שֶׁהָיוּ תָּמִיד נוֹקְשׁוֹת וְזָעוּמוֹת, לְפֶתַע הָיוּ חִוְרוֹת וְשָׁמוּטוֹת.”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “וִילָדִים, שָׁם, בְּאֶמְצַע הַלַּיְלָה הַקָּר וְהֶחָשׁוּךְ – קָרָה פֶלֶא שֶׁכְּמוֹתוֹ לֹא נִרְאָה!”.
דיווח: כתבו ידיעה קצרה שכותרתה “ר' זירא והבריונים”.
