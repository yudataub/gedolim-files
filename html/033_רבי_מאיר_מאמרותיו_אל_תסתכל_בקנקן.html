מאמרותיו: "אל תסתכל בקנקן..."

דף עבודה בעקבות רבי מאיר
שם: כיתה:

המעשה — הסיפור המלא

להלן מערך חינוכי, רגשי ומרתק על רבי מאיר (בעל הנס) ומאמרו המפורסם: "אַל תִּסְתַּכֵּל בַּקַּנְקַן, אֶלָּא בְּמַה שֶּׁיֵּשׁ בּוֹ" . הסיפור משלב את המשל המפורסם של הקנקנים עם המעשה ההיסטורי והמפעים של אהבת התורה של רבי מאיר (הליכתו אחר רבו 'אחר' כדי ללמוד תורה), מעובד ברגישות ובזהירות במיוחד לבנים בכיתות ב'-ד' בחינוך הדתי-לאומי.

הסיפור מנוקד במלואו, ועומד בכל כללי הסטוריטלינג הנדרשים.

פְּתִיחָה: רֵיחוֹת שֶׁל יַיִן וְסוֹדוֹת בַּיֶּקֶב

יְלָדִים, הַאִם פַּעַם יְרַדְתֶּם לְתוֹךְ מַרְתֵּף יָשָׁן, מָקוֹם קָרִיר שֶׁמָּלֵא בְּרֵיחוֹת מְתוּקִים שֶׁל עֲנָבִים?

דַּמְיְנוּ עַכְשָׁו מַרְתֵּף יַיִן גָּדוֹל מִזְּמַן הַתַּנָּאִים. עַל הַמַּדָּפִים כָּאן עָמְדוּ קַנְקַנִּים. בְּצַד אֶחָד עָמַד קַנְקַן זָהָב נוֹצֵץ, מְעֻטָּר בְּאַבְנֵי חֵן שֶׁהֵאִירוּ אֶת הַחֶדֶר. מַמָּשׁ כְּלִי מַלְכוּתִי! וּבַצַּד הַשֵּׁנִי? עָמַד כְּלִי חֶרֶס חִיֵּוּר, פָּשׁוּט, יָשָׁן וּמְעַט מְלֻכְלָךְ מִבּוֹץ וְאָבָק.

אִם הָיוּ מַצִּיעִים לָכֶם לִשְׁתּוֹת – מֵאֵיזֶה קַנְקַן הֱיִיתֶם בּוֹחֲרִים? בְּוַדַּאי מִקַּנְקַן הַזָּהָב!

אֲבָל... לוּ הֱיִיתֶם טוֹעֲמִים, הֱיִיתֶם מְגַלִּים סוֹד: בְּתוֹךְ קַנְקַן הַזָּהָב הַנּוֹצֵץ, הַיַּיִן הֶחְמִיץ וְהָפַךְ לְחֹמֶץ מַר מְאֹד. וְאִלּוּ בְּתוֹךְ כְּלִי הַחֶרֶס הַיָּשָׁן, הִסְתַּתֵּר הַיַּיִן הֲכִי מָתוֹק, הֲכִי בָּרִיא וַהֲכִי מְשֻׁבָּח שֶׁטְּעַמְתֶּם מֵעוֹדְכֶם!

אֶת הַסּוֹד הַזֶּה, יְלָדִים, יָדַע הַתַּנָּא הַקָּדוֹשׁ רַבִּי מֵאִיר .

לֹא סְתָם קָרְאוּ לוֹ "מֵאִיר". חֲזַ"ל אוֹמְרִים שֶׁשְּׁמוֹ הָיָה מֵאִיר מִפְּנֵי שֶׁ"הָיָה מֵאִיר עֵינֵי חֲכָמִים בַּהֲלָכָה". רַבִּי מֵאִיר יָדַע לְהִסְתַּכֵּל הֵיטֵב בְּתוֹךְ דְּבָרִים וּבְתוֹךְ אֲנָשִׁים, וְלִמְצֹא אֶת הָאוֹר שֶׁמִּסְתַּתֵּר בִּפְנִים. וְזֶה הַסִּפּוּר עַל הַמִּבְחָן הֲכִי גָּדוֹל שֶׁלּוֹ.

הַהֶלֶם בְּשַׁבַּת קֹדֶשׁ

הָיָה זֶה יוֹם שַׁבַּת קֹדֶשׁ בְּעִיר תְּבֶרְיָה. אֲוִיר הַשַּׁבָּת הָיָה שָׁקֵט וְרָגוּעַ. רֵיחַ שֶׁל שֶׁמֶן זַיִת טָהוֹר מֵהַמְּנוֹרוֹת עָמַד בָּאֲוִיר, וְקוֹלוֹת רַכִּים שֶׁל נִגּוּנֵי תּוֹרָה בָּקְעוּ מִבֵּית הַמִּדְרָשׁ. רַבִּי מֵאִיר יָשַׁב בְּרֹאשׁ הַשֻּׁלְחָן וְדָרַשׁ בְּפָרָשַׁת הַשָּׁבוּעַ, וּמֵאוֹת תַּלְמִידִים הִקְשִׁיבוּ מְרֻתָּקִים.

לְפֶתַע, קוֹל זָר הֵפֵר אֶת הַשֶּׁקֶט.

טְרַק-טְרַק, טְרַק-טְרַק .

קוֹל שַׁעֲטַת פְּרָסוֹת סוּס מְהִירוֹת עָלָה מִן הָרְחוֹב! הַתַּלְמִידִים הִבִּיטוּ זֶה בָּזֶה בְּהֶלֶם. בְּשַׁבָּת? מִי רוֹכֵב עַל סוּס בְּשַׁבָּת קֹדֶשׁ? זֶהוּ חִלּוּל שַׁבָּת נוֹרָא וְאָיוֹם!

אֶחָד הַתַּלְמִידִים הֵצִיץ מִבַּעַד לַחַלּוֹן, צֶבַע פָּנָיו הֶחְוִיר (נִהְיָה לָבָן כְּסִיד) וְהוּא זָעַק בִּלְחִישָׁה: "רַבִּי... רַבִּי מֵאִיר... זֶהוּ רַבְּךָ. זֶהוּ אֱלִישָׁע...".

יְלָדִים, פַּעַם הָיָה לְרַבִּי מֵאִיר רַב גָּדוֹל בַּתּוֹרָה שֶׁקָּרְאוּ לוֹ אֱלִישָׁע בֶּן אֲבוּיָה. הוּא הָיָה חָכָם עָצוּם. אַךְ בְּמַהֲלַךְ הַשָּׁנִים קָרָה מַשֶּׁהוּ נָורָא: אֱלִישָׁע הִתְבַּלְבֵּל, טָעָה, עָזַב אֶת דֶּרֶךְ הַתּוֹרָה, וְהֵחֵל לְחַלֵּל שַׁבָּת וְלַעֲשׂוֹת מַעֲשִׂים רָעִים. לָכֵן, שְׁאַר הַחֲכָמִים הִפְסִיקוּ לִקְרֹא לוֹ בִּשְׁמוֹ, וְקָרְאוּ לוֹ מֵאָז "אַחֵר" – כְּלוֹמַר, אִישׁ זָר. הַקַּנְקַן שֶׁל אֱלִישָׁע נִשְׁבַּר. מִבַּחוּץ, הוּא נִרְאָה רַע מְאֹד.

הַהֲלִיכָה בָּאָבָק אֶל הַתּוֹרָה

מָה הָיָה עוֹשֶׂה רַב רָגִיל בְּרֶגַע כָּזֶה? סוֹגֵר אֶת הַחַלּוֹן, מֵסִיט אֶת מַבָּטוֹ בְּכַעַס, וְאוֹמֵר: "אָסוּר לְהִסְתַּכֵּל עָלָיו! הוּא אִישׁ רָע!".

אֲבָל רַבִּי מֵאִיר בַּעַל הַנֵּס קָם מִכִּסְאוֹ, סָגַר בַּעֲדִינוּת אֶת סֵפֶר הַתּוֹרָה הַפָּתוּחַ, קָשַׁר אֶת גְּלִימָתוֹ, יָצָא מִבֵּית הַמִּדְרָשׁ... וְהִתְחִיל לָלֶכֶת מֵאֲחוֹרֵי הַסּוּס שֶׁל "אַחֵר"!

הַתַּלְמִידִים יָצְאוּ הַחוּצָה וְלֹא הֶאֱמִינוּ לְמַרְאֵה עֵינֵיהֶם. רַבָּם הַקָּדוֹשׁ צָעַד בָּרֶגֶל, נוֹשֵׁם אֶת הָאָבָק הַדַּק שֶׁעָלָה מִפַּרְסוֹת הַסּוּס שֶׁל הַחוֹטֵא. הַשֶּׁמֶשׁ קָפְחָה עַל רֹאשׁוֹ.

לָמָּה הוּא עָשָׂה זֹאת?

רַבִּי מֵאִיר הִתְקָרֵב אֶל "אַחֵר" וְשָׁאַל: "רַבִּי, כְּשֶׁלָּמַדְנוּ אֶת הַהֲלָכָה הַהִיא, כֵּיצַד פֵּרַשְׁתָּ אֶת הַפָּסוּק?".

"אַחֵר" מִמְּרוֹמֵי הַסּוּס עָנָה, וְדִבְרֵי תּוֹרָה מְתוּקִים כִּדְבַשׁ יָצְאוּ מִפִּיו, פֵּרוּשִׁים עֲמֻקִּים וּפִלְפּוּלִים אֲשֶׁר רַק חָכָם כָּמוֹהוּ יָכֹל הָיָה לְחַדֵּשׁ. וְרַבִּי מֵאִיר הָלַךְ, וְשָׁמַע, וּבָלַע כָּל מִלָּה מְקֻדֶּשֶׁת בְּצִמָּאוֹן אַדִּיר.

רַבִּי מֵאִיר רָאָה אֶת חִלּוּל הַשַּׁבָּת – שֶׁהוּא הַ"קַּנְקַן הֶהָרוּס", אַךְ הַשֵּׂכֶל וְהַיֶּדַע הַתּוֹרָנִי שֶׁבְּתוֹךְ שִׂכְלוֹ שֶׁל "אַחֵר" – הָיוּ הַיַּיִן הַטּוֹב. רַבִּי מֵאִיר בָּרַר אֶת הַתּוֹרָה וּשְׁתָה רַק אוֹתָהּ! מַמָּשׁ הַשְׁגָּחָה פְּרָטִית אִפְשְׁרָה לוֹ לִדְלוֹת אֶת הַנִּיצוֹצוֹת מִתּוֹךְ הַחֹשֶׁךְ.

הַגְּבוּל (תְּחוּם שַׁבָּת)

הֵם הָלְכוּ וְהָלְכוּ. הַתַּלְמִידִים כְּבָר נֶעֶצְרוּ מֵרָחוֹק, דּוֹאֲגִים מְאֹד.

לְפֶתַע, עָצַר "אַחֵר" אֶת סוּסוֹ. הוּא הִסְתּוֹבֵב אֶל רַבִּי מֵאִיר הַמֵּזִיעַ שֶׁעָמַד עַל שְׁבִיל הֶעָפָר, וְאָמַר בְּקוֹל שֶׁקֶט, רוֹעֵד מְעַט:

"מֵאִיר. עֲצֹר. חֲזֹר לַאֲחוֹרֶיךָ".

"מַדּוּעַ?" שָׁאַל רַבִּי מֵאִיר.

"מִפְּנֵי שֶׁסָּפַרְתִּי בִּמְדֻיָּק אֶת צַעֲדֵי פַּרְסוֹת הַסּוּס שֶׁלִּי. וַאֲנִי יוֹדֵעַ שֶׁכָּאן הִגַּעְנוּ לַגְּבוּל הַמֻּתָּר לַהֲלִיכָה בְּשַׁבָּת ('תְּחוּם שַׁבָּת'). אָסוּר לְךָ לָלֶכֶת אֲפִלּוּ צַעַד אֶחָד יוֹתֵר".

"אַחֵר" כְּבָר לֹא שָׁמַר שַׁבָּת בְּעַצְמוֹ, אֲבָל אֶת הַתּוֹרָה יָדַע הֵיטֵב, וְהוּא דָּאַג לִשְׁמֹר עַל תַּלְמִידוֹ הָאָהוּב.

רַבִּי מֵאִיר הִבִּיט בְּעֵינָיו הַכְּבוּיוֹת שֶׁל רַבּוֹ הַיָּשָׁן, וְזָעַק דֶּרֶךְ דְּמָעוֹת שֶׁהֵחֵלּוּ לָרֶדֶת מֵעֵינָיו:

"אַף אַתָּה – חֲזֹר בְּךָ! שׁוּב בִּתְשׁוּבָה! רְאֵה כַּמָּה תּוֹרָה יֵשׁ בְּךָ! הַקָּדוֹשׁ בָּרוּךְ הוּא מְחַכֶּה לְךָ!".

אֲבָל "אַחֵר" מִלְמֵל בְּצַעַר שֶׁהוּא כְּבָר שָׁקַע מִדַּי, וְדוֹהֵר הָלְאָה, עַד שֶׁנֶּעֱלַם כְּנִקֻדָּה קְטַנָּה בָּאֹפֶק.

סוֹד הָרִמּוֹן וְהַקַּנְקַן בְּבֵית הַמִּדְרָשׁ

רַבִּי מֵאִיר חָזַר לְבֵית הַמִּדְרָשׁ, מָלֵא בְּתוֹרָה עֲמֻקָּה.

הַתַּלְמִידִים הַצְּעִירִים נִגְּשׁוּ אֵלָיו. הֵם הָיוּ חֲשָׁשְׁנִים (פחדו לשאול), אֲבָל אֶחָד מֵהֶם תָּפַס אֹמֶץ:

"רַבֵּנוּ מֵאִיר! כֵּיצַד יָכוֹל הָיִיתָ לִלְמֹד תּוֹרָה, אֶת הַדָּבָר הֲכִי קָדוֹשׁ בָּעוֹלָם, מֵאָדָם כָּזֶה שֶׁמְּחַלֵּל שַׁבָּת? אֵיךְ נָגַעְתָּ בַּדָּבָר הַזֶּה?".

חִיֵּךְ הַתַּנָּא הַקָּדוֹשׁ חִיּוּךְ רַךְ וְעָנָה לָהֶם בְּמִשְׁפָּט אֶחָד, שֶׁנִּשְׁאַר מְפֻרְסָם לְכָל דּוֹרוֹת עַם יִשְׂרָאֵל:

"אַל תִּסְתַּכֵּל בַּקַּנְקַן – אֶלָּא בְּמַה שֶּׁיֵּשׁ בּוֹ! יֵשׁ קַנְקַן חָדָשׁ מָלֵא יָשָׁן, וְיָשָׁן שֶׁאֲפִלּוּ חָדָשׁ אֵין בּוֹ" . (אָבוֹת ד, כ).

וְאָז, כְּדֵי שֶׁיָּבִינוּ מַמָּשׁ טוֹב, הֵבִיא לָהֶם דֻּגְמָה עֲסִיסִית שֶׁל אֹכֶל (וּמִי לֹא מֵבִין שֶׁמְּדַבְּרִים עַל אֹכֶל?):

"יְלָדִים, אִם תִּמְצְאוּ רִמּוֹן יָפֶה בַּשָּׂדֶה. אֲבָל מִבַּחוּץ הַקְּלִפָּה שֶׁלּוֹ יְבֵשָׁה, טִפָּה רְקוּבָה, מְלֻכְלֶכֶת בְּבּוֹץ וְקוֹצִים. הַאִם תִּזְרְקוּ אוֹתוֹ לַפַּח הַזֶּבֶל בִּגְלַל הַקְּלִפָּה הַמְכֹעֶרֶת?".

הַתַּלְמִידִים צָעֲקוּ: "לֹא רַבִּי! אֲנַחְנוּ נִפְתַּח אוֹתוֹ!".

"נָכוֹן מְאֹד!" חִיֵּךְ רַבִּי מֵאִיר וְאָמַר לָהֶם: "רִמּוֹן מָצָאתִי... תּוֹכוֹ (בִּפְנִים) אָכַלְתִּי, וּקְלִפָּתוֹ (הַבַּחוּץ הָרַע) זָרַקְתִּי!" .

גַּם לְרַבִּי אֱלִישָׁע יֵשׁ קְלִפָּה קָשָׁה וְרָעָה – הַמַּעֲשִׂים שֶׁלּוֹ הֵם רָעִים וְשִׁקְרִיִּים. אַךְ בְּתוֹךְ תּוֹכוֹ יֵשׁ גַּרְעִינִים אֲדֻמִּים וּמְתוּקִים מִדְּבַשׁ – הַתּוֹרָה וְהַחָכְמָה הָאֲמִתִּית. אֲנִי, מֵאִיר, יוֹדֵעַ לֶאֱכֹל רַק אֶת הַגַּרְעִינִים, וְלִזְרֹק אֶת הַקְּלִפָּה הָרָעָה לַפַּח. אֲנִי לֹא מִסְתַּכֵּל עַל קַנְקַן הַחֶרֶס הַשָּׁבוּר, אֲנִי שׁוֹתֶה אֶת הַיַּיִן הַטּוֹב שֶׁבְּתוֹכוֹ!

הַתַּלְמִידִים הֵבִינוּ אֶת גְּדֻלָּתוֹ וְעַנְוְתָנוּתוֹ שֶׁל רַבָּם, וְהַבַּת-קוֹל הַיּוֹרֶדֶת מִן הַשָּׁמַיִם רָאֲתָה אֶת אַהֲבָתוֹ הַנְּקִיָּה לְתוֹרָה – שֶׁלֹּא נִרְתְּעָה מִשּׁוּם מִכְשׁוֹל, וְנֶחְקְקָה עַל שְׁמוֹ דֶּרֶךְ נֵצַח שֶׁל "אַל תִּסְתַּכֵּל בַּקַּנְקַן".

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “לָכֵן, שְׁאַר הַחֲכָמִים הִפְסִיקוּ לִקְרֹא לוֹ בִּשְׁמוֹ, וְקָרְאוּ לוֹ מֵאָז "אַחֵר" – כְּלוֹמַר, אִישׁ זָר.”.
מכתב: כתבו לרבי מאיר בעקבות “רַבִּי מֵאִיר הִבִּיט בְּעֵינָיו הַכְּבוּיוֹת שֶׁל רַבּוֹ הַיָּשָׁן, וְזָעַק דֶּרֶךְ דְּמָעוֹת שֶׁהֵחֵלּוּ לָרֶדֶת מֵעֵינָיו:”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “נָכוֹן מְאֹד!" חִיֵּךְ רַבִּי מֵאִיר וְאָמַר לָהֶם: "רִמּוֹן מָצָאתִי...”.
דיווח: כתבו ידיעה קצרה שכותרתה “מאמרותיו: "אל תסתכל בקנקן..."”.

המחזה

רגע מן הסיפורהדמויותמה יאמרו או יעשו?
קוֹל שַׁעֲטַת פְּרָסוֹת סוּס מְהִירוֹת עָלָה מִן הָרְחוֹב!
חִיֵּךְ הַתַּנָּא הַקָּדוֹשׁ חִיּוּךְ רַךְ וְעָנָה לָהֶם בְּמִשְׁפָּט אֶחָד, שֶׁנִּשְׁאַר מְפֻרְסָם…

7. חיבור לחיים מתוך המעשה

א. מה אפשר ללמוד מן הדרך שבה רבי מאיר פעל באירוע “הוּא הִסְתּוֹבֵב אֶל רַבִּי מֵאִיר הַמֵּזִיעַ שֶׁעָמַד עַל שְׁבִיל הֶעָפָר, וְאָמַר בְּקוֹל שֶׁקֶט, רוֹעֵד מְעַט:”?
