האילמים שביקש עליהם רחמים

דף עבודה בעקבות רבי יהודה הנשיא (רבי)
שם: כיתה:

המעשה — הסיפור המלא

פְּתִיחָה: הָרַעַשׁ הַקָּדוֹשׁ שֶׁל בֵּית הַמִּדְרָשׁ

יְלָדִים, הַאִם פַּעַם יְצָא לָכֶם לַעֲמֹד בַּחוּץ לְיַד בֵּית מִדְרָשׁ גָּדוֹל, אֲפִלּוּ בְּלִי לְהִכָּנֵס, רַק לַעֲצֹם עֵינַיִם וּלְהַקְשִׁיב? יֵשׁ לָזֶה קוֹל מְיֻחָד. קוֹל עָצוּם שֶׁל אֲלָפִים הַלּוֹמְדִים יַחַד – כְּמוֹ גַּלִּים חֲזָקִים שֶׁל יָם נִפְלָא שֶׁשּׁוֹטְפִים אֶת הָעוֹלָם.

זֶהוּ בְּדִיּוּק הַקּוֹל שֶׁהָיָה אֶפְשָׁר לִשְׁמֹעַ לִפְנֵי כְּאַלְפַּיִם שָׁנָה, בְּבֵית מִדְרָשׁוֹ שֶׁל גְּדוֹל הַדּוֹר, רַבִּי יְהוּדָה הַנָּשִׂיא – שֶׁכֻּלָּם קָרְאוּ לוֹ פָּשׁוּט "רַבִּי".

בְּתוֹךְ בֵּית הַמִּדְרָשׁ, הָאֲוִירָה הָיְתָה קְדוֹשָׁה וְחַמָּה. רֵיחַ נָעִים שֶׁל סִפְרֵי קְלָף יְשָׁנִים עָמַד בָּאֲוִיר, מְעוֹרָב בְּרֵיחַ נֵרוֹת הַשֶּׁמֶן הַדּוֹלְקִים. וְרַבִּי יוֹשֵׁב בָּרֹאשׁ. רַבִּי הָיָה נָשִׂיא וְאָדָם עָשִׁיר מְאֹד, אַךְ יוֹתֵר מִכָּל יַהֲלוֹם שֶׁלּוֹ – הַדָּבָר שֶׁהוּא אָהַב בְּיוֹתֵר בָּעוֹלָם, הָיוּ הַתַּלְמִידִים שֶׁלּוֹ. כְּשֶׁתַּלְמִיד שָׁאַל קֻשְׁיָה חֲזָקָה, עֵינָיו שֶׁל רַבִּי הָיוּ בּוֹרְקוֹת מִשִּׂמְחָה!

הָאַחִים בַּשּׁוּרָה הָרִאשׁוֹנָה

אֲבָל... בַּשּׁוּרָה הָרִאשׁוֹנָה, תָּמִיד בְּאוֹתוֹ מָקוֹם, יָשְׁבוּ שְׁנֵי בַּחוּרִים צְעִירִים. אֵלֶּה הָיוּ שְׁנֵי אַחִים, נֶכְדָּיו שֶׁל אָדָם קָדוֹשׁ מְאֹד בְּשֵׁם רַבִּי יוֹחָנָן בֶּן גּוּדְגְּדָא. הֵם תָּמִיד הִגִּיעוּ רִאשׁוֹנִים. הַבְּגָדִים שֶׁלָּהֶם הָיוּ נְקִיִּים וּמְסֻדָּרִים, וְהֵם לֹא הִפְסִידוּ אֲפִלּוּ שִׁעוּר אֶחָד.

הָיָה רַק דָּבָר אֶחָד חָסֵר לָהֶם.

הַשְּׁנַיִם הָאֵלֶּה הָיוּ אִלְּמִים .

מֵעוֹלָם לֹא יָצְאָה מִלָּה מִפִּיהֶם. כְּשֶׁהֵם קָמוּ בַּבֹּקֶר – הֵם לֹא יָכְלוּ לוֹמַר בְּקוֹל "מוֹדֶה אֲנִי". כְּשֶׁרָצוּ חֲתִיכַת לֶחֶם – הֵם סִמְּנוּ בַּיָּדַיִם.

וּמָה בְּבֵית הַמִּדְרָשׁ? יְלָדִים, לִלְמֹד תּוֹרָה זֶה לָשֶׁבֶת עִם חָבֵר (חַבְרוּתָא), לִשְׁאֹל קֻשְׁיָוֹת, לְתָרֵץ תֵּרוּצִים, לִצְעֹק מִתּוֹךְ הֲבָנָה: "אַהּ! עַכְשָׁו הֵבַנְתִּי!".

אֲבָל הָאַחִים הָאֵלֶּה לֹא יָכְלוּ. אֲנָשִׁים מִסָּבִיב הִסְתַּכְּלוּ עֲלֵיהֶם בְּרַחֲמִים. "אֵיזֶה חֲבָל," לָחֲשׁוּ הַתַּלְמִידִים, "לָמָּה הֵם יוֹשְׁבִים כָּאן שָׁעוֹת? הֲרֵי הֵם לֹא יוֹכְלוּ לְעוֹלָם לִהְיוֹת רַבָּנִים שֶׁמְּלַמְּדִים. מִי בִּכְלָל יוֹדֵעַ אִם הֵם מְבִינִים מַשֶּׁהוּ מִמָּה שֶׁרַבִּי אוֹמֵר?".

הַשְּׂפָתַיִם הָרוֹחֲשׁוֹת

אֲבָל רַבִּי יְהוּדָה הַנָּשִׂיא רָאָה מַשֶּׁהוּ אַחֵר לְגַמְרֵי. רַבִּי רָאָה לָהֶם לְתוֹךְ הַנְּשָׁמָה.

בְּכָל פַּעַם שֶׁרַבִּי הָיָה נִכְנָס לְבֵית הַמִּדְרָשׁ, מִיָּד הָיוּ הַשְּׁנַיִם נִכְנָסִים אַחֲרָיו בִּזְרִיזוּת. כְּשֶׁרַבִּי הָיָה יוֹשֵׁב וְדוֹרֵשׁ, בְּקוֹל מָלֵא הוֹד וַהֲדָר, הָאַחִים לֹא הִסְתַּכְּלוּ יָמִינָה, לֹא הִסְתַּכְּלוּ שְׂמֹאלָה, לֹא צִיְּרוּ עַל כְּרִיכַת הַסֵּפֶר וְלֹא הִשְׁתַּעַמְמוּ.

הֵם רָכְנוּ קָדִימָה. הָעֵינַיִם הַגְּדוֹלוֹת שֶׁלָּהֶם הִבִּיטוּ יָשָׁר אֶל פָּנָיו שֶׁל רַבִּי. רֹאשָׁם זָז לְמַעְלָה וּלְמַטָּה עִם כָּל מִלָּה! וּמָה הֵם עָשׂוּ עוֹד? חֲזַ"ל מְסַפְּרִים בַּגְּמָרָא (בַּשָּׂפָה הָאֲרָמִית): "וּמְרַחֲשִׁי בְּשִׂפְוָתַיְיהוּ!" – מְזִיזִים אֶת הַשְּׂפָתַיִם שֶׁלָּהֶם עוֹד וָעוֹד! "טּוּק, טּוּק, טּוּק", בְּלִי קוֹל, אֲבָל הַשְּׂפָתַיִם חוֹזְרוֹת בַּלֵּב עַל הַדְּבָרִים שֶׁהֵם לֹא יְכוֹלִים לְהוֹצִיא בַּפֶּה.

רַבִּי אָהַב אוֹתָם. הוּא הִרְגִּישׁ אֶת הַכְּאֵב שֶׁלָּהֶם. "כַּמָּה אָהֲבָה יֵשׁ לַבַּחוּרִים הָאֵלֶּה לַתּוֹרָה הַקְּדוֹשָׁה!" חָשַׁב רַבִּי בְּלִבּוֹ. "הֵם מִתְאַמְּצִים יוֹתֵר מִכֻּלָּם! אִם רַק הָיָה לָהֶם קוֹל... אֵיזֶה אוֹר הֵם הָיוּ מְבִיאִים לָעוֹלָם!".

יוֹם אֶחָד, כְּשֶׁרַבִּי לִמֵּד סֻגְיָה עֲמוּקָה בִּמְיֻחָד, הוּא הִבִּיט בַּשְּׁנַיִם. הוּא רָאָה אֵיךְ הֵם מְנַדְנְדִים בְּרֹאשָׁם בְּעָצְמָה, פְּנֵיהֶם מַאֲדִימוֹת מִמַּאֲמָץ לְהָבִין, לְנַסּוֹת לְתָרֵץ בְּשִׂפְתֵיהֶם, אַךְ שׁוּם קוֹל אֵינוֹ נִשְׁמָע. דִּמְעָה חַמָּה זָלְגָה עַל לֶחְיוֹ שֶׁל הַנָּשִׂיא. "עַד מָתַי בָּנַי הַיְּקָרִים יִסְבְּלוּ?" אָמַר לְעַצְמוֹ.

תְּפִלָּה מֵעוֹמֶק הַלֵּב

רַבִּי עָצַר אֶת הַשִּׁעוּר. בֵּית הַמִּדְרָשׁ כֻּלּוֹ שָׁתַק.

גְּדוֹל הַדּוֹר, שֶׁעָלָיו נֶאֱמַר "כָּל מַה שֶּׁרַבִּי גּוֹזֵר, הַקָּדוֹשׁ בָּרוּךְ הוּא מְקַיֵּם", קָם מִכִּסְּאוֹ. עֵינָיו הָיוּ עֲצוּמוֹת בְּרִכּוּז עָצוּם, וְהוּא פָּנָה לְבַקֵּשׁ רַחֲמִים לִפְנֵי אֲדוֹן הָעוֹלָם.

רַבִּי בִּקֵּשׁ עֲלֵיהֶם רַחֲמִים מִתּוֹךְ אַהֲבָה שֶׁל אַבָּא שֶׁכּוֹאֵב לוֹ עַל יְלָדָיו!

"רִבּוֹנוֹ שֶׁל עוֹלָם!" תּוֹרָה אִתְּךָ אוֹר מַאִיר, אָנָּא רַחֵם עַל הַתַּלְמִידִים הַקְּדוֹשִׁים הָאֵלֶּה. תִּרְאֶה אֶת הַמַּאֲמָץ שֶׁלָּהֶם, תִּרְאֶה כַּמָּה הֵם אוֹהֲבִים אוֹתְךָ וְאֶת הַתּוֹרָה שֶׁלְּךָ. פְּתַח לָהֶם אֶת הַשְּׂפָתַיִם שֶׁיּוּכְלוּ לְבָרֵךְ אוֹתְךָ בְּקוֹל!".

הַקָּדוֹשׁ בָּרוּךְ הוּא, אָבִינוּ הָרַחְמָן שֶׁשּׁוֹמֵעַ תְּפִלַּת צַדִּיקִים, כִּוֵּן אֶת הַהַשְׁגָּחָה הַפְּרָטִית בְּרֶגַע אֶחָד עָצוּם שֶׁל יְשׁוּעָה.

הַנֵּס וְהַגִּלּוּי הַמַּדְהִים

לְפֶתַע... זֶה קָרָה.

שְׁנֵי הָאַחִים הִסְתַּכְּלוּ זֶה עַל זֶה בְּתַדְהֵמָה. הֵם הִרְגִּישׁוּ שֶׁמַּשֶּׁהוּ בַּגָּרוֹן שֶׁלָּהֶם מִשְׁתַּחְרֵר. הַשְּׂפָתַיִם הָרוֹעֲדוֹת שֶׁלָּהֶם נִפְתְּחוּ. אֶחָד מֵהֶם הֵרִים אֶת קוֹלוֹ... יְלָדִים, שָׁנִים אֲרֻכּוֹת שֶׁאִישׁ לֹא שָׁמַע אוֹתָם, וְעַכְשָׁו, קוֹל זַךְ, מָתוֹק וּבָרוּר מִלֵּא אֶת חֲלַל בֵּית הַמִּדְרָשׁ!

"רַבִּי!" זָעַק הָאֶחָד בְּקוֹל נִרְגָּשׁ.

כָּל הַתַּלְמִידִים קָפְצוּ מִכִּסְאוֹתֵיהֶם שׁוֹקִים! נֵס!! נֵס גָּלוּי בְּבֵית הַמִּדְרָשׁ! "הֵם תְּרוּפְאוּ!" עָבְרָה לְחִישָׁה נִרְגֶּשֶׁת, "אֵלּוּ הָאִלְּמִים, הֵם דִּבְּרוּ!".

אַךְ זֶה עוֹד כְּלוּם לְעֻמַּת הַהַפְתָּעָה שֶׁחִכְּתָה לָהֶם.

כֻּלָּם הָיוּ בְּטוּחִים שֶׁבִּגְלַל שֶׁשָּׁנִים הֵם הָיוּ אִלְּמִים, בֶּטַח קָשֶׁה לָהֶם לְהָבִין דְּבָרִים עֲמֻקִּים. רַבִּי חִיֵּךְ לִקְרָאתָם וּבִקֵּשׁ מֵהֶם לוֹמַר מַשֶּׁהוּ בַּתּוֹרָה, רַק כְּדֵי לִשְׁמֹעַ אֶת קוֹלָם לִפְנֵי כֻּלָּם.

כְּשֶׁהֵם פָּתְחוּ אֶת הַפֶּה – יָצָא אֵשׁ שֶׁל קְדֻשָּׁה!

הֵם חָזְרוּ עַל הַשִּׁעוּר! וְלֹא רַק עַל הַשִּׁעוּר שֶׁל הַיּוֹם – הֵם הוֹכִיחוּ שֶׁהֵם זָכְרוּ הַכֹּל! "וְאִשְׁתְּכַח דַּהֲווֹ גְּמִירֵי הִלְכְתָא, וְסִפְרָא, וְסִפְרֵי וְתוֹסֶפְתָּא וְכוּלֵיהּ הַשַּׁ"ס!" (הִתְבָּרֵר שֶׁהֵם יָדוּעוּ אֶת הַהֲלָכָה, אֶת סִפְרֵי הַמִּדְרָשׁ הַמֻּרְכָּבִים מְאֹד, אֶת הַתּוֹסֶפְתָּא וְאֶת כָּל תּוֹרַת חֲזַ"ל עַד אוֹתוֹ יוֹם!).

אֵיךְ? אֵיךְ הֵם יָדְעוּ?

כִּי לִלְמֹד תּוֹרָה – הַשָּׁלָב הָרִאשׁוֹן וְהַכִּי חָשׁוּב בּוֹ, עוֹד לִפְנֵי הַדִּבּוּר – הִיא הַהַקְשָׁבָה מִתּוֹךְ צִמָּאוֹן . כְּשֶׁתַּלְמִיד אָסוּף כֻּלּוֹ, רָכוּן קָדִימָה וְשׂוֹתֶה אֶת הַמִּלִּים, הַתּוֹרָה חֲקוּקָה לוֹ בַּלֵּב.

1. שאלות ידע מתוך הסיפור

א. מה מלמד המעשה “האילמים שביקש עליהם רחמים” על רבי יהודה הנשיא (רבי)? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “טּוּק, טּוּק, טּוּק", בְּלִי קוֹל, אֲבָל הַשְּׂפָתַיִם חוֹזְרוֹת בַּלֵּב עַל הַדְּבָרִים שֶׁהֵם לֹא יְכוֹלִים לְהוֹצִיא בַּפֶּה.”.
ג. אילו רבי יהודה הנשיא (רבי) לא היה פועל כפי שמתואר ב־“הַקָּדוֹשׁ בָּרוּךְ הוּא, אָבִינוּ הָרַחְמָן שֶׁשּׁוֹמֵעַ תְּפִלַּת צַדִּיקִים, כִּוֵּן אֶת הַהַשְׁגָּחָה הַפְּרָטִית בְּרֶגַע…” — כיצד היה הסיפור עשוי להסתיים?
