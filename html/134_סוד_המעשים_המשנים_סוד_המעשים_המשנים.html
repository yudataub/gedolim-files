סוד המעשים המשנים

דף עבודה בעקבות סוד המעשים המשנים
שם: כיתה:

המעשה — הסיפור המלא

הסיפור מנוקד במלואו, כנדרש.

🔹 שָׁלָב 1: בֵּרוּר מַקְדִּים וְנִתּוּחַ (לַמּוֹרֶה/לַמְסַפֵּר)

1. בִּטָּחוֹן בַּחֲכָמִים: לֹא תָּמִיד אֲנַחְנוּ מְבִינִים מַעֲשִׂים שֶׁל צַדִּיקִים אוֹ הוֹרִים, אַךְ תָּמִיד יֵשׁ סִבָּה נְכוֹנָה וּקְדוֹשָׁה.

2. גֶּזֶל וְחֶפְצֵי אֲחֵרִים: זְהִירוּת עֲצוּמָה שֶׁלֹּא לְבַזְבֵּז זְמַן שֶׁל מַעֲבִיד, וְלִשְׁמֹר עַל חֵפֶץ שֶׁשָּׁאַלְתִּי יוֹתֵר מֵאֲשֶׁר עַל שֶׁלִּי.

🔹 שָׁלָב 2: הַסִּפּוּר הַמָּלֵא – "סוֹד הַמַּעֲשִׂים הַמְשֻׁנִּים"

פְּתִיחָה: אֵיפֹה הַנֶּכֶד שֶׁל חוֹנִי? יְלָדִים, אַתֶּם זוֹכְרִים אֶת חוֹנִי הַמְעַגֵּל שֶׁהוֹרִיד גֶּשֶׁם? שָׁנִים רַבּוֹת לְאַחַר מִכֵּן, שׁוּב הִגִּיעָה בַּצֹּרֶת לְאֶרֶץ יִשְׂרָאֵל. הָאֲדָמָה הָיְתָה יְבֵשָׁה, עֲצֵי הַפְּרִי נָבְלוּ, וְלַכְּבָשִׂים לֹא הָיָה מַיִם לִשְׁתּוֹת. הַחֲכָמִים, גְּדוֹלֵי יִשְׂרָאֵל, יָדְעוּ שֶׁצָּרִיךְ לְחַפֵּשׂ צַדִּיק מְיֻחָד שֶׁיִּתְפַּלֵּל. לְמִי יִפְנוּ? הֵם נִזְכְּרוּ בַּנֶּכֶד שֶׁל חוֹנִי – צַדִּיק קָדוֹשׁ בְּשֵׁם אַבָּא חִלְקִיָּה.

יָצְאָה מִשְׁלַחַת חֲכָמִים מְכֻבֶּדֶת לְחַפֵּשׂ אוֹתוֹ. הֵם דָּפְקוּ בְּדַלְתּוֹ, אַךְ בַּבַּיִת לֹא הָיָה אִישׁ. "הוּא וַדַּאי בַּשָּׂדֶה," אָמְרוּ זֶה לָזֶה.

הַהִתְעַלְּמוּת בַּשָּׂדֶה הַשֶּׁמֶשׁ יָקְדָה בְּחָזְקָה כַּאֲשֶׁר הִגִּיעוּ הַחֲכָמִים לַשָּׂדֶה. מֵרָחוֹק רָאוּ אֶת אַבָּא חִלְקִיָּה אוֹחֵז בְּמַעְדֵּר כָּבֵד וְעוֹדֵר אֶת הָאֲדָמָה הַקָּשָׁה. זֵעָה נִגְּרָה עַל פָּנָיו, וְהוּא עָבַד בִּמְהִירוּת עֲצוּמָה, בְּלִי לַעֲצֹר אֲפִלּוּ לְרֶגַע אֶחָד. נִגְּשׁוּ אֵלָיו הַחֲכָמִים וְקָרְאוּ בְּכָבוֹד: "שָׁלוֹם עָלֶיךָ, רַבִּי!" אַךְ מַה זֶּה?! אַבָּא חִלְקִיָּה בִּכְלָל לֹא סוֹבֵב אֶת רֹאשׁוֹ. הוּא לֹא עָנָה לָהֶם שָׁלוֹם! הוּא פָּשׁוּט הִמְשִׁיךְ לַחְפֹּר: "טְרַאח! טְרַאח!" הַחֲכָמִים הִתְפַּלְּאוּ מְאֹד. צַדִּיק כָּזֶה גָּדוֹל לֹא עוֹנֶה "שָׁלוֹם"? אֲבָל הֵם לֹא כָּעֲסוּ. הֵם הָיוּ בְּטוּחִים: וַדַּאי יֵשׁ לַצַּדִּיק סִבָּה טוֹבָה וּקְדוֹשָׁה. הֵם הֶחְלִיטוּ לְהַמְתִּין בַּצַּד.

הַהֲלִיכָה הַמּוּזָרָה לִפְנוֹת עֶרֶב, סִיֵּם אַבָּא חִלְקִיָּה אֶת עֲבוֹדָתוֹ. הוּא אָסַף חֲבִילַת עֵצִים, וְהִתְכּוֹנֵן לַחֲזֹר לְבֵיתוֹ. וְאָז רָאוּ הַחֲכָמִים סִדְרָה שֶׁל מַעֲשִׂים מוּזָרִים עוֹד יוֹתֵר:

הַחֲכָמִים הָלְכוּ מֵאֲחוֹרָיו, מַבִּיטִים וְשׁוֹתְקִים. הֵם יָדְעוּ – הַצַּדִּיק הַזֶּה הוּא כְּמוֹ חִידָה, וּבְקָרוֹב יִמְצְאוּ אֶת הַתְּשׁוּבוֹת.

הַסּוֹד עַל הַגַּג כְּשֶׁהִגִּיעוּ לְבֵיתוֹ שֶׁל אַבָּא חִלְקִיָּה, רָאוּ דָּבָר שֶׁקָּרַע אֶת לִבָּם. בַּבַּיִת הָיָה רַק מְעַט מְאֹד מָזוֹן. עֹנִי עָצוּם. אַבָּא חִלְקִיָּה חִלֵּק פְּרוּסוֹת קְטַנּוֹת לִילָדָיו, אֲבָל לָרַבָּנִים שֶׁהִגִּיעוּ כְּאוֹרְחִים – לֹא הִצִּיעַ כְּלוּם לֶאֱכֹל. בְּסוֹף הַסְּעוּדָה לָחַשׁ אַבָּא חִלְקִיָּה לְאִשְׁתּוֹ הַצַּדֶּקֶת: "אֲנִי יוֹדֵעַ לָמָּה הַחֲכָמִים בָּאוּ הֵנָּה. הֵם רוֹצִים שֶׁנִּתְפַּלֵּל עַל הַגֶּשֶׁם. בּוֹאִי נַעֲלֶה בְּשֶׁקֶט לַעֲלִיַּת הַגַּג, וְנִתְפַּלֵּל אֶל ה' יִתְבָּרַךְ שֶׁיּוֹשִׁיעַ אֶת עַם יִשְׂרָאֵל. אוּלַי ה' יַעֲנֶה לָנוּ מִיָּד, וְכָךְ הַחֲכָמִים בִּכְלָל לֹא יֵדְעוּ שֶׁזֶּה קָרָה בִּזְכוּתֵנוּ. אֲנַחְנוּ לֹא רוֹצִים כָּבוֹד".

הֵם עָלוּ לַגַּג. הַבַּעַל עָמַד בְּזָוִית אַחַת, וְהָאִשָּׁה עָמְדָה בַּזָּוִית הַשְּׁנִיָּה. שְׁנֵיהֶם הִתְחַנְּנוּ בִּדְמָעוֹת. פִּתְאוֹם – נָשְׁבָה רוּחַ, וְעֲנָנִים אֲפֹרִים וּשְׁחוֹרִים הוֹפִיעוּ בַּשָּׁמַיִם. וּמֵאֵיזֶה כִּוּוּן בָּאוּ הָעֲנָנִים הָרִאשׁוֹנִים? דַּוְקָא מֵהַפִּנָּה שֶׁבָּהּ עָמְדָה הָאִשָּׁה! (כִּי הִיא הָיְתָה נוֹתֶנֶת לָעֲנִיִּים מָזוֹן מוּכָן וְכָךְ הִצִּילָה אוֹתָם מִיָּד מֵרָעָב). מַבּוּל שֶׁל גִּשְׁמֵי בְּרָכָה הֵחֵל לָרֶדֶת עַל הָאָרֶץ!

פִּתְרוֹן הַחִידוֹת יָרְדוּ אַבָּא חִלְקִיָּה וְאִשְׁתּוֹ אֶל הַחֲכָמִים שֶׁבַּסָּלוֹן, וְשָׁאֲלוּ אוֹתָם כְּאִלּוּ לֹא קָרָה כְּלוּם: "בַּמֶּה נוּכַל לַעֲזֹר לָכֶם, רַבָּנִים נִכְבָּדִים?" אֲבָל הַחֲכָמִים חִיְּכוּ וְאָמְרוּ: "אֲנַחְנוּ יוֹדְעִים שֶׁהַגֶּשֶׁם הַזֶּה בָּא רַק בִּזְכוּת תְּפִלַּתְכֶם! אֲבָל בְּבַקָּשָׁה מִמְּךָ, רַבִּי – הַסְבֵּר לָנוּ אֶת הַמַּעֲשִׂים הַמּוּזָרִים שֶׁעָשִׂיתָ. תָּמִיד סָמַכְנוּ עָלֶיךָ שֶׁיֵּשׁ לְךָ סִבָּה טוֹבָה. מַדּוּעַ לֹא עָנִיתָ לָנוּ שָׁלוֹם?"

הֵשִׁיב לָהֶם אַבָּא חִלְקִיָּה: "פּוֹעֵל שָׂכִיר אֲנִי! שׂוֹכֵר אוֹתִי אָדָם בְּכֶסֶף כְּדֵי לַעֲבֹד בְּכָל רֶגַע וְרֶגַע. אִם אֶעֱצֹר לוֹמַר שָׁלוֹם, וְאַתְחִיל לְשׁוֹחֵחַ אִתְּכֶם – זֶה גֶּזֶל! אֲנִי אֶגְנֹב מֵהַזְּמַן שֶׁל בַּעַל הַבַּיִת שֶׁלִּי! בִּשְׁבִיל שֶׁתְּפִלָּה תִּתְקַבֵּל בַּשָּׁמַיִם, חַיָּבִים לִהְיוֹת יְשָׁרִים כְּמוֹ סַרְגֵּל". "וּמַדּוּעַ לֹא הִנַּחְתָּ אֶת מְעִילְךָ תַּחַת הָעֵצִים כְּדֵי לְרַפֵּד אֶת הַכָּתֵף?" קְשׁוּ הַחֲכָמִים. "כִּי הַבֶּגֶד הַזֶּה הוּא מֻשְׁאָל!" חִיֵּךְ הַצַּדִּיק. "חֲבֵרִי שֶׁלִּי הִשְׁאִיל לִי אוֹתוֹ שֶׁאֶלְבַּשׁ אוֹתוֹ וְלֹא יִהְיֶה לִי קַר. הוּא לֹא הִרְשָׁה לִי לָשִׂים עָלָיו עֵצִים קוֹצְנִיִּים שֶׁיִּקְרְעוּ אוֹתוֹ! אָסוּר לְשַׁנּוֹת בְּחֵפֶץ שֶׁל חָבֵר". "וְלָמָּה נָעַלְתָּ נַעֲלַיִם רַק בְּתוֹךְ הַמַּיִם?" "כִּי עַל שְׁבִיל הָעָפָר אֲנִי רוֹאֶה אֵיפֹה אֲנִי דּוֹרֵךְ. בְּתוֹךְ הַמַּיִם אֵינֶנִּי רוֹאֶה. אֲנִי חוֹשֵׁשׁ שֶׁמָּא יֵשׁ דָּג שֶׁנּוֹשֵׁךְ אוֹ עַקְרָב." "וּמַדּוּעַ הֵרַמְתָּ בְּגָדֶיךָ מֵעַל לַקּוֹצִים וְנָתַתָּ לְרַגְלֶיךָ לְהִשָּׂרֵט?" תָּמְהוּ לְבַסּוֹף. "כִּי רֶגֶל שֶׁנִּפְצַעַת – מִתְרַפֵּאת וְהָעוֹר צּוֹמֵחַ מֵחָדָשׁ," עָנָה אַבָּא חִלְקִיָּה בְּפַשְׁטוּת, "אֲבָל בֶּגֶד שֶׁנִּקְרַע לְאָדָם עָנִי כָּמוֹנִי, אִי אֶפְשָׁר לְרַפֵּא... "

הַחֲכָמִים חִיְּכוּ בְּלֵב מָלֵא הִתְפַּעֲלוּת. הֵם הֵבִינוּ: אַבָּא חִלְקִיָּה מְחַשֵּׁב כָּל פְּעֻלָּה וּפְעֻלָּה! הוּא לֹא מוּזָר – הוּא פָּשׁוּט מוּשְׁלָם בְּמִדּוֹת הַטּוֹבוֹת וּבִזְהִירוּת שֶׁל בֵּין אָדָם לַחֲבֵרוֹ. וּבִזְכוּת הַדִּקְדּוּק הַזֶּה שֶׁל כָּל דַּקָּה וְכָל חֵפֶץ – הַקָּדוֹשׁ בָּרוּךְ הוּא שׁוֹמֵעַ אֶת תְּפִלָּתוֹ וּמֵבִיא חַיִּים לָעוֹלָם!

🔹 שָׁלָב 4: מַדְרִיךְ לַמּוֹרֶה / לַמְסַפֵּר

א. שְׁאֵלַת פְּתִיחָה: "יְלָדִים, אִם הָיִיתִי בּוֹנֶה לִי כָּעֵת אוֹהֶל מֵהַמְּעִיל הֶחָדָשׁ שֶׁלָּכֶם עַל הָרִצְפָּה, הַאִם זֶה הָיָה בְּסֵדֶר? כַּמּוּבָן שֶׁלֹּא! כְּשֶׁשּׁוֹאֲלִים מַשֶּׁהוּ מֵחָבֵר, צָרִיךְ לְהִשְׁתַּמֵּשׁ בּוֹ רַק כְּפִי שֶׁמֻּתָּר לְפִי רְצוֹן הֶחָבֵר. הַיּוֹם נִפְגֹּשׁ צַדִּיק שֶׁשָּׁאַל מְעִיל מֵחָבֵר וְהָלַךְ בְּצוּרָה מְאֹד מוּזָרָה רַק כְּדֵי שֶׁלֹּא יֵהָרֵס".

ב. הֶעָרוֹת בִּצּוּעַ (לַמְסַפֵּר):

מקורות: המקורות מופיעים בקובץ המקורי.

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “בִּטָּחוֹן בַּחֲכָמִים: לֹא תָּמִיד אֲנַחְנוּ מְבִינִים מַעֲשִׂים שֶׁל צַדִּיקִים אוֹ הוֹרִים, אַךְ תָּמִיד יֵשׁ סִבָּה נְכוֹנָה וּקְדוֹשָׁה.” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “הֵם נִזְכְּרוּ בַּנֶּכֶד שֶׁל חוֹנִי – צַדִּיק קָדוֹשׁ בְּשֵׁם אַבָּא חִלְקִיָּה.”?
ג. כיצד הגיב סוד המעשים המשנים למצב המתואר: “וְאָז רָאוּ הַחֲכָמִים סִדְרָה שֶׁל מַעֲשִׂים מוּזָרִים עוֹד יוֹתֵר:”?
ד. מה השתנה בין “אוּלַי ה' יַעֲנֶה לָנוּ מִיָּד, וְכָךְ הַחֲכָמִים בִּכְלָל לֹא יֵדְעוּ שֶׁזֶּה קָרָה בִּזְכוּתֵנוּ.” לבין “שׂוֹכֵר אוֹתִי אָדָם בְּכֶסֶף כְּדֵי לַעֲבֹד בְּכָל רֶגַע וְרֶגַע.”?
ה. כיצד הגיע הסיפור אל האירוע: “הוּא לֹא מוּזָר – הוּא פָּשׁוּט מוּשְׁלָם בְּמִדּוֹת הַטּוֹבוֹת וּבִזְהִירוּת שֶׁל בֵּין אָדָם לַחֲבֵרוֹ.”?

3. מה היה ראשון?

כתבו ליד כל אירוע מספר מ־1 עד 5 לפי סדר התרחשותו בסיפור.

□ בּוֹאִי נַעֲלֶה בְּשֶׁקֶט לַעֲלִיַּת הַגַּג, וְנִתְפַּלֵּל אֶל ה' יִתְבָּרַךְ שֶׁיּוֹשִׁיעַ אֶת עַם יִשְׂרָאֵל.
□ בִּטָּחוֹן בַּחֲכָמִים: לֹא תָּמִיד אֲנַחְנוּ מְבִינִים מַעֲשִׂים שֶׁל צַדִּיקִים אוֹ הוֹרִים, אַךְ תָּמִיד יֵשׁ סִבָּה נְכוֹנָה וּקְדוֹשָׁה.
□ הַיּוֹם נִפְגֹּשׁ צַדִּיק שֶׁשָּׁאַל מְעִיל מֵחָבֵר וְהָלַךְ בְּצוּרָה מְאֹד מוּזָרָה רַק כְּדֵי שֶׁלֹּא יֵהָרֵס".
□ מֵרָחוֹק רָאוּ אֶת אַבָּא חִלְקִיָּה אוֹחֵז בְּמַעְדֵּר כָּבֵד וְעוֹדֵר אֶת הָאֲדָמָה הַקָּשָׁה.
□ אִם אֶעֱצֹר לוֹמַר שָׁלוֹם, וְאַתְחִיל לְשׁוֹחֵחַ אִתְּכֶם – זֶה גֶּזֶל!

לפני ואחרי

מה קרה לפני: “בּוֹאִי נַעֲלֶה בְּשֶׁקֶט לַעֲלִיַּת הַגַּג, וְנִתְפַּלֵּל אֶל ה' יִתְבָּרַךְ שֶׁיּוֹשִׁיעַ אֶת עַם יִשְׂרָאֵל.”?
מה קרה אחרי: “אִם אֶעֱצֹר לוֹמַר שָׁלוֹם, וְאַתְחִיל לְשׁוֹחֵחַ אִתְּכֶם – זֶה גֶּזֶל!”?

סיבה ותוצאה מתוך המעשה

האירוע הנתוןמה גרם לו?מה הייתה תוצאתו?
זֵעָה נִגְּרָה עַל פָּנָיו, וְהוּא עָבַד בִּמְהִירוּת עֲצוּמָה, בְּלִי לַעֲצֹר אֲפִלּוּ לְרֶגַע אֶחָד.
(כִּי הִיא הָיְתָה נוֹתֶנֶת לָעֲנִיִּים מָזוֹן מוּכָן וְכָךְ הִצִּילָה אוֹתָם מִיָּד מֵרָעָב).

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “סוד המעשים המשנים” על סוד המעשים המשנים? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “אַבָּא חִלְקִיָּה חִלֵּק פְּרוּסוֹת קְטַנּוֹת לִילָדָיו, אֲבָל לָרַבָּנִים שֶׁהִגִּיעוּ כְּאוֹרְחִים – לֹא הִצִּיעַ כְּלוּם לֶאֱכֹל.”.
ג. אילו סוד המעשים המשנים לא היה פועל כפי שמתואר ב־“שׂוֹכֵר אוֹתִי אָדָם בְּכֶסֶף כְּדֵי לַעֲבֹד בְּכָל רֶגַע וְרֶגַע.” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “הוּא לֹא מוּזָר – הוּא פָּשׁוּט מוּשְׁלָם בְּמִדּוֹת הַטּוֹבוֹת וּבִזְהִירוּת שֶׁל בֵּין אָדָם לַחֲבֵרוֹ.” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. בִּטָּחוֹן בַּחֲכָמִים: לֹא תָּמִיד אֲנַחְנוּ מְבִינִים מַעֲשִׂים שֶׁל…
2. הֵם דָּפְקוּ בְּדַלְתּוֹ, אַךְ בַּבַּיִת לֹא הָיָה אִישׁ.
3. הֵם יָדְעוּ – הַצַּדִּיק הַזֶּה הוּא כְּמוֹ חִידָה, וּבְקָרוֹב…
4. פִּתְאוֹם – נָשְׁבָה רוּחַ, וְעֲנָנִים אֲפֹרִים וּשְׁחוֹרִים הוֹפִיעוּ…
5. חֲבֵרִי שֶׁלִּי הִשְׁאִיל לִי אוֹתוֹ שֶׁאֶלְבַּשׁ אוֹתוֹ וְלֹא יִהְיֶה…
6. הַיּוֹם נִפְגֹּשׁ צַדִּיק שֶׁשָּׁאַל מְעִיל מֵחָבֵר וְהָלַךְ בְּצוּרָה…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “וְאָז רָאוּ הַחֲכָמִים סִדְרָה שֶׁל מַעֲשִׂים מוּזָרִים עוֹד יוֹתֵר:”.
מכתב: כתבו לסוד המעשים המשנים בעקבות “(כִּי הִיא הָיְתָה נוֹתֶנֶת לָעֲנִיִּים מָזוֹן מוּכָן וְכָךְ הִצִּילָה אוֹתָם מִיָּד מֵרָעָב).”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “הוּא לֹא הִרְשָׁה לִי לָשִׂים עָלָיו עֵצִים קוֹצְנִיִּים שֶׁיִּקְרְעוּ אוֹתוֹ!”.
דיווח: כתבו ידיעה קצרה שכותרתה “סוד המעשים המשנים”.
