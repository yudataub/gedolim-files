כיצד לימד ברכות — המברך והמלגלג

דף עבודה בעקבות בר קפרא
שם: כיתה:

המעשה — הסיפור המלא

פְּתִיחָה: קוֹלוֹת שֶׁל מַיִם וְתּוֹרָה

יְלָדִים, עִצְמוּ אֶת הָעֵינַיִם לְרֶגַע, וּדְמְיְנוּ אֶת עַצְמְכֶם בְּתוֹךְ בֵּית מִדְרָשׁ קָדוּם בְּאֶרֶץ יִשְׂרָאֵל, לִפְנֵי קָרוֹב לְאַלְפַּיִם שָׁנָה. קַרְנֵי שֶׁמֶשׁ שֶׁל צָהֳרַיִם צוֹנְחוֹת דֶּרֶךְ הַחַלּוֹנוֹת הַגְּבוֹהִים, וְגַרְגְּרֵי אָבָק קְטַנִּים מְרַקְּדִים בְּתוֹךְ הָאוֹר הַזָּהוּב. שֶׁקֶט אֵין שָׁם, כִּי אִם קוֹל אַדִּיר וּמָתוֹק שֶׁל תּוֹרָה! מֵאוֹת אֲנָשִׁים לוֹמְדִים בְּקוֹל רָם, וּבָאֶמְצַע יוֹשֵׁב אֶחָד מִגְּדוֹלֵי עֲנָקֵי הָרוּחַ שֶׁבְּאוֹתוֹ הַדּוֹר: הָרַב הַקָּדוֹשׁ בַּר קַפָּרָא .

בַּר קַפָּרָא הָיָה תַּלְמִידוֹ שֶׁל רַבִּי יְהוּדָה הַנָּשִׂיא. שִׂכְלוֹ הָיָה חַד כְּחֶרֶב שֶׁל אֵשׁ, וְכָל רָזֵי הַתּוֹרָה וְהַהֲלָכָה הָיוּ מֻנָּחִים בְּכִיסוֹ. לְצִדּוֹ מַמָּשׁ יָשְׁבוּ שְׁנֵי תַּלְמִידִים צְעִירִים. הֵם הָיוּ אֲהוּבִים וּמוּכְשָׁרִים, כָּאֵלּוּ שֶׁמּוֹסְרִים אֶת כָּל נַפְשָׁם עַל סוּגְיָה בַּגְּמָרָא, וּמִתְאַמְּצִים לְהָבִין כָּל מִלָּה וּמִלָּה מִפִּיו שֶׁל רַבָּם.

הַהֲכָנָה לַסְּעוּדָה וְרֵיחוֹת הַתַּבְשִׁילִים

הַיּוֹם אוֹתוֹ יוֹם אָרֹךְ שֶׁל לִמּוּד הִסְתַּיֵּם. הַבֶּטֶן מְקַרְקֶרֶת, וְהַשָּׁעָה לֶאֱכוֹל סְעוּדַת מִצְוָה הִגִּיעָה. עֲבָדִים הֵחֵלּוּ לַעֲרֹךְ אֶת הַשֻּׁלְחָן הֶעָרוּךְ עֵץ כָּבֵד. תְּחִלָּה יָצְקוּ מַיִם צְנוּנִים וּזְכוּכִיִּים מִתּוֹךְ נֵטְלָא (כְּלִי רַחְצָה) עַל יְדֵיהֶם שֶׁל הַלּוֹמְדִים. קֹר הַמַּיִם מְרַעֲנֵן אֶת מַחֲשָׁבוֹתֵיהֶם.

אַחַר כָּךְ, הֵחֵלּוּ הַמְשָׁרְתִים לְהַגִּישׁ אֶת קְעָרוֹת הַמָּזוֹן. אֵיזֶה רֵיחַ נִפְלָא הִתְפַּשֵּׁט בַּחֶדֶר! רֵיחַ כָּבֵד שֶׁל בָּשָׂר וְעוֹפוֹת צְלוּיִים ("פַּרְגִּיּוֹת") הָעוֹלֶה מֵהַקְּעָרוֹת. אֵדִים חַמִּים עָלוּ אֶל הָאֲוִיר מִתּוֹךְ תַּבְשִׁילִים מְהֻבָּלִים. לְצַד הָעוֹפוֹת הַצְּלוּיִים, הוּבָאוּ לַשֻּׁלְחָן קְעָרוֹת מְלֵאוֹת בִּכְרוּב מְבֻשָּׁל, חַם וְעָרֵב לַחֵךְ.

בַּר קַפָּרָא הִתְיַשֵּׁב, שָׁתַק שְׁתִיקָה שֶׁל דְּבֵקוּת, וְהִמְתִּין. גַּם שְׁנֵי תַּלְמִידָיו יָשְׁבוּ מוּלוֹ, מַמְתִּינִים בְּיִרְאַת שָׁמַיִם לִרְאוֹת כֵּיצַד יִנְהַג רַבָּם הַגָּדוֹל. הֲרֵי קֹדֶם שֶׁאוֹכְלִים – חוֹבָה עָלֵינוּ לְבָרֵךְ אֶת בּוֹרֵא הָעוֹלָם וּלְהוֹדוֹת לוֹ עַל הַמָּזוֹן. אַךְ כָּאן יְלָדִים... הִתְעוֹרְרָה שְׁאֵלָה הִלְכָתִית סְבוּכָה!

עַל הַכְּרוּב, שֶׁהוּא יָרָק, מְבָרְכִים "בּוֹרֵא פְּרִי הָאֲדָמָה". זוֹ בְּרָכָה מְיֻחֶדֶת וַחֲשׁוּבָה, כִּי הִיא קוֹבַעַת בְּדִיּוּק שֶׁהַמַּאֲכָל בָּא מִן הָאֲדָמָה.

לְעֻמַּת זֹאת, עַל בְּשַׂר הָעוֹפוֹת מְבָרְכִים "שֶׁהַכֹּל נִהְיָה בִּדְבָרוֹ" – בְּרָכָה כְּלָלִית יוֹתֵר. מִצַּד הַהֲלָכָה, יֵשׁ מַעֲלָה לְבָרֵךְ קֹדֶם עַל אוֹתוֹ מַאֲכָל שֶׁבִּרְכָּתוֹ חֲשׁוּבָה מִצַּד עַצְמָהּ (כְּלוֹמַר "הָאֲדָמָה"), אֲבָל מִצַּד שֵׁנִי – הָעוֹף הָיָה הַמַּאֲכָל הַ"חָבִיב" יוֹתֵר וְהַטָּעִים יוֹתֵר, שֶׁעָלָיו רָצוּ לְבָרֵךְ רִאשׁוֹן!

הַקְּפִיצָה הַפְּזִיזָה וְהַטָּעוּת

בַּר קַפָּרָא הִבִּיט בַּתַּלְמִידִים. הוּא רָצָה לְכַבֵּד אוֹתָם. הוּא פָּנָה לְאֶחָד מֵהֶם, וְאָמַר לוֹ בְּחִבָּה: "בְּנִי, מְבַקֵּשׁ אֲנִי מִמְּךָ שֶׁתִּטֹּל אֶת הָרְשׁוּת, וּתְבָרֵךְ עֲבוּרֵנוּ".

תָּאֲרוּ לָכֶם, אֵיזֶה כָּבוֹד עָצוּם שֶׁגְּדוֹל הַדּוֹר מְבַקֵּשׁ מִמְּךָ לְבָרֵךְ בְּנוֹכְחוּתוֹ! הַתַּלְמִיד כָּל כָּךְ הִתְרַגֵּשׁ, הַלֵּב שֶׁלּוֹ דָּפַק בְּחָזְקָה: "בּוּם-בּוּם-בּוּם". עֵינָיו נָצְצוּ, וּמֵרֹב הִתְרַגְּשׁוּת אוֹ פְּזִיזוּת, תַּחַת לַעֲצֹר רֶגַע וְלִשְׁאוֹל כַּדָּת וְכַדִּין: "רַבִּי, עַל מָה רָאוּי לְבָרֵךְ תְּחִלָּה? הַאִם עַל הַכְּרוּב הַחֲשׁוּב, אוֹ שֶׁמָּא עַל הָעוֹף הֶחָבִיב?", הוּא מִהֵר מְאֹד. הוּא פָּשַׁט אֶת יָדוֹ אַל הָעוֹף, וְקָרָא בְּקוֹל:

"בָּרוּךְ אַתָּה ה', אֱלֹקֵינוּ מֶלֶךְ הָעוֹלָם, שֶׁהַכֹּל נִהְיָה בִּדְבָרוֹ!"

יְלָדִים, שֶׁקֶט נִהְיָה בַּחֶדֶר. מִיָּד כְּשֶׁסִּיֵּם אֶת הַבְּרָכָה, יָדְעוּ הַיּוֹשְׁבִים שֶׁהוּא טָעָה בַּהֲלָכָה. הוּא דִּלֵּג עַל הַכְּרוּב שֶׁבִּרְכָּתוֹ חֲשׁוּבָה יוֹתֵר. עַכְשָׁו, אָנוּ יוֹדְעִים שֶׁגַּם לִגְדוֹלִים יֵשׁ רְגָעִים שֶׁל בִּלְבּוּל. הַתַּלְמִיד זֶה עַתָּה טָעָה וְהִרְגִּישׁ שֶׁהִתְבַּלְבֵּל. לִבּוֹ נָפַל בְּקִרְבּוֹ.

הַלַּעַג הַפּוֹצֵעַ כְּסַכִּין

אַךְ אָז, קָרָה דָּבָר חָמוּר הַרְבֵּה יוֹתֵר מִטָּעוּת הִלְכָתִית.

הַתַּלְמִיד הַשֵּׁנִי שֶׁיָּשַׁב בַּצַּד, זֶה שֶׁרָאָה אֶת הַטָּעוּת שֶׁל חֲבֵרוֹ – בִּמְקוֹם לִשְׁתֹּק, בִּמְקוֹם לָגֶשֶׁת בַּעֲדִינוּת לַחֲבֵרוֹ וּלְלַמֵּד אוֹתוֹ בְּסֵתֶר – קָפַץ אֶת אֶגְרוֹפָיו, חִיֵּךְ חִיּוּךְ רָחָב, וְהֵחֵל לִלְעֹג!

"חָה חָה חָה!" הוּא צָחַק לְעֵינֵי כָּל הַסּוֹעֲדִים. "תִּרְאוּ אוֹתוֹ! שׁוֹכֵחַ שֶׁבְּרָכַת 'בּוֹרֵא פְּרִי הָאֲדָמָה' קוֹדֶמֶת! עַל מָה מִהַרְתָּ לְקַפֵּץ כְּמוֹ תַּרְנְגוֹל אֶל הַפַּרְגִּיּוֹת? אֵיזֶה חֹסֶר יְדִיעָה!"

הַמִּלִּים שֶׁלּוֹ הָיוּ כְּמוֹ חִצִּים שֶׁל אֵשׁ. הַתַּלְמִיד הָרִאשׁוֹן, הַמְבָרֵךְ, הֵשְׁפִּיל אֶת מַבָּטוֹ, עֵינָיו דָּמְעוּ, וּפָנָיו הֶאֱדִימוּ מִבּוּשָׁה. לִבּוֹ נִשְׁבַּר בְּתוֹכוֹ.

הַשְּׁתִיקָה הַמַּקְפִּיאָה וּתְגוּבָתוֹ שֶׁל הָרַב

יְלָדִים, אִם עַד לְכָאן הָיָה שֶׁקֶט, עַכְשָׁו נִהְיְתָה בַּחֶדֶר הַדְּמָמָה דַּקָּה וְקָרָה כְּקֶרַח. בַּר קַפָּרָא, אוֹתוֹ תַּנָּא קָדוֹשׁ שֶׁפָּנָיו תָּמִיד שׁוֹפְעוֹת אַהֲבָה וְחִיּוּךְ, קָפַץ אֶת שְׂפָתָיו. מַבָּטוֹ הַטָּהוֹר נַעֲשָׂה חָמוּר סֵבֶר וְנוֹקֵב. לַהֲבַת כַּעַס קָדוֹשׁ דָּלְקָה בְּעֵינָיו. הוּא לֹא הִסְכִּים שֶׁבְּבֵית הַמִּדְרָשׁ שֶׁלּוֹ יִרְמְסוּ אֶת כְּבוֹדוֹ שֶׁל אָדָם בְּשֵׁם "הַלִּמּוּד", וְהוּא לֹא הִסְכִּים שֶׁתַּלְמִיד יִפְעַל מִתּוֹךְ גַּאֲוָה וּפְזִיזוּת.

הוּא מָחָא כַּף אֶל כַּף, וְהַקּוֹל הִדְהֵד עַל קִירוֹת הָאֶבֶן. אָז פָּתַח אֶת פִּיו, וּבְקוֹלוֹ הֶעָמֹק וְהַכּוֹאֵב הוֹכִיחַ אֶת שְׁנֵיהֶם – שְׁנֵי מִשְׁפָּטִים שֶׁנֶּחְקְרוּ לָנֶצַח בַּתַּלְמוּד שֶׁלָּנוּ.

תְּחִלָּה הִבִּיט אֶל הַתַּלְמִיד הַמְבָרֵךְ שֶׁקָּפַץ, וְאָמַר לוֹ מִלִּים חֲכָמוֹת אַךְ נוֹקְבוֹת:

"בְּנִי, עֲצֹר רֶגַע וַחְשֹׁב. אִם חָכְמָה אֵין כָּאן – זִקְנָה אֵין כָּאן?! "

הַסְּבָרָה שֶׁל הָרַב הָיְתָה פְּשׁוּטָה: בְּסֵדֶר, אַתָּה לֹא חָכָם גָּדוֹל עֲדַיִן, יֵשׁ הֲלָכוֹת שֶׁאַתָּה אֵינְךָ יוֹדֵעַ הֵיטֵב. וְזֶה בְּסֵדֶר לִטְעוֹת. אֲבָל כְּשֶׁאֵינְךָ יוֹדֵעַ – הַאִם אֵין פֹּה 'זָקֵן' וְרַב?! אֲנִי יָשַׁבְתִּי כָּאן לְצִדְּךָ! הָיָה עָלֶיךָ לִשְׁאוֹל וְלִלְמֹד, לִפְנוֹת בַּעֲנָוָה וְלוֹמַר: "רַבִּי, כֵּיצַד יֵשׁ לִנְהֹג כָּעֵת?" מַדּוּעַ אָצְתָּ לְבָרֵךְ מִבְּלִי לְבָרֵר אֶת הַדִּין כָּרָאוּי כְּשֶׁמּוֹרְךָ נוֹכֵחַ? פְּזִיזוּת אֵינָהּ דַּרְכָּהּ שֶׁל תּוֹרָה!

הַתַּלְמִיד הַמְבָרֵךְ הִכְנִיעַ אֶת רֹאשׁוֹ וְקִבֵּל אֶת הַתּוֹכָחָה מֵרַבּוֹ הַגָּדוֹל, מֵבִין שֶׁעָלָיו לְהַשְׁפִּיל מִגַּאֲוָתוֹ וְתָמִיד לִשְׁאוֹל אֶת הָרַב בְּעֵת סָפֵק.

הַחֵץ שֶׁל בַּר קַפָּרָא אַל הַלַּעַג הָרָע

וְאָז, הִסְתּוֹבֵב הָרַב אֶל הַתַּלְמִיד הַשֵּׁנִי, זֶה שֶׁצָּחַק. הַחִיּוּךְ הַלַּעֲגָן שֶׁל אוֹתוֹ תַּלְמִיד מַמָּשׁ קָפָא עַל שְׂפָתָיו כַּאֲשֶׁר מַבָּטוֹ הַיּוֹקֵד שֶׁל רַבּוֹ נָח עָלָיו.

"וְאַתָּה..." קוֹלוֹ שֶׁל בַּר קַפָּרָא רָעַד מִכְּאֵב, "חֲבֵרְךָ טָעָה, וְלֹא יָדַע כֵּיצַד לְבָרֵךְ... אֲבָל אַתָּה... לָמָּה לִגְלַגְתָּ?! "

בַּר קַפָּרָא זִעֲזֵעַ אֶת כָּל בֵּית הַמִּדְרָשׁ כְּשֶׁהוֹסִיף: "אִם חֲבֵרְךָ הוּא כְּמוֹ אִלֵּם שֶׁאֵינוֹ מֵבִין אֶת הַתּוֹרָה כָּרָאוּי – וְכִי אַתָּה חָכָם גָּדוֹל?! אִם יָדַעְתָּ אֶת הַהֲלָכָה הַנְּכוֹנָה, מַדּוּעַ לֹא לִמַּדְתָּ אוֹתוֹ בְּסֵתֶר וּבְעוֹד מוֹעֵד? הֲרֵי מִי שֶׁצּוֹחֵק עַל חֲבֵרוֹ מִתּוֹךְ כַּוָּנָה לְהַרְאוֹת שֶׁהוּא תַּלְמִיד חָכָם וְטוֹב יוֹתֵר – מוֹכִיחַ שֶׁאֵין בּוֹ טִפָּה שֶׁל תּוֹרָה לִשְׁמָהּ! הַלִּגְלוּג שֶׁלְּךָ הוּא עֲבֵרָה כְּבֵדָה הַרְבֵּה מֵהַטָּעוּת הַהִלְכָתִית שֶׁלּוֹ!"

הַמִּלִּים חָתְכוּ אֶת הָאֲוִיר כְּתַעַר. הַתַּלְמִיד הַשֵּׁנִי חָוַר. עַד אוֹתוֹ רֶגַע הוּא הָיָה בָּטוּחַ שֶׁהוּא "מֵגֵן עַל הַהֲלָכָה", שֶׁהוּא מַרְאֶה לַכֹּל שֶׁהוּא הַבָּקִי מִכֻּלָּם. אַךְ בַּר קַפָּרָא חָשַׂף בְּפָנָיו אֶת הָאֱמֶת הַכּוֹאֶבֶת: חָכְמָה אֲמִתִּית לְעוֹלָם אֵינָהּ הוֹלֶכֶת יַחַד עִם הַלְבָּנַת פָּנִים וְגַאֲוָה.

וְכָךְ, מִתּוֹךְ סְעוּדָה אַחַת הִסְתַּכְּלָה יְשִׁיבָה שְׁלֵמָה וְלָמְדָה אֶת מֻשָּׂגוֹ הֶעָמֹק שֶׁל הַכָּבוֹד בַּתּוֹרָה. בֵּית הַמִּדְרָשׁ לָמַד בְּאוֹתוֹ הַיּוֹם שֶׁרַבּוֹתֵינוּ דּוֹרְשִׁים שְׁלֵמוּת. אָסוּר לָנוּ לִקְפֹּץ וּלְפַעֲנֵחַ הוֹרָאוֹת לְבַד מִבְּלִי לְהִוָּעֵץ בַּזְּקֵנִים וּבַחֲכָמִים הַמְכֻוָּנִים אוֹתָנוּ, וּמֵעַל הַכֹּל – אָסוּר בְּאִסּוּר מֻחְלָט לְגַלְגֵּל לָשׁוֹן וְלִצְחֹק עַל חֲבֵרֵינוּ הַטּוֹעִים. הַתּוֹרָה נִתְּנָה כְּדֵי שֶׁנִּהְיֶה עֲנָוִים וְרוֹדְפֵי שָׁלוֹם, וְלֹא חָלִילָה בַּעֲלֵי גַּאֲוָה מְזַלְזְלִים.

וּמֵאָז, דִּבְרֵי בַּר קַפָּרָא מְהַדְהֲדִים בְּאָזְנֵי עַם יִשְׂרָאֵל, לְמַעַן לֹא יְגַלְגֵּל וְיִלְעַג אִישׁ אֶת רֵעֵהוּ לְעוֹלָם וָעֶד!

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “יְלָדִים, עִצְמוּ אֶת הָעֵינַיִם לְרֶגַע, וּדְמְיְנוּ אֶת עַצְמְכֶם בְּתוֹךְ בֵּית מִדְרָשׁ קָדוּם בְּאֶרֶץ יִשְׂרָאֵל, לִפְנֵי קָרוֹב לְאַלְפַּיִם שָׁנָה.” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “אַחַר כָּךְ, הֵחֵלּוּ הַמְשָׁרְתִים לְהַגִּישׁ אֶת קְעָרוֹת הַמָּזוֹן.”?
ג. כיצד הגיב בר קפרא למצב המתואר: “זוֹ בְּרָכָה מְיֻחֶדֶת וַחֲשׁוּבָה, כִּי הִיא קוֹבַעַת בְּדִיּוּק שֶׁהַמַּאֲכָל בָּא מִן הָאֲדָמָה.”?
ד. מה השתנה בין “הַתַּלְמִיד הַשֵּׁנִי שֶׁיָּשַׁב בַּצַּד, זֶה שֶׁרָאָה אֶת הַטָּעוּת שֶׁל חֲבֵרוֹ – בִּמְקוֹם לִשְׁתֹּק, בִּמְקוֹם…” לבין “הַסְּבָרָה שֶׁל הָרַב הָיְתָה פְּשׁוּטָה: בְּסֵדֶר, אַתָּה לֹא חָכָם גָּדוֹל עֲדַיִן, יֵשׁ הֲלָכוֹת שֶׁאַתָּה אֵינְךָ…”?
ה. כיצד הגיע הסיפור אל האירוע: “וְכָךְ, מִתּוֹךְ סְעוּדָה אַחַת הִסְתַּכְּלָה יְשִׁיבָה שְׁלֵמָה וְלָמְדָה אֶת מֻשָּׂגוֹ הֶעָמֹק שֶׁל הַכָּבוֹד בַּתּוֹרָה.”?

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “כיצד לימד ברכות — המברך והמלגלג” על בר קפרא? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “הַתַּלְמִיד כָּל כָּךְ הִתְרַגֵּשׁ, הַלֵּב שֶׁלּוֹ דָּפַק בְּחָזְקָה: "בּוּם-בּוּם-בּוּם".”.
ג. אילו בר קפרא לא היה פועל כפי שמתואר ב־“הַסְּבָרָה שֶׁל הָרַב הָיְתָה פְּשׁוּטָה: בְּסֵדֶר, אַתָּה לֹא חָכָם גָּדוֹל עֲדַיִן, יֵשׁ הֲלָכוֹת שֶׁאַתָּה אֵינְךָ יוֹדֵעַ…” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “וְכָךְ, מִתּוֹךְ סְעוּדָה אַחַת הִסְתַּכְּלָה יְשִׁיבָה שְׁלֵמָה וְלָמְדָה אֶת מֻשָּׂגוֹ הֶעָמֹק שֶׁל הַכָּבוֹד בַּתּוֹרָה.” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. יְלָדִים, עִצְמוּ אֶת הָעֵינַיִם לְרֶגַע, וּדְמְיְנוּ אֶת עַצְמְכֶם…
2. אַחַר כָּךְ, הֵחֵלּוּ הַמְשָׁרְתִים לְהַגִּישׁ אֶת קְעָרוֹת הַמָּזוֹן.
3. הוּא פָּנָה לְאֶחָד מֵהֶם, וְאָמַר לוֹ בְּחִבָּה: "בְּנִי, מְבַקֵּשׁ…
4. יְלָדִים, אִם עַד לְכָאן הָיָה שֶׁקֶט, עַכְשָׁו נִהְיְתָה בַּחֶדֶר…
5. וְאַתָּה..." קוֹלוֹ שֶׁל בַּר קַפָּרָא רָעַד מִכְּאֵב, "חֲבֵרְךָ טָעָה,…
6. וּמֵאָז, דִּבְרֵי בַּר קַפָּרָא מְהַדְהֲדִים בְּאָזְנֵי עַם יִשְׂרָאֵל,…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “זוֹ בְּרָכָה מְיֻחֶדֶת וַחֲשׁוּבָה, כִּי הִיא קוֹבַעַת בְּדִיּוּק שֶׁהַמַּאֲכָל בָּא מִן הָאֲדָמָה.”.
מכתב: כתבו לבר קפרא בעקבות “הוּא לֹא הִסְכִּים שֶׁבְּבֵית הַמִּדְרָשׁ שֶׁלּוֹ יִרְמְסוּ אֶת כְּבוֹדוֹ שֶׁל אָדָם בְּשֵׁם "הַלִּמּוּד", וְהוּא לֹא הִסְכִּים…”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “וְאַתָּה..." קוֹלוֹ שֶׁל בַּר קַפָּרָא רָעַד מִכְּאֵב, "חֲבֵרְךָ טָעָה, וְלֹא יָדַע כֵּיצַד לְבָרֵךְ...”.
דיווח: כתבו ידיעה קצרה שכותרתה “כיצד לימד ברכות — המברך והמלגלג”.

המחזה

רגע מן הסיפורהדמויותמה יאמרו או יעשו?
בַּר קַפָּרָא הִתְיַשֵּׁב, שָׁתַק שְׁתִיקָה שֶׁל דְּבֵקוּת, וְהִמְתִּין.
הַסְּבָרָה שֶׁל הָרַב הָיְתָה פְּשׁוּטָה: בְּסֵדֶר, אַתָּה לֹא חָכָם גָּדוֹל עֲדַיִן, יֵשׁ הֲלָכוֹת…

7. חיבור לחיים מתוך המעשה

א. מה אפשר ללמוד מן הדרך שבה בר קפרא פעל באירוע “הַתַּלְמִיד הַשֵּׁנִי שֶׁיָּשַׁב בַּצַּד, זֶה שֶׁרָאָה אֶת הַטָּעוּת שֶׁל חֲבֵרוֹ – בִּמְקוֹם לִשְׁתֹּק, בִּמְקוֹם לָגֶשֶׁת…”?
ב. תארו מצב מחיי הכיתה שבו אפשר ליישם את המסר של “כיצד לימד ברכות — המברך והמלגלג”.
