בקעת הזהב — נסיון העושר

דף עבודה בעקבות רבי שמעון בר יוחאי (רשב"י)
שם: כיתה:

המעשה — הסיפור המלא

פְּתִיחָה: קוֹלוֹת שֶׁל תּוֹרָה וּבֶטֶן מְקַרְקֶרֶת

יְלָדִים, אֲנַחְנוּ עוֹלִים עַכְשָׁו הַרְבֵּה נְשִׁימוֹת, לְמַעְלָה לְמַעְלָה אֶל הָרֵי הַגָּלִיל, אֶל הָעִיר מֵירוֹן שֶׁל לִפְנֵי כִּמְעַט אַלְפַּיִם שָׁנָה. הָאֲוִיר שָׁם צָלוּל וְקַר, אֲבָל אִם תִּתְקָרְבוּ לְבֵית הַמִּדְרָשׁ שֶׁל הַתַּנָּא הָאֱלֹקִי – רַבִּי שִׁמְעוֹן בַּר יוֹחַאי – תַּרְגִּישׁוּ חֹם גָּדוֹל שֶׁל קְדֻשָּׁה.

בְּתוֹךְ בֵּית הַמִּדְרָשׁ יָשְׁבוּ הַתַּלְמִידִים. קוֹל שִׁירַת הַתּוֹרָה הִתְנַגֵּן שָׁם יוֹמָם וָלַיְלָה. הָיָה שָׁם רֵיחַ נָעִים שֶׁל קְלָפִים עַתִּיקִים וְסִפְרֵי תּוֹרָה. אֲבָל... הָיָה שָׁם עוֹד מַשֶּׁהוּ.

הַתַּלְמִידִים שֶׁל רַבִּי שִׁמְעוֹן הָיוּ עֲנִיִּים מְאֹד. הַבְּגָדִים שֶׁלָּהֶם הָיוּ פְּשׁוּטִים וַעֲשׂוּיִים מִפִּשְׁתָּן גַּס, הַנַּעֲלַיִם שֶׁלָּהֶם הָיוּ קְרוּעוֹת, וּפַעַם אַחַר פַּעַם, בְּאֶמְצַע לִמּוּד סֻגְיָה קָשָׁה, הָיָה אֶפְשָׁר לִשְׁמֹעַ "קְרְרְרְ..." – קוֹל שֶׁל בֶּטֶן מְקַרְקֶרֶת מֵרָעָב. לַמְרוֹת הַכֹּל, הֵם אָהֲבוּ אֶת הַתּוֹרָה וְלֹא עָזְבוּ אוֹתָהּ אֲפִלּוּ לְדַקָּה.

הָאוֹרֵחַ הֶעָשִׁיר הַמְּסַנְוֵר

יוֹם אֶחָד, הַשֶּׁקֶט בָּרְחוֹב הֻפְרַע בְּקוֹל נְקִישׁוֹת פַּרְסוֹת סוּסִים כְּבֵדוֹת: טְרַק, טַק, טְרַק.

לְחָצֵר בֵּית הַמִּדְרָשׁ נִכְנַס סוּס אַבִּיר, לָבָן וְיָפֶה. עָלָיו רָכַב בָּחוּר צָעִיר הַלָּבוּשׁ בִּבְגָדִים מַבְרִיקִים מִמֶּשִׁי, מְעוֹרְרִים הִשְׁתָּאוּת, וְרֵיחַ שֶׁל בָּשְׂמֵי מְלָכִים יָקָר נָדַף מִמֶּנּוּ לְמֵרָחוֹק. בְּיָדָיו הוּא הֶחְזִיק שַׂקִּים מְלֵאִים וּכְבֵדִים עַד שֶׁהָיוּ מְצַלְצְלִים בְּקוֹל נְקִישַׁת מַטְבְּעוֹת.

הַתַּלְמִידִים הִבִּיטוּ בַּחַלּוֹן וְהִשְׁתָּאוּ: הֲרֵי זֶה מְנַחֵם! חֲבֵרָם לַסַּפְסָל שֶׁלָּמַד אִתָּם מַמָּשׁ לֹא מִזְּמַן! מְנַחֵם יָצָא לְחוּץ לָאָרֶץ לַעֲסֹק בְּמִסְחָר, וְהִנֵּה הוּא חוֹזֵר עָשִׁיר גָּדוֹל, עִם עֲבָדִים וּמָמוֹן רַב.

"שָׁלוֹם לָכֶם חֲבֵרַי!" חִיֵּךְ מְנַחֵם וְנִפְנֵף בְּיָדוֹ הֶעֱטוּיָה בְּטַבַּעַת זָהָב מְנַצְנֶצֶת, "אֵיךְ חַיֵּיכֶם? אֲנִי בָּרוּךְ ה' עָשִׂיתִי עֲסָקִים עֲצוּמִים לְמֵעֵבֶר לַיָּם. אֲנִי עַכְשָׁו אִישׁ חָשׁוּב!".

הַקִּנְאָה וְהַנִּסָּיוֹן

בְּאוֹתוֹ רֶגַע, הַיֵּצֶר הָרַע הֵחֵל לְלַחְשֵׁשׁ בְּאָזְנֵי הַתַּלְמִידִים. יְלָדִים, גַּם לְצַדִּיקִים גְּדוֹלִים יֵשׁ נִסְיוֹנוֹת!

הַתַּלְמִידִים הִבִּיטוּ בַּבְּגָדִים הַקְּרוּעִים שֶׁלָּהֶם, וְאָז בַּמֶּשִׁי הַנּוֹצֵץ שֶׁל מְנַחֵם. הֵם הִרְגִּישׁוּ אֶת הָרָעָב בַּבֶּטֶן, וְשָׁמְעוּ אֶת צִלְצוּל הַמַּטְבְּעוֹת מֵהַסּוּס שֶׁלּוֹ. לִבָּם נֶחְמַץ. הַכְּאֵב הָיָה אֲמִתִּי.

"לָמָּה הוּא יָכוֹל לֶאֱכֹל בָּשָׂר וְדָגִים, וַאֲנַחְנוּ רָעֵבִים?" לָחַשׁ תַּלְמִיד אֶחָד שֶׁפָּנָיו הָיוּ חִוְּרוֹת.

"אוּלַי נַעֲזֹב גַּם אֲנַחְנוּ? נִסַּע לְחוּץ לָאָרֶץ לְשָׁנָה-שְׁנָתַיִם, נַעֲשֶׂה כֶּסֶף גָּדוֹל, נִקְנֶה בָּתִּים וְסוּסִים, וְרַק אָז נַחְזֹר לִלְמֹד תּוֹרָה כְּשֶׁאֲנַחְנוּ עֲשִׁירִים רְגוּעִים?" שָׁאַל תַּלְמִיד אַחֵר.

הַמִּלְמוּלִים גָּבְרוּ בְּבֵית הַמִּדְרָשׁ. הַקּוֹלוֹת יָרְדוּ, הַסְּפָרִים נִסְגְּרוּ. שְׁלָבָה שֶׁל עֲזִיבַת הַתּוֹרָה רִחֲפָה בָּאֲוִיר הַקַּר.

עֵינָיו הַבּוֹעֲרוֹת שֶׁל רַבִּי שִׁמְעוֹן

פִּתְאוֹם, הַדֶּלֶת נִפְתְּחָה. בַּפֶּתַח עָמַד רַבָּם הַגָּדוֹל, הַתַּנָּא רַבִּי שִׁמְעוֹן בַּר יוֹחַאי. עֵינָיו הָיוּ קְדוֹשׁוֹת וְעוֹטְפוֹת אֶת כֻּלָּם בְּאַהֲבָה שֶׁל רוֹעֶה נֶאֱמָן, אֲבָל הָיְתָה בָּהֶן רְאִיָּה שֶׁחָודֶרֶת יָשָׁר אֶל תּוֹךְ הַנְּשָׁמָה.

רוּחַ הַקֹּדֶשׁ נָצְצָה בְּרַבִּי שִׁמְעוֹן, וְהוּא יָדַע מִיָּד מָה עוֹבֵר בְּלִבָּם שֶׁל תַּלְמִידָיו הַיְּקָרִים. הוּא יָדַע מַדּוּעַ הֵם עֲצוּבִים.

רַבִּי שִׁמְעוֹן לֹא צָעַק, וְלֹא כָּעַס עֲלֵיהֶם. הוּא יָדַע שֶׁקָּשֶׁה לָהֶם.

הוּא רַק סִמֵּן לָהֶם בְּיָדוֹ וְאָמַר בְּקוֹל עָמֹק וְשָׁקֵט: "תַּלְמִידַי בָּנַי, קוּמוּ מִמְּקוֹמְכֶם. קְחוּ אֶת סִפְרֵיכֶם, וּבֹאוּ אַחֲרַי".

הַתַּלְמִידִים קָמוּ בִּשְׁתִיקָה. הֵם הָלְכוּ אַחֲרָיו. רַבִּי שִׁמְעוֹן הוֹצִיא אוֹתָם מִבֵּית הַמִּדְרָשׁ, וְהוֹבִיל אוֹתָם דֶּרֶךְ שְׁבִילֵי עָפָר וֶהָרִים, עַד שֶׁהִגִּיעוּ לְבִקְעָה אַחַת גְּדוֹלָה (עֵמֶק רָחָב שֶׁבֵּין הֶהָרִים) בְּסָמוּךְ לְעִירוֹ.

זֶה הָיָה מָקוֹם צִיח וְיָבֵשׁ. שֶׁמֶשׁ יְשִׁירָה הִכְּתָה עַל חֲלוּקֵי הָאֲבָנִים שֶׁעַל הָאֲדָמָה, וְרוּחַ קַלָּה הִפְרִיחָה עַלְעֲלֵי שַׁלֶּכֶת יְבֵשִׁים שֶׁהִשְׁמִיעוּ קוֹל פִּצְפּוּץ.

הַנֵּס הַגָּדוֹל שֶׁל בִּקְעַת הַזָּהָב

רַבִּי שִׁמְעוֹן נֶעֱמַד בְּמֶרְכַּז הַבִּקְעָה. הַתַּלְמִידִים הִתְקַבְּצוּ סְבִיבוֹ, מוּפְתָּעִים, מַמְתִּינִים. הַהַשְׁגָּחָה הָעֶלְיוֹנָה רִחֲפָה שָׁם כְּמוֹ עֲנַן כָּבוֹד.

הַצַּדִּיק הֵרִים אֶת עֵינָיו לַשָּׁמַיִם הַכְּחֻלִּים, פָּרַשׂ אֶת יָדָיו, וּפָנָה לַבִּקְעָה. קוֹלוֹ רָעַם כְּאֲרִי אֲבָל מָלֵא בְּאַהֲבַת ה':

"בִּקְעָה, בִּקְעָה! הִמָּלְאִי דִּינְרֵי זָהָב! הִמָּלְאִי דִּינְרֵי זָהָב לְמַעַן תַּלְמִידַי הָאֲהוּבִים!"

וְאָז – הַבִּלְתִּי יֵאָמֵן קָרָה.

פִּתְאוֹם, אוֹר זָהָב מְסַנְוֵר הִתְפָּרֵץ מִן הָאֲדָמָה! אֲבָנִים וְרִגְבֵי עָפָר שִׁנּוּ אֶת צוּרָתָם בִּשְׁנִיָּה אַחַת. קוֹל פִּצְפּוּץ עָדִין שֶׁל מַתֶּכֶת הִשְׁמִיעַ צְלִיל עָרֵב שֶׁל כֶּסֶף נוֹפֵל. צִלְנְג, צְלִינְג, צְלִינְג!

בְּתוֹךְ רְגָעִים בּוֹדְדִים, הַבִּקְעָה הַיְּבֵשָׁה הָפְכָה לְעֵמֶק מְלֵא דִּינְרֵי זָהָב בּוֹהֲקִים, נְקִיִּים מִכָּל פְּסֹלֶת, שֶׁעָלוּ וְנֶעֶרְמוּ לַהֲרָרִים מַבְרִיקִים! אוֹרוֹת הַזָּהָב נִצְנְצוּ מוּל קַרְנֵי הַשֶּׁמֶשׁ, עַד שֶׁהַתַּלְמִידִים נֶאֶלְצוּ לְהַסְתִּיר אֶת עֵינֵיהֶם מֵרֹב בְּרַק הָעֹשֶׁר. זֶה הָיָה מַזְגָן עָצוּם, עֹשֶׁר שֶׁאֲפִלּוּ קֵיסָר רוֹמִי לֹא חָלַם עָלָיו מֵעוֹלָם!

הַתַּלְמִידִים נֶעֶתְקוּ מִמְּקוֹמָם. לִבָּם פָּעַם בְּחָזְקָה. הֵם הִבִּיטוּ בָּעוֹצְמָה הָעֲצוּמָה שֶׁל רַבָּם שֶׁשּׁוֹלֵט בַּטֶּבַע בִּזְכוּת תּוֹרָתוֹ. אִם כָּל כָּךְ הַרְבֵּה זָהָב כָּאן, לָמָּה שֶׁלֹּא יִקְחוּ?

הַהַבְטָחָה לָעוֹלָם הַבָּא

רַבִּי שִׁמְעוֹן סוֹבֵב אֶת רֹאשׁוֹ וְהִבִּיט בָּהֶם. קוֹלוֹ כָּעֵת הָיָה רַךְ וְעָמֹק, קוֹל שֶׁאוֹמֵר סוֹד עֶלְיוֹן:

"בָּנַי, רְאִיתֶם אֶת הַזָּהָב הַזֶּה? הַקָּדוֹשׁ בָּרוּךְ הוּא יָכוֹל לָתֵת לָנוּ אֶת כָּל הָעֹשֶׁר שֶׁבָּעוֹלָם בִּרְגַע קָט עַל שֶׁאֲנַחְנוּ עוֹסְקִים בַּתּוֹרָה... וְאִם רְצוֹנְכֶם בְּזָהָב – הִנֵּה הוּא לְפָנֵיכֶם! שְׁלְחוּ יְדֵיכֶם וְקַחוּ לָכֶם, מַלְאוּ אֶת שַׂקֵּיכֶם, תִּקְנוּ לָכֶם בָּתִּים וְסוּסִים כְּמוֹ שֶׁל מְנַחֵם...

אֲבָל! – " רַבִּי שִׁמְעוֹן עָצַר, הֵרִים אֶצְבַּע אֶחָת לַשָּׁמַיִם וְהִנְמִיךְ אֶת קוֹלוֹ – " דְּעוּ לָכֶם: כָּל מִי שֶׁנּוֹטֵל עַכְשָׁו אֶת הַזָּהָב הַזֶּה בָּעוֹלָם הַזֶּה, נוֹטֵל אֶת חֶלְקוֹ מֵהָעוֹלָם הַבָּא! בָּעוֹלָם הַזֶּה תִּהְיוּ עֲשִׁירִים מְשָׁרְתִים שֶׁל זָהָב, אֲבָל בַּשָּׁמַיִם יֶחְסַר לָכֶם מִכֶּתֶר הַתּוֹרָה".

יְלָדִים, שְׁתִיקָה עֲמֻקָּה יָרְדָה עַל הַבִּקְעָה. עַכְשָׁו, כְּשֶׁהַזָּהָב הָיָה מַמָּשׁ מוּל עֵינֵיהֶם, מַבְרִיק וּמוּכָן לִלְקִיחָה – הַתַּלְמִידִים הֵבִינוּ שֶׁהוּא שָׁוֶה כְּקִלּוּף שׁוּם לְעֻמַּת הַתּוֹרָה הַקְּדוֹשָׁה.

אִישׁ מֵהֶם לֹא זָז. אִישׁ לֹא פָּשַׁט אֶת יָדוֹ. הֵם הִבִּיטוּ זֶה בָּזֶה, אָז בְּרַבִּי שִׁמְעוֹן, וְעֵינֵיהֶם דָּמְעוּ מִשִּׂמְחָה. פִּתְאוֹם, הַבְּגָדִים הַבְּלוּיִים שֶׁלָּהֶם נִרְאוּ לָהֶם כְּבִגְדֵי מַלְכוּת, וְהַבֶּטֶן הָרְעֵבָה שֶׁלָּהֶם הִרְגִּישָׁה שְׂבֵעָה מִנֹּעַם צִחְצוּחַ הַקְּדֻשָּׁה.

הֵם עָשׂוּ אַחֲוֹרָה פְּנֶה, נָטְשׁוּ אֶת הֲרָרֵי הַזָּהָב מֵאֲחוֹרֵיהֶם, וְרָצוּ בְּשִׂמְחָה עֲצוּמָה וּבְשִׁירָה חֲזָרָה לְבֵית הַמִּדְרָשׁ! הֵם הֵבִינוּ – הָעֹשֶׁר הָאֲמִתִּי הוּא מָה שֶׁיֵּשׁ לָנוּ בַּשֵּׂכֶל וּבַלֵּב מִלִּמּוּד הַתּוֹרָה, וְאַף אֶחָד לֹא יִקַּח לָהֶם אֶת הָעוֹלָם הַבָּא בִּשְׁבִיל חֲתִיכַת מַתֶּכֶת מְנַצְנֶצֶת!

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “יְלָדִים, אֲנַחְנוּ עוֹלִים עַכְשָׁו הַרְבֵּה נְשִׁימוֹת, לְמַעְלָה לְמַעְלָה אֶל הָרֵי הַגָּלִיל, אֶל הָעִיר מֵירוֹן שֶׁל לִפְנֵי כִּמְעַט אַלְפַּיִם…” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “עָלָיו רָכַב בָּחוּר צָעִיר הַלָּבוּשׁ בִּבְגָדִים מַבְרִיקִים מִמֶּשִׁי, מְעוֹרְרִים הִשְׁתָּאוּת, וְרֵיחַ שֶׁל בָּשְׂמֵי מְלָכִים יָקָר נָדַף מִמֶּנּוּ…”?
ג. כיצד הגיב רבי שמעון בר יוחאי (רשב"י) למצב המתואר: “בַּפֶּתַח עָמַד רַבָּם הַגָּדוֹל, הַתַּנָּא רַבִּי שִׁמְעוֹן בַּר יוֹחַאי.”?
ד. מה השתנה בין “הַצַּדִּיק הֵרִים אֶת עֵינָיו לַשָּׁמַיִם הַכְּחֻלִּים, פָּרַשׂ אֶת יָדָיו, וּפָנָה לַבִּקְעָה.” לבין “הֵם הִבִּיטוּ בָּעוֹצְמָה הָעֲצוּמָה שֶׁל רַבָּם שֶׁשּׁוֹלֵט בַּטֶּבַע בִּזְכוּת תּוֹרָתוֹ.”?
ה. כיצד הגיע הסיפור אל האירוע: “הֵם הִבִּיטוּ זֶה בָּזֶה, אָז בְּרַבִּי שִׁמְעוֹן, וְעֵינֵיהֶם דָּמְעוּ מִשִּׂמְחָה.”?

3. מה היה ראשון?

כתבו ליד כל אירוע מספר מ־1 עד 5 לפי סדר התרחשותו בסיפור.

□ שֶׁמֶשׁ יְשִׁירָה הִכְּתָה עַל חֲלוּקֵי הָאֲבָנִים שֶׁעַל הָאֲדָמָה, וְרוּחַ קַלָּה הִפְרִיחָה עַלְעֲלֵי שַׁלֶּכֶת יְבֵשִׁים שֶׁהִשְׁמִיעוּ קוֹל…
□ יְלָדִים, אֲנַחְנוּ עוֹלִים עַכְשָׁו הַרְבֵּה נְשִׁימוֹת, לְמַעְלָה לְמַעְלָה אֶל הָרֵי הַגָּלִיל, אֶל הָעִיר מֵירוֹן שֶׁל לִפְנֵי כִּמְעַט אַלְפַּיִם…
□ הֵם הֵבִינוּ – הָעֹשֶׁר הָאֲמִתִּי הוּא מָה שֶׁיֵּשׁ לָנוּ בַּשֵּׂכֶל וּבַלֵּב מִלִּמּוּד הַתּוֹרָה, וְאַף אֶחָד לֹא יִקַּח לָהֶם אֶת הָעוֹלָם הַבָּא…
□ אֲנִי בָּרוּךְ ה' עָשִׂיתִי עֲסָקִים עֲצוּמִים לְמֵעֵבֶר לַיָּם.
□ קוֹלוֹ כָּעֵת הָיָה רַךְ וְעָמֹק, קוֹל שֶׁאוֹמֵר סוֹד עֶלְיוֹן:

לפני ואחרי

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “בקעת הזהב — נסיון העושר” על רבי שמעון בר יוחאי (רשב"י)? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “הוּא רַק סִמֵּן לָהֶם בְּיָדוֹ וְאָמַר בְּקוֹל עָמֹק וְשָׁקֵט: "תַּלְמִידַי בָּנַי, קוּמוּ מִמְּקוֹמְכֶם.”.
ג. אילו רבי שמעון בר יוחאי (רשב"י) לא היה פועל כפי שמתואר ב־“הֵם הִבִּיטוּ בָּעוֹצְמָה הָעֲצוּמָה שֶׁל רַבָּם שֶׁשּׁוֹלֵט בַּטֶּבַע בִּזְכוּת תּוֹרָתוֹ.” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “הֵם הִבִּיטוּ זֶה בָּזֶה, אָז בְּרַבִּי שִׁמְעוֹן, וְעֵינֵיהֶם דָּמְעוּ מִשִּׂמְחָה.” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. יְלָדִים, אֲנַחְנוּ עוֹלִים עַכְשָׁו הַרְבֵּה נְשִׁימוֹת, לְמַעְלָה…
2. בְּיָדָיו הוּא הֶחְזִיק שַׂקִּים מְלֵאִים וּכְבֵדִים עַד שֶׁהָיוּ…
3. עֵינָיו הָיוּ קְדוֹשׁוֹת וְעוֹטְפוֹת אֶת כֻּלָּם בְּאַהֲבָה שֶׁל רוֹעֶה…
4. קוֹל פִּצְפּוּץ עָדִין שֶׁל מַתֶּכֶת הִשְׁמִיעַ צְלִיל עָרֵב שֶׁל…
5. שְׁלְחוּ יְדֵיכֶם וְקַחוּ לָכֶם, מַלְאוּ אֶת שַׂקֵּיכֶם, תִּקְנוּ לָכֶם…
6. הֵם הֵבִינוּ – הָעֹשֶׁר הָאֲמִתִּי הוּא מָה שֶׁיֵּשׁ לָנוּ בַּשֵּׂכֶל…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “בַּפֶּתַח עָמַד רַבָּם הַגָּדוֹל, הַתַּנָּא רַבִּי שִׁמְעוֹן בַּר יוֹחַאי.”.
מכתב: כתבו לרבי שמעון בר יוחאי (רשב"י) בעקבות “בְּתוֹךְ רְגָעִים בּוֹדְדִים, הַבִּקְעָה הַיְּבֵשָׁה הָפְכָה לְעֵמֶק מְלֵא דִּינְרֵי זָהָב בּוֹהֲקִים, נְקִיִּים מִכָּל פְּסֹלֶת,…”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “– " רַבִּי שִׁמְעוֹן עָצַר, הֵרִים אֶצְבַּע אֶחָת לַשָּׁמַיִם וְהִנְמִיךְ אֶת קוֹלוֹ – " דְּעוּ לָכֶם: כָּל מִי שֶׁנּוֹטֵל…”.
דיווח: כתבו ידיעה קצרה שכותרתה “בקעת הזהב — נסיון העושר”.

המחזה

רגע מן הסיפורהדמויותמה יאמרו או יעשו?
בְּאוֹתוֹ רֶגַע, הַיֵּצֶר הָרַע הֵחֵל לְלַחְשֵׁשׁ בְּאָזְנֵי הַתַּלְמִידִים.
הֵם הִבִּיטוּ בָּעוֹצְמָה הָעֲצוּמָה שֶׁל רַבָּם שֶׁשּׁוֹלֵט בַּטֶּבַע בִּזְכוּת תּוֹרָתוֹ.
