🔹 שָׁלָב 1: בֵּרוּר מַקְדִּים וְנִתּוּחַ (לַמּוֹרֶה/לַמְסַפֵּר)

🔹 שָׁלָב 2: הַסִּפּוּר הַמָּלֵא – "הַסּוֹד בַּעֲלִיַּת הַגַּג"

פְּתִיחָה: הֶחָכָם שֶׁהִקְטִין אֶת עַצְמוֹ יְלָדִים, אֵיךְ לְדַעְתְּכֶם צָרִיךְ לִקְרֹא לְרַב עָנָק, צַדִּיק שֶׁיּוֹדֵעַ אֶת כָּל הַתּוֹרָה כֻּלָּהּ? בְּדֶרֶךְ כְּלָל קוֹרְאִים לוֹ "הַגָּדוֹל", נָכוֹן? הַיּוֹם נַכִּיר תַּנָּא מְיֻחָד מְאֹד בְּדוֹרוֹ שֶׁל רַבָּן גַּמְלִיאֵל. תַּנָּא כָּל כָּךְ עָצוּם, שֶׁפַּעַם אַחַת, כְּשֶׁיָּשְׁבוּ חֲכָמִים בָּעִיר יְרִיחוֹ, יָצְאָה בַּת קוֹל (קוֹל מִן הַשָּׁמַיִם) וְאָמְרָה: "יֵשׁ כָּאן אָדָם אֶחָד שֶׁרָאוּי שֶׁתִּשְׁרֶה עָלָיו שְׁכִינָה כְּמוֹ נָבִיא!". כֻּלָּם הִסְתַּכְּלוּ יָשָׁר עָלָיו. אַךְ אַף עַל פִּי שֶׁהָיָה עֲנָק – קָרְאוּ לוֹ חֲכָמִים: שְׁמוּאֵל הַקָּטָן.

מַדּוּעַ 'הַקָּטָן'? כִּי עַל אַף גְּדֻלָּתוֹ – הוּא תָּמִיד הִקְטִין אֶת עַצְמוֹ. הוּא בָּרַח מִן הַכָּבוֹד, עָזַר לַפְּשׁוּטִים, וְלְעוֹלָם לֹא הִתְגָּאָה. אֲבָל כַּמָּה הוּא הָיָה מוּכָן "לְהַקְטִין" אֶת עַצְמוֹ? אֶת זֶה אֲנַחְנוּ עֲתִידִים לְגַלּוֹת בְּבוֹקֶר קַר אֶחָד שֶׁל חֹרֶף.

הַסּוֹד שֶׁל עִבּוּר הַשָּׁנָה בְּאוֹתָם יָמִים, רַבָּן גַּמְלִיאֵל הַזָּקֵן, נְשִׂיא יִשְׂרָאֵל, הָיָה צָרִיךְ לְקַבֵּל הַחְלָטָה גּוֹרָלִית: לְעַבֵּר אֶת הַשָּׁנָה. (לְהוֹסִיף עוֹד חֹדֶשׁ אֲדָר, כְּדֵי שֶׁחַג הַפֶּסַח יִפֹּל בְּדִיּוּק בְּחֹדֶשׁ הָאָבִיב, כְּפִי שֶׁצִּוְּתָה הַתּוֹרָה). הַהַחְלָטָה הַזּוֹ דָּרְשָׁה קְדֻשָּׁה גְּדוֹלָה וְסוֹדִיּוֹת מֻחְלֶטֶת. לֹא כָּל אֶחָד יָכוֹל לְהִשְׁתַּתֵּף בִּישִׁיבָה כָּזוֹ. רַק הַחֲכָמִים שֶׁהַנָּשִׂיא זִמֵּן בִּמְיֻחָד בְּאוֹתוֹ בֹּקֶר, רַשָּׁאִים הָיוּ לְהִכָּנֵס.

לִפְנֵי שֶׁשָּׁקְעָה הַשֶּׁמֶשׁ, אָמַר רַבָּן גַּמְלִיאֵל لַשַּׁמָּשׁ שֶׁלּוֹ: "הַשְׁכִּימוּ לִי שִׁבְעָה תַּלְמִידֵי חֲכָמִים נִבְחָרִים לַעֲלִיַּת הַגַּג לְמָחָר בַּבֹּקֶר. רַק שִׁבְעָה!". וּשְׁמוּאֵל הַקָּטָן, כַּמּוּבָן, הָיָה בֵּין הַמֻּזְמָנִים שֶׁקִּבְּלוּ רְשׁוּת.

הָאִישׁ הַשְּׁמִינִי! הַבֹּקֶר הֵאִיר. קוֹר יְרוּשַׁלְמִי עָטַף אֶת בֵּית הַדִּין. הַתַּלְמִידִים הַנִּבְחָרִים טִפְּסוּ אֶחָד-אֶחָד עַל מַדְרֵגוֹת הָעֵץ הַחוֹרְקוֹת אֶל עֲלִיַּת הַגַּג. דֶּלֶת הָעֵץ הַכְּבֵדָה נִסְגְּרָה, וְהַשֶּׁקֶט בַּחֶדֶר הָיָה כָּזֶה, שֶׁאֶפְשָׁר הָיָה לִשְׁמֹעַ רִשְׁרוּשׁ שֶׁל דַּף. רַבָּן גַּמְלִיאֵל יָשַׁב בָּרֹאשׁ. הוּא עָבַר בְּמַבָּטוֹ עַל פְּנֵי הַיּוֹשְׁבִים, הֵזִיז אֶת שְׂפָתָיו בְּלַחַשׁ וְסָפַר אוֹתָם.

אֶחָד... שְׁנַיִם... אַרְבָּעָה... שִׁבְעָה... שְׁמוֹנָה! לְפֶתַע, עֵינָיו שֶׁל נְשִׂיא יִשְׂרָאֵל נִפְעֲרוּ. שְׁמוֹנָה חֲכָמִים מִסָּבִיב לַשֻּׁלְחָן! הוּא הֲרֵי זִמֵּן רַק שִׁבְעָה! הַמַּשְׁמָעוּת הָיְתָה בְּרוּרָה: אֶחָד מֵהָאֲנָשִׁים בַּחֶדֶר נִכְנַס סְתָם כָּךְ, לְלֹא רְשׁוּת, אוּלַי כִּי רָצָה לְהִתְגָּאוֹת אוֹ לְהִדָּחֵף לִמְקוֹם הַגְּדוֹלִים.

רַבָּן גַּמְלִיאֵל לֹא רָצָה לְהַעֲבִיר עַל כָּךְ לְסֵדֶר הַיּוֹם. תּוֹרָה צְרִיכָה סֵדֶר! קוֹלוֹ רָעַד בְּתֹקֶף צַדִּיקִים כַּאֲשֶׁר דָּפַק בְּעוֹז עַל הַשֻּׁלְחָן וְהִכְרִיז לְכָל עֲלִיַּת הַגַּג: "מִי הוּא שֶׁעָלָה שֶׁלֹּא בִּרְשׁוּת? יֵרֵד!!".

דְּפִיקוֹת הַלֵּב וְהַבּוּשָׁה הָאֲיֻמָּה דְּמָמָה מֻחְלֶטֶת. לֹא נִשְׁמְעָה נְשִׁימָה. אִישׁ מִן הַחֲכָמִים לֹא קָם. יְלָדִים, נַסּוּ לְתָאֵר לְעַצְמְכֶם מָה הִרְגִּישׁ בְּאוֹתָם רְגָעִים אוֹתוֹ חָכָם מִסְכֵּן שֶׁנִּכְנַס שֶׁלֹּא כַּדִּין. הַלֵּב שֶׁלּוֹ בֶּטַח דָּפַק בְּחָזְקָה... בּוּם! בּוּם! בּוּם! פָּנָיו הֶאֱדִימוּ כְּמוֹ אֵשׁ, עֵינָיו בָּעֲרוּ. עוֹד רֶגַע קָט הוּא יִצְטָרֵךְ לָקוּם מִכִּסְאוֹ, לַעֲבֹר בְּרֹאשׁ מֻרְכָּן לִפְנֵי כָּל גְּדוֹלֵי הַדּוֹר הַמַּבִּיטִים בּוֹ, לָגֶשֶׁת לַדֶּלֶת כְּאָדָם שֶׁגּוֹרַשׁ בְּבִזָּיוֹן, וְלָרֶדֶת מֵהַמַּדְרֵגוֹת! אֵיזוֹ בּוּשָׁה!! שֶׁנִּגְזְרוּ שָׁמַיִם וָאָרֶץ עַל הָאָדָם! לְהִשָּׂרֵף בַּחַיִּים מֵרֹב עֶלְבּוֹן!

וְרַבָּן גַּמְלִיאֵל מַמְתִּין. עֵינָיו דּוֹרְשׁוֹת. הַשְּׁנִיּוֹת נִמְתָּחוֹת כְּמוֹ נֶצַח. מִי יָקוּם?

"אֲנִי הוּא הָאָשֵׁם!" וְאָז, נִשְׁמַע רַחַשׁ שֶׁל כִּסֵּא שֶׁנִּדְחַף לְאָחוֹר. אָדָם אֶחָד קָם עַל רַגְלָיו. הָיָה זֶה הֶחָכָם הֶחָשׁוּב, תַּלְמִידוֹ שֶׁל הִלֵּל הַזָּקֵן, הָאִישׁ שֶׁבַּת קוֹל מִן הַשָּׁמַיִם הֵעִידָה עָלָיו – שְׁמוּאֵל הַקָּטָן.

שְׁמוּאֵל הַקָּטָן הִבִּיט בְּנָשִׂיא יִשְׂרָאֵל, הִשְׁפִּיל אֶת עֵינָיו מִתּוֹךְ בּוּשָׁה, וְאָמַר בְּקוֹל רוֹעֵד: "אֲנִי הוּא שֶׁעָלִיתִי שֶׁלֹּא בִּרְשׁוּת, כְּבוֹד הַנָּשִׂיא! וְלֹא מִפְּנֵי שֶׁרָצִיתִי לְהִתְגָּאוֹת וּלְעַבֵּר אֶת הַשָּׁנָה עָלִיתִי... אֶלָּא רָצִיתִי מְאֹד לִלְמֹד אֵיךְ נִפְסֶקֶת תּוֹרָה, לִלְמֹד הֲלָכָה לְמַעֲשֶׂה הוּצְרַכְתִּי. סְלַח לִי".

כְּשֶׁרַבָּן גַּמְלִיאֵל שָׁמַע שֶׁהָאָדָם הַמִּתְנַצֵּל שֶׁנִּכְנַס בְּלִי רְשׁוּת הוּא צַדִּיק טָהוֹר זֶה, מִיָּד נִרְגַּע כַּעֲסוֹ. חִיּוּךְ שֶׁל אַהֲבָה עָלָה עַל פָּנָיו, וְהוּא פָּנָה אֵלָיו בְּרַכּוּת: "שֵׁב, בְּנִי, שֵׁב! כָּל הַשָּׁנִים כֻּלָּן רְאוּיוֹת לְהִתְעַבֵּר עַל יָדְךָ בִּזְכוּת צִדְקָתְךָ וַחֲכָמָתְךָ הָעֲצוּמָה! אֶלָּא שֶׁחֲכָמִים אָמְרוּ כְּלָל שֶׁאֵין לַעֲשׂוֹת זֹאת אֶלָּא בְּמִי שֶׁהֻזְמַן מֵרֹאשׁ. שֵׁב יַקִּירִי!".

הַהֲלָכָה נִלְמְדָה. יְשִׁיבַת הַסַּנְהֶדְרִין עָבְרָה בְּשָׁלוֹם, וְחַג הַפֶּסַח נִקְבַּע בְּמוֹעֲדוֹ.

הַסּוֹד מִתְגַּלֶּה! אֲבָל חֲכָמֵינוּ זִכְרוֹנָם לִבְרָכָה, שֶׁכָּתְבוּ אֶת הַתַּלְמוּד, מְגַלִּים לָנוּ סוֹד מַדְהִים! הַאִם אַתֶּם הוֹלְכִים לְהַאֲמִין? לֹא שְׁמוּאֵל הַקָּטָן הָיָה זֶה שֶׁעָלָה שֶׁלֹּא בִּרְשׁוּת! שְׁמוּאֵל הַקָּטָן קִבֵּל הַזְמָנָה אִישִׁית מֵרַבָּן גַּמְלִיאֵל מֵרֹאשׁ! מִישֶׁהוּ אַחֵר, אָדָם שֵׁנִי, פָּשׁוּט נִדְחַף אִתָּם.

אָז לָמָּה שְׁמוּאֵל עָשָׂה עַצְמוֹ אָשֵׁם?! לָמָּה הוּא קָם וְשָׂם עַל עַצְמוֹ כֶּתֶם שֶׁל מִי שֶׁנִּכְנָס בְּלִי רְשׁוּת?! וְכָאן הַדְּמָעוֹת עוֹלוֹת לָעֵינַיִם מֵרֹב יֹפִי וּקְדֻשָּׁה: כְּשֶׁשָּׁמַע שְׁמוּאֵל אֶת רַבָּן גַּמְלִיאֵל כּוֹעֵס, הוּא הִבִּיט יָמִינָה וּשְׂמֹאלָה וְרָאָה אֶת הָאָשֵׁם הָאֲמִתִּי יְשׁוּב בַּפִּנָּה, חִוֵּר וְרוֹעֵד מֵרֹב אֵימָה. הוּא הֵבִין שֶׁאוֹתוֹ אָדָם עָתִיד לַחֲווֹת "מוות מֵרֹב בּוּשָׁה"! שְׁמוּאֵל חָשַׁב: אֲנִי – יֵשׁ לִי שֵׁם טוֹב. רַבָּן גַּמְלִיאֵל אוֹהֵב אוֹתִי וְיוֹדֵעַ אֶת צִדְקָתִי. אֲנִי אֶקַּח אֶת הַבּוּשָׁה עָלַי! אֲנִי אַעֲשֶׂה אֶת עַצְמִי הָאָשֵׁם, הָעִקָּר, הָעִקָּר, הָעִקָּר – שֶׁיְּהוּדִי שֵׁנִי לֹא יִתְבַּיֵּשׁ!!

זֶהוּ שְׁמוּאֵל הַקָּטָן. הַקָּטָן בְּעֵינֵי עַצְמוֹ, שֶׁהָיָה מוּכָן לִמְחֹק אֶת תְּהִלָּתוֹ, לָרֶדֶת לְרֶגַע מִגְּדֻלָּתוֹ, וּבִלְבַד שֶׁחֲבֵרוֹ יִשָּׁאֵר מְכֻבָּד בִּמְקוֹמוֹ.
