אבנים שחקו המים — הלימוד עם בנו

דף עבודה בעקבות רבי עקיבא
שם: כיתה:

המעשה — הסיפור המלא

פְּתִיחָה: הָרוֹעֶה הַשּׁוֹתֵק

יְלָדִים, עִצְמוּ אֶת הָעֵינַיִם וְנַסּוּ לְהָרִיחַ אֶת רֵיחַ הָאֲדָמָה הַיְּבֵשָׁה וְאֶת רֵיחַ צֶמֶר הַכְּבָשִׂים. שִׁמְעוּ אֶת הַפַּעֲמוֹנִים הַקְּטַנִּים: דִּינְג, דּוֹנְג, דִּינְג .

עַל גִּבְעָה תַּחַת שֶׁמֶשׁ חַמָּה יָשַׁב רוֹעֵה צֹאן. קָרְאוּ לוֹ עֲקִיבָא.

עֲקִיבָא לֹא הָיָה יֶלֶד. הוּא הָיָה גֶּבֶר גָּדוֹל, בֶּן אַרְבָּעִים שָׁנָה! יָדָיו הָיוּ חֲזָקוֹת מֵעֲבוֹדָה קָשָׁה, פָּנָיו הָיוּ שְׁזוּפוֹת מֵהַשֶּׁמֶשׁ, אֲבָל בְּתוֹךְ לִבּוֹ הוּא הָיָה עָצוּב וְרֵיק. עֲקִיבָא לֹא יָדַע לִקְרֹא. הוּא לֹא יָדַע מָה כָּתוּב בַּתּוֹרָה. אֲפִלּוּ אֶת הָאוֹת אָלֶ"ף הוּא לֹא הִכִּיר! כְּשֶׁרָאָה תַּלְמִידֵי חֲכָמִים עוֹבְרִים בַּדֶּרֶךְ וּמְדַבְּרִים בְּדִבְרֵי קְדֻשָּׁה, הוּא הָיָה מַשְׁפִּיל אֶת מַבָּטוֹ לָאֲדָמָה בְּבוּשָׁה.

רָחֵל, אִשְׁתּוֹ הַצַּדֶּקֶת, בִּתּוֹ שֶׁל הֶעָשִׁיר 'כַּלְבָּא שָׂבוּעַ', עָזְבָה אֶת אַרְמוֹן אָבִיהָ כְּדֵי לְהִתְחַתֵּן אִתּוֹ. הֵם חָיוּ בְּעֹנִי נוֹרָא וְיָשְׁנוּ עַל קַשׁ וָתֶבֶן. רָחֵל תָּמִיד הָיְתָה אוֹמֶרֶת לוֹ בְּקוֹל רַךְ וְאוֹהֵב: "עֲקִיבָא שֶׁלִּי, יֵשׁ לְךָ מִדּוֹת כָּל כָּךְ טוֹבוֹת. אִם רַק תֵּלֵךְ לִלְמֹד תּוֹרָה, אַתָּה תִּהְיֶה רַב גָּדוֹל בְּיִשְׂרָאֵל!".

אֲבָל עֲקִיבָא הָיָה אוֹחֵז בְּרֹאשׁוֹ וְנֶאֱנָח: "רָחֵל, זֶה מְאֻחָר מִדַּי! אֲנִי בֶּן אַרְבָּעִים! הָרֹאשׁ שֶׁלִּי קָשֶׁה כְּמוֹ בַּרְזֶל. הַשֵּׂכֶל שֶׁלִּי אָטוּם. אֵיךְ אֵשֵׁב עַכְשָׁו עִם תִּינוֹקוֹת בְּנֵי שָׁלוֹשׁ וְאַרְבַּע וְאֶלְמַד אָלֶף־בֵּית? כֻּלָּם יִצְחֲקוּ עָלַי!".

הַסּוֹד שֶׁל הַבְּאֵר

יוֹם אֶחָד, הַשֶּׁמֶשׁ קָפְחָה בְּחָזְקָה. עֲקִיבָא הָיָה צָמֵא מְאֹד. תּוֹךְ כְּדֵי הֲלִיכָה, הוּא הִגִּיעַ אֶל בְּאֵר מַיִם מְקוֹמִית. הוּא הִתְכּוֹפֵף כְּדֵי לִשְׁאֹב מַיִם צְנוּנִים, וּלְפֶתַע, עֵינָיו צָדוּ מַשֶּׁהוּ מוּזָר.

הוּא הִבִּיט עַל סֶלַע הָאֶבֶן הֶעָנָק וְהַקָּשֶׁה שֶׁנִּצַּב תַּחַת צִנּוֹר הַמַּיִם.

יְלָדִים, דַּמְיְנוּ אֶבֶן שֶׁלֹּא פַּטִּישׁ יָכוֹל לִשְׁבֹּר אוֹתָהּ. וְהִנֵּה, בְּתוֹךְ הָאֶבֶן הַזּוֹ הָיָה חוֹר עָמֹק, מְשֻׁפְשָׁף וְחָלָק גַּלִּים-גַּלִּים. עֲקִיבָא הֶעֱבִיר אֶת אֶצְבְּעוֹתָיו הַמְחֻסְפָּסוֹת בְּתוֹךְ הַחוֹר הֶחָלָק וְהִתְפַּלֵּא. הוּא הֵרִים מַבָּטוֹ וְרָאָה עֲרָבִי אֶחָד שֶׁשָּׁאַב שָׁם מַיִם.

"סְלִיחָה," שָׁאַל עֲקִיבָא בִּפְלִיאָה, "מִי חָקַק אֶת הָאֶבֶן הַזּוֹ? מִי עָשָׂה בָּהּ אֶת הַחוֹר הֶעָמֹק הַזֶּה?".

הָאִישׁ צָחַק. "עֲקִיבָא! אִי אַתָּה יוֹדֵעַ? תַּקְשִׁיב... טִיף, טַף... טִיף, טַף... "

עֲקִיבָא הִקְשִׁיב.

"הַמַּיִם, עֲקִיבָא! תִּסְתַּכֵּל עַל הַטִּפּוֹת! טִפָּה אַחַר טִפָּה, שֶׁנּוֹפְלוֹת תָּדִיר. הַמַּיִם הֵם שֶׁשָּׁחֲקוּ אֶת הָאֶבֶן הַקָּשָׁה!".

הַקָּדוֹשׁ בָּרוּךְ הוּא פָּתַח אֶת לִבּוֹ שֶׁל עֲקִיבָא בְּאוֹתוֹ הָרֶגַע מַמָּשׁ. כְּאִלּוּ בָּרָק הִכָּה מֵהַשָּׁמַיִם! עֲקִיבָא עָמַד שָׁם, נָשַׁם נְשִׁימָה עֲמֻקָּה, וְדָן בְּעַצְמוֹ "קַל וָחֹמֶר" (חִשּׁוּב שֶׁל הִגָּיוֹן טָהוֹר):

"מָה שֶׁרַךְ – הַמַּיִם, הִצְלִיחוּ לַחְפֹּר בְּתוֹךְ מָה שֶׁקָּשֶׁה מְאֹד – הָאֶבֶן... תּוֹרַת יִשְׂרָאֵל, שֶׁהִיא חֲזָקָה וּקְשָׁה כְּבַרְזֶל, הַאִם הִיא לֹא תוּכַל לַחְפֹּר עָמֹק וּלְרַכֵּךְ אֶת הַלֵּב שֶׁלִּי, שֶׁהוּא בְּסַךְ הַכֹּל בָּשָׂר וָדָם?".

עֲקִיבָא הֵבִין אֶת סוֹד הַטִּפּוֹת. טִפָּה אַחַת שֶׁל מַיִם לֹא עוֹשָׂה כְּלוּם, גַּם לֹא עֶשֶׂר. אֲבָל מִילְיוֹנֵי טִפּוֹת, שֶׁנּוֹפְלוֹת בְּהַתְמָדָה, לְלֹא הַפְסָקָה – חוֹצְבוֹת בַּסֶּלַע!

הָאַבָּא בְּכִתָּה א'

עֲקִיבָא עָזַב אֶת הַבְּאֵר. עֵינָיו בָּרְקוּ. הוּא לֹא הִמְתִּין וְלֹא חָשַׁב מָה יַגִּידוּ אֲנָשִׁים. הוּא מִהֵר לְבֵיתוֹ, תָּפַס אֶת יָדוֹ שֶׁל בְּנוֹ הַקָּטָן (שֶׁכְּבָר הִגִּיעַ לְגִיל הַחֶדֶר), וְהָלַךְ אִתּוֹ אֶל 'מְלַמֵּד הַתִּינוֹקוֹת' (רַב שֶׁל יְלָדִים קְטַנִּים).

נַסּוּ לְדַמְיֵן אֶת הַתְּמוּנָה: חֶדֶר קָטָן, מָלֵא בִּילָדִים קְטַנְטַנִּים שֶׁיּוֹשְׁבִים וּמִתְנַדְנְדִים. וּבַשּׁוּרָה הָרִאשׁוֹנָה, יוֹשֵׁב אָדָם בֶּן אַרְבָּעִים, חָסוֹן וְגָדוֹל, עִם זָקָן וְכַפּוֹת יָדַיִם מְחֻסְפָּסוֹת.

"רַבִּי," אָמַר עֲקִיבָא בַּעֲנָוָה עֲצוּמָה, "לַמְּדֵנִי תּוֹרָה!".

הַמְלַמֵּד הֵבִיא לָהֶם לְלוּחַ מֵעֵץ עָלָיו חֲקוּקוֹת הָאוֹתִיּוֹת. רַבִּי עֲקִיבָא אָחַז בְּצַד אֶחָד שֶׁל הַלּוּחַ, וּבְנוֹ הַקָּטָן אָחַז בַּצַּד הַשֵּׁנִי שֶׁל אוֹתוֹ הַלּוּחַ מַמָּשׁ.

הַמְלַמֵּד הִצְבִּיעַ: "זֶה - אָלֶף. וְזֶה - בֵּית".

וַעֲקִיבָא צָעַק יַחַד עִם הַיְלָדִים: "אָ-לֶף! בֵּ-ית!". הוּא שָׁנָה וְחָזַר, שׁוּב וָשׁוּב, וְהַלֵּב שֶׁלּוֹ כִּמְעַט הִתְפּוֹצֵץ מִשִּׂמְחָה עַל כָּל אוֹת קְדוֹשָׁה שֶׁקָּלַט בְּשִׂכְלוֹ. הוּא לָמַד אֶת הָאוֹתִיּוֹת הַפְּשׁוּטוֹת. אַחַר כָּךְ הִתְקַדֵּם לִלְמֹד חוּמָשׁ, מִלָּה אַחַר מִלָּה: "וַ-יִּ-קְ-רָ-א אֶ-ל מֹ-שֶׁ-ה".

הַהַתְמָדָה שֶׁנִּצְּחָה אֶת הַבּוּשָׁה

עֲקִיבָא הָיָה יְשֵׁן עַל קַשׁ, חוֹטֵב עֵצִים לְפַרְנָסָתוֹ וּלְאוֹרָם שֶׁל הָעֵצִים הַבּוֹעֲרִים, כְּשֶׁהֶעָשָׁן צוֹרֵב אֶת עֵינָיו, הָיָה לוֹמֵד וְלוֹמֵד, טִפָּה אַחַר טִפָּה.

מִשָּׁם זָכָה וְעָלָה, וְנָסַע לַיְשִׁיבוֹת שֶׁל הַחֲכָמִים הֲכִי גְּדוֹלִים בַּדּוֹר – רַבִּי אֱלִיעֶזֶר וְרַבִּי יְהוֹשֻׁעַ. שְׁתֵּים עֶשְׂרֵה שָׁנִים וְעוֹד שְׁתֵּים עֶשְׂרֵה שָׁנִים, הוּא חָפַר בַּתּוֹרָה, שָׁאַל קֻשְׁיוֹת, חִפֵּשׂ תְּשׁוּבוֹת, וְלֹא וִתֵּר לְעַצְמוֹ!

מִי הָיָה מַאֲמִין? הָרוֹעֶה הַפָּשׁוּט וְהֶעָנִי, שֶׁהִתְבַּיֵּשׁ לֹא לָדַעַת אָלֶף־בֵּית, הָפַךְ לְהִיוֹת רַבִּי עֲקִיבָא ! עֲנַק הָעֲנָקִים שֶׁל תּוֹרַת יִשְׂרָאֵל, שֶׁעֶשְׂרִים וְאַרְבָּעָה אֶלֶף תַּלְמִידִים גָּמְעוּ בְּצָמָא אֶת תּוֹרָתוֹ! וְהַכֹּל, יְלָדִים, הַכֹּל בִּזְכוּת הֲבָנָה שֶׁל סוֹד אֶחָד שֶׁל בָּרְא עֹלָם: טִפָּה וְעוֹד טִפָּה שֶׁל תּוֹרָה – וְהַלֵּב הֲכִי אָטוּם, נִפְתָּח וּמֵאִיר!

מקורות: תחילת לימודו וסוד האבן והמים: אבות דרבי נתן, פרק ו', משנה ב' ("מה היה תחילתו של רבי עקיבא..."). ישיבתו עם בנו אצל מלמד התינוקות: שם, "הלך הוא ובנו וישבו אצל מלמדי תינוקות, אמר לו רבי למדני תורה, אחז רבי עקיבא בראש הלוח ובנו בראש הלוח...". הלימוד העצמי ולימוד אצל גדולי הדור: התלמוד הבבלי, מסכת כתובות, דף ס"ב, ע"ב - ס"ג, ע"א (על דברי רחל: "אזיל יתיב תרי סרי שנין בבי רב...").

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “יְלָדִים, עִצְמוּ אֶת הָעֵינַיִם וְנַסּוּ לְהָרִיחַ אֶת רֵיחַ הָאֲדָמָה הַיְּבֵשָׁה וְאֶת רֵיחַ צֶמֶר הַכְּבָשִׂים.” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “אִם רַק תֵּלֵךְ לִלְמֹד תּוֹרָה, אַתָּה תִּהְיֶה רַב גָּדוֹל בְּיִשְׂרָאֵל!".”?
ג. כיצד הגיב רבי עקיבא למצב המתואר: “וְהִנֵּה, בְּתוֹךְ הָאֶבֶן הַזּוֹ הָיָה חוֹר עָמֹק, מְשֻׁפְשָׁף וְחָלָק גַּלִּים-גַּלִּים.”?
ד. מה השתנה בין “טִפָּה אַחַת שֶׁל מַיִם לֹא עוֹשָׂה כְּלוּם, גַּם לֹא עֶשֶׂר.” לבין “הַמְלַמֵּד הֵבִיא לָהֶם לְלוּחַ מֵעֵץ עָלָיו חֲקוּקוֹת הָאוֹתִיּוֹת.”?

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “אבנים שחקו המים — הלימוד עם בנו” על רבי עקיבא? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “הַקָּדוֹשׁ בָּרוּךְ הוּא פָּתַח אֶת לִבּוֹ שֶׁל עֲקִיבָא בְּאוֹתוֹ הָרֶגַע מַמָּשׁ.”.
ג. אילו רבי עקיבא לא היה פועל כפי שמתואר ב־“הַמְלַמֵּד הֵבִיא לָהֶם לְלוּחַ מֵעֵץ עָלָיו חֲקוּקוֹת הָאוֹתִיּוֹת.” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “שְׁתֵּים עֶשְׂרֵה שָׁנִים וְעוֹד שְׁתֵּים עֶשְׂרֵה שָׁנִים, הוּא חָפַר בַּתּוֹרָה, שָׁאַל קֻשְׁיוֹת, חִפֵּשׂ תְּשׁוּבוֹת, וְלֹא…” מחזק את המסר של הסיפור?

4. הוּא לֹא הִמְתִּין וְלֹא חָשַׁב מָה יַגִּידוּ אֲנָשִׁים.

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “וְהִנֵּה, בְּתוֹךְ הָאֶבֶן הַזּוֹ הָיָה חוֹר עָמֹק, מְשֻׁפְשָׁף וְחָלָק גַּלִּים-גַּלִּים.”.
מכתב: כתבו לרבי עקיבא בעקבות “הוּא מִהֵר לְבֵיתוֹ, תָּפַס אֶת יָדוֹ שֶׁל בְּנוֹ הַקָּטָן (שֶׁכְּבָר הִגִּיעַ לְגִיל הַחֶדֶר), וְהָלַךְ אִתּוֹ אֶל 'מְלַמֵּד…”.

דיווח: כתבו ידיעה קצרה שכותרתה “אבנים שחקו המים — הלימוד עם בנו”.
