במערה בפקיעין — 12 שנה (ואח"כ שנה נוספת)

דף עבודה בעקבות רבי שמעון בר יוחאי (רשב"י)
שם: כיתה:

המעשה — הסיפור המלא

הנה מערך חינוכי, רגשי ומרתק על רַבִּי שִׁמְעוֹן בַּר יוֹחַאי (רשב"י) במערה, הבנוי בדיוק לפי ההנחיות והמקורות, מותאם במיוחד לתלמידי כיתות ב'-ד' בחינוך הדתי-לאומי.

פְּתִיחָה: הַבְּרִיחָה אֶל הַחֹשֶׁךְ

יְלָדִים, תּאֲרוּ לָכֶם לַעֲזֹב אֶת הַבַּיִת הַחַם, אֶת הַמִּטָּה הָרַכָּה וְאֶת הָאֹכֶל הַטָּעִים, וְלִבְרֹחַ אֶל מָקוֹם שֶׁאִישׁ לֹא יוֹדֵעַ עָלָיו, רַק כְּדֵי שֶׁתּוּכְלוּ לְהַמְשִׁיךְ לִלְמֹד תּוֹרָה.

כָּךְ בְּדִיּוּק קָרָה לְגָדוֹל תַּלְמִידָיו שֶׁל רַבִּי עֲקִיבָא – הַצַּדִּיק הַקָּדוֹשׁ רַבִּי שִׁמְעוֹן בַּר יוֹחַאי , וְלִבְנוֹ רַבִּי אֶלְעָזָר .

הַיָּמִים הָיוּ יָמִים שֶׁל פַּחַד. הָרוֹמָאִים הָרְשָׁעִים שָׁלְטוּ בְּאֶרֶץ יִשְׂרָאֵל, וְכַאֲשֶׁר שָׁמַע הַקֵּיסָר הָרוֹמָאִי שֶׁרַבִּי שִׁמְעוֹן אָמַר עֲלֵיהֶם דְּבָרִים קָשִׁים, הוּא כָּעַס מְאֹד וְגָזַר דִּין מָוֶת עַל הַצַּדִּיק. רַבִּי שִׁמְעוֹן יָדַע: אִם יִתְפְּסוּ אוֹתוֹ, יִפָּסֵק לִמּוּד הַתּוֹרָה שֶׁלּוֹ. הוּא לָקַח אֶת בְּנוֹ רַבִּי אֶלְעָזָר, וּבְאִישׁוֹן לַיְלָה הֵם הָלְכוּ לַגָּלִיל, אֶל הַהָרִים, עַד שֶׁמָּצְאוּ מְעָרָה עֲמֻקָּה בַּכְּפָר פְּקִיעִין.

הֵם נִכְנְסוּ פְּנִימָה. הַחֹשֶׁךְ הָיָה מֻחְלָט. קִירוֹת הָאֶבֶן הָיוּ קָרִים מְאֹד, וּמִבַּחוּץ נִשְׁמְעוּ רַק יְלָלוֹת תַּנִּים וְשִׁקְשׁוּק עֲלֵי הָעֵצִים. לֹא הָיָה לָהֶם אֹכֶל, לֹא מַיִם וְלֹא בְּגָדִים לְהַחְלָפָה.

אַךְ הַקָּדוֹשׁ בָּרוּךְ הוּא, שֶׁשּׁוֹמֵר עַל תַּלְמִידֵי חֲכָמִים, הִשְׁגִּיחַ עֲלֵיהֶם בְּנֵס גָּלוּי: פִּתְאוֹם, מִתּוֹךְ הַסֶּלַע הַיָּבֵשׁ בְּפֶתַח הַמְּעָרָה, פִּקְפֵּק... בִּעְבֵּעַ... וּפָרַץ מַעְיַן מַיִם חַיִּים קְרִירִים וּמְתוּקִים! וּלְצִדּוֹ, בִּמְהִירוּת עֲצוּמָה, צָמַח וְעָלָה עֵץ עָבֶה מָלֵא בְּפֵרוֹת. עֵץ חָרוּבִים.

שְׁתֵּים עֶשְׂרֵה שָׁנִים בַּחוֹל

יְלָדִים, נַסּוּ לֶאֱכֹל רַק חָרוּבִים וְלִשְׁתּוֹת רַק מַיִם מֶשֶׁךְ יוֹם שָׁלֵם. זֶה קָשֶׁה מְאֹד, נָכוֹן? רַבִּי שִׁמְעוֹן וּבְנוֹ חָיוּ כָּךְ לֹא יוֹם... לֹא חֹדֶשׁ... אֶלָּא שְׁתֵּים עֶשְׂרֵה שָׁנִים רְצוּפוֹת! הַטַּעַם הַיָּבֵשׁ שֶׁל הֶחָרוּב קִיֵּם אֶת הַגּוּף הֶחָלָשׁ שֶׁלָּהֶם.

וּמָה הֵם עָשׂוּ כָּל הַזְּמַן הַזֶּה? הֵם לָמְדוּ תּוֹרָה! אֶת סוֹדוֹת הַתּוֹרָה הַגְּדוֹלִים בְּיוֹתֵר בָּעוֹלָם.

אֲבָל כֵּיוָן שֶׁלֹּא הָיוּ לָהֶם בְּגָדִים אֲחֵרִים, וְהֵם רָצוּ לִשְׁמֹר עַל בִּגְדֵיהֶם מִלְּהִקָּרַע וּלְהִבָּלוֹת, הֵם עָשׂוּ דָּבָר מַדְהִים: בְּכָל בֹּקֶר הֵם פָּשְׁטוּ אֶת הַבְּגָדִים, חָפְרוּ בּוֹר בְּחוֹל שֶׁבַּמְּעָרָה, וְנִכְנְסוּ לְתוֹכוֹ עַד לַצַּוָּאר. הַחוֹל הַקַּר וְהַמְחֻסְפָּס שָׂרַט אֶת עוֹרָם. זֶה צָרַב. זֶה כָּאַב. גּוּפָם נִהְיָה מָלֵא פְּצָעִים מֵהַחוֹל שֶׁהִתְחַכֵּךְ בִּבְשָׂרָם, אֲבָל הֵם לֹא הִפְסִיקוּ לִלְמֹד אֲפִלּוּ לְרֶגַע! קוֹל לִמּוּד הַתּוֹרָה בָּקַע מִתּוֹךְ הַבּוֹרוֹת בְּהִתְלַהֲבוּת שֶׁל אֵשׁ.

רַק כַּאֲשֶׁר הִגִּיעַ זְמַן הַתְּפִלָּה, הֵם הָיוּ יוֹצְאִים מֵהַחוֹל, מִתְנַעֲרִים, לוֹבְשִׁים אֶת בִּגְדֵיהֶם, מִתְפַּלְּלִים בְּאֵימָה וּבְיִרְאָה מִפְּנֵי מֶלֶךְ מַלְכֵי הַמְּלָכִים – וְאָז פּוֹשְׁטִים אֶת הַבְּגָדִים וְחוֹזְרִים אֶל הַחוֹל לִלְמֹד תּוֹרָה.

נְשָׁמוֹת שֶׁל אוֹר בְּתוֹךְ קִירוֹת שֶׁל אֶבֶן!

הַיְּצִיאָה וְהָאֵשׁ

כָּךְ עָבְרוּ שְׁתֵּים עֶשְׂרֵה שָׁנָה.

יוֹם אֶחָד, הַשֶּׁקֶט שֶׁל כְּפָר פְּקִיעִין נִשְׁבַּר. בְּפֶתַח הַמְּעָרָה עָמְדָה דְּמוּת מְאִירָה כְּמַלְאָךְ. זֶה הָיָה אֵלִיָּהוּ הַנָּבִיא! הוּא קָרָא בְּקוֹל שֶׁהִדְהֵד עַל הַסְּלָעִים: "מִי יוֹדִיעַ לְרַבִּי שִׁמְעוֹן בַּר יוֹחַאי שֶׁהַקֵּיסָר הָרוֹמָאִי מֵת, וְהַגְּזֵרָה בֻּטְּלָה?".

כְּשֶׁרַבִּי שִׁמְעוֹן וְרַבִּי אֶלְעָזָר שָׁמְעוּ זֹאת, הֵם לָבְשׁוּ אֶת בִּגְדֵיהֶם וְיָצְאוּ לְאִטָּם אֶל חֲלַל הָעוֹלָם הַגָּדוֹל. הַשֶּׁמֶשׁ סִנְוְרָה אֶת עֵינֵיהֶם שֶׁהִתְרַגְּלוּ לַחֲשֵׁכָה. הֵם הִבִּיטוּ יָמִינָה וּשְׂמֹאלָה, וְרָאוּ אֲנָשִׁים רְגִילִים.

הִנֵּה יְהוּדִי אֶחָד חוֹרֵשׁ אֶת הַשָּׂדֶה. הִנֵּה אַחֵר זוֹרֵעַ חִטָּה.

הִזְדַּעְזַע רַבִּי שִׁמְעוֹן. הָרֹאשׁ שֶׁלּוֹ הָיָה בַּשָּׁמַיִם, בְּסוֹדוֹת וּבְמַלְאָכִים! אֵיךְ יְהוּדִי יָכוֹל לְבַזְבֵּז זְמַן יָקָר כָּל כָּךְ עַל טְרַקְטוֹרִים וַעֲבוֹדַת אֲדָמָה?!

"מַנִּיחִים חַיֵּי עוֹלָם (הַתּוֹרָה), וְעוֹסְקִים בְּחַיֵּי שָׁעָה?!" קָרָא רַבִּי שִׁמְעוֹן בִּכְאֵב יָקֵד.

מֵרֹב קְדֻשָּׁתוֹ וְהַקְפָּדָתוֹ, כָּל מָקוֹם שֶׁרַבִּי שִׁמְעוֹן וְרַבִּי אֶלְעָזָר הִסְתַּכְּלוּ עָלָיו – מִיָּד עָלָה בְּאֵשׁ! תְּבוּאָה שְׁלֵמָה נִשְׂרְפָה! הֵם הָיוּ אֵשׁ שֶׁל תּוֹרָה שֶׁלֹּא יָכְלָה לִסְבֹּל אֶת הָעוֹלָם הַפָּשׁוּט.

לְפֶתַע, יָרַד קוֹל רַעַם אַדִּיר מִן הַשָּׁמַיִם, בַּת קוֹל מֵאֵת הַקָּדוֹשׁ בָּרוּךְ הוּא, שֶׁזִּעְזְעָה אֶת הֶהָרִים: "לְהַחְרִיב עוֹלָמִי יְצָאתֶם?! חִזְרוּ לִמְעָרַתְכֶם!"

הַקָּדוֹשׁ בָּרוּךְ הוּא רָצָה שֶׁהַתּוֹרָה לֹא תִּשְׂרֹף אֶת הָעוֹלָם, אֶלָּא תָּאִיר אוֹתוֹ. תְּבָרֵךְ אוֹתוֹ.

הַזָּקֵן עִם הַהֲדַסִּים

בְּעֶצֶב וּבַעֲנָוָה, חָזְרוּ שְׁנֵי הַצַּדִּיקִים אֶל הַחֹשֶׁךְ, אֶל הַחוֹל הַצּוֹרֵב, לְעוֹד שְׁנֵים עָשָׂר חֳדָשִׁים. שָׁנָה שְׁלֵמָה תִּקְּנוּ אֶת מַבָּטָם עַל הָעוֹלָם.

לְאַחַר שָׁנָה, שׁוּב יָצְאָה בַּת קוֹל: "צְאוּ מִמְּעָרַתְכֶם!".

הַפַּעַם, כְּשֶׁיָּצְאוּ, זֶה הָיָה בְּיוֹם שִׁשִּׁי אַחַר הַצָּהֳרַיִם. הַשֶּׁמֶשׁ הִתְחִילָה לָרֶדֶת וְצָבְעָה אֶת שְׁמֵי הַגָּלִיל בְּכָתֹם מֻזְהָב. בָּאֲוִיר מַמָּשׁ הֻרְגְּשָׁה קְדֻשָּׁתָהּ שֶׁל הַשַּׁבָּת הַמִּתְקָרֶבֶת.

רַבִּי שִׁמְעוֹן וְרַבִּי אֶלְעָזָר הָלְכוּ בַּדֶּרֶךְ, וְהִנֵּה הֵם רוֹאִים יְהוּדִי זָקֵן, שֶׁנִּרְאֶה עָיֵף מֵעֲבוֹדַת הַשָּׂדֶה הַקָּשָׁה שֶׁל כָּל הַשָּׁבוּעַ, רָץ וּמְמַהֵר בִּרְחוֹבוֹת הָעִיר, וּבְיָדוֹ – שְׁנֵי עֲנָפִים שֶׁל עֲצֵי הֲדַס יְרֻקִּים, שֶׁרֵיחָם מָתוֹק וּמְרַעֲנֵן.

"יְהוּדִי יָקָר," עוֹצֵר אוֹתוֹ רַבִּי שִׁמְעוֹן בַּעֲדִינוּת, וְעֵינָיו רַכּוֹת. "לְאָן אַתָּה מְמַהֵר? וְלָמָּה לְךָ הַהֲדַסִּים הָאֵלֶּה אֲשֶׁר עִמְּךָ?"

הַזָּקֵן נוֹשֵׁם בִּכְבֵדוּת אֲבָל חִיּוּךְ רָחָב וּמָאוֹר מְמַלֵּא אֶת פָּנָיו הַקְּמוּטוֹת: "לִכְבוֹד שַׁבָּת קֹדֶשׁ! כְּדֵי שֶׁיִּהְיֶה לִי רֵיחַ טוֹב בְּבֵיתִי בְּיוֹם הַמְּנוּחָה שֶׁל ה' יִתְבָּרַךְ!".

"וְלָמָּה לְךָ שְׁנַיִם?" שָׁאַל רַבִּי אֶלְעָזָר. "הַאִם לֹא מַסְפִּיק עָנָף אֶחָד שֶׁיָּפִיץ אֶת רֵיחוֹ?"

עָנָה הַזָּקֵן בְּשִׂמְחָה עֲצוּמָה: "אֶחָד כְּנֶגֶד 'זָכוֹר אֶת יוֹם הַשַּׁבָּת לְקַדְּשׁוֹ', וְאֶחָד כְּנֶגֶד 'שָׁמוֹר אֶת יוֹם הַשַּׁבָּת לְקַדְּשׁוֹ'!" וְהַזָּקֵן הִמְשִׁיךְ לָרוּץ בְּשִׂמְחָה לְבֵיתוֹ כְּדֵי לְקַבֵּל אֶת הַשַּׁבָּת.

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “גּוּפָם נִהְיָה מָלֵא פְּצָעִים מֵהַחוֹל שֶׁהִתְחַכֵּךְ בִּבְשָׂרָם, אֲבָל הֵם לֹא הִפְסִיקוּ לִלְמֹד אֲפִלּוּ לְרֶגַע!”.
מכתב: כתבו לרבי שמעון בר יוחאי (רשב"י) בעקבות “יְהוּדִי יָקָר," עוֹצֵר אוֹתוֹ רַבִּי שִׁמְעוֹן בַּעֲדִינוּת, וְעֵינָיו רַכּוֹת.”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “הוּא הֵבִין שֶׁעַם יִשְׂרָאֵל קָדוֹשׁ – לֹא רַק כְּשֶׁלּוֹמְדִים תּוֹרָה בַּמְּעָרָה כָּל הַיּוֹם וְהַלַּיְלָה, אֶלָּא גַּם…”.
דיווח: כתבו ידיעה קצרה שכותרתה “במערה בפקיעין — 12 שנה (ואח"כ שנה נוספת)”.

המחזה

רגע מן הסיפורהדמויותמה יאמרו או יעשו?
וּלְצִדּוֹ, בִּמְהִירוּת עֲצוּמָה, צָמַח וְעָלָה עֵץ עָבֶה מָלֵא בְּפֵרוֹת.
