סיפור 37 • רבי יהודה בר אלעאי • הוויכוח על התנהגות הרומאים

הוויכוח על התנהגות הרומאים

דף עבודה בעקבות רבי יהודה בר אלעאי
שם: כיתה:

המעשה — הסיפור המלא

פְּתִיחָה: בַּסֵּתֶר, לְאוֹר הַנֵּר

יְלָדִים מְתוּקִים, עִצְמוּ אֶת הָעֵינַיִם וְנַסּוּ לְהַקְשִׁיב. הַאִם אַתֶּם שׁוֹמְעִים אֶת הַשֶּׁקֶט שֶׁל הַלַּיְלָה?

אֲנַחְנוּ נִמְצָאִים בְּאֶרֶץ יִשְׂרָאֵל, לִפְנֵי כְּאַלְפַּיִם שָׁנָה, זְמַן קָצָר לְאַחַר שֶׁחָרַב בֵּית הַמִּקְדָּשׁ הַשֵּׁנִי. בַּחוּץ חֹשֶׁךְ מֻחְלָט. רֵיחַ עָשָׁן קַל וַאֲדָמָה לַחָה עוֹמֵד בָּאֲוִיר. פִּתְאוֹם, מִמֶּרְחָק, נִשְׁמָעִים קוֹלוֹת שֶׁל צְעָדִים כְּבֵדִים: טְרָאח... טְרָאח... טְרָאח... אֵלּוּ הֵם מַגְּפֵי הַבַּרְזֶל שֶׁל חַיָּלֵי רוֹמִי הָאַכְזָרִים מְסַיְּרִים בָּרְחוֹבוֹת, חַרְבוֹתֵיהֶם מְקַשְׁקְשׁוֹת זֶה בָּזֶה. כֻּלָּם רוֹעֲדִים מֵהֶם וּמִסְתַּגְּרִים בַּבָּתִּים.

אַךְ יֵשׁ מָקוֹם אֶחָד שֶׁבּוֹ לֹא פּוֹחֲדִים. בְּחֶדֶר קָטָן, לְאוֹר עֲשָׁשִׁית שֶׁמֶן מְהַבְהֶבֶת הַמְּפִיצָה רֵיחַ עָדִין שֶׁל זֵיתִים סְחוּטִים, יְשׁוּבִים שְׁלוֹשָׁה מִגְּדוֹלֵי עַם יִשְׂרָאֵל, קְדוֹשֵׁי עֶלְיוֹן, תַּלְמִידָיו שֶׁל רַבִּי עֲקִיבָא. אֵלּוּ הֵם: הַצַּדִּיק רַבִּי יְהוּדָה בַּר אִלְעָאִי , הַחָכָם הַגָּדוֹל רַבִּי יוֹסֵי , וְרַבִּי שִׁמְעוֹן בַּר יוֹחַאי אִישׁ הַסּוֹד וְהָאֱמֶת. יַחַד אִתָּם יָשַׁב אָדָם נוֹסָף שֶׁנִּקְרָא יְהוּדָה בֶּן גֵּרִים.

הַוִּכּוּחַ הַגָּדוֹל

הַחֲכָמִים יָשְׁבוּ וְשׂוֹחֲחוּ. הָרוֹמָאִים שָׁלְטוּ בָּאָרֶץ וְעָשׂוּ בָּהּ שִׁנּוּיִים רַבִּים.

רַבִּי יְהוּדָה, שֶׁלִּבּוֹ הָיָה מָלֵא תָּמִיד בְּאַהֲבָה וּבְעַיִן טוֹבָה, הִבִּיט בַּחֲבֵרָיו. הוּא יָדַע שֶׁהָרוֹמָאִים קָשִׁים, אַךְ הוּא חִפֵּשׂ צַד חִיּוּבִי, מַשֶּׁהוּ שֶׁמּוֹעִיל גַּם לַיְּהוּדִים לִחְיוֹת בְּשֶׁקֶט.

אָמַר רַבִּי יְהוּדָה בְּקוֹל שָׁקֵט וְרַךְ: "רְאוּ, כַּמָּה נָאִים מַעֲשֵׂיהֶם שֶׁל אֻמָּה זוֹ (הָרוֹמָאִים)! תִּקְּנוּ שְׁוָקִים יָפִים שֶׁיְּהוּדִים יְכוֹלִים לִקְנוֹת שָׁם מָזוֹן. תִּקְּנוּ גְּשָׁרִים רְחָבִים שֶׁלֹּא נִטְבַּע בַּנְּהָרוֹת, וְתִקְּנוּ בָּתֵּי מֶרְחָץ כְּדֵי שֶׁנּוּכַל לְרַחֵץ בְּמַיִם חַמִּים וְנָעִימִים לִכְבוֹד שַׁבָּת קֹדֶשׁ!".

רַבִּי יוֹסֵי שָׁמַע אֶת הַדְּבָרִים... וְשָׁתַק. הוּא לֹא סֵרֵב, וְלֹא הִסְכִּים.

אַךְ אָז... נַעֲנָה רַבִּי שִׁמְעוֹן בַּר יוֹחַאי! עֵינָיו בָּעֲרוּ כְּאֵשׁ לֶהָבָה. הוּא רָאָה אֶת הָאֱמֶת מִתַּחַת לַפָּנִים שֶׁל הָרוֹמָאִים. "לֹא!" אָמַר רַשְׁבִּ"י בְּתֹקֶף קְדוֹשָׁה. "כָּל מָה שֶׁתִּקְּנוּ וּבָנוּ – לֹא בָּנוּ אֶלָּא רַק לְצֹרֶךְ עַצְמָם וּלְתַאֲווֹתֵיהֶם! וְאֶת הַגְּשָׁרִים הֵם בָּנוּ רַק כְּדֵי לִקַּח מֵאִתָּנוּ עוֹד וְעוֹד מִסִּים עַל כָּל מַעֲבָר!".

יְלָדִים, זֶה לֹא הָיָה סְתָם וִכּוּחַ. זֶה הָיָה דִּיּוּן שֶׁל קְדֻשָּׁה بֵין תַּלְמִידֵי חֲכָמִים. יְהוּדָה בֶּן גֵּרִים, שֶׁיָּשַׁב שָׁם בַּפִּנָּה, שָׁמַע אֶת הַדְּבָרִים, הָלַךְ לְבֵיתוֹ וְסִפֵּר אוֹתָם. מִפֶּה לְאֹזֶן הִתְגַּלְגְּלוּ הַמִּלִּים, עַד שֶׁהִגִּיעוּ – לְאָזְנֵי הַמֶּמְשָׁלָה הָרוֹמָאִית!

הַגְּזֵרָה הַקָּשָׁה

כַּאֲשֶׁר שָׁמַע קֵיסַר רוֹמִי אֶת מָה שֶׁנֶּאֱמַר בְּאוֹתוֹ חֶדֶר, הוּא כָּעַס מְאֹד וְהוֹצִיא גְּזֵרָה אַכְזָרִית:

"שִׁמְעוֹן שֶׁגִּנָּה אוֹתָנוּ – יֵהָרֵג וְיוּמַת! יוֹסֵי שֶׁשָּׁתַק – יֵעָנֵשׁ וְיִגְלֶה מִבֵּיתוֹ לְעִיר צִפּוֹרִי!"

הַחַיָּלִים הָרוֹמָאִים יָצְאוּ לְחַפֵּשׂ אֶת הַחֲכָמִים. רַבִּי שִׁמְעוֹן, יַחַד עִם בְּנוֹ רַבִּי אֶלְעָזָר, נֶאֱלְצוּ לִבְרֹחַ מִיָּד בְּאִישׁוֹן לַיְלָה וּלְהִסְתַּתֵּר בַּמְּעָרָה, שָׁם לָמְדוּ תּוֹרָה בְּמֶשֶׁךְ שְׁלוֹשׁ עֶשְׂרֵה שָׁנִים אֲרֻכּוֹת (זֶהוּ הַסִּיפּוּר שֶׁאֲנַחְנוּ מַכִּירִים לִקְרַאת לַ"ג בָּעֹמֶר!).

אֲבָל מָה קָרָה לְרַבִּי יְהוּדָה?

"יְהוּדָה שֶׁשִּׁבֵּחַ אֶת בְּנִיָּתֵנוּ", הִכְרִיזוּ הָרוֹמָאִים, "הוּא יִתְעַלֶּה וִיקַבֵּל שָׂכָר! אֲנַחְנוּ מְמַנִּים אוֹתוֹ לִהְיוֹת 'רֹאשׁ הַמְדַבְּרִים בְּכָל מָקוֹם' !".

יְלָדִים, הָרוֹמָאִים חָשְׁבוּ שֶׁרַבִּי יְהוּדָה הוּא כָּעֵת שֶׁלָּהֶם. הֵם אָמְרוּ: "כָּל מָקוֹם שֶׁיֵּשׁ בּוֹ אֲסֵפָה אוֹ כִּנּוּס – תְּנוּ לְרַבִּי יְהוּדָה לְדַבֵּר רִאשׁוֹן! כָּךְ הוּא יַסְבִּיר לַיְּהוּדִים כַּמָּה אֲנַחְנוּ טוֹבִים, וְיַשְׁקִיט אוֹתָם".

הַנִּצָּחוֹן שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא

הַאִם רַבִּי יְהוּדָה בַּר אִלְעָאִי הָפַךְ לְעוֹזֵר שֶׁל הָרוֹמָאִים? חַס וְחָלִילָה!

יְלָדִים, כָּאן מִתְגַּלֵּית הַהַשְׁגָּחָה הַפְּרָטִית הָעֲצוּמָה . מַה שֶּׁהָרוֹמָאִים חָשְׁבוּ לַעֲשׂוֹת לְטוֹבָתָם – הַקָּדוֹשׁ בָּרוּךְ הוּא הָפַךְ לְנִצָּחוֹן שֶׁל הַתּוֹרָה הַקְּדוֹשָׁה!

רַבִּי יְהוּדָה קִבֵּל מֵהַשִּׁלְטוֹן תְּעוּדָה וְאִשּׁוּר מְיֻחָד לְדַבֵּר אֵיפֹה שֶׁרַק יִרְצֶה. אָז מָה עוֹשֶׂה גְּדוֹל יִשְׂרָאֵל שֶׁיֵּשׁ לוֹ אִשּׁוּר כָּזֶה? הוּא מַרְבִּיץ תּוֹרָה!!

מֵאוֹתוֹ יוֹם, רַבִּי יְהוּדָה עָבַר מִמָּקוֹם לְמָקוֹם, עָמַד לִפְנֵי קָהָל גָּדוֹל, וּבִמְקוֹם לְדַבֵּר עַל גְּשָׁרִים וְרוֹמָאִים, הוּא לִמֵּד הֲלָכוֹת: הִלְכוֹת שַׁבָּת, הִלְכוֹת צְדָקָה, בְּרָכוֹת וּמִצְווֹת! קוֹלוֹ נִשְׁמַע צָלוּל וּמָתוֹק לְמֵרָחוֹק, וְהַחַיָּלִים הָרוֹמָאִים אֲפִלּוּ שָׁמְרוּ שֶׁאַף אֶחָד לֹא יַפְרִיעַ לוֹ, כִּי הֲרֵי... קֵיסַר רוֹמִי בְּעַצְמוֹ נָתַן לוֹ רְשׁוּת!

כַּאֲשֶׁר הַחֲכָמִים יָשְׁבוּ וְסִדְּרוּ אֶת הַהֲלָכוֹת שֶׁל הַתּוֹרָה שֶׁבְּעַל-פֶּה לְתוֹךְ "הַמִּשְׁנָה", הֵם הִשְׁתַּמְּשׁוּ בַּכֹּחַ הָעָצוּם שֶׁל רַבִּי יְהוּדָה. בִּזְכוּת אוֹתָהּ עַיִן טוֹבָה שֶׁהָיְתָה לוֹ כְּשֶׁלִּמֵּד זְכוּת עַל מַעֲשֵׂיהֶם (כְּדֵי שֶׁיּוּכְלוּ לִחְיוֹת בְּשָׁלוֹם) – זָכָה רַבִּי יְהוּדָה לְכָבוֹד אֲמִתִּי שֶׁל תּוֹרָה.

וְכָךְ קָרָה, שֶׁשְּׁמוֹ שֶׁל "רַבִּי יְהוּדָה" מֻזְכָּר בַּמִּשְׁנָה הַקְּדוֹשָׁה אוּלַי יוֹתֵר מִכָּל תַּנָּא אַחֵר! יוֹתֵר מִ-600 הֲלָכוֹת רְשׁוּמוֹת עַל שְׁמוֹ!! וּכְשֶׁאַתֶּם תִּקְרְאוּ בַּגְּמָרָא אֶת הַמִּלִּים "אָמַר רַבִּי יְהוּדָה" - תֵּדְעוּ שֶׁמְּדֻבָּר עַל רַבִּי יְהוּדָה בַּר אִלְעָאִי, ראשׁ הַמְדַבְּרִים בְּכָל מָקוֹם! קוֹלָהּ הַנֶּאֱמָן שֶׁל הַתּוֹרָה.

מקורות: תלמוד בבלי, מסכת שבת דף ל"ג, עמוד ב' (הוויכוח בין רבי יהודה, רבי יוסי ורשב"י אודות בניית שווקים, גשרים ומרחצאות, המלשינות של יהודה בן גרים, וגזרת המלכות). העלאתו של רבי יהודה: הגמרא שם אומרת: "יהודה שעילה - יתעלה... יהא ראש המדברים בכל מקום". מספר ההלכות במשנה: כפי שמצוין במקורות הלימוד (אוסף "גדולי ישראל" של עמירם אלבה), רבי יהודה הפך בזכות כך לאחד מגדולי מוסרי המסורת עם כ-600 הלכות בשמו במשנה (בדרך כלל כשהמשנה אומרת "רבי יהודה" סתם - הכוונה לרבי יהודה בר אלעאי).

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “אֲנַחְנוּ נִמְצָאִים בְּאֶרֶץ יִשְׂרָאֵל, לִפְנֵי כְּאַלְפַּיִם שָׁנָה, זְמַן קָצָר לְאַחַר שֶׁחָרַב בֵּית הַמִּקְדָּשׁ הַשֵּׁנִי.” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “רַבִּי יְהוּדָה, שֶׁלִּבּוֹ הָיָה מָלֵא תָּמִיד בְּאַהֲבָה וּבְעַיִן טוֹבָה, הִבִּיט בַּחֲבֵרָיו.”?
ג. כיצד הגיב רבי יהודה בר אלעאי למצב המתואר: “הוּא רָאָה אֶת הָאֱמֶת מִתַּחַת לַפָּנִים שֶׁל הָרוֹמָאִים.”?

ה. כיצד הגיע הסיפור אל האירוע: “כַּאֲשֶׁר הַחֲכָמִים יָשְׁבוּ וְסִדְּרוּ אֶת הַהֲלָכוֹת שֶׁל הַתּוֹרָה שֶׁבְּעַל-פֶּה לְתוֹךְ "הַמִּשְׁנָה", הֵם הִשְׁתַּמְּשׁוּ בַּכֹּחַ הָעָצוּם שֶׁל…”?

3. מה היה ראשון?

כתבו ליד כל אירוע מספר מ־1 עד 5 לפי סדר התרחשותו בסיפור.

□ מִפֶּה לְאֹזֶן הִתְגַּלְגְּלוּ הַמִּלִּים, עַד שֶׁהִגִּיעוּ – לְאָזְנֵי הַמֶּמְשָׁלָה הָרוֹמָאִית!
□ אֲנַחְנוּ נִמְצָאִים בְּאֶרֶץ יִשְׂרָאֵל, לִפְנֵי כְּאַלְפַּיִם שָׁנָה, זְמַן קָצָר לְאַחַר שֶׁחָרַב בֵּית הַמִּקְדָּשׁ הַשֵּׁנִי.
□ וּכְשֶׁאַתֶּם תִּקְרְאוּ בַּגְּמָרָא אֶת הַמִּלִּים "אָמַר רַבִּי יְהוּדָה" - תֵּדְעוּ שֶׁמְּדֻבָּר עַל רַבִּי יְהוּדָה בַּר אִלְעָאִי, ראשׁ הַמְדַבְּרִים…
□ אָמַר רַבִּי יְהוּדָה בְּקוֹל שָׁקֵט וְרַךְ: "רְאוּ, כַּמָּה נָאִים מַעֲשֵׂיהֶם שֶׁל אֻמָּה זוֹ (הָרוֹמָאִים)!
□ כָּךְ הוּא יַסְבִּיר לַיְּהוּדִים כַּמָּה אֲנַחְנוּ טוֹבִים, וְיַשְׁקִיט אוֹתָם".

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “הוויכוח על התנהגות הרומאים” על רבי יהודה בר אלעאי? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “זֶה הָיָה דִּיּוּן שֶׁל קְדֻשָּׁה بֵין תַּלְמִידֵי חֲכָמִים.”.
ג. אילו רבי יהודה בר אלעאי לא היה פועל כפי שמתואר ב־“כָּךְ הוּא יַסְבִּיר לַיְּהוּדִים כַּמָּה אֲנַחְנוּ טוֹבִים, וְיַשְׁקִיט אוֹתָם".” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “כַּאֲשֶׁר הַחֲכָמִים יָשְׁבוּ וְסִדְּרוּ אֶת הַהֲלָכוֹת שֶׁל הַתּוֹרָה שֶׁבְּעַל-פֶּה לְתוֹךְ "הַמִּשְׁנָה", הֵם הִשְׁתַּמְּשׁוּ…” מחזק את המסר של הסיפור?
