ה' עונה לתפילת נכרייה

דף עבודה בעקבות רבי עקיבא
שם: כיתה:

המעשה — הסיפור המלא

להלן עיבוד חינוכי, רגשי ומרתק לסיפור המופלא על רבי עקיבא, העשירה והערבים (הקב"ה והים) . הסיפור נכתב בהתאמה מושלמת לבנים בכיתות ב'-ד' בחינוך הדתי-לאומי, תוך שימוש בעקרונות הסטוריטלינג שביקשת, כולל תיאורים חושיים, בניית מתח, והקפדה יתרה על שפה מכבדת ואווירה של קדושה וקידוש השם.

הסיפור מנוקד במלואו כנדרש.

קְצָת שֶׁקֶט לַלּוֹמְדִים

יְלָדִים יְקָרִים, כַּאֲשֶׁר אַתֶּם מִתְקָרְבִים לְבֵית הַכְּנֶסֶת אֹו לְבֵית הַמִּדְרָשׁ, אֵיזֶה צְלִיל מָתוֹק אַתֶּם שׁוֹמְעִים? נָכוֹן, אֶת קוֹל לִמּוּד הַתּוֹרָה! קוֹלוֹת מְלֵאִים שִׂמְחָה שֶׁל תַּלְמִידִים וְרַבָּנִים הַלּוֹמְדִים אֶת דְּבַר ה'.

לִפְנֵי שָׁנִים רַבּוֹת, בְּאֶרֶץ יִשְׂרָאֵל, הָיָה בֵּית מִדְרָשׁ עָצוּם. אַלְפֵי תַּלְמִידִים יָשְׁבוּ שָׁם. אִם הֱיִיתֶם נִכְנָסִים לַחֶדֶר, הֱיִיתֶם מְרִיחִים אֶת רֵיחַ קְלָפֵי הַסְּפָרִים הַיְשָׁנִים, וְשׁוֹמְעִים הֶמְיָה עֲדִינָה, כְּמוֹ שֶׁל כַּוֶּרֶת דְּבוֹרִים הָעוֹסְקוֹת בַּיִּצוּר הַדְּבַשׁ – אֲבָל כָּאן, הַתַּלְמִידִים יִצְּרוּ אֶת מְתִיקוּת הַתּוֹרָה.

וּמִי יָשַׁב בָּאֶמְצַע, עֵינָיו בּוֹרְקוֹת מֵאַהֲבָה וְחָכְמָה? זֶהוּ גְּדוֹל הַדּוֹר, הַתַּנָּא הַקָּדוֹשׁ רַבִּי עֲקִיבָא .

רַבִּי עֲקִיבָא אָהַב אֶת תַּלְמִידָיו אֲהֲבַת נֶפֶשׁ. אֲבָל בְּאוֹתָם יָמִים, הָיָה קָשֶׁה מְאֹד לְפַרְנֵס אֶת כֻּלָּם. תַּלְמִידִים רַבִּים הָיוּ עֲנִיִּים מְרוּדִים, הַבֶּטֶן שֶׁלָּהֶם קִרְקְרָה מֵרָעָב, וְהַנַּעֲלַיִם שֶׁלָּהֶם הָיוּ קְרוּעוֹת.

רַבִּי עֲקִיבָא חָשַׁב בְּלִבּוֹ: "הֵיאַךְ יוּכְלוּ בָּנַי הָאֲהוּבִים לַהֲגוֹת בַּתּוֹרָה בִּמְנוּחָה כַּאֲשֶׁר הֵם רְעֵבִים? אֲנִי חַיָּב לְהַשִּׂיג עֲבוּרָם מָמוֹן!". אֲבָל מֵאַיִן יָבִיא סְכוּם גָּדוֹל כָּל כָּךְ לְכָל כָּךְ הַרְבֵּה תַּלְמִידִים?

הַהַלְוָאָה וְהָעֲרֵבִים הַמֻּפְלָאִים

שָׁמַע רַבִּי עֲקִיבָא כִּי סָמוּךְ לְחוֹף הַיָּם הַגָּדוֹל, גָּרָה אִשָּׁה נוֹכְרִיָּה עֲשִׁירָה מְאֹד – קָרְאוּ לָהּ מַטְרוֹנִיתָא. הִיא הָיְתָה אִשָּׁה נִכְבָּדָה וּבַעֲלַת רְכוּשׁ רַב. הֶחְלִיט הַצַּדִּיק לָצֵאת לַדֶּרֶךְ וּלְבַקֵּשׁ מִמֶּנָּה הַלְוָאָה עֲבוּר תַּלְמִידֵי הַחֲכָמִים, עַד שֶׁיִּהְיֶה בְּיָדוֹ כֶּסֶף לְהַחְזִיר.

הוּא הִגִּיעַ לְאַרְמוֹנָהּ שֶׁעַל הַחוֹף. רוּחַ הַיָּם הַמְּלוּחָה נָשְׁבָה בַּחֲזָקָה, וְגַלִּים תְּכוּלִים וּלְבָנִים הִתְנַפְּצוּ עַל הַסְּלָעִים בְּקוֹל שָׁאוֹן: פְלָאשׁ! רַבִּי עֲקִיבָא, עֲטוּף בְּגַלִּימָתוֹ, עָמַד מוּל הָאִשָּׁה וּבִקֵּשׁ מִמֶּנָּה בַּעֲדִינוּת סְכוּם מָמוֹן אַדִּיר כְּהַלְוָאָה עֲבוּר תַּלְמִידָיו, כְּדֵי שֶׁיִּלְמְדוּ תּוֹרָה.

הַמַּטְרוֹנִיתָא הִבִּיטָה בִּפְנֵי הַצַּדִּיק, שֶׁהָיוּ מְאִירוֹת בְּאוֹר מְיֻחָד. הִיא הִסְכִּימָה בְּשִׂמְחָה לַעֲזֹר לוֹ, אַךְ הָיְתָה לָהּ שְׁאֵלָה אַחַת הֶגְיוֹנִית מְאֹד:

"חָכָם גָּדוֹל," הִיא אָמְרָה, "אֲנִי מוּכָנָה לָתֵת לְךָ אֶת הַכֶּסֶף מִיָּד. אֲבָל סְכוּם כָּזֶה הוּא עָצוּם! מִי יִהְיֶה הֶעָרֵב שֶׁלְּךָ? (עָרֵב הוּא אָדָם שֶׁמַּבְטִיחַ: אִם הַלּוֹוֶה לֹא מְשַׁלֵּם, אֲנִי אֲשַׁלֵּם בִּמְקוֹמוֹ). הֲרֵי צָרִיךְ לְהַעֲמִיד עָרֵב כְּדֵי שֶׁאֶהְיֶה בְּטוּחָה שֶׁאֲקַבֵּל אֶת הַכֶּסֶף בַּחֲזָרָה בַּזְּמַן שֶׁקָּבַעְנוּ!".

רַבִּי עֲקִיבָא עָצַר לְרֶגַע. מִי עָשִׁיר מַסְפִּיק לִהְיוֹת עָרֵב לַסְּכוּם הַזֶּה? הַצַּדִּיק הֵרִים אֶת עֵינָיו כְּלַפֵּי הַשָּׁמַיִם וְאַחַר כָּךְ הִבִּיט עַל הַיָּם הַגָּדוֹל וְהַפָּתוּחַ שֶׁמִּמּוּל.

בְּבִטָּחוֹן מוּחְלָט שֶׁאֵין כְּמוֹתוֹ בָּעוֹלָם, הוּא אָמַר: "מִי הָעֲרֵבִים שֶׁלִּי? הַקָּדוֹשׁ בָּרוּךְ הוּא... וְגַלֵּי הַיָּם! הֵם מֵעִידִים שֶׁאָשִׁיב לָךְ אֶת מְלוֹא הַסְּכוּם. ".

הַמַּטְרוֹנִיתָא הָיְתָה אִשָּׁה חֲכָמָה. הִיא הִכִּירָה בִּגְדֻלָּתוֹ וְהִסְכִּימָה בְּלֵב שָׁלֵם. "אִם אֵלּוּ הָעֲרֵבִים שֶׁלְּךָ, הֲרֵי אֲנִי בְּטוּחָה וּרְגוּעָה!". הִיא סָפְרָה לְיָדָיו אֶת מַטְבְּעוֹת הַזָּהָב: צִלְצוּל שֶׁל מַטְבֵּעַ אַחַר מַטְבֵּעַ – צְ'לִינְג, צְ'לִינְג. רַבִּי עֲקִיבָא קָבַע עִמָּהּ תַּאֲרִיךְ מְדֻיָּק לְהַחְזָרַת הַכֶּסֶף, וְחָזַר מִיָּד לְתַלְמִידָיו כְּדֵי שֶׁיּוּכְלוּ לִלְמֹד תּוֹרָה בְּנַחַת וּבְשִׂמְחָה.

מִבְחַן הָאֱמוּנָה שֶׁל הַצַּדִּיק

הַזְּמַן חָלַף עָבַר לוֹ. תַּלְמִידֵי רַבִּי עֲקִיבָא הָגוּ בַּתּוֹרָה יוֹמָם וָלַיְלָה. אֲבָל יְלָדִים, אַתֶּם יוֹדְעִים שֶׁבַּשָּׁמַיִם מְנַסִּים לִפְעָמִים אֶת הַצַּדִּיקִים הַגְּדוֹלִים בְּיוֹתֵר?

הִגִּיעַ הַתַּאֲרִיךְ שֶׁבּוֹ רַבִּי עֲקִיבָא הָיָה צָרִיךְ לְהַחְזִיר אֶת הַכֶּסֶף. הוּא הֵכִין אֶת תַּשְׁלוּם הַחוֹב עַד לַפְּרוּטָה הָאַחֲרוֹנָה, וְתִכְנֵן לָצֵאת מִיָּד לְבֵיתָהּ שֶׁל הַמַּטְרוֹנִיתָא. אֲבָל בְּדִיּוּק בְּאוֹתוֹ בֹּקֶר – רַבִּי עֲקִיבָא נָפַל לְמִשְׁכָּב.

מַחֲלָה חֲזָקָה וּכְאֵבִים עַזִּים תָּקְפוּ אֶת גּוּפוֹ הַקָּדוֹשׁ. הוּא בָּעַר מֵחֹם חָזָק וְהָיָה חַלָּשׁ מְאֹד, עַד שֶׁלֹּא יָכוֹל הָיָה לְהָזִיז אֶת רַגְלָיו כְּלָל.

עַל מִטַּת חָלְיוֹ, הַדָּבָר שֶׁהֲכִי הִכְאִיב לְרַבִּי עֲקִיבָא לֹא הָיָה הַחֹם אֹו הַחֻלְשָׁה. הוּא בָּכָה מִצַּעַר אֲמִתִּי עַל דָּבָר אַחֵר לְגַמְרֵי: "רִבּוֹנוֹ שֶׁל עוֹלָם! הִבְטַחְתִּי לְהַחְזִיר אֶת הַכֶּסֶף הַיּוֹם! מַה תֹּאמַר עַכְשָׁו אוֹתָהּ אִשָּׁה נוֹכְרִיָּה? הִיא תַּחְשֹׁב שֶׁחָלִילָה חֲכָמֵי יִשְׂרָאֵל אֵינָם עוֹמְדִים בְּדִבּוּרָם... מְדֻבָּר בְּכְבוֹד שָׁמַיִם! בְּקִדּוּשׁ שֵׁם ה'!". אֲבָל גּוּפוֹ בָּגַד בּוֹ, וְהוּא פָּשׁוּט לֹא יָכוֹל הָיָה לָקוּם.

הַזְּעָקָה אֶל הַיָּם

בְּאוֹתָהּ שָׁעָה, רְחוֹק שָׁם עַל שְׂפַת הַיָּם, יָצְאָה הַמַּטְרוֹנִיתָא הָעֲשִׁירָה מִבֵּיתָהּ. הִיא צָעֲדָה עַל הַחוֹל הָרָטֹב וְחִכְּתָה, אֲבָל רַבִּי עֲקִיבָא לֹא הִגִּיעַ. הַשֶּׁמֶשׁ כְּבָר לָנְטָה לִשְׁקֹעַ. מַה עָשְׂתָה אוֹתָהּ אִשָּׁה? הִיא זָכְרָה מִי הָיוּ הָעֲרֵבִים שֶׁל רַבִּי עֲקִיבָא.

הִיא הִתְקָרְבָה עַד לְקַו הַמַּיִם. רוּחַ סְעָרָה קְרִירָה הֵרִימָה אֶת מֵי הַיָּם לְגַלִּים שׁוֹצְפִים. הַנּוֹכְרִיָּה נָשְׂאָה אֶת קוֹלָהּ לְעֵבֶר הַשָּׁמַיִם וְהַיָּם וְקָרְאָה בְּקוֹל גָּדוֹל שֶׁהִתְגַּבֵּר עַל רַעַשׁ הַגַּלִּים:

"רִבּוֹנוֹ שֶׁל עוֹלָם! אַתָּה יוֹדֵעַ שֶׁעֲקִיבָא עַבְדְּךָ חָלָה, וְאֵינוֹ יָכוֹל לָבוֹא הַיּוֹם לְשַׁלֵּם לִי אֶת חֹבוֹ. אַתָּה, רִבּוֹן הָעוֹלָמִים, הָיִיתָ הֶעָרֵב שֶׁלּוֹ הַיַּחַד עִם הַיָּם! אֲנִי דּוֹרֶשֶׁת שֶׁאַתֶּם, הָעֲרֵבִים שֶׁלּוֹ, תְּשַׁלְּמוּ לִי אֶת חֹבוֹ עַכְשָׁו!".

יְלָדִים, נִשְׁמַע לָכֶם דִּמְיוֹנִי? נִשְׁמַע שֶׁהִיא מְבַקֶּשֶׁת בַּקָּשָׁה בִּלְתִּי אֶפְשָׁרִית? אֲבָל אֵצֶל בּוֹרֵא עוֹלָם שׁוּם דָּבָר אֵינוֹ בִּלְתִּי אֶפְשָׁרִי! הַקָּדוֹשׁ בָּרוּךְ הוּא, שֶׁשָּׁמַע אֶת צַעֲרוֹ הַטָּהוֹר שֶׁל רַבִּי עֲקִיבָא וְאֶת דְּרִישָׁתָהּ שֶׁל הָאִשָּׁה, סוֹבֵב אֶת מַעֲרֶכֶת הַהַשְׁגָּחָה הָעֶלְיוֹנָה בְּדֶרֶךְ נִפְלָאָה בְּיוֹתֵר.

תֵּבַת הַזָּהָב בַּגַּלִּים

בְּדִיּוּק בְּאוֹתָהּ שָׁעָה, מֶרְחָק עָצוּם מִשָּׁם, בְּאַרְמוֹנוֹ שֶׁל הַקֵּיסָר אוֹ מֶלֶךְ גָּדוֹל, קָרָה אֵרוּעַ קָשֶׁה: בִּתּוֹ שֶׁל הַקֵּיסָר יָרְדָה פִּתְאוֹם מִדַּעְתָּהּ. מֵרֹב בִּלְבּוּל מַחֲשַׁבְתִּי, הִיא נִגְּשָׁה לַכַּסֶּפֶת הַמַּלְכוּתִית, אָחֲזָה בְּקֻפְסַת עֵץ כְּבֵדָה וְעָבָה שֶׁהָיְתָה מְלֵאָה עַד שְׂפָתָהּ בִּזְהוּבִים, אַבְנֵי יְקָר וְיַהֲלוֹמִים, הִתְקָרְבָה לַחַלּוֹן הָאַרְמוֹן שֶׁעַל הַצּוּק וְ... זָרְקָה אֶת תֵּבַת הָאוֹצָר לְתוֹךְ הַיָּם הַגּוֹעֵשׁ! פְלָאק!

הַיָּם רָעַשׁ. גַּל דָּחַף גַּל. וְהִנֵּה, כְּמוֹ רַכֶּבֶת מְהִירָה הַמּוּנַעַת בְּיָד נֶעֱלָמָה, נָשְׂאוּ גַּלֵּי הַיָּם אֶת הַתֵּבָה הַכְּבֵדָה עַל פְּנֵי הַמַּיִם... וְהִשִּׁיטוּ אוֹתָהּ מֶרְחַקִּים עֲצוּמִים עַד... עַד לַחוֹף שֶׁל אוֹתָהּ אִשָּׁה מַטְרוֹנִיתָא!

הָאִשָּׁה עוֹמֶדֶת עַל הַחוֹף, שְׂעָרָהּ מִתְבַּדֵּר בָּרוּחַ. פִּתְאוֹם גַּל חָזָק שׁוֹטֵף אֶת הַחוֹל מַמָּשׁ לְמַרְגְּלוֹתֶיהָ, וּמַשְׁאִיר שָׁם קֻפְסַת עֵץ רְטוּבָה וּכְבֵדָה.

הִיא בְּקֹשִׁי מַצְלִיחָה לִפְתֹּחַ אֶת הַתֵּבָה הַחֲלָקָה מִמַּיִם. "קְלִיק". מִכְסֵה הַתֵּבָה נִפְתָּח.

אוֹר מְסַנְוֵר יָצָא מִבִּפְנִים! יְלָדִים, שָׁם בְּתוֹךְ הַקֻּפְסָה, נָצְצוּ אַלְפֵי מַטְבְּעוֹת זָהָב רְטוּבִים, עִם אַבְנֵי יְקָר שֶׁבָּרְקוּ בְּצִבְעֵי אָדוֹם, יָרֹק וְכָחוֹל כְּמוֹ נִיצוֹצוֹת אֵשׁ!

הַמַּטְרוֹנִיתָא נִשְׁאֲרָה חֲסְרַת מִלִּים. הִיא תָּפְסָה שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא, עֲרֵבוֹ הַנֶּאֱמָן שֶׁל רַבִּי עֲקִיבָא, אָכֵן סִלֵּק אֶת הַחוֹב בַּזְּמַן, וַאֲפִלּוּ בְּרֶוַח עָצוּם, הֲרֵי הַסְּכוּם שֶׁל הַקֻּפְסָה הָיָה הַרְבֵּה יוֹתֵר מִמַּה שֶׁנָּתְנָה כְּהַלְוָאָה! הִיא מָשְׁכָה אֶת הַתֵּבָה לְבֵיתָהּ בְּהִתְרַגְּשׁוּת רַבָּה.

קִדּוּשׁ שֵׁם שָׁמַיִם

עָבְרוּ כַּמָּה יָמִים. הַקָּדוֹשׁ בָּרוּךְ הוּא שָׁלַח רְפוּאָה שְׁלֵמָה לְרַבִּי עֲקִיבָא. גּוּפוֹ הִתְחַזֵּק, הַחֹם עָזַב אוֹתוֹ, וְהוּא קָם עַל רַגְלָיו. הַדָּבָר הָרִאשׁוֹן שֶׁעָשָׂה בְּהַחְלָמָתוֹ – הוּא לֹא הָלַךְ לָנוּחַ אוֹ לֶאֱכוֹל, אֶלָּא לָקַח אֶת צְרוֹר הַכֶּסֶף שֶׁהֵכִין לְתַשְׁלוּם הַחוֹב, עָלָה עַל הַחֲמוֹר, וְיָצָא לְבֵיתָהּ שֶׁל הַמַּטְרוֹנִיתָא עַל הַחוֹף.

כְּשֶׁהִגִּיעַ לְאַרְמוֹנָהּ, נִגַּשׁ בְּפָנִים קְצָת נְפוּלוֹת.

"אֲנִי מְבַקֵּשׁ סְלִיחָה וּמְחִילָה," אָמַר רַבִּי עֲקִיבָא מִתּוֹךְ בּוּשָׁה וְיִרְאַת שָׁמַיִם. "אָנָּא, אַל תִּכְעֲסִי עָלַי שֶׁלֹּא בָּאתִי בַּזְּמַן. מַחֲלָה קָשָׁה מָנְעָה מִמֶּנִּי לְהַגִּיעַ. הִנֵּה הַכֶּסֶף שֶׁלָּךְ שֶׁהֵכַנְתִּי מֵרֹאשׁ עַד לַפְּרוּטָה הָאַחֲרוֹנָה!".

לְתַדְהַמְתּוֹ, הַמַּטְרוֹנִיתָא לֹא כָּעֲסָה כְּלָל. לְהֶפֶךְ, הִיא צָחֲקָה מִשִּׂמְחָה וְדִמְעָה שֶׁל אוֹר עָמְדָה בְּעֵינָהּ.

"הַחְזֵר אֶת כַּסְפְּךָ לְכִיסְךָ, אִישׁ אֱלֹקִים קָדוֹשׁ!" חִיְּכָה וְאָמְרָה. "אַתָּה לֹא חַיָּב לִי מְאוּמָה! אֲנִי תָּבַעְתִּי אֶת הָעֲרֵבִים שֶׁלְּךָ בַּזְּמַן, וְהֵם שִׁלְּמוּ לִי מִיָּד, וַאֲפִלּוּ הוֹסִיפוּ לִי מַתָּנוֹת רַבּוֹת. הַקָּדוֹשׁ בָּרוּךְ הוּא, אֱלֹקֵי יִשְׂרָאֵל, שָׁלַח לִי לְעֵבֶר הַחוֹף עַל גַּבֵּי גַּלֵּי הַיָּם, קֻפְסַת אוֹצָר מְלֵאָה הָמוֹן זָהָב!".

וְלֹא מַסְפִּיק שֶׁמָּחֲלָה מִיָּד עַל הַחוֹב, הִיא עוֹד הוֹלִיכָה אוֹתוֹ לְתֵבַת הָאוֹצָר. הִיא נָתְנָה לְרַבִּי עֲקִיבָא אֶת כָּל תּוֹסֶפֶת הַכֶּסֶף שֶׁנּוֹתְרָה, וּבִקְּשָׁה לְכַבֵּד אוֹתוֹ בְּמַתָּנוֹת רַבּוֹת! בְּרֶגַע אֶחָד, נַעֲשָׂה רַבִּי עֲקִיבָא אָדָם עָשִׁיר עוֹד יוֹתֵר, וְיָכוֹל הָיָה לִתְמֹךְ בְּאַלְפֵי תַּלְמִידָיו בְּשֶׁפַע גָּדוֹל עֲבוּר לִמּוּד הַתּוֹרָה!

שָׁם בַּחֶדֶר, מוּל הָאִשָּׁה הַנּוֹכְרִיָּה, קִדֵּשׁ רַבִּי עֲקִיבָא שֵׁם שָׁמַיִם. כֻּלָּם רָאוּ שֶׁמִּי שֶׁדּוֹאֵג לַתּוֹרָה, וּמִי שֶׁעוֹמֵד בַּהַבְטָחוֹת שֶׁלּוֹ – בּוֹרֵא עוֹלָם וְכָל הַטֶּבַע (אֲפִלּוּ גַּלֵּי הַיָּם) נִרְתָּמִים לַעֲזֹר לוֹ!

א. שְׁאֵלַת פְּתִיחָה (לְעוֹרֵר סַקְרָנוּת):

"יְלָדִים, אִם חָבֵר מְבַקֵּשׁ לִלְווֹת מִכֶּם סְכוּם קָטָן עֲבוּר כְּרִיךְ מֵהַמַּזְנוֹן, הֲרֵי תַּסְכִּימוּ. אֲבָל מָה אִם מִישֶׁהוּ יְבַקֵּשׁ מִכֶּם מִילְיוֹן שְׁקָלִים? כַּמּוּבָן שֶׁתִּדְרְשׁוּ תְּעוּדוֹת וַעֲרֵבִים וּבִטְחוֹנוֹת עֲנָקִיִּים.

הַיּוֹם נִלְמַד עַל צַדִּיק שֶׁבִּקֵּשׁ סְכוּם עָצוּם, וּכְשֶׁבִּקְשׁוּ מִמֶּנּוּ עָרֵב (מִי שֶׁיְּשַׁלֵּם אִם תִּהְיֶה בְּעָיָה), הַתְּשׁוּבָה שֶׁהוּא נָתַן הָיְתָה כָּל כָּךְ עוֹצְמָתִית – עַד שֶׁהִיא זִעְזְעָה אֶת גַּלֵּי הַיָּם עַצְמָם!"

ב. הֶעָרוֹת בִּצּוּעַ וּסְטוֹרִיטֶלִינְג (לַמּוֹרֶה):

ג. הַמְלָצוֹת לַמַּצֶּגֶת (4 שְׁקוּפִיּוֹת קְדוֹשׁוֹת וּצְנוּעוֹת):

ד. פְּעִילוּת מְסַכֶּמֶת מֻמְלֶצֶת:

מִשְׂחַק מַצָּבִים "עוֹמְדִים בַּמִּלָּה": הַמּוֹרֶה כּוֹתֵב עַל פְּתָקִים מַצָּבִים מֵהַיּוֹם־יוֹם, מְחַלֵּק לִשְׁלִישִׁיּוֹת וְהַיְּלָדִים מְצִיגִים:

מַצָּב א': אִימָּא בִּקְּשָׁה לַעֲזֹר בְּסִדּוּר הַחֶדֶר. הִבְטַחְתִּי. אַחַר כָּךְ חָבֵר בָּא לִקְרֹא לִי לְשַׂחֵק.

מַצָּב ב': שָׁאַלְתִּי קַלְמָר לְיוֹמַיִם מֵחָבֵר. כָּעֵת הוּא צָרִיךְ אוֹתוֹ שׁוּב לְמִבְחָן בְּדִיּוּק שֶׁגַּם לִי יֵשׁ דַּף עֲבוֹדָה. אֵיךְ נָהַג רַבִּי עֲקִיבָא? (מָסַר נֶפֶשׁ שֶׁלֹּא תִּהְיֶה פְּגִיעָה בְּאֱמוּנַת הָאַחֵר).

מקורות: תַּלְמוּד בַּבְלִי, מַסֶּכֶת נְדָרִים דַּף נ' עַמּוּד ב' – תֵּאוּר אֵירוּעַ הַהַלְוָאָה וְקִדּוּשׁ שֵׁם שָׁמַיִם הַנַּעֲשָׂה עַל יְדֵי רַבִּי עֲקִיבָא בַּהַלְוָאָה מִן הַמַּטְרוֹנִיתָא וּבַעֲרֵבוּת ה' וְהַיָּם. תַּלְמוּד בַּבְלִי, מַסֶּכֶת עֲבוֹדָה זָרָה דַּף י' עַמּוּד ב' מָקוֹר רִקְעִי בְּפָנֶל הַלִּמּוּד הֶחִינוּכִי – סִדְרַת "כֹּה עָשׂוּ חֲכָמֵינוּ" (חֵלֶק ב', עמ' 32) וְהָעִבּוּד עַל "ה' עוֹנֶה לִתְפִלַּת נָכְרִיָּה". מַאֲמָרֵי חֲזַ"ל נוֹסָפִים שֶׁדָּנִים בַּעֲשִׁירוּתוֹ שֶׁל רַבִּי עֲקִיבָא לְאַחַר מַסַּע לִמּוּדוֹ מִתּוֹךְ עֲנִיּוּת (רַשִּׁ"י וְרַ"ן בַּמָּקוֹם).

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “להלן עיבוד חינוכי, רגשי ומרתק לסיפור המופלא על רבי עקיבא, העשירה והערבים (הקב"ה והים) .” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “הַמַּטְרוֹנִיתָא הִבִּיטָה בִּפְנֵי הַצַּדִּיק, שֶׁהָיוּ מְאִירוֹת בְּאוֹר מְיֻחָד.”?
ג. כיצד הגיב רבי עקיבא למצב המתואר: “מַחֲלָה חֲזָקָה וּכְאֵבִים עַזִּים תָּקְפוּ אֶת גּוּפוֹ הַקָּדוֹשׁ.”?
ד. מה השתנה בין “אַתָּה יוֹדֵעַ שֶׁעֲקִיבָא עַבְדְּךָ חָלָה, וְאֵינוֹ יָכוֹל לָבוֹא הַיּוֹם לְשַׁלֵּם לִי אֶת חֹבוֹ.” לבין “הַקָּדוֹשׁ בָּרוּךְ הוּא שָׁלַח רְפוּאָה שְׁלֵמָה לְרַבִּי עֲקִיבָא.”?
ה. כיצד הגיע הסיפור אל האירוע: “בְּרֶגַע אֶחָד, נַעֲשָׂה רַבִּי עֲקִיבָא אָדָם עָשִׁיר עוֹד יוֹתֵר, וְיָכוֹל הָיָה לִתְמֹךְ בְּאַלְפֵי תַּלְמִידָיו בְּשֶׁפַע גָּדוֹל עֲבוּר לִמּוּד…”?

האירוע הנתוןמה גרם לו?מה הייתה תוצאתו?

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “ה' עונה לתפילת נכרייה” על רבי עקיבא? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “אֲבָל גּוּפוֹ בָּגַד בּוֹ, וְהוּא פָּשׁוּט לֹא יָכוֹל הָיָה לָקוּם.”.
ג. אילו רבי עקיבא לא היה פועל כפי שמתואר ב־“הַקָּדוֹשׁ בָּרוּךְ הוּא שָׁלַח רְפוּאָה שְׁלֵמָה לְרַבִּי עֲקִיבָא.” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “בְּרֶגַע אֶחָד, נַעֲשָׂה רַבִּי עֲקִיבָא אָדָם עָשִׁיר עוֹד יוֹתֵר, וְיָכוֹל הָיָה לִתְמֹךְ בְּאַלְפֵי תַּלְמִידָיו בְּשֶׁפַע…” מחזק את המסר של הסיפור?
