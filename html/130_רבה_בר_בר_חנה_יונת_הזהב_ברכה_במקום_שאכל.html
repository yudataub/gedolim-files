יונת הזהב — ברכה במקום שאכל

דף עבודה בעקבות רבה בר בר חנה
שם: כיתה:

המעשה — הסיפור המלא

פְּתִיחָה: חֹם שֶׁל מִדְבָּר וְשַׁיָּרָה אֲרֻכָּה

יְלָדִים, סִגְרוּ אֶת הָעֵינַיִם וְנַסּוּ לְדַמְיֵן מִדְבָּר עָצוּם וְגָדוֹל. מִסָּבִיב יֵשׁ רַק דִּיּוּנוֹת שֶׁל חוֹל צָהֹב, הַשֶּׁמֶשׁ קוֹפַחַת מִלְמַעְלָה כְּמוֹ תַּנּוּר לוֹהֵט, וְהָאֲוִיר שֶׁנּוֹשְׁמִים הוּא יָבֵשׁ וְחַם. רֵיחַ שֶׁל אָבָק מְמַלֵּא אֶת הָאַף.

בְּתוֹךְ הַמִּדְבָּר הַזֶּה פּוֹסַעַת שַׁיֶּרֶת גְּמַלִּים אֲרֻכָּה. צְלִיל הַפַּעֲמוֹנִים שֶׁל הַגְּמַלִּים מְצַלְצֵל אִטִּי: צִיל... צִיל... עִם כָּל צַעַד שֶׁל רַגְלֵיהֶם הַגְּדוֹלוֹת בַּחוֹל. בְּרֹאשׁ הַשַּׁיָּרָה צוֹעֵד הַטַּיְּיעָא – סוֹחֵר עֲרָבִי שֶׁמַּכִּיר אֶת כָּל סוֹדוֹת הַמִּדְבָּר, אֵיפֹה יֵשׁ מַיִם וְאֵיפֹה טָמוּן סַכָּנָה.

וּבְאֶחָד הַגְּמַלִּים, מְכֻסֶּה מֵהַשֶּׁמֶשׁ, רָכוּב אָדָם גָּדוֹל וְקָדוֹשׁ – זֶהוּ הָאָמוֹרָא הַצַּדִּיק, רַבָּה בַּר בַּר חָנָה .

רַבָּה בַּר בַּר חָנָה הָיָה אִישׁ רַב עֲלִילוֹת, חָכָם עָצוּם, שֶׁאַף פַּעַם לֹא הִפְסִיק לִלְמֹד תּוֹרָה, גַּם כַּאֲשֶׁר נָסַע לְמֶרְחַקִּים עַל גַּבֵּי גְּמַלִּים וָאֳנִיּוֹת.

הָאֲרוּחָה בְּצֵל הַסְּלָעִים

לְהַלֵּךְ בַּמִּדְבָּר מְעַיֵּף מְאֹד. בִּשְׁעַת צָהֳרַיִם חַמָּה הִכְרִיז מַנְהִיג הַשַּׁיָּרָה בְּקוֹל צָרוּד: "עוֹצְרִים לִמְנוּחָה!". הַגְּמַלִּים כָּרְעוּ עַל הַחוֹל בַּאֲנָחָה. כֻּלָּם חִפְּשׂוּ פִּנַּת צֵל כְּדֵי לָנוּחַ וְלֶאֱכֹל מַשֶּׁהוּ.

גַּם רַבָּה בַּר בַּר חָנָה יָרַד מִגְּמַלּוֹ. הוּא מָצָא סֶלַע מִדְבָּרִי שֶׁנָּתַן מְעַט צֵל דַּק, יָשַׁב בְּנַחַת, נָטַל אֶת יָדָיו בִּמְעַט הַמַּיִם הַצּוֹנְנִים שֶׁשָּׁמַר בַּנֹּאד הָעוֹר שֶׁלּוֹ, וּבֵרַךְ בְּקוֹל שָׁקֵט: "הַמּוֹצִיא לֶחֶם מִן הָאָרֶץ". טַעַם הַלֶּחֶם הַיָּבֵשׁ הָיָה מָתוֹק מִדְּבַשׁ לְאָדָם שֶׁרָעֵב וְעָיֵף כָּל כָּךְ בַּדֶּרֶךְ הַקָּשָׁה.

הַזְּמַן בַּמִּדְבָּר קָצָר וְהַסַּכָּנוֹת מְרֻבּוֹת. עוֹד לִפְנֵי שֶׁהִסְפִּיק הַצַּדִּיק לְסַיֵּם בְּנַחַת, נִשְׁמְעָה צַעֲקַת הַסּוֹחֵר: "קָדִימָה! כֻּלָּם לַעֲלוֹת! הַשֶּׁמֶשׁ תִּשְׁקַע בְּקָרוֹב, אֲנַחְנוּ חַיָּבִים לְהַגִּיעַ לְתַחֲנַת הַמַּיִם הַבָּאָה!".

הָיְתָה בֶּהָלָה וּמְהִירוּת. קוֹלוֹת גְּמַלִּים, צְעָקוֹת בַּעֲרָבִית, שַׂקִּים נִקְשָׁרִים מַהֵר לָאֻכָּפִים. רַבָּה הִזְדָּרֵז, תָּפַס אֶת שַׂקּוֹ, טִפֵּס עַל הַגָּמָל, וְהַשַּׁיָּרָה יָצְאָה לַדֶּרֶךְ בִּדְהָרָה, מַשְׁאִירָה אַחֲרֶיהָ עֲנַן עָשָׁן אָפֹר וְסָמִיךְ.

הַהִזָּכְרוּת – סְעָרָה בַּלֵּב

הַגְּמַלִּים צָעֲדוּ שָׁעָה אֲרֻכָּה. קִילוֹמֶטֶר אַחַר קִילוֹמֶטֶר שֶׁל חוֹלוֹת שֶׁנִּרְאִים אוֹתוֹ הַדָּבָר. וְאָז, כְּמַכַּת בָּרָק, נֶחְרַד רַבָּה בַּר בַּר חָנָה לִמְקוֹמוֹ עַל הַגָּמָל!

אוֹי וַאֲבוֹי! הוּא אָכַל לֶחֶם לָשֹׂבַע... אֲבָל מֵרֹב הַמְּהִירוּת שֶׁל יְצִיאַת הַשַּׁיָּרָה... הוּא שָׁכַח לְבָרֵךְ בִּרְכַּת הַמָּזוֹן! הַצַּדִּיק נָתַן מַכָּה רַכָּה עַל מִצְחוֹ בְּיָדוֹ. "אֵיךְ שָׁכַחְתִּי לְהוֹדוֹת לְבוֹרֵא עוֹלָם עַל הַלֶּחֶם שֶׁנָּתַן לִי?"

רַבָּה יָדַע אֶת הַהֲלָכָה הֵיטֵב. אִם אָדָם שָׁכַח בִּרְכַּת הַמָּזוֹן וְנִזְכַּר אַחֲרֵי שֶׁעָזַב אֶת הַמָּקוֹם – מָה עָלָיו לַעֲשׂוֹת? בֵּית הִלֵּל אוֹמְרִים שֶׁאֶפְשָׁר לְבָרֵךְ בַּמָּקוֹם שֶׁבּוֹ נִזְכַּרְתָּ. אַךְ בֵּית שַׁמַּאי – מַחְמִירִים יוֹתֵר. בֵּית שַׁמַּאי דּוֹרְשִׁים: "חֲזֹר לַמָּקוֹם אֲשֶׁר אָכַלְתָּ בּוֹ, וּבָרֵךְ שָׁם!". רַבָּה רָצָה לְקַיֵּם אֶת הַמִּצְוָה בְּאֹפֶן הַמְהֻדָּר בְּיוֹתֵר, כְּפִי פְּסַק הַדִּין שֶׁל בֵּית שַׁמַּאי. לַחְזֹר אֲחוֹרָה!

אֲבָל רֶגַע... בּוֹאוּ נַחְשֹׁב יַחַד. הַשַּׁיָּרָה כְּבָר עָבְרָה מֶרְחָק גָּדוֹל. הַמִּדְבָּר מָלֵא בְּחַיּוֹת טֶרֶף וּבְשׁוֹדְדֵים. אִם רַבָּה יִפְנֶה עַכְשָׁו לְסוֹחֵר הָעֲרָבִי, הַטַּיְּיעָא, וְיֹאמַר לוֹ: "סְלִיחָה, אֲנִי חַיָּב לַחְזֹר כַּמָּה קִילוֹמֶטְרִים אֲחוֹרָה כִּי שָׁכַחְתִּי לוֹמַר כַּמָּה מִלִּים שֶׁל תְּפִלָּה", מָה יַעֲשֶׂה הַסּוֹחֵר? הוּא בְּוַדַּאי יִצְחַק עָלָיו! "תִּתְפַּלֵּל פֹּה! הַמִּדְבָּר שֶׁלְּךָ אוֹתוֹ חוֹל מִכָּל כִּוּוּן! אֲנִי לֹא עוֹצֵר אֶת הַשַּׁיָּרָה בִּשְׁבִיל מִלִּים!".

"שָׁכַחְתִּי יוֹנַת זָהָב!"

רַבָּה בַּר בַּר חָנָה יָדַע שֶׁאִם יֹאמַר אֶת הָאֱמֶת לַגּוֹי, הוּא לֹא יָבִין. עֲבוּר אָדָם שֶׁלֹּא שׁוֹמֵר תּוֹרָה, תְּפִלָּה הִיא "רַק דִּבּוּר בַּאֲוִיר". אֲבָל עֲבוּר הַצַּדִּיק, הַמִּצְוָה וְהַבְּרָכָה הֵם אוֹצָר יָקָר מִפָּז!

לָכֵן, רַבָּה בַּר בַּר חָנָה פָּנָה אֶל מַנְהִיג הַשַּׁיָּרָה וְאָמַר לוֹ בְּקוֹל דָּחוּף: "עֲצֹר! אֲנִי חַיָּב לַחְזֹר לַמָּקוֹם שֶׁבּוֹ יָשַׁבְנוּ תַּחַת הַסֶּלַע! שָׁכַחְתִּי שָׁם מַשֶּׁהוּ יָקָר מְאֹד... שָׁכַחְתִּי... יוֹנָה עֲשׂוּיָה מִזָּהָב! ".

כְּשֶׁשָּׁמַע הַסּוֹחֵר אֶת הַמִּלָּה "זָהָב", הָעֵינַיִם שֶׁלּוֹ נִפְעֲרוּ בִּפְלִיאָה. זָהָב טָהוֹר? בְּצוּרַת יוֹנָה? בְּוַדַּאי שֶׁזֶּה יָקָר! "הַבְטַח לִי שֶׁתַּחְזֹר מַהֵר," אָמַר הַסּוֹחֵר, "אֲנַחְנוּ נַמְתִּין לְךָ מְעַט, אֲבָל אַל תִּתְעַכֵּב!".

לָמָּה רַבָּה בַּר בַּר חָנָה אָמַר "יוֹנַת זָהָב"? כִּי לְאָדָם חָמְרִי – זָהָב הִיא הַסִּבָּה הַיְחִידָה לַחְזֹר אֲחוֹרָה בְּסַכָּנָה. אַךְ בְּלִבּוֹ יָדַע רַבָּה הַצַּדִּיק, שֶׁעַם יִשְׂרָאֵל מְשׁוּלִים לְיוֹנָה ("יוֹנָתִי בְּחַגְוֵי הַסֶּלַע"), וְהַבְּרָכָה שֶׁל יְהוּדִי אֶל ה' טְהוֹרָה וִיקָרָה יוֹתֵר מִכָּל זָהָב שֶׁבָּעוֹלָם!

הַתְּפִלָּה הַבּוֹדֵדָה עַל הַחוֹל

רַבָּה סוֹבֵב אֶת גְּמַלּוֹ, וְרָכַב לְבַדּוֹ, רָחוֹק מֵהַשַּׁיָּרָה. רַק הוּא, הַשֶּׁמֶשׁ, הַשָּׁמַיִם, וְרֵיחַ הַחוֹל שֶׁל הַמִּדְבָּר. הַדֶּרֶךְ חֲזָרָה הָיְתָה אֲרֻכָּה וּמַלְאֲת פַּחַד, שֶׁמָּא יָבוֹא שׁוֹדֵד... אֲבָל רַבָּה לֹא חָשַׁב עַל עַצְמוֹ, הַנְּשָׁמָה שֶׁלּוֹ שָׁאֲפָה לְהוֹדוֹת לַה' כַּהֲלָכָה.

אַחֲרֵי זְמַן נִכָּר, הוּא חָזַר לְאוֹתוֹ סֶלַע מְדֻיָּק שֶׁבּוֹ יָשַׁב בַּצָּהֳרַיִם.

רַבָּה יָרַד מֵהַגָּמָל. עַכְשָׁו אֵין לַחַץ. אֵין עֲרָבִים צוֹעֲקִים. מִסָּבִיב – שֶׁקֶט מֻחְלָט שֶׁל מִדְבָּר אֵינְסוֹפִי.

הַצַּדִּיק כָּרַע עַל בִּרְכָּיו בַּחוֹל הַצּוֹרֵב, עָצַם אֶת עֵינָיו בְּכַוָּנָה עֲצוּמָה, וְהֵחֵל לְבָרֵךְ אֶת בִּרְכַּת הַמָּזוֹן:

"בָּרוּךְ אַתָּה ה'... הַזָּן אֶת הָעוֹלָם כֻּלּוֹ בְּטוּבוֹ בְּחֵן בְּחֶסֶד וּבְרַחֲמִים!" – רַבָּה הִרְגִּישׁ מַמָּשׁ אֵיךְ ה' מַאֲכִיל כָּל יְצוּר וִיצוּר אֲפִלּוּ כָּאן, בַּמִּדְבָּר הַשּׁוֹמֵם.

"נוֹדֶה לְךָ ה' אֱלֹהֵינוּ... עַל בְּרִיתְךָ שֶׁחָתַמְתָּ בִּבְשָׂרֵנוּ, וְעַל תּוֹרָתְךָ שֶׁלִּמַּדְתָּנוּ!" – דְּמָעוֹת קְטַנּוֹת שֶׁל הוֹדָיָה וְשִׂמְחָה עָמְדוּ בְּקָצוֹת עֵינָיו. קוֹלוֹ נִשָּׂא בְּלֵב הַיְשִׁימוֹן הַגָּדוֹל, תְּפִלָּה זַכָּה שֶׁל אָדָם שֶׁמּוֹסֵר נַפְשׁוֹ עַל קוֹצוֹ שֶׁל יוֹד, כְּדֵי לַעֲשׂוֹת נַחַת רוּחַ לְבּוֹרְאוֹ. הוּא סִיֵּם אֶת הַבְּרָכָה מִתּוֹךְ אַהֲבַת ה' טְהוֹרָה.

הַהַשְׁגָּחָה וְהַפֶּלֶא

כַּאֲשֶׁר רַבָּה בַּר בַּר חָנָה פָּקַח אֶת עֵינָיו בְּסִיּוּם הַתְּפִלָּה, וּבָא לְהִתְרוֹמֵם, מַשֶּׁהוּ הִבְרִיק וְסִנְוֵר אֶת מַבָּטוֹ מִבֵּין גַּרְגְּרֵי הַחוֹל שֶׁלְּיַד רַגְלָיו.

הוּא מָצַמֵץ בְּעֵינָיו שׁוּב. "הַאִם הַשֶּׁמֶשׁ מְסַנְוֶרֶת אוֹתִי מֵהֶחָם?" חָשַׁב.

הוּא הִתְכּוֹפֵף. יָדוֹ נָגְעָה בְּמַשֶּׁהוּ קָשֶׁה וְכָבֵד.

יְלָדִים... זֶה לֹא הָיָה מִקְסַם שָׁוְא! שָׁם, בְּדִיּוּק בַּמָּקוֹם שֶׁבּוֹ כָּרַע לְבָרֵךְ, הִבְהִיקָה מֵהַחוֹל דְּמוּת מֻשְׁלֶמֶת שֶׁל יְצִירָה. דְּמוּת שֶׁל צִפּוֹר עֲשׂוּיָה זָהָב אֲמִתִּי וְטָהוֹר. זָהָב מָלֵא וְכָבֵד. יוֹנַת זָהָב הָיְתָה שָׁם!

מֵאַיִן הִגִּיעָה לְשָׁם יוֹנַת זָהָב בְּלֵב הַמִּדְבָּר שֶׁאִישׁ אֵינוֹ גָּר בּוֹ?!

רַבָּה הֵבִין בְּאוֹתוֹ הָרֶגַע אֶת הַחִבּוּק הֶעָצוּם שֶׁל הַהַשְׁגָּחָה הַפְּרָטִית. הוּא הֲרֵי אָמַר לַסּוֹחֵר "שכחתי יוֹנַת זָהָב" רַק כְּתֵרוּץ, בִּמְקוֹם לְהַגִּיד לוֹ אֶת סוֹד כַּוָּנָתוֹ לְבָרֵךְ כְּבֵית שַׁמַּאי. אֲבָל הַקָּדוֹשׁ בָּרוּךְ הוּא כִּבְיָכוֹל אָמַר מִשָּׁמַיִם: "רַבָּה בְּנִי הָאָהוּב, טָרַחְתָּ וְהִסְתַּכַּנְתָּ רַק כְּדֵי לְבָרֵךְ אוֹתִי כַּהֲלָכָה וּבְהִדּוּר? הַמִּצְוָה שֶׁלְּךָ כָּל כָּךְ יְקָרָה לִי! הִיא בֶּאֱמֶת יוֹנַת זָהָב! קַח, קַח אֶת תַּגְמוּלְךָ, גַּם בָּעוֹלָם הַזֶּה!"

רַבָּה תָּפַס אֶת שְׁתֵּי הַכְּנָפַיִם הַזְּהֻבּוֹת שֶׁל הַיּוֹנָה, שָׂם אוֹתָהּ בְּשַׂקּוֹ מְלֵא שִׂמְחָה וְהוֹדָיָה לַה', רָכַב חֲזָרָה לַשַּׁיָּרָה עִם חִיּוּךְ רָחָב, וְהֶרְאָה לַסּוֹחֵר שֶׁלֹּא שִׁקֵּר. הוּא בֶּאֱמֶת מָצָא שָׁם אֶת "יוֹנַת הַזָּהָב" – הַבְּרָכָה וְהַשָּׂכָר הֶעָצוּם עַל דִּקְדּוּק מִצְווֹת הַתּוֹרָה.

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “יְלָדִים, סִגְרוּ אֶת הָעֵינַיִם וְנַסּוּ לְדַמְיֵן מִדְבָּר עָצוּם וְגָדוֹל.” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “טַעַם הַלֶּחֶם הַיָּבֵשׁ הָיָה מָתוֹק מִדְּבַשׁ לְאָדָם שֶׁרָעֵב וְעָיֵף כָּל כָּךְ בַּדֶּרֶךְ הַקָּשָׁה.”?
ג. כיצד הגיב רבה בר בר חנה למצב המתואר: “בֵּית הִלֵּל אוֹמְרִים שֶׁאֶפְשָׁר לְבָרֵךְ בַּמָּקוֹם שֶׁבּוֹ נִזְכַּרְתָּ.”?
ד. מה השתנה בין “כְּשֶׁשָּׁמַע הַסּוֹחֵר אֶת הַמִּלָּה "זָהָב", הָעֵינַיִם שֶׁלּוֹ נִפְעֲרוּ בִּפְלִיאָה.” לבין “אַחֲרֵי זְמַן נִכָּר, הוּא חָזַר לְאוֹתוֹ סֶלַע מְדֻיָּק שֶׁבּוֹ יָשַׁב בַּצָּהֳרַיִם.”?
ה. כיצד הגיע הסיפור אל האירוע: “שָׁם, בְּדִיּוּק בַּמָּקוֹם שֶׁבּוֹ כָּרַע לְבָרֵךְ, הִבְהִיקָה מֵהַחוֹל דְּמוּת מֻשְׁלֶמֶת שֶׁל יְצִירָה.”?

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “יונת הזהב — ברכה במקום שאכל” על רבה בר בר חנה? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “עֲבוּר אָדָם שֶׁלֹּא שׁוֹמֵר תּוֹרָה, תְּפִלָּה הִיא "רַק דִּבּוּר בַּאֲוִיר".”.
ג. אילו רבה בר בר חנה לא היה פועל כפי שמתואר ב־“אַחֲרֵי זְמַן נִכָּר, הוּא חָזַר לְאוֹתוֹ סֶלַע מְדֻיָּק שֶׁבּוֹ יָשַׁב בַּצָּהֳרַיִם.” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “שָׁם, בְּדִיּוּק בַּמָּקוֹם שֶׁבּוֹ כָּרַע לְבָרֵךְ, הִבְהִיקָה מֵהַחוֹל דְּמוּת מֻשְׁלֶמֶת שֶׁל יְצִירָה.” מחזק את המסר של הסיפור?

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “בֵּית הִלֵּל אוֹמְרִים שֶׁאֶפְשָׁר לְבָרֵךְ בַּמָּקוֹם שֶׁבּוֹ נִזְכַּרְתָּ.”.
מכתב: כתבו לרבה בר בר חנה בעקבות “רַבָּה סוֹבֵב אֶת גְּמַלּוֹ, וְרָכַב לְבַדּוֹ, רָחוֹק מֵהַשַּׁיָּרָה.”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “עַל בְּרִיתְךָ שֶׁחָתַמְתָּ בִּבְשָׂרֵנוּ, וְעַל תּוֹרָתְךָ שֶׁלִּמַּדְתָּנוּ!" – דְּמָעוֹת קְטַנּוֹת שֶׁל הוֹדָיָה וְשִׂמְחָה…”.
דיווח: כתבו ידיעה קצרה שכותרתה “יונת הזהב — ברכה במקום שאכל”.

המחזה

רגע מן הסיפורהדמויותמה יאמרו או יעשו?
קוֹלוֹת גְּמַלִּים, צְעָקוֹת בַּעֲרָבִית, שַׂקִּים נִקְשָׁרִים מַהֵר לָאֻכָּפִים.
אַחֲרֵי זְמַן נִכָּר, הוּא חָזַר לְאוֹתוֹ סֶלַע מְדֻיָּק שֶׁבּוֹ יָשַׁב בַּצָּהֳרַיִם.
