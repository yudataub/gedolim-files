המגיפה של תלמידיו

דף עבודה בעקבות רבי עקיבא
שם: כיתה:

המעשה — הסיפור המלא

פְּתִיחָה: קוֹלוֹת שֶׁל תּוֹרָה כְּגַלֵּי הַיָּם

יְלָדִים, נַסּוּ לְדַמְיֵן אֶת הָרַעַשׁ הֲכִי גָּדוֹל שֶׁשְּׁמַעְתֶּם מֵעוֹדְכֶם. מַפַּל מַיִם אַדִּיר? מָטוֹס עֲנָק שֶׁמַּמְרִיא?

עַכְשָׁו תְּדַמְיְנוּ מַשֶּׁהוּ חָזָק יוֹתֵר, אֲבָל מָתוֹק מִדְּבַשׁ: עֶשְׂרִים וְאַרְבָּעָה אֶלֶף כִּסְאוֹת, עֶשְׂרִים וְאַרְבָּעָה אֶלֶף סְפָרִים, עֶשְׂרִים וְאַרְבָּעָה אֶלֶף בַּחוּרִים וַאֲנָשִׁים עֲמֵלִים, שֶׁיּוֹשְׁבִים יַחַד וְלוֹמְדִים תּוֹרָה!

זֶה הָיָה קוֹלָהּ שֶׁל הַיְשִׁיבָה שֶׁל גְּדוֹל יִשְׂרָאֵל, רַבִּי עֲקִיבָא.

רַבִּי עֲקִיבָא הַצַּדִּיק יָשַׁב בְּרֹאשׁ בֵּית הַמִּדְרָשׁ הָעֲנָק, וְהַלֵּב שֶׁלּוֹ רָחַב מִשִּׂמְחָה. מֵרְחוֹק אֶפְשָׁר הָיָה לְהָרִיחַ אֶת רֵיחָן הַטּוֹב שֶׁל מְגִלּוֹת הַקְּלָף, וְלִשְׁמֹעַ אֶת זִמְזוּם הַלִּמּוּד שֶׁנִּשְׁמַע מַמָּשׁ כְּמוֹ רַעַשׁ גַּלֵּי הַיָּם שֶׁמִּתְנַפְּצִים אֶל הַחוֹף. מִכָּל קְצוֹת אֶרֶץ יִשְׂרָאֵל בָּאוּ לִלְמֹד מִפִּיו – מִמָּקוֹם שֶׁנִּקְרָא "גַּבַּת" וְעַד מָקוֹם שֶׁנִּקְרָא "אַנְטִיפְרַס". הוּא יָשַׁב וְלִמֵּד אוֹתָם אֶת דִּבְרֵי ה', כֵּיצַד מְקַיְּמִים מִצְווֹת, אֶת סוֹדוֹת הַתּוֹרָה הַקְּדוֹשָׁה. נִרְאֶה הָיָה שֶׁאֵין דָּבָר חָזָק מִזֶּה. תּוֹרַת יִשְׂרָאֵל הָיְתָה בְּשִׂיא תִּפְאַרְתָּהּ. הָעוֹלָם הָיָה מָלֵא בְּאוֹר.

הַשֶּׁבֶר: הָעוֹלָם הָיָה שָׁמֵם

אֲבָל אָז, הִגִּיעוּ הַיָּמִים שֶׁבֵּין חַג הַפֶּסַח לְחַג הַשָּׁבוּעוֹת. חַג הָעֲצֶרֶת.

בְּבֵית הַמִּדְרָשׁ הִתְחִילָה לִשְׂרֹר אֲוִירָה כְּבֵדָה וְעַצְוּבָה. תַּלְמִידִים הֵחֵלּוּ לַחְלוֹת בְּמַחֲלָה קָשָׁה הַנִּקְרֵאת 'אַסְכָּרָה' – מַחֲלַת גָּרוֹן נְגִיפִית וּמַכְאִיבָה שֶׁחָנְקָה אֶת גְּרוֹנָם.

בַּיּוֹם הָרִאשׁוֹן חָלוּ מֵאוֹת. בַּיּוֹם הַשֵּׁנִי אֲלָפִים. סַפְסְלֵי הָעֵץ בְּבֵית הַמִּדְרָשׁ נִשְׁאֲרוּ רֵיקִים.

בִּמְקוֹם קוֹל שֶׁל שִׂמְחָה וְלִמּוּד, הֵחֵלּוּ לְהִשָּׁמַע קוֹלוֹת שֶׁל בֶּכִי וּקְבָרוֹת בָּרְחוֹבוֹת. יוֹם מְרַדֵּף יוֹם, וְהַמַּגֵּפָה אֵינָהּ פּוֹסֶקֶת! כָּל עֶשְׂרִים וְאַרְבָּעָה אֶלֶף הַתַּלְמִידִים, הַצָּבָא הָרוּחָנִי הַגָּדוֹל בְּיוֹתֵר שֶׁל עַם יִשְׂרָאֵל, מֵתוּ כֻּלָּם.

יְלָדִים, נַסּוּ לַחְשֹׁב עַל הַשֶּׁקֶט שֶׁהָיָה שָׁם. אֵיזֶה חֹשֶׁךְ.

הַגְּמָרָא מִשְׁתַּמֶּשֶׁת בְּמִלִּים כּוֹאֲבוֹת מְאֹד: "וְהָיָה הָעוֹלָם שָׁמֵם" . כְּאִלּוּ נִמְחַק מִמֶּנּוּ כָּל טַעַם לַחַיִּים. מִי יַמְשִׁיךְ לְלַמֵּד תּוֹרָה? מִי יַעֲבִיר אֶת הַתּוֹרָה שֶׁבְּעַל-פֶּה לַיְּלָדִים שֶׁיִּוָּלְדוּ מָחָר? הַכֹּל נֶהֱרַס!

וְלָמָּה עָשָׂה ה' אֶת הַדָּבָר הַנּוֹרָא הַזֶּה לַאֲנָשִׁים כָּל כָּךְ קְדוֹשִׁים שֶׁלּוֹמְדִים תּוֹרָה?

רַבִּי עֲקִיבָא, בְּחָכְמָתוֹ שֶׁל צַדִּיק, רָאָה אֶת הַשֹּׁרֶשׁ, אֶת הַלֵּב מִבִּפְנִים. חֲזַ"ל אוֹמְרִים: "מִפְּנֵי שֶׁלֹּא נָהֲגוּ כָּבוֹד זֶה בָּזֶה" . "הָיְתָה עֵינָם צָרָה בַּתּוֹרָה זֶה לָזֶה". כְּשֶׁתַּלְמִיד אֶחָד שָׁאַל שְׁאֵלָה יָפָה, הַתַּלְמִיד הַשֵּׁנִי לֹא שָׂמַח בִּשְׁבִילוֹ מִכָּל הַלֵּב. כְּשֶׁמִּישֶׁהוּ יָדַע לַעֲנוֹת תְּשׁוּבָה שֶׁהַשֵּׁנִי לֹא יָדַע, חֲבֵרוֹ לֹא נָתַן לוֹ בֶּאֱמֶת אֶת הַכָּבוֹד הָרָאוּי. תּוֹרָה הִיא דָּבָר שֶׁחַיָּב לָבוֹא עִם לֵב טָהוֹר. אִם מִי שֶׁלּוֹמֵד תּוֹרָה מְזַלְזֵל בֶּחָבֵר שֶׁלּוֹ – הַתּוֹרָה לֹא יְכוֹלָה לְהִשָּׁאֵר! הַקָּדוֹשׁ בָּרוּךְ הוּא בּוֹחֵן כְּלָיוֹת וָלֵב, וְהוּא תּוֹבֵעַ מִלּוֹמְדֵי הַתּוֹרָה אֶת שְׁלֵמוּת הַמִּדּוֹת.

הַקִּימָה מֵהֶעָפָר: הַמַּסָּע דָּרוֹמָה

דִּמְעָה חַמָּה יָרְדָה עַל לֶחְיוֹ הַמְקֻמֶּטֶת שֶׁל רַבִּי עֲקִיבָא. עֶשְׂרוֹת שָׁנִים שֶׁל עָמָל נֶעֶלְמוּ. כָּל אָדָם רָגִיל בַּמַּצָּב הַזֶּה הָיָה לוֹקֵחַ שְׂמִיכָה, מִתְכַּסֶּה, וְאוֹמֵר: "זֶהוּ. דַּי. אֲנִי מֵרִים יָדַיִם. אֲנִי זָקֵן מִדַּי לְהַתְחִיל מֵחָדָשׁ".

אֲבָל לֹא רַבִּי עֲקִיבָא.

רַבִּי עֲקִיבָא הֵבִין שֶׁהַתּוֹרָה גְּדוֹלָה מֵהַכֹּל. אִם הַתּוֹרָה חָסֵרָה – עַם יִשְׂרָאֵל לֹא יָכוֹל לִנְשֹׁם!

הוּא קָם מֵהָרִצְפָּה הַקָּרָה שֶׁל בֵּית הַמִּדְרָשׁ. הוּא לָקַח אֶת מַקְלוֹ. רָשַׁם לְפָנָיו אֶת אֲוִיר הַדָּרוֹם הַחַם, וְיָצָא לַדֶּרֶךְ.

הוּא הָלַךְ לְחַפֵּשׂ תַּלְמִידִים חֲדָשִׁים בַּדָּרוֹם שֶׁל אֶרֶץ יִשְׂרָאֵל. לֹא עֶשְׂרִים וְאַרְבָּעָה אֶלֶף. מִסְפָּר שֶׁאֶפְשָׁר לִסְפֹּר עַל כַּף יָד אַחַת.

חֲמִשָּׁה תַּלְמִידִים בִּלְבַד!

וְאֵלּוּ שְׁמוֹתָם הַקְּדוֹשִׁים: רַבִּי מֵאִיר, רַבִּי יְהוּדָה, רַבִּי יוֹסֵי, רַבִּי שִׁמְעוֹן (בַּר יוֹחַאי), וְרַבִּי אֶלְעָזָר בֶּן שַׁמּוּעַ.

הַשִּׁעוּר הָרִאשׁוֹן וְהַכְּלָל הַגָּדוֹל

רַבִּי עֲקִיבָא יָשַׁב אִתָּם. חֲמִשָּׁה בַּחוּרִים צְעִירִים סָבִיב לְרַב זָקֵן מְאֹד הֶעָטוּף בְּטַלִּית. צְלִיל קוֹלוֹ הָיָה שָׁקֵט אַךְ חָזָק וְחָדַר אֶל תּוֹךְ הַנְּשָׁמָה שֶׁלָּהֶם.

לִפְנֵי שֶׁרַבִּי עֲקִיבָא הִתְחִיל לְלַמֵּד אוֹתָם אֶת הֲלָכוֹת הַתּוֹרָה הַקָּשׁוֹת, הוּא אָמַר לָהֶם מִשְׁפָּט צַמְרְמֹרֶת:

"בָּנַי," מִלְמֵל רַבִּי עֲקִיבָא, וְעֵינָיו נָצְצוּ מִדְּמָעוֹת, "הָרִאשׁוֹנִים – שְׁנֵים עָשָׂר אֶלֶף הַזּוּגוֹת – לֹא מֵתוּ אֶלָּא מִפְּנֵי שֶׁהָיְתָה עֵינָם צָרָה בַּתּוֹרָה זֶה לָזֶה! תְּנוּ דַּעְתְּכֶם שֶׁלֹּא תַּעֲשׂוּ כְּמַעֲשֵׂיהֶם שֶׁל הָרִאשׁוֹנִים!".

הוּא הִבִּיט עֲלֵיהֶם בְּמַבָּט חָדוּר אַהֲבָה. הוּא הֵבִין שֶׁאִם יֵשׁ מַשֶּׁהוּ שֶׁיַּחֲזִיק אֶת הַתּוֹרָה, זֶה הָאַהֲבָה בֵּין הַחֲבֵרִים שֶׁלּוֹמְדִים אוֹתָהּ.

וְאָז, הִכְרִיז רַבִּי עֲקִיבָא אֶת מִשְׁפַּט חַיָּיו. הַמִּשְׁפָּט שֶׁהָפַךְ לַכּוֹתֶרֶת שֶׁל תּוֹרַת יִשְׂרָאֵל עַד עֶצֶם הַיּוֹם הַזֶּה:

"וְאָהַבְתָּ לְרֵעֲךָ כָּמוֹךָ – זֶה כְּלָל גָּדוֹל בַּתּוֹרָה!"

הַחֲמִשָּׁה יָשְׁבוּ וְשָׁתוּ בְּצִמָּאוֹן עָצוּם אֶת הַתּוֹרָה. הֵם אָהֲבוּ אֶחָד אֶת הַשֵּׁנִי, כִּבְּדוּ כָּל קֻשְׁיָא, שָׂמְחוּ כְּשֶׁהַחָבֵר הֵבִין דָּבָר חָדָשׁ. וּמֵהַחֲמִשָּׁה הָאֵלֶּה יְלָדִים, שֶׁנִּרְאֶה כְּאִלּוּ זֶה מְעַט מְאֹד לְעֻמַּת עֶשְׂרִים וְאַרְבָּעָה אֶלֶף – "עָמְדוּ הֵם, וּמִלְּאוּ אֶת כָּל אֶרֶץ יִשְׂרָאֵל תּוֹרָה". כָּל מָה שֶׁאֲנַחְנוּ לוֹמְדִים הַיּוֹם בַּמִּשְׁנָה, בַּמִּדְרָשׁ וּבַגְּמָרָא – הַכֹּל מַגִּיעַ מִמְּסִירוּת הַנֶּפֶשׁ וְהָאַהֲבָה שֶׁל אוֹתָם חֲמִשָּׁה תַּלְמִידִים וְרַבָּם הַגָּדוֹל שֶׁלֹּא וִתֵּר – רַבִּי עֲקִיבָא!

מקורות: המקורות מופיעים בקובץ המקורי.

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “יְלָדִים, נַסּוּ לְדַמְיֵן אֶת הָרַעַשׁ הֲכִי גָּדוֹל שֶׁשְּׁמַעְתֶּם מֵעוֹדְכֶם.” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “אֲבָל אָז, הִגִּיעוּ הַיָּמִים שֶׁבֵּין חַג הַפֶּסַח לְחַג הַשָּׁבוּעוֹת.”?
ג. כיצד הגיב רבי עקיבא למצב המתואר: “חֲזַ"ל אוֹמְרִים: "מִפְּנֵי שֶׁלֹּא נָהֲגוּ כָּבוֹד זֶה בָּזֶה" .”?
ד. מה השתנה בין “דִּמְעָה חַמָּה יָרְדָה עַל לֶחְיוֹ הַמְקֻמֶּטֶת שֶׁל רַבִּי עֲקִיבָא.” לבין “חֲמִשָּׁה בַּחוּרִים צְעִירִים סָבִיב לְרַב זָקֵן מְאֹד הֶעָטוּף בְּטַלִּית.”?

3. מה היה ראשון?

כתבו ליד כל אירוע מספר מ־1 עד 5 לפי סדר התרחשותו בסיפור.

□ הַקָּדוֹשׁ בָּרוּךְ הוּא בּוֹחֵן כְּלָיוֹת וָלֵב, וְהוּא תּוֹבֵעַ מִלּוֹמְדֵי הַתּוֹרָה אֶת שְׁלֵמוּת הַמִּדּוֹת.
□ יְלָדִים, נַסּוּ לְדַמְיֵן אֶת הָרַעַשׁ הֲכִי גָּדוֹל שֶׁשְּׁמַעְתֶּם מֵעוֹדְכֶם.
□ כָּל מָה שֶׁאֲנַחְנוּ לוֹמְדִים הַיּוֹם בַּמִּשְׁנָה, בַּמִּדְרָשׁ וּבַגְּמָרָא – הַכֹּל מַגִּיעַ מִמְּסִירוּת הַנֶּפֶשׁ וְהָאַהֲבָה שֶׁל אוֹתָם חֲמִשָּׁה…
□ בִּמְקוֹם קוֹל שֶׁל שִׂמְחָה וְלִמּוּד, הֵחֵלּוּ לְהִשָּׁמַע קוֹלוֹת שֶׁל בֶּכִי וּקְבָרוֹת בָּרְחוֹבוֹת.
□ צְלִיל קוֹלוֹ הָיָה שָׁקֵט אַךְ חָזָק וְחָדַר אֶל תּוֹךְ הַנְּשָׁמָה שֶׁלָּהֶם.

לפני ואחרי

מה קרה לפני: “הַקָּדוֹשׁ בָּרוּךְ הוּא בּוֹחֵן כְּלָיוֹת וָלֵב, וְהוּא תּוֹבֵעַ מִלּוֹמְדֵי הַתּוֹרָה אֶת שְׁלֵמוּת הַמִּדּוֹת.”?
מה קרה אחרי: “צְלִיל קוֹלוֹ הָיָה שָׁקֵט אַךְ חָזָק וְחָדַר אֶל תּוֹךְ הַנְּשָׁמָה שֶׁלָּהֶם.”?

סיבה ותוצאה מתוך המעשה

האירוע הנתוןמה גרם לו?מה הייתה תוצאתו?
כָּל עֶשְׂרִים וְאַרְבָּעָה אֶלֶף הַתַּלְמִידִים, הַצָּבָא הָרוּחָנִי הַגָּדוֹל בְּיוֹתֵר שֶׁל עַם יִשְׂרָאֵל, מֵתוּ כֻּלָּם.
רָשַׁם לְפָנָיו אֶת אֲוִיר הַדָּרוֹם הַחַם, וְיָצָא לַדֶּרֶךְ.

4. הבנה והעמקה

המסר העולה מן הסיפור: "ואהבת לרעך כמוך – זה כלל גדול בתורה", ואף פעם לא מאוחר להתחיל מחדש, גם כשנראה שהכל אבד
א. מה מלמד המעשה “המגיפה של תלמידיו” על רבי עקיבא? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “אִם מִי שֶׁלּוֹמֵד תּוֹרָה מְזַלְזֵל בֶּחָבֵר שֶׁלּוֹ – הַתּוֹרָה לֹא יְכוֹלָה לְהִשָּׁאֵר!”.
ג. אילו רבי עקיבא לא היה פועל כפי שמתואר ב־“חֲמִשָּׁה בַּחוּרִים צְעִירִים סָבִיב לְרַב זָקֵן מְאֹד הֶעָטוּף בְּטַלִּית.” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “וְאָהַבְתָּ לְרֵעֲךָ כָּמוֹךָ – זֶה כְּלָל גָּדוֹל בַּתּוֹרָה!” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. יְלָדִים, נַסּוּ לְדַמְיֵן אֶת הָרַעַשׁ הֲכִי גָּדוֹל שֶׁשְּׁמַעְתֶּם…
2. אֲבָל אָז, הִגִּיעוּ הַיָּמִים שֶׁבֵּין חַג הַפֶּסַח לְחַג…
3. כְּשֶׁתַּלְמִיד אֶחָד שָׁאַל שְׁאֵלָה יָפָה, הַתַּלְמִיד הַשֵּׁנִי לֹא…
4. אִם הַתּוֹרָה חָסֵרָה – עַם יִשְׂרָאֵל לֹא יָכוֹל לִנְשֹׁם!

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “חֲזַ"ל אוֹמְרִים: "מִפְּנֵי שֶׁלֹּא נָהֲגוּ כָּבוֹד זֶה בָּזֶה" .”.
מכתב: כתבו לרבי עקיבא בעקבות “רָשַׁם לְפָנָיו אֶת אֲוִיר הַדָּרוֹם הַחַם, וְיָצָא לַדֶּרֶךְ.”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “בָּנַי," מִלְמֵל רַבִּי עֲקִיבָא, וְעֵינָיו נָצְצוּ מִדְּמָעוֹת, "הָרִאשׁוֹנִים – שְׁנֵים עָשָׂר אֶלֶף הַזּוּגוֹת – לֹא מֵתוּ…”.
דיווח: כתבו ידיעה קצרה שכותרתה “המגיפה של תלמידיו”.
