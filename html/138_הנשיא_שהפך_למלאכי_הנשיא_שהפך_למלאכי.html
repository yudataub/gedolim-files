הנשיא שהפך למלאכי

דף עבודה בעקבות הנשיא שהפך למלאכי
שם: כיתה:

המעשה — הסיפור המלא

הסיפור מנוקד במלואו, כנדרש.

🔹 שָׁלָב 1: בֵּרוּר מַקְדִּים וְנִתּוּחַ (לַמּוֹרֶה/לַמְסַפֵּר)

1. עֲנָוָה: אָדָם גָּדוֹל בֶּאֱמֶת לֹא מַחְשִׁיב אֶת עַצְמוֹ כְּלָל יַחַס לְצַעֲרוֹ שֶׁל הַזּוּלַת.

2. אִכְפַּתִּיּוּת וְחֶסֶד: צְדָקָה הִיא לֹא רַק לִזְרֹק מַטְבֵּעַ לְקֻפָּה, אֶלָּא לִדְאֹג לַהֲרָגָשָׁה שֶׁל הֶעָנִי, לַ"כָּבוֹד" שֶׁלּוֹ, וּלְשַׂמֵּחַ אֶת לִבּוֹ.

🔹 שָׁלָב 2: הַסִּפּוּר הַמָּלֵא – "הַנָּשִׂיא שֶׁהָפַךְ לְמַלְאֲכִי"

פְּתִיחָה: הָעָשִׁיר שֶׁאִבֵּד אֶת הַכֹּל יְלָדִים, דַּמְיְנוּ לָכֶם אָדָם שֶׁיֵּשׁ לוֹ הַכֹּל. בַּיִת עֲנָק, מְשָׁרְתִים, בְּגָדִים מִמֶּשִׁי וְזָהָב. כַּאֲשֶׁר הָאִישׁ הַזֶּה הָיָה יוֹצֵא לָרְחוֹב, הוּא לֹא הָיָה הוֹלֵךְ בָּרֶגֶל כְּמוֹ כֻּלָּם. הוּא הָיָה רוֹכֵב עַל סוּס מְפֹאָר וְחָזָק, וְלִפְנֵי הַסּוּס הָיָה רָץ עֶבֶד מְיֻחָד, שֶׁהָיָה קוֹרֵא בְּקוֹל גָּדוֹל: "פַּנּוּ דֶּרֶךְ! פַּנּוּ קָדִימָה! הָאָדוֹן הַנִּכְבָּד מַגִּיעַ!". הָאִישׁ הַזֶּה הָיָה רָגִיל שֶׁכֻּלָּם מְכַבְּדִים אוֹתוֹ וּמַבִּיטִים בּוֹ בְּהַעֲרָצָה.

אֲבָל... יוֹם אֶחָד, הַגַּלְגַּל הִתְהַפֵּךְ. שְׂרֵפָה גְּדוֹלָה, אוֹ אוּלַי שׁוֹדְדִים שֶׁהִתְנַפְּלוּ עַל אֳנִיּוֹתָיו – אֲנַחְנוּ לֹא מְדֻיָּק יוֹדְעִים – גָּרְמוּ לְכָךְ שֶׁהָאִישׁ אִבֵּד אֶת כָּל רְכוּשׁוֹ. בְּיוֹם אֶחָד, הֶעָשִׁיר הַגָּדוֹל הָפַךְ לְעָנִי מָרוּד. אֲבָל הַדָּבָר שֶׁהֲכִי הִכְאִיב לוֹ, לֹא הָיָה הָרָעָב. מָה שֶׁהֲכִי כָּאַב לוֹ הָיָה הַבּוּשָׁה. הוּא הָיָה "עָנִי בֶּן טוֹבִים" – אָדָם שֶׁרָגִיל לְכָבוֹד, וְעַכְשָׁו אֲנָשִׁים רָאוּ אוֹתוֹ הוֹלֵךְ בָּרֶגֶל בָּרְחוֹב, עָיֵף וּמֻשְׁפָּל. עֵינָיו תָּמִיד הָיוּ בָּרִצְפָּה, שֶׁמָּא מִישֶׁהוּ יַכִּיר אוֹתוֹ וִירַחֵם עָלָיו. הַלֵּב שֶׁלּוֹ הָיָה שָׁבוּר לִרְסִיסִים.

הִלֵּל הַזָּקֵן שׁוֹמֵעַ וּפוֹעֵל הַשְּׁמוּעָה עַל הָעָשִׁיר שֶׁיָּרַד מִנְּכָסָיו הִגִּיעָה לְאָזְנָיו שֶׁל נְשִׂיא יִשְׂרָאֵל – הִלֵּל הַזָּקֵן. הִלֵּל הָיָה גְּדוֹל הַדּוֹר, רַבָּן שֶׁל כָּל תַּלְמִידֵי הַחֲכָמִים. הִלֵּל הֵבִין שֶׁאִם הוּא רַק יִשְׁלַח לָאִישׁ הַזֶּה קְצָת כֶּסֶף לְלֶחֶם וַחֲתִיכַת בָּשָׂר, זֶה אָמְנָם יַשְׂבִּיעַ אֶת בִּטְנוֹ, אֲבָל זֶה לֹא יְרַפֵּא אֶת לִבּוֹ הַבּוֹכֶה. הַתּוֹרָה מְצַוָּה עָלֵינוּ לָתֵת לֶעָנִי "דֵּי מַחְסֹרוֹ אֲשֶׁר יֶחְסַר לוֹ" – צָרִיךְ לָתֵת לוֹ אֶת מָה שֶׁחָסֵר לַנֶּפֶשׁ שֶׁלּוֹ! מָה עָשָׂה הִלֵּל הַזָּקֵן? מִתּוֹךְ קֻפַּת הַצְּדָקָה שֶׁהָיְתָה בִּידֵי הַנָּשִׂיא, הוּא קָנָה סוּס אַבִּיר וְיָפֶה! הוּא שָׂכַר נַעַר שֶׁיִּהְיֶה עֶבֶד שֶׁרָץ לְפָנָיו בָּרְחוֹב, וּמָסַר אֶת הַכֹּל לְאוֹתוֹ עָנִי! תָּאֲרוּ לָכֶם אֶת הַשִּׂמְחָה שֶׁל הָאִישׁ! שׁוּב הוּא יָכוֹל לָצֵאת לָרְחוֹב מוּרָם מֵעַם. הוּא רוֹכֵב, וְהָעֶבֶד רָץ וְצוֹעֵק: "פַּנּוּ דֶּרֶךְ לָאָדוֹן הַנִּכְבָּד!". הַלֵּב שֶׁלּוֹ חָזַר לִפְעֹם בְּשִׂמְחָה.

הָעֶבֶד נֶעֱלָם! וְהִנֵּה, יוֹם אֶחָד קָרָה מַשֶּׁהוּ בִּלְתִּי צָפוּי. הָעֶבֶד שֶׁהָיָה אָמוּר לָרוּץ לִפְנֵי הַסּוּס, חָלָה אוֹ נֶעֱלָם, וְלֹא הִגִּיעַ לָעֲבוֹדָה. הָאָדוֹן הָעָנִי יָשַׁב עַל הַסּוּס, קָרוֹב לְבֵיתוֹ שֶׁל הִלֵּל הַזָּקֵן, וּדְמָעוֹת הֵחֵלּוּ לִזְלֹג מֵעֵינָיו. "מָה אֶעֱשֶׂה עַכְשָׁו?" הוּא לָחַשׁ. "אִם אֵרָאֶה בָּרְחוֹב רוֹכֵב לְבַדִּי, בְּלִי מְשָׁרֵת שֶׁמְּפַנֶּה לִי אֶת הַדֶּרֶךְ, כֻּלָּם יִצְחֲקוּ עָלַי. יַגִּידוּ שֶׁאֲנִי זִיּוּף. שׁוּב יְרַחֲמוּ עָלַי. אֲנִי לֹא מְסֻגָּל לָזוּז מִפֹּה!".

בְּאוֹתָהּ שָׁעָה, יָצָא הִלֵּל הַזָּקֵן מִבֵּיתוֹ, עָטוּף בְּגְלִימַת הַנָּשִׂיא שֶׁלּוֹ, מוּכָן לָלֶכֶת לְבֵית הַמִּדְרָשׁ כְּדֵי לְלַמֵּד תּוֹרָה לְאַלְפֵי תַּלְמִידִים שֶׁהִמְתִּינוּ לוֹ. הוּא רָאָה אֶת הָאִישׁ עַל הַסּוּס, עוֹמֵד וּבוֹכֶה. "מָה קָרָה בְּנִי?" שָׁאַל הִלֵּל בְּנַחַת. "אֵין... אֵין מִי שֶׁיָּרוּץ לְפָנַי הַיּוֹם..." יִבֵּב הָאִישׁ.

הַנָּשִׂיא שֶׁרָץ כְּמוֹ עֶבֶד יְלָדִים, אָדָם רָגִיל הָיָה אוֹמֵר לוֹ: "תַּגִּיד לִי, אַתָּה חָצוּף?! כְּבָר נָתַתִּי לְךָ סוּס, שִׁלַּמְתִּי עַל מְשָׁרֵת! אָז הַיּוֹם תִּרְכַּב לְבַד! מָה הַפִּנּוּק הַזֶּה?!". אֲבָל הִלֵּל הַזָּקֵן הָיָה מָלֵא בְּאַהֲבַת יִשְׂרָאֵל וַעֲנָוָה עֲצוּמָה. הוּא יָדַע שֶׁהָאִישׁ מִתְבַּיֵּשׁ!

"אַל תִּדְאַג," חִיֵּךְ הִלֵּל, אוֹתוֹ חִיּוּךְ טָהוֹר שֶׁאִתּוֹ תָּמִיד קֵרֵב אֶת הַבְּרִיּוֹת אֶל הַתּוֹרָה. לְפֶתַע, הִלֵּל הַזָּקֵן הֵרִים שׁוּלֵי גְּלִימָתוֹ כְּדֵי שֶׁלֹּא תַּפְרִיעַ לְרַגְלָיו. אִישׁ חָכָם וּמְבֻגָּר זֶה, מִשְׁפַּחַת דָּוִד הַמֶּלֶךְ, נֶעֱמַד מוּל הַסּוּס. הוּא מָחָא כַּפַּיִם וְהִתְחִיל... לָרוּץ!

"פַּנּוּ דֶּרֶךְ! פַּנּוּ דֶּרֶךְ לָאָדוֹן!" צָעַק הִלֵּל הַזָּקֵן כְּשֶׁהוּא רָץ בִּרְחוֹבוֹת הָעִיר, וְהַסּוּס שֶׁעָלָיו יוֹשֵׁב הֶעָנִי, צוֹעֵד בְּגַאֲוָה מֵאֲחוֹרָיו. הָאֲנָשִׁים בָּרְחוֹב שִׁפְשְׁפוּ אֶת עֵינֵיהֶם וְלֹא הֶאֱמִינוּ. "זֶה הַנָּשִׂיא! הִלֵּל הַזָּקֵן! מָה הוּא עוֹשֶׂה?!". אֲבָל הִלֵּל לֹא שָׂם לֵב. הַנְּשִׁימוֹת שֶׁלּוֹ כָּבְדוּ, אֲגַלֵּי זֵעָה צָצוּ עַל מִצְחוֹ הַקָּדוֹשׁ, הָאֲבָק בָּרְחוֹב עָלָה בָּאֲוִיר, וְהוּא לֹא עָצַר. כַּמָּה הוּא רָץ? מֵאָה מֶטְרִים? רְחוֹב אֶחָד? מַמָּשׁ לֹא. הַתַּלְמוּד מְסַפֵּר שֶׁהוּא רָץ לְפָנָיו שְׁלוֹשָׁה מִילִין! שֶׁזֶּה בְּעֵרֶךְ חֲמִשָּׁה קִילוֹמֶטְרִים! הֲלִיכָה שֶׁל כִּמְעַט שָׁעָה שְׁלֵמָה תַּחַת הַשֶּׁמֶשׁ, רַק כְּדֵי שֶׁהַיְּהוּדִי הֶעָנִי יַרְגִּישׁ שֶׁכְּבוֹדוֹ נִשְׁמָר וְהוּא אֵינוֹ מוּשְׁפָּל.

וְכָךְ, הִלֵּל הַזָּקֵן לִמֵּד אֶת כָּל עַם יִשְׂרָאֵל, שֶׁכָּבוֹד אֲמִתִּי לֹא מְקַבְּלִים כְּשֶׁיּוֹשְׁבִים עַל כִּסֵּא רָם וּמְחַלְּקִים הוֹרָאוֹת. כָּבוֹד אֲמִתִּי וּגְדֻלָּה אֲמִתִּית נִמְדָּדִים בַּעֲנָוָה, בַּיְּכֹלֶת לוֹמַר: "אֲנִי לֹא חָשׁוּב – הַלֵּב שֶׁל הַיְּהוּדִי שֶׁעוֹמֵד מוּלִי חָשׁוּב הַרְבֵּה יוֹתֵר".

🔹 שָׁלָב 3: חִבּוּר עֶרְכִּי – ME-WE-YOU

🔹 שָׁלָב 4: מַדְרִיךְ לַמּוֹרֶה / לַמְסַפֵּר (לְשִׁלּוּב בַּמַּצֶּגֶת וּבַשִּׁעוּר)

א. שְׁאֵלַת פְּתִיחָה (לְעוֹרֵר זִהוּי וְהִשְׁתַּתְּפוּת): "יְלָדִים, אִם הֱיִיתֶם רוֹאִים אֶת 'הָרַב הָרָאשִׁי לְיִשְׂרָאֵל' רָץ בַּכְּבִישׁ, נוֹטֵף זֵעָה, קוֹפֵץ בְּתוֹךְ פְּקַק תְּנוּעָה וְצוֹעֵק לַמְּכוֹנִיּוֹת מִסָּבִיב 'פַּנּוּ דֶּרֶךְ!' רַק כְּדֵי שֶׁאֵיזֶה יְהוּדִי יָכוֹל לַעֲבֹר עִם הָרֶכֶב שֶׁלּוֹ – מָה הֱיִיתֶם חוֹשְׁבִים? הַיּוֹם נִפְגֹּשׁ נָשִׂיא אֶחָד, נָשִׂיא שֶׁל כָּל עַם יִשְׂרָאֵל, שֶׁעָשָׂה דָּבָר הַרְבֵּה יוֹתֵר מַפְתִּיעַ, וְהַכֹּל כְּדֵי לִשְׁמֹר עַל כָּבוֹד שֶׁל יְהוּדִי אַחֵר!".

ב. הֶעָרוֹת בִּצּוּעַ (לַמְסַפֵּר):

ג. הַמְלָצוֹת שִׁלּוּב בְּמַצֶּגֶת (עֲבוּר שְׁקוּפִיּוֹת כְּפִי שֶׁצֻּיַּן בַּדְּרִישׁוֹת הַקּוֹדְמוֹת):

ד. פְּעִילוּת מְסַכֶּמֶת - "צְדָקָה בְּשִׂמְחָה": בַּקֵּשׁ מִשְּׁנֵי יְלָדִים לָגֶשֶׁת. אֶחָד שׁוֹאֵל מֵהַשֵּׁנִי צֶבַע. בַּפַּעַם הָרִאשׁוֹנָה: "תַּדְגִּים כֵּיצַד אַתָּה נוֹתֵן לוֹ, אֲבָל פּוֹגֵעַ בּוֹ" (לִזְרֹק בְּעַצְבָּנוּת: "קַח, עֲזֹב אוֹתִי"). בַּפַּעַם הַשְּׁנִיָּה: "תַּדְגִּים עֲנָוָה, וּלְשַׂמֵּחַ אֶת הֶעָנִי" (לִמְסֹר בְּחִיּוּךְ: "בְּשִׂמְחָה אָחִי, אֲנִי זָכִיתִי לַעֲזֹר לְךָ!"). תּוֹבָנָה כִּתָּתִית: "הִלֵּל לִמֵּד אוֹתָנוּ שֶׁמָּה שֶׁנּוֹתְנִים זֶה חָשׁוּב, אֲבָל אֵיךְ נוֹתְנִים זֶה חָשׁוּב כִּפְלַיִם!".

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “עֲנָוָה: אָדָם גָּדוֹל בֶּאֱמֶת לֹא מַחְשִׁיב אֶת עַצְמוֹ כְּלָל יַחַס לְצַעֲרוֹ שֶׁל הַזּוּלַת.” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “שְׂרֵפָה גְּדוֹלָה, אוֹ אוּלַי שׁוֹדְדִים שֶׁהִתְנַפְּלוּ עַל אֳנִיּוֹתָיו – אֲנַחְנוּ לֹא מְדֻיָּק יוֹדְעִים – גָּרְמוּ לְכָךְ שֶׁהָאִישׁ אִבֵּד אֶת כָּל…”?
ג. כיצד הגיב הנשיא שהפך למלאכי למצב המתואר: “הִלֵּל הֵבִין שֶׁאִם הוּא רַק יִשְׁלַח לָאִישׁ הַזֶּה קְצָת כֶּסֶף לְלֶחֶם וַחֲתִיכַת בָּשָׂר, זֶה אָמְנָם יַשְׂבִּיעַ אֶת בִּטְנוֹ, אֲבָל זֶה לֹא…”?
ד. מה השתנה בין “אִם אֵרָאֶה בָּרְחוֹב רוֹכֵב לְבַדִּי, בְּלִי מְשָׁרֵת שֶׁמְּפַנֶּה לִי אֶת הַדֶּרֶךְ, כֻּלָּם יִצְחֲקוּ עָלַי.” לבין “הַנְּשִׁימוֹת שֶׁלּוֹ כָּבְדוּ, אֲגַלֵּי זֵעָה צָצוּ עַל מִצְחוֹ הַקָּדוֹשׁ, הָאֲבָק בָּרְחוֹב עָלָה בָּאֲוִיר, וְהוּא…”?
ה. כיצד הגיע הסיפור אל האירוע: “פְּעִילוּת מְסַכֶּמֶת - "צְדָקָה בְּשִׂמְחָה": בַּקֵּשׁ מִשְּׁנֵי יְלָדִים לָגֶשֶׁת.”?

3. מה היה ראשון?

כתבו ליד כל אירוע מספר מ־1 עד 5 לפי סדר התרחשותו בסיפור.

□ הָעֶבֶד שֶׁהָיָה אָמוּר לָרוּץ לִפְנֵי הַסּוּס, חָלָה אוֹ נֶעֱלָם, וְלֹא הִגִּיעַ לָעֲבוֹדָה.
□ עֲנָוָה: אָדָם גָּדוֹל בֶּאֱמֶת לֹא מַחְשִׁיב אֶת עַצְמוֹ כְּלָל יַחַס לְצַעֲרוֹ שֶׁל הַזּוּלַת.
□ תּוֹבָנָה כִּתָּתִית: "הִלֵּל לִמֵּד אוֹתָנוּ שֶׁמָּה שֶׁנּוֹתְנִים זֶה חָשׁוּב, אֲבָל אֵיךְ נוֹתְנִים זֶה חָשׁוּב כִּפְלַיִם!".
□ הוּא הָיָה "עָנִי בֶּן טוֹבִים" – אָדָם שֶׁרָגִיל לְכָבוֹד, וְעַכְשָׁו אֲנָשִׁים רָאוּ אוֹתוֹ הוֹלֵךְ בָּרֶגֶל בָּרְחוֹב, עָיֵף וּמֻשְׁפָּל.
□ הֲלִיכָה שֶׁל כִּמְעַט שָׁעָה שְׁלֵמָה תַּחַת הַשֶּׁמֶשׁ, רַק כְּדֵי שֶׁהַיְּהוּדִי הֶעָנִי יַרְגִּישׁ שֶׁכְּבוֹדוֹ נִשְׁמָר וְהוּא אֵינוֹ מוּשְׁפָּל.

לפני ואחרי

מה קרה לפני: “הָעֶבֶד שֶׁהָיָה אָמוּר לָרוּץ לִפְנֵי הַסּוּס, חָלָה אוֹ נֶעֱלָם, וְלֹא הִגִּיעַ לָעֲבוֹדָה.”?
מה קרה אחרי: “הֲלִיכָה שֶׁל כִּמְעַט שָׁעָה שְׁלֵמָה תַּחַת הַשֶּׁמֶשׁ, רַק כְּדֵי שֶׁהַיְּהוּדִי הֶעָנִי יַרְגִּישׁ שֶׁכְּבוֹדוֹ נִשְׁמָר וְהוּא אֵינוֹ מוּשְׁפָּל.”?

סיבה ותוצאה מתוך המעשה

האירוע הנתוןמה גרם לו?מה הייתה תוצאתו?
עֵינָיו תָּמִיד הָיוּ בָּרִצְפָּה, שֶׁמָּא מִישֶׁהוּ יַכִּיר אוֹתוֹ וִירַחֵם עָלָיו.
לְפֶתַע, הִלֵּל הַזָּקֵן הֵרִים שׁוּלֵי גְּלִימָתוֹ כְּדֵי שֶׁלֹּא תַּפְרִיעַ לְרַגְלָיו.

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “הנשיא שהפך למלאכי” על הנשיא שהפך למלאכי? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “הוּא רוֹכֵב, וְהָעֶבֶד רָץ וְצוֹעֵק: "פַּנּוּ דֶּרֶךְ לָאָדוֹן הַנִּכְבָּד!".”.
ג. אילו הנשיא שהפך למלאכי לא היה פועל כפי שמתואר ב־“הַנְּשִׁימוֹת שֶׁלּוֹ כָּבְדוּ, אֲגַלֵּי זֵעָה צָצוּ עַל מִצְחוֹ הַקָּדוֹשׁ, הָאֲבָק בָּרְחוֹב עָלָה בָּאֲוִיר, וְהוּא לֹא עָצַר.” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “פְּעִילוּת מְסַכֶּמֶת - "צְדָקָה בְּשִׂמְחָה": בַּקֵּשׁ מִשְּׁנֵי יְלָדִים לָגֶשֶׁת.” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. עֲנָוָה: אָדָם גָּדוֹל בֶּאֱמֶת לֹא מַחְשִׁיב אֶת עַצְמוֹ כְּלָל יַחַס…
2. שְׂרֵפָה גְּדוֹלָה, אוֹ אוּלַי שׁוֹדְדִים שֶׁהִתְנַפְּלוּ עַל…
3. מִתּוֹךְ קֻפַּת הַצְּדָקָה שֶׁהָיְתָה בִּידֵי הַנָּשִׂיא, הוּא קָנָה…
4. אֲבָל הִלֵּל הַזָּקֵן הָיָה מָלֵא בְּאַהֲבַת יִשְׂרָאֵל וַעֲנָוָה…
5. כָּבוֹד אֲמִתִּי וּגְדֻלָּה אֲמִתִּית נִמְדָּדִים בַּעֲנָוָה,…
6. תּוֹבָנָה כִּתָּתִית: "הִלֵּל לִמֵּד אוֹתָנוּ שֶׁמָּה שֶׁנּוֹתְנִים זֶה…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “הִלֵּל הֵבִין שֶׁאִם הוּא רַק יִשְׁלַח לָאִישׁ הַזֶּה קְצָת כֶּסֶף לְלֶחֶם וַחֲתִיכַת בָּשָׂר, זֶה אָמְנָם יַשְׂבִּיעַ אֶת…”.
מכתב: כתבו להנשיא שהפך למלאכי בעקבות “לְפֶתַע, הִלֵּל הַזָּקֵן הֵרִים שׁוּלֵי גְּלִימָתוֹ כְּדֵי שֶׁלֹּא תַּפְרִיעַ לְרַגְלָיו.”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “כָּבוֹד אֲמִתִּי וּגְדֻלָּה אֲמִתִּית נִמְדָּדִים בַּעֲנָוָה, בַּיְּכֹלֶת לוֹמַר: "אֲנִי לֹא חָשׁוּב – הַלֵּב שֶׁל הַיְּהוּדִי…”.
דיווח: כתבו ידיעה קצרה שכותרתה “הנשיא שהפך למלאכי”.
