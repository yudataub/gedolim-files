סוד היין ומשרתי המלך

דף עבודה בעקבות סוד היין ומשרתי המלך
שם: כיתה:

המעשה — הסיפור המלא

הסיפור מנוקד במלואו כנדרש.

🔹 שָׁלָב 1: בֵּרוּר מַקְדִּים וְנִתּוּחַ (לַמּוֹרֶה/לַמְסַפֵּר)

🔹 שָׁלָב 2: הַסִּפּוּר הַמָּלֵא – "סוֹד הַיַּיִן וּמְשָׁרְתֵי הַמֶּלֶךְ"

פְּתִיחָה: חָכְמָה עֲצוּמָה בְּגוּף פָּשׁוּט יְלָדִים, אֲנַחְנוּ כְּבָר יוֹדְעִים שֶׁגְּדוֹלֵי יִשְׂרָאֵל הֵם הָאֲנָשִׁים הֲכִי מְיֻחָדִים בָּעוֹלָם. אֲבָל הַאִם כֻּלָּם הָיוּ נִרְאִים כְּמוֹ שָׂרִים עֲשִׁירִים אוֹ נְסִיכִים? מַמָּשׁ לֹא. קְחוּ לְדֻגְמָה אֶת עֲנַק הַתּוֹרָה, רַבִּי יְהוֹשֻׁעַ בֶּן חֲנַנְיָה. רַבִּי יְהוֹשֻׁעַ הָיָה אָדָם פָּשׁוּט מְאֹד. אָמְנָם בְּתוֹךְ הָרֹאשׁ שֶׁלּוֹ הָיָה אוֹצָר עָצוּם שֶׁל כָּל תּוֹרַת יִשְׂרָאֵל, אֲבָל מִבַּחוּץ – הַבְּגָדִים שֶׁלּוֹ הָיוּ פְּשׁוּטִים, וְהַמַּרְאֶה הַחִיצוֹנִי שֶׁלּוֹ לֹא הָיָה מַרְשִׁים אוֹ נָאֶה כְּלָל. וּבְכָל זֹאת, הוּא זֶה שֶׁנִּבְחַר לַעֲמֹד מִמּוּל הַקֵּיסָר הָרוֹמָאִי כְּדֵי לְהָגֵן עַל עַם יִשְׂרָאֵל.

הָעֶלְבּוֹן שֶׁל הַנְּסִיכָה וְסוֹד הַיַּיִן יוֹם אֶחָד, הִגִּיעַ רַבִּי יְהוֹשֻׁעַ לְאַרְמוֹנוֹ הֶעָנָק שֶׁל קֵיסָר רוֹמִי. רִצְפַּת שַׁיִשׁ בָּרְקָה, עַמּוּדֵי זָהָב עָמְדוּ בַּפְּתָחִים. לְפֶתַע, יָצְאָה מִבַּעַד לַדְּלָתוֹת בַּת הַקֵּיסָר, הַנְּסִיכָה. הִיא לָבְשָׁה בִּגְדֵי פְּאֵר יְקָרִים מְאֹד. הִיא הִבִּיטָה עַל רַבִּי יְהוֹשֻׁעַ דֶּרֶךְ סַנְטֵרָהּ הַמּוּרָם, רָאֲתָה אֶת מַרְאֵהוּ הַפָּשׁוּט, וּפָרְצָה בִּצְחוֹק לַעֲגָנִי. "וָאוּ! אֵיזוֹ תּוֹרָה מְפֹאֶרֶת, וְאֵיזֶה כְּלִי מְכֹעָר!" לָעֲגָה הַנְּסִיכָה בְּקוֹל רָם לְפָנָיו, לְרֶמֶז עַל מַרְאֵהוּ הַחִיצוֹנִי.

יְלָדִים, אָדָם רָגִיל הָיָה נֶעֱלָב מְאֹד, נֶאֱדָם מִבּוּשָׁה וּבוֹרֵחַ. אֲבָל רַבִּי יְהוֹשֻׁעַ הָיָה מָלֵא בְּבִטָּחוֹן שֶׁל קְדֻשָּׁה. הוּא אֲפִלּוּ לֹא מִצְמֵץ. הוּא עָמַד בְּשֶׁקֶט וְאָמַר לָהּ בְּקוֹל רָגוּעַ: "נְסִיכָה, תַּגִּידִי לִי, הֵיכָן אָבִיךְ הַקֵּיסָר שׁוֹמֵר אֶת הַיַּיִן הַיָּקָר שֶׁלּוֹ?". "מָה שְׁאֵלָה?!" הִיא עָנְתָה, "בְּכַדֵּי חֶרֶס שֶׁל חֵמָר בַּמַּרְתֵּף, כַּמּוּבָן". רַבִּי יְהוֹשֻׁעַ עָשָׂה עַצְמוֹ מֻפְתָּע מְאֹד: "מָה?! קֵיסָר רוֹמָא הָאַדִּיר, שׁוֹמֵר אֶת מַשְׁקֵה הַיַּיִן הַטּוֹב שֶׁלּוֹ בְּכַדֵּי מִבֹּץ דַּל וּפָשׁוּט – בְּדִיּוּק כְּמוֹ אֲנָשִׁים פְּשׁוּטִים בַּשּׁוּק?! זֶה זִלְזוּל! מֶלֶךְ צָרִיךְ לִשְׁמֹר יַיִן אַךְ וְרַק בְּכַדִּים נוֹצְצִים מִזָּהָב וָכֶסֶף!". הַנְּסִיכָה חָשְׁבָה שֶׁהוּא צוֹדֵק. הִיא צִוְּתָה מִיָּד עַל כָּל הַמְשָׁרְתִים: "שִׁפְכוּ עַכְשָׁו אֶת כָּל הַיַּיִן מִכַּדֵּי הַחֶרֶס הַמְכֹעָרִים, וְהַעֲבִירוּ אוֹתוֹ לְכַדֵּי זָהָב כְּבֵדִים!".

עָבְרוּ כַּמָּה יָמִים. הַקֵּיסָר רָצָה לִשְׁתּוֹת. הוּא לָגַם לוֹגֶם מִכּוֹס הַיַּיִן שֶׁמָּזְגוּ לוֹ מִכַּד הַזָּהָב... וּמִיָּד יָרַק הַכֹּל הַחוּצָה בְּגֹעַל! "פִּיכְס! זֶה חָמוּץ! הַיַּיִן הִתְקַלְקֵל לְגַמְרֵי!" שָׁאַג הַקֵּיסָר. הַנְּסִיכָה הַנִּבְהֶלֶת רָצָה לְרַבִּי יְהוֹשֻׁעַ, רְתִיחָה בְּכַעַס: "לָמָּה נָתַתָּ לִי עֵצָה גְּרוּעָה כָּזוֹ?! בִּגְלָלְךָ הֶחְמִיץ לָנוּ כָּל הַיַּיִן הַטּוֹב שֶׁבַּמַּרְתֵּף!". חִיֵּךְ רַבִּי יְהוֹשֻׁעַ חִיּוּךְ רָחָב, וְאָמַר לָהּ בְּנַחַת: "עַכְשָׁו תָּבִינִי, נְסִיכָה. מַשְׁקֶה טוֹב נִשְׁמָר רַק בְּתוֹךְ כְּלִי חֶרֶס פָּשׁוּט, כִּי הַחֶרֶס מְקָרֵר אוֹתוֹ. הַזָּהָב שֶׁהוּא יָקָר וְגַאַוְתָן – מְקַלְקֵל אוֹתוֹ. כָּךְ הִיא הַתּוֹרָה, בַּת הַקֵּיסָר! הִיא מִשְׁתַּמֶּרֶת הֲכִי טוֹב אֵצֶל עֲנָוִים, אֲנָשִׁים פְּשׁוּטִים שֶׁלֹּא מִתְגָּאִים בְּיֹפִי חִיצוֹנִי!". הַנְּסִיכָה שָׁתְקָה, בּוֹשָׁה וְהֵבִינָה אֶת מַסְרוֹ הֶחָכָם.

אֲדוֹן הָעוֹלָם וְהַשֶּׁמֶשׁ הַקֵּיסָר בְּעַצְמוֹ הָיָה מְדַבֵּר הַרְבֵּה עִם רַבִּי יְהוֹשֻׁעַ עַל אֱמוּנָה בְּה'. יוֹם אֶחָד, הַקֵּיסָר (שֶׁהָיָה גּוֹי וְרָגִיל לִפְסָלִים מֵאֶבֶן שֶׁאֶפְשָׁר לִרְאוֹת), דָּרַשׁ מֵרַבִּי יְהוֹשֻׁעַ: "אֲנִי רוֹצֶה לִרְאוֹת אֶת הָאֱלֹקִים שֶׁלָּכֶם! תַּרְאֶה לִי אוֹתוֹ!". עָנָה לוֹ רַבִּי יְהוֹשֻׁעַ: "בּוֹרֵא עוֹלָם אֵין לוֹ גּוּף וּדְמוּת! אִי אֶפְשָׁר לִרְאוֹתוֹ בְּעֵינַיִם בְּשָׂרִיּוֹת. הָעֵינַיִם שֶׁלָּנוּ חַלָּשׁוֹת מִדַּי עֲבוּר הָאוֹר הָעָצוּם הַזֶּה". "תַּרְאֶה לִי, אֲנִי מִתְעַקֵּשׁ!" זָעַף הַקֵּיסָר. הוֹצִיא אוֹתוֹ רַבִּי יְהוֹשֻׁעַ לַבַּחוּץ, בְּאֶמְצַע חֹדֶשׁ תַּמּוּז, כְּשֶׁהַשֶּׁמֶשׁ בָּעֲרָה בַּשָּׁמַיִם כְּמוֹ תַּנּוּר לוֹהֵט. "הִסְתַּכֵּל יָשָׁר אֶל הַשֶּׁמֶשׁ!" צִוָּה עָלָיו הָרַב. הַקֵּיסָר הֵרִים אֶת מַבָּטוֹ, אַךְ מִיָּד עָצַם אֶת עֵינָיו בְּכֹחַ. "אֲנִי לֹא מְסֻגָּל! הַשֶּׁמֶשׁ מְסַנְוֶרֶת אוֹתִי מִדַּי!". חִיֵּךְ רַבִּי יְהוֹשֻׁעַ: "תִּרְאֶה אֵיזֶה קָטָן אַתָּה, קֵיסָר! הַשֶּׁמֶשׁ הַזּוֹ, הִיא רַק מְשָׁרֶתֶת אַחַת קְטַנָּה שֶׁל בּוֹרֵא עוֹלָם, וְאַתָּה לֹא מְסֻגָּל לְהַבִּיט בָּהּ... אֵיךְ תּוּכַל לְהַבִּיט בְּמֶלֶךְ מַלְכֵי הַמְּלָכִים, שֶׁהַזִּיו וְהָאוֹר שֶׁלּוֹ מְמַלְּאִים אֶת הָעוֹלָם כֻּלּוֹ?!".

הַסְּעוּדָה לַה' עַל שְׂפַת הַיָּם אֲבָל הַקֵּיסָר הִמְשִׁיךְ לְנַסּוֹת לְהָבִין. "אִם אֵינִי יָכוֹל לִרְאוֹתוֹ, אֲנִי שְׁלִיט הָעוֹלָם – אֲנִי רוֹצֶה לַעֲשׂוֹת לוֹ מְסִבָּה עֲנָקִית! אֲנִי אֶעֱרֹךְ סְעוּדָה לֶאֱלֹקֵי הַיְּהוּדִים!". עָנָה רַבִּי יְהוֹשֻׁעַ: "אִי אֶפְשָׁר. יֵשׁ לוֹ מְשָׁרְתִים וַחֲיָלִים עֲצוּמִים, הַסְּעוּדָה שֶׁלְּךָ לֹא תַּסְפִּיק". הַקֵּיסָר הִתְעַקֵּשׁ, וְהוֹרָה לְכָל צְבָאוֹ לְהָכִין בְּמֶשֶׁךְ שִׁשָּׁה חֳדָשִׁים שְׁלֵמִים אֶת מְסִבַּת הַמְּלָכִים הַגְּדוֹלָה בַּהִיסְטוֹרְיָה – מַמָּשׁ עַל שְׂפַת הַיָּם. אֲלָפִים שֶׁל שֻׁלְחָנוֹת, כּוֹסוֹת זָהָב, הֲרָרִים שֶׁל לֶחֶם, בָּשָׂר וּפֵרוֹת נִפְרְשׂוּ עַל הַחוֹל. הַכֹּל הָיָה מְפֹאָר מֵאֵין כָּמוֹהוּ.

"הִנֵּה!" הִכְרִיז הַקֵּיסָר בְּגַאֲוָה לְרַבִּי יְהוֹשֻׁעַ, "כַּמָּה אֹכֶל! הַסְּעוּדָה מוּכָנָה לַאֱלֹקִים!". וְאָז... לְפֶתַע... הַשָּׁמַיִם הֶחְשִׁיכוּ. עֲנָנִים עֲנָקִיִּים שֶׁל סְעָרָה כִּסּוּ אֶת הַשֶּׁמֶשׁ. רוּחַ וְהוּרִיקָן אַדִּיר, שָׁרַק וְזָעַם מִכִּוּוּן הַיָּם. פווווששש! סְוִוִישׁשׁ! הָרוּחַ סָחֲפָה וְהֵעִיפָה אֶת כָּל שֻׁלְחֲנוֹת הַזָּהָב הַשְּׁקִיעָה אוֹתָם בַּמַּיִם! עָפוּ מַגָּשִׁים, עָפוּ כּוֹסוֹת! וּמִיָּד אַחַר כָּךְ – הִתְחִיל גֶּשֶׁם זַעַף כָּבֵד, גַּלֵּי הַיָּם עָלוּ וְסָחֲפוּ אֶת כָּל מָה שֶׁנִּשְׁאַר אֶל תּוֹךְ הַמְּצוּלוֹת! שִׁשָּׁה חֳדָשִׁים שֶׁל הֲכָנוֹת – נֶעֶלְמוּ בְּדַקָּה וָחֵצִי!

4. הבנה והעמקה

א. מה מלמד המעשה “סוד היין ומשרתי המלך” על סוד היין ומשרתי המלך? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “עָנָה לוֹ רַבִּי יְהוֹשֻׁעַ: "בּוֹרֵא עוֹלָם אֵין לוֹ גּוּף וּדְמוּת!”.
ג. אילו סוד היין ומשרתי המלך לא היה פועל כפי שמתואר ב־“הִנֵּה!" הִכְרִיז הַקֵּיסָר בְּגַאֲוָה לְרַבִּי יְהוֹשֻׁעַ, "כַּמָּה אֹכֶל!” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “תִּרְאֶה עַכְשָׁו מָה קוֹרֶה רַק כְּשֶׁהַמְנַקִּים בָּאִים – כָּל מָה שֶׁהֵכַנְתָּ לֹא הִסְפִּיק אֲפִלּוּ כְּדֵי לְכַבֵּד אֶת…” מחזק את המסר של הסיפור?

2. יְלָדִים, אָדָם רָגִיל הָיָה נֶעֱלָב מְאֹד, נֶאֱדָם מִבּוּשָׁה…
3. אֲדוֹן הָעוֹלָם וְהַשֶּׁמֶשׁ הַקֵּיסָר בְּעַצְמוֹ הָיָה מְדַבֵּר…
4. הַסְּעוּדָה לַה' עַל שְׂפַת הַיָּם אֲבָל הַקֵּיסָר הִמְשִׁיךְ לְנַסּוֹת…
5. שִׁשָּׁה חֳדָשִׁים שֶׁל הֲכָנוֹת – נֶעֶלְמוּ בְּדַקָּה וָחֵצִי!
6. הַיּוֹם נִרְאֶה אֶת רַבִּי יְהוֹשֻׁעַ בֶּן חֲנַנְיָה שֶׁמְּלַמֵּד אֶת…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “הִיא מִשְׁתַּמֶּרֶת הֲכִי טוֹב אֵצֶל עֲנָוִים, אֲנָשִׁים פְּשׁוּטִים שֶׁלֹּא מִתְגָּאִים בְּיֹפִי חִיצוֹנִי!".”.
מכתב: כתבו לסוד היין ומשרתי המלך בעקבות “אִם אֵינִי יָכוֹל לִרְאוֹתוֹ, אֲנִי שְׁלִיט הָעוֹלָם – אֲנִי רוֹצֶה לַעֲשׂוֹת לוֹ מְסִבָּה עֲנָקִית!”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “רַבִּי יְהוֹשֻׁעַ, עוֹמֵד רָגוּעַ בַּסְּעָרָה, חִיֵּךְ וְעָנָה מִבַּעַד לָרוּחַ: "קֵיסָר, אַל תִּבָּהֵל!”.
דיווח: כתבו ידיעה קצרה שכותרתה “סוד היין ומשרתי המלך”.
