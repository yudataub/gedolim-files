הרבה לתת צדקה — הגבאים ברחו ממנו

דף עבודה בעקבות רבי אלעזר איש ברתותא
שם: כיתה:

המעשה — הסיפור המלא

להלן עיבוד חינוכי, עשיר ומרגש לסיפורו המופלא של רבי אלעזר איש ברתותא , המעובד במיוחד לבנים בכיתות ב'-ד' (חינוך דתי-לאומי). הסיפור מדגיש את מידת הביטחון המוחלט בה', ואת ההבנה שכל רכושנו שייך למעשה לבורא עולם.

הסיפור מנוקד במלואו כנדרש.

פְּתִיחָה: הַמִּסְתַּתְּרִים בַּשּׁוּק

יְלָדִים, צֵאוּ וּרְאוּ מַחֲזֶה מוּזָר מְאֹד בְּשׁוּק קָדוּם בְּאֶרֶץ יִשְׂרָאֵל. הַשּׁוּק הוֹמֶה אָדָם. סוֹחֲרִים צוֹעֲקִים אֶת מַרְכּוֹלָתָם, רֵיחוֹת שֶׁל תַּבְלִינִים חֲרִיפִים וּפֵרוֹת מְתוּקִים מְמַלְאִים אֶת הָאֲוִיר, וְקוֹל שִׁקְשׁוּק מַטְבְּעוֹת הַנְּחֹשֶׁת נִשְׁמָע מִכָּל דּוּכָן.

אֲבָל בְּתוֹךְ כָּל הָרַעַשׁ הַזֶּה, אִם תִּסְתַּכְּלוּ הֵיטֵב, תִּרְאוּ שְׁנֵי אֲנָשִׁים מְכֻבָּדִים עִם זְקָנִים לְבָנִים, מִתְחַבְּאִים מֵאֲחוֹרֵי דּוּכָן שֶׁל אֲבַטִּיחִים. הֵם מְצִיצִים בְּפַחַד, וּמִיָּד מִסְתַּתְּרִים שׁוּב.

מִי אֵלֶּה? אֵלּוּ הֵם גַּבָּאֵי הַצְּדָקָה שֶׁל הָעִיר! הָאֲנָשִׁים הַצַּדִּיקִים שֶׁאוֹסְפִים כֶּסֶף עֲבוּר עֲנִיִּים וִיתוֹמִים. וּמִמִּי הֵם בּוֹרְחִים? הַאִם הֵם בּוֹרְחִים מִשּׁוֹדְדִים? לֹא! הַאִם הֵם בּוֹרְחִים מֵחַיּוֹת רָעוֹת? מַמָּשׁ לֹא!

הֵם בּוֹרְחִים מֵאָדָם אֶחָד, צַדִּיק קָדוֹשׁ וְגָדוֹל. הֵם בּוֹרְחִים מִ רַבִּי אֶלְעָזָר אִישׁ בַּרְתּוֹתָא !

וְלָמָּה גַּבָּאֵי צְדָקָה בּוֹרְחִים מֵרַב גָּדוֹל?

כִּי לְרַבִּי אֶלְעָזָר אִישׁ בַּרְתּוֹתָא הָיָה לֵב זָהָב וֶאֱמוּנָה עֲצוּמָה. הָיָה לוֹ מִשְׁפָּט קָבוּעַ שֶׁהָיָה מְשַׁנֵּן לְתַלְמִידָיו: "תֵּן לוֹ מִשֶּׁלּוֹ, שֶׁאַתָּה וְשֶׁלְּךָ שֶׁלּוֹ!" כְּלוֹמַר – כָּל מָה שֶׁיֵּשׁ לִי בַּכִּיס, לֹא שַׁיָּךְ לִי! הַכֹּל שֶׁל בּוֹרֵא עוֹלָם! וְאִם בּוֹרֵא עוֹלָם שׁוֹלֵחַ עָנִי, זֶה סִימָן שֶׁהוּא מְבַקֵּשׁ אֶת הַכֶּסֶף שֶׁלּוֹ בַּחֲזָרָה.

וְלָכֵן, בְּכָל פַּעַם שֶׁרַבִּי אֶלְעָזָר הָיָה רוֹאֶה אֶת גַּבָּאֵי הַצְּדָקָה, הוּא הָיָה מַכְנִיס אֶת יָדוֹ לַכִּיס, מְרוֹקֵן אֶת כָּל מָה שֶׁיֵּשׁ לוֹ – עַד הַפְּרוּטָה הָאַחֲרוֹנָה – וְנוֹתֵן לָהֶם! הַגַּבָּאִים שֶׁיָּדְעוּ שֶׁהוּא אָדָם שֶׁחַי בְּצִמְצוּם, פָּשׁוּט רִחֲמוּ עָלָיו. הֵם לֹא רָצוּ שֶׁיִּשָּׁאֵר בְּלִי כֶּסֶף לְמִשְׁפַּחְתּוֹ, וְלָכֵן כְּשֶׁרָאוּ אוֹתוֹ מִתְקָרֵב – הָיוּ בּוֹרְחִים!

הַהֲכָנוֹת לַחֲתֻנָּה וְהַמִּרְדָּף הַגָּדוֹל

הַיּוֹם זֶה יוֹם מְיֻחָד בְּבֵיתוֹ שֶׁל רַבִּי אֶלְעָזָר. בִּתּוֹ הַיְקָרָה וְהַצַּדֶּקֶת עוֹמֶדֶת לְהִתְחַתֵּן בְּקָרוֹב!

שָׁנִים עַל גַּבֵּי שָׁנִים חָסַךְ רַבִּי אֶלְעָזָר מַטְבֵּעַ לְמַטְבֵּעַ. בַּקֻּפָּה הַקְּטַנָּה שֶׁלּוֹ הִצְטַבֵּר סְכוּם נִכְבָּד מְאֹד. הַיּוֹם הוּא הַיּוֹם שֶׁבּוֹ לָקַח אֶת צְרוֹר הַכֶּסֶף הַכָּבֵד שֶׁקָּשׁוּר לַחֲגוֹרָתוֹ, וְהָלַךְ לַשּׁוּק לִקְנוֹת לְבִתּוֹ "נְדוּנְיָה" – שִׂמְלַת כַּלָּה מֵהַבַּדִּים הֲכִי הֲדוּרִים, מַאֲכָלִים טוֹבִים לִסְעוּדַת הַחֲתֻנָּה, וְכֵלִים יָפִים לַבַּיִת הֶחָדָשׁ שֶׁלָּהּ.

רַבִּי אֶלְעָזָר פּוֹסֵעַ בַּשּׁוּק, חִיּוּךְ רָחָב שֶׁל שִׂמְחָה עַל פָּנָיו, וּלְפֶתַע – הוּא סוֹבֵב אֶת רֹאשׁוֹ וְקָלַט אֶת שְׁנֵי גַּבָּאֵי הַצְּדָקָה שֶׁל הָעִיר צְוָתָא מִסְתַּתְּרִים, כּוֹפְפִים אֶת רֹאשָׁם וּמַתְחִילִים לָרוּץ לַכִּוּוּן הַשֵּׁנִי וּלְהִבָּלַע בֵּין הַהֲמוֹנִים.

"עִמְדוּ אֲחַי! עִמְדוּ!" צָעַק רַבִּי אֶלְעָזָר, וְהִתְחִיל לָרוּץ אַחֲרֵיהֶם. הַגַּלָּבִּיָּה שֶׁלּוֹ מִתְנוֹפֶפֶת, צְרוֹר הַכֶּסֶף טוֹפֵחַ עַל רַגְלוֹ, אַךְ הוּא אֵינוֹ מְוַתֵּר. הוּא רָץ בֵּין הַדּוּכָנִים, נוֹשֵׁם בִּכְבֵדוּת, עַד שֶׁבְּסִמְטָה קְטַנָּה שָׂם אֶת יָדוֹ עַל כְּתֵפוֹ שֶׁל אֶחָד הַגַּבָּאִים. "תָּפַסְתִּי אֶתְכֶם!" אָמַר כְּשֶׁהוּא מִתְנַשֵּׁף וְאֶגְלֵי זֵעָה עַל מִצְחוֹ.

הַגַּבָּאִים עָמְדוּ נְבוֹכִים, מַשְׁפִּילִים אֶת מַבָּטָם.

"אֲנִי מַשְׁבִּיעַ אֶתְכֶם בְּשֵׁם ה'!" אָמַר רַבִּי אֶלְעָזָר בְּקוֹל תַּקִּיף וְאוֹהֵב, "סַפְּרוּ לִי, עֲבוּר אֵיזוֹ מִצְוָה אַתֶּם אוֹסְפִים הַיּוֹם כֶּסֶף שֶׁאַתֶּם מְנַסִּים לְהַסְתִּיר מִמֶּנִּי אֶת הַזְּכוּת לְהִשְׁתַּתֵּף בָּהּ?".

הַגַּבָּאִים שָׁתְקוּ לְרֶגַע, וְאָז לָחֲשׁוּ: "רַבֵּנוּ הַקָּדוֹשׁ... אֲנַחְנוּ אוֹסְפִים הַיּוֹם לְהַכְנָסַת כַּלָּה. יֵשׁ כָּאן חָתָן יָתוֹם וְכַלָּה יְתוֹמָה. שְׁנֵיהֶם עֲנִיִּים, אֵין לָהֶם אַבָּא וְאִמָּא שֶׁיִּדְאֲגוּ לָהֶם. אֲנַחְנוּ אוֹסְפִים לָהֶם כֶּסֶף לַחֲתֻנָּה וּלְבַיִת לְהָקִים".

הַהַחְלָטָה מִשּׁוּרַת הַדִּין

רַבִּי אֶלְעָזָר עָצַם אֶת עֵינָיו. צַעַר עָמוֹק מִלֵּא אֶת לִבּוֹ עַל אוֹתָם יְתוֹמִים גַּלְמוּדִים. לְבִתּוֹ כַּלָּתוֹ יֵשׁ אַבָּא – הֲרֵי זֶה הוּא עַצְמוֹ! יֵשׁ מִי שֶׁיִּדְאַג לָהּ. אֲבָל לַיְתוֹמִים הָאֵלֶּה... אֵין אִישׁ בָּעוֹלָם שֶׁלָּהֶם מִלְּבַד אֲבִיהֶם שֶׁבַּשָּׁמַיִם.

רַבִּי אֶלְעָזָר פָּקַח אֶת עֵינָיו, וּלְלֹא הִסּוּס שֶׁל רֶגַע קָרָא: "הָעֲבוֹדָה! שֶׁהֵם קוֹדְמִים לְבִתִּי!" (העבודה - לשון של שבועה, כמו עבודת בית המקדש).

הוּא שָׁלַח אֶת יָדוֹ אַמִּיצָה לַחֲגוֹרָתוֹ, הִתִּיר אֶת שַׂקִּית הָעוֹר הַכְּבֵדָה, זוֹ שֶׁהֵכִילוּ אֶת חֶסְכוֹנוֹת כָּל חַיָּיו, וְהִנִּיחַ אוֹתָהּ בְּיָדָיו שֶׁל הַגַּבַּאי. קוֹל שִׁקְשׁוּק הַמַּטְבְּעוֹת נִשְׁמַע הֵיטֵב.

"אֲבָל רַבֵּנוּ," רָעַד קוֹלוֹ שֶׁל הַגַּבַּאי, "מָה עִם שִׂמְחַת בִּתְּךָ?".

רַבִּי אֶלְעָזָר חִיֵּךְ. "ה' דּוֹאֵג לְכֻלָּם. קְחוּ אֶת הַכֹּל, וְשַׂמְּחוּ אֶת הַיְתוֹמִים".

רַק מַטְבֵּעַ אֶחָד קָטָן, מַטְבֵּעַ שֶׁל 'זוּז', נִשְׁאַר מִתְגַּלְגֵּל בְּתַחְתִּית הַכִּיס הַרֵיק שֶׁלּוֹ. בַּמַּטְבֵּעַ הַבּוֹדֵד הַזֶּה קָנָה רַבִּי אֶלְעָזָר כַּמָּה חֹפְנִים שֶׁל חִטָּה פְּשׁוּטָה וּצְהֻבָּה, רַק כְּדֵי לְהָבִיא מַשֶּׁהוּ הַבַּיְתָה, שָׂם אוֹתָם בְּשַׂק קָטָן, וְהָלַךְ לְבֵיתוֹ.

הַמַּחְסָן הַשָּׁקֵט וְהַבְּרָכָה הָרוֹעֶשֶׁת

כְּשֶׁהִגִּיעַ רַבִּי אֶלְעָזָר אִישׁ בַּרְתּוֹתָא לְבֵיתוֹ, שָׂרַר שֶׁקֶט. הוּא נִכְנַס בַּחֲשַׁאי לַמַּחְסָן הָרֵיק שֶׁלָּהֶם, וְשָׁפַךְ לְתוֹךְ קֶרֶן הַזָּוִית אֶת הַחִטָּה הַמְּעַטָּה שֶׁקָּנָה. רֵיחַ עָפָר וְחִטָּה טְרִיָּה עָלָה בַּחֲדַר הַמַּחְסָן. הוּא סָגַר אֶת הַדֶּלֶת. הוּא הָיָה שָׂמֵחַ בַּמִּצְוָה שֶׁל יְתוֹמֵי עַם יִשְׂרָאֵל, אֲבָל מְעַט הִתְבַּיֵּשׁ מֵאִשְׁתּוֹ וּמִבִּתּוֹ, לָכֵן הָלַךְ מִיָּד לְבֵית הַמִּדְרָשׁ לִשְׁקֹעַ בְּלִמּוּד הַתּוֹרָה, מֵנִיחַ אֶת כָּל דְּאָגוֹתָיו בִּידֵי ה'.

בַּבַּיִת, הִמְתִּינָה אִשְׁתּוֹ שֶׁל רַבִּי אֶלְעָזָר. הִיא יָדְעָה שֶׁהַיּוֹם הוּא יוֹם קְנִיַּת הַנְּדוּנְיָה. שָׁעוֹת עָבְרוּ, וְהוּא לֹא חָזַר.

קָרְאָה הָאֵם לְבִתָּהּ וְאָמְרָה לָהּ: "בִּתִּי, אַבָּא יָצָא לַשּׁוּק וַדַּאי קָנָה דְּבָרִים יָפִים וְשָׁמַר אוֹתָם בַּמַּחְסָן. בּוֹאִי, גְּשִׁי לַמַּחְסָן וּבִדְקִי מָה הֵבִיא לָנוּ לִכְבוֹד חֲתֻנָּתֵךְ".

הַבַּת נִגְּשָׁה אֶל דֶּלֶת הָעֵץ הַכְּבֵדָה שֶׁל הַמַּחְסָן. הִיא שִׁלְשְׁלָה אֶת הַמַּפְתֵּחַ, סוֹבְבָה... וְנִסְּתָה לִדְחֹף אֶת הַדֶּלֶת.

אֲבָל הַדֶּלֶת... לֹא נִפְתַּחַת! מַשֶּׁהוּ חָזָק וְכָבֵד דּוֹחֵף אוֹתָהּ מִבִּפְנִים!

"אִמָּא! אִמָּא! הַדֶּלֶת תְּקוּעָה!" קָרְאָה הַבַּת.

הָאֵם הִגִּיעָה מִיָּד, וּשְׁתֵּיהֶן נִשְׁעֲנוּ עַל הַדֶּלֶת בְּכָל כֹּחָן וְדָחֲפוּ חָזָק! חָזָק!

קְרִיק... קְרַאק... דֶּלֶת הָעֵץ נִפְתְּחָה מְעַט, וּלְפֶתַע...

מַפָּל שֶׁל זָהָב! עֲרֵמָה אֲדִירָה, שֶׁפַע עָצוּם וְלֹא נִתְפָּס שֶׁל חִטָּה זַכָּה וּטְהוֹרָה נִשְׁפַּךְ עֲלֵיהֶן מִתּוֹךְ הַמַּחְסָן! הַמַּחְסָן כֻּלּוֹ הָיָה מָלֵא מִן הָרִצְפָּה וְעַד הַתִּקְרָה, עַד שֶׁהַחִטָּה דָּחֲפָה אֶת צִירֵי הַדֶּלֶת! הַחִטָּה זָלְגָה אֶל הַסָּלוֹן, אֶל הַפֶּתַח, כְּמוֹ נָהָר שֶׁל בְּרָכָה! מֵהַשֶּׁפַע הַזֶּה אֶפְשָׁר הָיָה לַעֲשׂוֹת עֶשְׂרִים חֲתֻנּוֹת! הַהַשְׁגָּחָה הַפְּרָטִית שֶׁל בּוֹרֵא עוֹלָם הֵאִירָה אֶת הַבַּיִת כֻּלּוֹ.

רָצָה הַבַּת לְבֵית הַמִּדְרָשׁ, מְתוּנָה אַךְ מְלֵאַת הִתְרַגְּשׁוּת, וּמָצְאָה אֶת אָבִיהָ יוֹשֵׁב וְלוֹמֵד בִּדְבֵקוּת. "אַבָּא!" קָרְאָה לוֹ אֶל הַפֶּתַח. "אַבָּא, בּוֹא וּרְאֵה מָה עָשָׂה אוֹהַבְךָ עֲבוּרְךָ! מָה בּוֹרֵא עוֹלָם עָשָׂה לְךָ הַיּוֹם בַּמַּחְסָן! בּוֹא תִּרְאֶה אֶת אוֹצַר הַחִטָּה!".

רַבִּי אֶלְעָזָר חִיֵּךְ. שַׁלְוַת אֱלֹקִים נָחָה עַל פָּנָיו. הוּא הֵרִים מַבָּטוֹ לַשָּׁמַיִם, וְהֵשִׁיב לְבִתּוֹ אֶת הַמִּשְׁפָּט שֶׁמְּלַוֶּה אוֹתוֹ כָּל חַיָּיו:

"בִּתִּי הַיְקָרָה, אֲנִי מַשְׁבִּיעַ אוֹתָךְ... כָּל הַחִטָּה הַזּוֹ הִיא קֹדֶשׁ (הֶקְדֵּשׁ), מֻקְדֶּשֶׁת לַעֲנִיִּים כְּמוֹ בְּבֵית הַמִּקְדָּשׁ, וְלָנוּ אֵין בָּהּ חֵלֶק... חוּץ מִמְּנַת חִטָּה שֶׁלָּקַחְנוּ בִּזְכוּת, כִּי גַּם אֲנַחְנוּ מֵעֲנִיֵּי יִשְׂרָאֵל".

רַבִּי אֶלְעָזָר לֹא הִשְׁתַּנָּה, אַךְ הַקָּדוֹשׁ בָּרוּךְ הוּא דָּאַג לְבִתּוֹ לַחֲתֻנָּה הֲכִי שְׂמֵחָה וּמְבֹרֶכֶת מִתּוֹךְ כְּבוֹד שָׁמַיִם הָעָצוּם שֶׁהֵאִיר מִמַּעֲשָׂיו שֶׁל אָבִיהָ.

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “הרבה לתת צדקה — הגבאים ברחו ממנו” על רבי אלעזר איש ברתותא? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “רַבִּי אֶלְעָזָר פָּקַח אֶת עֵינָיו, וּלְלֹא הִסּוּס שֶׁל רֶגַע קָרָא: "הָעֲבוֹדָה!”.
ג. אילו רבי אלעזר איש ברתותא לא היה פועל כפי שמתואר ב־“הָאֵם הִגִּיעָה מִיָּד, וּשְׁתֵּיהֶן נִשְׁעֲנוּ עַל הַדֶּלֶת בְּכָל כֹּחָן וְדָחֲפוּ חָזָק!” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “חוּץ מִמְּנַת חִטָּה שֶׁלָּקַחְנוּ בִּזְכוּת, כִּי גַּם אֲנַחְנוּ מֵעֲנִיֵּי יִשְׂרָאֵל".” מחזק את המסר של הסיפור?

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “תָּפַסְתִּי אֶתְכֶם!" אָמַר כְּשֶׁהוּא מִתְנַשֵּׁף וְאֶגְלֵי זֵעָה עַל מִצְחוֹ.”.
מכתב: כתבו לרבי אלעזר איש ברתותא בעקבות “הוּא הָיָה שָׂמֵחַ בַּמִּצְוָה שֶׁל יְתוֹמֵי עַם יִשְׂרָאֵל, אֲבָל מְעַט הִתְבַּיֵּשׁ מֵאִשְׁתּוֹ וּמִבִּתּוֹ, לָכֵן הָלַךְ…”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “הַהַשְׁגָּחָה הַפְּרָטִית שֶׁל בּוֹרֵא עוֹלָם הֵאִירָה אֶת הַבַּיִת כֻּלּוֹ.”.
דיווח: כתבו ידיעה קצרה שכותרתה “הרבה לתת צדקה — הגבאים ברחו ממנו”.

המחזה

רגע מן הסיפורהדמויותמה יאמרו או יעשו?
הַיּוֹם זֶה יוֹם מְיֻחָד בְּבֵיתוֹ שֶׁל רַבִּי אֶלְעָזָר.
הָאֵם הִגִּיעָה מִיָּד, וּשְׁתֵּיהֶן נִשְׁעֲנוּ עַל הַדֶּלֶת בְּכָל כֹּחָן וְדָחֲפוּ חָזָק!

7. חיבור לחיים מתוך המעשה
