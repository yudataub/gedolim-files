כיצד לימד תורה בעיירות ללא מלמד — מנחמשים לחמשים

דף עבודה בעקבות ר' חייא
שם: כיתה:

המעשה — הסיפור המלא

פְּתִיחָה: הַשֶּׁקֶט שֶׁמַּפְחִיד אֶת הַצַּדִּיק

יְלָדִים, דַּמְיְנוּ עַצְמְכֶם הוֹלְכִים בִּרְחוֹבוֹת שֶׁל עֲיָרָה קְטַנָּה בְּאֶרֶץ יִשְׂרָאֵל, לִפְנֵי כְּאַלְפַּיִם שָׁנָה. מָה אַתֶּם שׁוֹמְעִים? צִיּוּץ שֶׁל צִפּוֹרִים, קוֹל שֶׁל פָּרוֹת גּוֹעוֹת, פַּטִּישׁ שֶׁל נַפָּח מַכֶּה עַל בַּרְזֶל... אֲבָל חָסֵר שָׁם הַקּוֹל הֲכִי חָשׁוּב בָּעוֹלָם.

רַבִּי חִיָּיא, הַתַּנָּא הַגָּדוֹל וְהַקָּדוֹשׁ, הָלַךְ בַּדְּרָכִים וְהִקְשִׁיב. הַשֶּׁקֶט הַזֶּה הִפְחִיד אוֹתוֹ. אֵין בָּעֲיָרָה הַזּוֹ קוֹל שֶׁל יְלָדִים שֶׁלּוֹמְדִים קְרִיאַת שְׁמַע. אֵין תַּלְמוּד תּוֹרָה. אֵין מְלַמֵּד. יְלָדִים מִשְׂחֲקִים בַּחוֹל, עוֹבְדִים בַּשָּׂדֶה, אֲבָל אֵינָם יוֹדְעִים אֲפִלּוּ אוֹת אַחַת מֵהַתּוֹרָה הַקְּדוֹשָׁה. הֵם לֹא יָדְעוּ מָה זֶה אָלֶף-בֵּית!

לִבּוֹ שֶׁל רַבִּי חִיָּיא כָּאַב. עֵינָיו דָּמְעוּ. "מָה יִהְיֶה עַל עַם יִשְׂרָאֵל?" חָשַׁב בְּצַעַר, "הַתּוֹרָה חָלִילָה עֲלוּלָה לְהִשְׁתַּכֵּחַ מֵהֶם! אֲנִי חַיָּב לַעֲשׂוֹת מַשֶּׁהוּ! אֲנִי אֶדְאַג שֶׁהַתּוֹרָה לֹא תִּשְׁתַּכֵּחַ מִיִּשְׂרָאֵל לְעוֹלָם!".

מָה הָיָה עוֹשֶׂה אָדָם רָגִיל? אוּלַי הָיָה הוֹלֵךְ לַחֲנוּת וְקוֹנֶה סְפָרִים? אֲבָל בְּאוֹתָם יָמִים לֹא הָיוּ בָּתֵּי דְּפוּס וְלֹא מִפְעָלִים. כְּדֵי שֶׁיִּהְיוּ סְפָרִים, מִישֶׁהוּ הָיָה צָרִיךְ לִיצֹר אוֹתָם. וְרַבִּי חִיָּיא – גְּדוֹל הַדּוֹר – הֶחְלִיט לַעֲשׂוֹת אֶת הַכֹּל מַמָּשׁ... מֵהַהַתְחָלָה!

הָרֶשֶׁת וְהַצְּבָאִים: מְסִירוּת מֵהַיְּסוֹד

יָצָא רַבִּי חִיָּיא אֶל הַשָּׂדֶה. הוּא לָקַח מַעֲדֵר, קִדֵּד בָּאֲדָמָה הַלַּחָה שֶׁרֵיחָהּ טוֹב עָלָה בָּאֲוִיר, וְטָמַן בָּהּ זְרָעִים קְטַנִּים שֶׁל צֶמַח הַפִּשְׁתָּן. הוּא הִשְׁקָה אוֹתָם יְוֹם יוֹם, וּבִקֵּשׁ בַּקָּשָׁה חֲרִישִׁית מֵהַשֵּׁם יִתְבָּרַךְ שֶׁיַּצְלִיחַ אֶת דַּרְכּוֹ. הַשֶּׁמֶשׁ חִמְּמָה, הַגֶּשֶׁם יָרַד, וְהַפִּשְׁתָּן צָמַח לְגִבְעוֹלִים אֲרֻכִּים וַחֲזָקִים.

רַבִּי חִיָּיא קָצַר אֶת הַפִּשְׁתָּן וְטָוָה מִמֶּנּוּ חוּטִים עָבִים וּמְחֻסְפָּסִים. אַצְבְּעוֹתָיו שֶׁל הַצַּדִּיק שֶׁהָיוּ רְגִילוֹת לַהֲפֹךְ בְּדַפֵּי הַתּוֹרָה, קָשְׁרוּ קֶשֶׁר אַחַר קֶשֶׁר עַד שֶׁהִתְקַבְּלָה רֶשֶׁת גְּדוֹלָה וַחֲזָקָה! "עַכְשָׁו," חִיֵּךְ רַבִּי חִיָּיא, "אֲנִי הוֹלֵךְ לַיַּעַר".

הוּא פָּרַשׂ אֶת הָרֶשֶׁת בֵּין הָעֵצִים, וְצָד בָּהּ צְבָאִים בְּרִיאִים וְזְרִיזִים. יְלָדִים, אַתֶּם בְּוַדַּאי שׁוֹאֲלִים – מָה עָשָׂה הַצַּדִּיק בִּבְשַׂר הַצְּבָאִים? הוּא בִּשֵּׁל אוֹתוֹ וְהֵכִין מָרָק מֵזִין, חַם וּמַהְבִּיל. אֶת הַמָּרָק הַזֶּה הוּא לֹא נָתַן לָעֲשִׁירִים. הוּא הָלַךְ לַבָּתִּים קְטַנִּים בָּעֲיָרָה שֶׁבָּהֶם גָּרוּ יְתוֹמִים עֲנִיִּים שֶׁלֹּא הָיָה לָהֶם אֹכֶל כְּלָל. הַיְלָדִים אָכְלוּ אֶת הַבָּשָׂר הֶחָם, וְהַשִּׂמְחָה חָזְרָה לְעֵינֵיהֶם! כָּךְ שִׁלֵּב רַבִּי חִיָּיא גְּמִילוּת חֲסָדִים נִפְלָאָה עִם מַטְּרָתוֹ הַקְּדוֹשָׁה.

וּמָה עָשָׂה עִם הָעוֹר שֶׁל הַצְּבָאִים? זֶה הָיָה הַסּוֹד שֶׁלּוֹ! הָעוֹר הַזֶּה הָיָה מְיֹעָד לִקְדֻשָּׁה עֶלְיוֹנָה.

הָאוֹתִיּוֹת הַמְּאִירוֹת עַל עֲרוֹת הַצְּבָאִים

רַבִּי חִיָּיא עִבֵּד אֶת הָעוֹרוֹת. הוּא נִקָּה אוֹתָם, מָתַח אוֹתָם וְיִבֵּשׁ אוֹתָם תַּחַת קַרְנֵי הַשֶּׁמֶשׁ, עַד שֶׁהָפְכוּ לִקְלָף אֲמִתִּי שֶׁל תּוֹרָה – חָלָק וְלָבָן.

הוּא יָשַׁב בְּבֵיתוֹ הַקָּטָן, לָקַח קוּלְמוֹס (נוֹצָה לִכְתִיבָה) וְדְיוֹ שָׁחוֹר כְּלַיְלָה. בִּקְדֻשָּׁה וּבְטָהֳרָה חָרַט רַבִּי חִיָּיא אֶת הָאוֹתִיּוֹת. הָאוֹתִיּוֹת נִרְאוּ כְּאִלּוּ יֵשׁ בָּהֶן אֵשׁ מֵאִירָה.

עַל הַקְּלָפִים שֶׁהֵכִין, הוּא כָּתַב אֶת חֲמִשָּׁה חֻמְשֵׁי תּוֹרָה: סֵפֶר בְּרֵאשִׁית עַל מְגִלָּה אַחַת, שְׁמוֹת, וַיִּקְרָא, בַּמִּדְבָּר וּדְבָרִים.

עוֹד שִׁשָּׁה קְלָפִים הוּא הֵכִין, וַעֲלֵיהֶם כָּתַב אֶת שִׁשָּׁה סִדְרֵי מִשְׁנָה: זְרָעִים, מוֹעֵד, נָשִׁים, נְזִיקִין, קָדָשִׁים וְטְהָרוֹת.

וְעַכְשָׁו, לְאַחַר שֶׁהַמְּגִלּוֹת הָיוּ מוּכָנוֹת... יָצָא רַבִּי חִיָּיא אֶל הָעֲיָרָה שֶׁלֹּא הָיָה בָּהּ מְלַמֵּד.

אֶחָד מְלַמֵּד אֶת הַשֵּׁנִי: קֶסֶם הַתּוֹרָה!

רַבִּי חִיָּיא לֹא אָסַף אֶת כָּל הָעִיר בְּבַת אַחַת. הוּא הָלַךְ לִשְׁכוּנָה וְאָסַף חֲמִשָּׁה יְלָדִים, שֶׁמֵּעוֹלָם לֹא יָדְעוּ קְרוֹא וּכְתֹב. הוּא הוֹשִׁיב אוֹתָם סְבִיבוֹ. הַמַּבָּט שֶׁלּוֹ הָיָה כָּל כָּךְ חַם וְאוֹהֵב!

הוּא הִגִּישׁ אֶת הַמְּגִלּוֹת וְאָמַר לָהֶם: "יְלָדִים יְקָרִים, אֲנִי אֲלַמֵּד אוֹתְכֶם אֶת מַתְּנַת הַשֵּׁם!".

וְאָז הוּא עָשָׂה דָּבָר פִּלְאִי! הוּא לֹא לִמֵּד אֶת עַצְמוֹ עִם כֻּלָּם. אֶת סֵפֶר 'בְּרֵאשִׁית' לִמֵּד רַק אֶת הַיֶּלֶד הָרִאשׁוֹן. אֶת סֵפֶר 'שְׁמוֹת' – לִמֵּד אֶת הַיֶּלֶד הַשֵּׁנִי. לַשְּׁלִישִׁי לִמֵּד רַק אֶת 'וַיִּקְרָא', לָרְבִיעִי רַק 'בַּמִּדְבָּר', וְלַחֲמִישִׁי رַק אֶת 'דְּבָרִים'!

אַחַר כָּךְ הָלַךְ וְאָסַף שִׁשָּׁה יְלָדִים אֲחֵרִים, בּוֹגְרִים יוֹתֵר. אֶת כָּל יֶלֶד הוּא לִמֵּד מַסֶּכֶת אַחַת שְׁלֵמָה מִתּוֹךְ שִׁשָּׁה סִדְרֵי מִשְׁנָה. רַק מַסֶּכֶת אַחַת לְיֶלֶד!

יְלָדִים, אַתֶּם בְּוַדַּאי מִתְפַּלְאִים – וּמָה יִקְרֶה עִם שְׁאָר הַתּוֹרָה? אֵיךְ יֵדַע זֶה שֶׁלָּמַד בְּרֵאשִׁית אֶת סֵפֶר שְׁמוֹת?

לִפְנֵי שֶׁרַבִּי חִיָּיא עָזַב אוֹתָם, הוּא חִיֵּךְ אֲלֵיהֶם וְהִנְחָה אוֹתָם בַּהוֹרָאָה שֶׁשִּׁנְּתָה אֶת הָעוֹלָם: "יְלָדַי הָאֲהוּבִים! עַכְשָׁו אֲנִי הָלַךְ לְעוֹד עֲיָרוֹת. עַד שֶׁאָשׁוּב – אֲנִי מְבַקֵּשׁ שֶׁכָּל אֶחָד מִכֶּם יְלַמֵּד אֶת מָה שֶׁהוּא יוֹדֵעַ לְחֲבֵרוֹ. הַיֶּלֶד שֶׁלָּמַד בְּרֵאשִׁית יְלַמֵּד אֶת חֲבֵרוֹ בְּרֵאשִׁית, וְיִלְמַד מִמֶּנּוּ אֶת מָה שֶׁהוּא יוֹדֵעַ. לַמְּדוּ אֶחָד אֶת הַשֵּׁנִי!".

חֲזָרָה מְפֹאֶרֶת וְשֶׁבַח נִצְחִי

כַּעֲבֹר זְמַן, חָזַר רַבִּי חִיָּיא לְאוֹתָהּ עֲיָרָה. בִּמְקוֹם הַשֶּׁקֶט שֶׁל צִיּוּצֵי הַצִּפּוֹרִים, מָה הוּא שָׁמַע עַכְשָׁו?

מֵרָחוֹק בָּקְעוּ קוֹלוֹת מְתוּקִים כִּדְבַשׁ שֶׁל עַשְׂרוֹת יְלָדִים הַלּוֹמְדִים תּוֹרָה בְּחֶבְרוּתָא, שׁוֹאֲלִים וְעוֹנִים, מַסְבִּירִים אֶחָד לַשֵּׁנִי וּמְשַׁנְּנִים אֶת פְּסוּקֵי הַתּוֹרָה וְהַמִּשְׁנָה בְּהִתְלַהֲבוּת גְּדוֹלָה! מֵאוֹתָם זְרָעִים קְטַנִּים שֶׁל פִּשְׁתָּן... קָם בֵּית מִדְרָשׁ מָלֵא תּוֹרָה שֶׁאַף אֶחָד לֹא יָכוֹל הָיָה לְכַבּוֹת.

כַּאֲשֶׁר שָׁמַע עַל כָּךְ עֲנַק הַדּוֹר, הַנָּשִׂיא רַבִּי יְהוּדָה – הַיָּדוּעַ כְּ"רַבִּי" – מִי שֶׁכָּתַב אֶת הַמִּשְׁנָה כֻּלָּהּ, הוּא קָרָא בְּהִתְרַגְּשׁוּת וּבְשִׂמְחָה עֲצוּמָה:

"כַּמָּה גְּדוֹלִים מַעֲשֵׂי חִיָּיא!"

רַבִּי חִיָּיא הֵבִין אֶת הַסּוֹד – מִי שֶׁרוֹצֶה שֶׁהַתּוֹרָה תִּחְיֶה בָּעוֹלָם, הוּא לֹא צָרִיךְ לְחַכּוֹת לַאֲחֵרִים שֶׁיַּעֲשׂוּ. הוּא מִתְאֲמֵּץ בְּעַצְמוֹ, עוֹשֶׂה גְּמִילוּת חֲסָדִים לִיתוֹמִים, וְלֹא שׁוֹמֵר אֶת הַיֶּדַע לְעַצְמוֹ, אֶלָּא דּוֹאֵג שֶׁכָּל אֶחָד יְלַמֵּד אֶת הַשֵּׁנִי.

מקורות: תלמוד בבלי, מסכת בבא מציעא, דף פ"ה, עמוד ב' (מעשה רבי חייא עם הפשתן, הצבאים והיתומים, כתיבת המגילות ותלמוד הילדים). תלמוד בבלי, מסכת כתובות, דף ק"ג, עמוד ב' (אמרתו של רבי חייא: "אני עושה שלא תשתכח תורה מישראל"). מדרשי חז"ל וספר האגדה (פירוטים על תקופת ר' יהודה הנשיא ותלמידיו).

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “יְלָדִים, דַּמְיְנוּ עַצְמְכֶם הוֹלְכִים בִּרְחוֹבוֹת שֶׁל עֲיָרָה קְטַנָּה בְּאֶרֶץ יִשְׂרָאֵל, לִפְנֵי כְּאַלְפַּיִם שָׁנָה.” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “וְרַבִּי חִיָּיא – גְּדוֹל הַדּוֹר – הֶחְלִיט לַעֲשׂוֹת אֶת הַכֹּל מַמָּשׁ...”?
ג. כיצד הגיב ר' חייא למצב המתואר: “הוּא פָּרַשׂ אֶת הָרֶשֶׁת בֵּין הָעֵצִים, וְצָד בָּהּ צְבָאִים בְּרִיאִים וְזְרִיזִים.”?
ד. מה השתנה בין “עַל הַקְּלָפִים שֶׁהֵכִין, הוּא כָּתַב אֶת חֲמִשָּׁה חֻמְשֵׁי תּוֹרָה: סֵפֶר בְּרֵאשִׁית עַל מְגִלָּה אַחַת, שְׁמוֹת,…” לבין “לַשְּׁלִישִׁי לִמֵּד רַק אֶת 'וַיִּקְרָא', לָרְבִיעִי רַק 'בַּמִּדְבָּר', וְלַחֲמִישִׁי رַק אֶת 'דְּבָרִים'!”?
ה. כיצד הגיע הסיפור אל האירוע: “קָם בֵּית מִדְרָשׁ מָלֵא תּוֹרָה שֶׁאַף אֶחָד לֹא יָכוֹל הָיָה לְכַבּוֹת.”?

הַיְלָדִים אָכְלוּ אֶת הַבָּשָׂר הֶחָם, וְהַשִּׂמְחָה חָזְרָה לְעֵינֵיהֶם!□□

3. מה היה ראשון?

כתבו ליד כל אירוע מספר מ־1 עד 5 לפי סדר התרחשותו בסיפור.

□ הוּא נִקָּה אוֹתָם, מָתַח אוֹתָם וְיִבֵּשׁ אוֹתָם תַּחַת קַרְנֵי הַשֶּׁמֶשׁ, עַד שֶׁהָפְכוּ לִקְלָף אֲמִתִּי שֶׁל תּוֹרָה – חָלָק וְלָבָן.
□ יְלָדִים, דַּמְיְנוּ עַצְמְכֶם הוֹלְכִים בִּרְחוֹבוֹת שֶׁל עֲיָרָה קְטַנָּה בְּאֶרֶץ יִשְׂרָאֵל, לִפְנֵי כְּאַלְפַּיִם שָׁנָה.
□ הוּא מִתְאֲמֵּץ בְּעַצְמוֹ, עוֹשֶׂה גְּמִילוּת חֲסָדִים לִיתוֹמִים, וְלֹא שׁוֹמֵר אֶת הַיֶּדַע לְעַצְמוֹ, אֶלָּא דּוֹאֵג שֶׁכָּל אֶחָד יְלַמֵּד אֶת…
□ הוּא הִשְׁקָה אוֹתָם יְוֹם יוֹם, וּבִקֵּשׁ בַּקָּשָׁה חֲרִישִׁית מֵהַשֵּׁם יִתְבָּרַךְ שֶׁיַּצְלִיחַ אֶת דַּרְכּוֹ.
□ אַחַר כָּךְ הָלַךְ וְאָסַף שִׁשָּׁה יְלָדִים אֲחֵרִים, בּוֹגְרִים יוֹתֵר.

לפני ואחרי

מה קרה לפני: “הוּא נִקָּה אוֹתָם, מָתַח אוֹתָם וְיִבֵּשׁ אוֹתָם תַּחַת קַרְנֵי הַשֶּׁמֶשׁ, עַד שֶׁהָפְכוּ לִקְלָף אֲמִתִּי שֶׁל תּוֹרָה – חָלָק וְלָבָן.”?
מה קרה אחרי: “אַחַר כָּךְ הָלַךְ וְאָסַף שִׁשָּׁה יְלָדִים אֲחֵרִים, בּוֹגְרִים יוֹתֵר.”?

סיבה ותוצאה מתוך המעשה

האירוע הנתוןמה גרם לו?מה הייתה תוצאתו?
הַשֶּׁמֶשׁ חִמְּמָה, הַגֶּשֶׁם יָרַד, וְהַפִּשְׁתָּן צָמַח לְגִבְעוֹלִים אֲרֻכִּים וַחֲזָקִים.
הוּא הָלַךְ לִשְׁכוּנָה וְאָסַף חֲמִשָּׁה יְלָדִים, שֶׁמֵּעוֹלָם לֹא יָדְעוּ קְרוֹא וּכְתֹב.

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “כיצד לימד תורה בעיירות ללא מלמד — מנחמשים לחמשים” על ר' חייא? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “הַיְלָדִים אָכְלוּ אֶת הַבָּשָׂר הֶחָם, וְהַשִּׂמְחָה חָזְרָה לְעֵינֵיהֶם!”.
ג. אילו ר' חייא לא היה פועל כפי שמתואר ב־“לַשְּׁלִישִׁי לִמֵּד רַק אֶת 'וַיִּקְרָא', לָרְבִיעִי רַק 'בַּמִּדְבָּר', וְלַחֲמִישִׁי رַק אֶת 'דְּבָרִים'!” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “קָם בֵּית מִדְרָשׁ מָלֵא תּוֹרָה שֶׁאַף אֶחָד לֹא יָכוֹל הָיָה לְכַבּוֹת.” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. יְלָדִים, דַּמְיְנוּ עַצְמְכֶם הוֹלְכִים בִּרְחוֹבוֹת שֶׁל עֲיָרָה…
2. וְרַבִּי חִיָּיא – גְּדוֹל הַדּוֹר – הֶחְלִיט לַעֲשׂוֹת אֶת הַכֹּל…
3. הוּא בִּשֵּׁל אוֹתוֹ וְהֵכִין מָרָק מֵזִין, חַם וּמַהְבִּיל.
4. יָצָא רַבִּי חִיָּיא אֶל הָעֲיָרָה שֶׁלֹּא הָיָה בָּהּ מְלַמֵּד.
5. לִפְנֵי שֶׁרַבִּי חִיָּיא עָזַב אוֹתָם, הוּא חִיֵּךְ אֲלֵיהֶם וְהִנְחָה…
6. הוּא מִתְאֲמֵּץ בְּעַצְמוֹ, עוֹשֶׂה גְּמִילוּת חֲסָדִים לִיתוֹמִים,…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “הוּא פָּרַשׂ אֶת הָרֶשֶׁת בֵּין הָעֵצִים, וְצָד בָּהּ צְבָאִים בְּרִיאִים וְזְרִיזִים.”.
מכתב: כתבו לר' חייא בעקבות “הוּא הָלַךְ לִשְׁכוּנָה וְאָסַף חֲמִשָּׁה יְלָדִים, שֶׁמֵּעוֹלָם לֹא יָדְעוּ קְרוֹא וּכְתֹב.”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “לִפְנֵי שֶׁרַבִּי חִיָּיא עָזַב אוֹתָם, הוּא חִיֵּךְ אֲלֵיהֶם וְהִנְחָה אוֹתָם בַּהוֹרָאָה שֶׁשִּׁנְּתָה אֶת הָעוֹלָם: "יְלָדַי…”.
דיווח: כתבו ידיעה קצרה שכותרתה “כיצד לימד תורה בעיירות ללא מלמד — מנחמשים לחמשים”.
