העיר שקנה רבי עקיבא (בכספי ר' טרפון)

דף עבודה בעקבות רבי טרפון
שם: כיתה:

המעשה — הסיפור המלא

פְּתִיחָה: הָעֹשֶׁר הַגָּדוֹל שֶׁל רַבִּי טַרְפוֹן

יְלָדִים, נַסּוּ לְדַמְיֵן בְּדִמְיוֹנְכֶם אֶת הָאִישׁ הֶעָשִׁיר בְּיוֹתֵר שֶׁאַתֶּם יְכוֹלִים לַחְשֹׁב עָלָיו. אִישׁ שֶׁיֵּשׁ לוֹ שָׂדוֹת רְחָבִים, כְּרָמִים הָעֲמוּסִים בַּעֲנָבִים מְתוּקִים, וְאַרְגְּזֵי עֵץ שֶׁמִּתְפּוֹצְצִים מֵרֹב מַטְבְּעוֹת שֶׁל זָהָב טָהוֹר.

בִּתְקוּפַת הַתַּנָּאִים הַקְּדוֹשִׁים, חַי בָּעִיר לוֹד צַדִּיק עָצוּם, עֲנַק רוּחַ, כֹּהֵן גָּדוֹל בַּתּוֹרָה וּבַמִּדּוֹת. קָרְאוּ לוֹ רַבִּי טַרְפוֹן . מִלְּבַד הֱיוֹתוֹ גָּאוֹן מֻפְלָג בַּתּוֹרָה, רַבִּי טַרְפוֹן הָיָה גַּם עָשִׁיר מֻפְלָג. הָיָה לוֹ כֶּסֶף רַב.

יוֹם אֶחָד, הָלַךְ בִּרְחוֹבוֹת הָעִיר חֲבֵרוֹ הַטּוֹב, הַתַּנָּא הַקָּדוֹשׁ רַבִּי עֲקִיבָא . רַבִּי עֲקִיבָא, שֶׁעָבַר בֵּין סִמְטְאוֹתֶיהָ שֶׁל הָעִיר, שָׂם לֵב לַדַּלּוּת וְלָעֹנִי. הוּא הֵרִיחַ אֶת רֵיחַ הַטְּחָב הַיּוֹצֵא מִקִּירוֹת בָּתֵּיהֶם שֶׁל תַּלְמִידֵי חֲכָמִים. הוּא רָאָה מִשְׁפָּחוֹת אֲצִילוֹת שֶׁקָּשֶׁה לָהֶן לִקְנוֹת לֶחֶם לַשַּׁבָּת.

רַבִּי עֲקִיבָא יָדַע שֶׁרַבִּי טַרְפוֹן הוּא אָדָם טוֹב שֶׁנּוֹתֵן צְדָקָה, אַךְ כְּבַעַל רְאִיָּה עֲמוּקָה, בִּקֵּשׁ רַבִּי עֲקִיבָא לְזַכּוֹת אֶת חֲבֵרוֹ הָעָשִׁיר בִּזְכוּת גְּדוֹלָה הַרְבֵּה יוֹתֵר וּלְקַיֵּם "פַּזֵּר נָתַן לָאֶבְיוֹנִים". "אֲנִי חַיָּב לַעֲזֹר לְרַבִּי טַרְפוֹן לְהַשְׁקִיעַ אֶת הַכֶּסֶף בַּמָּקוֹם הַנָּכוֹן," חָשַׁב רַבִּי עֲקִיבָא, וְלִבּוֹ הִתְמַלֵּא בְּאַהֲבָה לַה' וּלְיִשְׂרָאֵל.

הַהַצָּעָה הַעִסְקִית

כַּאֲשֶׁר נִפְגְּשׁוּ שְׁנֵי עֲנָקֵי הַתּוֹרָה, חִיֵּךְ רַבִּי עֲקִיבָא וּפָנָה לְרַבִּי טַרְפוֹן בְּהַצָּעָה מְפַתָּה:

"רַבִּי," אָמַר רַבִּי עֲקִיבָא בְּקוֹל נָעִים, "יֵשׁ לִי עֲסָקָה מְצֻיֶּנֶת עֲבוּרְךָ. רְצוֹנְךָ שֶׁאֶקְנֶה לְךָ עִיר אַחַת אוֹ שְׁתַּיִם?"

רַבִּי טַרְפוֹן הִבִּיט בַּחֲבֵרוֹ. לִקְנוֹת עִיר! עִם שָׂדוֹת, עִם בָּתִּים, עִם תְּבוּאָה רַבָּה! אֵיזוֹ הַשְׁקָעָה מֻצְלַחַת. הוּא סָמַךְ עַל חָכְמָתוֹ שֶׁל רַבִּי עֲקִיבָא בְּעֵינַיִם עֲצוּמוֹת. "הֵן! בְּוַדַּאי!" פָּסַק רַבִּי טַרְפוֹן בְּשִׂמְחָה.

הוּא קָם מִמְּקוֹמוֹ, נִגַּשׁ לַחֲדַר הָאוֹצָרוֹת שֶׁלּוֹ, וְהוֹצִיא מִשָּׁם שַׂקִּים כְּבֵדִים. קְלִינְג... קְלִינְג... צְלִיל מַתַּכְתִּי וְכָבֵד הִדְהֵד בַּחֶדֶר כַּאֲשֶׁר אַרְבַּעַת אֲלָפִים דִּינְרֵי זָהָב הוּנְחוּ עַל הַשֻּׁלְחָן הֶעָבֶה. זֶהוּ סְכוּם עָצוּם שֶׁל כֶּסֶף! רַבִּי עֲקִיבָא עָמַס אֶת שַׂקֵּי הַזָּהָב עַל כְּתֵפָיו, בֵּרֵךְ אֶת רַבִּי טַרְפוֹן לְשָׁלוֹם, וְיָצָא לַדֶּרֶךְ.

הַהַשְׁקָעָה הַאֲמִתִּית

הַאִם רַבִּי עֲקִיבָא הָלַךְ לְסוֹחֲרֵי קַרְקָעוֹת? הַאִם שָׂכַר פּוֹעֲלִים שֶׁיִּבְנוּ חוֹמוֹת שֶׁל אֶבֶן?

לֹא. רַבִּי עֲקִיבָא שָׂם אֶת פְּעָמָיו אֶל רְחוֹבוֹתֶיהָ הַצְּדָדִיִּים שֶׁל הָעִיר, שָׁם הִתְגּוֹרְרוּ הָעֲנִיִּים וְהָאֶבְיוֹנִים וְתַלְמִידֵי הַחֲכָמִים. מִישֶׁהוּ מֵהֶם צָרִיךְ לְהַשִּׂיא אֶת בִּתּוֹ וְאֵין לוֹ פְּרוּטָה? רַבִּי עֲקִיבָא דּוֹפֵק בְּדַלְתּוֹ, פּוֹתֵחַ אֶת הַשַּׂק וּמַעֲנִיק לוֹ דִּינָרֵי זָהָב מוֹבְהִיקִים!

יְתוֹמִים שֶׁאֵין לָהֶם מַה לֶּאֱכֹל? תַּלְמִידֵי יְשִׁיבָה שֶׁאֵין לָהֶם בֶּגֶד לְהִתְחַמֵּם בּוֹ מִפְּנֵי הַקֹּר?

רַבִּי עֲקִיבָא לֹא נָח וְלֹא שָׁקַט. שָׁעָה אַחַר שָׁעָה, יוֹם אַחַר יוֹם, הוּא סִיֵּר בְּרַחֲבֵי תְּחוּם הַמּוֹשָׁב, מְשַׁמֵּחַ לְבָבוֹת, מַאֲכִיל רְעֵבִים, עַד שֶׁהַשַּׂקִים הִתְרוֹקְנוּ לְגַמְרֵי. אַרְבַּעַת אֲלָפִים דִּינְרֵי זָהָב הוּשְׁקְעוּ כֻּלָּם בְּמִצְוַת "צְדָקָה" קְדוֹשָׁה. הַהַשְׁגָּחָה הָעֶלְיוֹנָה רָאֲתָה אֶת הַחֶסֶד וְשָׂמְחָה בִּבְנֵי יִשְׂרָאֵל.

הֵיכָן הָעִיר שֶׁלִּי?

עָבְרוּ כַּמָּה יָמִים, וְרַבִּי טַרְפוֹן יָצָא לְטַיֵּל. הַשֶּׁמֶשׁ לִטְּפָה אֶת פָּנָיו בְּחֹם, וְהוּא פָּגַשׁ אֶת רַבִּי עֲקִיבָא הוֹלֵךְ לִקְרָאתוֹ.

"שָׁלוֹם עָלֶיךָ, עֲקִיבָא אָחִי!" שָׂמַח רַבִּי טַרְפוֹן, "הֵיכָן הָעֲיָרוֹת שֶׁלָּקַחְתָּ (שֶׁקָּנִיתָ) לִי? אֲנִי מְעֻנְיָן לִרְאוֹת אֶת הַשָּׂדוֹת וְאֶת הַבָּתִּים שֶׁהִשְׁקַעְתִּי בָּהֶם אֶת זְהָבִי."

"בְּוַדַּאי, רַבִּי," אָמַר רַבִּי עֲקִיבָא, וְעֵינָיו הַטְּהוֹרוֹת בָּרְקוּ. "בּוֹא אַחֲרַי וְאַרְאֶה לְךָ אֶת נְכָסֶיךָ."

רַבִּי טַרְפוֹן הָלַךְ אַחֲרֵי רַבִּי עֲקִיבָא. אֲבָל לְהַפְתַּעְתּוֹ, הֵם לֹא הָלְכוּ לְעֵבֶר הֶהָרִים הַיְרֻקִּים אוֹ לְעֵבֶר שְׁכוּנוֹת הַעֲשִׁירִים. רַבִּי עֲקִיבָא הוֹבִיל אוֹתוֹ יְשִׁירוֹת אֶל... מִבְנֶה הָאֶבֶן שֶׁל בֵּית הַמִּדְרָשׁ!

כְּשֶׁנִּכְנְסוּ, הִכָּה בָּהֶם רֵיחַ מוּכָּר שֶׁל קְלָפִים יְשָׁנִים וְרֵיחַ שֶׁל שֶׁמֶן זַיִת מִן הַמְּנוֹרוֹת הַדּוֹלְקוֹת. הָאֲוִיר הָיָה אָפוּף בִּקְדוּשָׁה. קוֹל שֶׁל לִמּוּד תּוֹרָה מָתוֹק וְעָרֵב נִשְׁמַע מִכָּל עֵבֶר. עֲשָׂרוֹת תִּינוֹקוֹת שֶׁל בֵּית רַבָּן יָשְׁבוּ וְשִׁנְּנוּ פְּסוּקִים.

רַבִּי טַרְפוֹן הָיָה מְבֻלְבָּל. מָה קָשׁוּר בֵּית הַמִּדְרָשׁ לָעִיר שֶׁקָּנָה?

הָאַרְמוֹן שֶׁל רַבִּי טַרְפוֹן

רַבִּי עֲקִיבָא לֹא אָמַר מִלָּה. הוּא תָּפַס בְּיָדוֹ שֶׁל רַבִּי טַרְפוֹן בַּעֲדִינוּת, וְהוֹבִיל אוֹתוֹ קָרוֹב לְיֶלֶד קָטָן, שֶׁיָּשַׁב בַּפִּנָּה עִם מְגִלַּת קְלָף פְּתוּחָה שֶׁל סֵפֶר מִשְׁלֵי אוֹ תְּהִלִּים.

"קְרָא, יַלְדִּי, קְרָא!" בִּקֵּשׁ רַבִּי עֲקִיבָא מֵהַיֶּלֶד בְּקוֹל רַךְ וְאוֹהֵב.

הַיֶּלֶד מָתַח אֶת קוֹלוֹ הַדַּק וְהַזַּךְ, וּבְנִגּוּן מָתוֹק הִתְחִיל לִקְרֹא אֶת מָה שֶׁהָיָה כָּתוּב אֶצְלוֹ בַּסֵּפֶר, מִתּוֹךְ פֶּרֶק קִי"ב בִּתְהִלִּים:

"פִּזַּר... נָתַן לָאֶבְיוֹנִים... צִדְקָתוֹ עוֹמֶדֶת לָעַד... קַרְנוֹ תָּרוּם בְּכָבוֹד!"

הַיֶּלֶד סִיֵּם, וְהַשֶּׁקֶט נָפַל בְּבֵית הַמִּדְרָשׁ.

רַבִּי עֲקִיבָא הִסְתּוֹבֵב אֶל רַבִּי טַרְפוֹן, הִסְתַּכֵּל יָשָׁר לְתוֹךְ עֵינָיו הַטּוֹבוֹת, וְאָמַר לוֹ מִשְׁפָּט שֶׁמְּהַדְהֵד עַד הַיּוֹם הַזֶּה:

"רַבִּי... זוֹ הִיא הָעִיר שֶׁקָּנִיתִי לְךָ!"

"כָּל הַזָּהָב שֶׁנָּתַתָּ לִי," הִמְשִׁיךְ רַבִּי עֲקִיבָא, "לֹא הָלַךְ עַל עֵצִים וַאֲבָנִים שֶׁעֲתִידִים לְהִשָּׁבֵר. הוּא חֻלַּק לַעֲנִיִּים וְלִלְמֵדֵי תּוֹרָה! קָנִיתִי לְךָ 'עִיר שֶׁל זְכֻיּוֹת' – צִדְקָה שֶׁעוֹמֶדֶת לָנֶצַח קַיֶּמֶת לָךְ לְעוֹלָם הַבָּא!"

יְלָדִים, אֵיךְ לְדַעְתְּכֶם הֵגִיב אִישׁ עֲסָקִים עָשִׁיר שֶׁגִּלָּה שֶׁמִּילְיוֹנִים שֶׁל שְׁקָלִים שֶׁלּוֹ הָלְכוּ לִצְדָקָה בְּלִי שֶׁהוּא יָדַע מֵרֹאשׁ? הוּא הָיָה אָמוּר לִכְעֹס? לִצְעֹק? לִדְרֹשׁ אֶת הַכֶּסֶף בַּחֲזָרָה?

אֲבָל רַבִּי טַרְפוֹן הָיָה קָדוֹשׁ יִשְׂרָאֵל. כַּאֲשֶׁר נָפְלָה בּוֹ הַהֲבָנָה הָעֲצוּמָה עַל מַה שֶּׁעָשָׂה עֲבוּרוֹ רַבִּי עֲקִיבָא, דְּמָעוֹת שֶׁל שִׂמְחָה עָמְדוּ בְּעֵינָיו. הוּא הִרְגִּישׁ אֵיךְ נַפְשׁוֹ זוֹרַחַת הִנֵּה הַהַשְׁגָּחָה וְהַזְּכוּת לִתְמֹךְ בַּעֲנִיֵּי עִמּוֹ.

הוּא קָם, נִגַּשׁ לְרַבִּי עֲקִיבָא, עָמַד מוּלוֹ כֻּלּוֹ נִרְגָּשׁ, וּנְשָׁקוֹ בְּחֹם!

וְאָז אָמַר לוֹ מִתּוֹךְ כָּבוֹד אֲמִתִּי: "רַבִּי, אַלּוּפִי! רַבִּי בְּחָכְמָה – וְאַלּוּפִי בְּדֶרֶךְ אֶרֶץ!"

וּכְדֵי לְהַרְאוֹת עַד כַּמָּה הוּא שָׂמַח – הוֹסִיף רַבִּי טַרְפוֹן וְנָתַן לְרַבִּי עֲקִיבָא עוֹד מָמוֹן וְזָהָב לַחֲלֹק לִצְדָקָה וּלְאֶבְיוֹנִים. כִּי הוּא יָדַע: מָה שֶׁנּוֹתְנִים לִצְדָקָה – נִשְׁאָר שֶׁלָּנוּ לָנֶצַח.

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “יְלָדִים, נַסּוּ לְדַמְיֵן בְּדִמְיוֹנְכֶם אֶת הָאִישׁ הֶעָשִׁיר בְּיוֹתֵר שֶׁאַתֶּם יְכוֹלִים לַחְשֹׁב עָלָיו.” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “הוּא רָאָה מִשְׁפָּחוֹת אֲצִילוֹת שֶׁקָּשֶׁה לָהֶן לִקְנוֹת לֶחֶם לַשַּׁבָּת.”?
ג. כיצד הגיב רבי טרפון למצב המתואר: “רַבִּי עֲקִיבָא עָמַס אֶת שַׂקֵּי הַזָּהָב עַל כְּתֵפָיו, בֵּרֵךְ אֶת רַבִּי טַרְפוֹן לְשָׁלוֹם, וְיָצָא לַדֶּרֶךְ.”?
ד. מה השתנה בין “אֲנִי מְעֻנְיָן לִרְאוֹת אֶת הַשָּׂדוֹת וְאֶת הַבָּתִּים שֶׁהִשְׁקַעְתִּי בָּהֶם אֶת זְהָבִי.” לבין “קְרָא, יַלְדִּי, קְרָא!" בִּקֵּשׁ רַבִּי עֲקִיבָא מֵהַיֶּלֶד בְּקוֹל רַךְ וְאוֹהֵב.”?
ה. כיצד הגיע הסיפור אל האירוע: “הוּא קָם, נִגַּשׁ לְרַבִּי עֲקִיבָא, עָמַד מוּלוֹ כֻּלּוֹ נִרְגָּשׁ, וּנְשָׁקוֹ בְּחֹם!”?

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “העיר שקנה רבי עקיבא (בכספי ר' טרפון)” על רבי טרפון? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “שָׁעָה אַחַר שָׁעָה, יוֹם אַחַר יוֹם, הוּא סִיֵּר בְּרַחֲבֵי תְּחוּם הַמּוֹשָׁב, מְשַׁמֵּחַ לְבָבוֹת, מַאֲכִיל רְעֵבִים, עַד שֶׁהַשַּׂקִים הִתְרוֹקְנוּ…”.
ג. אילו רבי טרפון לא היה פועל כפי שמתואר ב־“קְרָא, יַלְדִּי, קְרָא!" בִּקֵּשׁ רַבִּי עֲקִיבָא מֵהַיֶּלֶד בְּקוֹל רַךְ וְאוֹהֵב.” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “הוּא קָם, נִגַּשׁ לְרַבִּי עֲקִיבָא, עָמַד מוּלוֹ כֻּלּוֹ נִרְגָּשׁ, וּנְשָׁקוֹ בְּחֹם!” מחזק את המסר של הסיפור?
