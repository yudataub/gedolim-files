ארבע מאות זהובים של סבלנות

דף עבודה בעקבות ארבע מאות זהובים של סבלנות
שם: כיתה:

המעשה — הסיפור המלא

הסיפור מנוקד במלואו, על פי ההנחיות.

🔹 שָׁלָב 1: בֵּרוּר מַקְדִּים וְנִתּוּחַ (לַמּוֹרֶה/לַמְסַפֵּר)

1. שְׁלִיטָה עַצְמִית: כַּעַס הוּא בְּחִירָה, וְאֶפְשָׁר לְהַחְלִיט לֹא לִכְעֹס אַף פַּעַם.

2. מַבָּט חִיּוּבִי וֶאֱמוּנָה: הִלֵּל לֹא רָאָה בַּשְּׁאֵלוֹת "שְׁטֻיּוֹת", אֶלָּא פֶּתַח לְהִתְבּוֹנֵן כֵּיצַד הַקָּדוֹשׁ בָּרוּךְ הוּא בּוֹרֵא אֶת בְּנֵי הָאָדָם בְּהַתְאָמָה מֻשְׁלֶמֶת לַסְּבִיבָה שֶׁלָּהֶם.

🔹 שָׁלָב 2: הַסִּפּוּר הַמָּלֵא – "אַרְבַּע מֵאוֹת זְהוּבִים שֶׁל סַבְלָנוּת"

פְּתִיחָה: הַהִתְעָרְבוּת שֶׁבַּשּׁוּק יְלָדִים, אֲנַחְנוּ יוֹדְעִים שֶׁהִלֵּל הַזָּקֵן הָיָה נְשִׂיא יִשְׂרָאֵל. הָאִישׁ הֲכִי גָּדוֹל, הֲכִי חָכָם וַהֲכִי מְכֻבָּד בְּכָל אֶרֶץ יִשְׂרָאֵל! כָּל גְּדוֹלֵי יִשְׂרָאֵל הָיוּ עוֹמְדִים לְפָנָיו בְּיִרְאַת כָּבוֹד. אֲבָל לַמְרוֹת גְּדֻלָּתוֹ, הִלֵּל הָיָה כָּל כָּךְ עָנָיו וְסַבְלָן, עַד שֶׁבִּירוּשָׁלַיִם הָיוּ אוֹמְרִים עָלָיו: "אֶת הִלֵּל הַנָּשִׂיא – אִי אֶפְשָׁר לְהַכְעִיס!".

בְּאוֹתָם יָמִים, יָשְׁבוּ שְׁנֵי אֲנָשִׁים בָּרְחוֹב, נִקְרָא לָהֶם אַהֲרֹן וּזְבוּלֻן. אָמַר אַהֲרֹן: "הִלֵּל הַזָּקֵן מֵעוֹלָם לֹא הִתְרַגֵּז! הוּא תָּמִיד שָׁלֵו!". קָפַץ זְבוּלֻן וְאָמַר: "מָה פִּתְאוֹם! אֵין אָדָם בָּעוֹלָם שֶׁאִי אֶפְשָׁר לְהַכְעִיס. תֵּן לִי שָׁעָה אַחַת, וַאֲנִי אַרְגִּיז אוֹתוֹ שֶׁיִּצְעַק עָלַי!". "אֲנִי לֹא מַאֲמִין לְךָ," חִיֵּךְ אַהֲרֹן. "בּוֹא נִתְעָרֵב. אִם אַתָּה מַצְלִיחַ לְהַכְעִיס וּלְהַרְגִּיז אֶת הִלֵּל הַזָּקֵן, אֲנִי מְשַׁלֵּם לְךָ מִכַּסְפִּי... אַרְבַּע מֵאוֹת זוּז!" (אַרְבַּע מֵאוֹת זוּז זֶה סְכוּם כֶּסֶף עָצוּם, שֶׁאֶפְשָׁר לִקְנוֹת בּוֹ שָׂדוֹת וּבָתִּים!). זְבוּלֻן צָחַק וְשִׁפְשֵׁף אֶת יָדָיו: "הַכֶּסֶף הַזֶּה כְּבָר שֶׁלִּי! אֲנִי יוֹדֵעַ בְּדִיּוּק מָתַי לָלֶכֶת אֵלָיו".

הַהַטְרָדָה הָרִאשׁוֹנָה אוֹתוֹ יוֹם הָיָה עֶרֶב שַׁבָּת. הַשֶּׁמֶשׁ כְּבָר הִתְחִילָה לָרֶדֶת, הַזְּמַן לָחוּץ, כֻּלָּם מְמַהֲרִים לֶאֱפוֹת, לְבַשֵּׁל, לְנַקּוֹת וְלַעֲרֹךְ אֶת הַשֻּׁלְחָן. הִלֵּל הַזָּקֵן נִכְנַס לַמֶּרְחָץ וְהִתְחִיל לַחְפֹּף אֶת רֹאשׁוֹ כְּדֵי לִהְיוֹת נָקִי לִכְבוֹד שַׁבָּת הַמַּלְכָּה. פִּתְאוֹם, נִשְׁמְעוּ דְּפִיקוֹת פֶּרְאִיּוֹת עַל הַדֶּלֶת הַחִיצוֹנִית, וְקוֹל חָצוּף זָעַק בָּרְחוֹב: "מִי כָּאן הִלֵּל?! מִי כָּאן הִלֵּל?!" (בְּלִי תֹּאַר "כְּבוֹד הָרַב", בְּלִי תֹּאַר "הַנָּשִׂיא"). אָדָם רָגִיל הָיָה צוֹעֵק מִבִּפְנִים: "אֲנִי מִתְרַחֵץ! תַּחְזְרוּ בְּיוֹם רִאשׁוֹן!". אֲבָל הִלֵּל הַזָּקֵן הָיָה מָלֵא בְּאַהֲבַת יִשְׂרָאֵל. "אוּלַי לַיְּהוּדִי הַזֶּה יֵשׁ שְׁאֵלָה הִלְכָתִית דְּחוּפָה לִכְבוֹד שַׁבָּת?" חָשַׁב. הוּא עָזַב הַכֹּל, הִתְעַטֵּף מַהֵר בִּמְעִילוֹ כְּדֵי לָצֵאת בְּדֶרֶךְ אֶרֶץ, פָּתַח אֶת הַדֶּלֶת, וְשָׁאַל בְּקּוֹל רַךְ וְנָעִים: "בְּנִי, מָה אַתָּה מְבַקֵּשׁ?". אָמַר לוֹ זְבוּלֻן: "יֵשׁ לִי שְׁאֵלָה לִשְׁאֹל!". "שְׁאַל בְּנִי, שְׁאַל!" הֵשִׁיב הִלֵּל, מוּכָן לִשְׁמֹעַ דִּבְרֵי תּוֹרָה. שָׁאַל זְבוּלֻן שְׁאֵלָה מַצְחִיקָה שֶׁאֵין לָהּ שׁוּם קֶשֶׁר לְשַׁבָּת: "תַּגִּיד, לָמָּה הָרֹאשׁ שֶׁל הָאֲנָשִׁים שֶׁגָּרִים בְּבָבֶל הוּא אָרֹךְ כְּמוֹ בֵּיצָה, וְלֹא עָגֹל כְּמוֹ שֶׁלָּנוּ?". הִלֵּל לֹא כָּעַס. הַפָּנִים שֶׁלּוֹ שָׁפְעוּ אוֹר, וְהוּא אָמַר: "בְּנִי! שְׁאֵלָה גְּדוֹלָה שָׁאַלְתָּ! מִפְּנֵי שֶׁהַמְיַלְּדוֹת (הָאֲחָיוֹת שֶׁעוֹזְרוֹת לַתִּינוֹקוֹת לְהִוָּלֵד) בְּבָבֶל אֵינָן זְרִיזוֹת וַחֲכָמוֹת כְּמוֹ בְּאֶרֶץ יִשְׂרָאֵל, וְלָכֵן הָרֹאשׁ שֶׁל הַתִּינוֹק מִתְאָרֵךְ שָׁם". זְבוּלֻן הִנְהֵן מְאֻכְזָב, וְהָלַךְ.

שׁוּב וָשׁוּב עָבְרָה שָׁעָה קַלָּה. שַׁבָּת עוֹד רֶגַע נִכְנֶסֶת. הִלֵּל שׁוּב נִכְנַס לַמֶּרְחָץ. וְשׁוּב – דְּפִיקוֹת זַעַם! "מִי כָּאן הִלֵּל?! מִי כָּאן הִלֵּל?!". הִלֵּל הַזָּקֵן שׁוּב מִתְעַטֵּף, יוֹצֵא הַחוּצָה וְאוֹמֵר בְּחִיּוּךְ: "בְּנִי, מָה אַתָּה מְבַקֵּשׁ?". "שְׁאֵלָה יֵשׁ לִי!" "שְׁאַל בְּנִי, שְׁאַל!" זְבוּלֻן נִסָּה מַשֶּׁהוּ מַעֲצְבֵּן עוֹד יוֹתֵר: "מִפְּנֵי מָה עֵינֵיהֶם שֶׁל תַּרְמוֹדִיִּים (אֲנָשִׁים הַגָּרִים בְּעִיר שֶׁשְּׁמָהּ תַּרְמוֹד) הֵן צָרוֹת וַעֲצוּמוֹת כִּמְעַט כָּל הַזְּמַן?". חִיֵּךְ הִלֵּל בְּאַהֲבָה לַשֵּׁם יִתְבָּרַךְ, וְעָנָה: "בְּנִי, שְׁאֵלָה עֲצוּמָה שָׁאַלְתָּ! תִּרְאֶה כַּמָּה הַקָּדוֹשׁ בָּרוּךְ הוּא דּוֹאֵג לַבְּרִיּוֹת שֶׁלּוֹ: הָאֲנָשִׁים הָאֵלֶּה גָּרִים בְּלֵב הַמִּדְבָּר. יֵשׁ שָׁם רוּחוֹת שֶׁמְּעִיפוֹת מְלֵא חוֹל וְאָבָק. אִם עֵינֵיהֶם הָיוּ פְּתוּחוֹת וּגְדוֹלוֹת, הֵם הָיוּ מִתְעַוְּרִים מֵהַחוֹל! לָכֵן ה' בָּרָא אוֹתָם עִם סֶדֶק קָטָן בָּעֵינַיִם, שֶׁיּוּכְלוּ לִרְאוֹת וְלֹא יִפָּגְעוּ!".

זְבוּלֻן כְּבָר הִתְחִיל לְהִלָּחֵץ עַל הַכֶּסֶף שֶׁלּוֹ. הוּא הָלַךְ, חִכָּה לָרֶגַע הַלָּחוּץ בְּיוֹתֵר כְּשֶׁהַשֶּׁמֶשׁ כִּמְעַט שָׁקְעָה, וְצָעַק שׁוּב: "מִי כָּאן הִלֵּל?! מִי כָּאן הִלֵּל?!". הִלֵּל שׁוּב יָצָא בְּנַחַת. "בְּנִי, מָה אַתָּה מְבַקֵּשׁ?" "לָמָּה שֶׁל הָאֲנָשִׁים בְּאַפְרִיקָה רַגְלֵיהֶם כָּל כָּךְ רְחָבוֹת וּפְחוּסוֹת?" "בְּנִי, שְׁאֵלָה נִפְלָאָה! מִפְּנֵי שֶׁהֵם גָּרִים בְּאֵזוֹר שֶׁמָּלֵא בְּבִצּוֹת וּבוֹץ הַרְבֵּה. אִם הָיוּ לָהֶם רַגְלַיִם דַּקּוֹת – הֵם הָיוּ שׁוֹקְעִים בַּבּוֹץ. ה' בָּרָא לָהֶם רַגְלַיִם רְחָבוֹת כְּדֵי שֶׁיּוּכְלוּ לִדְרֹךְ עַל הַמַּיִם בְּלִי לִשְׁקֹעַ!".

הַהֶפְסֵד שֶׁל זְבוּלֻן זְבוּלֻן תָּלַשׁ שְׂעָרוֹת מֵרֹאשׁוֹ. הוּא הִבִּיט עַל הִלֵּל הַזָּקֵן שֶׁעָמַד שָׁלֵו וְרָגוּעַ. שׁוּם דָּבָר לֹא הִכְעִיס אוֹתוֹ! זְבוּלֻן תָּקַע בּוֹ מַבָּט זַעְמָנִי וְאָמַר: "יֵשׁ לִי עוֹד הַרְבֵּה שְׁאֵלוֹת, אֲבָל אֲנִי מְפַחֵד שֶׁתִּכְעַס עָלַי!". הִלֵּל כָּרַךְ אֶת מְעִילוֹ לִכְבוֹדוֹ שֶׁל הָאִישׁ הַפָּשׁוּט הַזֶּה, הִתְיַשֵּׁב מוּלוֹ וְאָמַר: "לֹא, מַה פִּתְאוֹם. כָּל מָה שֶׁיֵּשׁ לְךָ לִשְׁאֹל – שְׁאַל!". "תַּגִּיד," אָמַר זְבוּלֻן מִבַּעַד לְשִׁנָּיו, "אַתָּה הִלֵּל? זֶה שֶׁקּוֹרְאִים אוֹתוֹ 'נְשִׂיא יִשְׂרָאֵל'?". "כֵּן, בְּנִי." "אִם כָּךְ," צָעַק זְבוּלֻן בְּחֻצְפָּה נוֹרָאָה, "הַלְוַאי שֶׁלֹּא יִרְבּוּ כְּמוֹתְךָ בְּיִשְׂרָאֵל!" אֲפִלּוּ אָז הִלֵּל לֹא אִבֵּד אֶת קוֹר הָרוּחַ שֶׁלּוֹ. בְּעֵינַיִם רַכּוֹת הוּא שָׁאַל: "לָמָּה אַתָּה אָמוּר כָּךְ, בְּנִי? לָמָּה אַתָּה מְאַחֵל לִי רַע?". יומן: כתבו מנקודת מבטו של מי שראה את האירוע “הַשֶּׁמֶשׁ כְּבָר הִתְחִילָה לָרֶדֶת, הַזְּמַן לָחוּץ, כֻּלָּם מְמַהֲרִים לֶאֱפוֹת, לְבַשֵּׁל, לְנַקּוֹת וְלַעֲרֹךְ אֶת…”.
מכתב: כתבו לארבע מאות זהובים של סבלנות בעקבות “מִפְּנֵי שֶׁהֵם גָּרִים בְּאֵזוֹר שֶׁמָּלֵא בְּבִצּוֹת וּבוֹץ הַרְבֵּה.”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “כֵּן, בְּנִי." "אִם כָּךְ," צָעַק זְבוּלֻן בְּחֻצְפָּה נוֹרָאָה, "הַלְוַאי שֶׁלֹּא יִרְבּוּ כְּמוֹתְךָ בְּיִשְׂרָאֵל!" אֲפִלּוּ…”.
דיווח: כתבו ידיעה קצרה שכותרתה “ארבע מאות זהובים של סבלנות”.

המחזה

רגע מן הסיפורהדמויותמה יאמרו או יעשו?
אִם אַתָּה מַצְלִיחַ לְהַכְעִיס וּלְהַרְגִּיז אֶת הִלֵּל הַזָּקֵן, אֲנִי מְשַׁלֵּם לְךָ…
הוּא הִבִּיט עַל הִלֵּל הַזָּקֵן שֶׁעָמַד שָׁלֵו וְרָגוּעַ.
