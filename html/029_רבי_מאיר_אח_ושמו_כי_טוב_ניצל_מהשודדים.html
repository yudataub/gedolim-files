אח ושמו "כי טוב" — ניצל מהשודדים

דף עבודה בעקבות רבי מאיר
שם: כיתה:

המעשה — הסיפור המלא

להלן מערך חינוכי, רגשי ומרתק על רבי מאיר (והאח "כי טוב") , מעובד במיוחד לבנים בכיתות ב'-ד' (חינוך דתי-לאומי). הסיפור מדגיש את כוחה של חכמת התורה, האמונה התמימה בהשגחה פרטית, וכיצד אור התורה מגרש את חושך הפחד והסכנה.

הסיפור מנוקד במלואו, עשיר בתיאורים חושיים, ובנוי לפי כללי הסטוריטלינג כפי שביקשת.

פְּתִיחָה: הָאִישׁ שֶׁפָּנָיו הֵאִירוּ

יְלָדִים, הַאִם רְאִיתֶם פַּעַם אָדָם שֶׁהַפָּנִים שֶׁלּוֹ מַמָּשׁ זוֹרְחוֹת כְּמוֹ שֶׁמֶשׁ?

בִּימֵי חֲכָמֵינוּ זִכְרוֹנָם לִבְרָכָה, חַי בְּאֶרֶץ יִשְׂרָאֵל חָכָם גָּדוֹל וְעָצוּם, שֶׁשְּׁמוֹ הָאֲמִתִּי הָיָה 'רַבִּי נְהוֹרַאי'. אֲבָל אַף אֶחָד כִּמְעַט לֹא קָרָא לוֹ כָּךְ. כֻּלָּם קָרְאוּ לוֹ: רַבִּי מֵאִיר .

וְלָמָּה קָרְאוּ לוֹ מֵאִיר? כִּי מִי שֶׁהָיָה נִכְנָס לְבֵית הַמִּדְרָשׁ וְשׁוֹמֵעַ אֶת הַקּוֹל הַמָּתוֹק שֶׁלּוֹ מַסְבִּיר הֲלָכָה קָשָׁה בַּתּוֹרָה, הָיָה מַרְגִּישׁ שֶׁאֲפִלּוּ בַּלַּיְלָה נִהְיָה פִּתְאוֹם אוֹר גָּדוֹל בַּחֶדֶר. הוּא הָיָה מֵאִיר אֶת הָעֵינַיִם שֶׁל חַכְמֵי יִשְׂרָאֵל מִתּוֹךְ הַתּוֹרָה הַקְּדוֹשָׁה. הַתּוֹרָה הָיְתָה כָּל הַחַיִּים שֶׁלּוֹ.

חֲשֵׁכָה בְּאֶרֶץ זָרָה

יוֹם אֶחָד, רַבִּי מֵאִיר נֶאֱלַץ לָצֵאת לְמַסָּע רָחוֹק מְאֹד מֵאֶרֶץ יִשְׂרָאֵל, לְמָקוֹם שֶׁנִּקְרָא 'אַסְיָא'.

הַדֶּרֶךְ הָיְתָה מְסֻכֶּנֶת. הָיָה זֶה בְּאֶמְצַע הַחֹרֶף. הַשֶּׁמֶשׁ שָׁקְעָה מַהֵר מִן הָרָגִיל, וַחֲשֵׁכָה כְּבֵדָה עָטְפָה אֶת הָעוֹלָם. רַבִּי מֵאִיר מָצָא אֶת עַצְמוֹ צוֹעֵד בְּתוֹךְ יַעַר עָבוֹת.

עֲנָפִים שְׁחוֹרִים שֶׁל עֵצִים עֲנָקִיִּים חָסְמוּ אֶת אוֹר הַיָּרֵחַ. רוּחַ קָרָה וַחֲזָקָה שָׁרְקָה בֵּין הֶעָלִים – ששששׁ... הוווו... – עֲלֵי שַׁלֶּכֶת יְבֵשִׁים נִרְמְסוּ תַּחַת רַגְלָיו בְּרַעַשׁ שֶׁל פְּרִיק... פְּרָאק... הָאֲוִיר הָיָה קַר וְדָחוּס, וְרֵיחַ כָּבֵד שֶׁל אֲדָמָה רְטֻבָּה וּבוֹץ מִלֵּא אֶת אַפּוֹ.

אֲנָשִׁים בְּאוֹתוֹ זְמַן הָיוּ רוֹעֲדִים מִפַּחַד לָלֶכֶת בַּיַּעַר בַּלַּיְלָה. לָמָּה? כִּי מִלְּבַד זְאֵבִים וְדֻבִּים, הַיְּעָרוֹת שָׁרְצוּ גַּנָּבִים וְשׁוֹדְדִים שֶׁהָיוּ תּוֹקְפִים עוֹבְרֵי אוֹרַח. אֲבָל רַבִּי מֵאִיר לֹא פָּחַד. הוּא הָלַךְ בְּנַחַת, צַעַד אַחַר צַעַד, וּבְלִבּוֹ לָחַשׁ דִּבְרֵי תוֹרָה. תּוֹרָתָם שֶׁל אֲבוֹתֵינוּ שָׁמְרָה עָלָיו, וְהוּא יָדַע שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא צוֹעֵד אִתּוֹ.

הַמַּאֲרָב הַמַּפְחִיד

לְפֶתַע... צְלִיל חַד קָרַע אֶת הַשֶּׁקֶט!

שְׁלִיף!

בָּרָק כַּסְפִּי הִבְזִיק בַּחֹשֶׁךְ. חֶרֶב גְּדוֹלָה נִשְׁלְפָה.

מִתּוֹךְ הַשִּׂיחִים קָפְצוּ לְפֶתַע דְּמֻיּוֹת שְׁחוֹרוֹת. הָיוּ אֵלֶּה חֲמִשָּׁה שׁוֹדְדִים גְּדוֹלִים וַחֲסֻנִּים. הֵם הָיוּ לְבוּשִׁים בִּבְגָדִים בְּלוּיִים וּמְלֻכְלָכִים, פְּנֵיהֶם מְכֻסּוֹת בִּבְדַל בַּד, וְרַק עֵינֵיהֶם אָצְוּ מִתּוֹךְ הַחֲשֵׁכָה, מְלֵאוֹת רֹעַ וְתַאֲוַת בֶּצַע. בִּידֵיהֶם הֵם הֶחְזִיקוּ חֲרָבוֹת וְסַכִּינִים. הֵם הִקִּיפוּ אֶת רַבִּי מֵאִיר מִכָּל הַכִּוּוּנִים, כְּדֵי שֶׁלֹּא יוּכַל לִבְרֹחַ.

"עֲצֹר מִיָּד, יְהוּדִי!" נָהַם מַנְהִיג הַשּׁוֹדְדִים בְּקוֹל צָרוּד שֶׁנִּשְׁמַע כְּמוֹ אֶבֶן הַנִּגְרֶרֶת עַל כְּבִישׁ. "תֵּן לָנוּ מִיָּד אֶת הַכֶּסֶף שֶׁלְּךָ! אֶת הַבְּגָדִים! אֶת הַכֹּל! וְאִם תִּתְנַגֵּד – זֶה יִהְיֶה יוֹמְךָ הָאַחֲרוֹן בָּעוֹלָם!".

יְלָדִים, תְּאֲרוּ לָכֶם – מָה עָשָׂה רַבִּי מֵאִיר? רָעַד? הֵרִים יָדַיִם? צָעַק לְעֶזְרָה הַצִּילוּ?

לֹא וְלֹא.

רַבִּי מֵאִיר עָמַד זָקוּף. הַפָּנִים הַקְּדוֹשׁוֹת שֶׁלּוֹ הָיוּ שְׁלֵווֹת מַמָּשׁ. הוּא הִסְתַּכֵּל יָשָׁר לְתוֹךְ עֵינָיו שֶׁל הַשּׁוֹדֵד, וּמִתּוֹךְ קְדֻשָּׁה עֲצוּמָה וּבִטָּחוֹן בַּה', צָץ בְּמֹחוֹ רַעְיוֹן פֶּלֶא שֶׁהִגִּיעַ יְשִׁירוֹת מִפְּסוּקֵי הַתּוֹרָה.

תִּזָּהֲרוּ מֵאָחִי!

"אַתֶּם חוֹשְׁבִים שֶׁאֲנִי לְבַד?" שָׁאַל רַבִּי מֵאִיר בְּקוֹל שָׁקֵט וְרוֹגֵעַ, קוֹל שֶׁהִבְלִיט אֶת הַבִּטָּחוֹן שֶׁלּוֹ. "תֵּדְעוּ לָכֶם שֶׁאַתֶּם טוֹעִים. אֲנִי בִּכְלָל לֹא צוֹעֵד לְבַד בַּיַּעַר הַזֶּה. יֵשׁ לִי אָח עֲנָק... וְהוּא אִישׁ חָזָק וְגִבּוֹר עָצוּם. אִם תִּגְּעוּ בִּי, אוֹ אֲפִלּוּ בְּשַׂעֲרָה מִשַּׂעֲרוֹת רֹאשִׁי, אָחִי יָבוֹא וְיַשְׁמִיד אֶתְכֶם מִיָּד. הוּא אָמַר לִי לְהַמְתִּין לוֹ הֵיטֵב, כִּי הוּא מַמָּשׁ קָרוֹב אֵלַי וְהוּא תֵּכֶף מַגִּיעַ!".

הַשּׁוֹדְדִים שָׁמְעוּ אֶת הַבִּטָּחוֹן בַּקּוֹל שֶׁל הָרַב. אָדָם שֶׁעוֹמֵד לָמוּת לֹא מְדַבֵּר כָּךְ. הֵם נִבְהֲלוּ! הֵם הִבִּיטוּ לַצְּדָדִים בְּפַחַד אֶל תּוֹךְ הָעֵצִים הַחֲשׁוּכִים. אִם יֵשׁ לַיְּהוּדִי הַזֶּה אָח עֲנָק, מְסֻכָּן וְאַכְזָר – קָרוֹב בַּיַּעַר, הֵם מַעֲדִיפִים לַחֲכוֹת. אוּלַי יִשְׁדְּדוּ אֶת שְׁנֵיהֶם בְּיַחַד בְּלִי הַפְתָּעוֹת.

"מָה הַשֵּׁם שֶׁל הָאָח הַגִּבּוֹר שֶׁלְּךָ?" חָקַר מַנְהִיג הַשּׁוֹדְדִים, כְּשֶׁהוּא לֹא מַחֲזִיר אֶת חַרְבּוֹ לַנָּרְתִּיק אֶלָּא מַחְזִיק אוֹתָהּ בְּכּוֹנְנוּת.

רַבִּי מֵאִיר חִיֵּךְ חִיּוּךְ זָעִיר, חִיּוּךְ שֶׁל קְדֻשָּׁה וֶאֱמוּנָה. "קְוֹרְאִים לְאָחִי... ' כִּי טוֹב '," הוּא הֵשִׁיב בְּנַחַת.

הַשּׁוֹדְדִים הִנְהֲנוּ. אֵיזֶה שֵׁם מוּזָר לְלוֹחֵם... "כִּי טוֹב". הֵם הֵבִיאוּ חֲבָלִים עָבִים, הִכְרִיחוּ אֶת רַבִּי מֵאִיר לָשֶׁבֶת עַל גֶּזַע עֵץ כָּרוּת בְּאֶמְצַע הַיַּעַר, וְהִתְיַשְּׁבוּ לְיָדוֹ, מַחְזִיקִים אֶת כְּלֵי הַנֶּשֶׁק שֶׁלָּהֶם וּמַמְתִּינִים בִּדְרִיכוּת.

תְּפִלָּה בַּחֲשֵׁכָה וְהוֹפָעַת הָ"אָח"

יְלָדִים, שָׁעוֹת אֲרֻכּוֹת, שָׁעוֹת שֶׁל לַיְלָה קַר וְשָׁחֹר, יָשְׁבוּ הַשּׁוֹדְדִים וְרָעֲדוּ מִקֹּר. הֵם צִפּוּ שֶׁמִּתּוֹךְ הָעֵצִים יִקְפֹּץ אֵיזֶה עֲנָק עִם חֶרֶב גְּדוֹלָה שֶׁיִּקְרְאוּ לוֹ "כִּי טוֹב".

וּמָה עָשָׂה רַבִּי מֵאִיר בַּזְּמַן הַזֶּה? הַאִם בָּכָה? הַאִם רָעַד? כְּלָל לֹא.

רַבִּי מֵאִיר יָשַׁב, עָצַם אֶת עֵינָיו, וּשְׁפָתָיו הֵחֵלּוּ לָנּוּעַ. בְּקּוֹל מָתוֹק וְשָׁקֵט הוּא יָשַׁב וְלָמַד תּוֹרָה בְּעַל-פֶּה וְהִתְפַּלֵּל לְבוֹרֵא עוֹלָם. הוּא יָדַע מִיהוּ בֶּאֱמֶת הָ"אָח" שֶׁלּוֹ.

לְאַט לְאַט, צִפּוֹרִים רִאשׁוֹנוֹת הֵחֵלּוּ לְצַיֵּץ בַּיַּעַר. רֵיחַ רַעֲנָן וְטָהוֹר שֶׁל בֹּקֶר הִתְפַּשֵּׁט בָּאֲוִיר. הַשָּׁמַיִם הַשְּׁחֹרִים הִתְחִילוּ לְקַבֵּל צֶבַע וָרֹד-סְגַלְגַּל, וְאָז – כַּדּוּר זָהָב גָּדוֹל, זוֹרֵחַ וּמַבְרִיק, פָּרַץ אֶת קַו הָאֹפֶק. הַשֶּׁמֶשׁ יָצְאָה! מִיָּד נֶעֶלְמוּ כָּל הַצְּלָלִים הַמַּפְחִידִים שֶׁל הַלַּיְלָה. קַרְנֵי שֶׁמֶשׁ מוּזְהָבוֹת וַחֲמִימוֹת חָדְרוּ מִבַּעַד לָעֵצִים וְלִטְּפוּ אֶת פָּנָיו הַזּוֹרְחוֹת שֶׁל רַבִּי מֵאִיר.

הַשּׁוֹדְדִים הָיוּ עֲיֵפִים, רְעֵבִים, וְעַכְשָׁו גַּם כּוֹעֲסִים מְאֹד.

"הֵי, אַתָּה תְּרַמֶּה אוֹתָנוּ?!" צָעַק מַנְהִיג הַכְּנוּפְיָה כְּשֶׁהוּא קָם עַל רַגְלָיו. הַלַּיְלָה נִגְמַר וְהַבֹּקֶר עָלָה. "אֵיפֹה הָאָח הַגִּבּוֹר שֶׁלְּךָ?! חִכִּינוּ לוֹ כָּל הַלַּיְלָה, וְאַף אֶחָד לֹא בָּא! אֵיפֹה הָאָח הַזֶּה שֶׁנִּקְרָא 'כִּי טוֹב'?!"

רַבִּי מֵאִיר פָּקַח אֶת עֵינָיו הַטְּהוֹרוֹת. הוּא הִצְבִּיעַ בְּאֶצְבָּעוֹ עַל הַשֶּׁמֶשׁ הָעוֹלָה, עַל שְׁמֵי הַבֹּקֶר הַכְּחֻלִּים, עַל אוֹר הַיּוֹם הַגָּדוֹל שֶׁגֵּרֵשׁ אֶת הַחֹשֶׁךְ.

"הִנֵּה הוּא," אָמַר רַבִּי מֵאִיר בְּקוֹל רוֹעֵד מֵהִתְרַגְּשׁוּת שֶׁל קְדֻשָּׁה. "הִנֵּה אָחִי. זֶה אוֹר הַבֹּקֶר! הֲרֵי בַּתּוֹרָה הַקְּדוֹשָׁה, בְּסֵפֶר בְּרֵאשִׁית נֶאֱמַר: ' וַיַּרְא אֱלֹקִים אֶת הָאוֹר – כִּי טוֹב !'. הָאוֹר הוּא "כִּי טוֹב"! וְעַכְשָׁו, כְּשֶׁהֵאִיר הַיּוֹם וַאֲנָשִׁים הוֹלְכִים בַּדְּרָכִים, אַתֶּם תִּפְחֲדוּ לִגַּעַת בִּי שֶׁמָּא יִרְאוּ אֶתְכֶם!".

הַשּׁוֹדְדִים נִדְהֲמוּ מֵחָכְמָתוֹ הַכַּבִּירָה שֶׁל הַיְהוּדִי וּמִבִּטְחוֹנוֹ בֶּאֱלֹקָיו. בְּאוֹתוֹ רֶגַע נִשְׁמְעוּ קוֹלוֹת שֶׁל שַׁיֶּרֶת סוֹחֲרִים שֶׁהִתְקָרְבָה לַמָּקוֹם לְאוֹר הַיּוֹם. הַשּׁוֹדְדִים תָּפְסוּ אֶת נִשְׁקָם, וּבְבֶהָלָה גְּדוֹלָה בָּרְחוּ אֶל מַעֲבֵה הַיַּעַר וְנֶעֶלְמוּ.

רַבִּי מֵאִיר נָשָׂא עֵינָיו לַשָּׁמַיִם וְהוֹדָה לְבוֹרֵא עוֹלָם שֶׁהִצִּיל אוֹתוֹ בְּהַשְׁגָּחָה פְּרָטִית מֻפְלָאָה, בִּזְכוּת חָכְמַת הַתּוֹרָה!

הוא (ME): רַבִּי מֵאִיר מָצָא אֶת עַצְמוֹ בַּסַּכָּנָה הֲכִי גְּדוֹלָה שֶׁאֶפְשָׁר לְדַמְיֵן – לְבַד, בַּחֹשֶׁךְ, מֻקָּף בְּשׁוֹדְדִים. בִּמְקוֹם לְהִתְמַלֵּא בְּהִיסְטֶרְיָה וּפַחַד, הוּא נִזְכַּר בַּתּוֹרָה וְיָדַע שֶׁה' אִתּוֹ תָּמִיד, מֵאִיר לוֹ כְּמוֹ פָּנָס גָּדוֹל שֶׁל תִּקְוָה.

אֲנַחְנוּ (WE): לִפְעָמִים אֲנַחְנוּ לְבַד בַּחֶדֶר וּפִתְאוֹם חָשׁוּךְ לָנוּ אוֹ מַפְחִיד. לִפְעָמִים אֲנַחְנוּ נִתְקָלִים בְּיֶלֶד שֶׁמֵּצִיק לָנוּ, בְּבֶהָלָה, אוֹ בִּבְעָיָה שֶׁנִּרְאֵית גְּדוֹלָה עָלֵינוּ, וַאֲנַחְנוּ רוֹצִים מִיָּד לִפְרֹץ בִּבְכִי אוֹ לְהֵילָחֵם בְּכַעַס וּבְפַחַד.

אַתֶּם (YOU): יְלָדִים מְתוּקִים, מָחָר בַּבֹּקֶר שֶׁקָּשֶׁה לָכֶם, אוֹ כְּשֶׁאַתֶּם קְצָת פּוֹחֲדִים, תִּזְכְּרוּ אֶת הָ"אָח" שֶׁל רַבִּי מֵאִיר. תֵּדְעוּ שֶׁאִם רַק תַּחְשְׁבוּ עַל מַשֶּׁהוּ קָדוֹשׁ שֶׁלְמַדְתֶּם בַּתּוֹרָה, עַל אֵיזֶה פָּסוּק יָפֶה אוֹ בְּרָכָה טוֹבָה – פִּתְאוֹם "כִּי טוֹב" יָאִיר לָכֶם, וְהַקָּדוֹשׁ בָּרוּךְ הוּא – שֶׁהוּא אֲבִינוּ הַגָּדוֹל שֶׁבַּשָּׁמַיִם – יִשְׁמֹר עֲלֵיכֶם וִיגָרֵשׁ מִכֶּם אֶת כָּל הַחֹשֶׁךְ וְהַפַּחַד.

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “אח ושמו "כי טוב" — ניצל מהשודדים” על רבי מאיר? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “אִם תִּגְּעוּ בִּי, אוֹ אֲפִלּוּ בְּשַׂעֲרָה מִשַּׂעֲרוֹת רֹאשִׁי, אָחִי יָבוֹא וְיַשְׁמִיד אֶתְכֶם מִיָּד.”.
ג. אילו רבי מאיר לא היה פועל כפי שמתואר ב־“הוּא הִצְבִּיעַ בְּאֶצְבָּעוֹ עַל הַשֶּׁמֶשׁ הָעוֹלָה, עַל שְׁמֵי הַבֹּקֶר הַכְּחֻלִּים, עַל אוֹר הַיּוֹם הַגָּדוֹל שֶׁגֵּרֵשׁ…” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “בִּמְקוֹם לְהִתְמַלֵּא בְּהִיסְטֶרְיָה וּפַחַד, הוּא נִזְכַּר בַּתּוֹרָה וְיָדַע שֶׁה' אִתּוֹ תָּמִיד, מֵאִיר לוֹ כְּמוֹ פָּנָס…” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. להלן מערך חינוכי, רגשי ומרתק על רבי מאיר (והאח "כי טוב") , מעובד במיוחד…
2. רוּחַ קָרָה וַחֲזָקָה שָׁרְקָה בֵּין הֶעָלִים – ששששׁ...
3. עֲצֹר מִיָּד, יְהוּדִי!" נָהַם מַנְהִיג הַשּׁוֹדְדִים בְּקוֹל צָרוּד…
4. הֵם צִפּוּ שֶׁמִּתּוֹךְ הָעֵצִים יִקְפֹּץ אֵיזֶה עֲנָק עִם חֶרֶב…
5. וְעַכְשָׁו, כְּשֶׁהֵאִיר הַיּוֹם וַאֲנָשִׁים הוֹלְכִים בַּדְּרָכִים,…
6. היום נשמע על אחד מגדולי התנאים שלנו, שמצא את עצמו בחושך מוחלט ומפחיד…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “הֵם הָיוּ לְבוּשִׁים בִּבְגָדִים בְּלוּיִים וּמְלֻכְלָכִים, פְּנֵיהֶם מְכֻסּוֹת בִּבְדַל בַּד, וְרַק עֵינֵיהֶם אָצְוּ מִתּוֹךְ…”.
מכתב: כתבו לרבי מאיר בעקבות “בְּקּוֹל מָתוֹק וְשָׁקֵט הוּא יָשַׁב וְלָמַד תּוֹרָה בְּעַל-פֶּה וְהִתְפַּלֵּל לְבוֹרֵא עוֹלָם.”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “הַשּׁוֹדְדִים נִדְהֲמוּ מֵחָכְמָתוֹ הַכַּבִּירָה שֶׁל הַיְהוּדִי וּמִבִּטְחוֹנוֹ בֶּאֱלֹקָיו.”.
דיווח: כתבו ידיעה קצרה שכותרתה “אח ושמו "כי טוב" — ניצל מהשודדים”.

המחזה

רגע מן הסיפורהדמויותמה יאמרו או יעשו?
אֲנָשִׁים בְּאוֹתוֹ זְמַן הָיוּ רוֹעֲדִים מִפַּחַד לָלֶכֶת בַּיַּעַר בַּלַּיְלָה.
הוּא הִצְבִּיעַ בְּאֶצְבָּעוֹ עַל הַשֶּׁמֶשׁ הָעוֹלָה, עַל שְׁמֵי הַבֹּקֶר הַכְּחֻלִּים, עַל אוֹר…

7. חיבור לחיים מתוך המעשה

א. מה אפשר ללמוד מן הדרך שבה רבי מאיר פעל באירוע “רַבִּי מֵאִיר חִיֵּךְ חִיּוּךְ זָעִיר, חִיּוּךְ שֶׁל קְדֻשָּׁה וֶאֱמוּנָה.”?
ב. תארו מצב מחיי הכיתה שבו אפשר ליישם את המסר של “אח ושמו "כי טוב" — ניצל מהשודדים”.
