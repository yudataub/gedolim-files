זהירות מלהשתמש בכבוד התורה

דף עבודה בעקבות רבי טרפון
שם: כיתה:

המעשה — הסיפור המלא

פְּתִיחָה: רֵיחַ הַתְּאֵנִים בַּשָּׂדֶה

יְלָדִים, עִצְמוּ אֶת הָעֵינַיִם וְנַסּוּ לְדַמְיֵן שָׂדֶה יְרוּק וְרָחָב תַּחַת קַרְנֵי שֶׁמֶשׁ זְהֻבּוֹת. הָאֲוִיר מָלֵא בְּרֵיחַ מָתוֹק וְנָעִים מְאֹד. זֶהוּ רֵיחַ שֶׁל תְּאֵנִים בְּשֵׁלוֹת, "קְצִיעוֹת" מְתוּקוֹת כִּדְבַשׁ. בֵּין הָעֵצִים מִתְעוֹפְפוֹת צִפֳּרִים מְצַיְּצוֹת, וּמֵרָחוֹק אֶפְשָׁר לִשְׁמֹעַ אֶת זְרִימַת הַמַּיִם הַקְּרִירִים שֶׁל הַנָּהָר הַגָּדוֹל שֶׁלְּיַד הַשָּׂדֶה.

הַשָּׂדֶה הַיָּפֶה הַזֶּה לֹא שַׁיָּךְ לְאָדָם רָגִיל. הוּא שַׁיָּךְ לְאֶחָד מִגְּדוֹלֵי יִשְׂרָאֵל, צַדִּיק שֶׁכֻּלָּם אָהֲבוּ – רַבִּי טַרְפוֹן.

רַבִּי טַרְפוֹן הָיָה עָשִׁיר גָּדוֹל בְּאוֹתָם יָמִים. הָיוּ לוֹ כְּרָמִים וּשְׂדוֹת-פְּרִי רַבִּים. אַךְ מָה שֶׁהָפַךְ אוֹתוֹ לֶאֱמֶת עָשִׁיר, לֹא הָיוּ מַטְבְּעוֹת הַזָּהָב שֶׁלּוֹ, אֶלָּא עֲמַל הַתּוֹרָה הַקְּדוֹשָׁה שֶׁבָּעֲרָה בְּלִבּוֹ לַיְלָה וָיוֹם.

בְּאוֹתוֹ בֹּקֶר הִגִּיעַ רַבִּי טַרְפוֹן אֶל אַחַד הַפַּרְדֵּסִים שֶׁלּוֹ. הָיְתָה זוֹ שְׁנַת שְׁמִטָּה (אוֹ לְפִי גִּרְסָה אַחֶרֶת, תְּקוּפָה שֶׁבָּהּ הַפֵּרוֹת כְּבָר הָיוּ מֻפְקָרִים). רַבִּי טַרְפוֹן לָבַשׁ בְּגָדִים פְּשׁוּטִים בְּאוֹתוֹ יוֹם. לֹא הָיוּ עָלָיו בִּגְדֵי רַבָּנִים מְפֹאָרִים. הוּא פָּסַע בַּשְּׁבִיל הַמֻּכָּר, הוֹשִׁיט אֶת יָדוֹ אֶל הֶעָנָף, תָּלַשׁ תְּאֵנָה מְתוּקָה, בֵּרַךְ לִפְנֵי שֶׁאָכַל בְּכַוָּנָה רַבָּה "בּוֹרֵא פְּרִי הָעֵץ", וְהִכְנִיס אֶת הַתְּאֵנָה לְפִיו. טַעַם דְּבַשׁ מִלֵּא אֶת פִּיו. הוּא הָלַךְ בְּשֶׁקֶט, טוֹעֵם מִפֵּרוֹת הַשָּׂדֶה שֶׁלּוֹ, וּמְהַרְהֵר בְּסֻגְיוֹת הַתַּלְמוּד.

הַהַתְקָפָה הַפִּתְאוֹמִית

לְפֶתַע, הַשֶּׁקֶט הֻפְרַע!

טְרַאח! קְרַאק!

שְׁמִיעַת צְעָדִים חֲזָקִים, כְּבֵדִים, מָעֲכוּ אֶת הֶעָלִים הַיְּבֵשִׁים עַל הָאֲדָמָה. אֶל בֵּין הָעֵצִים קָפַץ אִישׁ עֲנָק, מָלֵא בְּכַעַס, וּבְיָדוֹ מַקֵּל עָבֶה. זֶה הָיָה הַשּׁוֹמֵר שֶׁל הַשָּׂדֶה! הַשּׁוֹמֵר הַזֶּה נִשְׂכַּר עַל יְדֵי רַבִּי טַרְפוֹן לִשְׁמֹר עַל הַשָּׂדֶה מִפְּנֵי גַּנָּבִים, אֲבָל... הַשּׁוֹמֵר מֵעוֹלָם לֹא רָאָה אֶת רַבִּי טַרְפוֹן! רַבִּי טַרְפוֹן הָיָה תָּמִיד בְּבֵית הַמִּדְרָשׁ!

וְעַכְשָׁו, הַשּׁוֹמֵר רוֹאֶה אָדָם בְּבִגְדֵי חֹל פְּשׁוּטִים, קוֹטֵף תְּאֵנִים.

"תָּפַסְתִּי אוֹתְךָ, גַּנָּב שָׁפֵל!" צָעַק הַשּׁוֹמֵר בְּקוֹל שֶׁהִרְעִיד אֶת כָּל הַפַּרְדֵּס.

הוּא הֵנִיף אֶת מַקְלוֹ – וְהִכָּה! וְעוֹד מַכָּה!

יְלָדִים, רַבִּי טַרְפוֹן, הַצַּדִּיק וְהַקָּדוֹשׁ שֶׁכֻּלָּם רוֹעֲדִים מִפָּנָיו בְּיִרְאַת כָּבוֹד, עוֹמֵד כָּאן וּמְקַבֵּל מַכּוֹת מְכְאִיבוֹת מֵהַפּוֹעֵל שֶׁלּוֹ! הָכְּאֵב הָיָה נוֹרָא. אֲבָל רַבִּי טַרְפוֹן... שָׁתַק. הוּא לֹא אָמַר כְּלוּם.

לָמָּה? כִּי הוּא יָדַע שֶׁהַשּׁוֹמֵר סְתָם עוֹשֶׂה אֶת הָעֲבוֹדָה שֶׁלּוֹ, וְרַבִּי טַרְפוֹן עֲנָיו מְאֹד – הוּא לֹא רוֹצֶה שֶׁהַשּׁוֹמֵר יִבָּהֵל אוֹ יִתְבַּיֵּשׁ, וְהוּא בְּוַדַּאי שֶׁלֹּא רוֹצֶה לְהִשְׁתַּמֵּשׁ בַּשֵּׁם שֶׁלּוֹ, בַּכָּבוֹד הַתּוֹרָנִי שֶׁלּוֹ, רַק כְּדֵי שֶׁיַּפְסִיקוּ לְהַכּוֹת אוֹתוֹ.

הַשַּׂק הֶחָשׁוּךְ

אֲבָל הַשּׁוֹמֵר לֹא נִרְגַּע. מֵרֹב זַעַם עַל הַ"גַּנָּבִים" שֶׁמַּשְׁחִיתִים אֶת הַשָּׂדוֹת תָּמִיד, הוּא חָשַׁב בְּלִבּוֹ: "הַפַּעַם אֲנִי אֲלַמֵּד אֶת הַגַּנָּב הַזֶּה לֶקַח שֶׁהוּא לֹא יִשְׁכַּח לְעוֹלָם!".

הוּא פָּתַח שַׂק בַּד עָבֶה, מְחֻסְפָּס וּמָלֵא בְּאָבָק. בְּכֹחַ רַב הוּא דָּחַף אֶת רַבִּי טַרְפוֹן פְּנִימָה לְתוֹךְ הַשַּׂק, וְקָשַׁר אֶת הַפֶּתַח!

לְפֶתַע, נִהְיָה לְרַבִּי טַרְפוֹן חֹשֶׁךְ מֻחְלָט. הָאֲוִיר בְּתוֹךְ הַשַּׂק הָיָה דָּחוּס וְחָנוּק, מָלֵא בְּרֵיחַ שֶׁל שַׂק יָשָׁן וֶעָפָר. הוּא הִרְגִּישׁ אֵיךְ הַשּׁוֹמֵר הֶחָזָק מֵרִים אוֹתוֹ עַל הַכָּתֵף וּמַתְחִיל לִצְעֹד.

"לְאָן הוּא מְבִיא אוֹתִי?" חָשַׁב הָרַב.

תּוֹךְ כַּמָּה צְעָדִים, רַבִּי טַרְפוֹן שָׁמַע אֶת הַצְּלִיל הַמַּפְחִיד בְּיוֹתֵר: רַעַשׁ שֶׁל גַּלִּים מִתְנַפְּצִים וּמַיִם הַזּוֹרְמִים בְּעָצְמָה. זֶה הָיָה הַנָּהָר! הַשּׁוֹמֵר מִתְכַּוֵּן לִזְרֹק אֶת הַשַּׂק לְתוֹךְ הַמַּיִם הָעֲמֻקִּים! רַבִּי טַרְפוֹן עוֹמֵד לְטֻבַּע!

הַצְּעָקָה שֶׁקָּרְעָה אֶת הַשָּׁמַיִם

יְלָדִים, זֶה הָיָה רֶגַע שֶׁל "פִּקּוּחַ נֶפֶשׁ" – חַיִּים אוֹ מָוֶת. לְרַבִּי טַרְפוֹן כְּבָר לֹא הָיְתָה בְּרֵרָה. אִם הוּא יַמְשִׁיךְ לִשְׁתֹּק עַכְשָׁו – הוּא יָמוּת, וְהַשּׁוֹמֵר יֵעָנֵשׁ בַּשָּׁמַיִם עַל רֶצַח!

מִתּוֹךְ הַשַּׂק הֶחָשׁוּךְ, רַבִּי טַרְפוֹן אָסַף אֶת כָּל כֹּחוֹ, וְצָעַק בְּקוֹל שָׁבוּר שֶׁבָּקַע מֵעוֹמֶק נִשְׁמָתוֹ:

"אוֹי לוֹ לְטַרְפוֹן! אוֹי לָהֶם לִבְנֵי בֵּיתוֹ שֶׁל טַרְפוֹן, שֶׁעוֹמְדִים עַכְשָׁו לְהָבִיא לוֹ תַּכְרִיכִים לִקְבוּרָה!".

הַשּׁוֹמֵר עָצַר.

הוּא קָפָא בִּמְקוֹמוֹ. מָה?! טַרְפוֹן?! הַגַּנָּב הַזֶּה הוּא רַבִּי טַרְפוֹן בִּכְבוֹדוֹ וּבְעַצְמוֹ?! קְדוֹשׁ יִשְׂרָאֵל, רַבּוֹ שֶׁל רַבִּי עֲקִיבָא?!

הַשּׁוֹמֵר הֵנִיחַ אֶת הַשַּׂק עַל הָאֲדָמָה בְּיָדַיִם רוֹעֲדוֹת. הוּא מִהֵר לְהַתִּיר אֶת הַקֶּשֶׁר. מִתּוֹךְ הַשַּׂק יָצָא אָדָם קָדוֹשׁ, פָּנָיו מְאִירוֹת אֲבָל כְּאֵב נוֹצֵץ מֵעֵינָיו.

הַשּׁוֹמֵר נָפַל עַל פָּנָיו בֶּעָפָר. הוּא תָּלַשׁ אֶת שְׂעָרוֹתָיו מֵרֹב צַעַר וּבָכָה בְּקוֹל רָם: "רַבִּי! רַבִּי יַקִּירִי! סְלַח לִי! מְחַל לִי! מָה עָשִׂיתִי?! הִכֵּיתִי אֶת 'אֲבִיהֶם שֶׁל יִשְׂרָאֵל'! אוֹי לִי עַל עֲווֹנִי!".

רַבִּי טַרְפוֹן, הַמָּלֵא בְּאַהֲבַת יִשְׂרָאֵל, הִתְכּוֹפֵף אֵלָיו, הִנִּיחַ יָד עַל כְּתֵפוֹ, וְאָמַר לוֹ בְּרוֹךְ: "חַיֶּיךָ (אֲנִי מִשָּׁבָע לְךָ), שֶׁעַל כָּל מַכָּה וּמַכָּה שֶׁהִכֵּיתָ אוֹתִי מִקֹּדֶם – כְּבָר סָלַחְתִּי וּמָחַלְתִּי לְךָ בְּלִבִּי בְּאוֹתוֹ הָרֶגַע!".

הַדְּמָעוֹת שֶׁלֹּא פָּסְקוּ לְעוֹלָם

הַשּׁוֹמֵר נִרְגַּע וְהָלַךְ לְדַרְכּוֹ. רַבִּי טַרְפוֹן חָזַר לְבֵיתוֹ שָׁלֵם וּבָרִיא. אָדָם רָגִיל הָיָה עוֹשֶׂה כָּעֵת שְׂמָחוֹת וְרוֹקֵד: "אֵיזֶה נֵס! נִצַּלְתִּי מִמָּוֶת כִּמְעַט בָּרֶגַע הָאַחֲרוֹן!".

אֲבָל... מֵאוֹתוֹ יוֹם, וְעַד סוֹף יָמָיו, רַבִּי טַרְפוֹן הָיָה מְצֵר וְעָצוּב עַל מָה שֶׁקָּרָה שָׁם. הַרְבֵּה פְּעָמִים רָאוּ אֶת רַבִּי טַרְפוֹן בוֹכֶה בְּשֶׁקֶט, וּכְשֶׁשָּׁאֲלוּ אוֹתוֹ לָמָּה הוּא בוֹכֶה, הוּא אָמַר מִשְׁפָּט אֶחָד שֶׁזִּעְזֵעַ אֶת כָּל בֵּית הַמִּדְרָשׁ:

"אוֹי לִי שֶׁנִּשְׁתַּמַּשְׁתִּי בְּכִתְרָהּ שֶׁל תּוֹרָה!".

יְלָדִים, אֲפִלּוּ שֶׁהָיָה מֻתָּר לְרַבִּי טַרְפוֹן לְצָעוֹק שֶׁהוּא רַב כְּדֵי לְהַצִּיל אֶת חַיָּיו (כִּי 'פִּקּוּחַ נֶפֶשׁ מְדַחֶה אֶת הַכֹּל'), עֲדַיִן כָּאַב לוֹ מְאֹד. הוּא הִרְגִּישׁ שֶׁהַתּוֹרָה שֶׁלָּמַד צְרִיכָה לִהְיוֹת רַק לִכְבוֹד ה', מֵאַהֲבָה לַה'. הַעֻבְדָּה שֶׁבִּגְלַל הַתּוֹרָה שֶׁלּוֹ הוּא נִצַּל וְקִבֵּל שֵׁרוּת וְהַנָּחָה מֵהַשּׁוֹמֵר, הָיְתָה כְּמוֹ שְׂרִיטָה בַּלֵּב הַטָּהוֹר שֶׁלּוֹ.

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “יְלָדִים, עִצְמוּ אֶת הָעֵינַיִם וְנַסּוּ לְדַמְיֵן שָׂדֶה יְרוּק וְרָחָב תַּחַת קַרְנֵי שֶׁמֶשׁ זְהֻבּוֹת.” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “הָיְתָה זוֹ שְׁנַת שְׁמִטָּה (אוֹ לְפִי גִּרְסָה אַחֶרֶת, תְּקוּפָה שֶׁבָּהּ הַפֵּרוֹת כְּבָר הָיוּ מֻפְקָרִים).”?
ג. כיצד הגיב רבי טרפון למצב המתואר: “וְעַכְשָׁו, הַשּׁוֹמֵר רוֹאֶה אָדָם בְּבִגְדֵי חֹל פְּשׁוּטִים, קוֹטֵף תְּאֵנִים.”?
ד. מה השתנה בין “הוּא הִרְגִּישׁ אֵיךְ הַשּׁוֹמֵר הֶחָזָק מֵרִים אוֹתוֹ עַל הַכָּתֵף וּמַתְחִיל לִצְעֹד.” לבין “אוֹי לָהֶם לִבְנֵי בֵּיתוֹ שֶׁל טַרְפוֹן, שֶׁעוֹמְדִים עַכְשָׁו לְהָבִיא לוֹ תַּכְרִיכִים לִקְבוּרָה!".”?
ה. כיצד הגיע הסיפור אל האירוע: “הַרְבֵּה פְּעָמִים רָאוּ אֶת רַבִּי טַרְפוֹן בוֹכֶה בְּשֶׁקֶט, וּכְשֶׁשָּׁאֲלוּ אוֹתוֹ לָמָּה הוּא בוֹכֶה, הוּא אָמַר מִשְׁפָּט אֶחָד שֶׁזִּעְזֵעַ אֶת…”?

3. מה היה ראשון?

כתבו ליד כל אירוע מספר מ־1 עד 5 לפי סדר התרחשותו בסיפור.

□ בְּכֹחַ רַב הוּא דָּחַף אֶת רַבִּי טַרְפוֹן פְּנִימָה לְתוֹךְ הַשַּׂק, וְקָשַׁר אֶת הַפֶּתַח!
□ יְלָדִים, עִצְמוּ אֶת הָעֵינַיִם וְנַסּוּ לְדַמְיֵן שָׂדֶה יְרוּק וְרָחָב תַּחַת קַרְנֵי שֶׁמֶשׁ זְהֻבּוֹת.
□ הַעֻבְדָּה שֶׁבִּגְלַל הַתּוֹרָה שֶׁלּוֹ הוּא נִצַּל וְקִבֵּל שֵׁרוּת וְהַנָּחָה מֵהַשּׁוֹמֵר, הָיְתָה כְּמוֹ שְׂרִיטָה בַּלֵּב הַטָּהוֹר שֶׁלּוֹ.
□ הוּא הָלַךְ בְּשֶׁקֶט, טוֹעֵם מִפֵּרוֹת הַשָּׂדֶה שֶׁלּוֹ, וּמְהַרְהֵר בְּסֻגְיוֹת הַתַּלְמוּד.
□ הַשּׁוֹמֵר הֵנִיחַ אֶת הַשַּׂק עַל הָאֲדָמָה בְּיָדַיִם רוֹעֲדוֹת.

לפני ואחרי

מה קרה לפני: “בְּכֹחַ רַב הוּא דָּחַף אֶת רַבִּי טַרְפוֹן פְּנִימָה לְתוֹךְ הַשַּׂק, וְקָשַׁר אֶת הַפֶּתַח!”?
מה קרה אחרי: “הַשּׁוֹמֵר הֵנִיחַ אֶת הַשַּׂק עַל הָאֲדָמָה בְּיָדַיִם רוֹעֲדוֹת.”?

סיבה ותוצאה מתוך המעשה

האירוע הנתוןמה גרם לו?מה הייתה תוצאתו?
שְׁמִיעַת צְעָדִים חֲזָקִים, כְּבֵדִים, מָעֲכוּ אֶת הֶעָלִים הַיְּבֵשִׁים עַל הָאֲדָמָה.
יְלָדִים, זֶה הָיָה רֶגַע שֶׁל "פִּקּוּחַ נֶפֶשׁ" – חַיִּים אוֹ מָוֶת.

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “זהירות מלהשתמש בכבוד התורה” על רבי טרפון? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “מֵרֹב זַעַם עַל הַ"גַּנָּבִים" שֶׁמַּשְׁחִיתִים אֶת הַשָּׂדוֹת תָּמִיד, הוּא חָשַׁב בְּלִבּוֹ: "הַפַּעַם אֲנִי אֲלַמֵּד אֶת הַגַּנָּב הַזֶּה לֶקַח שֶׁהוּא…”.
ג. אילו רבי טרפון לא היה פועל כפי שמתואר ב־“אוֹי לָהֶם לִבְנֵי בֵּיתוֹ שֶׁל טַרְפוֹן, שֶׁעוֹמְדִים עַכְשָׁו לְהָבִיא לוֹ תַּכְרִיכִים לִקְבוּרָה!".” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “הַרְבֵּה פְּעָמִים רָאוּ אֶת רַבִּי טַרְפוֹן בוֹכֶה בְּשֶׁקֶט, וּכְשֶׁשָּׁאֲלוּ אוֹתוֹ לָמָּה הוּא בוֹכֶה, הוּא אָמַר מִשְׁפָּט…” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. יְלָדִים, עִצְמוּ אֶת הָעֵינַיִם וְנַסּוּ לְדַמְיֵן שָׂדֶה יְרוּק…
2. הוּא פָּסַע בַּשְּׁבִיל הַמֻּכָּר, הוֹשִׁיט אֶת יָדוֹ אֶל הֶעָנָף,…
3. יְלָדִים, רַבִּי טַרְפוֹן, הַצַּדִּיק וְהַקָּדוֹשׁ שֶׁכֻּלָּם רוֹעֲדִים…
4. תּוֹךְ כַּמָּה צְעָדִים, רַבִּי טַרְפוֹן שָׁמַע אֶת הַצְּלִיל…
5. מִתּוֹךְ הַשַּׂק יָצָא אָדָם קָדוֹשׁ, פָּנָיו מְאִירוֹת אֲבָל כְּאֵב…
6. הַעֻבְדָּה שֶׁבִּגְלַל הַתּוֹרָה שֶׁלּוֹ הוּא נִצַּל וְקִבֵּל שֵׁרוּת…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “וְעַכְשָׁו, הַשּׁוֹמֵר רוֹאֶה אָדָם בְּבִגְדֵי חֹל פְּשׁוּטִים, קוֹטֵף תְּאֵנִים.”.
מכתב: כתבו לרבי טרפון בעקבות “יְלָדִים, זֶה הָיָה רֶגַע שֶׁל "פִּקּוּחַ נֶפֶשׁ" – חַיִּים אוֹ מָוֶת.”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “הוּא תָּלַשׁ אֶת שְׂעָרוֹתָיו מֵרֹב צַעַר וּבָכָה בְּקוֹל רָם: "רַבִּי!”.
דיווח: כתבו ידיעה קצרה שכותרתה “זהירות מלהשתמש בכבוד התורה”.
