הצלת שר העיר

דף עבודה בעקבות בר קפרא
שם: כיתה:

המעשה — הסיפור המלא

להלן עיבוד חינוכי, רגשי ומרתק לסיפור המופלא על בר קפרא והצלת שר העיר , המעובד במיוחד לבנים בכיתות ב'-ד' (חינוך דתי-לאומי). הסיפור מדגיש את כוחו העצום של חסד של אמת שנעשה ללא ציפייה לתמורה, וכיצד הקב"ה מסובב את כל האירועים בעולם (השגחה פרטית) כדי להגן על עם ישראל.

הסיפור מנוקד במלואו, מועשר בתיאורים חושיים, ובנוי על פי עקרונות הסטוריטלינג וההנחיות הפדגוגיות.

פְּתִיחָה: רֵיחוֹת שֶׁל סְעָרָה

יְלָדִים, הַאִם פַּעַם יְשַׁבְתֶּם בַּבַּיִת כְּשֶׁבַּחוּץ הִשְׁתּוֹלְלָה סְעָרָה? הַחַלּוֹנוֹת רוֹעֲדִים, הָרוּחַ שׁוֹרֶקֶת "הוּוווּ... הוּוווּ...", וְהַגֶּשֶׁם מַכֶּה בַּתְּרִיסִים. עַכְשָׁו תְּדַמְיְנוּ אֶת הַסְּעָרָה הַזּוֹ – אֲבָל עַל שְׂפַת הַיָּם הַגָּדוֹל שֶׁל הָעִיר קֵיסָרְיָה, לִפְנֵי קָרוֹב לְאַלְפַּיִם שָׁנָה.

הַתַּנָּא הַקָּדוֹשׁ, בַּר קַפָּרָא, הָלַךְ בְּאוֹתוֹ יוֹם חָרְפִּי עַל שְׂפַת הַיָּם. רֵיחַ חָזָק וּמָלוּחַ שֶׁל אֵדֵי מַיִם מִלֵּא אֶת הָאֲוִיר. הַגַּלִּים קָפְצוּ לְגֹבַהּ שֶׁל הָרִים וְהִתְנַפְּצוּ עַל הַסְּלָעִים בְּרַעַשׁ אַדִּיר: בּוּם! קְרָאח! בַּר קַפָּרָא הִתְעַטֵּף בִּמְעִילוֹ, וְעֵינָיו הַחֲכָמוֹת הִבִּיטוּ אֶל הָאֹפֶק הֶעָשֵׁן.

לְפֶתַע, הוּא שָׁמַע קוֹל זְעָקָה. צְעָקָה חַלָּשָׁה שֶׁנָּבְעָה מִתּוֹךְ מַיִם זֶה עַתָּה. הוּא הִתְקָרֵב קָדִימָה. אֲדָמַת הַחוֹל הָיְתָה בֻּצִּית וְקָרָה. שָׁם, עַל קַו הַמַּיִם, רָאָה בַּר קַפָּרָא מַחֲזֶה מְזַעֲזֵעַ: שְׁבָרִים שֶׁל עֵצִים קְרוּעִים הָיוּ מְפֻזָּרִים עַל הַחוֹל. סְפִינָה גְּדוֹלָה שֶׁהִגִּיעָה מֵאֶרֶץ רְחוֹקָה, טָבְעָה בַּסְּעָרָה.

וּבֵין הַקְּרָשִׁים הַשְּׁבוּרִים, שָׁכַב אָדָם. אָדָם אֶחָד שֶׁנִּצַּל.

חֶסֶד בְּלִי לִשְׁאֹל שְׁאֵלוֹת

הָאִישׁ הַזֶּה הָיָה שַׁיָּךְ לְעַם אַחֵר, נוֹכְרִי גָּמוּר. הַבְּגָדִים שֶׁלּוֹ נִקְרְעוּ לְחַלּוּטִין בַּגַּלִּים. הוּא רָעַד מִקֹּר, מַמָּשׁ שִׁנָּיו נָקְשׁוּ זוֹ בָּזוֹ ( טַק-טַק-טַק ). הָיָה נִרְאֶה שֶׁהוּא עָמַד לָמוּת מִקֹּר וּמֵרָעָב בְּתוֹךְ דַּקּוֹת.

יְלָדִים, אָדָם רָגִיל שֶׁעוֹבֵר שָׁם הָיָה חוֹשֵׁב: "מִי זֶה? אוּלַי הוּא גַּנָּב? אוּלַי הוּא סְתָם קַבְּצָן? אוּלַי זֶה רוֹמָאִי שֶׁשּׂוֹנֵא יְהוּדִים? כְּדַאי לִי לִבְרֹחַ מִפֹּה!".

אֲבָל בַּר קַפָּרָא הָיָה תַּלְמִיד חָכָם, מָלֵא בְּאַהֲבַת הַשֵּׁם וּבְרַחֲמִים. הוּא לֹא שָׁאַל שְׁאֵלוֹת. הוּא מִיָּד פָּשַׁט אֶת בִּגְדּוֹ הַחַם עַל הַחוֹף הַקַּר, וְעָטַף בּוֹ אֶת הָאִישׁ הַנּוֹכְרִי הָרוֹעֵד. "קוּם, אָחִי הַצּוֹרֵר בְּצָרָתוֹ," אָמַר לוֹ הָרַב בְּקוֹל רַךְ כְּשֶׁהוּא מֵרִים אוֹתוֹ. "בּוֹא אִתִּי לְבֵיתִי".

בַּר קַפָּרָא הִכְנִיס אוֹתוֹ לְתוֹךְ בֵּיתוֹ הֶחָמִים. הוּא הוֹשִׁיב אוֹתוֹ לְיַד הַתַּנּוּר, הִגִּישׁ לוֹ קְעָרָה שֶׁל מָרָק מַהְבִּיל וְרֵיחָנִי, וְנָתַן לוֹ לְהִתְרַחֵץ וְלִלְבֹּשׁ בְּגָדִים נְקִיִּים שֶׁהָיוּ שַׁיָּכִים לָרַב עַצְמוֹ!

כְּשֶׁהָאִישׁ הִתְאוֹשֵׁשׁ, עֵינָיו זָלְגוּ דְּמָעוֹת שֶׁל הוֹדָיָה. "אֲנִי... אֲנִי לֹא יוֹדֵעַ אֵיךְ לְהוֹדוֹת לְךָ, הַיְּהוּדִי," הוּא לָחַשׁ. "אִבַּדְתִּי הַכֹּל בַּיָּם. אֵין לִי אֲפִלּוּ מַטְבֵּעַ אֶחָד כְּדֵי לַחְזֹר לְעִירִי!".

בַּר קַפָּרָא יָכוֹל הָיָה לוֹמַר לוֹ: "נָתַתִּי לְךָ אֹכֶל וּבְגָדִים. עַכְשָׁו לֵךְ תִּסְתַּדֵּר".

אַךְ מַהוּ חֶסֶד שֶׁל אֱמֶת? חֶסֶד עַד הַסּוֹף! בַּר קַפָּרָא נִגַּשׁ לְקֻפַּת הַחִסָּכוֹן שֶׁלּוֹ, הוֹצִיא מִשָּׁם שְׁלוֹשָׁה מַטְבְּעוֹת שֶׁל "סֶלַע" (סְכוּם כֶּסֶף גָּדוֹל מְאֹד בְּאוֹתָם יָמִים), וְשָׂם בְּיָדוֹ שֶׁל הַנּוֹכְרִי.

"קַח אֶת זֶה, אֲדוֹנִי. סַע לְשָׁלוֹם, וְהַשֵּׁם יִשְׁמֹר דְּרָכֶיךָ," אָמַר הָרַב בְּחִיּוּךְ מֵאִיר.

הַנּוֹכְרִי יָצָא לְדַרְכּוֹ, וּבַר קַפָּרָא חָזַר לְבֵית הַמִּדְרָשׁ, לִסְפָרָיו הַקְּדוֹשִׁים, בְּלִי לַחְשֹׁב עַל כָּךְ שׁוּב. הוּא עָשָׂה אֶת הַטּוֹב, רַק בִּגְלַל שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא צִוָּה לַעֲשׂוֹת טוֹב.

הַגְּזֵרָה הַנּוֹרָאָה עַל קֵיסָרְיָה

עָבְרוּ שָׁנִים רַבּוֹת. הַשֶּׁקֶט שֶׁבְּאֶרֶץ יִשְׂרָאֵל אוֹיַם שׁוּב עַל יְדֵי הָרוֹמָאִים.

יוֹם אֶחָד, הִגִּיעַ לְקֵיסָרְיָה שַׂר רוֹמָאִי חָדָשׁ אֲשֶׁר מֻנָּה עַל יְדֵי מֶלֶךְ רוֹמִי, הַקֵּיסָר, לִשְׁלֹט בָּאֵזוֹר. הַשַּׂר הַזֶּה הָיָה אָדָם תַּקִּיף וְחָזָק. מִסִּבָּה כָּלְשֶׁהִי, כָּעֲסוּ הָרוֹמָאִים עַל חֲבוּרָה שֶׁל יְהוּדִים בְּאֵזוֹר קֵיסָרְיָה, וְהַשַּׂר הַחָדָשׁ אָמַר בְּזַעַם: "אֲנִי מֵטִיל עָנְשִׁים כְּבֵדִים עַל יְהוּדֵי הָעִיר! וּמִי שֶׁיָּעֵז לְהִתְנַגֵּד – אֶשְׁלַח אֶת חַיָּלַי לֶאֱסֹר אוֹתוֹ!"

פַּחַד גָּדוֹל נָפַל בָּרְחוֹבוֹת. חֲנוּיוֹת נִסְגְּרוּ. אִמָּהוֹת חִבְּקוּ אֶת הַיְּלָדִים שֶׁלָּהֶן בַּבָּתִּים וּבָכוּ. זְקֵנֵי הָעִיר הִתְאַסְּפוּ בְּבֵית הַכְּנֶסֶת.

"מָה נַעֲשֶׂה?" קָרָא אֶחָד מִזִּקְנֵי הַקְּהִלָּה. "הַשַּׂר הַזֶּה הוּא אַכְזָר! הַיְּהוּדִים כָּאן בְּסַכָּנָה עֲצוּמָה!"

"מִי יוּכַל לָלֶכֶת לָאַרְמוֹן כְּדֵי לְדַבֵּר עַל לִבּוֹ?" שָׁאַל זָקֵן אַחֵר. "מִי שֶׁיִּתְקָרֵב לַשַּׂר עָלוּל לְהֵרָצֵחַ עַל יְדֵי הַשּׁוֹמְרִים שֶׁלּוֹ!"

כֻּלָּם הִשְׁפִּילוּ מַבָּט בְּאֵימָה.

וְאָז קָם בַּר קַפָּרָא. פָּנָיו שָׁלְווֹת, זוֹרְחוֹת מִקְּדֻשָּׁה שֶׁל תּוֹרָה. "הַקָּדוֹשׁ בָּרוּךְ הוּא שׁוֹמֵר אֶת יִשְׂרָאֵל," הוּא אָמַר בְּקוֹל יַצִּיב. "אֲנִי אֵלֵךְ. לֹא אֶעֱזֹב אֶת הָעָם שֶׁלִּי בִּשְׁעַת צָרָה. אֵלֵךְ אֶל הָאַרְמוֹן, יִקְרֶה מָה שֶׁיִּקְרֶה!".

בְּמֶשֶׁךְ הַלַּיְלָה יָשְׁבוּ הַיְּהוּדִים וְקָרְאוּ תְּהִלִּים בִּדְמָעוֹת, מִתְפַּלְלִים לִשְׁלוֹמוֹ שֶׁל הַצַּדִּיק.

מִדְרְכוֹת הַשַּׁיִשׁ וְהַשּׁוֹמְרִים הָרוֹמָאִים

לְמָחֳרָת מֻקְדָּם בַּבֹּקֶר, צָעַד בַּר קַפָּרָא מִבַּעַד לְשַׁעֲרֵי הָאַרְמוֹן הָרוֹמָאִי.

הַכֹּל כָּאן שִׁדֵּר כֹּחַ וְאֵימָה. מִדְרְכוֹת הַשַּׁיִשׁ הָיוּ קָרוֹת וְחֲלָקוֹת. עַל הַקִּירוֹת הָיוּ תְּלוּיִים חֲרָבוֹת וּמָגִנִּים כְּבֵדִים שֶׁנִּצְנְצוּ בָּאוֹר. מִשְּׁנֵי הַצְּדָדִים עָמְדוּ חַיָּלִים רוֹמָאִים גְּבוֹהִים, לוֹבְשִׁים שִׁרְיוֹנוֹת מַתֶּכֶת מַבְרִיקִים, קַסְדּוֹת בַּרְזֶל עַל רָאשֵׁיהֶם, וּבִידֵיהֶם חֲנִיתוֹת אֲרֻכּוֹת וְחַדּוֹת. נְשִׁימָתָם נִשְׁמְעָה בַּחֲלַל הָאַרְמוֹן הַגָּדוֹל: הָא! הוּא!

אֲבָל בַּר קַפָּרָא צָעַד זָקוּף. הוּא לֹא פָּחַד מִבָּשָׂר וָדָם, אֶלָּא רַק מִמֶּלֶךְ מַלְכֵי הַמְּלָכִים. הוּא נִכְנַס לְתוֹךְ דַּלְתוֹת הָעֵץ הָעֲנָקִיּוֹת – יְשִׁירוֹת אֶל אֲוּלַם הַמִּשְׁפָּט.

שָׁם, עַל כִּסֵּא עֲנָקִי מְצֻפֶּה זָהָב, יָשַׁב שַׂר הָעִיר, הַמּוֹשֵׁל הָרוֹמָאִי הַתַּקִּיף. שׁוֹמְרֵי הָרֹאשׁ שֶׁלּוֹ כִּמְעַט וּשְׁלָפוּ חֲרָבוֹת כְּדֵי לִפְגֹּעַ בַּיְּהוּדִי שֶׁהֵעֵז לְהִכָּנֵס בְּלִי לִקְרֹא לוֹ!

הַשַּׂר הָרוֹמָאִי סוֹבֵב אֶת רֹאשׁוֹ וְהִבִּיט בַּזָּקֵן הַיְּהוּדִי הָעוֹמֵד מוּלוֹ. עֵינָיו נִפְעֲרוּ לִרְוָחָה!

הוּא קָם בְּמַפְתִּיעַ מִכִּסְאוֹ, מֵרִים אֶת יָדוֹ וְצוֹעֵק לְחַיָּלָיו: "עִצְרוּ! אַל תִּגְּעוּ בּוֹ הָרָעָה! הָרִידוּ אֶת הַנֶּשֶׁק שֶׁלָּכֶם, מִיָּד!"

הַחַיָּלִים הַמֻּפְתָּעִים מִהֲרוּ לְהוֹרִיד אֶת הַחֲנִיתוֹת וְהַחֲרָבוֹת. שֶׁקֶט מֻחְלָט ( הַס... ) הִשְׁתָּרֵר בָּאוּלָם הָעֲנָקִי.

הַשַּׂר יָרַד מִכִּסְאוֹ הָרָם, נִגַּשׁ בְּקִדָּה עֲמֻקָּה אֶל בַּר קַפָּרָא, וְשָׁאַל בְּקוֹל שֶׁרָעַד מֵהִתְרַגְּשׁוּת:

"רַבִּי יְהוּדִי! מָה מֵבִיא אוֹתְךָ כָּאן אֵלַי? מָה מְבַקֵּשׁ כְּבוֹדוֹ? רַק תַּגִּיד – וְהַכֹּל אֶעֱשֶׂה בִּשְׁבִילְךָ!"

הַהַשְׁגָּחָה הַפְּרָטִית מִתְגַּלֵּית

בַּר קַפָּרָא הָיָה הָמוּם. מִי זֶה הַשַּׂר הַזֶּה שֶׁמְּכַבֵּד אוֹתוֹ כָּךְ?

הוּא הִבִּיט לְתוֹךְ פָּנָיו שֶׁל הַמּוֹשֵׁל הָרוֹמָאִי, וְאָז – הֵבִין!

לְפָנָיו עָמַד לֹא אַחֵר מֵאֲשֶׁר אוֹתוֹ נִצּוֹל מִן הַסְּפִינָה הַטּוֹבַעַת! אוֹתוֹ אָדָם שֶׁהוּא כִּסָּה בִּמְעִילוֹ, שֶׁנָּתַן לוֹ מָרָק חַם, שֶׁשָּׂם בְּיָדוֹ שְׁלוֹשָׁה סְלָעִים שֶׁל כֶּסֶף כְּדֵי שֶׁיּוּכַל לַחְזֹר לְעִירוֹ.

אוֹתוֹ נִצּוֹל שֶׁחָזַר לְרוֹמָא, מֻנָּה עַל יְדֵי הַקֵּיסָר לִהְיוֹת שַׂר כָּאן בְּאֶרֶץ יִשְׂרָאֵל!

"אֲנִי מְבַקֵּשׁ דָּבָר אֶחָד, אֲדוֹנִי הַמּוֹשֵׁל," לָחַשׁ בַּר קַפָּרָא בַּעֲנָוָה, "אָנָּא מִמְּךָ, בַּטֵּל אֶת הַגְּזֵרָה וְאֶת הָעֳנָשִׁים הַקָּשִׁים שֶׁגָּזַרְתָּ עַל עַמִּי הַיְּהוּדִים בְּעִירְךָ".

"מֻסְכָּם!" קָרָא הַשַּׂר בְּקוֹל אֶל מוּל כָּל מְשָׁרְתָיו וְחַיָּלָיו. "אֶתֵּן אֶת הַפְּקֻדָּה לְבַטֵּל אֶת הַכֹּל! כֵּיצַד אוּכַל לְסָרֵב לְבַקָּשָׁתוֹ שֶׁל הָאָדָם שֶׁהִצִּיל אֶת חַיַּי וּמֵעוֹלָם לֹא דָּרַשׁ עַל כָּךְ דָּבָר?!"

בְּאוֹתוֹ יוֹם, חָזַר בַּר קַפָּרָא אֶל קְהִלָּתוֹ הַמְּחַכָּה בְּבֵית הַכְּנֶסֶת. לִבּוֹ פָּעַם בְּשִׂמְחָה וְהוֹדָיָה לְבּוֹרֵא עוֹלָם. מָה חָשְׁבוּ עַמּוֹ יִשְׂרָאֵל? שֶׁהוּא הָלַךְ לְסַכֵּן אֶת נַפְשׁוֹ. אַךְ הַקָּדוֹשׁ בָּרוּךְ הוּא, עִם הַהַשְׁגָּחָה הַפְּרָטִית הַמַּדְהִימָה שֶׁלּוֹ, הֵכִין אֶת הַתְּרוּפָה שָׁנִים רַבּוֹת לִפְנֵי הַמַּכָּה.

כְּשֶׁבַּר קַפָּרָא עָשָׂה חֶסֶד נָקִי וְטָהוֹר בְּחוֹף הַיָּם – הוּא בְּעֶצֶם שָׁתַל אֶת הַיְּשׁוּעָה שֶׁתַּצִּיל אֶת הַיְּהוּדִים שָׁנִים אַחַר כָּךְ! עַל כָּךְ אָמַר שְׁלֹמֹה הַמֶּלֶךְ בְּחָכְמָתוֹ: "שַׁלַּח לַחְמְךָ עַל פְּנֵי הַמָּיִם – כִּי בְרֹב הַיָּמִים תִּמְצָאֶנּוּ!".

א. שְׁאֵלַת פְּתִיחָה (לִפְנֵי שֶׁמַּתְחִילִים):

"ילדים, האם פעם עזרתם למישהו סתם ככה, ולאחר המון זמן אותו ילד עזר לכם בדיוק ברגע שהכי הייתם צריכים?

היום אנחנו הולכים לשמוע על גדול בישראל, שעשה חסד קטן עם אדם שלא הכיר, ולא ידע שהחסד הזה – הולך להציל עיר שלמה!"

ב. הֶעָרוֹת בִּצּוּעַ וּסְטוֹרִיטֵלִינְג:

ג. הַמְלָצוֹת לְמַצֶּגֶת (4 שְׁקוּפִיּוֹת בִּקְפִידָה עַל צְנִיעוּת):

ד. פְּעִילוּת מְסַכֶּמֶת (לַבְּחִירָה):

משחק תפקידים – "שרשרת החסד":

מחלקים את הילדים לשלשות. ילד א' הוא במצוקה (נפלה לו קלמר/נשבר לו העיפרון). ילד ב' עוזר לו מכל הלב (נותן לו עיפרון/אוסף איתו). אחרי דקה, מתחלפים, וילד ב' צריך עזרה ממשהו שילד א' יכול לספק. המטרה: להראות פיזית בכיתה איך גלגל החסד מסתובב, ולהדגיש שהעזרה ניתנת תמיד בשמחה וללא ציפייה.

מקורות: מדרש קהלת רבה, פרשה י"א, פסוק א' (על הפסוק: "שַׁלַּח לַחְמְךָ עַל פְּנֵי הַמָּיִם"). אזכור קורותיו של בר קפרא: תלמוד ירושלמי (כמובא ומצוטט במדרשים על חכמינו). ספרות עיבוד חז"ל ("בדרכי אבותינו" וכדו'). (מבוסס על שורות המקור 19, 35, 104 מתוך "שורשי גדולי ישראל / הצעה לתוכנית לימודים" – מעשי החסד של בר קפרא וביטול הגזירה לשר העיר).

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “להלן עיבוד חינוכי, רגשי ומרתק לסיפור המופלא על בר קפרא והצלת שר העיר , המעובד במיוחד לבנים בכיתות ב'-ד' (חינוך דתי-לאומי).” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “הָאִישׁ הַזֶּה הָיָה שַׁיָּךְ לְעַם אַחֵר, נוֹכְרִי גָּמוּר.”?
ג. כיצד הגיב בר קפרא למצב המתואר: “בַּר קַפָּרָא נִגַּשׁ לְקֻפַּת הַחִסָּכוֹן שֶׁלּוֹ, הוֹצִיא מִשָּׁם שְׁלוֹשָׁה מַטְבְּעוֹת שֶׁל "סֶלַע" (סְכוּם כֶּסֶף גָּדוֹל מְאֹד בְּאוֹתָם יָמִים),…”?
ד. מה השתנה בין “הַקָּדוֹשׁ בָּרוּךְ הוּא שׁוֹמֵר אֶת יִשְׂרָאֵל," הוּא אָמַר בְּקוֹל יַצִּיב.” לבין “הוּא קָם בְּמַפְתִּיעַ מִכִּסְאוֹ, מֵרִים אֶת יָדוֹ וְצוֹעֵק לְחַיָּלָיו: "עִצְרוּ!”?
ה. כיצד הגיע הסיפור אל האירוע: “אַךְ הַקָּדוֹשׁ בָּרוּךְ הוּא, עִם הַהַשְׁגָּחָה הַפְּרָטִית הַמַּדְהִימָה שֶׁלּוֹ, הֵכִין אֶת הַתְּרוּפָה שָׁנִים רַבּוֹת לִפְנֵי הַמַּכָּה.”?

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “הצלת שר העיר” על בר קפרא? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “יוֹם אֶחָד, הִגִּיעַ לְקֵיסָרְיָה שַׂר רוֹמָאִי חָדָשׁ אֲשֶׁר מֻנָּה עַל יְדֵי מֶלֶךְ רוֹמִי, הַקֵּיסָר, לִשְׁלֹט בָּאֵזוֹר.”.
ג. אילו בר קפרא לא היה פועל כפי שמתואר ב־“הוּא קָם בְּמַפְתִּיעַ מִכִּסְאוֹ, מֵרִים אֶת יָדוֹ וְצוֹעֵק לְחַיָּלָיו: "עִצְרוּ!” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “אַךְ הַקָּדוֹשׁ בָּרוּךְ הוּא, עִם הַהַשְׁגָּחָה הַפְּרָטִית הַמַּדְהִימָה שֶׁלּוֹ, הֵכִין אֶת הַתְּרוּפָה שָׁנִים רַבּוֹת…” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. להלן עיבוד חינוכי, רגשי ומרתק לסיפור המופלא על בר קפרא והצלת שר העיר ,…
2. הוּא רָעַד מִקֹּר, מַמָּשׁ שִׁנָּיו נָקְשׁוּ זוֹ בָּזוֹ ( טַק-טַק-טַק ).
3. הַנּוֹכְרִי יָצָא לְדַרְכּוֹ, וּבַר קַפָּרָא חָזַר לְבֵית הַמִּדְרָשׁ,…
4. עַל הַקִּירוֹת הָיוּ תְּלוּיִים חֲרָבוֹת וּמָגִנִּים כְּבֵדִים…
5. אוֹתוֹ אָדָם שֶׁהוּא כִּסָּה בִּמְעִילוֹ, שֶׁנָּתַן לוֹ מָרָק חַם,…
6. המטרה: להראות פיזית בכיתה איך גלגל החסד מסתובב, ולהדגיש שהעזרה ניתנת…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “בַּר קַפָּרָא נִגַּשׁ לְקֻפַּת הַחִסָּכוֹן שֶׁלּוֹ, הוֹצִיא מִשָּׁם שְׁלוֹשָׁה מַטְבְּעוֹת שֶׁל "סֶלַע" (סְכוּם כֶּסֶף גָּדוֹל…”.
מכתב: כתבו לבר קפרא בעקבות “הוּא לֹא פָּחַד מִבָּשָׂר וָדָם, אֶלָּא רַק מִמֶּלֶךְ מַלְכֵי הַמְּלָכִים.”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “אוֹתוֹ נִצּוֹל שֶׁחָזַר לְרוֹמָא, מֻנָּה עַל יְדֵי הַקֵּיסָר לִהְיוֹת שַׂר כָּאן בְּאֶרֶץ יִשְׂרָאֵל!”.
דיווח: כתבו ידיעה קצרה שכותרתה “הצלת שר העיר”.

המחזה

רגע מן הסיפורהדמויותמה יאמרו או יעשו?
קוּם, אָחִי הַצּוֹרֵר בְּצָרָתוֹ," אָמַר לוֹ הָרַב בְּקוֹל רַךְ כְּשֶׁהוּא מֵרִים אוֹתוֹ.
הוּא קָם בְּמַפְתִּיעַ מִכִּסְאוֹ, מֵרִים אֶת יָדוֹ וְצוֹעֵק לְחַיָּלָיו: "עִצְרוּ!
