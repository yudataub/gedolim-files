דעתו בוויכוח על הרומאים

דף עבודה בעקבות רבי שמעון בר יוחאי (רשב"י)
שם: כיתה:

המעשה — הסיפור המלא

להלן מערך חינוכי מקיף, רגשי ומרתק על רבי שמעון בר יוחאי (רשב"י) ודעתו בוויכוח על הרומאים , המעובד במיוחד לבנים בכיתות ב'-ד' (חינוך דתי-לאומי). המערך בנוי על פי עקרונות הסטוריטלינג שביקשת, תוך הקפדה חמורה על כבוד התורה, תיאורים חושיים, והשגחה פרטית.

הסיפור מנוקד במלואו כנדרש.

פְּתִיחָה: הָעוֹלָם הַנּוֹצֵץ מִבַּחוּץ

יְלָדִים, דַּמְיְנוּ שֶׁאַתֶּם הוֹלְכִים בִּרְחוֹב עַתִּיק. כָּל אֶבֶן בָּרְחוֹב גְּדוֹלָה מֵהַגֹּבַהּ שֶׁלָּכֶם. אַתֶּם שׁוֹמְעִים סוּסִים דּוֹהֲרִים בְּכֹחַ, אֲנָשִׁים צוֹעֲקִים בְּשָׂפָה זָרָה, וּמִסָּבִיב – בִּנְיְנֵי עֲנָק. גְּשָׁרִים אֲדִירִים מֵאֶבֶן חֲזָקָה, בָּתֵּי מֶרְחָץ עֲנָקִיִּים מְלֵאִים בְּמַיִם חַמִּים וְשַׁיִשׁ מַבְרִיק, וּשְׁוָקִים עֲמוּסִים בְּכָל טוּב. אֵיזֶה יֹפִי, נָכוֹן? הַכֹּל נִרְאֶה כָּל כָּךְ נוֹצֵץ!

מִי בָּנָה אֶת הַבִּנְיָנִים הָאֵלֶּה בְּאֶרֶץ יִשְׂרָאֵל בְּאוֹתָם יָמִים? הָרוֹמָאִים.

הָרוֹמָאִים הָיוּ עַם אַכְזָרִי, גֵּאֶה וְחָזָק. הֵם שָׁלְטוּ בָּאָרֶץ שֶׁלָּנוּ לִפְנֵי כְּאַלְפַּיִם שָׁנָה, רָדְפוּ יְהוּדִים וְגָזְרוּ גְּזֵרוֹת. אֲבָל כְּלַפֵּי חוּץ – הֵם בָּנוּ בִּנְיָנִים מַרְשִׁימִים. וְהָיוּ אֲנָשִׁים שֶׁהִסְתַּנְוְרוּ (הִתְרַשְּׁמוּ יוֹתֵר מִדַּי) מֵהַיֹּפִי הַזֶּה.

הַוִּכּוּחַ מִתַּחַת לַקְּשָׁתוֹת

בְּאוֹתוֹ זְמַן, בְּחֶדֶר שָׁקֵט בְּתוֹךְ בֵּית הַמִּדְרָשׁ, בְּמָקוֹם שֶׁבּוֹ הֵרִיחוּ אֶת רֵיחַ הַקְּלָפִים הַיְּשָׁנִים שֶׁל סִפְרֵי הַתּוֹרָה, יָשְׁבוּ שְׁלוֹשָׁה מִגְּדוֹלֵי עַם יִשְׂרָאֵל. שְׁלוֹשָׁה תַּנָּאִים קְדוֹשִׁים, תַּלְמִידָיו שֶׁל רַבִּי עֲקִיבָא הַגָּדוֹל.

יָשַׁב שָׁם רַבִּי יְהוּדָה, יָשַׁב רַבִּי יוֹסֵי, וְיָשַׁב... הַצַּדִּיק הַגָּדוֹל, רַבִּי שִׁמְעוֹן בַּר יוֹחַאי .

בַּחֶדֶר יָשַׁב בְּשֶׁקֶט גַּם אָדָם נוֹסָף, שֶׁשְּׁמוֹ יְהוּדָה בֶּן גֵּרִים, שֶׁהִקְשִׁיב לַשִּׂיחָה בֵּינֵיהֶם.

הֵם דִּבְּרוּ עַל הַמַּצָּב שֶׁל עַם יִשְׂרָאֵל. לְפֶתַע אָמַר רַבִּי יְהוּדָה: "כַּמָּה נָאִים מַעֲשֵׂיהֶן שֶׁל אֻמָּה זוֹ (שֶׁל הָרוֹמָאִים)! רְאוּ מָה הֵם תִּקְּנוּ וּבָנוּ כָּאן. הֵם עָשׂוּ שְׁוָקִים כְּדֵי שֶׁנּוּכַל לִקְנוֹת, בָּנוּ גְּשָׁרִים גְּדוֹלִים שֶׁנּוּכַל לַחֲצוֹת אֶת הַנְּהָרוֹת מִבְּלִי לִטְבֹּעַ, וְתִקְּנוּ לָנוּ בָּתֵּי מֶרְחָץ כְּדֵי לְהִתְרַחֵץ... אִי אֶפְשָׁר לוֹמַר שֶׁאֵין בְּזֶה דְּבָרִים טוֹבִים".

רַבִּי יוֹסֵי שָׁמַע אֶת הַדְּבָרִים... וְשָׁתַק. הוּא לֹא הִסְכִּים, אֲבָל בָּחַר שֶׁלֹּא לוֹמַר כְּלוּם. הוּא יָדַע שֶׁהָרוֹמָאִים רְשָׁעִים וְאָכְזָרִיִּים.

אֲבָל אָז, נַעֲמַד רַבִּי שִׁמְעוֹן בַּר יוֹחַאי .

הָעֵינַיִם שֶׁלּוֹ זָרְחוּ בְּאֵשׁ שֶׁל אֱמֶת וּקְדֻשָּׁה. עֲבוּר רַבִּי שִׁמְעוֹן, הָאֱמֶת שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא הָיְתָה יְקָרָה מִכָּל זָהָב רוֹמָאִי וּמִכָּל קִיר שֶׁל שַׁיִשׁ. הוּא קָרָא בְּקוֹל צָלוּל:

"כָּל מָה שֶׁתִּקְּנוּ - לֹא תִּקְּנוּ אֶלָּא לְצֹרֶךְ עַצְמָם! הֵם לֹא בָּנוּ אֶת זֶה מִתּוֹךְ אַהֲבָה אֵלֵינוּ! אֶת הַשְּׁוָקִים הֵם בָּנוּ בִּשְׁבִיל חֲטָאִים מְכֹעָרִים! אֶת בָּתֵּי הַמֶּרְחָץ הֵם בָּנוּ רַק כְּדֵי לְפַנֵּק וּלְעַדֵּן אֶת עַצְמָם! וְאֶת הַגְּשָׁרִים... לָמָּה לְדַעְתְּכֶם הֵם בָּנוּ אֶת הַגְּשָׁרִים? כְּדֵי שֶׁכָּל מִי שֶׁיַּעֲבֹר – הֵם יוּכְלוּ לִגְבּוֹת מִמֶּנּוּ מֶכֶס וְכֶסֶף בְּכֹחַ, כְּדֵי לְהַתְעַשֵּׁר מֵהַסֵּבֶל שֶׁל אֲחֵרִים!".

רַבִּי שִׁמְעוֹן קָרַע אֶת הַמַּסֵּכָה הַנּוֹצֶצֶת שֶׁל הָרוֹמָאִים. הוּא הוֹכִיחַ שֶׁמֵּאֲחוֹרֵי הַבִּנְיָנִים הַגְּדוֹלִים, יֵשׁ רַק כֹּחַ, גַּאֲוָה וְרִשְׁעוּת.

הַהַלְשָׁנָה וְהַבְּרִיחָה הַגְּדוֹלָה

אוֹתוֹ אָדָם שֶׁיָּשַׁב בַּחֶדֶר, יְהוּדָה בֶּן גֵּרִים, הָלַךְ לְבֵיתוֹ וְסִפֵּר לִבְנֵי מִשְׁפַּחְתּוֹ אֶת מָה שֶׁנֶּאֱמַר. מִשָּׁם זֶה עָבַר מֵאוֹזֶן לְאוֹזֶן... עַד שֶׁהַמִּלִּים הִגִּיעוּ יָשָׁר לָאַרְמוֹן הַקֵּיסָר הָרוֹמָאִי!

הַקֵּיסָר הָרוֹמָאִי רָתַח מִזַּעַם! "אֵיךְ הַחָכָם הַיְּהוּדִי הַזֶּה מֵעֵז לְזַלְזֵל בָּאִמְפֶּרְיָה שֶׁלָּנוּ?!" שָׁאַג הַקֵּיסָר, קוֹלוֹ מְהַדְהֵד בְּאַרְמוֹן הַשַּׁיִשׁ הַקַּר. מִיָּד יָצְאָה פְּקֻדָּה אַכְזָרִית: "יְהוּדָה שֶׁשִּׁבֵּחַ - יִתְעַלֶּה (יְקַבֵּל תַּפְקִיד חָשׁוּב). יוֹסֵי שֶׁשָּׁתַק - יִגְלֶה לְעִיר צִפּוֹרִי. אֲבָל שִׁמְעוֹן שֶׁגִּנָּה - יֵהָרֵג! ".

יְלָדִים, אַתֶּם מְבִינִים מָה זֶה אוֹמֵר? אֶלְפֵי חַיָּלִים רוֹמָאִים, חֲמוּשִׁים בַּחֲרָבוֹת בַּרְזֶל מַבְרִיקוֹת, כּוֹבַע פְּלָדָה עַל רֹאשָׁם וּמָגִנִּים כְּבֵדִים בִּידֵיהֶם, הִתְחִילוּ לְסַיֵּר בִּרְחוֹבוֹת יְרוּשָׁלַיִם. צַעֲדֵיהֶם "טְרָאח! טְרָאח!" נִשְׁמְעוּ עַל אֲבָנֵי הָרְחוֹב. הֵם חִפְּשׂוּ רַק אָדָם אֶחָד – אֶת רַבִּי שִׁמְעוֹן.

רַבִּי שִׁמְעוֹן לָקַח אֶת בְּנוֹ הַקָּדוֹשׁ, רַבִּי אֶלְעָזָר, וּשְׁנֵיהֶם מִהֲרוּ לְהִתְחַבֵּא בְּבֵית הַמִּדְרָשׁ. כָּל יוֹם שֶׁעָבַר הָיָה סַכָּנָה אֲיֻמָּה. רַעְיָתוֹ, אִשְׁתּוֹ שֶׁל רַבִּי שִׁמְעוֹן, הָיְתָה מַגִּיעָה בַּסֵּתֶר, בְּחֶשְׁכַת הַלַּיְלָה, מְבִיאָה לָהֶם קְצָת לֶחֶם טָרִי שֶׁהֵרִיחַ כְּמוֹ נֶחָמָה, וְכַד קָטָן שֶׁל מַיִם. כָּךְ יָשְׁבוּ שָׁם הָאָב וּבְנוֹ, וְלָמְדוּ תּוֹרָה. אֲבָל כְּשֶׁהַחִפּוּשִׂים הָפְכוּ קָשִׁים יוֹתֵר, יָדַע רַבִּי שִׁמְעוֹן שֶׁזֶּה מְסֻכָּן מִדַּי.

"בְּנִי," לָחַשׁ רַבִּי שִׁמְעוֹן לְרַבִּי אֶלְעָזָר, "נָשִׁים – דַּעְתָּן קַלָּה עֲלֵיהֶן. אִם הָרוֹמָאִים יִתְפְּסוּ אֶת אִמָּא וְיַעֲשׂוּ לָהּ יִסּוּרִים, מֵרֹב כְּאֵב הִיא עֲלוּלָה לְגַלּוֹת הֵיכָן אֲנַחְנוּ. עָלֵינוּ לִבְרֹחַ עַכְשָׁו, מִיָּד, לְמָקוֹם שֶׁאַף אֶחָד בָּעוֹלָם אֵינוֹ מַכִּיר!".

הַנֵּס הַגָּדוֹל בַּמְּעָרָה שֶׁבַּגָּלִיל

בְּחֹשֶׁךְ מֻחְלָט, הִתְגַּנְּבוּ רַבִּי שִׁמְעוֹן וּבְנוֹ מִחוּץ לָעִיר. הֵם הָלְכוּ שָׁעוֹת אֲרֻכּוֹת. הָרַגְלַיִם שֶׁלָּהֶם דָּרְכוּ עַל אֲבָנִים קָשׁוֹת וְעַל קוֹצִים. הֵם עָזְבוּ אֶת הַבָּתִּים, אֶת הַמִּשְׁפָּחָה, אֶת הַשְּׁוָקִים, כְּדֵי שֶׁיּוּכְלוּ לְהַמְשִׁיךְ לִחְיוֹת וְלִלְמֹד אֶת הַתּוֹרָה בִּקְדֻשָּׁה.

בְּסוֹף הַמַּסָּע הַמְּיַגֵּעַ, הֵם מָצְאוּ מְעָרָה עֲמֻקָּה וַחֲשׁוּכָה, אֵי שָׁם בְּהָרֵי הַגָּלִיל. "כָּאן נִסְתַּתֵּר," אָמַר רַבִּי שִׁמְעוֹן.

הֵם נִכְנְסוּ פְּנִימָה. הָאֲוִיר הָיָה קַר וְטָחוּב. הָיָה מְלֹא הַחֶדֶר חוֹל וַעֲפַר סְלָעִים.

"אַבָּא, מָה נֹאכַל? מָה נִשְׁתֶּה כָּאן בְּאֶמְצַע שׁוּם מָקוֹם?" כִּמְעַט בָּכָה רַבִּי אֶלְעָזָר.

יְלָדִים, כָּאן נִכְנֶסֶת לַסִּפּוּר הַהַשְׁגָּחָה הַפְּרָטִית שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא! בּוֹרֵא עוֹלָם לֹא נוֹטֵשׁ אֶת בָּנָיו הַצַּדִּיקִים הַמּוֹסְרִים נַפְשָׁם עַל שְׁמוֹ!

בְּאוֹתוֹ הָרֶגַע, מַמָּשׁ בְּפֶתַח הַמְּעָרָה... פְּלַק! אֲדָמָה קְשִׁיחָה נִבְקְעָה. עֵץ חָרוּבִים פֶּלְאִי צָמַח וְעָלָה, פּוֹרֵשׂ אֶת עֲנָפָיו הַיְרֻקִּים! מִיָּד הֵנֵץ פֵּרוֹת חֲרוּבִים מְתוּקִים! וְלֹא רַק זֶה – קוֹל שֶׁל שִׁקְשׁוּק מַיִם זַכִּים, "פְּשְׁשְׁשׁ... צְלִילִילִיל...", נִשְׁמַע פִּתְאוֹם. מַעְיַן מַיִם טְהוֹרִים וְקָרִים בָּקַע מִתּוֹךְ הַסֶּלַע! יֵשׁ לָהֶם אֹכֶל! יֵשׁ לָהֶם מַיִם! הַכֹּל מֵהַקָּדוֹשׁ בָּרוּךְ הוּא!

רַבִּי שִׁמְעוֹן וְרַבִּי אֶלְעָזָר יָדְעוּ שֶׁהֵם יֵשְׁבוּ כָּאן שָׁנִים אֲרֻכּוֹת, וְהַבְּגָדִים שֶׁלָּהֶם יִקָּרְעוּ וְיִבְלוּ. מָה עָשׂוּ? בְּלֶכְתָּם לִלְמֹד תּוֹרָה כָּל שְׁעוֹת הַיּוֹם, פָּשְׁטוּ הַצַּדִּיקִים אֶת בִּגְדֵיהֶם וּקְבָרוּ עַצְמָם בְּתוֹךְ חוֹל הַמְּעָרָה כִּמְעַט עַד צַוָּארָם. הַחוֹל שָׂרַט אֶת עוֹרָם, אֲבָל לָהֶם לֹא הָיָה אִכְפַּת! הֵם לָמְדוּ אֶת הַסּוֹדוֹת הֲכִי עֲמֻקִּים שֶׁל הַתּוֹרָה. רַק כְּשֶׁהִגִּיעַ זְמַן תְּפִלָּה – לָבְשׁוּ הַבְּגָדִים, הִתְפַּלְּלוּ בְּקוֹל שֶׁקָּרַע רְקִיעִים, וְשָׁבוּ שׁוּב לְחוֹל וְלַעֲפַר הַמְּעָרָה.

שְׁתֵּים עֶשְׂרֵה שָׁנָה שֶׁל קְדֻשָּׁה

שְׁתֵּים עֶשְׂרֵה שָׁנָה יָשְׁבוּ הַצַּדִּיקִים בַּמְּעָרָה. בַּחוּץ – הַקֵּיסָר כָּעַס וְחִפֵּשׂ. בִּפְנִים – אוֹר נִפְלָא שֶׁל קְדֻשָּׁה עָטַף אוֹתָם. הֵם לֹא הָיוּ בּוֹדְדִים וַעֲצוּבִים. נִשְׁמוֹת הַצַּדִּיקִים שֶׁבְּגַן עֵדֶן מַמָּשׁ יָרְדוּ לִלְמֹד אִתָּם!

יוֹם אֶחָד, אַחֲרֵי לֹא פָּחוֹת מִשְּׁתֵּים עֶשְׂרֵה שָׁנִים אֲרֻכּוֹת, הִגִּיעַ לְפֶתַח הַמְּעָרָה אֵלִיָּהוּ הַנָּבִיא בִּכְבוֹדוֹ וּבְעַצְמוֹ!

אֵלִיָּהוּ עָמַד מִחוּץ לַמְּעָרָה, קוֹלוֹ כְּקַרְנֵי שֶׁמֶשׁ שֶׁחוֹדְרוֹת אֶת הַחֹשֶׁךְ, וְקָרָא: "מִי יוֹדִיעַ לְבַר יוֹחַאי שֶׁמֵּת הַקֵּיסָר וּבָטְלָה הַגְּזֵרָה?!".

רַבִּי שִׁמְעוֹן וְרַבִּי אֶלְעָזָר שָׁמְעוּ. בִּדְמָעוֹת שֶׁל הִתְרַגְּשׁוּת וְשִׂמְחָה, מְכֻסִּים בִּצְלָקוֹת אֲבָל מְאִירִים כְּמַלְאָכִים, הֵם יָצְאוּ הַחוּצָה לָאֲוִיר הָעוֹלָם. צַדִּיקִים גְּדוֹלִים שֶׁנִּצְּחוּ אֶת הָאִמְפֶּרְיָה הָרוֹמָאִית הָעֲנָקִית, לֹא עִם חֶרֶב אוֹ רוֹמַח – אֶלָּא עִם עֵץ חָרוּבִים וְאֵשׁ שֶׁל תּוֹרָה!

ME (רַשְׁבִּ"י): רַבִּי שִׁמְעוֹן לֹא הִתְבַּלְבֵּל מֵהַיֹּפִי וּמֵהַגֹּדֶל שֶׁל הָרוֹמָאִים. הוּא יָדַע שֶׁאִם מַשֶּׁהוּ יָפֶה מִבַּחוּץ אֲבָל בִּפְנִים הוּא עָשׂוּי מֵאַכְזָרִיּוּת, גַּאֲוָה וְשִׂנְאָה – אָז הוּא שֶׁקֶר! בִּשְׁבִיל לְשַׁמֵּר אֶת הָאֱמֶת (הַתּוֹרָה) הוּא הָיָה מוּכָן לוֹתֵר עַל מַיִם קָרִים וּבָתִּים נוֹצְצִים, וְלָלֶכֶת לִמְעָרָה וּבִלְבַד שֶׁלֹּא לְוַתֵּר עַל תּוֹרַת ה'.

WE (אֲנַחְנוּ): לִפְעָמִים אֲנַחְנוּ מִתְבַּלְבְּלִים. אֲנַחְנוּ רוֹאִים רְחוֹב יָפֶה, צַעֲצוּעַ יָקָר, אוֹ הִתְנַהֲגוּת שֶׁל אֲנָשִׁים שֶׁנִּרְאִים לָנוּ קוּלִים וַחֲזָקִים (גַּם אִם הֵם מְדַבְּרִים בְּשָׂפָה זָרָה לָנוּ וְרָעָה). אֲנַחְנוּ רוֹצִים לִהְיוֹת כְּמוֹתָם, וְלִפְעָמִים גַּם לְוַתֵּר קְצָת עַל הַתְּפִלָּה שֶׁלָּנוּ, כְּדֵי لְמַצְא חֵן בְּעֵינֵיהֶם שֶׁל "הַחֲזָקִים".

YOU (אַתֶּם): מָחָר, כְּשֶׁאַתֶּם הוֹלְכִים לְבֵית הַסֵּפֶר, אוֹ קוֹרְאִים אֵיזֶה חֲבֵרִים בּוֹחֲרִים – תִּזְכְּרוּ אֶת רַשְׁבִּ"י! אַל תִּבְחֲרוּ אֶת הֶחָבֵר רַק כִּי יֵשׁ לוֹ צַעֲצוּעִים יָפִים אוֹ שְׁרִירִים וְגַאֲוָה כְּמוֹ הָרוֹמָאִים. תִּבְחֲרוּ אֶת מִי שֶׁיֵּשׁ לוֹ אֱמֶת בַּלֵּב. דַּעוּ שֶׁמִּי שֶׁלּוֹמֵד תּוֹרָה וּמִתְחַבֵּר לְבוֹרֵא עוֹלָם – בּוֹרֵא עוֹלָם שׁוֹמֵר עָלָיו, יְסַדֵּר לוֹ מַיִם וְחָרוּבִים, וְאַף מֶלֶךְ רוֹמָאִי לֹא יוּכַל לְנַצֵּחַ אוֹתוֹ שֶׁל עַם יִשְׂרָאֵל אֲשֶׁר דָּבֵק בַּתּוֹרָה!

א. שְׁאֵלַת פְּתִיחָה (לִיצִירַת סַקְרָנוּת):

"יְלָדִים, סַפְּרוּ לִי: מָה קוֹרֶה אִם יֶלֶד אֶחָד גָּדוֹל, חָזָק וְאַכְזָרִי מַרְבִּיץ לְמִישֶׁהוּ, וְכֻלָּם שׁוֹתְקִים מִפַּחַד – אֲבָל יֵשׁ יֶלֶד אֶחָד שֶׁאֵינֶנּוּ מְפַחֵד מֵהַגֹּדֶל שֶׁלּוֹ, וְאוֹמֵר בְּקוֹל רָם: 'אַתָּה טוֹעֶה!'?

הַיּוֹם נִלְמַד עַל עֲנַק הַתּוֹרָה הַקָּדוֹשׁ שֶׁעָשָׂה בְּדִיּוּק אֶת זֶה מול הָאִמְפֶּרְיָה הֲכִי חֲזָקָה בָּעוֹלָם – רַבִּי שִׁמְעוֹן בַּר יוֹחַאי!"

ב. הֶעָרוֹת בִּצּוּעַ וּסְטוֹרִיטֶלִינְג (איך להקריא נכון?):

ג. הַמְלָצוֹת לְמַצֶּגֶת (4 שְׁקוּפִיּוֹת צְנוּעוֹת וְאִיכוּתִיּוֹת):

ד. פְּעִילוּת מְסַכֶּמֶת ("הַמִּשְׁפָּט שֶׁל הַמְּעָרָה"):

תֵּן לַיְּלָדִים מְשִׂימָה. "יְלָדִים, אִם לָכֶם הָיָה הֶחָרוּב וְהַמַּיִם, אֶל מִי הֱיִיתֶם אוֹמְרִים 'תּוֹדָה רַבָּה הַקָּדוֹשׁ בָּרוּךְ הוּא'? כָּל אֶחָד כּוֹתֵב עַל פֶּתֶק שֶׁבְּצוּרַת 'עֲלֵה חָרוּב' מִשְׁפָּט אֶחָד תּוֹרָנִי אוֹ תְּפִלָּה יָפָה שֶׁרַשְׁבִּ"י מְלַמֵּד אוֹתָנוּ." אֶפְשָׁר תּוֹךְ כְּדֵי לְחַלֵּק לָהֶם כַּמָּה חֲרוּבִים אֲמִתִּיִּים שֶׁיִּטְעֲמוּ מִטַּעַם נִסֵּי הַמְּעָרָה.

מקורות: תלמוד בבלי, מסכת שבת, דף ל"ג עמוד ב': ("ואמאי קרו ליה ראש המדברים בכל מקום? דיתבי רבי יהודה ורבי יוסי ורבי שמעון, ויתיב יהודה בן גרים גבייהו... פתח ר' יהודה ואמר כמה נאים מעשיהן של אומה זו... נענה רשב"י ואמר כל מה שתקנו לא תקנו אלא לצורך עצמן... אזלו טשו במערתא, איתרחיש ניסא איברי להו חרובא ועינא דמיא...").

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “להלן מערך חינוכי מקיף, רגשי ומרתק על רבי שמעון בר יוחאי (רשב"י) ודעתו בוויכוח על הרומאים , המעובד במיוחד לבנים בכיתות ב'-ד' (חינוך דתי-לאומי).” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “בַּחֶדֶר יָשַׁב בְּשֶׁקֶט גַּם אָדָם נוֹסָף, שֶׁשְּׁמוֹ יְהוּדָה בֶּן גֵּרִים, שֶׁהִקְשִׁיב לַשִּׂיחָה בֵּינֵיהֶם.”?
ג. כיצד הגיב רבי שמעון בר יוחאי (רשב"י) למצב המתואר: “אוֹתוֹ אָדָם שֶׁיָּשַׁב בַּחֶדֶר, יְהוּדָה בֶּן גֵּרִים, הָלַךְ לְבֵיתוֹ וְסִפֵּר לִבְנֵי מִשְׁפַּחְתּוֹ אֶת מָה שֶׁנֶּאֱמַר.”?
ד. מה השתנה בין “בְּחֹשֶׁךְ מֻחְלָט, הִתְגַּנְּבוּ רַבִּי שִׁמְעוֹן וּבְנוֹ מִחוּץ לָעִיר.” לבין “הַחוֹל שָׂרַט אֶת עוֹרָם, אֲבָל לָהֶם לֹא הָיָה אִכְפַּת!”?
ה. כיצד הגיע הסיפור אל האירוע: “אֲנַחְנוּ רוֹאִים רְחוֹב יָפֶה, צַעֲצוּעַ יָקָר, אוֹ הִתְנַהֲגוּת שֶׁל אֲנָשִׁים שֶׁנִּרְאִים לָנוּ קוּלִים וַחֲזָקִים (גַּם אִם הֵם מְדַבְּרִים בְּשָׂפָה…”?

אֲבָל כְּשֶׁהַחִפּוּשִׂים הָפְכוּ קָשִׁים יוֹתֵר, יָדַע רַבִּי שִׁמְעוֹן שֶׁזֶּה מְסֻכָּן מִדַּי.□□
בסיפור לא התרחש הדבר הבא: בּוֹרֵא עוֹלָם לֹא נוֹטֵשׁ אֶת בָּנָיו הַצַּדִּיקִים הַמּוֹסְרִים נַפְשָׁם עַל שְׁמוֹ!□□
בִּדְמָעוֹת שֶׁל הִתְרַגְּשׁוּת וְשִׂמְחָה, מְכֻסִּים בִּצְלָקוֹת אֲבָל מְאִירִים כְּמַלְאָכִים, הֵם יָצְאוּ הַחוּצָה לָאֲוִיר הָעוֹלָם.□□

3. מה היה ראשון?

כתבו ליד כל אירוע מספר מ־1 עד 5 לפי סדר התרחשותו בסיפור.

□ אִם הָרוֹמָאִים יִתְפְּסוּ אֶת אִמָּא וְיַעֲשׂוּ לָהּ יִסּוּרִים, מֵרֹב כְּאֵב הִיא עֲלוּלָה לְגַלּוֹת הֵיכָן אֲנַחְנוּ.
□ להלן מערך חינוכי מקיף, רגשי ומרתק על רבי שמעון בר יוחאי (רשב"י) ודעתו בוויכוח על הרומאים , המעובד במיוחד לבנים בכיתות ב'-ד' (חינוך דתי-לאומי).
□ הַיּוֹם נִלְמַד עַל עֲנַק הַתּוֹרָה הַקָּדוֹשׁ שֶׁעָשָׂה בְּדִיּוּק אֶת זֶה מול הָאִמְפֶּרְיָה הֲכִי חֲזָקָה בָּעוֹלָם – רַבִּי שִׁמְעוֹן בַּר יוֹחַאי!
□ עֲבוּר רַבִּי שִׁמְעוֹן, הָאֱמֶת שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא הָיְתָה יְקָרָה מִכָּל זָהָב רוֹמָאִי וּמִכָּל קִיר שֶׁל שַׁיִשׁ.
□ הֵם לָמְדוּ אֶת הַסּוֹדוֹת הֲכִי עֲמֻקִּים שֶׁל הַתּוֹרָה.

לפני ואחרי

מה קרה לפני: “אִם הָרוֹמָאִים יִתְפְּסוּ אֶת אִמָּא וְיַעֲשׂוּ לָהּ יִסּוּרִים, מֵרֹב כְּאֵב הִיא עֲלוּלָה לְגַלּוֹת הֵיכָן אֲנַחְנוּ.”?
מה קרה אחרי: “הֵם לָמְדוּ אֶת הַסּוֹדוֹת הֲכִי עֲמֻקִּים שֶׁל הַתּוֹרָה.”?

סיבה ותוצאה מתוך המעשה

האירוע הנתוןמה גרם לו?מה הייתה תוצאתו?
כָּל מָה שֶׁתִּקְּנוּ - לֹא תִּקְּנוּ אֶלָּא לְצֹרֶךְ עַצְמָם!
בּוֹרֵא עוֹלָם לֹא נוֹטֵשׁ אֶת בָּנָיו הַצַּדִּיקִים הַמּוֹסְרִים נַפְשָׁם עַל שְׁמוֹ!

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “דעתו בוויכוח על הרומאים” על רבי שמעון בר יוחאי (רשב"י)? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “אֲבָל כְּשֶׁהַחִפּוּשִׂים הָפְכוּ קָשִׁים יוֹתֵר, יָדַע רַבִּי שִׁמְעוֹן שֶׁזֶּה מְסֻכָּן מִדַּי.”.
ג. אילו רבי שמעון בר יוחאי (רשב"י) לא היה פועל כפי שמתואר ב־“הַחוֹל שָׂרַט אֶת עוֹרָם, אֲבָל לָהֶם לֹא הָיָה אִכְפַּת!” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “אֲנַחְנוּ רוֹאִים רְחוֹב יָפֶה, צַעֲצוּעַ יָקָר, אוֹ הִתְנַהֲגוּת שֶׁל אֲנָשִׁים שֶׁנִּרְאִים לָנוּ קוּלִים וַחֲזָקִים (גַּם אִם…” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. להלן מערך חינוכי מקיף, רגשי ומרתק על רבי שמעון בר יוחאי (רשב"י) ודעתו…
2. לְפֶתַע אָמַר רַבִּי יְהוּדָה: "כַּמָּה נָאִים מַעֲשֵׂיהֶן שֶׁל אֻמָּה…
3. אֶלְפֵי חַיָּלִים רוֹמָאִים, חֲמוּשִׁים בַּחֲרָבוֹת בַּרְזֶל…
4. בְּסוֹף הַמַּסָּע הַמְּיַגֵּעַ, הֵם מָצְאוּ מְעָרָה עֲמֻקָּה…
5. יוֹם אֶחָד, אַחֲרֵי לֹא פָּחוֹת מִשְּׁתֵּים עֶשְׂרֵה שָׁנִים אֲרֻכּוֹת,…
6. הַיּוֹם נִלְמַד עַל עֲנַק הַתּוֹרָה הַקָּדוֹשׁ שֶׁעָשָׂה בְּדִיּוּק אֶת…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “אוֹתוֹ אָדָם שֶׁיָּשַׁב בַּחֶדֶר, יְהוּדָה בֶּן גֵּרִים, הָלַךְ לְבֵיתוֹ וְסִפֵּר לִבְנֵי מִשְׁפַּחְתּוֹ אֶת מָה שֶׁנֶּאֱמַר.”.
מכתב: כתבו לרבי שמעון בר יוחאי (רשב"י) בעקבות “בּוֹרֵא עוֹלָם לֹא נוֹטֵשׁ אֶת בָּנָיו הַצַּדִּיקִים הַמּוֹסְרִים נַפְשָׁם עַל שְׁמוֹ!”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “בִּדְמָעוֹת שֶׁל הִתְרַגְּשׁוּת וְשִׂמְחָה, מְכֻסִּים בִּצְלָקוֹת אֲבָל מְאִירִים כְּמַלְאָכִים, הֵם יָצְאוּ הַחוּצָה לָאֲוִיר…”.
דיווח: כתבו ידיעה קצרה שכותרתה “דעתו בוויכוח על הרומאים”.

המחזה

רגע מן הסיפורהדמויותמה יאמרו או יעשו?
כָּל מָה שֶׁתִּקְּנוּ - לֹא תִּקְּנוּ אֶלָּא לְצֹרֶךְ עַצְמָם!
הַחוֹל שָׂרַט אֶת עוֹרָם, אֲבָל לָהֶם לֹא הָיָה אִכְפַּת!
