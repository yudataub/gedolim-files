13 הפנים שנשכחו — ר' חייא החזיר 7 והכובס 6

דף עבודה בעקבות רבי יהודה הנשיא (רבי)
שם: כיתה:

המעשה — הסיפור המלא

פְּתִיחָה: קוֹל הַתּוֹרָה בְּאַרְמוֹן הַנָּשִׂיא

יְלָדִים, נַסּוּ לִשְׁמֹעַ אֶת הַקּוֹלוֹת שֶׁל מַה שֶׁאֲנִי הוֹלֵךְ לְתָאֵר לָכֶם עַכְשָׁו.

אֲנַחְנוּ נִמְצָאִים בְּאֶרֶץ יִשְׂרָאֵל, לִפְנֵי כְּאַלְפַּיִם שָׁנָה. בְּעִיר יָפָה וּשְׁקֵטָה גָּר מַנְהִיג עַם יִשְׂרָאֵל. קוֹרְאִים לוֹ רַבִּי יְהוּדָה הַנָּשִׂיא, אֲבָל כֻּלָּם כָּל כָּךְ אָהֲבוּ וְכִבְּדוּ אוֹתוֹ, עַד שֶׁקָּרְאוּ לוֹ פָּשׁוּט "רַבִּי", אוֹ "רַבֵּינוּ הַקָּדוֹשׁ".

הַבַּיִת שֶׁל רַבִּי הָיָה מַמָּשׁ כְּמוֹ אַרְמוֹן. רַבִּי הָיָה עָשִׁיר מְאֹד. עַל הַשֻּׁלְחָן הָיוּ מֻנָּחִים תָּמִיד פֵּרוֹת נְדִירִים וְטָעִימִים, רֵיחַ שֶׁל לֶחֶם טָרִי עָלָה מֵהַמִּטְבָּח, וְהַבְּגָדִים שֶׁל רַבִּי הָיוּ עֲשׂוּיִים מִבַּד רַךְ וְנָעִים.

אֲבָל כָּל הָעֹשֶׁר הַזֶּה לֹא עִנְיֵן אֶת רַבִּי יְהוּדָה כְּלָל. אִם הֱיִיתֶם נִכְנָסִים לַחֶדֶר הַפְּרָטִי שֶׁלּוֹ, לֹא הֱיִיתֶם מוֹצְאִים אוֹתוֹ סוֹפֵר מַטְבְּעוֹת זָהָב. לֹא וָלֹא!

מֵהַבֹּקֶר עַד שְׁעוֹת הַלַּיְלָה הַמְאֻחָרוֹת, בָּקַע מֵהַחֶדֶר קוֹל מָתוֹק וְעָרֵב. זֶה הָיָה קוֹל לִמּוּד הַתּוֹרָה!

בְּאוֹתָם יָמִים עָסַק רַבִּי יְהוּדָה הַנָּשִׂיא בְּהֲלָכָה אַחַת עֲמֻקָּה וְקָשָׁה מְאֹד. הַהֲלָכָה הַזּוֹ הָיְתָה כָּל כָּךְ מְסֻבֶּכֶת, עַד שֶׁרַבִּי מָצָא לָהּ שְׁלוֹשָׁה עָשָׂר (13) הֶסְבֵּרִים שׁוֹנִים! בִּשְׂפַת תַּלְמִידֵי הַחֲכָמִים זֶה נִקְרָא שֶׁהוּא מָצָא "שְׁלוֹשָׁה עָשָׂר פָּנִים לַהֲלָכָה".

וְכָךְ, יוֹמָם וָלַיְלָה, יָשַׁב רַבִּי, נִגֵּן אֶת מִלּוֹת הַתּוֹרָה מִתּוֹךְ מְגִילּוֹת הַקְּלָף, שֶׁרֵיחַ שֶׁל קֹדֶשׁ נָדַף מֵהֶן, וְשִׁנֵּן שׁוּב וָשׁוּב אֶת י"ג (13) הַהֶסְבֵּרִים, מֵהָרִאשׁוֹן וְעַד לָאַחֲרוֹן, כְּדֵי שֶׁלֹּא יִשָּׁכְחוּ.

הָאִישׁ מֵאֲחוֹרֵי הַקִּיר

אֲבָל רַבִּי יְהוּדָה לֹא יָדַע שֶׁהוּא אֵינוֹ לּוֹמֵד לְבַדּוֹ!

מִחוּץ לַחַלּוֹן הַגָּדוֹל שֶׁל בֵּית הַמִּדְרָשׁ, בֶּחָצֵר, עָמַד אִישׁ פָּשׁוּט. זֶה הָיָה הַכּוֹבֵס שֶׁל הָעִיר (בָּאֲרַמִּית קוֹרְאִים לוֹ 'קַצְרָא'). תַּפְקִידוֹ שֶׁל הַכּוֹבֵס הָיָה עֲבוֹדָה פִיזִית קָשָׁה שֶׁל עֲבוֹדַת כַּפַּיִם מְעַיֶּפֶת. הוּא הָיָה טוֹבֵל אֶת הַבְּגָדִים בְּגִיגִיּוֹת שֶׁל מַיִם מְלֵאִים בְּסַבּוֹן שֶׁל פַּעַם שֶׁנִּקְרָא בּוֹרִית. הָיָה לְסַבּוֹן הַזֶּה רֵיחַ חָזָק וְחָרִיף מְאֹד. הַכּוֹבֵס הָיָה מְשַׁפְשֵׁף בְּכֹחַ אֶת הַבְּגָדִים, סוֹחֵט אוֹתָם עַד שֶׁשְּׁרִירֵי יָדָיו כָּאֲבוּ, וּמַכֶּה עֲלֵיהֶם עַל הַסְּלָעִים כְּדֵי לְהוֹצִיא מֵהֶם אֶת הַלִּכְלוּךְ.

פְּלִיק... פְּלַק... סְוִוישׁ... אֵלּוּ הָיוּ הַקּוֹלוֹת שֶׁל הַמַּיִם וְהַבְּגָדִים בֶּחָצֵר.

אֲבָל הַכּוֹבֵס הַזֶּה הָיָה אוֹהֵב תּוֹרָה אֲמִתִּי. כְּשֶׁהָיָה שׁוֹטֵף וּמְשַׁפְשֵׁף אֶת הַבְּגָדִים, הוּא לֹא חָשַׁב עַל כֶּסֶף, וְלֹא חָשַׁב עַל הֶעָיֵפוּת. הוּא כִּוֵּן אֶת הָאָזְנַיִם שֶׁלּוֹ עַד לַקָּצֶה... רַק כְּדֵי לְהַקְשִׁיב לַקּוֹל הַבּוֹקֵעַ מֵהַחַלּוֹן – קוֹלוֹ הַמָּתוֹק שֶׁל רַבִּי יְהוּדָה הַנָּשִׂיא!

בְּכָל פַּעַם שֶׁרַבִּי חָזַר עַל הֲלָכָה, הַכּוֹבֵס הָיָה לוֹחֵשׁ אוֹתָהּ לְעַצְמוֹ. תוֹךְ כְּדֵי כִּבּוּס וּסְחִיטָה, הוּא חָזַר בְּדִיּוּק מֻפְלָא עַל כָּל שְׁלוֹשָׁה עָשָׂר הַהֶסְבֵּרִים הָעֲמֻקִּים! מַמָּשׁ "תּוֹרָה וַעֲבוֹדָה". יָדַיִם בַּמַּיִם – וְלֵב בַּשָּׁמַיִם!

חֹשֶׁךְ שֶׁל שִׁכְחָה

וְאָז, קָרָה דָּבָר נוֹרָא.

בְּאֶחָד הַיָּמִים, הִתְחִיל רַבִּי לְהַרְגִּישׁ רַע. חֹם גּוּפוֹ עָלָה, פָּנָיו הֶאֱדִימוּ, וְזֵעָה קָרָה כִּסְּתָה אֶת מִצְחוֹ. רַבֵּינוּ הַקָּדוֹשׁ חָלָה בְּמַחֲלָה קָשָׁה מְאֹד. תַּלְמִידָיו בָּכוּ וְהִתְפַּלְּלוּ, וְכָל בֵּית הַמִּדְרָשׁ עָצַר מִלֶּכֶת. שֶׁקֶט עָמוֹק וְעָצוּב יָרַד עַל הָעִיר כֻּלָּהּ. אֲפִלּוּ הַכּוֹבֵס בַּחוּץ עָצַר מִמְּלַאכְתּוֹ וְנָשָׂא תְּפִלָּה גְּדוֹלָה לְרִבּוֹנוֹ שֶׁל עוֹלָם.

בְּחַסְדֵּי ה', לְאַחַר תְּקוּפָה קָשָׁה, רַבִּי הִבְרִיא! הַחֹם יָרַד, וְכוֹחוֹ חָזַר אֵלָיו מְעַט מְעַט.

רַבִּי קָם מִמִּטָּתוֹ, נָטַל אֶת יָדָיו, וּמִיָּד מִהֵר לַסְּפָרִים שֶׁלּוֹ. הוּא רָצָה לַחְזֹר אֶל 13 הַהֶסְבֵּרִים שֶׁל הַהֲלָכָה הַקָּשָׁה. הוּא פָּתַח אֶת הַסֵּפֶר, עָצַם עֵינַיִם, רָצָה לְהוֹצִיא מִפִּיו אֶת הַהֲלָכָה – אֲבָל... שֶׁקֶט!

הַמִּלִּים... נֶעֶלְמוּ. הַזִּכָּרוֹן נִמְחַק. בִּגְלַל הַמַּחֲלָה הַקָּשָׁה, שָׁכַח רַבִּי יְהוּדָה אֶת כָּל תְּלַת-עֶשְׂרֵה (13) הַפָּנִים שֶׁל הַהֲלָכָה שֶׁעָלֶיהָ עָמַל כָּל כָּךְ קָשֶׁה!

רַבִּי הָיָה שָׁבוּר מִצַּעַר. הוּא לֹא בָּכָה שֶׁהוּא חָלָה, הוּא לֹא בָּכָה שֶׁכּוֹאֵב לוֹ בַּגּוּף. הוּא בָּכָה – עַל הַתּוֹרָה שֶׁהָלְכָה לְאִבּוּד!

קֶרֶן אוֹר רִאשׁוֹנָה – רַבִּי חִיָּא

רַבִּי קָרָא לַתַּלְמִיד הֲכִי קָרוֹב וְהֲכִי גָּדוֹל שֶׁלּוֹ, רַבִּי חִיָּא.

"חִיָּא בְּנִי," בִּקֵּשׁ רַבִּי בְּקוֹל רוֹעֵד, "הַאִם הִסְפַּקְתָּ לִלְמֹד מִמֶּנִּי אֶת הַהֲלָכָה הַהִיא עִם שְׁלוֹשָׁה עָשָׂר הַהֶסְבֵּרִים? הַאִם אַתָּה זוֹכֵר מַשֶּׁהוּ?".

רַבִּי חִיָּא הִתְאַמֵּץ לְהִזָּכֵר. "מוֹרִי וְרַבִּי," אָמַר, "לָמַדְתִּי מִמְּךָ רַק חֵלֶק מִמֶּנָּה. אֲנִי זוֹכֵר שִׁבְעָה הֶסְבֵּרִים מִתּוֹךְ הַשְּׁלוֹשָׁה עָשָׂר. שִׁשָּׁה יְסוֹדוֹת חֲסֵרִים לִי".

רַבִּי יְהוּדָה שָׂמַח עַל הַשִּׁבְעָה שֶׁנִּצְּלוּ, אֲבָל לִבּוֹ נִצְבַּט בִּכְאֵב. מַה עִם הַשִּׁשָּׁה הַחֲסֵרִים? אֵיךְ יוֹדְעוּ עַכְשָׁו אֶת הָאֱמֶת הָאֱלֹקִית? אֵיךְ נַחֲזִיר אֶת הַתּוֹרָה?

הַהַפְתָּעָה הַגְּדוֹלָה שֶׁל הַכּוֹבֵס

לְפֶתַע, אֶחָד הַתַּלְמִידִים נִזְכַּר בְּמַשֶּׁהוּ. "רַבִּי," אָמַר כִּמְעַט בִּלְחִישָׁה, "כְּשֶׁלָּמַדְתָּ אֶת הַהֲלָכָה הַזּוֹ לִפְנֵי שֶׁחָלִיתָ, הַחַלּוֹן הָיָה פָּתוּחַ... בַּחוּץ עָמַד הַכּוֹבֵס הַפָּשׁוּט... אֲנִי זָכוּר שֶׁרָאִיתִי אֶת הַשְּׂפָתַיִם שֶׁלּוֹ זָזוֹת בַּהֶקְשֵׁב רַב".

לֹא הָיָה צָרִיךְ לַחְשֹׁב פַּעֲמַיִם. רַבִּי חִיָּא יָצָא מִיָּד הַחוּצָה, אֶל הֶחָצֵר עִם הַגִּיגִיּוֹת.

הַכּוֹבֵס עָמַד שָׁם, יָדָיו קְפוּאוֹת מֵהַמַּיִם וְרַטּוֹבוֹת הָיוּ בִּבְגָדִים כְּבֵדִים.

"אָחִי הַיָּקָר," פָּנָה אֵלָיו רַבִּי חִיָּא בְּכָבוֹד הָרָאוּי. "תַּגִּיד לִי, מִמַּה שֶּׁשָּׁמַעְתָּ מִקּוֹלוֹ שֶׁל נְשִׂיא יִשְׂרָאֵל, הַאִם אַתָּה זוֹכֵר אֶת י"ג הַהֶסְבֵּרִים שֶׁל הַהֲלָכָה הַהִיא?".

הַכּוֹבֵס הִסְמִיק, הוריד את ראשו בענווה ואמר: "כֵּן, כְּבוֹדוֹ. חָזַרְתִּי עֲלֵיהֶם הַרְבֵּה, שֶׁמָּא אֶזְכֶּה לָדַעַת תּוֹרָה".

הַכּוֹבֵס חָזַר מִלָּה בְּמִלָּה עַל אוֹתָם שִׁשָּׁה (6) הֶסְבֵּרִים שֶׁנֶּעֶלְמוּ! רַבִּי חִיָּא שָׁמַע, וְזִכְרוֹנוֹ חָזַר אֵלָיו מִיָּד. שִׂמְחָה גְּדוֹלָה פָּרְצָה בְּלִבּוֹ! הַכּוֹבֵס, עִם הַבְּגָדִים הָרְטֻבִּים וְרֵיחַ הַסַּבּוֹן – הִצִּיל אֶת סוֹדוֹת הַתּוֹרָה!

"אַתָּה עָשִׂיתָ אוֹתִי!"

רַבִּי חִיָּא רָץ מִיָּד לַחַדְרוֹ שֶׁל רַבִּי יְהוּדָה הַנָּשִׂיא. עִם חִיּוּךְ רָחָב, הוּא חָזַר בְּאָזְנָיו עַל אוֹתָם שִׁשָּׁה פָּנִים קְדוֹשִׁים, וְיַחַד אִתָּם שְׁלָשׁ-עֶשְׂרֵה (13) הַפָּנִים הֻשְׁלְמוּ! התּוֹרָה חָזְרָה לִמְקוֹמָהּ, שלמה ומאירה.

ילדים, שימו לב היטב למה שקרה עכשיו.

אדם רגיל, שהיה נשיא חשוב ועשיר, היה שולח משרת שייתן לכובס כמה מטבעות כסף ויגיד לו "תודה, תמשיך לכבס". אֲבָל רַבִּי יְהוּדָה הַנָּשִׂיא הָיָה עָנָיו וְטָהוֹר אֲמִתִּי!

רַבִּי שָׂם אֶת הַבְּגָדִים הַיָּפִים שֶׁלּוֹ, יָרַד בְּעַצְמוֹ מִמִּטָּתוֹ אֶל מִחוּץ לַבַּיִת, עַד לֶחָצֵר. כְּשֶׁהוּא עָמַד מוּל הַכּוֹבֵס הַפָּשׁוּט, שֶׁהָיָה עֲדַיִן רָטוֹב וְלוֹבֵשׁ בִּגְדֵי עֲבוֹדָה מְלֻכְלָכִים, הִבִּיט בּוֹ נְשִׂיא יִשְׂרָאֵל בְּעֵינַיִם בּוֹרְקוֹת, כִּמְעַט עִם דְּמָעוֹת שֶׁל אוֹר, וְאָמַר לוֹ מִלִּים מַדְהִימוֹת מִלֵּב וְנֶפֶשׁ:

"אַתָּה – עָשִׂיתָ אֶת חִיָּא! וְחִיָּא – עָשָׂה אוֹתִי!"

אֵיזֶה כָּבוֹד עָצוּם! מַנְהִיג הַדּוֹר אָמַר לְאִישׁ הַכּוֹבֵס הַפָּשׁוּט: "בִּזְכוּתְךָ יֵשׁ לִי שׁוּב תּוֹרָה! בִּזְכוּתְךָ, הַתּוֹרָה חָיָה!". וְהַכֹּל – מִשּׁוּם שֶׁהַכּוֹבֵס שֶׁהַיָּדַיִם שֶׁלּוֹ עָבְדוּ כָּל כָּךְ קָשֶׁה – דָּאַג שֶׁהָאָזְנַיִם שֶׁלּוֹ יִשְׁמְעוּ תּוֹרָה, וְהַלֵּב שֶׁלּוֹ הָיָה נָקִי כְּמוֹ הַבְּגָדִים שֶׁהוּא כִּבֵּס!
