המעשה — הסיפור המלא

להלן מערך חינוכי, רגשי ומרתק על רבי יוסי בן חלפתא וסוד 300 ההלכות . הסיפור מעובד בקפידה עבור בנים בכיתות ב'-ד' (חינוך דתי-לאומי), תוך שימוש בעקרונות פדגוגיים, תיאורים חושיים, סטוריטלינג מותח, והקפדה גמורה על כבוד התורה וחכמיה.

הסיפור מנוקד במלואו כנדרש.

פְּתִיחָה: רַעַם קָדוֹשׁ בַּגָּלִיל

יְלָדִים, סִגְרוּ אֶת הָעֵינַיִם רֶגַע אֶחָד.

נַסּוּ לִדְמְיֵן שֶׁאַתֶּם עוֹמְדִים מוּל סֶלַע עֲנָק בְּאֶמְצַע הַיָּם, וְגַלִּים עֲצוּמִים שֶׁל מַיִם מִתְנַפְּצִים עָלָיו בְּרַעַשׁ אַדִּיר: טְרַאח! בּוּם! קוֹלוֹת חֲזָקִים שֶׁמַּרְעִידִים לָכֶם אֶת הַבֶּטֶן.

כָּךְ בְּדִיּוּק, וַאֲפִלּוּ יוֹתֵר מִזֶּה, נִשְׁמַע בֵּית הַמִּדְרָשׁ בָּעִיר צִפּוֹרִי שֶׁבַּגָּלִיל, לִפְנֵי קָרוֹב לְאַלְפַּיִם שָׁנָה.

אַךְ זֶה לֹא הָיָה רַעַשׁ שֶׁל סְעָרָה, זֶה הָיָה רַעַשׁ קָדוֹשׁ, הָרַעַשׁ הֲכִי יָפֶה בָּעוֹלָם – רַעַשׁ שֶׁל תּוֹרָה! עֲשָׂרוֹת תַּלְמִידֵי חֲכָמִים, תַּלְמִידָיו הַקְּדוֹשִׁים שֶׁל רַבִּי עֲקִיבָא, יָשְׁבוּ עַל סַפְסְלֵי עֵץ עָבִים. רֵיחַ נָעִים שֶׁל גְּוִילִים יְשָׁנִים שֶׁל סִפְרֵי קֹדֶשׁ וְשֶׁל שֶׁמֶן זַיִת טָהוֹר מִלֵּא אֶת חֲלַל הָאֲוִיר. קַרְנֵי שֶׁמֶשׁ זְהֻבּוֹת חָדְרוּ דֶּרֶךְ הַחַלּוֹנוֹת וְהֵאִירוּ אֶת גַּרְגְּרֵי הָאָבָק שֶׁרִחֲפוּ בָּאֲוִיר, וְאֶת הַפָּנִים הַזּוֹרְחוֹת שֶׁל הַלּוֹמְדִים.

הַתַּנָּאִים הַקְּדוֹשִׁים אָהֲבוּ אֶת הַתּוֹרָה יוֹתֵר מִכָּל דָּבָר אֲחֵר. כַּאֲשֶׁר דָּנוּ בִּשְׁאֵלָה הִלְכָתִית – לְמָשָׁל, הַאִם מַאֲכָל מְסֻיָּם מֻתָּר אוֹ אָסוּר לַאֲכִילָה בְּשַׁבָּת? – הֵם הָיוּ מִתְוַכְּחִים בְּלַהַט עָצוּם.

רַבִּי מֵאִיר – שֶׁהָיָה אָדָם שֶׁשִּׂכְלוֹ מָהִיר כְּמוֹ בָּרָק – הָיָה טוֹעֵן: "מֻתָּר! מֻתָּר גָּמוּר! וְיֵשׁ לִי עֶשְׂרִים רְאָיוֹת לְכָךְ!".

וּמִיָּד קָפַץ רַבִּי יְהוּדָה כְּנֶגְדּוֹ, בְּקוֹל רוֹעֵם וּבְעֵינַיִם בּוֹרְקוֹת: "מָה פִּתְאוֹם! זֶה אָסוּר! וַאֲנִי אֶסְתֹּר אֶת כָּל הָרְאָיוֹת שֶׁלְּךָ!".

זוֹ הָיְתָה מַחֲלֹקֶת לְשֵׁם שָׁמַיִם . הֵם לֹא רָבוּ חָלִילָה כְּדֵי שֶׁמִּישֶׁהוּ יַרְגִּישׁ חָזָק וְחָכָם, הֵם רָבוּ כִּי הֵם חִפְּשׂוּ אֶת הָאֱמֶת, רָצוּ לָדַעַת בְּדִיּוּק מָה הַקָּדוֹשׁ בָּרוּךְ הוּא צִוָּה בַּתּוֹרָה!

הֶחָכָם הַשּׁוֹתֵק בַּפִּנָּה

אֲבָל בְּתוֹךְ כָּל הָרַעַשׁ וְהַוִּכּוּחִים, אִם הֱיִיתֶם בָּאִים לַפִּנָּה וּמִסְתַּכְּלִים הֵיטֵב, הֱיִיתֶם רוֹאִים שָׁם חָכָם אֶחָד קָדוֹשׁ. שְׁמוֹ הָיָה רַבִּי יוֹסֵי בֶּן חֲלַפְתָּא .

בַּזְּמַן שֶׁכֻּלָּם הֵרִימוּ קוֹל וְהֵבִיאוּ הוֹכָחוֹת מְהִירוֹת, רַבִּי יוֹסֵי עָשָׂה מַשֶּׁהוּ אַחֵר לְגַמְרֵי.

רַבִּי יוֹסֵי הָיָה... שׁוֹתֵק.

הוּא הָיָה מַעֲצִים אֶת עֵינָיו בְּחָזְקָה. כַּפּוֹת יָדָיו חִבְּקוּ אֶת הַגְּמָרָא. הוּא הִקְשִׁיב בְּרִכּוּז עָצוּם לְכָל מִלָּה שֶׁאָמַר רַבִּי מֵאִיר, וְאָז הִקְשִׁיב לְכָל מִלָּה שֶׁאָמַר רַבִּי יְהוּדָה. הוּא שָׁמַע אֶת הַכֹּל, צְלִיל אַחַר צְלִיל.

רַבִּי יוֹסֵי יָדַע כִּי כְּדֵי לִמְצֹא אֶת הָאֱמֶת הַטְּהוֹרָה שֶׁל הַתּוֹרָה, לֹא צָרִיךְ לְמַהֵר. הוּא יָרַד בְּמַחֲשַׁבְתּוֹ עָמֹק-עָמֹק, לַשֹּׁרֶשׁ שֶׁל הַהֲלָכָה. הוּא שָׁקַל בְּדַעְתּוֹ: לָמָּה רַבִּי מֵאִיר אָמַר כָּךְ? הַאִם הַהוֹכָחָה שֶׁלּוֹ מֻשְׁלֶמֶת? הַאִם דִּבְרֵי רַבִּי יְהוּדָה יְשָׁרִים לַהֲלָכָה?

אֲנָשִׁים פְּשׁוּטִים שֶׁהָיוּ נִכְנָסִים, הָיוּ אולי חוֹשְׁבִים עָלָיו: "אוּלַי רַבִּי יוֹסֵי לֹא יוֹדֵעַ אֶת הַתְּשׁוּבָה? לָמָּה הוּא לֹא מְדַבֵּר?". אֲבָל תַּלְמִידֵי הַחֲכָמִים יָדְעוּ, שֶׁשְּׁתִיקָתוֹ שֶׁל רַבִּי יוֹסֵי עֲמֻקָּה וַחֲזָקָה יוֹתֵר מִמֵּאָה צְעָקוֹת.

הָרֶגַע שֶׁבּוֹ הַמַּיִם מַשְׁקִיטִים אֶת הָאֵשׁ

וְאָז, הִגִּיעַ הָרֶגַע הַגָּדוֹל.

רַבִּי יוֹסֵי הָיָה פּוֹקֵחַ אֶת עֵינָיו. קוֹמָתוֹ זָקְפָה, וּמַבָּטוֹ הָיָה שָׁלֵו כְּמַיִם שְׁקֵטִים מֵעַל אֲגַם רָגוּעַ. הוּא הֵרִים מְעַט אֶת אֶצְבָּעוֹ.

בְּאוֹתוֹ רֶגַע קָרָה קֶסֶם בְּבֵית הַמִּדְרָשׁ שֶׁל צִפּוֹרִי.

כָּל הַחֲכָמִים – כֻּלָּם! – הִשְׁתַּתְּקוּ בְּבַת אַחַת. רַבִּי מֵאִיר עָצַר אֶת דְּבָרָיו בְּאֶמְצַע מִשְׁפָּט. רַבִּי יְהוּדָה יָשַׁב בִּנְשִׁימָה עֲצוּרָה. דְּמָמָה דַּקָּה יָרְדָה עַל הָאוּלָם הֶעָנָק.

"מָה שְׁמוּעָה יֵשׁ בְּפִי רַבִּי יוֹסֵי?" לָחֲשׁוּ הַתַּלְמִידִים.

רַבִּי יוֹסֵי לֹא צָעַק. הוּא לֹא דָּפַק בְּזַעַם עַל הַשֻּׁלְחָן. בְּקוֹל שָׁקֵט, נָעִים וְרַךְ, הוּא חִלֵּק אֶת הַהֲלָכָה: "רַבּוֹתַי הַקְּדוֹשִׁים, מָה שֶׁאָמַר אָחִי, רַבִּי מֵאִיר, נָכוֹן הוּא לְמִקְרֶה אֶחָד. אַךְ בַּמִּקְרֶה שֶׁלָּנוּ – הַדִּין הוּא אָסוּר, וּמַדּוּעַ? כִּי יֵשׁ טַעַם עָמֹק בַּתּוֹרָה שֶׁאוֹמֵר כָּךְ וְכָךְ...".

הַמִּלִּים שֶׁלּוֹ הָיוּ מְסֻדָּרוֹת. הַהִגָּיוֹן שֶׁלּוֹ הָיָה בָּרוּר וְיָשָׁר כְּמוֹ חֵץ. אִי אֶפְשָׁר הָיָה לְהִתְוַכֵּחַ עִם הַסְּבָרָה הַטְּהוֹרָה וְהַחַדָּה שֶׁלּוֹ.

הַתַּנָּאִים הַגְּדוֹלִים, שֶׁלִּפְנֵי רֶגַע הִתְוַכְּחוּ בְּלַהַט מִי צוֹדֵק, רָאוּ אֶת הָאֱמֶת. הֵם לֹא הִתְבַּיְּשׁוּ לְהוֹדוֹת לְרַבִּי יוֹסֵי. הֵם רָאוּ שֶׁרַבִּי יוֹסֵי אוֹמֵר תּוֹרָה שֶׁיָּרְדָה מֵהַשָּׁמַיִם.

הַהַחְלָטָה שֶׁל רַבִּי יְהוּדָה הַנָּשִׂיא

שָׁנִים לְאַחַר מִכֵּן, קָם הַמַּנְהִיג הֶעָצוּם, רַבִּי יְהוּדָה הַנָּשִׂיא. הוּא זֶה שֶׁעָרַךְ אֶת הַ"מִּשְׁנָה" – הַסֵּפֶר שֶׁמְּאַגֵּד בְּתוֹכוֹ אֶת כָּל סוֹדוֹת הַתּוֹרָה שֶׁבְּעַל-פֶּה שֶׁעָבְרוּ מִדּוֹר לְדוֹר עַד מֹשֶׁה רַבֵּנוּ!

רַבִּי יְהוּדָה הַנָּשִׂיא קָבַע כְּלָל יְסוֹד עֲצּוּם וּמַרְעִישׁ, שֶׁנִּשְׁמַר עַד הַיּוֹם הַזֶּה: "בְּכָל מָקוֹם שֶׁרַבִּי יוֹסֵי בֶּן חֲלַפְתָּא מִתְוַכֵּחַ עִם אֶחָד מֵחֲבֵרָיו... הֲלָכָה כְּרַבִּי יוֹסֵי! ".

תָּבִינוּ, יְלָדִים, אֵיזֶה כָּבוֹד אָדִיר זֶה! הַקָּדוֹשׁ בָּרוּךְ הוּא צִוָּה עָלֵינוּ מִסִּינַי שֶׁנִּפְסֹק הֲלָכָה כְּמוֹ רַבִּי יוֹסֵי בֶּן חֲלַפְתָּא בְּכָל דָּבָר שֶׁהוּא אוֹמֵר!

וּמַדּוּעַ? הַגְּמָרָא הַקְּדוֹשָׁה מַסְבִּירָה בִּשְׁתֵּי מִלִּים סוֹדִיּוֹת שֶׁשָּׁווֹת הַכֹּל: "דְּנִימּוּקוֹ עִמּוֹ" (שֶׁהַטַּעַם שֶׁלּוֹ, הַסְּבָרָה שֶׁלּוֹ, אִתּוֹ). רַבִּי יוֹסֵי לְעוֹלָם לֹא זָרַק מִלִּים מֵהַשַּׁרְווּל בִּשְׁבִיל לְנַצֵּחַ; לְכָל הֲלָכָה שֶׁאָמַר, הָיָה לוֹ מָקוֹר, יְסוֹד וְטַעַם עָמֹק וְאָמִתִּי.

וּמָה הָיְתָה הַתּוֹצָאָה?

בַּמִּשְׁנָה הַקְּדוֹשָׁה שֶׁאֲנַחְנוּ לוֹמְדִים הַיּוֹם, יֵשְׁנָן לֹא פָּחוֹת מִ- 300 הֲלָכוֹת (שָׁלֹשׁ מֵאוֹת מִשְׁנָיוֹת!) שֶׁנִּפְסְקוּ לְמַעֲשֶׂה לְדוֹרוֹת עַל פִּי פְּסָק הַהֲלָכָה הַטָּהוֹר, הַשָּׁקֵט וְהֶחָכָם שֶׁל רַבִּי יוֹסֵי בֶּן חֲלַפְתָּא הֶעָנָיו מֵהָעִיר צִפּוֹרִי.

א. שְׁאֵלַת פְּתִיחָה (לִפְנֵי הַסִּפּוּר):

"יְלָדִים, שְׁאֵלָה שֶׁבּוֹחֶנֶת הִגָּיוֹן: כְּשֶׁיֵּשׁ וִכּוּחַ בֵּין שְׁנֵי יְלָדִים בַּכִּתָּה אוֹ בַּחָצֵר, מִי בְּדֶרֶךְ כְּלָל מְנַצֵּחַ בַּוִּכּוּחַ? מִי שֶׁצּוֹעֵק הֲכִי חָזָק? מִי שֶׁמְּדַבֵּר הֲכִי מַהֵר?
