כיבוד אמו — שני הסיפורים

דף עבודה בעקבות רבי טרפון
שם: כיתה:

המעשה — הסיפור המלא

פְּתִיחָה: הָעָשִׁיר הַגָּדוֹל מִלּוֹד

יְלָדִים יְקָרִים, כְּשֶׁאֲנַחְנוּ שׁוֹמְעִים עַל אָדָם שֶׁהוּא עָשִׁיר גָּדוֹל, כָּזֶה שֶׁיֵּשׁ לוֹ שָׂדוֹת, כְּרָמִים, מַטְבְּעוֹת זָהָב וְאַרְמוֹנוֹת... אֵיךְ אֲנַחְנוּ מְדַמְיְנִים אֶת הַחַיִּים שֶׁלּוֹ? בְּוַדַּאי יֵשׁ לוֹ עֲבָדִים שֶׁמַּגִּישִׁים לוֹ אֹכֶל, שׁוֹמְרִים שֶׁפּוֹתְחִים לוֹ אֶת הַדֶּלֶת, וְהוּא אֵינוֹ צָרִיךְ לַעֲשׂוֹת שׁוּם עֲבוֹדָה קָשָׁה, נָכוֹן?

וְאִם אוֹתוֹ אָדָם עָשִׁיר הוּא גַּם... אֶחָד מִגְּדוֹלֵי צַדִּיקֵי יִשְׂרָאֵל? תַּנָּא קָדוֹשׁ שֶׁמֵּאוֹת תַּלְמִידִים הוֹלְכִים אַחֲרָיו בְּיִרְאַת כָּבוֹד וְשׁוֹתִים בְּצִמָּאוֹן אֶת דִּבְרֵי תּוֹרָתוֹ?

כָּזֶה בְּדִיּוּק הָיָה הַתַּנָּא הַקָּדוֹשׁ, רַבִּי טַרְפוֹן . הוּא הָיָה מִגְּדוֹלֵי הַחֲכָמִים בָּעִיר לוֹד, עָשִׁיר מֻפְלָג, וְכֹהֵן מְיֻחָס שֶׁזָּכָה בְּיַלְדוּתוֹ לִרְאוֹת אֶת בֵּית הַמִּקְדָּשׁ בְּתִפְאַרְתּוֹ. כֻּלָּם כִּבְּדוּ אוֹתוֹ וְרָחֲשׁוּ לוֹ אַהֲבָה עֲצוּמָה. יָכוֹל הָיָה רַבִּי טַרְפוֹן לָשֶׁבֶת עַל כֻּרְסַת הַכָּבוֹד שֶׁלּוֹ וְלִתְתּ כָּל מְלָאכָה קָשָׁה לַמְשָׁרְתִים שֶׁלּוֹ.

אֲבָל כְּשֶׁהָיָה מְדֻבָּר בְּאִמּוֹ הַזְּקֵנָה – הַכֹּל הָיָה אַחֶרֶת לְגַמְרֵי.

מַדְרֵגָה מִבָּשָׂר וָדָם

אִמּוֹ שֶׁל רַבִּי טַרְפוֹן הָיְתָה אִשָּׁה יְקָרָה וְצַדֶּקֶת, אַךְ עִם הַשָּׁנִים גּוּפָהּ נֶחֱלַשׁ. הִיא חָלְתָה וּבְקֹשִׁי יָכְלָה לַעֲמֹד עַל רַגְלֶיהָ. בְּחַדְרָהּ, עָמְדָה מִטָּה גְּבוֹהָה עֲשׂוּיָה מֵעֵץ מְגֻלָּף – מִטָּה רַכָּה וּנְעִימָה. אַךְ בִּגְלַל חֻלְשָׁתָהּ, הִיא לֹא הִצְלִיחָה לְטַפֵּס וְלַעֲלוֹת לַמִּטָּה לְבַדָּהּ, וְגַם לֹא לָרֶדֶת מִמֶּנָּה.

יְלָדִים, אִם לְרַבִּי טַרְפוֹן הָיוּ עֲבָדִים וּשְׁפָחוֹת, לָמָּה לֹא קָרָא לָהֶם? "הָיּוּ בּוֹאוּ הֵנָּה! עִזְרוּ לְאִמִּי לַעֲלוֹת לַמִּטָּה!"... לֹא!

רַבִּי טַרְפוֹן יָדַע סוֹד גָּדוֹל: מִצְוַת כִּבּוּד אֵם אוֹהֶבֶת שֶׁעוֹשִׂים אוֹתָהּ בַּיָּדַיִם מַמָּשׁ!

בְּכָל עֶרֶב, כַּאֲשֶׁר אִמּוֹ רָצְתָה לִישֹׁן, רַבִּי טַרְפוֹן, עֲנַק הַתּוֹרָה בִּכְבוֹדוֹ וּבְעַצְמוֹ, עָזַב אֶת סְפָרָיו. הוּא נִכְנַס לְחַדְרָהּ בְּחִיּוּךְ מֵאִיר, הִתְכּוֹפֵף כְּלַפֵּי הָרִצְפָּה הַקָּרָה, הִנִּיחַ אֶת יָדָיו עַל הַקַּרְקַע עַד שֶׁגַּבּוֹ הָפַךְ שָׁטוּחַ מַמָּשׁ כְּמוֹ שְׁרַפְרַף, מַדְרֵגָה.

"עֲלִי, אִמִּי הַיְקָרָה," הָיָה לוֹחֵשׁ לָהּ בַּאֲהָבָה.

הַזְּקֵנָה הַחַלָּשָׁה הָיְתָה מַנִּיחָה אֶת רַגְלָהּ עַל גַּבּוֹ שֶׁל בְּנָהּ הַצַּדִּיק, דּוֹרֶכֶת עָלָיו בַּעֲדִינוּת, וּמְטַפֶּסֶת לַמִּטָּה.

וּבַבֹּקֶר? כְּשֶׁהַשֶּׁמֶשׁ יָצְאָה וְקָרְאָה לָעוֹלָם לָקוּם – מִי מִהֵר לְחַדְרָהּ? רַבִּי טַרְפוֹן! שׁוּב מִתְכּוֹפֵף, שׁוּב הוֹפֵךְ עַצְמוֹ לְמַדְרֵגָה, כְּדֵי שֶׁאִמּוֹ תֵּרֵד בְּשָׁלוֹם וּבְנַחַת לַקַּרְקַע. וְלֹא הָיָה מְאֻשָּׁר מִמֶּנּוּ בָּעוֹלָם! כִּי לְכַבֵּד אִמָּא זֶה הַזְּכוּת הֲכִי נִפְלָאָה שֶׁיֵּשׁ.

הַטִּיּוּל בַּחֲצֵר הָאֶבֶן בְּשַׁבָּת קֹדֶשׁ

עָבְרוּ הַיָּמִים, וְהִנֵּה הִגִּיעַ יוֹם שַׁבָּת קֹדֶשׁ. בְּתוֹךְ בֵּיתוֹ שֶׁל רַבִּי טַרְפוֹן הָיָה שֶׁקֶט קָדוֹשׁ. רֵיחוֹת מְתוּקִים שֶׁל תַּבְלִינִים, תַּבְשִׁילִים חַמִּים וַהֲדַסִּים רֵיחָנִיִּים מִלְּאוּ אֶת חֲלַל הַבַּיִת. בַּחוּץ, בֶּחָצֵר שֶׁל מִשְׁפַּחַת רַבִּי טַרְפוֹן, זָרְחָה שֶׁמֶשׁ חֲמִימָה. קַרְנֵי שֶׁמֶשׁ זְהוּבוֹת לִטְּפוּ אֶת רִצְפַת הָאֶבֶן שֶׁבֶּחָצֵר, וְצִיּוּץ הַצִּפּוֹרִים נִשְׁמַע מִבֵּין עֲנָפֵי עֲצֵי הַתְּאֵנָה.

אִמּוֹ הַחוֹלָה שֶׁל רַבִּי טַרְפוֹן הִרְגִּישָׁה מְעַט כֹּחַ הַיּוֹם. "בְּנִי הַיָּקָר," אָמְרָה לוֹ, "אֲנִי רוֹצָה לָרֶדֶת וְלִנְשֹׁם קְצָת אֲוִיר צַח וּנְקִי. אֲנִי רוֹצָה לְטַיֵּל מְעַט בֶּחָצֵר שֶׁלָּנוּ".

בְּשִׂמְחָה רַבָּה תָּמַךְ בָּהּ רַבִּי טַרְפוֹן, וְהִיא פָּסְעָה אַט-אַט אֶל הֶחָצֵר. אֶלָּא שֶׁהֶחָצֵר, יְלָדִים, לֹא הָיְתָה מְרֻצֶּפֶת בְּרַצְפּוֹת שַׁיִשׁ חֲלָקוֹת כְּמוֹ אֶצְלֵנוּ בַּסָּלוֹן. הֶחָצֵר הָיְתָה מְלֵאָה בַּאֲבָנִים קְטַנּוֹת, בְּרִגְבֵי אֲדָמָה, וְכָאן וָשָׁם בָּלְטוּ אֲבָנִים שֶׁהָיוּ מְעַט קָשׁוֹת וַחֲדָדוֹת.

הִיא הָלְכָה לְאַט, סַנְדָּלֶיהָ עַל רַגְלֶיהָ מְגִנִּים עָלֶיהָ מִפְּנֵי הָאֲבָנִים.

אַךְ לְפֶתַע – פְּרָאק! קְרַע!

נִשְׁמַע רַעַשׁ קָטָן. רְצוּעַת הָעוֹר בְּסַנְדָּלָהּ שֶׁל אִמּוֹ – נִקְרְעָה! הַסַּנְדָּל הֶחְלִיק וְנָפַל, וְרַגְלָהּ הַחַלָּשָׁה נִשְׁאֲרָה יְחֵפָה לְגַמְרֵי עַל אֲבָנֵי הֶחָצֵר הַקָּשׁוֹת.

אִמּוֹ שֶׁל רַבִּי טַרְפוֹן עָצְרָה בְּבֶהָלָה. מָה תַּעֲשֶׂה? לְהַמְשִׁיךְ לָלֶכֶת? הָאֲבָנִים כָּל כָּךְ שְׁפִּיצִיּוֹת וְדוֹקְרוֹת! לְתַקֵּן אֶת הַנַּעַל הַנִּקְרַעַת? אָסוּר! הַיּוֹם יוֹם שַׁבָּת קֹדֶשׁ, הֲרֵי זֶה חִלּוּל שַׁבָּת.

שְׁטִיחַ שֶׁל יָדַיִם

רַבִּי טַרְפוֹן קָלַט אֶת הַמַּצָּב בִּשְׁנִיָּה אַחַת. הַשֵּׁם יִתְבָּרַךְ מְזַמֵּן לוֹ כָּעֵת נִסָּיוֹן – הַאִם יַצְלִיחַ לְהַצִּיל אֶת אִמּוֹ מִכְּאֵב?

הוּא יָכוֹל הָיָה לְבַקֵּשׁ שֶׁיָּבִיאוּ לָהּ נַעַל אַחֶרֶת, אַךְ זֶה יִקַּח זְמַן, וְהִיא עוֹמֶדֶת כָּאן וּמִתְקַשָּׁה לִדְרֹךְ עַל הָרִצְפָּה!

בְּלִי לוֹמַר מִלָּה, רַבִּי טַרְפוֹן מִתְכּוֹפֵף.

יְלָדִים, תְּדַמְיְנוּ אֶת הַמַּחֲזֶה הַמַּדְהִים הַזֶּה: שַׂר הַתּוֹרָה, הָאָדָם שֶׁכָּל גְּדוֹלֵי דּוֹרוֹ חֲרֵדִים מִפָּנָיו, יוֹרֵד עַל בִּרְכָּיו עַל אֲדָמַת הֶחָצֵר. הוּא פּוֹרֵשׂ קָדִימָה אֶת שְׁתֵּי כַּפּוֹת יָדָיו הַחַמּוֹת... וּמַנִּיחַ אוֹתָן מַמָּשׁ עַל הָאֲבָנִים.

"אִמָּא," הוּא לוֹחֵשׁ, "הַנִּיחִי אֶת הָרֶגֶל הַיְּחֵפָה שֶׁלָּךְ עַל הַיָּד שֶׁלִּי."

אִמּוֹ נִשְׁעֲנָה וְהִנִּיחָה אֶת כַּף רַגְלָהּ עַל גַּב יָדוֹ שֶׁל רַבִּי טַרְפוֹן. לֹא הָאֲבָנִים הַחַדּוֹת דָּקְרוּ בָּהּ, אֶלָּא יָדָיו שֶׁל בְּנָהּ רִפְּדוּ אֶת דַּרְכָּהּ.

וְכָךְ זֶה נִמְשַׁךְ. הִיא עוֹשָׂה צַעַד – וְרַבִּי טַרְפוֹן שׁוֹלֵף אֶת הַיָּד הָאַחֲרוֹנָה וּמַנִּיחַ אוֹתָהּ קָדִימָה. צַעַד... וְשׁוּב יָד מִלְּפָנִים. צַעַד... וְעוֹד יָד מִלְּפָנִים.

הַיָּדַיִם שֶׁל רַבִּי טַרְפוֹן שְׂרוּטוֹת וְכוֹאֲבוֹת מֵהָאֲבָנִים הַקָּשׁוֹת, הוּא מִתְנַשֵּׁם בִּכְבֵדוּת, כָּל הַגַּב שֶׁלּוֹ כּוֹאֵב מֵהַהֲלִיכָה עַל הָרִצְפָּה, אֲבָל פָּנָיו זָרְחוּ מֵאוֹר שֶׁל קְדֻשָּׁה! הוּא לֹא נָתַן לְאִמּוֹ לִדְרֹךְ אֲפִלּוּ עַל עָפָר רַךְ. עַד שֶׁהִגִּיעוּ חֲזָרָה לְתוֹךְ הַבַּיִת, עַד לַמִּטָּה, רַגְלָהּ שֶׁל הָאֵם דָּרְכָה רַק עַל כַּפּוֹת יָדָיו שֶׁל בְּנָהּ הַגָּדוֹל.

הַהַפְתָּעָה שֶׁל חֲכָמִים

עָבַר זְמַן מָה, וּמִן הַשָּׁמַיִם רָצוּ לְלַמֵּד לָנוּ שִׁעוּר אֲמִתִּי מַהוּ כְּבוֹד אֵם בֶּאֱמֶת.

רַבִּי טַרְפוֹן חָלָה לְפֶתַע, וּמַחֲלָתוֹ הָיְתָה קָשָׁה מְאֹד. גְּדוֹלֵי וְחַכְמֵי יִשְׂרָאֵל, חֲבֵרָיו לַתּוֹרָה, מִהֲרוּ לָבוֹא וּלְבַקֵּר אוֹתוֹ (בִּקּוּר חוֹלִים), לִרְאוֹת בִּשְׁלוֹמוֹ שֶׁל שַׂר הַתּוֹרָה.

הֵם נִכְנְסוּ לַבַּיִת בְּצַעַר וּבִדְאָגָה עֲמֻקָּה.

אִמּוֹ הַזְּקֵנָה יָצְאָה לִקְרָאתָם, כְּשֶׁדְּמָעוֹת נוֹצְצוֹת בְּעֵינֶיהָ. "רַבּוֹתַי," הִיא תָּפְסָה בִּבְגָדָם שֶׁל הַחֲכָמִים הַקְּדוֹשִׁים, "אָנָּא! הִתְפַּלְּלוּ עַל בְּנִי, עַל רַבִּי טַרְפוֹן! שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא יְרַפֵּא אוֹתוֹ!".

וְאָז הוֹסִיפָה מִשְׁפָּט: "וְתַגִּידוּ לַה' בַּתְּפִלָּה שֶׁלָּכֶם, שֶׁמַּגִּיעַ לִבְנִי לִחְיוֹת, מִפְּנֵי שֶׁהוּא אוֹהֵב אוֹתִי כָּל כָּךְ! אֵין לָכֶם מֻשָּׂג כַּמָּה כָּבוֹד הוּא נוֹהֵג בִּי!"

שָׁאֲלוּ אוֹתָהּ הַחֲכָמִים בַּעֲדִינוּת: "אִמָּא צַדֶּקֶת, אֵיזֶה כָּבוֹד מְיֻחָד הוּא מְכַבֵּד אוֹתָךְ?"

וְהָאֵם הַזְּקֵנָה סִפְּרָה לָהֶם – בְּקוֹל רוֹעֵד מֵהִתְרַגְּשׁוּת – הֵן עַל מַדְרֵגַת הַגַּב בְּכָל עֶרֶב וָבֹקֶר, וְהֵן עַל הַשַּׁבָּת הַהִיא, עַל הַסַּנְדָּל הַקָּרוּעַ וְעַל כַּפּוֹת יָדָיו שֶׁל רַבִּי טַרְפוֹן שֶׁשִּׁמְּשׁוּ כְּרִצְפָּה חַמָּה לְרַגְלֶיהָ הַיְּחֵפוֹת. הִיא צִפְּתָה שֶׁהַחֲכָמִים יִתְפַּלְאוּ מְאֹד.

אֲבָל הַחֲכָמִים הִבִּיטוּ זֶה בָּזֶה, וְאָז אָמְרוּ מִשְׁפָּט מְזַעֲזֵעַ:

"אִמָּא הַיְקָרָה... דְּעִי לָךְ... גַּם אִם בְּנֵךְ כִּבֵּד אוֹתָךְ כָּךְ אֶלֶף וְאַלְפֵי פְּעָמִים... אֲפִלּוּ אָז – לַחֲצִי הַכָּבוֹד שֶׁבִּקְּשָׁה מֵאִתָּנוּ הַתּוֹרָה – עֲדַיִן לֹא הִגִּיעַ!".

מָה פֶּרֶשׁ?

הַחֲכָמִים לֹא זִלְזְלוּ, חָלִילָה! הֵם בָּאוּ לְלַמֵּד אוֹתָנוּ (וְאֶת הָאִמָּא), שֶׁמִּצְוַת "כַּבֵּד אֶת אָבִיךָ וְאֶת אִמֶּךָ" הִיא מִצְוָה כָּל כָּךְ גְּדוֹלָה, כָּל כָּךְ רְחָבָה וּקְדוֹשָׁה עַד הַשָּׁמַיִם, שֶׁאֲפִלּוּ מַעֲשִׂים אֲדִירִים כָּאֵלֶּה שֶׁל צַדִּיק כְּרַבִּי טַרְפוֹן – הֵם רַק הַהַתְחָלָה שֶׁל אֵיךְ שֶׁהַתּוֹרָה רוֹצָה שֶׁנְּכַבֵּד אֶת הַהוֹרִים שֶׁהֵבִיאוּ אוֹתָנוּ לָעוֹלָם! וּבִזְכוּת הַהֲבָנָה הַזּוֹ וְהַמַּעֲשִׂים הַקְּדוֹשִׁים הָאֵלֶּה, זָכָה רַבִּי טַרְפוֹן לְהִתְרַפֵּא וּלְהַאֲרִיךְ יָמִים, וְלִהְיוֹת סֵמֶל וּמוֹפֵת לְעַם יִשְׂרָאֵל כֻּלּוֹ.

מקורות: תלמוד בבלי, מסכת קידושין, דף ל"א, עמוד ב' (סיפור הסנדל שנקרע בחצר וידיו תחת רגליה). תלמוד ירושלמי, מסכת פאה, פרק א', הלכה א' (סיפור הירידה על גבו כהדום, וכן אזכור כיבוד אמו). פסיקתא רבתי, פיסקא כ"ג (תגובת החכמים בעת חוליו: "לא הגיע לחצי הכבוד").

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “יְלָדִים יְקָרִים, כְּשֶׁאֲנַחְנוּ שׁוֹמְעִים עַל אָדָם שֶׁהוּא עָשִׁיר גָּדוֹל, כָּזֶה שֶׁיֵּשׁ לוֹ שָׂדוֹת, כְּרָמִים, מַטְבְּעוֹת זָהָב וְאַרְמוֹנוֹת...” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “רַבִּי טַרְפוֹן יָדַע סוֹד גָּדוֹל: מִצְוַת כִּבּוּד אֵם אוֹהֶבֶת שֶׁעוֹשִׂים אוֹתָהּ בַּיָּדַיִם מַמָּשׁ!”?
ג. כיצד הגיב רבי טרפון למצב המתואר: “בַּחוּץ, בֶּחָצֵר שֶׁל מִשְׁפַּחַת רַבִּי טַרְפוֹן, זָרְחָה שֶׁמֶשׁ חֲמִימָה.”?
ד. מה השתנה בין “הַסַּנְדָּל הֶחְלִיק וְנָפַל, וְרַגְלָהּ הַחַלָּשָׁה נִשְׁאֲרָה יְחֵפָה לְגַמְרֵי עַל אֲבָנֵי הֶחָצֵר הַקָּשׁוֹת.” לבין “הִיא עוֹשָׂה צַעַד – וְרַבִּי טַרְפוֹן שׁוֹלֵף אֶת הַיָּד הָאַחֲרוֹנָה וּמַנִּיחַ אוֹתָהּ קָדִימָה.”?
ה. כיצד הגיע הסיפור אל האירוע: “וְאָז הוֹסִיפָה מִשְׁפָּט: "וְתַגִּידוּ לַה' בַּתְּפִלָּה שֶׁלָּכֶם, שֶׁמַּגִּיעַ לִבְנִי לִחְיוֹת, מִפְּנֵי שֶׁהוּא אוֹהֵב אוֹתִי כָּל כָּךְ!”?

□ הֶחָצֵר הָיְתָה מְלֵאָה בַּאֲבָנִים קְטַנּוֹת, בְּרִגְבֵי אֲדָמָה, וְכָאן וָשָׁם בָּלְטוּ אֲבָנִים שֶׁהָיוּ מְעַט קָשׁוֹת וַחֲדָדוֹת.

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “כיבוד אמו — שני הסיפורים” על רבי טרפון? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “בְּשִׂמְחָה רַבָּה תָּמַךְ בָּהּ רַבִּי טַרְפוֹן, וְהִיא פָּסְעָה אַט-אַט אֶל הֶחָצֵר.”.
ג. אילו רבי טרפון לא היה פועל כפי שמתואר ב־“הִיא עוֹשָׂה צַעַד – וְרַבִּי טַרְפוֹן שׁוֹלֵף אֶת הַיָּד הָאַחֲרוֹנָה וּמַנִּיחַ אוֹתָהּ קָדִימָה.” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “וְאָז הוֹסִיפָה מִשְׁפָּט: "וְתַגִּידוּ לַה' בַּתְּפִלָּה שֶׁלָּכֶם, שֶׁמַּגִּיעַ לִבְנִי לִחְיוֹת, מִפְּנֵי שֶׁהוּא אוֹהֵב…” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. יְלָדִים יְקָרִים, כְּשֶׁאֲנַחְנוּ שׁוֹמְעִים עַל אָדָם שֶׁהוּא עָשִׁיר…
2. בְּכָל עֶרֶב, כַּאֲשֶׁר אִמּוֹ רָצְתָה לִישֹׁן, רַבִּי טַרְפוֹן, עֲנַק…
3. אִמּוֹ הַחוֹלָה שֶׁל רַבִּי טַרְפוֹן הִרְגִּישָׁה מְעַט כֹּחַ הַיּוֹם.
4. הוּא יָכוֹל הָיָה לְבַקֵּשׁ שֶׁיָּבִיאוּ לָהּ נַעַל אַחֶרֶת, אַךְ זֶה…
5. עַד שֶׁהִגִּיעוּ חֲזָרָה לְתוֹךְ הַבַּיִת, עַד לַמִּטָּה, רַגְלָהּ שֶׁל…
6. וּבִזְכוּת הַהֲבָנָה הַזּוֹ וְהַמַּעֲשִׂים הַקְּדוֹשִׁים הָאֵלֶּה,…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “בַּחוּץ, בֶּחָצֵר שֶׁל מִשְׁפַּחַת רַבִּי טַרְפוֹן, זָרְחָה שֶׁמֶשׁ חֲמִימָה.”.
מכתב: כתבו לרבי טרפון בעקבות “הוּא פּוֹרֵשׂ קָדִימָה אֶת שְׁתֵּי כַּפּוֹת יָדָיו הַחַמּוֹת...”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “עָבַר זְמַן מָה, וּמִן הַשָּׁמַיִם רָצוּ לְלַמֵּד לָנוּ שִׁעוּר אֲמִתִּי מַהוּ כְּבוֹד אֵם בֶּאֱמֶת.”.
דיווח: כתבו ידיעה קצרה שכותרתה “כיבוד אמו — שני הסיפורים”.

המחזה

רגע מן הסיפורהדמויותמה יאמרו או יעשו?
שׁוּב מִתְכּוֹפֵף, שׁוּב הוֹפֵךְ עַצְמוֹ לְמַדְרֵגָה, כְּדֵי שֶׁאִמּוֹ תֵּרֵד בְּשָׁלוֹם וּבְנַחַת…
הִיא עוֹשָׂה צַעַד – וְרַבִּי טַרְפוֹן שׁוֹלֵף אֶת הַיָּד הָאַחֲרוֹנָה וּמַנִּיחַ אוֹתָהּ קָדִימָה.

7. חיבור לחיים מתוך המעשה

א. מה אפשר ללמוד מן הדרך שבה רבי טרפון פעל באירוע “הַסַּנְדָּל הֶחְלִיק וְנָפַל, וְרַגְלָהּ הַחַלָּשָׁה נִשְׁאֲרָה יְחֵפָה לְגַמְרֵי עַל אֲבָנֵי הֶחָצֵר הַקָּשׁוֹת.”?
ב. תארו מצב מחיי הכיתה שבו אפשר ליישם את המסר של “כיבוד אמו — שני הסיפורים”.
