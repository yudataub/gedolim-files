עירו בבית שערים ועבר לציפורי עקב חוליו

דף עבודה בעקבות רבי יהודה הנשיא (רבי)
שם: כיתה:

המעשה — הסיפור המלא

להלן מערך חינוכי, רגשי ומרתק במיוחד, הבנוי על פי עקרונות הסטוריטלינג שביקשת. הסיפור עוסק ב רבי יהודה הנשיא (רבנו הקדוש) , וממוקד בגבורתו העצומה, עריכת המשנה, ומעברו מבית שערים לציפורי בעקבות חוליו, כדי להמשיך להנהיג את ישראל למשך 17 שנותיו האחרונות.

הסיפור מותאם לבנים בכיתות ב'-ד', בחינוך הדתי-לאומי, ומנוקד במלואו.

פְּתִיחָה: רַעַשׁ קְדוֹשׁ בְּבֵית שְׁעָרִים

יְלָדִים, צִמְצוּ אֶת הָעֵינַיִם לְרֶגַע וְנַסּוּ לְהַקְשִׁיב. מָה אַתֶּם שׁוֹמְעִים? אִם הָיִינוּ עוֹמְדִים עַכְשָׁו בָּעִיר הַיְּהוּדִית הַגְּדוֹלָה 'בֵּית שְׁעָרִים' לִפְנֵי קְרוֹב לְאַלְפַּיִם שָׁנָה, לֹא הָיִינוּ שׁוֹמְעִים מְכוֹנִיּוֹת אוֹ טֵלֵפוֹנִים מְצַלְצְלִים. הָיִינוּ שׁוֹמְעִים קוֹל אַדִּיר, קוֹל שֶׁהִרְעִיד אֶת קִירוֹת הָאֶבֶן שֶׁל הַבָּתִּים: קוֹל שֶׁל תּוֹרָה!

"אָמַר רַבִּי מֵאִיר... אָמַר רַבִּי יְהוּדָה... חַיָּב! פָּטוּר! מֻתָּר! אָסוּר!".

מֵאוֹת תַּלְמִידֵי חֲכָמִים יָשְׁבוּ עַל סַפְסַלֵּי עֵץ, מִתְנַדְנְדִים הָלוֹךְ וָשׁוֹב, כְּשֶׁרֵיחַ נֵרוֹת הַשֶּׁמֶן הַדּוֹלְקִים מְמַלֵּא אֶת הַחֶדֶר הַגָּדוֹל.

וּבַמֶּרְכָּז, יוֹשֵׁב רַבֵּנוּ הַקָּדוֹשׁ – רַבִּי יְהוּדָה הַנָּשִׂיא. אִישׁ שֶׁפָּנָיו מְאִירוֹת כְּמוֹ מַלְאָךְ, לָבוּשׁ בִּבְגָדִים נְקִיִּים וּמְכֻבָּדִים. מְלָכִים מֵרַחֲבֵי הָעוֹלָם הָיוּ בָּאִים לְהִתְיָעֵץ אִתּוֹ, אֲבָל אוֹתוֹ עִנְיֵן רַק דָּבָר אֶחָד: הַתּוֹרָה הַקְּדוֹשָׁה.

רַבִּי יְהוּדָה יָדַע שֶׁאִם לֹא יִכְתֹּב עַכְשָׁו אֶת הַהֲלָכוֹת, הַיְהוּדִים בַּגָּלוּת עֲלוּלִים, חָלִילָה, לִשְׁכֹּחַ אֶת הַתּוֹרָה שֶׁבְּעַל-פֶּה. עַל שֻׁלְחָנוֹ הָיוּ מֻנָּחוֹת מְגִלּוֹת קְלָף עָבוֹת. אֶצְבְּעוֹתָיו הָיוּ מְלֵאוֹת בִּכְתָמִים שֶׁל דְּיוֹ שָׁחוֹר. שָׁנִים אֲרֻכּוֹת הוּא עָבַד, לִקֵּט, מִיֵּן וְסִדֵּר אֶת "שִׁשָּׁה סִדְרֵי מִשְׁנָה".

הַיִּסּוּרִים בָּאִים

אֲבָל אָז, לְפֶתַע, מַשֶּׁהוּ קָרָה.

יוֹם אֶחָד, הַתַּלְמִידִים שָׂמוּ לֵב שֶׁהַקּוֹל הַצָּלוּל שֶׁל רַבִּי נִהְיָה חַלָּשׁ יוֹתֵר. הוּא עָצַר אֶת הַשִּׁעוּר שׁוּב וָשׁוּב בִּגְלַל שִׁעוּל כָּבֵד וְכוֹאֵב שֶׁטִּלְטֵל אֶת גּוּפוֹ הַקָּדוֹשׁ. רַבִּי סָבַל מִיִּסּוּרִים נוֹרָאִים – מַחֲלוֹת שִׁנַּיִם וּמַחֲלוֹת מֵעַיִם – אֲבָל הוּא קִבֵּל הַכֹּל בְּאַהֲבָה. מֵעוֹלָם לֹא הִתְלוֹנֵן.

בַּלַּיְלָה, כְּשֶׁכֻּלָּם יָשְׁנוּ, הָיוּ תַּלְמִידָיו שׁוֹמְעִים מִבַּעַד לַחַלּוֹן שִׁעוּל דַּק, וּמִיָּד אַחֲרָיו אֶת קוֹל הַלִּמּוּד נִמְשָׁךְ בִּמְסִירוּת. אַךְ הַמַּחֲלָה הָלְכָה וְהֶחְמִירָה. פָנָיו הַזּוֹרְחוֹת חָוְרוּ. הָאֲוִיר בְּבֵית שְׁעָרִים, שֶׁהָיְתָה עִיר עֲמוּסָה בַּעֲמָקִים, הָיָה לַח וְכָבֵד, וְזֶה רַק הִקְשָׁה עַל רַבִּי לִנְשֹׁם.

רוֹפְאִים גְּדוֹלִים מִכָּל הָאָרֶץ הֻזְעֲקוּ לְבֵיתוֹ. הֵם מָשְׁשׁוּ אֶת מִצְחוֹ הַלּוֹהֵט, רָקְחוּ (הֵכִינוּ) לְמַעֲנוֹ סַמְּמָנִים וְעֵשֶׂב הַמַּר וְטָעֲמוּ אֶת טַעֲמָם הֶחָרִיף לְפָנָיו, אַךְ כְּלוּם לֹא עָזַר.

בַּסּוֹף, קָם גְּדוֹל הָרוֹפְאִים וְאָמַר לַתַּלְמִידִים בְּקוֹל שָׁקֵט: "הָאֲוִיר שֶׁל בֵּית שְׁעָרִים רַע לָרַב. כְּדֵי לְהַצִּיל אֶת חַיָּיו שֶׁל נְשִׂיא יִשְׂרָאֵל, הוּא חַיָּב אֲוִיר הָרִים צָלוּל, נָקִי וְקָרִיר. הוּא חַיָּב לַעֲבֹר מִכָּאן אֶל הָעִיר צִפּוֹרִי הַנִּמְצֵאת עַל הָהָר".

הַפְּרֵדָה הַקָּשָׁה מִבֵּית שְׁעָרִים

רַבִּי שָׁמַע אֶת הַהַחְלָטָה. הַאִם עַכְשָׁו, בְּגִילוֹ הַמְּבֻגָּר, כְּשֶׁהוּא כָּל כָּךְ חַלָּשׁ, יַעֲזֹב אֶת מְקוֹם מוֹשָׁבוֹ? אֶת בֵּית הַמִּדְרָשׁ הָאָהוּב שֶׁלּוֹ? אֶת הַמָּקוֹם שֶׁבּוֹ לָמַד תּוֹרָה עִם טוֹבֵי הַתַּנָּאִים?

אֲבָל רַבִּי יָדַע: הַחַיִּים נִתְּנוּ לָנוּ כְּדֵי לִלְמֹד תּוֹרָה. אִם כְּדֵי לִגְמֹר לִכְתֹּב אֶת הַמִּשְׁנָה צָרִיךְ לָקוּם וְלָלֶכֶת – קָמִים וְהוֹלְכִים!

עֲגָלָה גְּדוֹלָה מְרֻפֶּדֶת בְּכָרִיּוֹת רַכּוֹת הוּכְנָה לַנָּשִׂיא.

קוֹל חֲרִיקַת הַגַּלְגַּלִּים נִשְׁמַע בָּרְחוֹב. כָּל תּוֹשָׁבֵי בֵּית שְׁעָרִים יָצְאוּ וּבָכוּ: "אַל תַּעֲזֹב אוֹתָנוּ, רַבֵּנוּ!". אַךְ הָעֲגָלָה הִתְחִילָה לָנוּעַ בִּדְרָכֵי הֶעָפָר.

הַמַּסָּע הָיָה קָשֶׁה. כָּל אֶבֶן וּקְפִיצָה בַּדֶּרֶךְ הִכְאִיבוּ לְגוּפוֹ הֶחָלוּשׁ שֶׁל הַנָּשִׂיא. אֲבָל רַבִּי סָגַר אֶת עֵינָיו, וּשְׂפָתָיו כְּמוֹ רָעֲדוּ הַלּוֹךְ וָשׁוֹב – הוּא חָזַר בְּרֹאשׁוֹ עַל מַסֶּכֶת שַׁבָּת, עַל מַסֶּכֶת סַנְהֶדְרִין, תּוֹרָה וְעוֹד תּוֹרָה. הַתּוֹרָה הִיא תְּרוּפָתוֹ הָאֲמִתִּית!

הָאֲוִיר שֶׁל צִפּוֹרִי – הַשְׁגָּחָה פְּרָטִית

לְאַט לְאַט הָעֲגָלָה טִפְּסָה עַל הֶהָרִים. וְאָז, פָתַח רַבִּי אֶת עֵינָיו.

מֵחַלּוֹן הָעֲגָלָה הָיָה אֶפְשָׁר לִרְאוֹת נוֹף הָרִים יָרֹק וּמַדְהִים. רֵיחַ חָדָשׁ וּנְקִי עָמַד בָּאֲוִיר – רֵיחַ שֶׁל עֲצֵי אוֹרֶן מְרַעֲנְנִים. רוּחַ צְנוֹנָה וּנְעִימָה לִטְּפָה אֶת פָּנָיו הַלּוֹהֲטוֹת. רַבִּי הַנָּשִׂיא שָׁאַף נְשִׁימָה אֲרֻכָּה... עֲמֻקָּה.

לְפֶתַע, צֶבַע שֶׁל חַיִּים חָזַר לְלֶחָיָיו. הַשִּׁעוּל הָיָה רָגוּעַ יוֹתֵר. מַיִם זַכִּים וְצוֹנְנִים מִמַּעְיְנוֹת הָעִיר צִפּוֹרִי הֻגְּשׁוּ לוֹ, וְטַעֲמָם הָיָה טַעַם שֶׁל חַיִּים חֲדָשִׁים. אֲוִיר הֶהָרִים הָרְפוּאִי עָשָׂה אֶת שֶׁלּוֹ.

בָּרֶגַע הַזֶּה הֵבִינוּ הַתַּלְמִידִים: אֵין מִקְרֶה בָּעוֹלָם! הַקָּדוֹשׁ בָּרוּךְ הוּא אָהַב אֶת רַבִּי כָּל כָּךְ, עַד שֶׁבָּרָא אֶת הָעִיר צִפּוֹרִי בִּגְבָהִים נְקִיִּים בְּדִיּוּק בִּשְׁבִילוֹ, בִּשְׁבִיל שֶׁיּוּכַל לְהַמְשִׁיךְ אֶת מְלֶאכֶת הַקֹּדֶשׁ שֶׁל הַמִּשְׁנָה!

שְׁבַע עֶשְׂרֵה שָׁנִים שֶׁל הַצָּלַת דּוֹרוֹת

בְּצִפּוֹרִי נִפְתַּח מֵחָדָשׁ הַיְשִׁיבָה הַגְּדוֹלָה שֶׁל רַבִּי. הַתַּלְמִידִים עָלוּ וּבָאוּ לָהָר, וְקוֹל הַתּוֹרָה הִדְהֵד בְּעָצְמָה.

יְלָדִים, אַתֶּם יוֹדְעִים כַּמָּה שָׁנִים חַי רַבִּי מֵהָרֶגַע שֶׁהִגִּיעַ לְצִפּוֹרִי? בְּדִיּוּק – שְׁבַע עֶשְׂרֵה שָׁנִים!

מִי מִגְּדוֹלֵי הָאֻמָּה חַי בִּמְדִינָה אַחֶרֶת גַּם כֵּן שְׁבַע עֶשְׂרֵה שָׁנִים? יַעֲקֹב אָבִינוּ! הַתּוֹרָה מְסַפֶּרֶת "וַיְחִי יַעֲקֹב בְּאֶרֶץ מִצְרַיִם שְׁבַע עֶשְׂרֵה שָׁנָה". אוֹתָן שָׁנִים שֶׁיַּעֲקֹב חַי בְּמִצְרַיִם הָיוּ הַשָּׁנִים הֲכִי טוֹבוֹת וּשְׁלֵווֹת שֶׁלּוֹ בְּחֵיק מִשְׁפַּחְתּוֹ, לְאַחַר כָּל צָרוֹתָיו.

כָּךְ הָיָה גַּם אֵצֶל רַבֵּנוּ הַקָּדוֹשׁ. שְׁבַע עֶשְׂרֵה שְׁנוֹתָיו בְּצִפּוֹרִי הָיוּ שְׁנוֹת תִּפְאֶרֶת. שָׁם, בַּאֲוִיר הֶהָרִים הַצָּלוּל, הֻשְׁלְמָה הַמִּשְׁנָה הַקְּדוֹשָׁה שֶׁאָנוּ לוֹמְדִים הַיּוֹם. אִלְמָלֵא הַנְּסִיעָה הַקָּשָׁה הַזּוֹ, וְאִלְמָלֵא הַהַשְׁגָּחָה הַפְּרָטִית שֶׁהֵבִיאָה אוֹתוֹ לְצִפּוֹרִי, מִי יוֹדֵעַ מָה הָיָה עָלֵינוּ וְעַל הַתּוֹרָה שֶׁלָּנוּ!

א. שְׁאֵלַת פְּתִיחָה (לְעוֹרֵר זִהוּי סִקְרָנוּת):

"יְלָדִים, מִי מִכֶּם פַּעַם עָבַר דִּירָה לְעִיר חֲדָשָׁה? זֶה קָשֶׁה לַעֲזֹב אֶת הַחֲבֵרִים וְהַבַּיִת הַמֻּכָּר, נָכוֹן? הַיּוֹם נִשְׁמַע עַל נְשִׂיא עַם יִשְׂרָאֵל, שֶׁהָיָה חַיָּב לַעֲזֹב אֶת בֵּיתוֹ וְאֶת הַמָּקוֹם שֶׁלּוֹ כְּדֵי לְהַצִּיל אֶת הַתּוֹרָה כֻּלָּהּ."

ב. הֶעָרוֹת בִּצּוּעַ וְסְטוֹרִיטֶלִינְג לַמְסַפֵּר:

ג. הַמְלָצוֹת לְשִׁלּוּב בְּמַצֶּגֶת (4 שְׁקוּפִיּוֹת):

ד. פְּעִילוּת מְסַכֶּמֶת / יְצִירָה:

מקורות: יְרוּשַׁלְמִי תַּלְמוּד (מַסֶּכֶת כִּלְאַיִם, פֶּרֶק ט, הֲלָכָה ג): עַל מַחֲלָתוֹ שֶׁל רַבִּי בְּבֵית שְׁעָרִים וַעֲצַת הָרוֹפְאִים לְהַעֲבִירוֹ לְצִפּוֹרִי בִּגְלַל אֲוִיר הֶהָרִים הַזַּךְ, וּמְגִיקָתוֹ לְשָׁם (מקור המאשר שחי שם 17 שנה עקב חוליו). תַּלְמוּד בַּבְלִי (מַסֶּכֶת כְּתֻבּוֹת, דַּף קג, עַמּוּד ב): עַל כָּךְ שֶׁרַבִּי חַי בְּצִפּוֹרִי שְׁבַע-עֶשְׂרֵה שָׁנָה, בְּדִיּוּק כְּפִי שֶׁיַּעֲקֹב אָבִינוּ חַי בְּמִצְרַיִם, וַעֲלֵיהֶן נֶאֱמַר הַפָּסוּק "וַיְחִי יַעֲקֹב בְּאֶרֶץ מִצְרַיִם שְׁבַע עֶשְׂרֵה שָׁנָה". תַּלְמוּד בַּבְלִי (סַנְהֶדְרִין, לב, ב): "צֶדֶק צֶדֶק תִּרְדֹּף - הֲלֹךְ אַחַר הֲלֹךְ רַבִּי לְבֵית שְׁעָרִים, אַחַר רַבִּי לְצִפּוֹרִי" - כָּךְ נָדְדָה הַסַּנְהֶדְרִין.

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “להלן מערך חינוכי, רגשי ומרתק במיוחד, הבנוי על פי עקרונות הסטוריטלינג שביקשת.” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “וּבַמֶּרְכָּז, יוֹשֵׁב רַבֵּנוּ הַקָּדוֹשׁ – רַבִּי יְהוּדָה הַנָּשִׂיא.”?
ג. כיצד הגיב רבי יהודה הנשיא (רבי) למצב המתואר: “הוּא עָצַר אֶת הַשִּׁעוּר שׁוּב וָשׁוּב בִּגְלַל שִׁעוּל כָּבֵד וְכוֹאֵב שֶׁטִּלְטֵל אֶת גּוּפוֹ הַקָּדוֹשׁ.”?
ד. מה השתנה בין “הוּא חַיָּב לַעֲבֹר מִכָּאן אֶל הָעִיר צִפּוֹרִי הַנִּמְצֵאת עַל הָהָר".” לבין “מֵחַלּוֹן הָעֲגָלָה הָיָה אֶפְשָׁר לִרְאוֹת נוֹף הָרִים יָרֹק וּמַדְהִים.”?
ה. כיצד הגיע הסיפור אל האירוע: “אוֹתָן שָׁנִים שֶׁיַּעֲקֹב חַי בְּמִצְרַיִם הָיוּ הַשָּׁנִים הֲכִי טוֹבוֹת וּשְׁלֵווֹת שֶׁלּוֹ בְּחֵיק מִשְׁפַּחְתּוֹ, לְאַחַר כָּל צָרוֹתָיו.”?

לפני ואחרי

מה קרה לפני: “בַּסּוֹף, קָם גְּדוֹל הָרוֹפְאִים וְאָמַר לַתַּלְמִידִים בְּקוֹל שָׁקֵט: "הָאֲוִיר שֶׁל בֵּית שְׁעָרִים רַע לָרַב.”?
מה קרה אחרי: “רֵיחַ חָדָשׁ וּנְקִי עָמַד בָּאֲוִיר – רֵיחַ שֶׁל עֲצֵי אוֹרֶן מְרַעֲנְנִים.”?

סיבה ותוצאה מתוך המעשה

האירוע הנתוןמה גרם לו?מה הייתה תוצאתו?
רַבִּי יְהוּדָה יָדַע שֶׁאִם לֹא יִכְתֹּב עַכְשָׁו אֶת הַהֲלָכוֹת, הַיְהוּדִים בַּגָּלוּת עֲלוּלִים, חָלִילָה, לִשְׁכֹּחַ אֶת הַתּוֹרָה שֶׁבְּעַל-פֶּה.
כָּל תּוֹשָׁבֵי בֵּית שְׁעָרִים יָצְאוּ וּבָכוּ: "אַל תַּעֲזֹב אוֹתָנוּ, רַבֵּנוּ!".

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “עירו בבית שערים ועבר לציפורי עקב חוליו” על רבי יהודה הנשיא (רבי)? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “הָאֲוִיר בְּבֵית שְׁעָרִים, שֶׁהָיְתָה עִיר עֲמוּסָה בַּעֲמָקִים, הָיָה לַח וְכָבֵד, וְזֶה רַק הִקְשָׁה עַל רַבִּי לִנְשֹׁם.”.
ג. אילו רבי יהודה הנשיא (רבי) לא היה פועל כפי שמתואר ב־“מֵחַלּוֹן הָעֲגָלָה הָיָה אֶפְשָׁר לִרְאוֹת נוֹף הָרִים יָרֹק וּמַדְהִים.” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “אוֹתָן שָׁנִים שֶׁיַּעֲקֹב חַי בְּמִצְרַיִם הָיוּ הַשָּׁנִים הֲכִי טוֹבוֹת וּשְׁלֵווֹת שֶׁלּוֹ בְּחֵיק מִשְׁפַּחְתּוֹ, לְאַחַר…” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. להלן מערך חינוכי, רגשי ומרתק במיוחד, הבנוי על פי עקרונות הסטוריטלינג…
2. אִישׁ שֶׁפָּנָיו מְאִירוֹת כְּמוֹ מַלְאָךְ, לָבוּשׁ בִּבְגָדִים…
3. בַּלַּיְלָה, כְּשֶׁכֻּלָּם יָשְׁנוּ, הָיוּ תַּלְמִידָיו שׁוֹמְעִים…
4. אֲבָל רַבִּי יָדַע: הַחַיִּים נִתְּנוּ לָנוּ כְּדֵי לִלְמֹד תּוֹרָה.
5. מַיִם זַכִּים וְצוֹנְנִים מִמַּעְיְנוֹת הָעִיר צִפּוֹרִי הֻגְּשׁוּ לוֹ,…
6. הַיּוֹם נִשְׁמַע עַל נְשִׂיא עַם יִשְׂרָאֵל, שֶׁהָיָה חַיָּב לַעֲזֹב…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “הוּא עָצַר אֶת הַשִּׁעוּר שׁוּב וָשׁוּב בִּגְלַל שִׁעוּל כָּבֵד וְכוֹאֵב שֶׁטִּלְטֵל אֶת גּוּפוֹ הַקָּדוֹשׁ.”.
מכתב: כתבו לרבי יהודה הנשיא (רבי) בעקבות “כָּל תּוֹשָׁבֵי בֵּית שְׁעָרִים יָצְאוּ וּבָכוּ: "אַל תַּעֲזֹב אוֹתָנוּ, רַבֵּנוּ!".”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “הַקָּדוֹשׁ בָּרוּךְ הוּא אָהַב אֶת רַבִּי כָּל כָּךְ, עַד שֶׁבָּרָא אֶת הָעִיר צִפּוֹרִי בִּגְבָהִים נְקִיִּים בְּדִיּוּק…”.
דיווח: כתבו ידיעה קצרה שכותרתה “עירו בבית שערים ועבר לציפורי עקב חוליו”.

המחזה

רגע מן הסיפורהדמויותמה יאמרו או יעשו?
רַבִּי יְהוּדָה יָדַע שֶׁאִם לֹא יִכְתֹּב עַכְשָׁו אֶת הַהֲלָכוֹת, הַיְהוּדִים בַּגָּלוּת…
מֵחַלּוֹן הָעֲגָלָה הָיָה אֶפְשָׁר לִרְאוֹת נוֹף הָרִים יָרֹק וּמַדְהִים.
