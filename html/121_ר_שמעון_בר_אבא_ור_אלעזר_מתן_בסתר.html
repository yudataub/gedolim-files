מתן בסתר

דף עבודה בעקבות ר' שמעון בר אבא ור' אלעזר
שם: כיתה:

המעשה — הסיפור המלא

פְּתִיחָה: רֵיחַ שֶׁל קְדֻשָּׁה וְטַעַם שֶׁל רָעָב

יְלָדִים, עִצְמוּ אֶת הָעֵינַיִם וְנַסּוּ לְדַמְיֵן אֶת בֵּית הַמִּדְרָשׁ הַגָּדוֹל שֶׁל אֶרֶץ יִשְׂרָאֵל לִפְנֵי כְּאַלְפַּיִם שָׁנָה. אֲוִיר קָרִיר נָשַׁב מִן הַחַלּוֹנוֹת, אַךְ בִּפְנִים – הַכֹּל בָּעַר כְּמוֹ אֵשׁ! מֵאוֹת תַּלְמִידֵי חֲכָמִים יָשְׁבוּ עַל סַפְסַלֵּי הָעֵץ הַיְּשָׁנִים, הִתְנַדְנְדוּ קָדִימָה וְאָחוֹרָה, וְקוֹלָם נִשְׁמַע כְּמוֹ רַעַם נָעִים וּמָתוֹק. רֵיחַ שֶׁל גְּוִילִים יְשָׁנִים, מֵהֶם לָמְדוּ אֶת הַתּוֹרָה, עָמַד בָּאֲוִיר.

בַּשּׁוּרָה הָרִאשׁוֹנָה שֶׁל בֵּית הַמִּדְרָשׁ יָשַׁב אָדָם מְיֻחָד בְּמִינוֹ. שְׁמוֹ הָיָה רַבִּי שִׁמְעוֹן בַּר אַבָּא .

אִם הֱיִיתֶם מַבִּיטִים עַל פָּנָיו, הֱיִיתֶם רוֹאִים אוֹר מַמָּשׁ. חִיּוּךְ שָׁלֵו וְרָגוּעַ לֹא מָשׁ מִשְּׂפָתָיו, וְעֵינָיו הַטְּהוֹרוֹת בָּרְקוּ מֵאַהֲבַת הַשֵּׁם. אַךְ אִם הֱיִיתֶם מַבִּיטִים עַל בְּגָדָיו... אוֹי, אֵיזֶה צַעַר. הַגְּלִימָה שֶׁלּוֹ הָיְתָה קְרוּעָה וַעֲשׂוּיָה טְלָאִים טְלָאִים. סַנְדָּלָיו הָיוּ בְּלוּיִים כָּל כָּךְ, עַד שֶׁבְּכָל פַּעַם שֶׁדָּרַךְ עַל אַבְנֵי הָרְחוֹב, יָכוֹל הָיָה לְהַרְגִּישׁ אֶת הַקֹּר הַחֲדַרְדָּר עוֹלֶה בְּרַגְלָיו.

רַבִּי שִׁמְעוֹן הָיָה עָנִי מָרוּד. מִשְּׁעַת בֹּקֶר מֻקְדֶּמֶת לֹא בָּא פֵּרוּר לֶחֶם אֶל פִּיו. הַבֶּטֶן שֶׁלּוֹ קִרְקְרָה, אַךְ הוּא לֹא שָׂם לֵב לָרָעָב כְּלָל, כִּי מְתִיקוּת הַתּוֹרָה מִלְּאָה אֶת הַנְּשָׁמָה שֶׁלּוֹ.

הַלֵּב הַכּוֹאֵב שֶׁל הֶחָבֵר

לֹא הַרְחֵק מִמֶּנּוּ, מֵעֵבֶר לַשֻּׁלְחָן, יָשַׁב חֲבֵרוֹ הַטּוֹב, גְּדוֹל הַדּוֹר – רַבִּי אֶלְעָזָר .

רַבִּי אֶלְעָזָר הִבִּיט בַּחֲבֵרוֹ רַבִּי שִׁמְעוֹן וְלִבּוֹ נִכְמַר מֵרַחֲמִים. הוּא שָׁמַע אֶת נְשִׁימוֹתָיו הַחַלָּשׁוֹת. הוּא יָדַע שֶׁאֵין לְרַבִּי שִׁמְעוֹן מָה לֶאֱכֹל, וְשֶׁגַּם יְלָדָיו בַּבַּיִת וַדַּאי רְעֵבִים. רַבִּי אֶלְעָזָר הִכְנִיס אֶת יָדוֹ לְכִיסוֹ הַיְמָנִי, וְהִרְגִּישׁ תַּחַת אֶצְבְּעוֹתָיו כַּמָּה "דִּינְרֵי זָהָב" – מַטְבְּעוֹת זָהָב כְּבֵדִים, חֲלָקִים וִיקָרִים.

"אֲנִי חַיָּב לָתֵת לוֹ אֶת הַכֶּסֶף הַזֶּה!" חָשַׁב רַבִּי אֶלְעָזָר. "הַכֶּסֶף הַזֶּה יָכוֹל לִקְנוֹת לֶחֶם, דָּגִים, יְרָקוֹת וְתַפּוּחִים מְתוּקִים עֲבוּר מִשְׁפַּחְתּוֹ שֶׁל רַבִּי שִׁמְעוֹן לְשָׁבוּעוֹת אֲרֻכִּים!".

אֲבָל... הָיְתָה בְּעָיָה שֶׁנִּרְאֲתָה בִּלְתִּי אֶפְשָׁרִית לְפִתְרוֹן.

רַבִּי אֶלְעָזָר הִכִּיר הֵיטֵב אֶת חֲבֵרוֹ. הוּא יָדַע שֶׁאִם הוּא יִגַּשׁ לְרַבִּי שִׁמְעוֹן וְיַגִּיד לוֹ: "חֲבֵרִי הַיָּקָר, קַח בְּבַקָּשָׁה כֶּסֶף, מַתָּנָה מִמֶּנִּי" – רַבִּי שִׁמְעוֹן לְעוֹלָם לֹא יַסְכִּים לָקַחַת!

מַדּוּעַ? לֹא מִגַּאֲוָה, אֶלָּא מִיִּרְאַת שָׁמַיִם. שְׁלֹמֹה הַמֶּלֶךְ אָמַר בְּסֵפֶר מִשְׁלֵי: "וְשׂוֹנֵא מַתָּנֹת יִחְיֶה". אָדָם גָּדוֹל בּוֹטֵחַ רַק בַּקָּדוֹשׁ בָּרוּךְ הוּא שֶׁיְּפַרְנֵס אוֹתוֹ, וְאֵינוֹ מוּכָן לִקַּחַת מַתְּנוֹת חִנָּם מִבְּנֵי אָדָם.

"אִם כָּךְ," לָחַשׁ רַבִּי אֶלְעָזָר לְעַצְמוֹ בְּסוֹד, עֵינָיו בּוֹרְקוֹת בְּחָכְמָה, "אֶעֱשֶׂה זֹאת בְּסֵתֶר. אֲנִי אֶדְאַג שֶׁהוּא יִקַּח אֶת הַכֶּסֶף, אֲבָל בְּדֶרֶךְ שֶׁאֲפִלּוּ לֹא יֵדַע שֶׁאֲנִי הוּא הַנּוֹתֵן!".

הַתָּכְנִית הַסּוֹדִית בַּסִּמְטָה

לְהַשֵּׁם יִתְבָּרַךְ תָּמִיד יֵשׁ תִּזְמוּן מֻשְׁלָם.

שְׁעוֹת הַלִּמּוּד שֶׁל הַבֹּקֶר הִסְתַּיְּמוּ. תַּלְמִידֵי הַחֲכָמִים יָצְאוּ מִבֵּית הַמִּדְרָשׁ אֶל רְחוֹבוֹת הָעִיר, שֶׁהָיוּ מְלֵאִים בַּהֲמֻלָּה וּצְעָקוֹת קוֹנִים וּמוֹכְרִים. רַבִּי שִׁמְעוֹן בַּר אַבָּא יָצָא וְהֵחֵל לִפְסֹעַ מַעֲדַנּוֹת (לְאַט וּבְרֹגַע) בְּאַחַת הַסִּמְטָאוֹת הַצָּרוֹת.

רַבִּי אֶלְעָזָר עָקַב אַחֲרָיו מֵרָחוֹק, בְּלִי שֶׁרַבִּי שִׁמְעוֹן יַרְגִּישׁ.

"זֶה הָרֶגַע שֶׁלִּי," חָשַׁב רַבִּי אֶלְעָזָר. הוּא הִזְדָּרֵז, עָקַף אֶת רַבִּי שִׁמְעוֹן מֵרְחוֹב אַחֵר, וְהִגִּיעַ לַסִּמְטָה לְפָנָיו. שָׁם, בְּמָקוֹם פָּנוּי שֶׁאֵין בּוֹ אִישׁ, הוּא הוֹצִיא אֶת דִּינְרֵי הַזָּהָב הַנּוֹצְצִים מִכִּיסוֹ... וְהִפִּיל אוֹתָם עַל הָרִצְפָּה.

צְלִינְג... צְלִינְג... קוֹל מַתַּכְתִּי וּנְקִי נִשְׁמַע בַּסִּמְטָה. הַמַּטְבְּעוֹת נָפְלוּ בֵּין אַבְנֵי הָרְחוֹב וְנִצְנְצוּ בְּאוֹר הַשֶּׁמֶשׁ הַחַמָּה.

רַבִּי אֶלְעָזָר עָמַד בַּצַּד, הִסְתַּתֵּר מְעַט מֵאֲחוֹרֵי פִּנַּת אֶבֶן, וְהִמְתִּין בְּלֵב פּוֹעֵם.

"עַכְשָׁו," חִיֵּךְ רַבִּי אֶלְעָזָר לְעַצְמוֹ, "יַגִּיעַ רַבִּי שִׁמְעוֹן. הוּא יִרְאֶה כֶּסֶף מֻפְקָר עַל הָרִצְפָּה (בְּלִי שׁוּם סִימָן הֶכֵּר). עַל פִּי הַהֲלָכָה, כֶּסֶף שֶׁנָּפַל וְאֵין בּוֹ סִימָן – מֻתָּר לַמּוֹצֵא לָקַחַת אוֹתוֹ! כָּךְ הַקָּדוֹשׁ בָּרוּךְ הוּא יַעֲנִיק לְרַבִּי שִׁמְעוֹן אֶת הַכֶּסֶף מִבְּלִי שֶׁהוּא יַרְגִּישׁ שֶׁקִּבֵּל מַתָּנָה מִבֶּן אָדָם!".

הָאֱמֶת שֶׁחֲזָקָה מִזָּהָב

וְאָכֵן, רַגְלָיו בְּעוֹת הַסַּנְדָּלִים שֶׁל רַבִּי שִׁמְעוֹן נִשְׁמְעוּ עַל הָאֲבָנִים. טַק, טַק, טַק.

הוּא נִכְנַס לַסִּמְטָה. לְפֶתַע, עֵינָיו קָלְטוּ הֶבְזֵק מַבְרִיק מִן הָרִצְפָּה. הוּא עָצַר.

יְלָדִים, אָנוּ מְדַבְּרִים עַל אָדָם שֶׁאֵין לוֹ פְּרוּסַת לֶחֶם מְרֻחָה בִּמְעַט שֶׁמֶן. מוּל עֵינָיו מֻנָּח אוֹצָר! מָה אָדָם רָגִיל הָיָה עוֹשֶׂה? הָיָה צוֹעֵק "יֵשׁ! תּוֹדָה ה'!", חוֹטֵף אֶת הַכֶּסֶף וְרָץ יָשָׁר לַמַּאֲפִיָּה!

אֲבָל רַבִּי שִׁמְעוֹן הָיָה עֲנַק הָרוּחַ.

הוּא אָמְנָם הִתְכּוֹפֵף. יָדָיו חִפְּשׂוּ עַל הָרִצְפָּה וְאָסְפוּ בַּעֲדִינוּת אֶת מַטְבְּעוֹת הַזָּהָב. אַךְ תּוֹךְ כְּדֵי שֶׁהוּא מֵרִים אוֹתָם, עֵינָיו חִפְּשׂוּ סָבִיב. הוּא הִבְחִין בְּרַבִּי אֶלְעָזָר מֵצִיץ מֵהַפִּנָּה בְּמֶרְחָק קָצָר לְפָנָיו.

הַמֹּחַ הֶחָרִיף שֶׁל רַבִּי שִׁמְעוֹן עָבַד כְּבָרָק. "אִם רַבִּי אֶלְעָזָר הָלַךְ כָּאן לְפָנַי," חָשַׁב רַבִּי שִׁמְעוֹן, "סָבִיר מְאֹד לְהַנִּיחַ שֶׁהַכֶּסֶף הַזֶּה שַׁיָּךְ לְרַבִּי אֶלְעָזָר וְנָפַל מִכִּיסוֹ!".

רַבִּי שִׁמְעוֹן לֹא חָשַׁב לְרֶגַע לִשְׁמֹר אֶת הַכֶּסֶף! הוּא לָקַח אֶת הַמַּטְבְּעוֹת, קָם עַל רַגְלָיו, וְהֵחֵל לָרוּץ – עַד כַּמָּה שֶׁאִפְשְׁרוּ לוֹ כּוֹחוֹתָיו הַדַּלִּים – לְכִוּוּן רַבִּי אֶלְעָזָר.

"רַבִּי אֶלְעָזָר! הַמְתֵּן!" קָרָא רַבִּי שִׁמְעוֹן, קוֹלוֹ צָרוּד אַךְ יָצִיב. הוּא הִגִּיעַ לְפָנָיו, הִתְנַשֵּׁף מְעַט, וְהוֹשִׁיט לוֹ אֶת כַּף יָדוֹ הַפְּתוּחָה, שָׁם נָחוּ מַטְבְּעוֹת הַזָּהָב. "נָפַל לְךָ, חֲבֵרִי. זֶה שֶׁלְּךָ. מָצָאתִי זֹאת מִיָּד אַחֲרֶיךָ."

רֶגַע שֶׁל מְסִירוּת עַל גַּאֲוַת הַקֹּדֶשׁ

רַבִּי אֶלְעָזָר הִבִּיט בַּזָּהָב שֶׁבְּיַד חֲבֵרוֹ. הַתָּכְנִית שֶׁלּוֹ כִּמְעַט נִכְשְׁלָה! הוּא מִהֵר לוֹמַר בְּאַהֲבָה גְּדוֹלָה: "רַבִּי שִׁמְעוֹן, אָחִי הַיָּקָר. אָמְנָם זֶה הָיָה הַכֶּסֶף שֶׁלִּי, אַךְ אַל תִּצְטַעֵר. דַּע לְךָ שֶׁאָנֹכִי הִפְקַרְתִּי אֶת הַכֶּסֶף הַזֶּה לַחֲלוּטִין! אָמַרְתִּי מְפֹרָשׁ 'הֲרֵי זֶה הֶפְקֵר', וְעַל פִּי דִּין תּוֹרָה – מִי שֶׁמּוֹצֵא זֹאת הָרִאשׁוֹן, זוֹכֶה בָּזֶה מִן הַדִּין! הַכֶּסֶף שֶׁלְּךָ!".

אֲבָל רַבִּי שִׁמְעוֹן שֶׁחַי בִּקְדֻשָּׁה, רָאָה מִיָּד אֶת הָאֱמֶת. הַחִיּוּךְ שֶׁלּוֹ מִמֻּקְדָּם חָזַר אֶל פָּנָיו. הוּא הִבִּיט בְּעֵינָיו הָאוֹהֲבוֹת שֶׁל רַבִּי אֶלְעָזָר, וְהֵבִין שֶׁכָּל זֶה נַעֲשָׂה רַק בִּשְׁבִילוֹ.

"וְכִי סוֹבֵר אַתָּה שֶׁאֵינִי מֵבִין?" אָמַר רַבִּי שִׁמְעוֹן בְּקוֹל עָדִין שֶׁלֹּא הָיָה בּוֹ טִפַּת כַּעַס, "יוֹדֵעַ אֲנִי שֶׁלֹּא הִפְקַרְתָּ אֶת הַמַּטְבְּעוֹת הָאֵלּוּ אֶלָּא מִפְּנֵי שֶׁרָאִיתָ אוֹתִי הוֹלֵךְ מֵאֲחוֹרֶיךָ. רָצִיתָ לַעֲשׂוֹת לִי מַתָּנָה מִבְּלִי שֶׁאֶתְבַּיֵּשׁ".

"אִם כָּךְ, קַח אוֹתָם!" הִתְחַנֵּן רַבִּי אֶלְעָזָר. "הֲרֵי זֶה כְּבָר בְּיָדְךָ!".

"לֹא, אָחִי הַטּוֹב," סָגַר רַבִּי שִׁמְעוֹן אֶת יָדוֹ שֶׁל רַבִּי אֶלְעָזָר עַל הַזָּהָב. "אֲנִי מוֹדֶה לְךָ עַל הָאַהֲבָה וְהַחֶסֶד שֶׁבְּלִבְּךָ. אֲבָל כְּבָר הֶחְלַטְתִּי: אֵינֶנִּי לוֹקֵחַ מַתָּנוֹת. הַקָּדוֹשׁ בָּרוּךְ הוּא הוּא זֶה שֶׁזָּן וּמְפַרְנֵס לַכֹּל, הוּא לֹא שָׁכַח אוֹתִי. וְאִם הוּא רוֹצֶה שֶׁאֶחְיֶה עַתָּה בְּעֹנִי, אֲקַבֵּל זֹאת בְּאַהֲבָה! בַּתּוֹרָה שֶׁלִּי יֵשׁ אֶת כָּל הָעֹשֶׁר שֶׁאֲנִי צָרִיךְ".

רַבִּי שִׁמְעוֹן סוֹבֵב אֶת גַּבּוֹ, וְהִמְשִׁיךְ לִפְסֹעַ מַעֲדַנּוֹת לְבֵיתוֹ. בִּטְנוֹ קִרְקְרָה, סַנְדָּלָיו הָיוּ קְרוּעִים, אֲבָל... יְלָדִים, בְּאוֹתוֹ רֶגַע הוּא הָיָה הָאָדָם הֶעָשִׁיר בְּיוֹתֵר בָּעוֹלָם! כִּי לַגַּאֲוָה שֶׁל קְדֻשָּׁה, וְלָאֱמוּנָה בַּה', אֵין מְחִיר שֶׁל זָהָב!

וְרַבִּי אֶלְעָזָר? הוּא עָמַד שָׁם, מַבִּיט עַל חֲבֵרוֹ הַמִּתְרַחֵק, וְדִמְעָה קְטַנָּה שֶׁל הִתְרַגְּשׁוּת יָרְדָה מֵעֵינוֹ. דִּמְעָה שֶׁל זְכוּת גְּדוֹלָה – הַזְּכוּת לִרְאוֹת צַדִּיק אֲמִתִּי שֶׁבֶּאֱמֶת, אֲבָל בֶּאֱמֶת, סוֹמֵךְ רַק עַל הַקָּדוֹשׁ בָּרוּךְ הוּא.

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “יְלָדִים, עִצְמוּ אֶת הָעֵינַיִם וְנַסּוּ לְדַמְיֵן אֶת בֵּית הַמִּדְרָשׁ הַגָּדוֹל שֶׁל אֶרֶץ יִשְׂרָאֵל לִפְנֵי כְּאַלְפַּיִם שָׁנָה.” — מה קרה מיד לאחר מכן?

ה. כיצד הגיע הסיפור אל האירוע: “בַּתּוֹרָה שֶׁלִּי יֵשׁ אֶת כָּל הָעֹשֶׁר שֶׁאֲנִי צָרִיךְ".”?

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “מתן בסתר” על ר' שמעון בר אבא ור' אלעזר? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “הַמַּטְבְּעוֹת נָפְלוּ בֵּין אַבְנֵי הָרְחוֹב וְנִצְנְצוּ בְּאוֹר הַשֶּׁמֶשׁ הַחַמָּה.”.
ג. אילו ר' שמעון בר אבא ור' אלעזר לא היה פועל כפי שמתואר ב־“הַמְתֵּן!" קָרָא רַבִּי שִׁמְעוֹן, קוֹלוֹ צָרוּד אַךְ יָצִיב.” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “בַּתּוֹרָה שֶׁלִּי יֵשׁ אֶת כָּל הָעֹשֶׁר שֶׁאֲנִי צָרִיךְ".” מחזק את המסר של הסיפור?

3. רַבִּי אֶלְעָזָר עָקַב אַחֲרָיו מֵרָחוֹק, בְּלִי שֶׁרַבִּי שִׁמְעוֹן…
4. תּוֹדָה ה'!", חוֹטֵף אֶת הַכֶּסֶף וְרָץ יָשָׁר לַמַּאֲפִיָּה!
5. אָמַרְתִּי מְפֹרָשׁ 'הֲרֵי זֶה הֶפְקֵר', וְעַל פִּי דִּין תּוֹרָה – מִי…
6. דִּמְעָה שֶׁל זְכוּת גְּדוֹלָה – הַזְּכוּת לִרְאוֹת צַדִּיק אֲמִתִּי…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “תַּלְמִידֵי הַחֲכָמִים יָצְאוּ מִבֵּית הַמִּדְרָשׁ אֶל רְחוֹבוֹת הָעִיר, שֶׁהָיוּ מְלֵאִים בַּהֲמֻלָּה וּצְעָקוֹת קוֹנִים…”.
מכתב: כתבו לר' שמעון בר אבא ור' אלעזר בעקבות “אַךְ תּוֹךְ כְּדֵי שֶׁהוּא מֵרִים אוֹתָם, עֵינָיו חִפְּשׂוּ סָבִיב.”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “אֲבָל רַבִּי שִׁמְעוֹן שֶׁחַי בִּקְדֻשָּׁה, רָאָה מִיָּד אֶת הָאֱמֶת.”.
דיווח: כתבו ידיעה קצרה שכותרתה “מתן בסתר”.

המחזה

רגע מן הסיפורהדמויותמה יאמרו או יעשו?
הוּא יָדַע שֶׁאִם הוּא יִגַּשׁ לְרַבִּי שִׁמְעוֹן וְיַגִּיד לוֹ: "חֲבֵרִי הַיָּקָר, קַח…
הַמְתֵּן!" קָרָא רַבִּי שִׁמְעוֹן, קוֹלוֹ צָרוּד אַךְ יָצִיב.

7. חיבור לחיים מתוך המעשה

א. מה אפשר ללמוד מן הדרך שבה ר' שמעון בר אבא ור' אלעזר פעל באירוע “כָּךְ הַקָּדוֹשׁ בָּרוּךְ הוּא יַעֲנִיק לְרַבִּי שִׁמְעוֹן אֶת הַכֶּסֶף מִבְּלִי שֶׁהוּא יַרְגִּישׁ שֶׁקִּבֵּל מַתָּנָה מִבֶּן…”?
