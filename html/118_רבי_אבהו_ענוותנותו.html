דף עבודה בעקבות רבי אבהו
שם: כיתה:

המעשה — הסיפור המלא

בס"ד

להלן עיבוד חינוכי, רגשי ומרתק על ענוותנותו העצומה של רבי אבהו , מעובד במיוחד לבנים בכיתות ב'-ד' (חינוך דתי-לאומי). הסיפור משלב את המובא במקורות שסיפקת [1] על פניו המאירות, יחד עם סיפור הענווה המפורסם ממסכת סוטה.

הסיפור מנוקד במלואו כנדרש.

פְּתִיחָה: הָאוֹר שֶׁל עִיר הַנָּמֵל

יְלָדִים, בּוֹאוּ נִסַּע יַחַד בַּדִּמְיוֹן לְעִיר יָפָה וַעֲתִיקָה עַל חֹוף הַיָּם הַתִּיכוֹן. שְׁמָהּ - קֵיסָרִין (קֵיסָרְיָה).

אִם תַּעַמְדוּ שָׁם, תּוּכְלוּ לְהָרִיחַ אֶת הָרֵיחַ הַמָּלוּחַ שֶׁל מֵי הַיָּם הַנִּתָּזִים עַל הַסְּלָעִים, וְלִשְׁמֹעַ אֶת רַעַשׁ הָאֳנִיּוֹת הַגְּדוֹלוֹת שֶׁל הָרוֹמָאִים הַפּוֹרְקוֹת סְחוֹרָה. בְּקֵיסָרִין הָיוּ אַרְמוֹנוֹת עֲנָקִיִּים שֶׁל שָׂרֵי רוֹמִי, עַמּוּדֵי שַׁיִשׁ מְפֹאָרִים וְרְחוֹבוֹת הֲומִי אָדָם.

אֲבָל הַדָּבָר הֲכִי נוֹצֵץ בְּקֵיסָרִין לֹא הָיָה מֵי הַיָּם וְלֹא זְהַב הָרוֹמָאִים. הַדָּבָר הֲכִי מוֹאֵר בָּעִיר הָיָה... הַפָּנִים שֶׁל רַב הָעִיר הַגָּדוֹל, רַבִּי אֲבָהוּ .

רַבִּי אֲבָהוּ הָיָה תַּלְמִיד חָכָם עָצוּם, וְאָדָם עָשִׁיר וּמְכֻבָּד מְאֹד. אֲפִלּוּ הַקֵּיסָר הָרוֹמָאִי רָחַשׁ לוֹ כָּבוֹד מְלָכִים. אֲבָל מָה שֶׁשִּׂמַּח אֶת רַבִּי אֲבָהוּ יוֹתֵר מִכֹּל, הָיָה לִמּוּד הַתּוֹרָה. תַּלְמִידָיו הָיוּ מְסַפְּרִים, שֶׁכַּאֲשֶׁר רַבִּי אֲבָהוּ הָיָה לוֹמֵד "תּוֹסֶפְתָּא" חֲדָשָׁה שֶׁלֹּא יָדַע קֹדֶם, הַפָּנִים שֶׁלּוֹ הָיוּ פָּשׁוּט מְאִירוֹת! מַמָּשׁ זוֹרְחוֹת כְּמוֹ פַּנָּס קָדוֹשׁ! עָלָיו אָמַר שְׁלֹמֹה הַמֶּלֶךְ: "חָכְמַת אָדָם תָּאִיר פָּנָיו" [1].

שְׁנֵי הַשִּׁעוּרִים בְּבֵית הַמִּדְרָשׁ

בְּאוֹתָם יָמִים, הַשִּׁעוּרִים בְּבֵית הַמִּדְרָשׁ הָיוּ הַמָּקוֹם שֶׁאֵלָיו כֻּלָּם הָיוּ הוֹלְכִים. רַבִּי אֲבָהוּ הָיָה יוֹשֵׁב וּמְלַמֵּד אֶת אַנְשֵׁי עִירוֹ הֲלָכוֹת קָשׁוֹת וַעֲמֻקּוֹת. דִּינֵי שַׁבָּת, כַּשְׁרוּת וּמִשְׁפָּטִים מְסֻבָּכִים. לִמּוּד הֲלָכָה דּוֹרֵשׁ רִכּוּז רַב, וְהָאֲנָשִׁים רָחֲשׁוּ כָּבוֹד רַב לְרַבָּם הַמַּעֲמִיק.

וְהִנֵּה, יוֹם אֶחָד, הִגִּיעַ לְקֵיסָרִין אוֹרֵחַ חָשׁוּב: רַבִּי חִיָּא בַּר אַבָּא .

חֲכָמֵי הָעִיר הֶחְלִיטוּ שֶׁשְּׁנֵי הָרַבָּנִים יִמְסְרוּ שִׁעוּר בְּאוֹתוֹ הַזְּמַן, בִּשְׁנֵי חֲדָרִים שׁוֹנִים שֶׁל בֵּית הַמִּדְרָשׁ.

רַבִּי אֲבָהוּ הִתְיַשֵּׁב בְּחַדְרוֹ, הֵרִיחַ אֶת רֵיחַ קְלָפֵי הָעוֹר הַיְשָׁנִים שֶׁל הַסְּפָרִים, וְהֵחֵל לִדְרֹשׁ בְּ"הֲלָכָה" – הַחֻקִּים הַמְדֻיָּקִים שֶׁל הַתּוֹרָה.

בַּחֶדֶר הַשֵּׁנִי, הִתְיַשֵּׁב רַבִּי חִיָּא בַּר אַבָּא, וְהֵחֵל לִדְרֹשׁ בְּ"אַגָּדָה" – סִפּוּרִים מְרַתְּקִים, מִדְרָשִׁים עַל יְצִיאַת מִצְרַיִם, וּמְשָׁלִים שֶׁמּוֹשְׁכִים אֶת הַלֵּב.

יְלָדִים, אֲנָשִׁים פְּשׁוּטִים אוֹהֲבִים מְאֹד סִפּוּרִים. כְּשֶׁהֵם עֲיֵפִים מֵהָעֲבוֹדָה בַּשָּׂדֶה וּבַנָּמֵל, קָשֶׁה לָהֶם לְהִתְרַכֵּז בַּהֲלָכוֹת סְבוּכוֹת. בְּתוֹךְ דַּקּוֹת סְפוּרוֹת, קָרָה דָּבָר מֵבִיךְ.

רַבִּי אֲבָהוּ עוֹמֵד וְדוֹרֵשׁ בַּחֶדֶר שֶׁלּוֹ... וּפִתְאוֹם, הוּא שׁוֹמֵעַ צְעָדִים. שְׁרְרְק... שְׁרְרְק... קוֹל גְּרִירַת סַנְדָּלִים. הוּא מֵרִים אֶת עֵינָיו וְרוֹאֶה אֲנָשִׁים שֶׁקָּמִים מֵהַסַּפְסָלִים, וְיוֹצְאִים בְּשֶׁקֶט מֵהַדֶּלֶת. עוֹד חֲמִשָּׁה קָמִים, וְעוֹד עֲשָׂרָה קָמִים וְיוֹצְאִים.

הֵם עָבְרוּ כֻּלָּם לַחֶדֶר הַשֵּׁנִי, כְּדֵי לִשְׁמֹעַ אֶת הַסִּפּוּרִים הַמְּתוּקִים שֶׁל רַבִּי חִיָּא!

הֶחָדֶר הָרֵיק וְהַבּוּשָׁה שֶׁל הָאוֹרֵחַ

בְּתוֹךְ זְמַן קָצָר, הַחֶדֶר שֶׁל רַב הָעִיר, רַבִּי אֲבָהוּ, נִשְׁאַר כִּמְעַט רֵיק מֵאָדָם. רַק קֹמֶץ שֶׁל תַּלְמִידֵי חֲכָמִים עֲמֻקִּים נִשְׁאֲרוּ אִתּוֹ. קוֹלוֹ שֶׁל רַבִּי אֲבָהוּ הִדְהֵד עַכְשָׁו בֵּין הַקִּירוֹת הָרֵיקִים.

בּוֹאוּ נַעֲצֹר רֶגַע וְנַחְשֹׁב. אֵיךְ אָדָם רָגִיל הָיָה מַרְגִּישׁ? הָרַב שֶׁל קֵיסָרִין, מְכֻבָּד אֵצֶל הַקֵּיסָר, וּפִתְאוֹם כָּל אַנְשֵׁי הָעִיר עוֹזְבִים אוֹתוֹ שָׁם לְבַד מִפְּנֵי שֶׁרַב אַחֵר הִגִּיעַ? כָּל הַכָּבוֹד שֶׁלּוֹ נִרְמַס. אָדָם רָגִיל הָיָה מַאְדִּים מִכַּעַס. אוּלַי הָיָה צוֹעֵק, אֲפִלּוּ סוֹגֵר אֶת הַסֵּפֶר וְעוֹזֵב אֶת הַמָּקוֹם.

כְּשֶׁהִסְתַּיֵּם הַשִּׁעוּר, נוֹדַע הַדָּבָר לְרַבִּי חִיָּא בַּר אַבָּא. כְּשֶׁרַבִּי חִיָּא הֵבִין שֶׁכָּל הַקָּהָל עָזַב אֶת רַב הָעִיר וּבָא אֵלָיו, לִבּוֹ נָפַל. צַעַר גָּדוֹל עֲטָפוֹ. הוּא הִרְגִּישׁ שֶׁהוּא עָלַב בְּרַבִּי אֲבָהוּ, אֲפִלּוּ בְּלִי כַּוָּנָה. הוּא הָיָה בָּטוּחַ שֶׁרַבִּי אֲבָהוּ שׂוֹנֵא אוֹתוֹ עַכְשָׁו עַל הַבּוּשָׁה הַגְּדוֹלָה שֶׁנִּגְרְמָה לוֹ בְּעִירוֹ שֶׁלּוֹ. חֲלִישׁוּת הַדַּעַת נָפְלָה עַל רַבִּי חִיָּא.

סוֹחֵר הַמַּרְגָּלִיּוֹת – נִצָּחוֹן שֶׁל אַהֲבָה

כְּשֶׁרַבִּי אֲבָהוּ הֵבִין שֶׁחֲבֵרוֹ עָצוּב, הוּא מִהֵר לָלֶכֶת אֵלָיו. הַאִם פָּנָיו שֶׁל רַבִּי אֲבָהוּ הָיוּ כּוֹעֲסוֹת? זְעוּפוֹת?

לְגַמְרֵי לֹא! פָּנָיו הָיוּ מְאִירוֹת כַּשֶּׁמֶשׁ שֶׁל צָהֳרַיִם.

הוּא נִגַּשׁ לְרַבִּי חִיָּא, שָׂם אֶת יָדָיו עַל כְּתֵפָיו בְּחֹם, וְאָמַר לוֹ בְּחִיּוּךְ מָלֵא אַהֲבָה: "רַבִּי חִיָּא, אָחִי הַיָּקָר! לָמָּה פָּנֶיךָ נְפוּלוֹת? בּוֹא אַמְשִׁיל לְךָ מָשָׁל. לְמָה הַדָּבָר דּוֹמֶה?"

"דַּמְיֵן לְךָ מֶרְכַּז קְנִיּוֹת בַּשּׁוּק הַגָּדוֹל. שְׁנֵי סוֹחֲרִים הִגִּיעוּ לָעִיר וּפָתְחוּ אֶת הַדּוּכָנִים שֶׁלָּהֶם זֶה מִמּוּל זֶה.

הַסּוֹחֵר הָאֶחָד הוֹצִיא מֵהַתִּיק שֶׁלּוֹ אֲבָנִים טוֹבוֹת וּמַרְגָּלִיּוֹת יְקָרוֹת מְאֹד. אֲבָנִים מְנַצְנְצוֹת מִזָּהָב שֶׁכָּל אַחַת שָׁוָה מִילְיוֹנִים!

וְהַסּוֹחֵר הַשֵּׁנִי פָּתַח דּוּכָן וְשָׁפַךְ שָׁם סִדְקִית . מָה זֶה סִדְקִית? צַעֲצוּעִים קְטַנִּים מִפְּלַסְטִיק, סִכּוֹת צִבְעוֹנִיּוֹת, מַסְרְקִים נוֹצְצִים וּזְכוּכִיּוֹת מְקֻשָּׁטוֹת מַשֶּׁהוּ שֶׁשָּׁוֶה שְׁקָלִים בּוֹדְדִים."

עָצַר רַבִּי אֲבָהוּ, תָּלָה עֵינַיִם מְאִירוֹת בְּרַבִּי חִיָּא וְשָׁאַל: "תַּגִּיד לִי אַתָּה, עַל מִי הֲמוֹן הָעָם יִקְפְּצוּ? לְאָן יָרוּצוּ כָּל הַיְלָדִים, וְהַפּוֹעֲלִים, וְהָאֲנָשִׁים הַפְּשׁוּטִים?"

"וַדַּאי שֶׁיָּרוּצוּ לְדוּכַן הַסִּדְקִית וְהַצַּעֲצוּעִים!" עָנָה רַבִּי חִיָּא. "זֶה מַשֶּׁהוּ יָפֶה שֶׁמּוֹשֵׁךְ אֶת הָעַיִן, וּקְלִיל לִקְנוֹת אוֹתוֹ".

"וּמִי יִגַּשׁ לְסוֹחֵר הַמַּרְגָּלִיּוֹת?"

"רַק אֲנָשִׁים עֲשִׁירִים מְאֹד, מְבִינֵי עִנְיָן, שֶׁיּוֹדְעִים לְהַעֲרִיךְ יַהֲלוֹם אֲמִתִּי וִיקַר עֵרֶךְ."

חִיֵּךְ רַבִּי אֲבָהוּ חִיּוּךְ רָחָב: "זֶה בְּדִיּוּק מָה שֶׁקָּרָה לָנוּ הַיּוֹם אָחִי!

אֲנִי דָּרַשְׁתִּי בַּהֲלָכָה. הֲלָכָה הִיא מַרְגָּלִיּוֹת! הִיא חֻקִּים קָשִׁים וַעֲמֻקִּים מְאֹד. לֹא כָּל אֶחָד יָכוֹל לְהָבִין וְלִלְמֹד אוֹתָהּ. אַתָּה דָּרַשְׁתָּ בְּאַגָּדָה – דְּבָרִים מְתוּקִים שֶׁמּוֹשְׁכִים אֶת הַלֵּב וְשָׂוִים לְכָל נֶפֶשׁ כְּמוֹ הַסִּדְקִית הַיָּפָה.

בְּוַדַּאי שֶׁכֻּלָּם כָּפְצוּ עָלֶיךָ! אֲנָשִׁים אוֹהֲבִים מְשָׁלִים מְשַׂמְּחִים! וּמִי נִשְׁאַר אֶצְלִי? רַק אוֹתָם קְצָת תַּלְמִידֵי חֲכָמִים גְּדוֹלִים שֶׁמְּבִינִים בְּמַרְגָּלִיּוֹת! הַאִם עַל זֶה אֶכְעַס? הֲרֵי זוֹ גַּאֲוָתִי, שֶׁאַתָּה הִצְלַחְתָּ לְקָרֵב כָּל כָּךְ הַרְבֵּה אֲנָשִׁים לְאָבִינוּ שֶׁבַּשָּׁמַיִם!"

הַלִּוּוּי בָּרְחוֹבוֹת

הַהַשְׁגָּחָה הַפְּרָטִית דָּאֲגָה לְכָךְ, שֶׁלֹּא רַק בְּמִלִּים יַרְאֶה רַבִּי אֲבָהוּ שֶׁאֵין בְּלִבּוֹ טִיפַּת קִנְאָה.

בְּמֶשֶׁךְ כָּל אוֹתָם הַיָּמִים שֶׁבָּהֶם שָׁהָה רַבִּי חִיָּא בְּקֵיסָרִין, מָה עָשָׂה רַבִּי אֲבָהוּ הַגָּדוֹל? בְּמָקוֹם לָשֶׁבֶת בְּמִשְׂרָדוֹ הַמְּפֹאָר, הוּא לָקַח עַל עַצְמוֹ תַּפְקִיד שֶׁל מְלַוֶּה.

בְּכָל בֹּקֶר וָעֶרֶב, הוּא עָזַב אֶת עִסּוּקָיו הָרַבִּים, וְהָלַךְ בְּעַצְמוֹ לָאכְסַנְיָה (הַמָּלוֹן) שֶׁל רַבִּי חִיָּא, כְּדֵי לְהַעֲנִיק לוֹ כָּבוֹד אִישִׁי! הוּא דָּאַג לְכָל מַחְסוֹרוֹ, לִוָּה אוֹתוֹ בָּרְחוֹבוֹת כְּשֶׁתַּלְמִיד רָץ אַחַר מַנְהִיגוֹ, וְדָאַג שֶׁכָּל הָעִיר תִּרְאֶה וְתִלְמַד: כָּךְ אוֹהֲבִים אֶת כְּבוֹד הַתּוֹרָה! כָּךְ מְבַטְּלִים אֶת הַגַּאֲוָה וְהַקִּנְאָה!

וְלָכֵן, יְלָדִים, פָּנָיו שֶׁל רַבִּי אֲבָהוּ תָּמִיד הֵאִירוּ [1]. כִּי כְּשֶׁאָדָם נָקִי מִקִּנְאָה לַחֲבֵרוֹ – כָּל שִׂמְחַת הַתּוֹרָה וְהָאַהֲבָה לַה' פּוֹרֶצֶת מִתּוֹךְ נִשְׁמָתוֹ לַחוּץ, וּמְאִירָה אֶת הָעוֹלָם כֻּלּוֹ.

א. שְׁאֵלַת פְּתִיחָה (לְעוֹרֵר זִהוּי וְרֶגֶשׁ):

"יְלָדִים, יָצָא לָכֶם פַּעַם שֶׁעָבַדְתֶּם קָשֶׁה עַל מַשֶּׁהוּ – אוּלַי מִגְדַּל קוּבִּיּוֹת, הַצָּגָה בַּכִּתָּה, אוֹ צִיּוּר מֻשְׁקָע – וּפִתְאוֹם כֻּלָּם סוֹבְבוּ אֶת הָרֹאשׁ וְהָלְכוּ לְהִסְתַּכֵּל עַל יֶלֶד אַחֵר בַּצַּד הַשֵּׁנִי שֶׁל הַחֶדֶר?

אֵיךְ הִרְגַּשְׁתֶּם בְּאוֹתוֹ רֶגַע? מְעַצְבֵּן, נָכוֹן? עָלוּב? קָשֶׁה לִשְׂמֹחַ בְּרֶגַע כָּזֶה.

הַיּוֹם נִלְמַד עַל חָכָם אֶחָד, גָּדוֹל בְּיִשְׂרָאֵל, שֶׁעָשׂוּ לוֹ בְּדִיּוּק אֶת זֶה... אֲבָל מָה שֶׁהוּא בָּחַר לַעֲשׂוֹת – יַפְתִּיעַ אֶתְכֶם לְגַמְרֵי".

ב. הֶעָרוֹת בִּצּוּעַ וּסְטוֹרִיטֶלִינְג (לַמּוֹרֶה):

ג. הַמְלָצוֹת לְתַמְלִיל 4 שְׁקוּפִיּוֹת עֲבוּר מַצֶּגֶת (וִיזוּאַל בְּדֶגֶשׁ עַל קְדֻשָּׁה):

ד. פְּעִילוּת מְסַכֶּמֶת (מִשְׂחַק תַּפְקִידִים וַהֲבָנָה):

"בּוֹאוּ נְשַׂחֵק: אֲנִי זָקוּק לִשְׁנֵי מִתְנַדְּבִים.

מִתְנַדֵּב 1 (מְיַצֵּג אֶת רַבִּי אֲבָהוּ) יִמְכֹּר בַּדּוּכָן "סְפָרִים שֶׁצָּרִיךְ לַחְשֹׁב עֲלֵיהֶם הַרְבֵּה וְקָשֶׁה".

מִתְנַדֵּב 2 (מְיַצֵּג אֶת רַבִּי חִיָּא) יִמְכֹּר בַּדּוּכָן "סִפּוּרִים קְלִילִים שֶׁכֻּלָּם מְבִינִים".

עַכְשָׁו נְבַקֵּשׁ מִ-10 יְלָדִים מֵהַכִּתָּה לָרוּץ לַדּוּכָן הַשֵּׁנִי וְלַעֲזֹב אֶת הָרִאשׁוֹן.

עַכְשָׁו הַמִּתְנַדֵּב הָרִאשׁוֹן צָרִיךְ לְנַסּוֹת לְתָאֵר אֵיךְ נִרְאֶה חִיּוּךְ גָּדוֹל (לֹא מְזֻיָּף אֶלָּא מֵהַלֵּב) וְלוֹמַר לַחֲבֵרוֹ: אֲנִי מַמָּשׁ שָׂמֵחַ בִּשְׁבִילְךָ שֶׁעָזַרְתָּ לְכֻלָּם!".

מקורות: פָּנָיו הַמְּאִירוֹת שֶׁל רַבִּי אֲבָהוּ כְּשֶׁלּוֹמֵד תּוֹרָה חֲדָשָׁה: תַּלְמוּד יְרוּשַׁלְמִי, שַׁבָּת (פ"ח ה"א), (וּמוּבָא בְּמַאֲגַר הַמְּקוֹרוֹת הָאִישִׁי שֶׁלְּךָ – מקור 512 מתוך 'כה עשו חכמינו'). סִפּוּר הַדְּרָשׁוֹת, נְהִירַת הַקָּהָל לְרַבִּי חִיָּא בַּר אַבָּא, מְשַׁל הַמַּרְגָּלִיּוֹת הַסִּדְקִית וְכָבוֹד וְלִוּוּי הָרַב בְּקֵיסָרִין: מְבוּסָּס עַל תַּלְמוּד בַּבְלִי, מַסֶּכֶת סוֹטָה, דַּף מ' עַמּוּד א' ("רבי אבהו ורבי חייא בר אבא איקלעו לההוא אתרא, פתח ר' אבהו בדרשה ופתח ר' חייא בר אבא באגדתא..."). מקור חינוכי חיצוני שהובא במיוחד להשלמת בקשתך על ענוותנותו של רבי אבהו.

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “להלן עיבוד חינוכי, רגשי ומרתק על ענוותנותו העצומה של רבי אבהו , מעובד במיוחד לבנים בכיתות ב'-ד' (חינוך דתי-לאומי).” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “בְּאוֹתָם יָמִים, הַשִּׁעוּרִים בְּבֵית הַמִּדְרָשׁ הָיוּ הַמָּקוֹם שֶׁאֵלָיו כֻּלָּם הָיוּ הוֹלְכִים.”?
ג. כיצד הגיב רבי אבהו למצב המתואר: “הֵם עָבְרוּ כֻּלָּם לַחֶדֶר הַשֵּׁנִי, כְּדֵי לִשְׁמֹעַ אֶת הַסִּפּוּרִים הַמְּתוּקִים שֶׁל רַבִּי חִיָּא!”?
ד. מה השתנה בין “הוּא נִגַּשׁ לְרַבִּי חִיָּא, שָׂם אֶת יָדָיו עַל כְּתֵפָיו בְּחֹם, וְאָמַר לוֹ בְּחִיּוּךְ מָלֵא אַהֲבָה: "רַבִּי…” לבין “רַק אוֹתָם קְצָת תַּלְמִידֵי חֲכָמִים גְּדוֹלִים שֶׁמְּבִינִים בְּמַרְגָּלִיּוֹת!”?
ה. כיצד הגיע הסיפור אל האירוע: “אֲבָל מָה שֶׁהוּא בָּחַר לַעֲשׂוֹת – יַפְתִּיעַ אֶתְכֶם לְגַמְרֵי".”?

לפני ואחרי

מה קרה לפני: “הוּא הִרְגִּישׁ שֶׁהוּא עָלַב בְּרַבִּי אֲבָהוּ, אֲפִלּוּ בְּלִי כַּוָּנָה.”?
מה קרה אחרי: “הֲרֵי זוֹ גַּאֲוָתִי, שֶׁאַתָּה הִצְלַחְתָּ לְקָרֵב כָּל כָּךְ הַרְבֵּה אֲנָשִׁים לְאָבִינוּ שֶׁבַּשָּׁמַיִם!”?

סיבה ותוצאה מתוך המעשה

האירוע הנתוןמה גרם לו?מה הייתה תוצאתו?
חֲכָמֵי הָעִיר הֶחְלִיטוּ שֶׁשְּׁנֵי הָרַבָּנִים יִמְסְרוּ שִׁעוּר בְּאוֹתוֹ הַזְּמַן, בִּשְׁנֵי חֲדָרִים שׁוֹנִים שֶׁל בֵּית הַמִּדְרָשׁ.
וַדַּאי שֶׁיָּרוּצוּ לְדוּכַן הַסִּדְקִית וְהַצַּעֲצוּעִים!" עָנָה רַבִּי חִיָּא.

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “ענוותנותו” על רבי אבהו? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “כְּשֶׁהִסְתַּיֵּם הַשִּׁעוּר, נוֹדַע הַדָּבָר לְרַבִּי חִיָּא בַּר אַבָּא.”.
ג. אילו רבי אבהו לא היה פועל כפי שמתואר ב־“רַק אוֹתָם קְצָת תַּלְמִידֵי חֲכָמִים גְּדוֹלִים שֶׁמְּבִינִים בְּמַרְגָּלִיּוֹת!” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “אֲבָל מָה שֶׁהוּא בָּחַר לַעֲשׂוֹת – יַפְתִּיעַ אֶתְכֶם לְגַמְרֵי".” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. להלן עיבוד חינוכי, רגשי ומרתק על ענוותנותו העצומה של רבי אבהו , מעובד…
2. רַבִּי אֲבָהוּ הָיָה יוֹשֵׁב וּמְלַמֵּד אֶת אַנְשֵׁי עִירוֹ הֲלָכוֹת…
3. רַק קֹמֶץ שֶׁל תַּלְמִידֵי חֲכָמִים עֲמֻקִּים נִשְׁאֲרוּ אִתּוֹ.
4. וְהַסּוֹחֵר הַשֵּׁנִי פָּתַח דּוּכָן וְשָׁפַךְ שָׁם סִדְקִית .
5. בְּמָקוֹם לָשֶׁבֶת בְּמִשְׂרָדוֹ הַמְּפֹאָר, הוּא לָקַח עַל עַצְמוֹ…
6. עַכְשָׁו הַמִּתְנַדֵּב הָרִאשׁוֹן צָרִיךְ לְנַסּוֹת לְתָאֵר אֵיךְ…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “הֵם עָבְרוּ כֻּלָּם לַחֶדֶר הַשֵּׁנִי, כְּדֵי לִשְׁמֹעַ אֶת הַסִּפּוּרִים הַמְּתוּקִים שֶׁל רַבִּי חִיָּא!”.
מכתב: כתבו לרבי אבהו בעקבות “וַדַּאי שֶׁיָּרוּצוּ לְדוּכַן הַסִּדְקִית וְהַצַּעֲצוּעִים!" עָנָה רַבִּי חִיָּא.”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “בְּכָל בֹּקֶר וָעֶרֶב, הוּא עָזַב אֶת עִסּוּקָיו הָרַבִּים, וְהָלַךְ בְּעַצְמוֹ לָאכְסַנְיָה (הַמָּלוֹן) שֶׁל רַבִּי חִיָּא,…”.
דיווח: כתבו ידיעה קצרה שכותרתה “ענוותנותו”.

המחזה

רגע מן הסיפורהדמויותמה יאמרו או יעשו?
חֲכָמֵי הָעִיר הֶחְלִיטוּ שֶׁשְּׁנֵי הָרַבָּנִים יִמְסְרוּ שִׁעוּר בְּאוֹתוֹ הַזְּמַן, בִּשְׁנֵי…
רַק אוֹתָם קְצָת תַּלְמִידֵי חֲכָמִים גְּדוֹלִים שֶׁמְּבִינִים בְּמַרְגָּלִיּוֹת!

7. חיבור לחיים מתוך המעשה

א. מה אפשר ללמוד מן הדרך שבה רבי אבהו פעל באירוע “הוּא נִגַּשׁ לְרַבִּי חִיָּא, שָׂם אֶת יָדָיו עַל כְּתֵפָיו בְּחֹם, וְאָמַר לוֹ בְּחִיּוּךְ מָלֵא אַהֲבָה: "רַבִּי חִיָּא, אָחִי…”?
ב. תארו מצב מחיי הכיתה שבו אפשר ליישם את המסר של “ענוותנותו”.
