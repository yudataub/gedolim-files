כיצד ניצלו תלמידיו מהשודדים

דף עבודה בעקבות רבי עקיבא
שם: כיתה:

המעשה — הסיפור המלא

הנה מערך חינוכי וסיפורי מקיף, מרגש ועשיר, הבנוי בדיוק לפי בקשתך ולפי הכללים הנוירו-מדעיים של סטוריטלינג לילדים.

פְּתִיחָה: הָאוֹר שֶׁיּוֹצֵא מִבֵּית הַמִּדְרָשׁ

יְלָדִים, נַסּוּ לְדַמְיֵן אֶת בֵּית הַמִּדְרָשׁ שֶׁל גְּדוֹל דּוֹרֵנוּ, רַבִּי עֲקִיבָא. אַלְפֵי תַּלְמִידִים, יְשׁוּבִים עַל סַפְסְלֵי עֵץ וְעַל מַחְצָלוֹת, לוֹמְדִים מִפִּיו תּוֹרָה מִבֹּקֶר וְעַד הַשָּׁעוֹת הַקְּטַנּוֹת שֶׁל הַלַּיְלָה. קוֹל לִמּוּד הַתּוֹרָה הָיָה כָּל כָּךְ עָצְמָתִי, שֶׁהוּא נִשְׁמַע לְמֵרָחוֹק כְּמוֹ שְׁאָגָה שֶׁל אַרְיֵה קָדוֹשׁ. רַבִּי עֲקִיבָא לֹא לִמֵּד אוֹתָם רַק הֲלָכוֹת. הוּא לִמֵּד אוֹתָם מַשֶּׁהוּ גָּדוֹל יוֹתֵר: כֵּיצַד לִהְיוֹת אֲנָשִׁים חֲכָמִים, זְהִירִים, תְּמִימִים עִם ה', אֲבָל לֹא פְּתָיִים! הַתּוֹרָה צְרִיכָה לְלַמֵּד אוֹתָנוּ שֵׂכֶל וַהֲבָנָה בַּחַיִּים.

בְּבֹקֶר קַיִץ אֶחָד, הַשֶּׁמֶשׁ זָרְחָה וְהֵאִירָה אֶת הֲרָרֶיהָ שֶׁל אֶרֶץ יִשְׂרָאֵל. קְבוּצָה קְטַנָּה וּמְבֹרֶכֶת מִתַּלְמִידָיו שֶׁל רַבִּי עֲקִיבָא נֶאֶלְצָה לַעֲזֹב לְכַמָּה יָמִים אֶת בֵּית הַמִּדְרָשׁ. הָיָה עֲלֵיהֶם לִנְסֹעַ צָפוֹנָה, אֶל עִיר הַחוֹף שֶׁנִּקְרֵאת 'כְּזִיב' (הַיּוֹם דְּרוֹמִית לְרֹאשׁ הַנִּקְרָה), כְּדֵי לַעֲסֹק בְּמִצְוָה חֲשׁוּבָה שֶׁהִטִּיל עֲלֵיהֶם רַבָּם.

הֵם אָרְזוּ מְעַט מָזוֹן בְּתַרְמִילֵיהֶם – פִּתּוֹת יְבֵשׁוֹת, מְעַט זֵיתִים שְׁחוֹרִים, וְנֹאדוֹת מַיִם מֵעוֹר שֶׁל עִזִּים. הֵם חָגְרוּ אֶת מָתְנֵיהֶם, נָטְלוּ אֶת מַקְלוֹת הַהֲלִיכָה שֶׁלָּהֶם, וְיָצְאוּ לַדֶּרֶךְ.

הַהֲלִיכָה בַּדֶּרֶךְ הַשּׁוֹמֵמָה

הַדֶּרֶךְ הָיְתָה אֲרֻכָּה וּמְעַיֶּפֶת. רוּחַ חַמָּה נָשְׁבָה בְּפָנֵיהֶם. בְּכָל פַּעַם שֶׁהֵרִימוּ אֶת רַגְלֵיהֶם, תִּמְרוֹת קְטַנּוֹת שֶׁל אָבָק לָבָן פָּרְחוּ בָּאֲוִיר וְנִדְבְּקוּ לְסַנְדְּלֵיהֶם. רֵיחַ עָצוּם שֶׁל תְּאֵנִים שֶׁהִבְשִׁילוּ בַּשָּׂדוֹת הִתְעַרְבֵּב עִם מְלִיחוּת רֵיחַ הַיָּם הַתִּיכוֹן שֶׁנִּשְׁקַף מֵרָחוֹק.

אַךְ תַּלְמִידֵי רַבִּי עֲקִיבָא לֹא הִבִּיטוּ יָמִינָה וּשְׂמֹאלָה בְּבַטָּלָה.

תּוֹךְ כְּדֵי הֲלִיכָה, כְּשֶׁזֵּעָה נוֹטֶפֶת מִמִּצְחָם, הֵם חָזְרוּ בְּקוֹל נָעִים וּבְנִגּוּן עַל מַסֶּכְתּוֹת אֲשֶׁר לָמְדוּ בַּיְשִׁיבָה.

הַזְּמַן חָלַף, הַשֶּׁמֶשׁ הִתְחִילָה לָרֶדֶת, וְהֵם נִכְנְסוּ לְאֵזוֹר מְסֻכָּן: גֵּיא סְלָעִים מְבֻדָּד. מִשְּׁנֵי צִדֵּי הַשְּׁבִיל נִצְּבוּ אֲבָנִים עֲנָקִיּוֹת שֶׁהִטִּילוּ צֵל אָרֹךְ וְאָפֵל. צִיּוּץ הַצִּפּוֹרִים הִשְׁתַּתֵּק לְפֶתַע, וְשִׂיחֵי הַקּוֹצִים רִשְׁרְשׁוּ, אַף עַל פִּי שֶׁלֹּא הָיְתָה רוּחַ. הַשֶּׁקֶט הָיָה מַלְחִיץ.

הַתַּלְמִיד הַבּוֹגֵר שֶׁבַּחֲבוּרָה הֵרִים אֶת יָדוֹ וְסִמֵּן לַחֲבֵרָיו לַעֲצֹר. לִבָּם פָּעַם בְּחָזְקָה... מַשֶּׁהוּ לֹא קָשׁוּר לַנּוֹף זָז בֵּין הַסְּלָעִים.

אֵימַת הַשּׁוֹדְדִים!

לְפֶתַע, מִתּוֹךְ מַחֲבוֹא הַסְּלָעִים, קָפְצוּ הַחוּצָה חֲמִשָּׁה אֲנָשִׁים מְגֻדָּלִים, רְחָבִים כְּהָרִים וּפְנֵיהֶם מְכֻסּוֹת בִּבְדַלֵּי בַּד שְׁחוֹרִים. הָיוּ אֵלּוּ 'לִיסְטִים' – שׁוֹדְדֵי דְּרָכִים אַכְזָרִיִּים. בִּידֵיהֶם הֵם הֶחְזִיקוּ סַכִּינִים אֲרֻכּוֹת וְחַרְבוֹת נְחֹשֶׁת כְּבֵדוֹת שֶׁנָּצְצוּ בְּאוֹר הַשֶּׁמֶשׁ. עֵינֵיהֶם רָשְׁפוּ רֹעַ, וְהֵם חָסְמוּ אֶת הַשְּׁבִיל לַחֲלוּטִין.

תַּלְמִידֵי רַבִּי עֲקִיבָא נָשְׁמוּ נְשִׁימָה עֲמֻקָּה. הֵם לֹא הָיוּ לְמוּדֵי מִלְחָמוֹת, אַךְ זָכְרוּ אֶת תּוֹרַת רַבָּם – תּוֹרָה שֶׁל בִּטָּחוֹן בַּה'. הֵם לֹא אִבְּדוּ אֶת עֶשְׁתּוֹנוֹתֵיהֶם. הַשּׁוֹדְדִים הִתְקָרְבוּ בִּצְעָדִים מְאַיְּמִים. מְפַקֵּד הַשּׁוֹדְדִים עָמַד מוּל הַתַּלְמִידִים, סָקַר אוֹתָם מִכַּף רֶגֶל וְעַד רֹאשׁ, וְאָז נָחַר בְּקוֹל גַּס וְרָע:

"מִי אַתֶּם? וּלְאָן אַתֶּם הוֹלְכִים?!"

חָכְמָה עֶלְיוֹנָה שֶׁל תּוֹרָה

יְלָדִים, מָה אֲנַחְנוּ הָיִינוּ עוֹשִׂים בְּמַצָּב כָּזֶה? בְּוַדַּאי הָיִינוּ בּוֹכִים מִפַּחַד, אוֹ אוֹמְרִים בִּתְמִימוּת: "אֲנַחְנוּ הוֹלְכִים לְעִיר הַקְּרוֹבָה כְּזִיב, לִישׁוֹן הַלַּיְלָה. יֵשׁ לָנוּ קְצָת כֶּסֶף בַּשַּׂק, בְּבַקָּשָׁה אַל תִּפְגְּעוּ בָּנוּ!"

אֲבָל אִם הָיוּ אוֹמְרִים זֹאת, הַשּׁוֹדְדִים הָיוּ שׁוֹדְדִים אוֹתָם מִיָּד, אוֹ עָרְבִים לָהֶם בִּשְׁעֲרֵי 'כְּזִיב'.

תַּלְמִידֵי רַבִּי עֲקִיבָא זָכְרוּ אֶת הַכְּלָל: אָסוּר לְשַׁקֵּר בְּמֵזִיד, אַךְ מֻתָּר (וְאַף חוֹבָה) לִנְהֹג בְּחָכְמָה כְּדֵי לְהַצִּיל נְפָשׁוֹת.

אֶחָד מֵהַתַּלְמִידִים יָדַע שֶׁהַדֶּרֶךְ לָעִיר 'כְּזִיב' (שֶׁהִיא קְרוֹבָה), מַמְשִׁיכָה אַחַר כָּךְ לָעִיר הַגְּדוֹלָה 'עַכּוֹ' (שֶׁהִיא רְחוֹקָה מְאֹד וְלֹא הָיָה בְּכַוָּנָתָם לַעֲצֹר בָּהּ).

לְלֹא כָּל רַעַד בְּקוֹלוֹ, יָשִׁיר וּבָטוּחַ, הוּא חִיֵּךְ מְעַט לַשּׁוֹדֵד וְאָמַר: "אֲנַחְנוּ אֲנָשִׁים פְּשׁוּטִים, וְדַרְכֵּנוּ לְעַכּוֹ".

הַשּׁוֹדֵד הִבִּיט בָּהֶם בְּזִלְזוּל. הֵם לֹא נִרְאוּ עֲשִׁירִים כָּרֶגַע, אַךְ הוּא חָשַׁב בְּלִבּוֹ הָרַע: "לְעַכּוֹ הַדֶּרֶךְ עוֹד אֲרֻכָּה מְאֹד. הֵם בְּוַדַּאי יִישְׁנוּ הַלַּיְלָה בַּוָּאדִי הֶחָשׁוּךְ לִפְנֵי עַכּוֹ. נִתֵּן לָהֶם לְהַמְשִׁיךְ לָלֶכֶת, נַעֲקֹף אוֹתָם מִלְמַעְלָה דֶּרֶךְ הֶהָרִים, וְנֶאֱרֹב לָהֶם בַּלַּיְלָה לִפְנֵי עַכּוֹ, כְּשֶׁהֵם יִהְיוּ עֲיֵפִים. אָז נִשְׁדֹּד אוֹתָם הַרְבֵּה יוֹתֵר בְּקַלּוּת!".

הַשּׁוֹדֵד צָחַק צְחוֹק מְרֻשָּׁע וְסִמֵּן לַחֲבֵרָיו: "עִזְבוּ אוֹתָם עַכְשָׁו, תְּנוּ לָהֶם לַעֲבֹר".

הַפְּנִיָּה לִכְזִיב – נִצָּחוֹן תּוֹרָנִי

הַתַּלְמִידִים הִמְשִׁיכוּ לִצְעֹד קָדִימָה. הֵם הִרְגִּישׁוּ אֶת מַבְּטֵיהֶם שֶׁל הַשּׁוֹדְדִים הַנּוֹעֲצִים לָהֶם בַּגַּב. הֵם לֹא רָצוּ, אֶלָּא צָעֲדוּ בְּקוֹמָה זְקוּפָה וּבְשֶׁקֶט, תּוֹךְ שֶׁהֵם מִתְפַּלְּלִים בְּלִבָּם שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא יִשְׁמֹר אֶת דַּרְכָּם.

אַחֲרֵי שְׁעָתַיִם שֶׁל הֲלִיכָה נִגְלוּ לְפָנֵיהֶם חוֹמוֹת הָעִיר 'כְּזִיב'. הַשַּׁעַר מִתַּמֵּר מֵעֲלֵיהֶם, מָגֵן וּבָטוּחַ. הַתַּלְמִידִים (שֶׁבֶּאֱמֶת צָרִיכוּ רַק לְהַגִּיעַ לִכְזִיב) נִכְנְסוּ אֶל תּוֹךְ הָעִיר. הַדְּלָתוֹת הַכְּבֵדוֹת נִסְגְּרוּ מֵאֲחוֹרֵיהֶם. הֵם הִגִּיעוּ לְחוֹף מִבְטַחִים! הֵם מָצְאוּ מָקוֹם בְּבֵית הַמִּדְרָשׁ הַמְּקוֹמִי שֶׁל הָעִיר, שָׁתוּ מַיִם קָרִים מִן הַבְּאֵר, רָחֲצוּ אֶת פְּנֵיהֶם הַמְּאֻבָּקוֹת, פָּתְחוּ גְּוִילֵי תּוֹרָה וְהֵחֵלּוּ לִלְמֹד בְּשִׂמְחָה עֲצוּמָה וּבְהוֹדָאָה לַה'.

וּמָה קָרָה עִם הַשּׁוֹדְדִים?

בְּחֶשְׁכַת הַלַּיְלָה הֵם הִמְתִּינוּ עַל הַדֶּרֶךְ לְעַכּוֹ. הֵם חִכּוּ שָׁעָה, שְׁעָתַיִם... וְאִישׁ לֹא הִגִּיעַ. "לְאָן הֵם נֶעֶלְמוּ?" רָתְחוּ הַשּׁוֹדְדִים. "אוּלַי הֵם עָצְרוּ בִּכְזִיב?". בַּבֹּקֶר הַמָּחֳרָת, זְעוּמִים וְעַצְבָּנִיִּים, הֵם חָזְרוּ בְּצַעַד כָּבֵד עַד לָעִיר 'כְּזִיב' לְחַפֵּשׂ אֶת קָרְבְּנוֹתֵיהֶם.

הֵם הִסְתּוֹבְבוּ בָּרְחוֹבוֹת עַד שֶׁהִגִּיעוּ לְחַלּוֹנוֹת בֵּית הַמִּדְרָשׁ הַגָּדוֹל. הַשּׁוֹדְדִים הֵצִיצוּ פְנִימָה. וְהִנֵּה – הֵם רוֹאִים אֶת אוֹתָם צְעִירִים שֶׁפָּגְשׁוּ אֶתְמוֹל! רְגוּעִים, שְׁלֵוִים, וְלוֹמְדִים תּוֹרָה לְלֹא כָּל פַּחַד!

הַהוֹדָאָה בָּאֱמֶת שֶׁל הָרְשָׁעִים

הַשּׁוֹדֵד הָרָאשִׁי בָּלַע אֶת רֻקּוֹ. הוּא פִּתְאוֹם הֵבִין אֶת הַתַּכְסִיס.

"הָאַבְרֵכִים הָאֵלֶּה הִכְשִׁילוּ אוֹתָנוּ!" הוּא לָחַשׁ לַחֲבֵרָיו. "הֵם בְּכַוָּנָה עָנוּ 'עַכּוֹ', כִּי הֵם יָדְעוּ שֶׁהַדֶּרֶךְ לְעַכּוֹ עוֹבֶרֶת דֶּרֶךְ כְּזִיב. כָּךְ הֵם לֹא שִׁקְּרוּ דָּבָר, אֲבָל הֵם סִדְּרוּ אוֹתָנוּ. אֵיזוֹ חָכְמָה... אֵיזוֹ פִּקְּחוּת...".

הַשּׁוֹדֵד קָרָא לְאֶחָד הַתַּלְמִידִים מֵהַחַלּוֹן, בְּקוֹל שֶׁכְּבָר לֹא הָיָה מְאַיֵּם אֶלָּא מָלֵא בִּפְלִיאָה וּבְהַעֲרָצָה.

"תַּגִּידוּ לִי..." שָׁאַל הַשּׁוֹדֵד, "תַּלְמִידִים שֶׁל מִי אַתֶּם? מִי הָרַב שֶׁלִּמֵּד אֶתְכֶם לִהְיוֹת כָּל כָּךְ פִּקְּחִים וְעַרְמוּמִיִּים כְּנֶגֶד אֲנָשִׁים רָעִים?"

הַתַּלְמִיד סָגַר אֶת גְּוִיל הַתּוֹרָה, זָקַף אֶת קוֹמָתוֹ וְעָנָה בְּגַאֲוָה קְדוֹשָׁה:

"אֲנַחְנוּ – תַּלְמִידָיו שֶׁל רַבִּי עֲקִיבָא!"

שָׁתַק הַשּׁוֹדֵד לְרֶגַע, הִבִּיט בַּחֲבֵרָיו, וְאָז הִכְרִיז מִשְׁפָּט שֶׁנִּשְׁאַר מְהַדְהֵד בַּגְּמָרָא לְדוֹרוֹת עַד הַיּוֹם. הוּא – הָאָדָם הָרַע, בֵּרֵךְ אוֹתָם!

"אַשְׁרֵי רַבִּי עֲקִיבָא וְתַלְמִידָיו! שֶׁלֹּא פָּגַע בָּהֶם אָדָם רַע מֵעוֹלָם!"

הוּא הֵבִין שֶׁתּוֹרַת יִשְׂרָאֵל, כַּאֲשֶׁר לוֹמְדִים אוֹתָהּ בֶּאֱמֶת, נוֹתֶנֶת גַּם חָכְמָה אֵיךְ לִשְׁמֹר עַל הַחַיִּים. הַשּׁוֹדְדִים הִסְתּוֹבְבוּ וְהָלְכוּ לְדַרְכָּם כְּשֶׁהֵם מוּבָסִים לַחֲלוּטִין עַל יְדֵי כּוֹחָם שֶׁל בְּנֵי הַתּוֹרָה.

הַדְּמוּת (הֵם): תַּלְמִידֵי רַבִּי עֲקִיבָא הָיוּ בְּסַכָּנָה גְּדוֹלָה, אֲבָל הֵם לֹא נִבְהֲלוּ וְלֹא מָסְרוּ בִּתְמִימוּת מֵידָע שֶׁהָיָה מְסַכֵּן אֶת חַיֵּיהֶם. הֵם הִשְׁתַּמְּשׁוּ בְּ"חָכְמַת הַתּוֹרָה" כְּדֵי לַעֲנוֹת תְּשׁוּבָה חֲכָמָה וּמַצִּילַת חַיִּים שֶׁהִכְשִׁילָה אֶת הָרְשָׁעִים.

אֲנַחְנוּ (WE): הַרְבֵּה פְּעָמִים אֲנַחְנוּ נוֹטִים "לְהִתְלַהֵב" וְלְסַפֵּר אֶת כָּל מָה שֶׁאֲנַחְנוּ יֹודְעִים וּמְתַכְנְנִים לְאָדָם זָר, אֲפִלּוּ מִבְּלִי לְזַהוֹת אִם הָאָדָם הַזֶּה הוּא חָבֵר אוֹ שֶׁמָּא יֵשׁ לוֹ כַּוָּנוֹת רָעוֹת.

אַתָּה (YOU): מָחָר בַּבֹּקֶר – תִּזְכֹּר שֶׁ"תּוֹרָה הִיא תּוֹרַת חַיִּים". כְּשֶׁאַתָּה לוֹמֵד תּוֹרָה – הִיא בּוֹנָה לְךָ שֵׂכֶל חָזָק וְנָכוֹן. תִּהְיֶה זָהִיר! חֲשֹׁב פַּעֲמַיִם לִפְנֵי שֶׁאַתָּה עוֹנֶה לְאָדָם שֶׁאַתָּה לֹא מַכִּיר בָּרְחוֹב, וּשְׁמֹר עַל הָעֵינַיִם פְּקוּחוֹת כְּדֵי לְהִשָּׁמֵר מִסַּכָּנוֹת. ה' אוֹהֵב וְשׁוֹמֵר עַל לומדי התורה!

א. שְׁאֵלַת פְּתִיחָה (לְעוֹרֵר סַקְרָנוּת):

"יְלָדִים, אִם אָחֲדוּ מִכֶּם הָיָה הוֹלֵךְ הַיּוֹם בָּרְחוֹב, וּפִתְאוֹם נִגָּשׁ אֵלָיו אִישׁ מְפֻקְפָּק, רָחָב וּמַפְחִיד, וְשׁוֹאֵל אוֹתוֹ: 'אֵיפֹה אַתָּה גָּר וּלְאָן אַתָּה הוֹלֵךְ עַכְשָׁו?' – הַאִם נָכוֹן תָּמִיד לַעֲנוֹת לְכוּלָם אֶת הָאֱמֶת הַמְּדֻיֶּקֶת וְלִמְסֹר הַכֹּל?

הַיּוֹם נִלְמַד מִתַּלְמִידָיו שֶׁל רַבִּי עֲקִיבָא – אֵיךְ חָכְמָה שֶׁל תּוֹרָה נִצְּחָה קְבוּצַת שׁוֹדְדִים וְחָסְכָה שְׁפִיכוּת דָּמִים!"

ד. פְּעִילוּת מְסַכֶּמֶת / הַפְעָלָה קְצָרָה:

מִשְׂחַק תַּפְקִידִים – "חָכְמָה דִּקְדֻשָּׁה"

הַמּוֹרֶה כּוֹתֵב כַּמָּה מַצָּבִים שֶׁל דִּלֶמָּה יוֹמְיוֹמִית לְיֶלֶד בְּכִתָּה ג', מַזְמִין כָּל פַּעַם יֶלֶד אֶחָד וְשׁוֹאֵל אוֹתוֹ שְׁאֵלָה שֶׁל אָדָם רָע/זָר. הַיֶּלֶד צָרִיךְ לְהַפְגִּין 'חָכְמַת רַבִּי עֲקִיבָא', לַחְשֹׁב שְׁנִיָּה וְלַעֲנוֹת בְּתַחְכּוּם וּבִזְהִירוּת שֶׁשּׁוֹמֶרֶת עָלָיו, מִבְּלִי לְשַׁקֵּר וּמִבְּלִי לִמְסֹר סוֹדוֹת.

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “הנה מערך חינוכי וסיפורי מקיף, מרגש ועשיר, הבנוי בדיוק לפי בקשתך ולפי הכללים הנוירו-מדעיים של סטוריטלינג לילדים.” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “בְּכָל פַּעַם שֶׁהֵרִימוּ אֶת רַגְלֵיהֶם, תִּמְרוֹת קְטַנּוֹת שֶׁל אָבָק לָבָן פָּרְחוּ בָּאֲוִיר וְנִדְבְּקוּ לְסַנְדְּלֵיהֶם.”?
ג. כיצד הגיב רבי עקיבא למצב המתואר: “הֵם לֹא הָיוּ לְמוּדֵי מִלְחָמוֹת, אַךְ זָכְרוּ אֶת תּוֹרַת רַבָּם – תּוֹרָה שֶׁל בִּטָּחוֹן בַּה'.”?
ד. מה השתנה בין “נִתֵּן לָהֶם לְהַמְשִׁיךְ לָלֶכֶת, נַעֲקֹף אוֹתָם מִלְמַעְלָה דֶּרֶךְ הֶהָרִים, וְנֶאֱרֹב לָהֶם בַּלַּיְלָה לִפְנֵי…” לבין “הֵם בְּכַוָּנָה עָנוּ 'עַכּוֹ', כִּי הֵם יָדְעוּ שֶׁהַדֶּרֶךְ לְעַכּוֹ עוֹבֶרֶת דֶּרֶךְ כְּזִיב.”?
ה. כיצד הגיע הסיפור אל האירוע: “כְּשֶׁאַתָּה לוֹמֵד תּוֹרָה – הִיא בּוֹנָה לְךָ שֵׂכֶל חָזָק וְנָכוֹן.”?

□ לְלֹא כָּל רַעַד בְּקוֹלוֹ, יָשִׁיר וּבָטוּחַ, הוּא חִיֵּךְ מְעַט לַשּׁוֹדֵד וְאָמַר: "אֲנַחְנוּ אֲנָשִׁים פְּשׁוּטִים, וְדַרְכֵּנוּ לְעַכּוֹ".
□ הנה מערך חינוכי וסיפורי מקיף, מרגש ועשיר, הבנוי בדיוק לפי בקשתך ולפי הכללים הנוירו-מדעיים של סטוריטלינג לילדים.
□ הַיֶּלֶד צָרִיךְ לְהַפְגִּין 'חָכְמַת רַבִּי עֲקִיבָא', לַחְשֹׁב שְׁנִיָּה וְלַעֲנוֹת בְּתַחְכּוּם וּבִזְהִירוּת שֶׁשּׁוֹמֶרֶת עָלָיו, מִבְּלִי לְשַׁקֵּר…
□ הַזְּמַן חָלַף, הַשֶּׁמֶשׁ הִתְחִילָה לָרֶדֶת, וְהֵם נִכְנְסוּ לְאֵזוֹר מְסֻכָּן: גֵּיא סְלָעִים מְבֻדָּד.
□ הַשּׁוֹדֵד קָרָא לְאֶחָד הַתַּלְמִידִים מֵהַחַלּוֹן, בְּקוֹל שֶׁכְּבָר לֹא הָיָה מְאַיֵּם אֶלָּא מָלֵא בִּפְלִיאָה וּבְהַעֲרָצָה.

לפני ואחרי

מה קרה לפני: “לְלֹא כָּל רַעַד בְּקוֹלוֹ, יָשִׁיר וּבָטוּחַ, הוּא חִיֵּךְ מְעַט לַשּׁוֹדֵד וְאָמַר: "אֲנַחְנוּ אֲנָשִׁים פְּשׁוּטִים, וְדַרְכֵּנוּ לְעַכּוֹ".”?
מה קרה אחרי: “הַשּׁוֹדֵד קָרָא לְאֶחָד הַתַּלְמִידִים מֵהַחַלּוֹן, בְּקוֹל שֶׁכְּבָר לֹא הָיָה מְאַיֵּם אֶלָּא מָלֵא בִּפְלִיאָה וּבְהַעֲרָצָה.”?

סיבה ותוצאה מתוך המעשה

האירוע הנתוןמה גרם לו?מה הייתה תוצאתו?
צִיּוּץ הַצִּפּוֹרִים הִשְׁתַּתֵּק לְפֶתַע, וְשִׂיחֵי הַקּוֹצִים רִשְׁרְשׁוּ, אַף עַל פִּי שֶׁלֹּא הָיְתָה רוּחַ.
הַתַּלְמִידִים (שֶׁבֶּאֱמֶת צָרִיכוּ רַק לְהַגִּיעַ לִכְזִיב) נִכְנְסוּ אֶל תּוֹךְ הָעִיר.

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “כיצד ניצלו תלמידיו מהשודדים” על רבי עקיבא? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “תַּלְמִידֵי רַבִּי עֲקִיבָא זָכְרוּ אֶת הַכְּלָל: אָסוּר לְשַׁקֵּר בְּמֵזִיד, אַךְ מֻתָּר (וְאַף חוֹבָה) לִנְהֹג בְּחָכְמָה כְּדֵי לְהַצִּיל נְפָשׁוֹת.”.
ג. אילו רבי עקיבא לא היה פועל כפי שמתואר ב־“הֵם בְּכַוָּנָה עָנוּ 'עַכּוֹ', כִּי הֵם יָדְעוּ שֶׁהַדֶּרֶךְ לְעַכּוֹ עוֹבֶרֶת דֶּרֶךְ כְּזִיב.” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “כְּשֶׁאַתָּה לוֹמֵד תּוֹרָה – הִיא בּוֹנָה לְךָ שֵׂכֶל חָזָק וְנָכוֹן.” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. הנה מערך חינוכי וסיפורי מקיף, מרגש ועשיר, הבנוי בדיוק לפי בקשתך ולפי…
2. רֵיחַ עָצוּם שֶׁל תְּאֵנִים שֶׁהִבְשִׁילוּ בַּשָּׂדוֹת הִתְעַרְבֵּב עִם…
3. בְּוַדַּאי הָיִינוּ בּוֹכִים מִפַּחַד, אוֹ אוֹמְרִים בִּתְמִימוּת:…
4. הֵם לֹא רָצוּ, אֶלָּא צָעֲדוּ בְּקוֹמָה זְקוּפָה וּבְשֶׁקֶט, תּוֹךְ…
5. הוּא הֵבִין שֶׁתּוֹרַת יִשְׂרָאֵל, כַּאֲשֶׁר לוֹמְדִים אוֹתָהּ…
6. הַיֶּלֶד צָרִיךְ לְהַפְגִּין 'חָכְמַת רַבִּי עֲקִיבָא', לַחְשֹׁב…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “הֵם לֹא הָיוּ לְמוּדֵי מִלְחָמוֹת, אַךְ זָכְרוּ אֶת תּוֹרַת רַבָּם – תּוֹרָה שֶׁל בִּטָּחוֹן בַּה'.”.
מכתב: כתבו לרבי עקיבא בעקבות “הַתַּלְמִידִים (שֶׁבֶּאֱמֶת צָרִיכוּ רַק לְהַגִּיעַ לִכְזִיב) נִכְנְסוּ אֶל תּוֹךְ הָעִיר.”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “הַשּׁוֹדְדִים הִסְתּוֹבְבוּ וְהָלְכוּ לְדַרְכָּם כְּשֶׁהֵם מוּבָסִים לַחֲלוּטִין עַל יְדֵי כּוֹחָם שֶׁל בְּנֵי הַתּוֹרָה.”.
דיווח: כתבו ידיעה קצרה שכותרתה “כיצד ניצלו תלמידיו מהשודדים”.

המחזה

רגע מן הסיפורהדמויותמה יאמרו או יעשו?
צִיּוּץ הַצִּפּוֹרִים הִשְׁתַּתֵּק לְפֶתַע, וְשִׂיחֵי הַקּוֹצִים רִשְׁרְשׁוּ, אַף עַל פִּי שֶׁלֹּא…
הֵם בְּכַוָּנָה עָנוּ 'עַכּוֹ', כִּי הֵם יָדְעוּ שֶׁהַדֶּרֶךְ לְעַכּוֹ עוֹבֶרֶת דֶּרֶךְ כְּזִיב.

7. חיבור לחיים מתוך המעשה

א. מה אפשר ללמוד מן הדרך שבה רבי עקיבא פעל באירוע “נִתֵּן לָהֶם לְהַמְשִׁיךְ לָלֶכֶת, נַעֲקֹף אוֹתָם מִלְמַעְלָה דֶּרֶךְ הֶהָרִים, וְנֶאֱרֹב לָהֶם בַּלַּיְלָה לִפְנֵי עַכּוֹ,…”?
ב. תארו מצב מחיי הכיתה שבו אפשר ליישם את המסר של “כיצד ניצלו תלמידיו מהשודדים”.
