הדינרים של בני האחות

דף עבודה בעקבות רבי שמעון בר יוחאי (רשב"י)
שם: כיתה:

המעשה — הסיפור המלא

להלן מערך חינוכי, סיפורי ומרגש על התנא הקדוש רבי שמעון בר יוחי (רשב"י) וסוד הדינרים . הסיפור מדגיש את כוחה של מצוות הצדקה, ומעל הכול – את החשיבות של עשיית מצווה "לשם שמים", מבלי לחפש פרס או תועלת אישית. הסיפור מותאם בדיוק לבנים בכיתות ב'-ד' בחינוך הדתי-לאומי, ערוך עם תיאורים חושיים, מתח רגשי וניקוד מלא.

פְּתִיחָה: הַחֲלוֹם שֶׁל גְּדוֹל הַדּוֹר

יְלָדִים, שֶׁקֶט גָּמוּר עָטַף אֶת בֵּית הַמִּדְרָשׁ. הָיָה זֶה אֶמְצַע הַלַּיְלָה בְּאֶרֶץ יִשְׂרָאֵל. רֵיחַ עָדִין שֶׁל שֶׁמֶן זַיִת זַךְ שֶׁנִּשְׂרַף בַּמְּנוֹרָה מִלֵּא אֶת הַחֶדֶר הַקָּטָן.

לְיַד הַשֻּׁלְחָן, רָכוּן עַל הַסְּפָרִים הַקְּדוֹשִׁים, יָשַׁב רַבִּי שִׁמְעוֹן בַּר יוֹחַאי. הַתַּנָּא הַקָּדוֹשׁ שֶׁהָיָה מְלֻמָּד בְּנִסִּים. עֵינָיו נֶעֶצְמוּ לְרֶגַע מִתּוֹךְ עֲיֵפוּת, וְהוּא שָׁקַע בְּתַרְדֵּמָה קְצָרָה.

וְהִנֵּה, בַּחֲלוֹמוֹ, הוּא שׁוֹמֵעַ בְּרוּחַ הַקֹּדֶשׁ קוֹל כָּבֵד וְקָשֶׁה. הַקּוֹל הִכְרִיז עַל גְּזֵרָה שֶׁנִּגְזְרָה בַּשָּׁמַיִם: "מָחָר בַּבֹּקֶר, יוֹצִיא קֵיסַר רוֹמִי פְּקֻדָּה. הָרוֹמָאִים הָרְשָׁעִים יִקְנְסוּ אֶת בְּנֵי אֲחוֹתוֹ שֶׁל רַבִּי שִׁמְעוֹן בִּסְכוּם עָצוּם – שֵׁשׁ מֵאוֹת דִּינְרֵי זָהָב! וְאִם לֹא יְשַׁלְּמוּ – יִשְׁלְכוּ אוֹתָם לְבֵית הָאֲסוּרִים!".

רַבִּי שִׁמְעוֹן הִתְעוֹרֵר בְּבֶהָלָה. שֵׁשׁ מֵאוֹת דִּינְרֵי זָהָב! זֶהוּ הוֹן עָתֵק. מַטְבְּעוֹת זָהָב נוֹצְצִים וּכְבֵדִים, הָרְכוּשׁ שֶׁל בְּנֵי אֲחוֹתוֹ הַצַּדִּיקִים. הַלֵּב שֶׁל רַבִּי שִׁמְעוֹן דָּפַק. הוּא הֵבִין שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא גִּלָּה לוֹ אֶת הַסּוֹד כְּדֵי שֶׁיּוּכַל לְהַצִּיל אוֹתָם, אַךְ הָיָה כָּאן מִבְחָן עָצוּם.

הוּא יָכוֹל הָיָה לָרוּץ אֲלֵיהֶם וְלוֹמַר: "שִׁמְעוּ! הָרוֹמָאִים מַגִּיעִים! תַּחְבִּיאוּ אֶת הַכֶּסֶף!". אוֹ לוֹמַר לָהֶם: "תִּתְרְמוּ מַהֵר שֵׁשׁ מֵאוֹת דִּינָרִים לִצְדָקָה כְּדֵי שֶׁהַכֶּסֶף לֹא יִלָּקַח מִכֶּם". אֲבָל אִם כָּךְ יַעֲשׂוּ – הֵם לֹא יִתְּנוּ מִתּוֹךְ אַהֲבַת ה', אֶלָּא מִתּוֹךְ פַּחַד מֵהָרוֹמָאִים!

הַשְּׁלִיחוּת שֶׁל בְּנֵי הָאָחוֹת

בַּבֹּקֶר הַמָּחֳרָת, הַשֶּׁמֶשׁ חִמְּמָה אֶת רְחוֹבוֹת אֶרֶץ יִשְׂרָאֵל. רַבִּי שִׁמְעוֹן קָרָא לִשְׁנֵי בְּנֵי אֲחוֹתוֹ לְבֵית הַמִּדְרָשׁ. הֵם הִגִּיעוּ, נַעֲלֵיהֶם מְכֻסּוֹת מְעַט בַּאֲבַק הַדְּרָכִים, וְקָדוּ קִידָה עֲמֻקָּה לְדוֹדָם הַגָּדוֹל.

"דּוֹדִי, קָרָאתָ לָנוּ?" שָׁאֲלוּ.

"כֵּן, בָּנַי הַיְקָרִים," מְשַׂעֲרוֹ וּפָנָיו זָרַח אוֹר שֶׁל קְדֻשָּׁה. "יֵשׁ לִי בַּקָּשָׁה חֲשׁוּבָה אֲלֵיכֶם. אֲנִי רוֹצֶה שֶׁמֵּהַיּוֹם תִּהְיוּ 'גַּבָּאֵי צְדָקָה'. עֲבֹרוּ בֵּין אֲנָשִׁים, אִסְפוּ כֶּסֶף לָעֲנִיִּים שֶׁאֵין לָהֶם לֶחֶם לֶאֱכֹל וְלֹא כְּסוּת לְהִתְכַּסּוֹת. וְכַמּוּבָן, הַתְחִילוּ קֹדֶם כֹּל מֵהַכֶּסֶף שֶׁלָּכֶם, מִכַּסְפִּיכֶם שֶׁלָּכֶם לִצְדָקָה... וְכִתְבוּ לָכֶם עַל דַּף כָּל סְכוּם שֶׁאַתֶּם נוֹתְנִים".

בְּנֵי הָאָחוֹת, שֶׁהָיוּ תַּלְמִידֵי חֲכָמִים טְהוֹרִים, לֹא שָׁאֲלוּ שְׁאֵלוֹת. "אִם גְּדוֹל הַדּוֹר דּוֹדֵנוּ מְבַקֵּשׁ – כָּךְ נַעֲשֶׂה!".

יְלָדִים, אֵיזוֹ מְסִירוּת! בְּכָל יוֹם הֵם לָקְחוּ מִקֻּפָּתָם, מַטְבֵּעַ אַחַר מַטְבֵּעַ שֶׁל זָהָב. צִילְל... צְלִיל... נִשְׁמַע נְקִישׁוֹת הַזָּהָב בְּתוֹךְ קֻפַּת הַצְּדָקָה. הֵם חִלְּקוּ לִיתוֹמִים, הֵבִיאוּ לֶחֶם וּבָשָׂר לְאַלְמָנוֹת, דָּאֲגוּ לִתְלָמִידֵי חֲכָמִים עֲנִיִּים. עָבְרוּ שָׁבוּעוֹת, וְהֵם רָשְׁמוּ בַּפִּנְקָס: הַיּוֹם נָתַנּוּ 50 דִּינָרִים... הַיּוֹם נָתַנּוּ 100 דִּינָרִים...

עַד שֶׁהִגִּיעוּ לִסְכוּם שֶׁל חֲמֵשׁ מֵאוֹת תִּשְׁעִים וְאַרְבָּעָה (594) דִּינְרֵי זָהָב. הָיָה חָסֵר לָהֶם כִּמְעַט סְכוּם שֶׁל שִׁשָּׁה דִּינָרִים שְׁלֵמִים.

בְּבֵית הָאֲסוּרִים הָרוֹמָאִי

וְאָז – זֶה קָרָה.

עֶרֶב אֶחָד, הֵם יָשְׁבוּ בְּבֵיתָם וְסָעֲדוּ מְעַט לֶחֶם וְשֶׁמֶן. פִּתְאוֹם – בּוּם! בּוּם! בּוּם!

דְּפִיקוֹת אֵימָה הִרְעִידוּ אֶת דֶּלֶת הָעֵץ הַכְּבֵדָה! הַדֶּלֶת נִפְרְצָה וּפְנִימָה פָּרְצוּ חַיָּלִים רוֹמָאִים עִם שִׁרְיוֹנֵי בַּרְזֶל מַבְרִיקִים וְחֲרָבוֹת שְׁלוּפוֹת! הַקַּסְדּוֹת שֶׁלָּהֶם נָצְצוּ בְּאוֹר הַמְּנוֹרָה.

"קֻמוּ מִיָּד!" שָׁאֲג הַקָּצִין הָרוֹמָאִי, קוֹלוֹ הִדְהֵד בַּחֶדֶר כְּמוֹ רַעַם. "יֵשׁ עֲלֵיכֶם הַלְשָׁנָה שֶׁאַתֶּם סוֹחֲרִים בְּמֶשִׁי לְלֹא רִשָּׁיוֹן! הַקֵּיסָר אוֹסֵר אֶתְכֶם!".

הַחַיָּלִים לָפְתוּ אֶת יְדֵיהֶם שֶׁל הָאַחְיָנִים וְכָבְלוּ אוֹתָם בְּשַׁרְשְׁרָאוֹת בַּרְזֶל – קְלַאנְק... קְלַאנְק... הַבַּרְזֶל הַקַּר נָגַע בְּעוֹרָם, וְהֵם נִגְרְרוּ הַחוּצָה לְתוֹךְ הַחֹשֶׁךְ.

הֵם הֻשְׁלְכוּ לְתוֹךְ תָּא מַעֲצָר תַּת-קַרְקָעִי בְּבֵית הָאֲסוּרִים. זֶה הָיָה מָקוֹם נוֹרָא. קַר, רָטֹב וְחָשׁוּךְ כִּמְעַט לְגַמְרֵי, רַק קֶרֶן אוֹר חִוֶּרֶת נִכְנְסָה מֵחֲרַךְ קָטָן בַּתִּקְרָה. רֵיחַ שֶׁל טְחַב וַאֲדָמָה רְטֻבָּה מִלֵּא אֶת הָאֲוִיר. בְּנֵי הָאָחוֹת יָשְׁבוּ עַל הָרִצְפָּה הַקָּרָה וְהֵחֵלּוּ לְהִתְפַּלֵּל לְבוֹרֵא עוֹלָם. הֵם הָיוּ בְּפַחַד גָּדוֹל, יוֹדְעִים שֶׁהָרוֹמָאִים אַכְזָרִיִּים.

הַיְּשׁוּעָה וְהַסּוֹד הַגָּדוֹל

הַשְּׁמוּעָה מִהֲרָה לְהַגִּיעַ לְבֵית הַמִּדְרָשׁ. עֵינָיו שֶׁל רַבִּי שִׁמְעוֹן בַּר יוֹחַאי בָּרְקוּ. הוּא הֵבִין שֶׁהַגְּזֵרָה שֶׁרָאָה בַּחֲלוֹם יָרְדָה לָעוֹלָם! הָרוֹמָאִים הִגִּיעוּ לִגְבּוֹת אֶת כַּסְפָּם.

הוּא לָקַח אֶת גְּלִימָתוֹ, וּצְעָדָיו הָיוּ בְּטוּחִים וְשְׁקֵטִים בַּדֶּרֶךְ לְבֵית הַמִּדְרָשׁ. "הַשְּׁגָחַת ה' הִיא עֶלְיוֹנָה," מִלְמֵל לְעַצְמוֹ בְּבִטָּחוֹן אֵיתָן. הוּא הִגִּיעַ לְבֵית הָאֲסוּרִים, מַרְאֵהוּ הַקָּדוֹשׁ וְהֶהָדָר שֶׁנָּטַף מִמֶּנּוּ גָּרְמוּ לַשּׁוֹמְרִים הָרוֹמָאִים לָתֵת לוֹ רְשׁוּת לְדַבֵּר עִם הָאֲסִירִים מִבַּעַד לַסּוֹרָגִים.

הוּא נִגַּשׁ לַתָּא הֶחָשׁוּךְ.

"בָּנַי," לָחַשׁ רַבִּי שִׁמְעוֹן, קוֹלוֹ רַךְ וּמַרְגִּיעַ, תּוֹמֵךְ כְּקִיר בַּרְזֶל.

"דּוֹדֵנוּ הַקָּדוֹשׁ!" זָעֲקוּ שְׁנֵיהֶם בְּדִמְעוֹת תִּקְוָה, אֲנַחְנוּ בְּסַכָּנָה עֲצוּמָה! קָצִין מְרֻשָּׁע דּוֹרֵשׁ כֶּסֶף מְפֻבְרָק וְכָל רְכוּשֵׁנוּ בְּסַכָּנָה!".

רַבִּי שִׁמְעוֹן שָׁאַל בְּנַחַת: "הַגִּידוּ לִי, כַּמָּה כֶּסֶף נְתַתֶּם לִצְדָקָה בַּחֳדָשִׁים הָאַחֲרוֹנִים? הַאִם רְשַׁמְתֶּם הַכֹּל כְּפִי שֶׁבִּקַּשְׁתִּי?".

הֵם הוֹצִיאוּ אֶת הַפִּנְקָס הַקָּטָן שֶׁהָיָה בְּכִיסָם, חִשְּׁבוּ שׁוּב וְאָמְרוּ: "חֲמֵשׁ מֵאוֹת תִּשְׁעִים וְאַרְבָּעָה (594) דִּינָרִים!".

חִיּוּךְ רָחָב וְזוֹרֵחַ עָלָה עַל פָּנָיו שֶׁל רַבִּי שִׁמְעוֹן!

"אַל תִּפְחֲדוּ כְּלָל!" הוּא קָרָא.

הוּא שָׁלַח יָד אֶל שַׂקִּיק קָטָן שֶׁהֵבִיא עִמּוֹ, וְהוֹצִיא בְּדִיּוּק... שִׁשָּׁה דִּינְרֵי זָהָב (וְשִׁשָּׁה דִּינָרִים הוּא סְכוּם נִכְבָּד, אֲבָל מְעַט מְאֹד לְעֻמַּת שֵׁשׁ מֵאוֹת). הוּא קָרָא לְשַׂר בֵּית הָאֲסוּרִים, שִׁלְשֵׁל אֶת הַמַּטְבְּעוֹת לְתוֹךְ יָדוֹ. "זֶהוּ קְנָס קָטָן עֲבוּר הַשִּׁחְרוּר שֶׁלָּהֶם. תְּשַׁחְרֵר אוֹתָם. הַהַלְשָׁנָה הָיְתָה תְּלוּת-שָׁוְא!". הַקָּצִין, שֶׁעֵינָיו נָצְצוּ לְמַרְאֵה הַזָּהָב, לָקַח אֶת הַכֶּסֶף, פָּתַח אֶת דֶּלֶת הַבַּרְזֶל וְשִׁחְרֵר אֶת בְּנֵי הָאָחוֹת מִיָּד!

הָעֲשִׂיָּה לְשֵׁם הַמִּצְוָה

כְּשֶׁיָּצְאוּ מֵאֲפֵלַת הַצִּינוֹק אֶל אוֹר הַשֶּׁמֶשׁ, שָׁאֲלוּ בְּנֵי הָאָחוֹת בִּפְלִיאָה: "דּוֹדֵנוּ הַגָּדוֹל, רַבִּי שִׁמְעוֹן! עַכְשָׁו אֲנַחְנוּ מְבִינִים. אַתָּה יָדַעְתָּ! יָדַעְתָּ שֶׁעוֹמְדִים לְהַחְרִים מֵאִתָּנוּ 600 דִּינָרִים! הֲרֵי נָתַנּוּ 594 לִצְדָקָה, וְאַתָּה שִׁלַּמְתָּ עַכְשָׁו אֶת הַ-6 שֶׁחָסְרוּ בְּדִיּוּק כְּדֵי לְהַשְׁלִים אֶת הַקְּנָס! זֶה פֶּלֶא נֶשְׂגָּב! אֲבָל... יֵשׁ לָנוּ קֻשְׁיָה: מַדּוּעַ לֹא אָמַרְתָּ לָנוּ מֵרֹאשׁ לָתֵת 600 דִּינָרִים וְזֶהוּ? לָמָּה שָׁלַחְתָּ אוֹתָנוּ לֶאֱסֹף צְדָקָה בְּלִי לְגַלּוֹת אֶת הַמִּסְפָּר? הָיִינוּ נִצָּלִים מִפַּחַד מָוֶת שֶׁל שְׁעוֹת הַמַּעֲצָר!".

עָצַר רַבִּי שִׁמְעוֹן בַּר יוֹחַאי אֶת צְעָדָיו, הִסְתַּכֵּל עֲלֵיהֶם מִתּוֹךְ אַהֲבָה טְהוֹרָה, וְאָמַר לָהֶם אֶת אֶחָד מֵהַסּוֹדוֹת הַגְּדוֹלִים שֶׁל הַתּוֹרָה:

"בָּנַי הָאֲהוּבִים. אִלּוּ הָיִיתִי בָּא אֲלֵיכֶם מֵרֹאשׁ וְאוֹמֵר 'תִּתְרְמוּ 600 דִּינָרִים וְכָךְ תִּנָּצְלוּ מֵהָרוֹמָאִים', הֱיִיתֶם אוֹסְפִים אֶת הַכֶּסֶף מַהֵר... אֲבָל לָמָּה הֱיִיתֶם אוֹסְפִים? הֱיִיתֶם אוֹסְפִים בִּשְׁבִיל לְהַצִּיל אֶת הַכֶּסֶף שֶׁלָּכֶם, כְּדֵי שֶׁלֹּא יִפֹּל בִּידֵי הַמַּלְכוּת הָרְשָׁעָה! הַמִּצְוָה שֶׁלָּכֶם הָיְתָה כְּדֵי לִשְׁמֹר עַל עַצְמְכֶם.

אֲבָל כַּאֲשֶׁר שָׁלַחְתִּי אֶתְכֶם לְחַלֵּק צְדָקָה לִיתוֹמִים וְאַלְמָנוֹת, מִבְּלִי לְגַלּוֹת לָכֶם עַל הָעֹנֶשׁ הַמִּתְקָרֵב... כָּל מַטְבֵּעַ שֶׁנְּתַתֶּם, נְתַתֶּם לִשְׁמָהּ ! נְתַתֶּם לְשֵׁם שָׁמַיִם! עֲשִׂיתֶם אֶת הַמִּצְוָה מִתּוֹךְ חֶסֶד וְאַהֲבַת הַקָּדוֹשׁ בָּרוּךְ הוּא. וְזוֹהִי, צְדָקָה אֲמִתִּית!".

א. שְׁאֵלַת פְּתִיחָה (לִשְׁאֹל לִפְנֵי שֶׁמַּתְחִילִים):

"יְלָדִים, קָרָה לָכֶם פַּעַם שֶׁעֲזַרְתֶּם בַּבַּיִת הֲמוֹן... וְאַף אֶחָד לֹא רָאָה, וְלֹא אָמַר כָּל הַכָּבוֹד, וְלֹא הֵבִיא פְּרָס? אֵיךְ זֶה מַרְגִּישׁ? הַיּוֹם אֲנַחְנוּ הוֹלְכִים לִשְׁמֹעַ מֵרַבִּי שִׁמְעוֹן בַּר יוֹחַאי – מַהוּ שְׁוִויוֹ שֶׁל הַמַּעֲשֶׂה שֶׁעוֹשִׂים אוֹתוֹ בְּלִי לְחַכּוֹת לְתְמוּרָה!"

ב. הֶעָרוֹת בִּצּוּעַ וּסְטוֹרִיטֶלִינְג:

ג. הַמְלָצוֹת לְמַצֶּגֶת שְׁקֻפִּיּוֹת (מְכֻבֶּדֶת וּצְנוּעָה):

ד. פְּעִילוּת מְסַכֶּמֶת מֻמְלֶצֶת:

"קֻפַּת חֶסֶד בַּסֵּתֶר" (אוֹ: עֲשִׂיָּה לְשֵׁם שָׁמַיִם)

הָכֵן מֵרֹאשׁ קֻפְסָה יָפָה עִם חֶרֶךְ סָגוּר. תֵּן לְכָל יֶלֶד פֶּתֶק וְעִפָּרוֹן. בַּקֵּשׁ מֵהַיְלָדִים לִכְתֹּב אוֹ לְצַיֵּר עַל הַפֶּתֶק מִצְוָה קְטַנָּה אוֹ טוֹבָה שֶׁהֵם עָשׂוּ בַּיּוֹם-יוֹמַיִם הָאַחֲרוֹנִים בְּלִי שֶׁאַף אֶחָד יָדַע מִזֶּה וּבְלִי קַבָּלַת פְּרָס . הַיְלָדִים מְקַפְּלִים אֶת הַפֶּתֶק וּמַשְׁלִיכִים לַקֻּפָּה. בסיום, שקשק את הקופה וְאָמַר: "כָּאן בַּקֻּפְסָה הַזּוֹ מִסְתַּתֵּר 'הַזָּהָב שֶׁל רַבִּי שִׁמְעוֹן בַּר יוֹחַאי' – זָהָב שֶׁל מִצְווֹת לִשְׁמָן!".

מקורות: מִדְרַשׁ רַבָּה (וַיִּקְרָא רַבָּה ל"ד, י"ב): סִפּוּר זֶה מְבֻסָּס עַל הַמִּדְרַשׁ הַמְתָאֵר כֵּיצַד רַשְׁבִּ"י רָאָה בַּחֲלוֹם לֵיל רֹאשׁ הַשָּׁנָה כִּי עַל בְּנֵי אֲחוֹתוֹ נִגְזַר קְנָס שֶׁל שֵׁשׁ מֵאוֹת דִּינָרִים שֶׁיִּלָּקַח עַל יְדֵי הַמַּלְכוּת, וְעַל עֲצָתוֹ לַהֲפֹךְ אוֹתָם לְגַבָּאֵי צְדָקָה כְּדֵי שֶׁיִּתְּנוּ מִמָּמוֹנָם "לִשְׁמָהּ". פְּסִיקְתָּא דְּרַב כָּהֲנָא (פֶּרֶק י'): תֵּאוּר דּוֹמֶה שֶׁל אוֹתוֹ מַעֲשֶׂה גְּבוּרָה וְאַהֲבַת הַצְּדָקָה שֶׁל רַשְׁבִּ"י וּבְנֵי אֲחוֹתוֹ. סִפְרֵי מַעֲשִׂיּוֹת: ("כֹּה עָשׂוּ חֲכָמֵינוּ" – כֶּרֶךְ ג', וְסֵפֶר הָאַגָּדָה עַל אוֹתָהּ סוּגְיָה שֶׁל "עֲשִׂיָּה לְשֵׁם שָׁמַיִם").

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “להלן מערך חינוכי, סיפורי ומרגש על התנא הקדוש רבי שמעון בר יוחי (רשב"י) וסוד הדינרים .” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “הוּא הֵבִין שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא גִּלָּה לוֹ אֶת הַסּוֹד כְּדֵי שֶׁיּוּכַל לְהַצִּיל אוֹתָם, אַךְ הָיָה כָּאן מִבְחָן עָצוּם.”?
ג. כיצד הגיב רבי שמעון בר יוחאי (רשב"י) למצב המתואר: “בְּנֵי הָאָחוֹת, שֶׁהָיוּ תַּלְמִידֵי חֲכָמִים טְהוֹרִים, לֹא שָׁאֲלוּ שְׁאֵלוֹת.”?
ד. מה השתנה בין “הַחַיָּלִים לָפְתוּ אֶת יְדֵיהֶם שֶׁל הָאַחְיָנִים וְכָבְלוּ אוֹתָם בְּשַׁרְשְׁרָאוֹת בַּרְזֶל – קְלַאנְק...” לבין “קָצִין מְרֻשָּׁע דּוֹרֵשׁ כֶּסֶף מְפֻבְרָק וְכָל רְכוּשֵׁנוּ בְּסַכָּנָה!".”?
ה. כיצד הגיע הסיפור אל האירוע: “הֱיִיתֶם אוֹסְפִים בִּשְׁבִיל לְהַצִּיל אֶת הַכֶּסֶף שֶׁלָּכֶם, כְּדֵי שֶׁלֹּא יִפֹּל בִּידֵי הַמַּלְכוּת הָרְשָׁעָה!”?

לפני ואחרי

מה קרה לפני: “קֻמוּ מִיָּד!" שָׁאֲג הַקָּצִין הָרוֹמָאִי, קוֹלוֹ הִדְהֵד בַּחֶדֶר כְּמוֹ רַעַם.”?
מה קרה אחרי: “הֵם הוֹצִיאוּ אֶת הַפִּנְקָס הַקָּטָן שֶׁהָיָה בְּכִיסָם, חִשְּׁבוּ שׁוּב וְאָמְרוּ: "חֲמֵשׁ מֵאוֹת תִּשְׁעִים וְאַרְבָּעָה (594) דִּינָרִים!".”?

סיבה ותוצאה מתוך המעשה

האירוע הנתוןמה גרם לו?מה הייתה תוצאתו?
הֵם הִגִּיעוּ, נַעֲלֵיהֶם מְכֻסּוֹת מְעַט בַּאֲבַק הַדְּרָכִים, וְקָדוּ קִידָה עֲמֻקָּה לְדוֹדָם הַגָּדוֹל.
הוּא לָקַח אֶת גְּלִימָתוֹ, וּצְעָדָיו הָיוּ בְּטוּחִים וְשְׁקֵטִים בַּדֶּרֶךְ לְבֵית הַמִּדְרָשׁ.

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “הדינרים של בני האחות” על רבי שמעון בר יוחאי (רשב"י)? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “הָיָה חָסֵר לָהֶם כִּמְעַט סְכוּם שֶׁל שִׁשָּׁה דִּינָרִים שְׁלֵמִים.”.
ג. אילו רבי שמעון בר יוחאי (רשב"י) לא היה פועל כפי שמתואר ב־“קָצִין מְרֻשָּׁע דּוֹרֵשׁ כֶּסֶף מְפֻבְרָק וְכָל רְכוּשֵׁנוּ בְּסַכָּנָה!".” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “הֱיִיתֶם אוֹסְפִים בִּשְׁבִיל לְהַצִּיל אֶת הַכֶּסֶף שֶׁלָּכֶם, כְּדֵי שֶׁלֹּא יִפֹּל בִּידֵי הַמַּלְכוּת הָרְשָׁעָה!” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. להלן מערך חינוכי, סיפורי ומרגש על התנא הקדוש רבי שמעון בר יוחי (רשב"י)…
2. אוֹ לוֹמַר לָהֶם: "תִּתְרְמוּ מַהֵר שֵׁשׁ מֵאוֹת דִּינָרִים לִצְדָקָה…
3. הֵם חִלְּקוּ לִיתוֹמִים, הֵבִיאוּ לֶחֶם וּבָשָׂר לְאַלְמָנוֹת, דָּאֲגוּ…
4. קַר, רָטֹב וְחָשׁוּךְ כִּמְעַט לְגַמְרֵי, רַק קֶרֶן אוֹר חִוֶּרֶת…
5. שִׁשָּׁה דִּינְרֵי זָהָב (וְשִׁשָּׁה דִּינָרִים הוּא סְכוּם נִכְבָּד,…
6. בסיום, שקשק את הקופה וְאָמַר: "כָּאן בַּקֻּפְסָה הַזּוֹ מִסְתַּתֵּר…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “בְּנֵי הָאָחוֹת, שֶׁהָיוּ תַּלְמִידֵי חֲכָמִים טְהוֹרִים, לֹא שָׁאֲלוּ שְׁאֵלוֹת.”.
מכתב: כתבו לרבי שמעון בר יוחאי (רשב"י) בעקבות “הוּא לָקַח אֶת גְּלִימָתוֹ, וּצְעָדָיו הָיוּ בְּטוּחִים וְשְׁקֵטִים בַּדֶּרֶךְ לְבֵית הַמִּדְרָשׁ.”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “הוּא קָרָא לְשַׂר בֵּית הָאֲסוּרִים, שִׁלְשֵׁל אֶת הַמַּטְבְּעוֹת לְתוֹךְ יָדוֹ.”.
דיווח: כתבו ידיעה קצרה שכותרתה “הדינרים של בני האחות”.

המחזה

רגע מן הסיפורהדמויותמה יאמרו או יעשו?
הֵם הִגִּיעוּ, נַעֲלֵיהֶם מְכֻסּוֹת מְעַט בַּאֲבַק הַדְּרָכִים, וְקָדוּ קִידָה עֲמֻקָּה לְדוֹדָם…
קָצִין מְרֻשָּׁע דּוֹרֵשׁ כֶּסֶף מְפֻבְרָק וְכָל רְכוּשֵׁנוּ בְּסַכָּנָה!".
