אוהב מחקר ובעלי חיים — עיסוק בנמלים, דוכיפת ועוד

דף עבודה בעקבות ר' שמעון בן חלפתא
שם: כיתה:

המעשה — הסיפור המלא

פְּתִיחָה: הָרַב שֶׁיָּצָא אֶל הַשָּׂדֶה

יְלָדִים, כַּאֲשֶׁר אֲנַחְנוּ חוֹשְׁבִים עַל תַּלְמִיד חָכָם גָּדוֹל, אָנוּ מְדַמְיְנִים אוֹתוֹ יוֹשֵׁב כָּל הַיּוֹם בְּבֵית הַמִּדְרָשׁ, מִתְנַדְנֵד מוּל סֵפֶר פָּתוּחַ וְשׁוֹקֵעַ בַּהֲלָכוֹת. וְזֶה נָכוֹן! אַךְ הַיּוֹם נַכִּיר תַּלְמִיד חָכָם עָצוּם, רַבִּי שִׁמְעוֹן בֶּן חֲלַפְתָּא, שֶׁהָיָה יוֹצֵא מִבֵּית הַמִּדְרָשׁ אֶל הַשָּׂדוֹת, נוֹשֵׁם עָמֹק אֶת רֵיחָהּ שֶׁל הָאֲדָמָה הָרְטֻבָּה, וּמִתְבּוֹנֵן בַּפְּרָחִים וּבְבַעֲלֵי הַחַיִּים.

רַבִּי שִׁמְעוֹן אָהַב לַחְקֹר אֶת הַטֶּבַע. הוּא לֹא חָקַר רַק כְּדֵי לְהַשְׂכִּיל, אֶלָּא מִפְּנֵי שֶׁיָּדַע סוֹד גָּדוֹל: הַקָּדוֹשׁ בָּרוּךְ הוּא כָּתַב אֶת הַתּוֹרָה, וְהַקָּדוֹשׁ בָּרוּךְ הוּא גַּם בָּרָא אֶת הַטֶּבַע. אִם נִלְמַד עַל הַטֶּבַע – נָבִין כַּמָּה גְּדוֹלָה הִיא תּוֹרַת ה'! חֲזַ"ל קָרְאוּ לוֹ "עַסְקָן בִּדְבָרִים", כְּלוֹמַר, אָדָם שֶׁעוֹשֶׂה נִסּוּיִים כְּדֵי לִרְאוֹת אֶת אֱמֶת הַתּוֹרָה בְּעֵינָיו.

הַנִּסּוּי הָרִאשׁוֹן: סוֹד הַנְּמָלִים

יוֹם אֶחָד, יָשַׁב רַבִּי שִׁמְעוֹן וְלָמַד אֶת סֵפֶר מִשְׁלֵי שֶׁכָּתַב שְׁלֹמֹה הַמֶּלֶךְ. הוּא קָרָא אֶת הַפָּסוּק: "לֵךְ אֶל נְמָלָה עָצֵל, רְאֵה דְרָכֶיהָ וַחֲכָם... אֲשֶׁר אֵין לָהּ קָצִין שֹׁטֵר וּמֹשֵׁל". שְׁלֹמֹה הַמֶּלֶךְ אוֹמֵר שֶׁלַּנְּמָלִים אֵין מֶלֶךְ שֶׁמְּפַקֵּד עֲלֵיהֶן וְאוֹמֵר לָהֶן מָה לַעֲשׂוֹת, וּבְכָל זֹאת כֻּלָּן עֲמֵלוֹת וַחֲרוּצוֹת!

חָשַׁב רַבִּי שִׁמְעוֹן: "אֲנִי רוֹצֶה לִבְדֹּק אֶת זֶה. הַאִם בֶּאֱמֶת אֵין לָהֶן מֶלֶךְ עֶלְיוֹן אוֹ שׁוֹפֵט?"

יָצָא רַבִּי שִׁמְעוֹן אֶל הַשָּׂדֶה הַחַם בִּשְׁעַת צָהֳרַיִם. קַרְנֵי הַשֶּׁמֶשׁ צָרְבוּ, אַךְ הוּא רָכַן עַל בִּרְכָּיו מוּל קֵן הֲמוֹנֶה נְמָלִים שְׁחוֹרוֹת. הוּא הֵסִיר אֶת מְעִילוֹ וּפָרַס אוֹתוֹ מֵעַל הַקֵּן. לְפֶתַע, יָרַד צֵל עַל קֵן הַנְּמָלִים.

נְמָלָה אַחַת הֵצִיצָה הַחוּצָה, חָשָׁה בַּצֵּל הַקָּרִיר וְנִבְהֲלָה – "אוֹי, יֵשׁ צֵל! מִישֶׁהוּ מַסְתִּיר אֶת הַשֶּׁמֶשׁ!" הִיא מִהֲרָה אֶל תּוֹךְ הַקֵּן לְבַשֵּׂר לַחֲבֵרוֹתֶיהָ: "צְאוּ הַחוּצָה בְּחֻמּוֹ שֶׁל יוֹם, מָצָאתִי מָקוֹם עִם צֵל!"

בָּרֶגַע שֶׁהַנְּמָלָה חָזְרָה פְּנִימָה, הֵרִים רַבִּי שִׁמְעוֹן אֶת הַמְּעִיל לְמַעְלָה. קַרְנֵי הַשֶּׁמֶשׁ הַחַמּוֹת שׁוּב הִכּוּ בָּאֲדָמָה הַחֲשׂוּפָה.

הַנְּמָלָה הָרִאשׁוֹנָה יָצְאָה, וְאַחֲרֶיהָ שַׁיָּרָה עֲנָקִית שֶׁל נְמָלִים, רָצוֹת לִמְצֹא צֵל. אֲבָל כְּשֶׁהֵן יָצְאוּ... הָיְתָה רַק שֶׁמֶשׁ לוֹהֶטֶת! הַנְּמָלִים סָבְרוּ שֶׁהַנְּמָלָה הָרִאשׁוֹנָה שִׁקְּרָה לָהֶן. מִיָּד, בְּלִי לְהַמְתִּין לְמִשְׁפָּט, בְּלִי לִשְׁאֹל מֶלֶךְ אוֹ קָצִין, הֵן הִתְנַפְּלוּ עָלֶיהָ וְהָרְגוּ אוֹתָהּ עַל מְקוֹמָהּ.

חִיֵּךְ רַבִּי שִׁמְעוֹן שֶׁחָקַר וְגִלָּה: "שְׁלֹמֹה הַמֶּלֶךְ צָדַק! 'אֲשֶׁר אֵין לָהּ קָצִין שֹׁטֵר וּמֹשֵׁל'. אִם הָיָה לָהֶן שׁוֹפֵט, הֵן הָיוּ מְבִיאוֹת אוֹתָהּ לְמִשְׁפָּט, אַךְ כֵּיוָן שֶׁפָּעֲלוּ מִיָּד בְּעַצְמָן, סִימָן שֶׁאֵין עֲלֵיהֶן מֶלֶךְ!". (אַף שֶׁחֲכָמִים אֲחֵרִים טָעֲנוּ שֶׁאוּלַי דַּוְקָא אוֹתָהּ נְמָלָה הָיְתָה הַמַּלְכָּה, רַבִּי שִׁמְעוֹן עָשָׂה הַכֹּל כְּדֵי לִרְאוֹת אֶת סוֹדוֹת הַתּוֹרָה בָּעֵינַיִם).

הַנִּסּוּי הַשֵּׁנִי: קוֹלוֹת הַזְּכוּכִית וְעֵשֶׂב הַפֶּלֶא

אַךְ אַהֲבָתוֹ הַגְּדוֹלָה בְּיוֹתֵר לְטֶבַע, הָיְתָה לַמְּעוֹפְפִים.

יָמִים רַבִּים הִתְבּוֹנֵן רַבִּי שִׁמְעוֹן בְּצִפּוֹר הַדּוּכִיפַת (חֲזַ"ל קָרְאוּ לָהּ "תַּרְנְגוֹלֶת הַבָּר" אוֹ נְגַר הֶהָרִים). מְדֻבָּר בְּצִפּוֹר חֲכָמָה שֶׁחִפְּשָׂה עֵץ כְּדֵי לִבְנוֹת בּוֹ קֵן לְגוֹזָלֶיהָ. הִיא מָצְאָה חֹר בְּגֶזַע עֵץ, הֵבִיאָה עֲנָפִים רַכִּים, וְיָצְרָה קֵן. אָז הִטִּילָה בּוֹ אֶת בֵּיצֶיהָ, וְיָצְאָה לְחַפֵּשׂ תּוֹלָעִים מְתוּקוֹת לַאֲרוּחָה.

אָז בָּא רַבִּי שִׁמְעוֹן. הָיָה לוֹ רַעְיוֹן לְנִסּוּי! הוּא בָּחַן לוּחַ זְכוּכִית חָלָק, קַר וְעָבֶה. הוּא הִנִּיחַ אֶת מִשְׁטַח הַזְּכוּכִית הֶחָזָק עַל פֶּתַח הַקֵּן, וּמִסְמֵר אוֹתוֹ שֶׁלֹּא יִפֹּל. כָּעֵת, הַקֵּן הָיָה סָגוּר. הָיָה אֶפְשָׁר לִרְאוֹת פְּנִימָה, אַךְ לֹא לְהִכָּנֵס!

רַבִּי שִׁמְעוֹן הִסְתַּתֵּר מֵאֲחוֹרֵי שִׂיחַ רַעֲנָן וְהִמְתִּין.

צְוִיץ... צְוִיץ...

הַדּוּכִיפַת חָזְרָה. הִיא נִסְּתָה לְהִכָּנֵס, אַךְ – טוּק! – הַמָּקוֹר שֶׁלָּהּ נִתְקַע בַּזְּכוּכִית. הִיא דָּפְקָה וְדָפְקָה, אֲבָל הַזְּכוּכִית לֹא מֵנֵצָה.

מָה תַּעֲשֶׂה צִפּוֹר עַכְשָׁו? תִּתְיָאֵשׁ?

לֹא צִפּוֹר הַפֶּלֶא שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא! הַצִּפּוֹר פָּרְשָׂה כְּנָפַיִם וְעָפָה הַרְחֵק הַרְחֵק. עָבְרוּ שָׁעוֹת. רַבִּי שִׁמְעוֹן לֹא עָזַב. עֵינָיו עָקְבוּ אַחֲרֵי הַשָּׁמַיִם.

לְפֶתַע, שָׁבָה הַדּוּכִיפַת. בְּמַקּוֹרָהּ, הִיא הֶחְזִיקָה עָלֶה קָטָן, יָרֹק וּמְיֻחָד. עֵשֶׂב פֶּלֶא! זֶהוּ צֶמַח שֶׁיָּכוֹל לַחְתֹּךְ גַּם זְכוּכִית וּבַרְזֶל. הַדּוּכִיפַת נָחֲתָה עַל צֶמֶרֶת הָעֵץ וְהִתְכּוֹנְנָה לְהַנִּיחַ אֶת הֶעָלֶה עַל הַזְּכוּכִית כְּדֵי שֶׁתִּתְנַפֵּץ.

בָּרֶגַע זֶה, מָחָא רַבִּי שִׁמְעוֹן כַּפַּיִם בְּעָצְמָה! פְּלַאק!

הַצִּפּוֹר נִבְהֲלָה מֵהָרַעַשׁ הַפִּתְאוֹמִי. מִקּוֹל הַפַּחַד – הִיא שָׁמְטָה אֶת הֶעָלֶה מִמַּקּוֹרָהּ.

הֶעָלֶה צָנַח מַטָּה, וְרַבִּי שִׁמְעוֹן תָּפַס אוֹתוֹ בְּיָדוֹ!

הַצִּפּוֹר הַמִּסְכֵּנָה, שֶׁרָאֲתָה שֶׁאִבְּדָה אֶת הָעֵשֶׂב הַנָּדִיר שֶׁעִמּוֹ הָיְתָה צְרִיכָה לְבַקֵּעַ אֶת עֲצֵי הָעוֹלָם (וְהִיא נִשְׁבְּעָה לֹא לְאַבֵּד אוֹתוֹ), צָעֲקָה וְחָנְקָה אֶת עַצְמָהּ, וְרַבִּי שִׁמְעוֹן לָקַח אֶת אֶת אוֹתוֹ עֵשֶׂב פֶּלֶא (כְּעֵין הַשָּׁמִיר הַמֻּפְלָא) וְרָאָה בְּעֵינָיו אֶת הַכֹּחוֹת הָעֲצוּמִים שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא טָמַן בַּצְּמָחִים וּבַחַיּוֹת. (כַּמּוּבָן שֶׁלְּאַחַר מִכֵּן שִׁחְרֵר אֶת בֵּיצֵי הַדּוּכִיפַת, כִּי תַּלְמִידֵי חֲכָמִים רַחְמָנִים הֵם).

הַמְּעִיל שֶׁל הַתַּרְנְגֹלֶת

אַהֲבָתוֹ שֶׁל רַבִּי שִׁמְעוֹן אֶל בַּעֲלֵי הַחַיִּים גָּרְמָה לוֹ גַּם לַעֲזֹר לָהֶם כְּשֶׁסָּבְלוּ.

בְּחַצְרוֹ הָיְתָה תַּרְנְגֹלֶת קְטַנָּה שֶׁנָּשְׁרוּ לָהּ כָּל נוֹצוֹתֶיהָ. יְלָדִים, הַנּוֹצוֹת הֵן אֵלּוּ שֶׁמְּחַמְּמוֹת אֶת הַצִּפּוֹר. בְּלִי נוֹצוֹת, בִּימֵי קֹר, הַתַּרְנְגֹלֶת פָּשׁוּט קָפְאָה. הִיא עָמְדָה בְּפִנַּת הֶחָצֵר, וְהַגּוּף הַקָּטָן שֶׁלָּהּ רָעַד: בְּרְרְרְרְרְ... קוֹר עַז עָבַר בְּגוּפָהּ.

אָדָם רָגִיל הָיָה אָמוֹר: "אֵין מָה לַעֲשׂוֹת. זוֹ רַק תַּרְנְגֹלֶת".

אֲבָל רַבִּי שִׁמְעוֹן, גְּדוֹל הַדּוֹר, שֶׁעָסַק בְּלִמּוּד גְּמָרָא וַהֲלָכָה, רָאָה צַעַר שֶׁל בַּעַל חַיִּים וְלֹא יָכוֹל הָיָה לְהִתְעַלֵּם. מָה הוּא עָשָׂה?

הוּא הֵבִיא עַוֹר רַךְ וְחַם (שֶׁל גָּמָל, אוֹ עוֹר רַךְ אַחֵר) שֶׁמְּשַׁמֵּשׁ לִתְפִירָה, חָתַךְ אוֹתוֹ בְּמִדָּה מְדֻיֶּקֶת, וְתָפַר לַתַּרְנְגֹלֶת... מְעִיל עֶלְיוֹן מִינְיָאטוּרִי!

הוּא הִלְבִּישׁ אֶת הַתַּרְנְגֹלֶת בַּמְּעִיל הַקָּטָן. לְפֶתַע, הִיא הִפְסִיקָה לִרְעֹד. חֹם עָטַף אֶת גּוּפָהּ. וּרְאוּ זֶה פֶּלֶא – בִּזְכוּת הַחֹם הַמְּלָאכוּתִי הַזֶּה שֶׁרַבִּי שִׁמְעוֹן עָשָׂה לָהּ, הַגּוּף שֶׁלָּהּ הִתְחַזֵּק, וְלְאַט לְאַט הֵחֵלּוּ צּוֹמְחוֹת לָהּ נוֹצוֹת חֲדָשׁוֹת יָפוֹת וּמַבְרִיקוֹת, עַד שֶׁנִּרְפְּאָה לְגַמְרֵי!

יְלָדִים יְקָרִים, מֵרַבִּי שִׁמְעוֹן בֶּן חֲלַפְתָּא אָנוּ לוֹמְדִים דָּבָר עָצוּם. רַבָּנִים לֹא רַק יוֹדְעִים לִקְרֹא בַּסְּפָרִים. אֲנָשִׁים שֶׁאוֹהֲבִים אֶת ה', אוֹהֲבִים לַחְקֹר אֶת שָׁמַיִם, אֶת הָאֲדָמָה, אֶת הַנְּמָלִים הַקְּטַנּוֹת וְאֶת הַצִּפֳּרִים. כִּי הַכֹּל – אֲבָל הַכֹּל – מֵעִיד עַל הַחָכְמָה הָעֲצוּמָה שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא בָּעוֹלָם!

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “יְלָדִים, כַּאֲשֶׁר אֲנַחְנוּ חוֹשְׁבִים עַל תַּלְמִיד חָכָם גָּדוֹל, אָנוּ מְדַמְיְנִים אוֹתוֹ יוֹשֵׁב כָּל הַיּוֹם בְּבֵית הַמִּדְרָשׁ, מִתְנַדְנֵד מוּל…” — מה קרה מיד לאחר מכן?

ה. כיצד הגיע הסיפור אל האירוע: “וּרְאוּ זֶה פֶּלֶא – בִּזְכוּת הַחֹם הַמְּלָאכוּתִי הַזֶּה שֶׁרַבִּי שִׁמְעוֹן עָשָׂה לָהּ, הַגּוּף שֶׁלָּהּ הִתְחַזֵּק, וְלְאַט לְאַט הֵחֵלּוּ צּוֹמְחוֹת…”?
