בגיל 80 עומד על רגל אחת וחולץ ונועל מנעלו

דף עבודה בעקבות ר' חנינא בר חמא
שם: כיתה:

המעשה — הסיפור המלא

פְּתִיחָה: רֵיחוֹת הַגָּלִיל וְקוֹל הַתּוֹרָה

יְלָדִים, אִם הֱיִיתֶם מְטַיְּלִים לִפְנֵי כְּאַלְפַּיִם שָׁנָה בָּעִיר צִפּוֹרִי שֶׁבַּגָּלִיל, הֱיִיתֶם מַרְגִּישִׁים מִיָּד שֶׁזּוֹ עִיר מְיֻחֶדֶת מְאֹד. אֲוִיר הַבֹּקֶר הָיָה צָלוּל וְקָרִיר, וְרֵיחַ מָתוֹק שֶׁל עֲצֵי זַיִת וּתְאֵנִים נָשַׁב מִן הֶהָרִים וּמִלֵּא אֶת הָרְחוֹבוֹת.

אֲבָל הַדָּבָר הֲכִי יָפֶה בְּצִפּוֹרִי לֹא הָיָה הַנּוֹף אוֹ הָרֵיחַ. הָיָה זֶה הַקּוֹל.

מִתּוֹךְ בֵּית הַמִּדְרָשׁ הַגָּדוֹל, עָלָה קוֹל שִׁירָה מְתוּקָה – קוֹלָם שֶׁל מֵאוֹת תַּלְמִידֵי חֲכָמִים שֶׁיָּשְׁבוּ וְלָמְדוּ תּוֹרָה בְּהִתְלַהֲבוּת. הֵם לָמְדוּ כָּל כָּךְ בְּרֶצֶף, עַד שֶׁהַקּוֹל שֶׁלָּהֶם נִשְׁמַע כְּמוֹ רַעַשׁ שֶׁל מַפָּל מַיִם אֲדִירִים הַנִּשְׁפָּכִים עַל סְלָעִים.

בְּרֹאשׁ כָּל הַתַּלְמִידִים הָאֵלֶּה, עַל כִּסֵּא מְכֻבָּד מֵעֵץ, יָשַׁב רַבָּם הַגָּדוֹל, רַבִּי חֲנִינָא בַּר חָמָא .

רַבִּי חֲנִינָא הָיָה תַּלְמִידוֹ הָאָהוּב שֶׁל רַבִּי יְהוּדָה הַנָּשִׂיא. הוּא הָיָה אָדָם קָדוֹשׁ עַד מְאֹד, וּפָנָיו מַמָּשׁ הֵאִירוּ כְּמוֹ שֶׁמֶשׁ זְהוּבָה. עֵינָיו חָקְרוּ כָּל דַּף שֶׁל תּוֹרָה, וּשְׂפָתָיו מִלְמְלוּ פְּסוּקִים גַּם בְּאֶמְצַע הַלַּיְלָה.

הַזָּקֵן שֶׁחָזָק כְּבָחוּר

וְהִנֵּה, הָיָה דָּבָר אֶחָד שֶׁמְּאֹד הִפְלִיא אֶת כָּל תַּלְמִידָיו שֶׁל רַבִּי חֲנִינָא.

רַבִּי חֲנִינָא כְּבָר הָיָה אָדָם זָקֵן. זָקֵן מְאֹד. הוּא הָיָה בֶּן שְׁמוֹנִים שָׁנָה! הַזָּקָן שֶׁלּוֹ הָיָה לָבָן כְּשֶׁלֶג הַיּוֹרֵד עַל הַר הַחֶרְמוֹן.

יְלָדִים, אַתֶּם בְּוַדַּאי יוֹדְעִים אֵיךְ נִרְאֶה וּמַרְגִּישׁ אָדָם רָגִיל בֶּן שְׁמוֹנִים, נָכוֹן? הַגַּב מְעַט כָּפוּף, הָרַגְלַיִם חַלָּשׁוֹת, לִפְעָמִים צָרִיךְ לְהִשָּׁעֵן עַל מַקֵּל עָבֶה כְּדֵי לָלֶכֶת מִמָּקוֹם לְמָקוֹם, וְהַיָּדַיִם מְעַט רוֹעֲדוֹת כְּשֶׁמַּחֲזִיקִים כּוֹס מַיִם. זֶהוּ טֶבַע הָעוֹלָם שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא בָּרָא.

אֲבָל אֵצֶל רַבִּי חֲנִינָא בַּר חָמָא הַכֹּל הָיָה שׁוֹנֶה לְגַמְרֵי!

הַגּוּף שֶׁלּוֹ הָיָה זָקוּף כְּמוֹ חוֹמַת אֶבֶן חֲזָקָה. הוּא הָלַךְ מַהֵר וּבְקַלּוּת כְּמוֹ בָּחוּר צָעִיר וְנִמְרָץ. קוֹלוֹ הָיָה חָזָק וְצָלוּל כְּשֶׁלִּמֵּד תּוֹרָה, וְאֶצְבְּעוֹתָיו הָיוּ אֵיתָנוֹת וִיצִיבוֹת. הַתַּלְמִידִים הָיוּ מַבִּיטִים בּוֹ, מוֹחִים זֵעָה מִמִּצְחָם אַחֲרֵי שֶׁרָצוּ כְּדֵי לְהַשִּׂיג אוֹתוֹ בָּרְחוֹב, וְלוֹחֲשִׁים אֶחָד לַשֵּׁנִי: "אֵיךְ הָרַב בֶּן הַשְּׁמוֹנִים הוֹלֵךְ מַהֵר יוֹתֵר מֵאִתָּנוּ, בְּנֵי הָעֶשְׂרִים? אֵיזֶה מִן פֶּלֶא זֶה? מֵאֵיפֹה יֵשׁ לוֹ כֹּחַ אֲדִיר כָּזֶה?!".

הַמַּחֲזֶה הַמַּדְהִים עַל רֶגֶל אַחַת

יוֹם אֶחָד, יָשְׁבוּ הַתַּלְמִידִים עַל סַפְסַלֵּי הָעֵץ בְּבֵית הַמִּדְרָשׁ, מְמַתִּינִים שֶׁהָרַב יַתְחִיל אֶת הַשִּׁעוּר.

רַבִּי חֲנִינָא נִכְנַס לַחֶדֶר. מַשֶּׁהוּ כַּנִּרְאֶה הִפְרִיעַ לוֹ בְּנַעֲלוֹ. אוּלַי רְצוּעַת הָעוֹר הַחֲזָקָה שֶׁל הַסַּנְדָּל הִשְׁתַּחְרְרָה, אוֹ שֶׁאֶבֶן קְטַנָּה נִכְנְסָה לְתוֹכָהּ.

וְאָז... קָרָה מַשֶּׁהוּ שֶׁהִשְׁאִיר אֶת כָּל עֵינֵי הַתַּלְמִידִים פְּעוּרוֹת מֵרֹב תַּדְהֵמָה!

רַבִּי חֲנִינָא בַּר חָמָא, זְקַן חַכְמֵי יִשְׂרָאֵל, לֹא יָשַׁב עַל הַכִּסֵּא כְּדֵי לַחְלֹץ מַעֲלוֹ. הוּא לֹא נִשְׁעַן עַל הַקִּיר כְּדֵי לְהַחְזִיק אֶת עַצְמוֹ.

הוּא פָּשׁוּט קִפֵּל רֶגֶל אַחַת שֶׁלּוֹ בָּאֲוִיר, וְנִשְׁאַר לַעֲמֹד... עַל רֶגֶל אַחַת בִּלְבַד! יְצִיב לְגַמְרֵי, בְּלִי לְהִתְנַדְנֵד אֲפִלּוּ כִּמְלוֹא נִימָה.

בְּעוֹדוֹ עוֹמֵד כָּךְ עַל רֶגֶל אַחַת, הוּא הִתְכּוֹפֵף, פָּתַח בְּאֶצְבְּעוֹתָיו אֶת הַקֶּשֶׁר שֶׁל עֲאוֹת הָעוֹר, חָלַץ אֶת הַנַּעַל שֶׁלּוֹ, סִדֵּר אֶת מָה שֶׁצָּרִיךְ, שָׁב וְנָעַל אֶת הַנַּעַל, קָשַׁר אוֹתָהּ הֵיטֵב בְּחָזְקָה, וְרַק אָז הֵנִיחַ מִחָדָשׁ אֶת הָרֶגֶל הַשְּׁנִיָּה עַל הָאֲדָמָה.

וְכָל זֶה – בְּגִיל שְׁמוֹנִים שָׁנָה! בְּלִי לִפֹּל, בְּלִי לָמֹּעַ, וּבְלִי עֶזְרָה מֵאִישׁ!

הַשֶּׁקֶט בְּבֵית הַמִּדְרָשׁ הָיָה כָּל כָּךְ גָּדוֹל, עַד שֶׁהָיָה אֶפְשָׁר לִשְׁמֹעַ זְבוּב עוֹף.

הַתַּלְמִידִים הִסְתַּכְּלוּ בְּרַבָּם, וְאָז הִסְתַּכְּלוּ אֶחָד עַל הַשֵּׁנִי. הֵם לֹא הֶאֱמִינוּ לְמַרְאֵה עֵינֵיהֶם. אֲפִלּוּ בָּחוּר צָעִיר וְחָזָק מִתְקַשֶּׁה לַעֲמֹד עַל רֶגֶל אַחַת לִזְמַן כָּל כָּךְ מְמֻשָּׁךְ, לְהִתְכּוֹפֵף וְלִקְשֹׁר אֶת מַעֲלוֹ, מִבְּלִי לְאַבֵּד אֶת שִׁוּוּי הַמִּשְׁקָל!

"רַבֵּנוּ וּמוֹרֵנוּ הַקָּדוֹשׁ..." לָחַשׁ אֶחָד הַתַּלְמִידִים, קוֹלוֹ רוֹעֵד מִיִּרְאַת כָּבוֹד, "אֵיךְ... אֵיךְ יֵשׁ לָרַב כָּל כָּךְ הַרְבֵּה כֹּחַ? הֲרֵי זֶה פֶּלֶא שֶׁל מַמָּשׁ! מָה סוֹד הַגְּבוּרָה וְהַבְּרִיאוּת שֶׁל כְּבוֹדוֹ?".

כֻּלָּם הִמְתִּינוּ לַתְּשׁוּבָה. אולי הָרַב יַגִּיד שֶׁהוּא שׁוֹתֶה תְּרוּפַת קֶסֶם נְדִירָה? אוּלַי יֹאמַר שֶׁזֶּה בִּזְכוּת צִיּוֹם אוֹ שֶׁלָּמַד שַׁ"ס בְּעַל פֶּה אַלְפֵי פְּעָמִים?

סוֹדָהּ שֶׁל אִמָּא וְהַכָּרַת הַטּוֹב

רַבִּי חֲנִינָא בַּר חָמָא חִיֵּךְ. זֶה הָיָה חִיּוּךְ רַךְ וְחָם. הוּא הִבִּיט עַל תַּלְמִידָיו הָאֲהוּבִים, עֵינָיו נִצְנְצוּ, וְתָלָה אֶת מַבָּטוֹ לְרֶגַע כְּמוֹ נִזְכָּר בְּמַשֶּׁהוּ שֶׁקָּרָה לִפְנֵי שָׁנִים עַל גַּבֵּי שָׁנִים, כְּשֶׁהוּא בְּעַצְמוֹ עוֹד הָיָה יֶלֶד קָטָן.

"סוֹד הַגְּבוּרָה שֶׁלִּי?" שָׁאַל בְּקוֹל רוֹגֵעַ. "לֹא, יְלָדַי הַיְּקָרִים. אֵין כָּאן קֶסֶם, וְאֵין כָּאן סוֹד אִישִׁי שֶׁלִּי."

רַבִּי חֲנִינָא סִדֵּר אֶת גְּלִימָתוֹ, נָשַׁם עֲמֻקּוֹת וְאָמַר לָהֶם אֶת הַמִּשְׁפָּט הַמַּדְהִים הַבָּא:

"חַמִּין וְשֶׁמֶן שֶׁסָּכַתְנִי אִמִּי בְּיַלְדוּתִי – הֵן עָמְדוּ לִי בְּעֵת זִקְנוּתִי" .

"מָה הַפֵּרוּשׁ?" תָּמְהוּ הַתַּלְמִידִים.

"כְּשֶׁהָיִיתִי יֶלֶד קָטָן," הִסְבִּיר הַצַּדִּיק, "אִמִּי עָלֶיהָ הַשָּׁלוֹם, הָיְתָה דּוֹאֶגֶת לִי בִּמְסִירוּת שֶׁאֵין כְּמוֹתָהּ. הִיא לֹא רַק אָהֲבָה אוֹתִי בַּלֵּב, הִיא הִשְׁקִיעָה בִּי הֲמוֹן כֹּחַ! בְּיָמִים הַקָּרִים, הִיא הָיְתָה רוֹחֶצֶת אֶת גּוּפִי הַקָּטָן בְּמַיִם חַמִּים ('חַמִּין') כְּדֵי שֶׁאֶהְיֶה נָקִי וּבָרִיא. אַחַר כָּךְ, הִיא הָיְתָה מְחַמֶּמֶת מְעַט שֶׁמֶן זַיִת טָהוֹר, שֶׁמֶן כָּזֶה שֶׁמֵּרִיחַ מְעֻלֶּה, וּמוֹרַחַת אוֹתוֹ ('סָכַתְנִי') עַל בְּשָׂרִי הָרַךְ כְּמִשְׁחָה פֶלֶא, מְעַסָּה אֶת זְרוֹעוֹתַי וְאֶת רַגְלַי, כְּדֵי לְחַזֵּק אֶת עֲצָמוֹתַי שֶׁיִּגְדְּלוּ מְלֵאוֹת כֹּחַ.

הִיא עָשְׂתָה אֶת הַכֹּל מִתּוֹךְ תְּפִלָּה גְּדוֹלָה שֶׁאֶגְדַּל לִהְיוֹת לְמַדְרֵגַת תַּלְמִיד חָכָם וְשֶׁיִּהְיֶה לִי גּוּף בָּרִיא כְּדֵי שֶׁאוּכַל לַעֲמֹל בַּתּוֹרָה."

רַבִּי חֲנִינָא בַּר חָמָא סִיֵּם אֶת דְּבָרָיו כְּשֶׁדִּמְעָה קְטַנָּה נוֹצֶצֶת בְּזָוִית עֵינוֹ: "הַהַשְׁקָעָה הַזּוֹ שֶׁל אִמִּי שֶׁטָּרְחָה בִּשְׁבִילִי וְהִשְׁקִיעָה בִּי אֶת חַיֶּיהָ לִפְנֵי שִׁבְעִים אוֹ שְׁמוֹנִים שָׁנָה... הִיא זוֹ שֶׁשּׁוֹמֶרֶת עָלַי הַיּוֹם! וּבִזְכוּתָהּ הַקָּדוֹשׁ בָּרוּךְ הוּא נוֹתֵן לִי כֹּחַ לַעֲמֹד אֵיתָן כַּבַּרְזֶל גַּם בְּגִילִי הַמֻּפְלָג!".

הַתַּלְמִידִים הִקְשִׁיבוּ מְרֻתָּקִים. הֵם הֵבִינוּ לְפֶתַע אֶת הַשִּׁעוּר הַגָּדוֹל בְּיוֹתֵר: כַּמָּה גְּדוֹלָה זְכוּתָן שֶׁל הָאִמָּהוֹת (וְהָאָבוֹת) שֶׁמְּטַפְּלוֹת בְּיַלְדֵיהֶן בְּמְסִירוּת, וְאֵיזוֹ מִדָּה עֲצוּמָה שֶׁל הַכָּרַת הַטּוֹב יֵשׁ לְרַבָּם שֶׁלֹּא שָׁכַח זֹאת לְעוֹלָם, וְיָדַע לוֹמַר "תּוֹדָה" לְאִמּוֹ עַל הַטִּפּוּל גַּם לְאַחַר עֲשָׂרוֹת שָׁנִים!

מקורות: תלמוד בבלי, מסכת חולין, דף כ"ד, עמוד ב': "רבי חנינא בן שמונים שנה היה, והיה עומד על רגלו אחת, וחולץ מנעלו ונועל מנעלו. אמר רבי חנינא: חמין ושמן שסכתני אמי בילדותי הן עמדו לי בעת זקנותי." תולדות רבי חנינא בר חמא: ירושלמי פאה פ"ז (אהבתו לארץ, רפואה וחקלאות), ועוד מדרשים המאירים את הנהגתו כראש הישיבה בצפורי.

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “יְלָדִים, אִם הֱיִיתֶם מְטַיְּלִים לִפְנֵי כְּאַלְפַּיִם שָׁנָה בָּעִיר צִפּוֹרִי שֶׁבַּגָּלִיל, הֱיִיתֶם מַרְגִּישִׁים מִיָּד שֶׁזּוֹ עִיר מְיֻחֶדֶת…” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “רַבִּי חֲנִינָא הָיָה תַּלְמִידוֹ הָאָהוּב שֶׁל רַבִּי יְהוּדָה הַנָּשִׂיא.”?
ג. כיצד הגיב ר' חנינא בר חמא למצב המתואר: “אֲבָל אֵצֶל רַבִּי חֲנִינָא בַּר חָמָא הַכֹּל הָיָה שׁוֹנֶה לְגַמְרֵי!”?
ד. מה השתנה בין “רַבִּי חֲנִינָא בַּר חָמָא, זְקַן חַכְמֵי יִשְׂרָאֵל, לֹא יָשַׁב עַל הַכִּסֵּא כְּדֵי לַחְלֹץ מַעֲלוֹ.” לבין “רַבֵּנוּ וּמוֹרֵנוּ הַקָּדוֹשׁ..." לָחַשׁ אֶחָד הַתַּלְמִידִים, קוֹלוֹ רוֹעֵד מִיִּרְאַת כָּבוֹד, "אֵיךְ...”?
ה. כיצד הגיע הסיפור אל האירוע: “בְּיָמִים הַקָּרִים, הִיא הָיְתָה רוֹחֶצֶת אֶת גּוּפִי הַקָּטָן בְּמַיִם חַמִּים ('חַמִּין') כְּדֵי שֶׁאֶהְיֶה נָקִי וּבָרִיא.”?

סיבה ותוצאה מתוך המעשה

האירוע הנתוןמה גרם לו?מה הייתה תוצאתו?
וְהִנֵּה, הָיָה דָּבָר אֶחָד שֶׁמְּאֹד הִפְלִיא אֶת כָּל תַּלְמִידָיו שֶׁל רַבִּי חֲנִינָא.
הַשֶּׁקֶט בְּבֵית הַמִּדְרָשׁ הָיָה כָּל כָּךְ גָּדוֹל, עַד שֶׁהָיָה אֶפְשָׁר לִשְׁמֹעַ זְבוּב עוֹף.

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “בגיל 80 עומד על רגל אחת וחולץ ונועל מנעלו” על ר' חנינא בר חמא? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “קוֹלוֹ הָיָה חָזָק וְצָלוּל כְּשֶׁלִּמֵּד תּוֹרָה, וְאֶצְבְּעוֹתָיו הָיוּ אֵיתָנוֹת וִיצִיבוֹת.”.
ג. אילו ר' חנינא בר חמא לא היה פועל כפי שמתואר ב־“רַבֵּנוּ וּמוֹרֵנוּ הַקָּדוֹשׁ..." לָחַשׁ אֶחָד הַתַּלְמִידִים, קוֹלוֹ רוֹעֵד מִיִּרְאַת כָּבוֹד, "אֵיךְ...” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “בְּיָמִים הַקָּרִים, הִיא הָיְתָה רוֹחֶצֶת אֶת גּוּפִי הַקָּטָן בְּמַיִם חַמִּים ('חַמִּין') כְּדֵי שֶׁאֶהְיֶה נָקִי וּבָרִיא.” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. יְלָדִים, אִם הֱיִיתֶם מְטַיְּלִים לִפְנֵי כְּאַלְפַּיִם שָׁנָה בָּעִיר…
2. הוּא הָיָה אָדָם קָדוֹשׁ עַד מְאֹד, וּפָנָיו מַמָּשׁ הֵאִירוּ כְּמוֹ…
3. הוּא הָלַךְ מַהֵר וּבְקַלּוּת כְּמוֹ בָּחוּר צָעִיר וְנִמְרָץ.
4. הוּא לֹא נִשְׁעַן עַל הַקִּיר כְּדֵי לְהַחְזִיק אֶת עַצְמוֹ.
5. רַבִּי חֲנִינָא סִדֵּר אֶת גְּלִימָתוֹ, נָשַׁם עֲמֻקּוֹת וְאָמַר לָהֶם…
6. וּבִזְכוּתָהּ הַקָּדוֹשׁ בָּרוּךְ הוּא נוֹתֵן לִי כֹּחַ לַעֲמֹד אֵיתָן…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “אֲבָל אֵצֶל רַבִּי חֲנִינָא בַּר חָמָא הַכֹּל הָיָה שׁוֹנֶה לְגַמְרֵי!”.
מכתב: כתבו לר' חנינא בר חמא בעקבות “הַשֶּׁקֶט בְּבֵית הַמִּדְרָשׁ הָיָה כָּל כָּךְ גָּדוֹל, עַד שֶׁהָיָה אֶפְשָׁר לִשְׁמֹעַ זְבוּב עוֹף.”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “חַמִּין וְשֶׁמֶן שֶׁסָּכַתְנִי אִמִּי בְּיַלְדוּתִי – הֵן עָמְדוּ לִי בְּעֵת זִקְנוּתִי" .”.
דיווח: כתבו ידיעה קצרה שכותרתה “בגיל 80 עומד על רגל אחת וחולץ ונועל מנעלו”.
