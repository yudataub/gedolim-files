סיפור 89 • ר' שמעון בן חלפתא • הנערים שזלזלו בצדיק ועונשם

הנערים שזלזלו בצדיק ועונשם

דף עבודה בעקבות ר' שמעון בן חלפתא
שם: כיתה:

המעשה — הסיפור המלא

פְּתִיחָה: חָכְמַת הַבּוֹרֵא בַּטֶּבַע

יְלָדִים, דַּמְיְנוּ עַצְמְכֶם צוֹעֲדִים בִּשְׁבִילֵי אֶרֶץ יִשְׂרָאֵל שֶׁל יְמֵי קֶדֶם. קַרְנֵי שֶׁמֶשׁ זְהֻבּוֹת מְלַטְּפוֹת אֶת הֶהָרִים, רֵיחַ טָרִי שֶׁל אֲדָמָה רְטֻבָּה וַעֲלֵי תְּאֵנָה מְמַלֵּא אֶת הָאֲוִיר, וְצִיּוּץ עַלִּיז שֶׁל צִפֳּרִים נִשְׁמָע מִכָּל עֵבֶר.

בַּשְּׁבִיל הַזֶּה, הָלַךְ בְּנַחַת סָבָא קָדוֹשׁ וְטָהוֹר, עֲטוּף בְּטַלִּית. שְׁמוֹ הָיָה רַבִּי שִׁמְעוֹן בֶּן חֲלַפְתָּא .

רַבִּי שִׁמְעוֹן לֹא הָיָה רַק תַּלְמִיד חָכָם שֶׁיּוֹשֵׁב בְּבֵית הַמִּדְרָשׁ. הַקָּדוֹשׁ בָּרוּךְ הוּא חָנַן אוֹתוֹ בְּאַהֲבָה עֲצוּמָה לַטֶּבַע וְלִבְרִיאוֹתָיו שֶׁל ה'. בְּכָל נְמָלָה שֶׁסָּחֲבָה גַּרְגִּיר חִטָּה, בְּכָל קוֹל רִשְׁרוּשׁ שֶׁל עֲלֵי הַזַּיִת בָּרוּחַ, הוּא שָׁמַע אֶת הַשִּׁירָה שֶׁהָעוֹלָם שָׁר לְבוֹרְאוֹ.

רַבִּי שִׁמְעוֹן הָלַךְ בַּדֶּרֶךְ, עֵינָיו הַקְּדוֹשׁוֹת בּוֹחֲנוֹת פֶּרַח קָטָן בְּצַד הַדֶּרֶךְ, וּשְׂפָתָיו רוֹחֲשׁוֹת דִּבְרֵי תּוֹרָה בְּנִגּוּן חֲרִישִׁי וּמָתוֹק. הַשֶּׁקֶט שֶׁל הַטֶּבַע הָיָה מֻשְׁלָם.

הַצְּחוֹק שֶׁקָּרַע אֶת הַשֶּׁקֶט

אַךְ לְפֶתַע, קוֹלוֹת רָמִים קָרְעוּ אֶת הַשַּׁלְוָה.

“חַחַחַחַ! הָהָהָהָ!”

מִמּוֹרַד הַגִּבְעָה, הִגִּיעָה חֲבוּרָה שֶׁל נְעָרִים. הֵם בָּעֲטוּ בָּאֲבָנִים, תָּלְשׁוּ עֲנָפִים יְרֻקִּים מֵהָעֵצִים סְתָם מִתּוֹךְ שִׁעֲמוּם, וְצָחֲקוּ בְּקוֹל גָּדוֹל וְחָצוּף. אֵלּוּ הָיוּ נְעָרִים שֶׁבָּחֲרוּ לְבַזְבֵּז אֶת זְמַנָּם מִחוּץ לְבֵית הַמִּדְרָשׁ, בְּלִי תּוֹרָה וּבְלִי דֶּרֶךְ אֶרֶץ.

כַּאֲשֶׁר הֵם הִבְחִינוּ בְּרַבִּי שִׁמְעוֹן בֶּן חֲלַפְתָּא הַזָּקֵן, בִּמְקוֹם לַעֲמֹד בְּדֶרֶךְ אֶרֶץ וְלִנְהֹג כָּבוֹד בַּצַּדִּיק הַגָּדוֹל, נִכְנְסָה בָּהֶם רוּחַ שְׁטוּת שֶׁל יֵצֶר הָרַע.

אֶחָד הַנְּעָרִים הִצְבִּיעַ עָלָיו, וּצְחוֹק מְזַלְזֵל בָּקַע מִפִּיו: "הַבִּיטוּ עַל הַזָּקֵן הַזֶּה! מָה הוּא מִסְתַּכֵּל עַל הָאֲדָמָה? אוּלַי הוּא מְדַבֵּר עִם הַנְּמָלִים?!".

נַעַר אַחֵר זָרַק גַּרְגִּיר שֶׁל בֹּץ שֶׁנָּפַל קָרוֹב לְרַגְלָיו שֶׁל הָרַב. אֲחֵרִים מָחֲאוּ כַּפַּיִם בְּלַעַג. "הָרַבָּנִים הָאֵלֶּה," אָמַר נַעַר שְׁלִישִׁי גְּבוֹהַּ, "יוֹדְעִים רַק לִלְמֹד בִּסְפָרִים מְשַׁעַמְמִים. אֲנַחְנוּ – אֲנַחְנוּ נֶהֱנִים מֵהַחַיִּים! מְשַׂחֲקִים, אוֹכְלִים, חָפְשִׁיִּים כְּמוֹ הָרוּחַ!".

הָאַזְהָרָה וְהַגְּזֵרָה הַשְּׁמֵימִית

יְלָדִים, עַכְשָׁו תַּקְשִׁיבוּ טוֹב. רַבִּי שִׁמְעוֹן לֹא כָּעַס בִּגְלַל הַכָּבוֹד הָאִישִׁי שֶׁלּוֹ. הוּא הָיָה עָנָיו מְאֹד. אֲבָל כָּאַב לוֹ מְאֹד עַל כְּבוֹד שָׁמַיִם הַמְּחֻלָּל, וְעַל הַנְּשָׁמוֹת הַטְּהוֹרוֹת שֶׁל הַנְּעָרִים הָאֵלֶּה שֶׁהוֹלְכוֹת לְאִבּוּד.

הוּא עָצַר מֵהֲלִיכָתוֹ. הוּא הִזְדַּקֵּף, וְהִבִּיט בָּהֶם בְּעֵינַיִם חֲכָמוֹת וַעֲמֻקּוֹת שֶׁרָאוּ כְּבָר הַכֹּל

"בָּנַי אֲהוּבַי," אָמַר קוֹלוֹ הֶעָמֹק וְהַמְּהַדְהֵד עַל פְּנֵי הַגִּבְעָה, קוֹל שֶׁגָּרַם לָרוּחַ מַמָּשׁ לַעֲצֹר מִנְּשִׁיבָתָהּ. "אַתֶּם חוֹשְׁבִים שֶׁהַחֵטְא וְהַזִּלְזוּל מְבִיאִים לָכֶם שִׂמְחָה. אַתֶּם מַרְגִּישִׁים מְנַצְּחִים. אֲבָל דְּעוּ לָכֶם כְּלָל גָּדוֹל

'הַחֵטְא, נָעִים וּמָתוֹק רַק בִּתְחִלָּתוֹ – אַךְ בְּסוֹפוֹ הוּא רַע וָמַר מִמָּוֶת!' "

הַנְּעָרִים גִּלְגְּלוּ אֶת עֵינֵיהֶם. הֵם לֹא רָצוּ לְהַקְשִׁיב לַתּוֹכֵחָה. אֲבָל הֵם לֹא יָדְעוּ שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא אוֹהֵב אֶת רַבִּי שִׁמְעוֹן אֲהוּבוֹ, כְּפִי שֶׁכָּתוּב "צַדִּיק גּוֹזֵר – וְהַקָּדוֹשׁ בָּרוּךְ הוּא מְקַיֵּם!". מַה שֶּׁיֵּצֵא מִפִּיו שֶׁל הַצַּדִּיק – מִיָּד יִקְרֶה בַּמְּצִיאוּת!.

"בּוֹאוּ נִבְרַח מִמֶּנּוּ לַיַּעַר!" צָעַק מַנְהִיג הַנְּעָרִים, וְרָץ קָדִימָה.

לְפֶתַע, רָאוּ הַנְּעָרִים לִפְנֵיהֶם שִׂיחַ עֲנָק, מָלֵא בְּפֵרוֹת עֲסִיסִיִּים הַנִּרְאִים מְתוּקִים כִּדְבַשׁ. צֶבַעְ הַפֵּרוֹת הָיָה אָדֹם-סָגֹל וּמַבְרִיק, וְרֵיחַ שֶׁל מַמְתַּקִּים עָלָה מֵהֶם. זֶה הָיָה נִרְאֶה הַמָּקוֹם הֲכִי נָעִים וְכֵיפִי לָשֶׁבֶת בּוֹ וְלִזְלֹל!.

הַלְּכִידָה וְהַמְּרִירוּת

"וָאוּ! הַבִּיטוּ!" צָהֲלוּ הַנְּעָרִים בְּשִׂמְחָה. הֵם רָצוּ פְּנִימָה אֶל תּוֹךְ הַשִּׂיחַ הֶעָבֶה, צוֹחֲקִים עַל הַזָּקֵן שֶׁהִשְׁאִירוּ מֵאָחוֹר. הֵם קָטְפוּ חָפְנֵי פֵּרוֹת וְהִכְנִיסוּ לְפִיהֶם בְּזֹלְלָנוּת.

"אֵיזֶה מָתוֹק!" אָמַר הָאֶחָד. "זֶה הַיּוֹם הֲכִי טוֹב בַּחַיִּים שֶׁלִּי!" לָעַג הַשֵּׁנִי. הַמְּתִיקוּת הָיְתָה כָּל כָּךְ נְעִימָה... בְּדִיּוּק כְּמוֹ שֶׁרַבִּי שִׁמְעוֹן תֵּאֵר! הַתְּחָלָה נְעִימָה.

אֲבָל אָז... קְרַאק! סְנַאפּ!

לְפֶתַע, הַשִּׂיחִים שֶׁסְּבִיבָם כְּאִלּוּ סָגְרוּ עֲלֵיהֶם! הֵם שָׂמוּ לֵב לְפֶתַע שֶׁהַשִּׂיחַ הוּא בְּעֶצֶם סְבַךְ נוֹרָא שֶׁל סִירִים וְקוֹצִים חַדִּים כְּמַחֲטִים מִבַּרְזֶל. הַנְּעָרִים נִסּוּ לָזוּז – אַךְ כָּל תְּנוּעָה קְטַנָּה גָּרְמָה לַקּוֹצִים לִשְׂרֹט אוֹתָם עָמֹק בִּבְשָׂרָם וְלִקְרֹעַ אֶת בִּגְדֵיהֶם!

וְזֶה עוֹד לֹא הַכֹּל! טַעַם הַפְּרִי שֶׁבְּפִיהֶם הִשְׁתַּנָּה לְפֶתַע מֵקָצֶה לְקָצֶה. הַמְּתִיקוּת הִתְחַלְּפָה בְּטַעַם מַר, צוֹרֵב וְנִתְעָב! פִּיהֶם הִתְמַלֵּא בִּמְרִירוּת שֶׁעָשְׂתָה לָהֶם בְּחִילָה נוֹרָאִית. הֵם נִסּוּ לִירֹק, לִצְעֹק, אַךְ כָּל צְעָקָה רַק סָבְכָה אוֹתָם יוֹתֵר בַּקּוֹצִים הֶחָזָקִים. הַנְּעָרִים הַגֵּאִים, שֶׁעַד לִפְנֵי שְׁנִיָּה צָחֲקוּ וְזִלְזְלוּ בַּתּוֹרָה – עָמְדוּ כָּעֵת בּוֹכִים, אֲחוּזֵי אֵימָה, וְדִמְעוֹתֶיהֶם שׁוֹתְתוֹת עַל לֶחְיֵיהֶם הַשְּׂרוּטוֹת.

יַד שֶׁל חֶסֶד וַהֲבָנָה

"הַצִּילוּ! מִישֶׁהוּ! בְּבַקָּשָׁה!" קָרְאוּ הַנְּעָרִים בְּקוֹל שָׁבוּר.

צְעָדִים שְׁקֵטִים נִשְׁמְעוּ מֵאֲחוֹרֵיהֶם. רַבִּי שִׁמְעוֹן בֶּן חֲלַפְתָּא הִתְקָרֵב הָיָה זֶה הָאִישׁ שֶׁבּוֹ פָּגְעוּ רַק עַכְשָׁו. אָדָם רָגִיל הָיָה אָמַר לָהֶם: "מַגִּיעַ לָכֶם! אָמַרְתִּי לָכֶם!".

אַךְ רַבִּי שִׁמְעוֹן הָיָה צַדִּיק מָלֵא בְּרַחֲמִים. הוּא פָּסַע בַּעֲדִינוּת בֵּין הַקּוֹצִים (וּלְפֶלֶא, הַקּוֹצִים מַמָּשׁ פִּנּוּ לוֹ אֶת הַדֶּרֶךְ וְלֹא נָגְעוּ בְּקָצֵה בִּגְדּוֹ). הוּא הוֹשִׁיט אֶת יָדוֹ הַחַמָּה אֶל מַנְהִיג הַנְּעָרִים הַבּוֹכֶה, וְחִלֵּץ אוֹתוֹ בָּזֶה אַחַר זֶה לְלֹא פֶּגַע מִתּוֹךְ הַסְּבַךְ.

הַנְּעָרִים יָשְׁבוּ עַל הָאֲדָמָה הַטּוֹבָה, מִתְנַשְּׁמִים וְרוֹעֲדִים.

הֵם נָשְׂאוּ אֶת עֵינֵיהֶם לְמַעְלָה וְרָאוּ אֶת פָּנָיו הַמְּאִירוֹת שֶׁל רַבִּי שִׁמְעוֹן בֶּן חֲלַפְתָּא, עוֹמֵד מֵעֲלֵיהֶם. שֶׁקֶט עָמֹק שָׂרַר, עַד שֶׁנִּתָּן הָיָה לִשְׁמֹעַ צִרְצַר יְחִידִי מְנַסֵּר בַּדֶּשֶׁא.

הָרַב הִתְכּוֹפֵף אֲלֵיהֶם, לִטֵּף אֶת רֹאשָׁם בְּרֹךְ, וְלָחַשׁ: "הַאִם הֵבַנְתֶּם כָּעֵת, בָּנַי הַיְּקָרִים? הַחֵטְא, הַזִּלְזוּל, הַגַּאֲוָה... הֵם נִרְאִים נוֹצְצִים וּמְתוּקִים כְּשֶׁרָצִים אֲלֵיהֶם. אֲבָל מַה הֵם הִשְׁאִירוּ לָכֶם? פֶּצַע, שְׂרִיטוֹת, וְטַעַם מַר. רַק תּוֹרַת אֱמֶת, רַק כְּבוֹד שָׁמַיִם וְאַהֲבַת יִשְׂרָאֵל, הֵם דְּבָרִים שֶׁהֵם תָּמִיד טוֹבִים, עַכְשָׁו וּלְנֶצַח."

הַנְּעָרִים בָּכוּ – אַךְ הַפַּעַם מִתּוֹךְ חֲרָטָה עֲמֻקָּה. הֵם בִּקְּשׁוּ מֵרַבִּי שִׁמְעוֹן סְלִיחָה עַל חֻצְפָּתָם, וְהִבְטִיחוּ מֵהַיּוֹם וָהָלְאָה לִנְהֹג בְּכָבוֹד וּבְיִרְאַת שָׁמַיִם מִפְּנֵי תַּלְמִידֵי חֲכָמִים. וְרַבִּי שִׁמְעוֹן? הוּא שָׂמַח – כִּי הִצְלִיחַ, בְּחָכְמָתוֹ וּבְאַהֲבָתוֹ, לְהַצִּיל כַּמָּה נְשָׁמוֹת אֲבוּדוֹת.

מקורות: מדרש קהלת רבה פרשה ג, ו. (על הפסוק "את הכל עשה יפה בעתו"). מדרש שוחר טוב (תהילים) מזמור צ"ב, ח. מופיע במרוכז גם ב "ספר האגדה" - מעשי חכמים. סדרת "בדרכי אבותינו" - מנחם גץ, חלק ג' עמוד 92 ("רבי שמעון בן חלפתא - הנערים שזלזלו בצדיק ועונשם").

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “יְלָדִים, דַּמְיְנוּ עַצְמְכֶם צוֹעֲדִים בִּשְׁבִילֵי אֶרֶץ יִשְׂרָאֵל שֶׁל יְמֵי קֶדֶם.” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “הֵם בָּעֲטוּ בָּאֲבָנִים, תָּלְשׁוּ עֲנָפִים יְרֻקִּים מֵהָעֵצִים סְתָם מִתּוֹךְ שִׁעֲמוּם, וְצָחֲקוּ בְּקוֹל גָּדוֹל וְחָצוּף.”?
ג. כיצד הגיב ר' שמעון בן חלפתא למצב המתואר: “אֲבָל כָּאַב לוֹ מְאֹד עַל כְּבוֹד שָׁמַיִם הַמְּחֻלָּל, וְעַל הַנְּשָׁמוֹת הַטְּהוֹרוֹת שֶׁל הַנְּעָרִים הָאֵלֶּה שֶׁהוֹלְכוֹת לְאִבּוּד.”?
ד. מה השתנה בין “לְפֶתַע, רָאוּ הַנְּעָרִים לִפְנֵיהֶם שִׂיחַ עֲנָק, מָלֵא בְּפֵרוֹת עֲסִיסִיִּים הַנִּרְאִים מְתוּקִים כִּדְבַשׁ.” לבין “הֵם נִסּוּ לִירֹק, לִצְעֹק, אַךְ כָּל צְעָקָה רַק סָבְכָה אוֹתָם יוֹתֵר בַּקּוֹצִים הֶחָזָקִים.”?
ה. כיצד הגיע הסיפור אל האירוע: “רַק תּוֹרַת אֱמֶת, רַק כְּבוֹד שָׁמַיִם וְאַהֲבַת יִשְׂרָאֵל, הֵם דְּבָרִים שֶׁהֵם תָּמִיד טוֹבִים, עַכְשָׁו וּלְנֶצַח.”?

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “הנערים שזלזלו בצדיק ועונשם” על ר' שמעון בן חלפתא? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “'הַחֵטְא, נָעִים וּמָתוֹק רַק בִּתְחִלָּתוֹ – אַךְ בְּסוֹפוֹ הוּא רַע וָמַר מִמָּוֶת!' ”.
ג. אילו ר' שמעון בן חלפתא לא היה פועל כפי שמתואר ב־“הֵם נִסּוּ לִירֹק, לִצְעֹק, אַךְ כָּל צְעָקָה רַק סָבְכָה אוֹתָם יוֹתֵר בַּקּוֹצִים הֶחָזָקִים.” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “רַק תּוֹרַת אֱמֶת, רַק כְּבוֹד שָׁמַיִם וְאַהֲבַת יִשְׂרָאֵל, הֵם דְּבָרִים שֶׁהֵם תָּמִיד טוֹבִים, עַכְשָׁו וּלְנֶצַח.” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. יְלָדִים, דַּמְיְנוּ עַצְמְכֶם צוֹעֲדִים בִּשְׁבִילֵי אֶרֶץ יִשְׂרָאֵל…
2. אֵלּוּ הָיוּ נְעָרִים שֶׁבָּחֲרוּ לְבַזְבֵּז אֶת זְמַנָּם מִחוּץ לְבֵית…
3. הוּא הִזְדַּקֵּף, וְהִבִּיט בָּהֶם בְּעֵינַיִם חֲכָמוֹת וַעֲמֻקּוֹת…
4. זֶה הָיָה נִרְאֶה הַמָּקוֹם הֲכִי נָעִים וְכֵיפִי לָשֶׁבֶת בּוֹ…
5. רַבִּי שִׁמְעוֹן בֶּן חֲלַפְתָּא הִתְקָרֵב הָיָה זֶה הָאִישׁ שֶׁבּוֹ…
6. הוּא שָׂמַח – כִּי הִצְלִיחַ, בְּחָכְמָתוֹ וּבְאַהֲבָתוֹ, לְהַצִּיל…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “אֲבָל כָּאַב לוֹ מְאֹד עַל כְּבוֹד שָׁמַיִם הַמְּחֻלָּל, וְעַל הַנְּשָׁמוֹת הַטְּהוֹרוֹת שֶׁל הַנְּעָרִים הָאֵלֶּה שֶׁהוֹלְכוֹת…”.
מכתב: כתבו לר' שמעון בן חלפתא בעקבות “הֵם רָצוּ פְּנִימָה אֶל תּוֹךְ הַשִּׂיחַ הֶעָבֶה, צוֹחֲקִים עַל הַזָּקֵן שֶׁהִשְׁאִירוּ מֵאָחוֹר.”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “הוּא פָּסַע בַּעֲדִינוּת בֵּין הַקּוֹצִים (וּלְפֶלֶא, הַקּוֹצִים מַמָּשׁ פִּנּוּ לוֹ אֶת הַדֶּרֶךְ וְלֹא נָגְעוּ בְּקָצֵה…”.
דיווח: כתבו ידיעה קצרה שכותרתה “הנערים שזלזלו בצדיק ועונשם”.

המחזה

רגע מן הסיפורהדמויותמה יאמרו או יעשו?
אֶחָד הַנְּעָרִים הִצְבִּיעַ עָלָיו, וּצְחוֹק מְזַלְזֵל בָּקַע מִפִּיו: "הַבִּיטוּ עַל הַזָּקֵן…
הֵם נִסּוּ לִירֹק, לִצְעֹק, אַךְ כָּל צְעָקָה רַק סָבְכָה אוֹתָם יוֹתֵר בַּקּוֹצִים הֶחָזָקִים.

7. חיבור לחיים מתוך המעשה

א. מה אפשר ללמוד מן הדרך שבה ר' שמעון בן חלפתא פעל באירוע “לְפֶתַע, רָאוּ הַנְּעָרִים לִפְנֵיהֶם שִׂיחַ עֲנָק, מָלֵא בְּפֵרוֹת עֲסִיסִיִּים הַנִּרְאִים מְתוּקִים כִּדְבַשׁ.”?
