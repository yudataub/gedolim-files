תלמידו ר' יוחנן הסנדלר למד בהבלעה

דף עבודה בעקבות רבי עקיבא
שם: כיתה:

המעשה — הסיפור המלא

פְּתִיחָה: חֹשֶׁךְ בְּבֵית הַמִּדְרָשׁ וְחֹשֶׁךְ בַּכֶּלֶא

יְלָדִים, נַסּוּ לְדַמְיֵן רֵיחַ שֶׁל מָקוֹם שֶׁלֹּא פָּתְחוּ בּוֹ חַלּוֹן שָׁנִים רַבּוֹת. רֵיחַ טְחָב, אֲדָמָה רְטֻבָּה וַאֲוִיר מַחֲנִיק. עַכְשָׁו, הוֹסִיפוּ לַדִּמְיוֹן גַּם קוֹלוֹת שֶׁל צִלְצוּל שַׁרְשְׁרָאוֹת בַּרְזֶל כְּבֵדוֹת שֶׁנִּגְרָרוֹת עַל רִצְפַּת אֶבֶן קָרָה, וּצְעָקוֹת בְּשָׂפָה זָרָה שֶׁל חַיָּלִים רוֹמָאִים מְפַחֲדִים שֶׁשּׁוֹמְרִים בַּפֶּתַח. כָּךְ נִרְאָה הַכֶּלֶא הָרוֹמָאִי תַּחַת הָאֲדָמָה, בּוֹ הָיָה כָּלוּא לֹא אַחֵר מֵאֲשֶׁר... רַבִּי עֲקִיבָא .

לָמָּה הוּא שָׁם? כִּי הוּא מָסַר אֶת נַפְשׁוֹ! הָרוֹמָאִים הָרְשָׁעִים גָּזְרוּ שֶׁאָסוּר לִלְמֹד וּלְלַמֵּד תּוֹרָה, אַךְ רַבִּי עֲקִיבָא אָמַר: "יְהוּדִי בְּלִי תּוֹרָה, הוּא כְּמוֹ דָּג מִחוּץ לַמַּיִם. הוּא לֹא יָכוֹל לִחְיוֹת!". הוּא הִקְהִיל קְהִלּוֹת בָּרַבִּים וְלִמֵּד תּוֹרָה, עַד שֶׁהָרוֹמָאִים תָּפְסוּ אוֹתוֹ וְהִשְׁלִיכוּ אוֹתוֹ לְבֵית הָאֲסוּרִים.

בְּאוֹתָהּ עֵת, בְּבֵית הַמִּדְרָשׁ, הָאֲוִירָה הָיְתָה עֲצוּבָה. הַתַּלְמִידִים יָשְׁבוּ מוּל שֻׁלְחָנוֹת הָעֵץ הָעַתִּיקִים, קוֹל לִמּוּדָם הָיָה שָׁקֵט מִתָּמִיד. הַכִּסֵּא שֶׁל רַבִּי עֲקִיבָא, בָּרֹאשׁ, הָיָה רֵיק.

לְפֶתַע, בְּאֶמְצַע הַלִּמּוּד, נִתְקְלוּ הַתַּלְמִידִים בִּשְׁאֵלָה הִלְכָתִית קָשָׁה מְאֹד. זֶה הָיָה מִקְרֶה מְסֻבָּךְ שֶׁל כַּפִּית חֲלָבִית שֶׁהִשְׁתַּמְּשׁוּ בָּהּ בְּסִיר בָּשָׂר רָתוּחַ בְּמַצָּבִים מְסֻיָּמִים.

אֶחָד הַתַּלְמִידִים טָעַן: "הַסִּיר הַזֶּה אָסוּר!". תַּלְמִיד שֵׁנִי קָם וְאָמַר: "מַה פִּתְאוֹם, זֶה כָּשֵׁר!".

הֵם חִפְּשׂוּ בַּסְּפָרִים, הֵם מָסְרוּ סְבָרוֹת חֲזָקוֹת, אַךְ אִישׁ לֹא הִצְלִיחַ לְהַגִּיעַ לְמַסְקָנָה וַדָּאִית. הֵם הָיוּ צְמֵאִים לָדַעַת אֶת רְצוֹן ה' בְּדִיּוּק גָּמוּר.

"אֵין מִי שֶׁיּוּכַל לַעֲנוֹת לָנוּ עַל הַשְּׁאֵלָה הַזּוֹ," אָמַר תַּלְמִיד אֶחָד וּדְמָעוֹת נוֹצְצוּ בְּעֵינָיו, "מִלְּבַד רַבֵּנוּ, רַבִּי עֲקִיבָא. אַךְ מִי יָכוֹל לָגֶשֶׁת אֵלָיו? הַכֶּלֶא שׁוֹרֵץ חַיָּלִים רוֹמָאִים חֲמוּשִׁים בַּחֲרָבוֹת!".

הַהַחְלָטָה שֶׁל רַבִּי יוֹחָנָן

מִי שֶׁיָּשַׁב כָּל אוֹתָהּ שָׁעָה וְהִקְשִׁיב בְּשֶׁקֶט, הָיָה תַּלְמִיד אָהוּב וּמָסוּר עַד מְאֹד – רַבִּי יוֹחָנָן הַסַּנְדְּלָר .

רַבִּי יוֹחָנָן לֹא הָיָה מוּכָן לְוַתֵּר. "אָסוּר שֶׁהַתּוֹרָה תֵּעָצֵר!" הוּא קָרָא לְפֶתַע, קוֹלוֹ צָלוּל וּמָלֵא אֱמוּנָה. "כַּאֲשֶׁר אֲנַחְנוּ רוֹצִים לְבָרֵר אֶת הָאֱמֶת, אֵין דָּבָר שֶׁיַּעֲמֹד בְּדַרְכֵּנוּ. הַקָּדוֹשׁ בָּרוּךְ הוּא יִשְׁמֹר עָלַי. אֲנִי – אֵלֵךְ אֶל רַבִּי עֲקִיבָא לִשְׁאֹל אוֹתוֹ!".

יְלָדִים, דַּמְיְנוּ כַּמָּה אַמִּיץ הָיָה רַבִּי יוֹחָנָן! אִם שׁוֹמֵר רוֹמָאִי אֶחָד תּוֹפֵס אוֹתוֹ מְדַבֵּר דִּבְרֵי תּוֹרָה עִם רַבִּי עֲקִיבָא, הוּא עַצְמוֹ הָיָה מֻשְׁלָךְ לַצִּינוֹק מִיָּד בְּלִי רַחֲמִים.

מֶה עָשָׂה רַבִּי יוֹחָנָן? הוּא הֵכִין לְעַצְמוֹ תַּחְפֹּשֶׂת. הוּא לָבַשׁ בְּגָדִים שֶׁל סוֹחֵר פָּשׁוּט בַּשּׁוּק, חָבַשׁ מִגְבַּעַת גְּדוֹלָה שֶׁהִסְתִּירָה אֶת פָּנָיו הַטְּהוֹרוֹת, לָקַח אַרְגָּז עֵץ עַל הַכָּתֵף שֶׁלּוֹ, וּמִלֵּא אוֹתוֹ בְּסַנְדָּלִים מֵעוֹר, מְחָטִים חַדּוֹת, מַסְרֵגוֹת, כִּישׁוֹרִים (כְּלִי לִטְוִיַּת חֹטִים) וּשְׁאָר כְּלֵי מְלָאכָה.

הוּא יָצָא אֶל רְחוֹבוֹת הָעִיר, שׁוֹאֵף אֲוִיר עָמֹק בְּלִבּוֹ, וְתַחַת שְׂפָתָיו לָחַשׁ תְּפִלָּה עֲמֻקָּה: "רִבּוֹנוֹ שֶׁל עוֹלָם, עֲשֵׂה שֶׁמַּסָּע זֶה לִכְבוֹד תּוֹרָתְךָ, יַצְלִיחַ".

הַשִּׁירָה מוּל חֲדַר הַכֶּלֶא

רַבִּי יוֹחָנָן הִתְקָרֵב אֶל מִבְנֵה הָאֶבֶן הֶעָצוּם וְהֶאָפֵל שֶׁל בֵּית הָאֲסוּרִים. שְׁנֵי שׁוֹמְרִים גְּדוֹלִים לְבוּשִׁים בְּשִׁרְיוֹנוֹת בַּרְזֶל, עִם קְסָדוֹת נוֹצְצוֹת וַחֲנִיתוֹת שְׁלוּפוֹת בִּידֵיהֶם, עָמְדוּ בַּפֶּתַח וּבָחֲנוּ בְּמַבָּט אַכְזָרִי כָּל מִי שֶׁחָלַף עַל פְּנֵיהֶם. רַבִּי יוֹחָנָן לֹא הִרְאָה סִימָן שֶׁל פַּחַד.

הוּא הִתְחִיל לָלֶכֶת מִסָּבִיב לַחוֹמוֹת, מוֹצֵא אֶת הַחַלּוֹן הַקָּטָן שֶׁנִּמְצָא מַמָּשׁ צָמוּד לְצִינוֹקוֹ שֶׁל רַבִּי עֲקִיבָא. שָׁם הוּא עָצַר.

הוּא יָדַע שֶׁאִם יִגַּשׁ לַחַלּוֹן וְיִשְׁאַל שְׁאֵלָה בַּהֲלָכָה, הַשּׁוֹמְרִים יִשְׁמְעוּ אוֹתוֹ מִיָּד. מֶה עָשָׂה אוֹתוֹ תַּלְמִיד פִּקֵּחַ? הוּא הִתְחִיל לְשַׁוֵּק (לִמְכֹּר) אֶת הַסְּחוֹרָה שֶׁלּוֹ כְּמוֹ כָּל רוֹכֵל בַּשּׁוּק, אֲבָל... בְּתוֹךְ דִּבְרֵי הַמְּכִירָה הוּא הִבְלִיעַ (הֶחְבִּיא) אֶת הַשְּׁאֵלָה הַהִלְכָתִית.

הוּא שָׂם כַּף יָד עַל פִּיו כְּדֵי שֶׁקּוֹלוֹ יִשָּׁמַע לְמֵרָחוֹק, וְהִתְחִיל לְנַגֵּן בְּקוֹל שֶׁל מוֹכֵר:

"סַנְדָּלִים חֲדָשִׁים! מַסְרֵגוֹת יָפוֹת חֲדָשׁוֹת לַמְּכִירָה! מִי חָסֵר לוֹ מַחַט? הַאִם לַסּוֹהֵר חָסֵר כִּישׁוֹר לִקְנִיָּה?!..."

וְאָז – בְּלִי לַעֲצֹר אֶת הַמַּנְגִּינָה, הוּא צָעַק בְּאוֹתוֹ קוֹל:

"...כַּפִּית חֲלָבִית שֶׁנָּפְלָה לְסִיר רָתוּחַ שֶׁל בָּשָׂר מַה דִּינָהּ?!... מִי רוֹצֶה לִקְנוֹת סַנְדָּלִים יָפִים וְחֲדָשִׁים הַיּוֹם?!..."

הַשּׁוֹמְרִים הָרוֹמָאִים עָמְדוּ שָׁם, פִּהֲקוּ מִשִּׁעֲמוּם, וּבִכְלָל לֹא שָׂמוּ לֵב לַמִּשְׁפָּט הַמּוּזָר בְּאֶמְצַע שִׁירַת הַרוֹכֵל. בִּשְׁבִילָם זֶה נִשְׁמַע כְּמוֹ רְשִׁימָה אֲרֻכָּה שֶׁל סִירִים וְכַפִּיּוֹת לַמְּכִירָה בַּשּׁוּק.

אֲבָל... בְּתוֹךְ בֵּית הָאֲסוּרִים הֶחָשׁוּךְ... קָמָה דְּמוּת צְנוּמָה. זֶה הָיָה רַבִּי עֲקִיבָא. אָזְנָיו הַטְּהוֹרוֹת הָיוּ רְגִישׁוֹת לְדִבְרֵי תּוֹרָה יוֹתֵר מִכָּל דָּבָר אַחֵר. הוּא הִכִּיר אֶת הַקּוֹל הַמִּתְנַגֵּן! זֶהוּ קוֹלוֹ שֶׁל תַּלְמִידוֹ הָאָהוּב יוֹחָנָן הַסַּנְדְּלָר! רַבִּי עֲקִיבָא הֵבִין מִיָּד, שֶׁהַתַּלְמִיד סִכֵּן אֶת חַיָּיו שֶׁלּוֹ כְּדֵי לְהַגִּיעַ עַד כָּאן. שְׂפָתָיו שֶׁל רַבִּי עֲקִיבָא הֶעֱלוּ חִיּוּךְ קָטָן שֶׁל אַהֲבָה וּמְסִירוּת שֶׁל רַב לְתַלְמִיד. "אֵיזוֹ מְסִירוּת נֶפֶשׁ יֵשׁ לַתַּלְמִידִים שֶׁלִּי לָדַעַת אֶת אֱמֶת הַתּוֹרָה," הִרְהֵר בְּלִבּוֹ.

תְּשׁוּבָה מֵאֲחוֹרֵי הַסּוֹרְגִים

רַבִּי עֲקִיבָא יָדַע מִיָּד אֶת הַתְּשׁוּבָה לַשְּׁאֵלָה הַסְּבוּכָה בְּדִינֵי בָּשָׂר וְחָלָב. אַךְ אֵיךְ יוֹדִיעַ אוֹתָהּ לְרַבִּי יוֹחָנָן מִבְּלִי שֶׁהַשּׁוֹמְרִים יִשְׁמְעוּ עַל מָה מְדֻבָּר וְיִכְעֲסוּ שֶׁהוּא עֲדַיִן מְלַמֵּד תּוֹרָה?

גַּם רַבִּי עֲקִיבָא הִשְׁתַּמֵּשׁ בְּחָכְמָה עֲצוּמָה וּ"בְהַבְלָעָה".

הוּא הִתְקָרֵב אֶל אֶשְׁנַב הַבַּרְזֶל הַקָּטָן וּפָנָה לַ"רוֹכֵל" שֶׁלְּמַטָּה, וְכָךְ קָרָא לְעֶבְרוֹ:

"הֵי אַתָּה! סוֹחֵר נִכְבָּד! הַאִם יֵשׁ לְךָ מַחַט? הַאִם יֵשׁ לְךָ כִּישׁוֹר?... כָּשֵׁר! ... הַאִם יֵשׁ לְךָ כְּלֵי בַּרְזֶל לִמְכֹּר?..."

רַבִּי עֲקִיבָא שָׁאַל לִכְאוֹרָה אִם יֵשׁ כִּישׁוֹר עֲבוּרוֹ לַכֶּלֶא, וְאָז הוֹסִיף "כָּשֵׁר!".

הַשּׁוֹמְרִים צָחֲקוּ וְחָשְׁבוּ שֶׁהַזָּקֵן מְבַקֵּשׁ צִיּוּד כָּשֵׁר וְנָקִי מֵהַסּוֹחֵר בַּשּׁוּק. אַךְ רַבִּי יוֹחָנָן שָׁמַע אֶת הַתְּשׁוּבָה הַנִּכְסֶפֶת. הוּא הֵבִין! רַבִּי עֲקִיבָא חָתַךְ אֶת הַהֲלָכָה וְאָמַר – כָּשֵׁר! הַכַּפִּית לֹא אָסְרָה אֶת הַמָּרָק בְּאוֹתוֹ מַצָּב סְפֵּצִיפִי.

רַבִּי יוֹחָנָן לִטֵּף אֶת תִּיק הָעוֹר שֶׁלּוֹ. הוּא צָעַק צְעָקָה אַחֲרוֹנָה שֶׁל שִׂמְחָה לִכְאוֹרָה אֶל הַקּוֹנִים: "הַסְּחוֹרָה מְעֻלָּה, הַכֹּל כָּשֵׁר וְנָקִי!" וְהִסְתּוֹבֵב אָחוֹרָה לַחֲזֹר לְבֵית הַמִּדְרָשׁ.

כַּאֲשֶׁר נִכְנַס רַבִּי יוֹחָנָן הַסַּנְדְּלָר עִם קֻפְסַת הָעֵץ עַל הַכָּתֵף חֲזָרָה לְבֵית הַמִּדְרָשׁ, הַדְּמָמָה הָפְכָה לְהַתְרַגְּשׁוּת עֲצוּמָה. הוּא חָלַץ אֶת הַכּוֹבַע, הִבִּיט בְּחַבְרַי הַתַּלְמִידִים וְאָמַר לָהֶם דָּבָר אֶחָד: "יֵשׁ לָנוּ פְּסַק הֲלָכָה מֵרַבּוֹתֵינוּ. זֶה כָּשֵׁר!".

הַתַּלְמִידִים יָשְׁבוּ מוּל הַסְּפָרִים בְּשִׂמְחָה וּבְיִרְאַת כָּבוֹד לְרַבָּם שֶׁגַּם בְּתוֹךְ קִיר הָאֶבֶן שֶׁל הַמַּחֲשַׁכִּים – אוֹרוֹ וּפְסָקָיו הוֹסִיפוּ לְהָאִיר לְעַם יִשְׂרָאֵל אֶת הַדֶּרֶךְ.

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “תלמידו ר' יוחנן הסנדלר למד בהבלעה” על רבי עקיבא? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “שְׁנֵי שׁוֹמְרִים גְּדוֹלִים לְבוּשִׁים בְּשִׁרְיוֹנוֹת בַּרְזֶל, עִם קְסָדוֹת נוֹצְצוֹת וַחֲנִיתוֹת שְׁלוּפוֹת בִּידֵיהֶם, עָמְדוּ בַּפֶּתַח וּבָחֲנוּ…”.
ג. אילו רבי עקיבא לא היה פועל כפי שמתואר ב־“שְׂפָתָיו שֶׁל רַבִּי עֲקִיבָא הֶעֱלוּ חִיּוּךְ קָטָן שֶׁל אַהֲבָה וּמְסִירוּת שֶׁל רַב לְתַלְמִיד.” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “הַכַּפִּית לֹא אָסְרָה אֶת הַמָּרָק בְּאוֹתוֹ מַצָּב סְפֵּצִיפִי.” מחזק את המסר של הסיפור?

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “אִם שׁוֹמֵר רוֹמָאִי אֶחָד תּוֹפֵס אוֹתוֹ מְדַבֵּר דִּבְרֵי תּוֹרָה עִם רַבִּי עֲקִיבָא, הוּא עַצְמוֹ הָיָה מֻשְׁלָךְ לַצִּינוֹק…”.
מכתב: כתבו לרבי עקיבא בעקבות “הַשּׁוֹמְרִים הָרוֹמָאִים עָמְדוּ שָׁם, פִּהֲקוּ מִשִּׁעֲמוּם, וּבִכְלָל לֹא שָׂמוּ לֵב לַמִּשְׁפָּט הַמּוּזָר בְּאֶמְצַע שִׁירַת…”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “רַבִּי עֲקִיבָא שָׁאַל לִכְאוֹרָה אִם יֵשׁ כִּישׁוֹר עֲבוּרוֹ לַכֶּלֶא, וְאָז הוֹסִיף "כָּשֵׁר!".”.
דיווח: כתבו ידיעה קצרה שכותרתה “תלמידו ר' יוחנן הסנדלר למד בהבלעה”.

המחזה

רגע מן הסיפורהדמויותמה יאמרו או יעשו?
אֵין מִי שֶׁיּוּכַל לַעֲנוֹת לָנוּ עַל הַשְּׁאֵלָה הַזּוֹ," אָמַר תַּלְמִיד אֶחָד וּדְמָעוֹת…
שְׂפָתָיו שֶׁל רַבִּי עֲקִיבָא הֶעֱלוּ חִיּוּךְ קָטָן שֶׁל אַהֲבָה וּמְסִירוּת שֶׁל רַב לְתַלְמִיד.
