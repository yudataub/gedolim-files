סיפור 101 • ר' חנינא בר חמא • החרוב הענק והפרי שמילא ידו בדבש

החרוב הענק והפרי שמילא ידו בדבש

דף עבודה בעקבות ר' חנינא בר חמא
שם: כיתה:

המעשה — הסיפור המלא

פְּתִיחָה: הַמַּסָּע הַגָּדוֹל מִבָּבֶל

יְלָדִים, נַסּוּ לְדַמְיֵן עִיר גְּדוֹלָה, עֲשִׁירָה וּמְלֵאָה בְּנַהֲרוֹת מַיִם מְתוּקִים. כָּךְ נִרְאֲתָה אֶרֶץ בָּבֶל לִפְנֵי שָׁנִים רַבּוֹת. הָיוּ שָׁם בָּתִּים יָפִים, שְׁוָקִים עֲמוּסִים בְּכָל טוּב וּפַרְנָסָה מְצוּיָה בְּשֶׁפַע.

בְּבָבֶל גָּר תַּלְמִיד חָכָם קָדוֹשׁ וְעָצוּם, שְׁמוֹ הָיָה רַבִּי חֲנִינָא בַּר חָמָא . אֲבָל לַמְרוֹת שֶׁהָיָה לוֹ כָּל מָה שֶׁאָדָם צָרִיךְ, הַלֵּב שֶׁלּוֹ לֹא מָצָא מְנוּחָה. בַּלֵּילוֹת, כַּאֲשֶׁר הָיָה לוֹמֵד תּוֹרָה לְאוֹר הַנֵּר, הָיָה מֵרִים אֶת עֵינָיו לַשָּׁמַיִם וּמִתְגַּעְגֵּעַ לְמָקוֹם אֶחָד – לְאֶרֶץ יִשְׂרָאֵל.

"מָה אֲנִי עוֹשֶׂה כָּאן בְּאֶרֶץ נָכְרִיָּה?" אָמַר יוֹם אֶחָד לִבְנֵי בֵּיתוֹ. "הַקָּדוֹשׁ בָּרוּךְ הוּא נָתַן לָנוּ אֶרֶץ קְדוֹשָׁה, אֶרֶץ זָבַת חָלָב וּדְבַשׁ. אֲנַחְנוּ עוֹזְבִים הַכֹּל וְעוֹלִים לְאֶרֶץ יִשְׂרָאֵל!"

הַדֶּרֶךְ מִבָּבֶל לְאֶרֶץ יִשְׂרָאֵל הָיְתָה קָשָׁה וּמַפְחִידָה. חֳדָשִׁים שֶׁל הֲלִיכָה בַּמִּדְבָּר, תַּחַת שֶׁמֶשׁ קוֹפַחַת וְלוֹהֶטֶת שֶׁשָּׂרְפָה אֶת הָעוֹר. רוּחַ חַמָּה נָשְׁבָה וְהֵעִיפָה גַּרְגְּרֵי חוֹל יָבֵשׁ יָשָׁר לְתוֹךְ הָעֵינַיִם, וְהַגָּרוֹן הָיָה נִחָר מִצָּמָא. אֲבָל רַבִּי חֲנִינָא וּבְנוֹ, שֶׁצָּעֲדוּ יַחַד, לֹא וִתְּרוּ. כָּל צַעַד קֵרֵב אוֹתָם לְאַרְמוֹנוֹ שֶׁל הַמֶּלֶךְ, לְאֶרֶץ הַקֹּדֶשׁ.

נְשִׁיקוֹת לָאֲבָנִים וְתִקּוּן הַשְּׁבִילִים

וְהִנֵּה, בְּבֹקֶר מְעֻרְפָּל אֶחָד, הֵם עָבְרוּ אֶת הַגְּבוּל וְנִכְנְסוּ לְאֶרֶץ יִשְׂרָאֵל. הָרֵי הַגָּלִיל הִתְנַשְּׂאוּ מוּלָם, וְרֵיחַ מְיֻחָד, רֵיחַ שֶׁל אֲדָמָה קְדוֹשָׁה רְטֻבָּה מִטַּל בֹּקֶר, מִלֵּא אֶת הָאֲוִיר. בָּרֶגַע שֶׁדָּרְכוּ רַגְלָיו שֶׁל רַבִּי חֲנִינָא עַל הָאֲדָמָה, הוּא לֹא עָמַד בַּהִתְרַגְּשׁוּת. הוּא נָפַל עַל בִּרְכָּיו, קָרָא: "מוֹדֶה אֲנִי לְפָנֶיךָ, רִבּוֹנוֹ שֶׁל עוֹלָם!", וְהִתְחִיל לְנַשֵּׁק אֶת סַלְעֵי הָאָרֶץ אֶחָד אֶחָד.

אֲבָל כְּשֶׁהִתְחִילוּ לָלֶכֶת בַּשְּׁבִילִים, שָׂם לֵב רַבִּי חֲנִינָא שֶׁהַדְּרָכִים מְשֻׁבָּשׁוֹת. הָאָרֶץ הָיְתָה עֲדַיִן תַּחַת שִׁלְטוֹן הָרוֹמָאִים הָרְשָׁעִים, שֶׁהֶחֱרִיבוּ מְקוֹמוֹת רַבִּים. אֲבָנִים גְּדוֹלוֹת, בּוֹרוֹת וְסַלְעִים חַדִּים חָסְמוּ אֶת הַדֶּרֶךְ.

אָדָם רָגִיל הָיָה מִתְלוֹנֵן: "אוּף! אֵיזוֹ דֶּרֶךְ רָעָה! הָרַגְלַיִם שֶׁלִּי כּוֹאֲבוֹת!".

אֲבָל הַצַּדִּיק רַבִּי חֲנִינָא בַּר חָמָא לֹא דִּבֵּר כָּךְ מֵעוֹלָם עַל אֶרֶץ יִשְׂרָאֵל. מָה עָשָׂה? מִתּוֹךְ כָּבוֹד עָצוּם לָאָרֶץ, הוּא הָיָה הוֹלֵךְ בַּדְּרָכִים, וּבְמוֹ יָדָיו סִקֵּל אֶת הָאֲבָנִים וְסָתַם אֶת הַבּוֹרוֹת בַּעֲפַר הָאָרֶץ. "כְּדֵי שֶׁשּׁוּם יְהוּדִי לֹא יִמְעַד," אָמַר לִבְנוֹ בְּעוֹדוֹ גּוֹרֵר סֶלַע כָּבֵד שֶׁשָּׂרַט אֶת יָדָיו, "וּכְדֵי שֶׁכֻּלָּם יִרְאוּ כַּמָּה שְׁבִילֵי אֶרֶץ יִשְׂרָאֵל יָפִים וִישָׁרִים!". הַקָּדוֹשׁ בָּרוּךְ הוּא רָאָה אֶת הָאַהֲבָה הָעֲצוּמָה הַזּוֹ, וְשָׂמַח בְּבָנָיו הַשָּׁבִים לִגְבוּלָם.

הֶחָרוּב הָעֲנָק בְּאֶמְצַע הַשָּׂדֶה

בְּאַחַד הַיָּמִים, טִיְּלוּ רַבִּי חֲנִינָא וּבְנוֹ בְּאֵזוֹר מְיֻחָד בְּאֶרֶץ יִשְׂרָאֵל. הֵם צָעֲדוּ בַּשָּׂדֶה עַד שֶׁלְּפֶתַע, עָצַר רַבִּי חֲנִינָא מִלֶּכֶת וְעֵינָיו נִפְעֲרוּ מִפְּלִיאָה.

מוּלָם הִתְנַשֵּׂא עֵץ חָרוּב הֶגְמוֹנִי, עָצוּם בְּגָדְלוֹ! הַגֶּזַע שֶׁלּוֹ הָיָה עָבֶה מְאֹד, קְלִפָּתוֹ חוּמָה וּמְחֻסְפֶּסֶת, וְעֲנָפָיו הָרְחָבִים הִצִּילּוּ עַל הָאֲדָמָה בְּמֶרְחָב עֲנָק. צִיּוּץ שֶׁל צִפּוֹרִים נִשְׁמַע מִבֵּין הֶעָלִים הַיְרֻקִּים, וְרֵיחַ מָתוֹק עָמַד בָּאֲוִיר.

"בְּנִי," לָחַשׁ רַבִּי חֲנִינָא בְּיִרְאַת כָּבוֹד, "רְאֵה אֵיזוֹ בְּרָכָה הִנְחִיל הַקָּדוֹשׁ בָּרוּךְ הוּא לְאֶרֶץ יִשְׂרָאֵל. עֵץ עֲנָק כָּזֶה מֵעוֹלָם לֹא רָאִיתִי בְּבָבֶל! הָבָה נִמְדֹּד כַּמָּה עָבֶה וְחָזָק הַגֶּזַע הַזֶּה".

בְּנוֹ שֶׁל רַבִּי חֲנִינָא הִתְקָרֵב. הוּא פָּתַח אֶת הָאַבְנֵט (הַחֲגוֹרָה) שֶׁהָיָה קָשׁוּר סְבִיב מָתְנָיו – וְזוֹ הָיְתָה חֲגוֹרָה אֲרֻכָּה מְאֹד. הַבֵּן נִגַּשׁ לַגֶּזַע שֶׁל עֵץ הֶחָרוּב וְנִסָּה לְהַקִּיף אוֹתוֹ בַּחֲגוֹרָה, לִמְדֹּד אֶת הָרֹחַב שֶׁלּוֹ. הוּא הָלַךְ סְבִיב הָעֵץ, מָתַח אֶת הַחֲגוֹרָה עוֹד וָעוֹד, אֲבָל יְלָדִים, אַתֶּם יוֹדְעִים מָה קָרָה? הַחֲגוֹרָה שֶׁל הַבֵּן הִסְפִּיקָה בְּקֹשִׁי לְהַקִּיף רַק חֲצִי מֵהַגֶּזַע!

כָּל כָּךְ עָבֶה, כָּל כָּךְ גָּדוֹל, כָּל כָּךְ שׁוֹפֵעַ חַיִּים הָיָה הָעֵץ הַזֶּה.

סוֹד הַמְּתִיקוּת מִתְגַּלֶּה

רַבִּי חֲנִינָא נָשָׂא עֵינָיו אֶל הָעֲנָפִים הָעֶלְיוֹנִים. הֵם הָיוּ עֲמוּסִים בְּתַרְמִילֵי חֲרוּבִים. רַבִּי חֲנִינָא הוֹשִׁיט אֶת יָדוֹ, בְּזֶהִירוּת שֶׁל מוֹצֵא אוֹצָר, וְקָטַף חָרוּב אֶחָד.

הֶחָרוּב הָיָה כֵּהֶה וּנְפוּחַ. בְּדֶרֶךְ כְּלָל, כְּשֶׁאֲנַחְנוּ חוֹשְׁבִים עַל חָרוּב, אֲנַחְנוּ חוֹשְׁבִים עַל מַשֶּׁהוּ קָשֶׁה שֶׁצָּרִיךְ לִלְעֹס בְּכֹחַ. אֲבָל אֶצְבְּעוֹתָיו שֶׁל הַצַּדִּיק לָחֲצוּ קַלּוֹת עַל קְלִפַּת הֶחָרוּב, וְשָׁמְעוּ 'פְּרִיק' קָטָן שֶׁל פְּתִיחָה.

וְאָז זֶה קָרָה.

מִתּוֹךְ סֶדֶק הֶחָרוּב שֶׁהִתְבַּקֵּעַ, פָּרַץ זֶרֶם חַם וְסָמִיךְ שֶׁל מַשֶּׁהוּ זָהוּב וְנִפְלָא. זֶה לֹא הָיָה סְתָם מִיץ. זֶה הָיָה דְּבַשׁ! דְּבַשׁ חֲרוּבִים טָהוֹר וּמָתוֹק יָצָא מִתּוֹךְ הַפְּרִי וּמִלֵּא אֶת כַּפּוֹת יָדָיו שֶׁל רַבִּי חֲנִינָא! הַדְּבַשׁ גָּלַשׁ בֵּין אֶצְבְּעוֹתָיו, נוֹטֵף מַטָּה עַל הָאֲדָמָה, וְרֵיחַ מְתִיקוּת שֶׁכְּמוֹתוֹ לֹא הֵרִיחוּ מֵעוֹלָם הִקִּיף אוֹתָם.

רַבִּי חֲנִינָא טָעַם מֵהַדְּבַשׁ שֶׁעַל יָדָיו, וְטַעַם גַּן עֵדֶן הִתְפַּשֵּׁט בְּפִיו. דְּמָעוֹת עָמְדוּ בְּעֵינָיו. "בְּנִי," הוּא אָמַר בְּקוֹל רוֹעֵד מֵהִתְרַגְּשׁוּת וְהוֹדָיָה, "בָּבֶל גְּדוֹלָה, אֲבָל בְּאֶרֶץ יִשְׂרָאֵל, הַקָּדוֹשׁ בָּרוּךְ הוּא גָּנַז מְתִיקוּת שֶׁאֵין בְּשׁוּם מָקוֹם אַחֵר! אֲפִלּוּ בְּתוֹךְ עֵץ חָרוּב פָּשׁוּט, זָבַת חָלָב וּדְבַשׁ שֶׁל אֶרֶץ יִשְׂרָאֵל פּוֹרֶצֶת הַחוּצָה לְמִי שֶׁאוֹהֵב אוֹתָהּ בֶּאֱמֶת!".

מֵאוֹתוֹ יוֹם, כְּדֵי לִזְכֹּר תָּמִיד אֶת קְדֻשַּׁת אֶרֶץ יִשְׂרָאֵל וּפֵרוֹתֶיהָ, בָּחַר רַבִּי חֲנִינָא בַּר חָמָא מַשֶּׁהוּ מְיֻחָד שֶׁיִּהְיֶה "הַחוֹתָם" שֶׁלּוֹ (הַסֵּמֶל הָאִישִׁי הֶחָרוּט שֶׁלּוֹ). לֹא אֲרָיוֹת, לֹא כְּתָרִים שֶׁל זָהָב, אֶלָּא "חָרוּתָא" – שֶׁהִיא עֲנַף שֶׁל עֵץ דֶּקֶל. כְּדֵי לִזְכֹּר שֶׁאֲנַחְנוּ שַׁיָּכִים לָאֲדָמָה הַקְּדוֹשָׁה הַזּוֹ, אֲדָמָה שֶׁנּוֹתֶנֶת פֵּרוֹת מְתוּקִים כִּדְבַשׁ לְאוֹהֲבֶיהָ.

מקורות: החרוב והדבש ומדידת הגזע: ירושלמי פאה, פרק ז' הלכה ג' ("ר' חנינא בריה דר' אבהו אמר ר' חנינא בר חמא עבדנא: כד סלקית להכא, נסבית אזוריה דברי והקפתיה על חרובה ולא מטא, פצעתיה ומלאו ידיי דבש..."). סיקול ותיקון הדרכים בארץ ישראל: מסכת כתובות, קי"ב ע"א ("ר' חנינא ממתקן מתקליה" - היה מתקן את השבילים שלא ייכשלו בהם, מתוך חיבת הארץ). חותמו (סמלו) של ר' חנינא: מסכת בבא בתרא קס"א ע"א ("חותמו של רבי חנינא – חרותא" - ענף דקל).

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “יְלָדִים, נַסּוּ לְדַמְיֵן עִיר גְּדוֹלָה, עֲשִׁירָה וּמְלֵאָה בְּנַהֲרוֹת מַיִם מְתוּקִים.” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “רוּחַ חַמָּה נָשְׁבָה וְהֵעִיפָה גַּרְגְּרֵי חוֹל יָבֵשׁ יָשָׁר לְתוֹךְ הָעֵינַיִם, וְהַגָּרוֹן הָיָה נִחָר מִצָּמָא.”?
ג. כיצד הגיב ר' חנינא בר חמא למצב המתואר: “הָאָרֶץ הָיְתָה עֲדַיִן תַּחַת שִׁלְטוֹן הָרוֹמָאִים הָרְשָׁעִים, שֶׁהֶחֱרִיבוּ מְקוֹמוֹת רַבִּים.”?
ד. מה השתנה בין “הֵם צָעֲדוּ בַּשָּׂדֶה עַד שֶׁלְּפֶתַע, עָצַר רַבִּי חֲנִינָא מִלֶּכֶת וְעֵינָיו נִפְעֲרוּ מִפְּלִיאָה.” לבין “רַבִּי חֲנִינָא הוֹשִׁיט אֶת יָדוֹ, בְּזֶהִירוּת שֶׁל מוֹצֵא אוֹצָר, וְקָטַף חָרוּב אֶחָד.”?
ה. כיצד הגיע הסיפור אל האירוע: “בְּנִי," הוּא אָמַר בְּקוֹל רוֹעֵד מֵהִתְרַגְּשׁוּת וְהוֹדָיָה, "בָּבֶל גְּדוֹלָה, אֲבָל בְּאֶרֶץ יִשְׂרָאֵל, הַקָּדוֹשׁ בָּרוּךְ הוּא גָּנַז מְתִיקוּת…”?

דְּבַשׁ חֲרוּבִים טָהוֹר וּמָתוֹק יָצָא מִתּוֹךְ הַפְּרִי וּמִלֵּא אֶת כַּפּוֹת יָדָיו שֶׁל רַבִּי חֲנִינָא!□□

3. מה היה ראשון?

כתבו ליד כל אירוע מספר מ־1 עד 5 לפי סדר התרחשותו בסיפור.

□ הַקָּדוֹשׁ בָּרוּךְ הוּא רָאָה אֶת הָאַהֲבָה הָעֲצוּמָה הַזּוֹ, וְשָׂמַח בְּבָנָיו הַשָּׁבִים לִגְבוּלָם.
□ יְלָדִים, נַסּוּ לְדַמְיֵן עִיר גְּדוֹלָה, עֲשִׁירָה וּמְלֵאָה בְּנַהֲרוֹת מַיִם מְתוּקִים.
□ כְּדֵי לִזְכֹּר שֶׁאֲנַחְנוּ שַׁיָּכִים לָאֲדָמָה הַקְּדוֹשָׁה הַזּוֹ, אֲדָמָה שֶׁנּוֹתֶנֶת פֵּרוֹת מְתוּקִים כִּדְבַשׁ לְאוֹהֲבֶיהָ.
□ וְהִנֵּה, בְּבֹקֶר מְעֻרְפָּל אֶחָד, הֵם עָבְרוּ אֶת הַגְּבוּל וְנִכְנְסוּ לְאֶרֶץ יִשְׂרָאֵל.
□ בְּדֶרֶךְ כְּלָל, כְּשֶׁאֲנַחְנוּ חוֹשְׁבִים עַל חָרוּב, אֲנַחְנוּ חוֹשְׁבִים עַל מַשֶּׁהוּ קָשֶׁה שֶׁצָּרִיךְ לִלְעֹס בְּכֹחַ.

לפני ואחרי

מה קרה לפני: “הַקָּדוֹשׁ בָּרוּךְ הוּא רָאָה אֶת הָאַהֲבָה הָעֲצוּמָה הַזּוֹ, וְשָׂמַח בְּבָנָיו הַשָּׁבִים לִגְבוּלָם.”?
מה קרה אחרי: “בְּדֶרֶךְ כְּלָל, כְּשֶׁאֲנַחְנוּ חוֹשְׁבִים עַל חָרוּב, אֲנַחְנוּ חוֹשְׁבִים עַל מַשֶּׁהוּ קָשֶׁה שֶׁצָּרִיךְ לִלְעֹס בְּכֹחַ.”?

סיבה ותוצאה מתוך המעשה

האירוע הנתוןמה גרם לו?מה הייתה תוצאתו?
הָרֵי הַגָּלִיל הִתְנַשְּׂאוּ מוּלָם, וְרֵיחַ מְיֻחָד, רֵיחַ שֶׁל אֲדָמָה קְדוֹשָׁה רְטֻבָּה מִטַּל בֹּקֶר, מִלֵּא אֶת הָאֲוִיר.
הוּא פָּתַח אֶת הָאַבְנֵט (הַחֲגוֹרָה) שֶׁהָיָה קָשׁוּר סְבִיב מָתְנָיו – וְזוֹ הָיְתָה חֲגוֹרָה אֲרֻכָּה מְאֹד.

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “החרוב הענק והפרי שמילא ידו בדבש” על ר' חנינא בר חמא? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “כְּדֵי שֶׁשּׁוּם יְהוּדִי לֹא יִמְעַד," אָמַר לִבְנוֹ בְּעוֹדוֹ גּוֹרֵר סֶלַע כָּבֵד שֶׁשָּׂרַט אֶת יָדָיו, "וּכְדֵי שֶׁכֻּלָּם יִרְאוּ כַּמָּה שְׁבִילֵי…”.
ג. אילו ר' חנינא בר חמא לא היה פועל כפי שמתואר ב־“רַבִּי חֲנִינָא הוֹשִׁיט אֶת יָדוֹ, בְּזֶהִירוּת שֶׁל מוֹצֵא אוֹצָר, וְקָטַף חָרוּב אֶחָד.” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “בְּנִי," הוּא אָמַר בְּקוֹל רוֹעֵד מֵהִתְרַגְּשׁוּת וְהוֹדָיָה, "בָּבֶל גְּדוֹלָה, אֲבָל בְּאֶרֶץ יִשְׂרָאֵל, הַקָּדוֹשׁ בָּרוּךְ…” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. יְלָדִים, נַסּוּ לְדַמְיֵן עִיר גְּדוֹלָה, עֲשִׁירָה וּמְלֵאָה…
2. אֲבָל רַבִּי חֲנִינָא וּבְנוֹ, שֶׁצָּעֲדוּ יַחַד, לֹא וִתְּרוּ.
3. אֲבָנִים גְּדוֹלוֹת, בּוֹרוֹת וְסַלְעִים חַדִּים חָסְמוּ אֶת הַדֶּרֶךְ.
4. בְּנִי," לָחַשׁ רַבִּי חֲנִינָא בְּיִרְאַת כָּבוֹד, "רְאֵה אֵיזוֹ…
5. מִתּוֹךְ סֶדֶק הֶחָרוּב שֶׁהִתְבַּקֵּעַ, פָּרַץ זֶרֶם חַם וְסָמִיךְ…
6. כְּדֵי לִזְכֹּר שֶׁאֲנַחְנוּ שַׁיָּכִים לָאֲדָמָה הַקְּדוֹשָׁה הַזּוֹ,…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “הָאָרֶץ הָיְתָה עֲדַיִן תַּחַת שִׁלְטוֹן הָרוֹמָאִים הָרְשָׁעִים, שֶׁהֶחֱרִיבוּ מְקוֹמוֹת רַבִּים.”.
מכתב: כתבו לר' חנינא בר חמא בעקבות “הוּא פָּתַח אֶת הָאַבְנֵט (הַחֲגוֹרָה) שֶׁהָיָה קָשׁוּר סְבִיב מָתְנָיו – וְזוֹ הָיְתָה חֲגוֹרָה אֲרֻכָּה מְאֹד.”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “דְּבַשׁ חֲרוּבִים טָהוֹר וּמָתוֹק יָצָא מִתּוֹךְ הַפְּרִי וּמִלֵּא אֶת כַּפּוֹת יָדָיו שֶׁל רַבִּי חֲנִינָא!”.
דיווח: כתבו ידיעה קצרה שכותרתה “החרוב הענק והפרי שמילא ידו בדבש”.

המחזה

רגע מן הסיפורהדמויותמה יאמרו או יעשו?
הָרֵי הַגָּלִיל הִתְנַשְּׂאוּ מוּלָם, וְרֵיחַ מְיֻחָד, רֵיחַ שֶׁל אֲדָמָה קְדוֹשָׁה רְטֻבָּה מִטַּל…
רַבִּי חֲנִינָא הוֹשִׁיט אֶת יָדוֹ, בְּזֶהִירוּת שֶׁל מוֹצֵא אוֹצָר, וְקָטַף חָרוּב אֶחָד.
