פעם אחת הגיע מאוחר לבית המדרש

דף עבודה בעקבות רבי עקיבא
שם: כיתה:

המעשה — הסיפור המלא

להלן מערך חינוכי, רגשי ומרתק על רבי עקיבא וענוותנותו העצומה (כאשר איחר לבית המדרש) . הסיפור נכתב בהקפדה על הדרכתך: שפה מכבדת, תיאורים חושיים, בניית מתח, וניקוד מלא לכל מילה בסיפור.

פְּתִיחָה: בֵּית הַמִּדְרָשׁ כְּמִבְצַר אֵשׁ

יְלָדִים, נַסּוּ לְדַמְיֵן מָקוֹם עָצוּם, מָלֵא בְּהוֹד וְקְדֻשָּׁה. לֹא אַרְמוֹן שֶׁל מֶלֶךְ, לֹא מִבְצָר שֶׁל חַיָּלִים, אֶלָּא מָקוֹם שֶׁבּוֹ נוֹלֶדֶת הַחָכְמָה שֶׁל עַם יִשְׂרָאֵל.

בֵּית הַמִּדְרָשׁ הַגָּדוֹל.

בַּחוּץ נוֹשֶׁבֶת רוּחַ קָרָה, אַךְ כְּשֶׁמִּתְקָרְבִים לַדְּלָתוֹת שֶׁל הַבִּנְיָן, יֵשׁ תְּחוּשָׁה שֶׁל חֹם שֶׁלֹּא מַגִּיעַ מִתַּנּוּר... הַחֹם מַגִּיעַ מִקּוֹלוֹת שֶׁל תּוֹרָה!

מֵאוֹת תַּלְמִידֵי חֲכָמִים יוֹשְׁבִים שָׁם יַחַד. רֵיחַ נָעִים שֶׁל קְלָפִים יְשָׁנִים וּטְהוֹרִים נִשָּׂא בָּאֲוִיר, צְלִיל עֲדִין שֶׁל דַּפִּים שֶׁמִּתְהַפְּכִים הָפְכָה, וְקוֹלוֹתָם הָאַדִּירִים שֶׁל הַלּוֹמְדִים עוֹלִים יַחַד וְנִשְׁמָעִים מַמָּשׁ כְּמוֹ רַעַם, כְּמוֹ הַמַּיִם הָרַבִּים שֶׁל הַיָּם הַגָּדוֹל. "אָמַר הַתַּנָּא... פָּסַק הַהֲלָכָה... מֻתָּר... אָסוּר...".

בִּמְקוֹם הַכָּבוֹד, מִלְּפָנִים, יוֹשֵׁב אֶחָד מִגְּדוֹלֵי עַנְקֵי הַתּוֹרָה בְּכָל הַדּוֹרוֹת – רַבִּי אֱלִיעֶזֶר בֶּן הוֹרְקְנוֹס. פָּנָיו מְאִירוֹת, וְהַשְּׁקִידָה שֶׁלּוֹ אֲצוּרָה בְּכָל מִלָּה שֶׁהוּא אוֹמֵר. כָּל תַּלְמִיד רָצָה מָקוֹם קָרוֹב כַּמָּה שֶׁיּוֹתֵר לָרַב, כְּדֵי שֶׁלֹּא יַפְסִיד אֲפִלּוּ אוֹת אַחַת מִתּוֹרָתוֹ.

וְרַק כִּסֵּא אֶחָד – הַכִּסֵּא הֲכִי חָשׁוּב בַּשּׁוּרָה הָרִאשׁוֹנָה, זֶה שֶׁשָּׁמוּר לְגָדוֹל הַתַּלְמִידִים – עָמַד הַיּוֹם רֵיק.

הָאִיחוּר שֶׁל גְּדוֹל הַדּוֹר

זֶה הָיָה מְקוֹמוֹ שֶׁל רַבִּי עֲקִיבָא.

יְלָדִים, רַבִּי עֲקִיבָא בְּאוֹתָם יָמִים כְּבָר לֹא הָיָה הָרוֹעֶה הֶעָנִי. הוּא הָיָה הַתַּנָּא הַקָּדוֹשׁ, הָאִישׁ שֶׁיּוֹדֵעַ אֶת כָּל רָזֵי הַתּוֹרָה, שֶׁעֶשְׂרִים וְאַרְבָּעָה אֶלֶף תַּלְמִידִים שׁוֹתִים אֶת דְּבָרָיו בְּצִמָּאוֹן. אַךְ בְּבֵית מִדְרָשׁוֹ שֶׁל רַבִּי אֱלִיעֶזֶר, הָיָה רַבִּי עֲקִיבָא רַק תַּלְמִיד שֶׁאוֹהֵב אֶת רַבּוֹ בְּאַהֲבָה עֲצוּמָה וְיִרְאַת כָּבוֹד.

וְהִנֵּה, בְּאוֹתוֹ יוֹם, רַבִּי עֲקִיבָא לֹא הִגִּיעַ בַּזְּמַן. אוּלַי קִיֵּם בַּדֶּרֶךְ מִצְוַת בִּקּוּר חוֹלִים אוֹ הֲשָׁבַת אֲבֵדָה, אוֹ שֶׁמָּא הָיָה עָמוּס מִדַּי בְּדִינֵי צְדָקָה שֶׁל עֲנִיֵּי הָעִיר – אֵינֶנּוּ יוֹדְעִים.

אֲבָל כְּשֶׁרַבִּי עֲקִיבָא הִגִּיעַ לְפֶתַח בֵּית הַמִּדְרָשׁ, הַדֶּלֶת כְּבָר הָיְתָה פְּתוּחָה מֶחֱצָה.

רַבִּי עֲקִיבָא עָצַר עַל הַמִּפְתָּן. הוּא הִבִּיט פְּנִימָה. מֵאוֹת זוּגוֹת עֵינַיִם רְתוּקוֹת לְרַבִּי אֱלִיעֶזֶר שֶׁיּוֹשֵׁב בָּאֶמְצַע וְדוֹרֵשׁ. מֶתַח דַּק מָלֵא בַּאֲוִיר – כֻּלָּם מְנַסִּים לְהָבִין אֶת הַסֻּגְיָה הַקָּשָׁה.

רַבִּי עֲקִיבָא מַבִּיט עַל הַכִּסֵּא שֶׁמְּחַכֶּה רַק לוֹ בַּשּׁוּרָה הָרִאשׁוֹנָה. אִם הוּא יִכָּנֵס עַכְשָׁו, מָה יִקְרֶה? הֲרֵי כָּל הַתַּלְמִידִים יַעַמְדוּ בִּכְבוֹדוֹ! רַבִּי אֱלִיעֶזֶר יַפְסִיק לְדַבֵּר לְרֶגַע כְּדֵי שֶׁכֻּלָּם יִסְתַּדְּרוּ! וְהַתּוֹרָה? שֶׁרְשֶׁרֶת הַתּוֹרָה תֵּעָצֵר לְדַקָּה אוֹ שְׁתַּיִם.

"לֹא," חָשַׁב רַבִּי עֲקִיבָא בְּלִבּוֹ הַטָּהוֹר, "הַתּוֹרָה אֵינָהּ שֶׁלִּי. וְלֹא כְּבוֹדִי חָשׁוּב פֹּה אֶלָּא כְּבוֹד שָׁמַיִם."

בְּלִי לְהוֹצִיא הֶגֶה, רַבִּי עֲקִיבָא יָרַד בְּשֶׁקֶט עַל בִּרְכָּיו, וְהִתְיַשֵּׁב בַּחוּץ. מַמָּשׁ כָּךְ. מֵאֲחוֹרֵי דֶּלֶת הָעֵץ הַכְּבֵדָה, עַל אֶבֶן סַף הַכְּנִיסָה הַקָּרָה. בְּלִי כִּסֵּא חָשׁוּב, בְּלִי סְפָרִים מֻנָּחִים לְפָנָיו עַל שֻׁלְחָן כְּבוֹד. הַקָּדוֹשׁ שֶׁבַּתַּנָּאִים יָשַׁב עַל הָאָרֶץ וְהִקְשִׁיב – רַק כְּדֵי לֹא לְהַפְרִיעַ לְתַלְמִיד אֶחָד קָטָן לִשְׁמֹעַ אֶת רַבּוֹ!

הַשְּׁאֵלָה שֶׁנּוֹתְרָה יְתוֹמָה

בְּתוֹךְ בֵּית הַמִּדְרָשׁ, הַשִּׁיעוּר שֶׁל רַבִּי אֱלִיעֶזֶר מַמְשִׁיךְ.

לְפֶתַע, עוֹלָה קֻשְׁיָה עֲצוּמָה! שְׁאֵלָה הִלְכָתִית שֶׁמְּסַבֶּכֶת אֶת כָּל הַהֲבָנָה שֶׁל הַדִּין.

רַבִּי אֱלִיעֶזֶר שׁוֹאֵל, וְ... שֶׁקֶט בְּבֵית הַמִּדְרָשׁ.

דְּמָמָה כָּל כָּךְ כְּבֵדָה, שֶׁהָיָה מַמָּשׁ אֶפְשָׁר לִשְׁמֹעַ אֶת לֶהָבוֹת הַנֵּרוֹת מְרַצְּדוֹת עַל הַקִּירוֹת.

רַבִּי הוֹשֵׁעַ צוֹפֶה, חֲכָמִים אֲחֵרִים מִסְתַּכְּלִים אֶחָד עַל הַשֵּׁנִי. הַמַּבָּטִים מוּשְׁפָּלִים. אַף אֶחָד לֹא מוֹצֵא תְּשׁוּבָה מְדֻיֶּקֶת וּנְכוֹנָה שֶׁתַּתְאִים לְתוֹרָתוֹ שֶׁל רַבִּי אֱלִיעֶזֶר.

רַבִּי אֱלִיעֶזֶר בֶּן הוֹרְקְנוֹס, עֵינָיו מְשׁוֹטְטוֹת עַל פְּנֵי כָּל הַשּׁוּרוֹת הָאֲרֻכּוֹת שֶׁל תַּלְמִידָיו, עַד שֶׁמַּבָּטוֹ נוֹפֵל עַל הַכִּסֵּא הָרֵיק בַּשּׁוּרָה הָרִאשׁוֹנָה.

"הֵיכָן עֲקִיבָא?" הוּא לְאַט שׁוֹאֵל. "כְּלוּם הוּא אֵינֶנּוּ כָּאן הַיּוֹם? אִלּוּ רַבִּי עֲקִיבָא הָיָה פֹּה, הָיָה מֵאִיר אֶת עֵינֵינוּ בַּסֻּגְיָה הַזּוֹ."

אַנְשִׁים מַתְחִילִים לְלַחְשֵׁשׁ. אוּלַי חָלָה? אוּלַי קָרָה מַשֶּׁהוּ בַּדֶּרֶךְ?

וְאָז, מַבְחִין אֶחָד הַתַּלְמִידִים בְּצֵל עָדִין מֵאֲחוֹרֵי הַדֶּלֶת. הוּא מִתְרוֹמֵם, נִגָּשׁ לְעֵבֶר הַכְּנִיסָה, וְרוֹאֶה אֶת הַמַּחֲזֶה שֶׁלֹּא יִשְׁכַּח לְעוֹלָם.

רַבִּי עֲקִיבָא... גְּדוֹל הַדּוֹר, שֶׁעֲשִׁיְרֵי יְרוּשָׁלַיִם מְכַבְּדִים בְּזָהָב, יוֹשֵׁב עַל הַמִּדְרֵגָה מִבַּחוּץ, מַקְשִׁיב בְּרִכּוּז וּבַעֲנָוָה, כְּדֵי לֹא לְהַפְרִיעַ לְאַף אֶחָד.

"רַבִּי!" קוֹרֵא הַתַּלְמִיד בְּהִתְרַגְּשׁוּת, "רַבִּי עֲקִיבָא פֹּה! בַּחוּץ!"

שֶׁלִּי וְשֶׁלָּכֶם - שֶׁלּוֹ הִיא

הַהֲמוֹן מַתְחִיל לָזוּז. הֵם פּוֹתְחִים אֶת הַדְּלָתוֹת לִרְוָחָה.

"כָּבוֹד הָרַב! הִכָּנֵס נָא לְכִסְאֲךָ! שֵׁב לְפָנֵינוּ!" מִתְחַנְּנִים חֲבֵרָיו וְתַלְמִידָיו, מְפַנִּים לוֹ מָקוֹם רָחָב לָצְעֹד דֶּרֶךְ שְׁבִיל שֶׁל אֲנָשִׁים אוֹהֲבִים. רַבִּי אֱלִיעֶזֶר עַצְמוֹ מַמְתִּין בְּמֶתַח וּבְשִׂמְחָה שֶׁיִּכָּנֵס תַּלְמִידוֹ הָאָהוּב.

רַבִּי עֲקִיבָא קָם מִמְּקוֹמוֹ עַל אֶבֶן הַסַּף, מְנַעֵר אֶת גְּלִימָתוֹ בַּעֲדִינוּת. הוּא פּוֹסֵעַ פְּנִימָה.

הַאִם הוּא רָץ לְכִסְאוֹ הַמְכֻבָּד בַּשּׁוּרָה הָרִאשׁוֹנָה? הַאִם הוּא אוֹמֵר: "הִנֵּה אֲנִי, גְּדוֹל הַדּוֹר, בָּאתִי לְהַצִּיל אֶת כֻּלְּכֶם עִם תְּשׁוּבָתִי הַנְּבוֹנָה"?

לֹא דֻּבִּים וְלֹא יַעַר.

רַבִּי עֲקִיבָא צוֹעֵד בְּרֹאשׁ מֻרְכָּן בְּיִרְאַת כָּבוֹד, מִתְקָרֵב אֶל שֻׁלְחָנוֹ שֶׁל רַבִּי אֱלִיעֶזֶר, וּמִתְיַשֵּׁב... לְמַרְגְּלוֹתָיו. עַל הָרִצְפָּה, מַמָּשׁ קָרוֹב אֵלָיו, כְּתַלְמִיד חָדָשׁ שֶׁרַק נִכְנַס בַּפַּעַם הָרִאשׁוֹנָה לְבֵית הַסֵּפֶר וְיוֹשֵׁב צָמוּד לְמוֹרֵהוּ כְּדֵי לִשְׁתּוֹת מֵחָכְמָתוֹ.

הוּא שָׁמַע אֶת הַשְּׁאֵלָה שֶׁנִּשְׁאֲלָה בַּחוּץ. וּבְעוֹדוֹ שָׁם, לְמַרְגְּלוֹת רַבּוֹ, בְּקוֹל עָנָו אֲבָל מָלֵא בְּבִטָּחוֹן תּוֹרָנִי זַךְ, פָּתַח רַבִּי עֲקִיבָא אֶת פִּיו. מִלָּה אַחַר מִלָּה, רְאָיָה אַחַר רְאָיָה, הוּא חִבֵּר אֶת כָּל קְצוֹת הַתּוֹרָה וְהֵאִיר אֶת הַהֲלָכָה הַקָּשָׁה כְּאוֹר הַשֶּׁמֶשׁ הַמַּבְהִיק בְּאֶמְצַע הַלַּיְלָה.

הַקֻּשְׁיָה שֶׁל רַבִּי אֱלִיעֶזֶר נִפְתְּרָה.

חִיּוּךְ רָחָב וְזוֹרֵחַ מִלֵּא אֶת פָּנָיו שֶׁל רַבִּי אֱלִיעֶזֶר. וְכָל תַּלְמִידֵי בֵּית הַמִּדְרָשׁ יָדְעוּ בְּאוֹתוֹ רֶגַע אֶת הַסּוֹד הָאֲמִתִּי שֶׁל רַבִּי עֲקִיבָא:

הוּא הָיָה הַגָּדוֹל מִכֻּלָּם - רַק בִּגְלַל שֶׁהָיָה הָעָנָיו מִכֻּלָּם! בִּזְכוּת זֶה שֶׁכְּבוֹד הַתּוֹרָה וּכְבוֹד מְלַמְּדָיו - הָיָה יָקָר לוֹ יוֹתֵר מִכְּבוֹדוֹ שֶׁלּוֹ.

א. שְׁאֵלַת פְּתִיחָה (לִפְנֵי שֶׁמַּתְחִילִים):

"יְלָדִים, נַסּוּ לְדַמְיֵן שֶׁיֵּשׁ הַפְסָקָה פְּעִילָה עִם מִשְׂחָק מְטֹרָף אוֹ מְסִבַּת כִּתָּה. מָה אַתֶּם עוֹשִׂים אִם בַּטָּעוּת אֵחַרְתֶּם לַכִּתָּה? רָצִים מַהֵר-מַהֵר וּמִתְפָּרְצִים פְּנִימָה מֵרֹב הִתְרַגְּשׁוּת, נָכוֹן?

אַךְ מָה קוֹרֶה כְּשֶׁמְּאַחֲרִים לַשִּׁיעוּר תּוֹרָה שֶׁל הָרַב הֲכִי גָּדוֹל בְּיִשְׂרָאֵל? הַיּוֹם נִשְׁמַע מָה עָשָׂה רַבִּי עֲקִיבָא בְּבֹקֶר אֶחָד כָּזֶה, וְנַגִּיעַ לְסוֹד עָצוּם."

ב. הֶעָרוֹת בִּצּוּעַ וְסְטוֹרִיטֵלִינְג:

ג. הַמְלָצוֹת לְמַצֶּגֶת (4 שְׁקוּפִיּוֹת):

(כְּלַל בַּרְזֶל: גְּבָרִים וִילָדִים מְעֻטָּרִים בְּכִפּוֹת, זְקָנִים וּפֵאוֹת כַּנִּדְרָשׁ. שַׁרְווּלִים אֲרֻכִּים. אוֹר קָדוֹשׁ שֶׁל תּוֹרָה, לְלֹא נָשִׁים אוֹ דְּמֻיּוֹת מֻפְשָׁטוֹת).

ד. פְּעִילוּת מְסַכֶּמֶת (5 דַּקּוֹת):

"מִשְׂחַק כְּנִיסָה בְּכָבוֹד"

נַעֲשֶׂה "סִמוּלַצְיָה". נִבְחַר 2 יְלָדִים, שֶׁצְּרִיכִים לָצֵאת לְרֶגַע מִחוּץ לַכִּתָּה (וּכְבָיָכוֹל תַּלְמִידִים שֶׁאֵחֲרוּ לַשִּׁיעוּר). שְׁאַר הַכִּתָּה בִּפְנִים תַּקְשִׁיב בְּרִכּוּז לַמּוֹרֶה הַמְסַפֵּר/מְלַמֵּד דָּבָר חָדָשׁ.

כַּאֲשֶׁר הַמּוֹרֶה נוֹתֵן סִימָן, הַיְלָדִים מִבַּחוּץ צְרִיכִים לְהִכָּנֵס, לִסְגֹּר אֶת הַדֶּלֶת בַּעֲדִינוּת שֶׁל תַּלְמִיד חָכָם, לִפְסֹעַ וְלִתְפֹּס מָקוֹם "מִבְּלִי לְעוֹרֵר עֲלֵיהֶם אֶת צְמִיּוֹת הָעַיִן אוֹ לַעֲצֹר אֶת הַתּוֹרָה שֶׁבַּכִּתָּה". הַיֶּלֶד שֶׁנִּכְנָס וּמִתְיַשֵּׁב מִבְּלִי שֶׁעָצַר אֶת הַלִּמּוּד מְקַבֵּל כַּפַּיִם בַּסּוֹף (אוֹ תְּעוּדַת "עֲנָוָה שֶׁל גָּדוֹל").

הַסִּפּוּר מְבֻסָּס בְּדִיּוּק עַל מִדְרָשׁ חֲזַ"ל כְּפִי שֶׁהוּבָא מִמְּקוֹרוֹתֵינוּ:

מקורות: הַסִּפּוּר מְבֻסָּס בְּדִיּוּק עַל מִדְרָשׁ חֲזַ"ל כְּפִי שֶׁהוּבָא מִמְּקוֹרוֹתֵינוּ: שִׁיר הַשִּׁירִים רַבָּה (פָּרָשָׁה א', פִּסְקָה י"ב): "מַעֲשֶׂה בְּרַבִּי עֲקִיבָא שֶׁאֵחַר קְצָת לְבֵית הַמִּדְרָשׁ וּבָא וְיָשַׁב לוֹ בַּחוּץ... אָמַר לוֹ הֲלָכָה זוֹ תְּלוּיָה בְּעַקִיבָא...". עַל הֲלִיכָתוֹ שֶׁל רַבִּי עֲקִיבָא וִישִׁיבָתוֹ תַּחַת רַגְלֵי רַבּוֹ אַף שֶׁיָּדַע אֶת כָּל הַתּוֹרָה. סֵפֶר "כֹּה עָשׂוּ חֲכָמֵינוּ" (חֵלֶק א'/ב'): עִבּוּד מוּכָר עַל רַבִּי עֲקִיבָא הַמְּאַחֵר, שֶׁלֹּא חָפֵץ בְּהַפְרָעָה לַמְּלַמֵּד כַּמּוּבָא בְּסֵפֶר הָאַגָּדָה ק"נ.

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “להלן מערך חינוכי, רגשי ומרתק על רבי עקיבא וענוותנותו העצומה (כאשר איחר לבית המדרש) .” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “יְלָדִים, רַבִּי עֲקִיבָא בְּאוֹתָם יָמִים כְּבָר לֹא הָיָה הָרוֹעֶה הֶעָנִי.”?
ג. כיצד הגיב רבי עקיבא למצב המתואר: “לֹא," חָשַׁב רַבִּי עֲקִיבָא בְּלִבּוֹ הַטָּהוֹר, "הַתּוֹרָה אֵינָהּ שֶׁלִּי.”?
ד. מה השתנה בין “אַף אֶחָד לֹא מוֹצֵא תְּשׁוּבָה מְדֻיֶּקֶת וּנְכוֹנָה שֶׁתַּתְאִים לְתוֹרָתוֹ שֶׁל רַבִּי אֱלִיעֶזֶר.” לבין “עַל הָרִצְפָּה, מַמָּשׁ קָרוֹב אֵלָיו, כְּתַלְמִיד חָדָשׁ שֶׁרַק נִכְנַס בַּפַּעַם הָרִאשׁוֹנָה לְבֵית הַסֵּפֶר…”?
ה. כיצד הגיע הסיפור אל האירוע: “(כְּלַל בַּרְזֶל: גְּבָרִים וִילָדִים מְעֻטָּרִים בְּכִפּוֹת, זְקָנִים וּפֵאוֹת כַּנִּדְרָשׁ.”?

גְּדוֹל הַדּוֹר, שֶׁעֲשִׁיְרֵי יְרוּשָׁלַיִם מְכַבְּדִים בְּזָהָב, יוֹשֵׁב עַל הַמִּדְרֵגָה מִבַּחוּץ, מַקְשִׁיב בְּרִכּוּז וּבַעֲנָוָה, כְּדֵי לֹא…

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “פעם אחת הגיע מאוחר לבית המדרש” על רבי עקיבא? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “בְּתוֹךְ בֵּית הַמִּדְרָשׁ, הַשִּׁיעוּר שֶׁל רַבִּי אֱלִיעֶזֶר מַמְשִׁיךְ.”.
ג. אילו רבי עקיבא לא היה פועל כפי שמתואר ב־“עַל הָרִצְפָּה, מַמָּשׁ קָרוֹב אֵלָיו, כְּתַלְמִיד חָדָשׁ שֶׁרַק נִכְנַס בַּפַּעַם הָרִאשׁוֹנָה לְבֵית הַסֵּפֶר וְיוֹשֵׁב צָמוּד…” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “(כְּלַל בַּרְזֶל: גְּבָרִים וִילָדִים מְעֻטָּרִים בְּכִפּוֹת, זְקָנִים וּפֵאוֹת כַּנִּדְרָשׁ.” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. להלן מערך חינוכי, רגשי ומרתק על רבי עקיבא וענוותנותו העצומה (כאשר איחר…
2. הוּא הָיָה הַתַּנָּא הַקָּדוֹשׁ, הָאִישׁ שֶׁיּוֹדֵעַ אֶת כָּל רָזֵי…
3. מֵאֲחוֹרֵי דֶּלֶת הָעֵץ הַכְּבֵדָה, עַל אֶבֶן סַף הַכְּנִיסָה הַקָּרָה.
4. וְאָז, מַבְחִין אֶחָד הַתַּלְמִידִים בְּצֵל עָדִין מֵאֲחוֹרֵי הַדֶּלֶת.
5. וְכָל תַּלְמִידֵי בֵּית הַמִּדְרָשׁ יָדְעוּ בְּאוֹתוֹ רֶגַע אֶת הַסּוֹד…
6. הַסִּפּוּר מְבֻסָּס בְּדִיּוּק עַל מִדְרָשׁ חֲזַ"ל כְּפִי שֶׁהוּבָא…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “לֹא," חָשַׁב רַבִּי עֲקִיבָא בְּלִבּוֹ הַטָּהוֹר, "הַתּוֹרָה אֵינָהּ שֶׁלִּי.”.
מכתב: כתבו לרבי עקיבא בעקבות “גְּדוֹל הַדּוֹר, שֶׁעֲשִׁיְרֵי יְרוּשָׁלַיִם מְכַבְּדִים בְּזָהָב, יוֹשֵׁב עַל הַמִּדְרֵגָה מִבַּחוּץ, מַקְשִׁיב בְּרִכּוּז…”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “הוּא הָיָה הַגָּדוֹל מִכֻּלָּם - רַק בִּגְלַל שֶׁהָיָה הָעָנָיו מִכֻּלָּם!”.
דיווח: כתבו ידיעה קצרה שכותרתה “פעם אחת הגיע מאוחר לבית המדרש”.

המחזה

רגע מן הסיפורהדמויותמה יאמרו או יעשו?
אֲבָל כְּשֶׁרַבִּי עֲקִיבָא הִגִּיעַ לְפֶתַח בֵּית הַמִּדְרָשׁ, הַדֶּלֶת כְּבָר הָיְתָה פְּתוּחָה…
עַל הָרִצְפָּה, מַמָּשׁ קָרוֹב אֵלָיו, כְּתַלְמִיד חָדָשׁ שֶׁרַק נִכְנַס בַּפַּעַם הָרִאשׁוֹנָה…

7. חיבור לחיים מתוך המעשה

א. מה אפשר ללמוד מן הדרך שבה רבי עקיבא פעל באירוע “אַף אֶחָד לֹא מוֹצֵא תְּשׁוּבָה מְדֻיֶּקֶת וּנְכוֹנָה שֶׁתַּתְאִים לְתוֹרָתוֹ שֶׁל רַבִּי אֱלִיעֶזֶר.”?
