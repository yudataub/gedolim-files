סוד הרגל האחת וסרגל הבנאים

דף עבודה בעקבות סוד הרגל האחת וסרגל הבנאים
שם: כיתה:

המעשה — הסיפור המלא

הסיפור מנוקד במלואו כפי שנדרש.

פְּתִיחָה: שְׁנֵי שׁוֹמְרֵי הַתּוֹרָה יְלָדִים, אִם הֱיִיתֶם אוֹחֲזִים בַּיָּד שֶׁלָּכֶם יַהֲלוֹם יָקָר שֶׁשָּׁוֶה מִילְיוֹנֵי שְׁקָלִים, אֵיךְ הֱיִיתֶם שׁוֹמְרִים עָלָיו? הַאִם הֱיִיתֶם מְשַׂחֲקִים בּוֹ בַּחוּץ בַּבֹּץ, אוֹ שׁוֹמְרִים אוֹתוֹ בְּתוֹךְ קֻפְסַת קְטִיפָה מִזָּהָב בְּתוֹךְ כַּסֶּפֶת? לְפְנֵי שָׁנִים רַבּוֹת בִּירוּשָׁלַיִם, הָיוּ שְׁנֵי חֲכָמִים אֲשֶׁר שָׁמְרוּ עַל הַיַּהֲלוֹם הַיָּקָר בְּיוֹתֵר בָּעוֹלָם – עַל תּוֹרַת יִשְׂרָאֵל. לִשְׁנֵי הָרַבָּנִים הַגְּדוֹלִים הָאֵלֶּה קָרְאוּ: שַׁמַּאי הַזָּקֵן וְהִלֵּל הַזָּקֵן.

שְׁנֵיהֶם אָהֲבוּ אֶת הַתּוֹרָה וְאֶת הַקָּדוֹשׁ בָּרוּךְ הוּא אַהֲבָה עֲצוּמָה, אַךְ לְכָל אֶחָד הָיְתָה דֶּרֶךְ מְיֻחֶדֶת לִשְׁמֹר עָלֶיהָ. שַׁמַּאי הַזָּקֵן הָיָה קַפְּדָן לִכְבוֹדָהּ שֶׁל תּוֹרָה. הוּא זָעַק: "הַתּוֹרָה הִיא קְדוֹשָׁה וַחֲמוּרָה כְּאֵשׁ! אָסוּר לְאַף אָדָם לְזַלְזֵל בָּהּ אֲפִלּוּ כְּחוּט הַשַּׂעֲרָה!". מִי שֶׁבָּא לִלְמֹד אֶצְלוֹ הָיָה חַיָּב לִהְיוֹת רְצִינִי וְיָרֵא שָׁמַיִם. וְהִלֵּל הַזָּקֵן? הִלֵּל הָיָה אִישׁ שֶׁל מַיִם וְחֶסֶד. הוּא הָיָה סַבְלָן כְּלַפֵּי כָּל אָדָם, תָּמִיד חִיֵּךְ לְכֻלָּם, וְאָמַר: "עָלֵינוּ לְקַבֵּל כָּל אָדָם בְּסֵבֶר פָּנִים יָפוֹת, אֲפִלּוּ אִם הוּא רָחוֹק, וְכָךְ נְקָרֵב אוֹתוֹ לַתּוֹרָה".

הַנָּכְרִי הַמְמַהֵר יוֹם אֶחָד, הָלַךְ בִּרְחוֹבוֹת יְרוּשָׁלַיִם הַהוֹמִים אָדָם נָכְרִי (גּוֹי). הָאִישׁ הַזֶּה הָיָה כַּנִּרְאֶה בַּנַּאי שֶׁבּוֹנֶה בָּתִּים, כִּי בְּיָדוֹ הוּא הֶחְזִיק קֶרֶשׁ עֵץ מְיֻחָד שֶׁנִּקְרָא "אַמַּת הַבִּנְיָן" – סַרְגֵּל אָרֹךְ עִם סִימוּנִים שֶׁבְּעֶזְרָתוֹ מוֹדְדִים אֶת הַקִּירוֹת כְּדֵי שֶׁיִּהְיוּ יְשָׁרִים. הַנָּכְרִי הַזֶּה רָאָה אֶת הַיְהוּדִים לְבוּשִׁים בִּבְגָדִים נְקִיִּים, לוֹמְדִים בְּבָתֵּי הַמִּדְרָשׁ, מִתְפַּלְלִים לַה', וְהַלֵּב שֶׁלּוֹ רָצָה גַּם. הוּא אָמַר לְעַצְמוֹ: "אֲנִי רוֹצֶה לִהְיוֹת יְהוּדִי! אֲנִי רוֹצֶה לִלְמֹד אֶת כָּל הַתּוֹרָה שֶׁלָּהֶם!". אֲבָל לָאִישׁ הַזֶּה הָיְתָה בְּעָיָה אַחַת. לֹא הָיְתָה לוֹ סַבְלָנוּת כְּלָל. "אֵין לִי זְמַן לָשֶׁבֶת עַכְשָׁו עֶשְׂרִים שָׁנָה וְלִלְמֹד סְפָרִים עָבִים," מִלְמֵל לְעַצְמוֹ, "אֲנִי רוֹצֶה לִלְמֹד הַכֹּל מַהֵר-מַהֵר. בְּכַמָּה דַּקּוֹת!".

אֵצֶל שַׁמַּאי הַזָּקֵן הָלַךְ הַנָּכְרִי עִם סַרְגֵּל הָעֵץ הֶעָבֶה שֶׁלּוֹ לְבֵיתוֹ שֶׁל שַׁמַּאי הַזָּקֵן. הָרַב הַקָּדוֹשׁ יָשַׁב שָׁם רָכוּן עַל גְּוִילֵי הַתּוֹרָה, מְדַקְדֵּק בְּכָל קוֹץ וְתָג. נִכְנַס הַנָּכְרִי, עָמַד מוּל שַׁמַּאי, וְאָמַר בְּקוֹל דּוֹרֵשׁ: "רַבִּי! אֲנִי רוֹצֶה לִהְיוֹת יְהוּדִי. גַּיֵּר אוֹתִי! אֲבָל... רַק עַל תְּנַאי אֶחָד!". הֵרִים שַׁמַּאי אֶת עֵינָיו הַקְּדוֹשׁוֹת: "וּמַהוּ הַתְּנַאי?". הַנָּכְרִי הֵרִים אֶת רַגְלוֹ הָאַחַת בָּאֲוִיר, עָמַד בְּשִׁוּוּי מִשְׁקָל רַק עַל רֶגֶל אַחַת, וְהִכְרִיז עִם חִיּוּךְ מִזְלַזֵּל: "תְּלַמֵּד אוֹתִי אֶת כָּל הַתּוֹרָה כֻּלָּהּ... כְּשֶׁאֲנִי עוֹמֵד עַל רֶגֶל אַחַת! בְּקִצּוּר, בְּמִשְׁפָּט אֶחָד, בְּלִי לְהַאֲרִיךְ!".

יְלָדִים, שַׁמַּאי הַזָּקֵן הָיָה הָמוּם! הַתּוֹרָה, שֶׁעֲלֵיהֶם עָמַל מֹשֶׁה רַבֵּנוּ אַרְבָּעִים יוֹם וְאַרְבָּעִים לַיְלָה בְּהַר סִינַי! הַתּוֹרָה שֶׁיֵּשׁ בָּהּ שִׁשָּׁה סִדְרֵי מִשְׁנָה, תַּלְמוּד, חֻמָּשִׁים, דִּינֵי שַׁבָּת וְכַשְׁרוּת! הַגּוֹי הַזֶּה בָּא מֵהָרְחוֹב, חוֹשֵׁב שֶׁהַתּוֹרָה הִיא תַּפְרִיט שֶׁל מִסְעָדָה אוֹ שִׁיר קָצָר, וּמְבַקֵּשׁ לִלְמֹד אוֹתָהּ תּוֹךְ כְּדֵי קְפִיצָה עַל רֶגֶל אַחַת?! אֵיזֶה זִלְזוּל דָּחוּף וְנוֹרָא בִּכְבוֹדוֹ שֶׁל מֶלֶךְ מַלְכֵי הַמְּלָכִים! שַׁמַּאי הַקַּפְּדָן וְהַטָּהוֹר לֹא הָיָה מוּכָן לִסְבֹּל זִלְזוּל כָּזֶה. הוּא הֵרִים אֶת "אַמַּת הַבִּנְיָן" – אוֹתוֹ קֶרֶשׁ שֶׁל מְדִידוֹת שֶׁהָיָה בְּיָדוֹ שֶׁל הַבַּנַּאי, וְדָחַף אוֹתוֹ הַחוּצָה לָרְחוֹב. "צֵא מִכָּאן!" קָרָא שַׁמַּאי, "הַתּוֹרָה הִיא לֹא בְּדִיחָה לְעוֹמְדִים עַל רֶגֶל אַחַת! אִם אֵין לְךָ כָּבוֹד لַקֹּדֶשׁ, אֵין לְךָ מָקוֹם כָּאן!". שַׁמַּאי שָׁמַר עַל כְּבוֹד הַתּוֹרָה כְּמוֹ אֲרִי.

אֵצֶל הִלֵּל הַזָּקֵן הַנָּכְרִי עָמַד בָּרְחוֹב מְאֻכְזָב, אֲבָל לֹא וִתֵּר. "אֵלֵךְ לְנָשִׂיא הַשֵּׁנִי, אֶל הִלֵּל הַזָּקֵן," אָמַר בְּלִבּוֹ. הוּא הִגִּיעַ לְבֵיתוֹ שֶׁל הִלֵּל, נִכְנַס פְּנִימָה, וְשׁוּב... הֵרִים רֶגֶל אַחַת לָאֲוִיר. "שָׁלוֹם לְךָ, הִלֵּל!" אָמַר. "אֲנִי רוֹצֶה לְהִתְגַּיֵּר! אֲבָל רַק עַל תְּנַאי שֶׁתְּלַמֵּד אוֹתִי אֶת כָּל הַתּוֹרָה כֻּלָּהּ עַכְשָׁו, בַּשְּׁנִיּוֹת שֶׁבָּהֶן אֲנִי מַצְלִיחַ לַעֲמֹד עַל רֶגֶל אַחַת!".

הִלֵּל הַזָּקֵן הִבִּיט בָּאִישׁ. הִלֵּל ראה בְּתוֹךְ לִבּוֹ שֶׁל הָאִישׁ אֶת הַבִּלְבּוּל וְהַבּוּרוּת, אֲבָל גַּם רָאָה נִיצוֹץ קָטָן שֶׁרוֹצֶה לָדַעַת אֶת הָאֱמֶת. הִלֵּל תָּמִיד אָמַר "הֱוֵי מְקַבֵּל אֶת כָּל הָאָדָם בְּסֵבֶר פָּנִים יָפוֹת". בִּמְקוֹם לִכְעֹס עַל הַזִּלְזוּל, הִלֵּל חִיֵּךְ אֵלָיו חִיּוּךְ רָחָב, חַם וְאוֹהֵב שֶׁל סָבָא דַּאֲגָן.

"אַתָּה רוֹצֶה לָדַעַת אֶת כָּל סוֹד הַתּוֹרָה בְּמִשְׁפָּט אֶחָד?" שָׁאַל הִלֵּל בְּרַכּוּת. "תִּשָּׁאֵר עַל רֶגֶל אַחַת. אֲנִי אַגִּיד לְךָ אֶת סוֹד הַסּוֹדוֹת שֶׁל הַתּוֹרָה". הַנָּכְרִי פָּתַח עֵינַיִם מְצַפּוֹת. הַאִם זֶה אֶפְשָׁרִי?!

וְאָז אָמַר הִלֵּל אֶת הַמִּשְׁפָּט הֶחָזָק וְהַמְּפֻרְסָם בְּיוֹתֵר: "מַה שֶּׁשָּׂנוּא עָלֶיךָ – לַחֲבֵרְךָ לֹא תַּעֲשֶׂה!" (בָּאֲרָמִית: "דְּעָלָךְ סְנֵי – לְחַבְרָךְ לָא תַּעֲבִיד"). וְהִלֵּל הִמְשִׁיךְ: "זוֹ הִיא כָּל הַתּוֹרָה כֻּלָּהּ. אִם אַתָּה לֹא רוֹצֶה שֶׁיְּצַעֲרוּ אוֹתְךָ, שֶׁיִּגְנְבוּ מִמְּךָ, אוֹ שֶׁיַּעֲלִיבוּ אוֹתְךָ – אַל תַּעֲשֶׂה זֹאת לְאַף אָדָם אַחֵר! הָאַהֲבָה הַזּוֹ לַשֵּׁנִי, הִיא הַשֹּׁרֶשׁ הֶעָמֹק שֶׁל כָּל הַתּוֹרָה שֶׁנָּתַן ה'!". "וּמָה עִם כָּל שְׁאָר הַהֲלָכוֹת?" שָׁאַל הַנָּכְרִי הַמֻּפְתָּע שֶׁנִּשְׁאַר בָּאֲוִיר עַל רֶגֶל אַחַת. הִלֵּל טָפַח לְאַט עַל כְּתֵפוֹ, וְאָמַר לוֹ עִם אוֹתוֹ חִיּוּךְ מוֹאֵר: "וּשְׁאָר הַתּוֹרָה? אֵלּוּ כְּבָר הַהֶסְבֵּרִים שֶׁל הַמִּשְׁפָּט הַזֶּה... וְעַכְשָׁו, יְדִידִי – זִיל גְּמוֹר! (לֵךְ וְתִלְמַד! תּוֹרִיד אֶת הָרֶגֶל, שֵׁב עַל הַסַּפְסָל, וּבְעֶזְרַת ה' מִתּוֹךְ אַהֲבָה תַּתְחִיל לִלְמֹד אֶת כָּל שְׁאָר הַפְּרָטִים)".

סוֹף דָּבָר: הָאֱמֶת וְהַשָּׁלוֹם הַנָּכְרִי הַזֶּה הוֹרִיד אֶת רַגְלוֹ, יָשַׁב אֵצֶל הִלֵּל הַזָּקֵן, וְהָפַךְ לְגֵר צֶדֶק קָדוֹשׁ – אָדָם שֶׁלָּמַד אֶת כָּל הַתּוֹרָה כֻּלָּהּ וְהָפַךְ לִיהוּדִי כָּשֵׁר לְמֶשֶׁךְ כָּל חַיָּיו.

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “הַתּוֹרָה שֶׁיֵּשׁ בָּהּ שִׁשָּׁה סִדְרֵי מִשְׁנָה, תַּלְמוּד, חֻמָּשִׁים, דִּינֵי שַׁבָּת וְכַשְׁרוּת!”.
מכתב: כתבו לסוד הרגל האחת וסרגל הבנאים בעקבות “הִלֵּל ראה בְּתוֹךְ לִבּוֹ שֶׁל הָאִישׁ אֶת הַבִּלְבּוּל וְהַבּוּרוּת, אֲבָל גַּם רָאָה נִיצוֹץ קָטָן שֶׁרוֹצֶה לָדַעַת אֶת…”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “תּוֹרִיד אֶת הָרֶגֶל, שֵׁב עַל הַסַּפְסָל, וּבְעֶזְרַת ה' מִתּוֹךְ אַהֲבָה תַּתְחִיל לִלְמֹד אֶת כָּל שְׁאָר הַפְּרָטִים)".”.
דיווח: כתבו ידיעה קצרה שכותרתה “סוד הרגל האחת וסרגל הבנאים”.
