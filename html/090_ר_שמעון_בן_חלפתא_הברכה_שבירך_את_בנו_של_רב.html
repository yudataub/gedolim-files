הברכה שבירך את בנו של רב

דף עבודה בעקבות ר' שמעון בן חלפתא
שם: כיתה:

המעשה — הסיפור המלא

פְּתִיחָה: רֵיחַ הַפַּרְדֵּסִים וְקוֹל הַתּוֹרָה

יְלָדִים, צְאוּ אִתִּי לְמַסָּע קָסוּם אֶל אֶרֶץ יִשְׂרָאֵל שֶׁל יְמֵי קֶדֶם. אֲנַחְנוּ נִמְצָאִים בָּעִיר צִפּוֹרִי שֶׁבַּגָּלִיל. הַשֶּׁמֶשׁ מַתְחִילָה לָרֶדֶת וְצוֹבַעַת אֶת הַשָּׁמַיִם בִּצְבָעִים מַרְהִיבִים שֶׁל זָהָב, כָּתֹם וְאַרְגָּמָן. בָּאֲוִיר עוֹמֵד רֵיחַ מָתוֹק שֶׁל פַּרְדֵּסֵי תַּפּוּזִים וְרֵיחַ עָטִיק וְנָעִים שֶׁל גְּוִילֵי קְלָף מִבָּתֵּי הַמִּדְרָשׁ. מִכָּל חַלּוֹן נִשְׁמָע קוֹל מִתְנַגֵּן, קוֹלָם שֶׁל חֲכָמִים הַלּוֹמְדִים תּוֹרָה בְּחֵשֶׁק רָב.

בְּאוֹתָהּ עִיר גָּר נְשִׂיא יִשְׂרָאֵל – רַבִּי יְהוּדָה הַנָּשִׂיא, שֶׁכֻּלָּם קָרְאוּ לוֹ "רַבֵּנוּ הַקָּדוֹשׁ". בְּאוֹתוֹ הַיּוֹם, הִגִּיעַ לָעִיר אוֹרֵחַ חָשׁוּב מְאֹד. זֶה הָיָה צַדִּיק עָצוּם שֶׁשְּׁמוֹ הָיָה רַבִּי שִׁמְעוֹן בֶּן חֲלַפְתָּא.

רַבִּי שִׁמְעוֹן בֶּן חֲלַפְתָּא הָיָה מְלֻמָּד בְּנִסִּים. הוּא הָיָה כָּל כָּךְ אָהוּב בַּשָּׁמַיִם, עַד שֶׁפַּעַם אַחַת, כַּאֲשֶׁר הָיָה עָנִי מָרוּד וְלֹא הָיָה לוֹ כֶּסֶף לִקְנוֹת מַאֲכָלִים לִכְבוֹד חַג הַפֶּסַח, הוּא הִתְפַּלֵּל אֶל הַקָּדוֹשׁ בָּרוּךְ הוּא, וּמִן הַשָּׁמַיִם... צָנְחָה לְתוֹךְ כַּפּוֹת יָדָיו מַרְגָּלִית יְקָרָה וְנוֹצֶצֶת שֶׁשָּׁוָה הוֹן עָתֵק! אָדָם שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא שׁוֹמֵעַ כָּךְ אֶת תְּפִלָּתוֹ – הוּא אָדָם שֶׁבִּרְכָתוֹ הִיא כְּמוֹ אֵשׁ שֶׁיּוֹרֶדֶת מֵהַשָּׁמַיִם!

הַשְּׁלִיחוּת שֶׁל הַבֵּן

קָרָא רַבִּי יְהוּדָה הַנָּשִׂיא לִבְנוֹ הַצָּעִיר. גַּם לַבֵּן קָרְאוּ שִׁמְעוֹן.

"שִׁמְעוֹן בְּנִי הַיָּקָר," אָמַר נְשִׂיא יִשְׂרָאֵל, בְּקוֹלוֹ הֶעָמֹק וְהָאוֹהֵב, "הַצַּדִּיק הַגָּדוֹל, רַבִּי שִׁמְעוֹן בֶּן חֲלַפְתָּא נִמְצָא עַכְשָׁו בְּבֵית הַמִּדְרָשׁ שֶׁלָּנוּ. לֵךְ אֵלָיו מַהֵר, בְּנִי! כּוֹף אֶת רֹאשְׁךָ תַּחַת כַּפּוֹת יָדָיו, וּבַקֵּשׁ מִמֶּנּוּ שֶׁיְּבָרֵךְ אוֹתְךָ. מִי יוֹדֵעַ אֵיזוֹ יְשׁוּעָה עֲצוּמָה תִּצְמַח לְךָ מִבִּרְכָּתוֹ!".

הַנַּעַר שִׁמְעוֹן הָיָה נִרְגָּשׁ. הַלֵּב שֶׁלּוֹ דָּפַק בְּחָזְקָה: טּוּם-טּוּם, טּוּם-טּוּם . הוּא מִהֵר לְעֵבֶר בֵּית הַמִּדְרָשׁ. עַל אַבְנֵי הָרְחוֹב הָעַתִּיקוֹת נִשְׁמְעוּ צְעָדָיו הַמְהִירִים. תּוֹךְ כְּדֵי הֲלִיכָה, לֹא יָכוֹל הָיָה שִׁמְעוֹן לְהַפְסִיק לְדַמְיֵן בְּרֹאשׁוֹ.

"אֵיזוֹ בְּרָכָה הוּא יִתֵּן לִי?" חָשַׁב הַנַּעַר. "אוּלַי הוּא יְבָרֵךְ אוֹתִי שֶׁאֶזְכֶּה לִהְיוֹת הָעָשִׁיר הֲכִי גָּדוֹל בָּעוֹלָם? שֶׁאֶזְכֶּה לִמְצֹא תֵּבוֹת אוֹצָר מְלֵאוֹת בְּמַטְבְּעוֹת זָהָב וְיַהֲלוֹמִים? לֹא, אֲנִי בֶּן תּוֹרָה. בֶּטַח הוּא יְבָרֵךְ אוֹתִי בִּבְרָכָה רוּחָנִית עֲצוּמָה! הוּא בְּוַדַּאי יְבָרֵךְ שֶׁאֶדַּע אֶת כָּל הַתּוֹרָה כֻּלָּהּ בְּעַל פֶּה בְּלִי לִשְׁכֹּחַ אֲפִלּוּ אוֹת אַחַת! שֶׁיִּהְיוּ לִי עֲשָׂרוֹת אַלְפֵי תַּלְמִידִים! שֶׁאֶחְיֶה מֵאָה וְעֶשְׂרִים שָׁנִים!".

עִם תִּקְווֹת הַרְקִיעַ שְׁחָקִים, נִכְנַס הַנַּעַר שִׁמְעוֹן לְבֵית הַמִּדְרָשׁ.

הַמִּפְגָּשׁ וְהַבְּרָכָה

הַשֶּׁקֶט שֶׁל קְדֻשָּׁה עָטַף אֶת הֶחָלָל. לְיַד אֶחָד הַשֻּׁלְחָנוֹת, מֵעַל מְגִלַּת קְלָף פְתוּחָה, יָשַׁב הַצַּדִּיק הַזָּקֵן. זְקָנוֹ לָבָן כַּשֶּׁלֶג, וְעֵינָיו בּוֹרְקוֹת כִּשְׁנֵי כּוֹכָבִים מְאִירִים.

הַנַּעַר יָרַד עַל בִּרְכָּיו מִתּוֹךְ כָּבוֹד, הִרְכִּין אֶת רֹאשׁוֹ בַּעֲנָוָה לִפְנֵי הַצַּדִּיק וְלָחַשׁ: "בָּרְכֵנִי, כְּבוֹד הָרַב. אֲנִי בְּנוֹ שֶׁל רַבִּי יְהוּדָה הַנָּשִׂיא".

רַבִּי שִׁמְעוֹן בֶּן חֲלַפְתָּא שָׁלַח מַבָּט חָם אֶל הַנַּעַר. הוּא הֵרִים אֶת שְׁתֵּי יָדָיו הַקְּדוֹשׁוֹת, יָדַיִם שֶׁהֶחְזִיקוּ מַרְגָּלִית מִגַּן עֵדֶן, וְהִנִּיחַ אוֹתָן בְּרֹךְ עַל רֹאשׁוֹ שֶׁל הַנַּעַר. יָדָיו שֶׁל הָרַב הָיוּ חַמּוֹת. הַנַּעַר עָצַם עֵינַיִם. הוּא הִמְתִּין לִשְׁמֹעַ אֶת הַבְּרָכָה שֶׁתַּשְׁפִּיעַ שֶׁפַע גַּשְׁמִי וְרוּחָנִי עַל תְּחוּמֵי חַיָּיו. עַכְשָׁו, חָשַׁב, עַכְשָׁו תֵּרֵד הַבְּרָכָה הַגְּדוֹלָה!

אֲבָל אָז, נִשְׁמַע קוֹלוֹ הַשָּׁקֵט שֶׁל הָרַב:

"יְהִי רָצוֹן מִלְּפָנֶיךָ שֶׁלֹּא תְּבַיֵּשׁ אֶת אֲחֵרִים, וְשֶׁלֹּא תִּתְבַּיֵּשׁ אַתָּה בְּעַצְמְךָ".

הַנַּעַר הִמְתִּין. הוּא חִכָּה לְהֶמְשֵׁךְ. אַךְ זֶה הָיָה הַסּוֹף. הָרַב הִסִּיר אֶת יָדָיו. הַבְּרָכָה הִסְתַּיְּמָה.

הָאַכְזָבָה הַמָּרָה

שִׁמְעוֹן הַנַּעַר קָם מִמְּקוֹמוֹ, בִּמְבוּכָה. הוּא מִלְמֵל "תּוֹדָה רַבָּה, רַבִּי", וְסָב עַל עֲקֵבָיו לָצֵאת מִבֵּית הַמִּדְרָשׁ.

הַהֲלִיכָה שֶׁלּוֹ חֲזָרָה הַבַּיְתָה כְּבָר לֹא הָיְתָה מְהִירָה וּקְלִילָה. הַכְּתֵפַיִם שֶׁלּוֹ נִשְׁמְטוּ מַטָּה, וְכָל צַעַד שֶׁלּוֹ נִרְאָה כָּבֵד כְּאִלּוּ סָחַב אֲבָנִים עַל גַּבּוֹ. לִבּוֹ הָיָה מָלֵא בְּאַכְזָבָה.

"זֶהוּ?!" אָמַר בְּלִבּוֹ, "בִּשְׁבִיל זֶה נִשְׁלַחְתִּי לַצַּדִּיק? 'לֹא לְבַיֵּשׁ וְלֹא לְהִתְבַּיֵּשׁ'? אֲבָל... זוֹ בִּכְלָל לֹא נִשְׁמַעַת כְּמוֹ בְּרָכָה! זֶה נִשְׁמַע רַק כְּמוֹ מִשְׁפַּט מוּסָר שֶׁאוֹמְרִים לְיֶלֶד כְּשֶׁהוּא פּוֹגֵעַ בְּחָבֵר שֶׁלּוֹ. 'שִׁמְעוֹן, תִּזָּהֵר לֹא לְהַעֲלִיב'. בִּשְׁבִיל זֶה בָּאתִי עַד אֵלָיו?"

גִּלּוּי סוֹדָהּ שֶׁל בְּרָכָה

כַּאֲשֶׁר חָזַר שִׁמְעוֹן לְבֵיתוֹ שֶׁל נְשִׂיא יִשְׂרָאֵל, רָאָה אוֹתוֹ רַבִּי יְהוּדָה וּמִיָּד הִבְחִין בְּפָנָיו הַנְּפוּלוֹת.

"מָה קָרָה, בְּנִי הַיָּקָר?" שָׁאַל הָרַב. "מַדּוּעַ אַתָּה כֹּה עָצוּב? הַאִם לֹא זָכִיתָ לִרְאוֹת אֶת הַצַּדִּיק? הַאִם לֹא בֵּרַךְ אוֹתְךָ רַבִּי שִׁמְעוֹן בֶּן חֲלַפְתָּא?"

אָנַח שִׁמְעוֹן, וְעָנָה בְּקוֹל צָרוּד קְצָת מִצַּעַר: "רָאִיתִי אוֹתוֹ, אַבָּא. אֲבָל... הוּא כִּמְעַט לֹא בֵּרַךְ אוֹתִי".

"מַה פֵּרוּשׁ לֹא בֵּרַךְ?" הִתְפַּלֵּא רַבִּי יְהוּדָה. "מָה הוּא אָמַר לְךָ?"

"הוּא לֹא בֵּרַךְ אוֹתִי לֹא בְּכֶסֶף, לֹא בִּבְרִיאוּת וְלֹא בְּהַצְלָחָה בַּתּוֹרָה. הוּא רַק אָמַר לִי בְּשֶׁקֶט צִוּוּי פָּשׁוּט כָּזֶה: 'יְהִי רָצוֹן שֶׁלֹּא תְּבַיֵּשׁ וְלֹא תִּתְבַּיֵּשׁ'".

לְפֶתַע, אוֹר גָּדוֹל זָרַח עַל פָּנָיו שֶׁל רַבִּי יְהוּדָה הַנָּשִׂיא. חִיּוּךְ רָחָב, הֲכִי רָחָב שֶׁשִּׁמְעוֹן רָאָה מֵעוֹדוֹ, הֵאִיר אֶת פְּנֵי אָבִיו. רַבִּי יְהוּדָה קָם, חִבֵּק אֶת בְּנוֹ בְּאַהֲבָה רַבָּה, וְקָרָא מִתּוֹךְ שִׂמְחָה עֲצוּמָה:

"בְּנִי שֶׁלִּי! עַל מָה אַתָּה מֵצֵר? הֲרֵי זוֹהִי הַבְּרָכָה הַגְּדוֹלָה בְּיוֹתֵר שֶׁיֵּשׁ בָּעוֹלָם! רַבִּי שִׁמְעוֹן בֶּן חֲלַפְתָּא לֹא בֵּרַךְ אוֹתְךָ בִּבְרָכָה שֶׁל אֲנָשִׁים. הוּא נָתַן לְךָ אֶת הַבְּרָכָה שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא בִּכְבוֹדוֹ וּבְעַצְמוֹ! "

הַנַּעַר שִׁמְעוֹן הֵרִים מַבָּט מֻפְתָּע: "הַבְּרָכָה שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא? מָה הַכַּוָּנָה, אַבָּא?"

"הַקְשֵׁב הֵיטֵב, בְּנִי," הוֹשִׁיב אוֹתוֹ רַבִּי יְהוּדָה לְצִדּוֹ. "חֲשֹׁב לְרֶגַע מָה קוֹרֶה כַּאֲשֶׁר אָדָם רָעֵב וְאֵין לוֹ כֶּסֶף לִקְנוֹת לֶחֶם? הוּא בָּא לְבַקֵּשׁ צְדָקָה, וּפָנָיו מַאְדִּימוֹת. הוּא מִתְבַּיֵּשׁ! אִם תִּזְכֶּה לֹא לְהִתְבַּיֵּשׁ מֵעוֹלָם, זֶה אוֹמֵר שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא יִדְאַג לְךָ לְפַרְנָסָה טוֹבָה כָּל יְמֵי חַיֶּיךָ, וְלֹא תִּצְטָרֵךְ שׁוּם טוֹבָה מִשּׁוּם אָדָם!

וּמָה קוֹרֶה כַּאֲשֶׁר תַּלְמִיד עוֹמֵד בְּבֵית הַמִּדְרָשׁ, הָרַב שׁוֹאֵל אוֹתוֹ שְׁאֵלָה בַּתּוֹרָה, וְהַתַּלְמִיד אֵינוֹ יוֹדֵעַ לַעֲנוֹת? הוּא מִתְבַּיֵּשׁ! אִם קִבַּלְתָּ בְּרָכָה שֶׁלֹּא תִּתְבַּיֵּשׁ, פֵּרוּשׁ הַדָּבָר שֶׁה' יָאִיר אֶת שִׂכְלְךָ, וְאַתָּה תִּהְיֶה תַּלְמִיד חָכָם עָצוּם, עִם זִכָּרוֹן מוּפְלָא שֶׁכָּל הַתּוֹרָה כֻּלָּהּ מֻנַּחַת בְּרֹאשׁוֹ!

הַלֹּא כָּךְ הִבְטִיחַ הַקָּדוֹשׁ בָּרוּךְ הוּא לְעַם יִשְׂרָאֵל עַל יְדֵי יוֹאֵל הַנָּבִיא בַּפָּסוּק: 'וְלֹא יֵבוֹשׁוּ עַמִּי לְעוֹלָם'! לֹא לְהִתְבַּיֵּשׁ – זוֹהִי הַשְּׁלֵמוּת הֲכִי גְּדוֹלָה שֶׁל הַחַיִּים. זֶה אוֹמֵר שֶׁיֵּשׁ לְךָ הַכֹּל!".

שִׁמְעוֹן הַנַּעַר קָפַץ מִמְּקוֹמוֹ, עֵינָיו בָּרְקוּ עַכְשָׁו בְּהֲבָנָה: "וְלָמָּה אָמַר לִי 'שֶׁלֹּא תְּבַיֵּשׁ אֲחֵרִים'?"

"מִשּׁוּם שֶׁהַדָּבָר הֲכִי נֶחְשָׁב בְּעֵינֵי רִבּוֹנוֹ שֶׁל עוֹלָם אֵינוֹ כַּמָּה הָאָדָם חָכָם, אֶלָּא כַּמָּה לֵב זַךְ וְטָהוֹר יֵשׁ לוֹ! אִם עַד סוֹף חַיֶּיךָ תִּזְכֶּה לֹא לְצַעֵר אַף אֶחָד, לֹא לִגְרֹם לְאַף חָבֵר לַהֲרִיד דִּמְעָה, לֹא לִצְחֹק עַל חָבֵר בַּכִּתָּה... אִם לֹא תְּבַיֵּשׁ אַף אָדָם מֵעוֹלָם – אַתָּה הַצַּדִּיק הֲכִי גָּדוֹל! אָדָם שֶׁזָּהִיר בִּכְבוֹדָם שֶׁל אֲחֵרִים, ה' שׁוֹמֵר עָלָיו יוֹתֵר מִכֻּלָּם!".

מִיָּד הֵבִין שִׁמְעוֹן אֶת גֹּדֶל הָאוֹצָר שֶׁנָּתַן לוֹ הַתַּנָּא הַקָּדוֹשׁ. הוּא לָמַד בְּאוֹתוֹ יָמִים אֶת הַסּוֹד הַגָּדוֹל בְּיוֹתֵר שֶׁל תּוֹרַת יִשְׂרָאֵל: לֹא הַכֶּסֶף עוֹשֶׂה אֶת הָאָדָם לְגָדוֹל. הַדָּבָר הֲכִי חָשׁוּב, הֲכִי קָדוֹשׁ וַהֲכִי יָקָר – הוּא לִשְׁמֹר עַל הַכָּבוֹד, עַל הַשִּׂמְחָה וְעַל הַחִיּוּךְ שֶׁל הֶחָבֵר, וּלְעוֹלָם לֹא לְבַיֵּשׁ.

מקורות: תלמוד בבלי, מסכת מועד קטן, דף ט' עמוד ב' ("רבי אמר ליה לרבי שמעון בריה: זיל לגבי רבי שמעון בן חלפתא דליברכך... אמר ליה מאי ברכך? אמר ליה מילי דהדיוטות אמר לי, דאמר לי יהי רצון שלא תבייש ולא תתבייש... אמר ליה ברכתא דברכוך קודשא בריך הוא לישראל"). מקורות עזר (אווירה והמחשה): שיר השירים רבה ח' - "אוהב מחקר ובע"ח", מדרש רות רבה פרשה ג' - סיפור המרגלית שירדה משמיים לרבי שמעון בן חלפתא.

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “יְלָדִים, צְאוּ אִתִּי לְמַסָּע קָסוּם אֶל אֶרֶץ יִשְׂרָאֵל שֶׁל יְמֵי קֶדֶם.” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “אָדָם שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא שׁוֹמֵעַ כָּךְ אֶת תְּפִלָּתוֹ – הוּא אָדָם שֶׁבִּרְכָתוֹ הִיא כְּמוֹ אֵשׁ שֶׁיּוֹרֶדֶת מֵהַשָּׁמַיִם!”?
ג. כיצד הגיב ר' שמעון בן חלפתא למצב המתואר: “הַנַּעַר יָרַד עַל בִּרְכָּיו מִתּוֹךְ כָּבוֹד, הִרְכִּין אֶת רֹאשׁוֹ בַּעֲנָוָה לִפְנֵי הַצַּדִּיק וְלָחַשׁ: "בָּרְכֵנִי, כְּבוֹד הָרַב.”?
ד. מה השתנה בין “זֶה נִשְׁמַע רַק כְּמוֹ מִשְׁפַּט מוּסָר שֶׁאוֹמְרִים לְיֶלֶד כְּשֶׁהוּא פּוֹגֵעַ בְּחָבֵר שֶׁלּוֹ.” לבין “רַבִּי שִׁמְעוֹן בֶּן חֲלַפְתָּא לֹא בֵּרַךְ אוֹתְךָ בִּבְרָכָה שֶׁל אֲנָשִׁים.”?
ה. כיצד הגיע הסיפור אל האירוע: “אִם לֹא תְּבַיֵּשׁ אַף אָדָם מֵעוֹלָם – אַתָּה הַצַּדִּיק הֲכִי גָּדוֹל!”?

מה קרה אחרי: “הוּא נָתַן לְךָ אֶת הַבְּרָכָה שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא בִּכְבוֹדוֹ וּבְעַצְמוֹ!”?

הוּא רַק אָמַר לִי בְּשֶׁקֶט צִוּוּי פָּשׁוּט כָּזֶה: 'יְהִי רָצוֹן שֶׁלֹּא תְּבַיֵּשׁ וְלֹא תִּתְבַּיֵּשׁ'".

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “הברכה שבירך את בנו של רב” על ר' שמעון בן חלפתא? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “יְהִי רָצוֹן מִלְּפָנֶיךָ שֶׁלֹּא תְּבַיֵּשׁ אֶת אֲחֵרִים, וְשֶׁלֹּא תִּתְבַּיֵּשׁ אַתָּה בְּעַצְמְךָ".”.
ג. אילו ר' שמעון בן חלפתא לא היה פועל כפי שמתואר ב־“רַבִּי שִׁמְעוֹן בֶּן חֲלַפְתָּא לֹא בֵּרַךְ אוֹתְךָ בִּבְרָכָה שֶׁל אֲנָשִׁים.” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “אִם לֹא תְּבַיֵּשׁ אַף אָדָם מֵעוֹלָם – אַתָּה הַצַּדִּיק הֲכִי גָּדוֹל!” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. יְלָדִים, צְאוּ אִתִּי לְמַסָּע קָסוּם אֶל אֶרֶץ יִשְׂרָאֵל שֶׁל יְמֵי…
2. שִׁמְעוֹן בְּנִי הַיָּקָר," אָמַר נְשִׂיא יִשְׂרָאֵל, בְּקוֹלוֹ הֶעָמֹק…
3. רַבִּי שִׁמְעוֹן בֶּן חֲלַפְתָּא שָׁלַח מַבָּט חָם אֶל הַנַּעַר.
4. הוּא לֹא בֵּרַךְ אוֹתִי לֹא בְּכֶסֶף, לֹא בִּבְרִיאוּת וְלֹא…
5. אִם קִבַּלְתָּ בְּרָכָה שֶׁלֹּא תִּתְבַּיֵּשׁ, פֵּרוּשׁ הַדָּבָר שֶׁה'…
6. הַדָּבָר הֲכִי חָשׁוּב, הֲכִי קָדוֹשׁ וַהֲכִי יָקָר – הוּא לִשְׁמֹר עַל…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “הַנַּעַר יָרַד עַל בִּרְכָּיו מִתּוֹךְ כָּבוֹד, הִרְכִּין אֶת רֹאשׁוֹ בַּעֲנָוָה לִפְנֵי הַצַּדִּיק וְלָחַשׁ: "בָּרְכֵנִי,…”.
מכתב: כתבו לר' שמעון בן חלפתא בעקבות “הוּא רַק אָמַר לִי בְּשֶׁקֶט צִוּוּי פָּשׁוּט כָּזֶה: 'יְהִי רָצוֹן שֶׁלֹּא תְּבַיֵּשׁ וְלֹא תִּתְבַּיֵּשׁ'".”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “הַלֹּא כָּךְ הִבְטִיחַ הַקָּדוֹשׁ בָּרוּךְ הוּא לְעַם יִשְׂרָאֵל עַל יְדֵי יוֹאֵל הַנָּבִיא בַּפָּסוּק: 'וְלֹא יֵבוֹשׁוּ עַמִּי…”.
דיווח: כתבו ידיעה קצרה שכותרתה “הברכה שבירך את בנו של רב”.

המחזה

רגע מן הסיפורהדמויותמה יאמרו או יעשו?
הוּא בְּוַדַּאי יְבָרֵךְ שֶׁאֶדַּע אֶת כָּל הַתּוֹרָה כֻּלָּהּ בְּעַל פֶּה בְּלִי לִשְׁכֹּחַ…
רַבִּי שִׁמְעוֹן בֶּן חֲלַפְתָּא לֹא בֵּרַךְ אוֹתְךָ בִּבְרָכָה שֶׁל אֲנָשִׁים.

7. חיבור לחיים מתוך המעשה

א. מה אפשר ללמוד מן הדרך שבה ר' שמעון בן חלפתא פעל באירוע “זֶה נִשְׁמַע רַק כְּמוֹ מִשְׁפַּט מוּסָר שֶׁאוֹמְרִים לְיֶלֶד כְּשֶׁהוּא פּוֹגֵעַ בְּחָבֵר שֶׁלּוֹ.”?
