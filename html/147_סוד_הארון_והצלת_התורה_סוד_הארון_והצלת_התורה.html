🔹 שָׁלָב 1: בֵּרוּר מַקְדִּים וְנִתּוּחַ (לַמּוֹרֶה/לַמְסַפֵּר)

🔹 שָׁלָב 2: הַסִּפּוּר הַמָּלֵא – "סוֹד הָאָרוֹן וְהַצָּלַת הַתּוֹרָה"

פְּתִיחָה: חֹשֶׁךְ עַל יְרוּשָׁלַיִם יְלָדִים, אֲנַחְנוּ חוֹזְרִים כִּמְעַט אַלְפַּיִם שָׁנָה אָחוֹרָה, לַיָּמִים הַקָּשִׁים בְּיוֹתֵר בְּתוֹלְדוֹת יְרוּשָׁלַיִם. צְבָא רוֹמִי הָאַכְזָר, עִם אַלְפֵי חַיָּלִים, סוּסִים וּכְלֵי מִלְחָמָה, צָר עַל יְרוּשָׁלַיִם. אִישׁ לֹא נִכְנַס וְאִישׁ לֹא יוֹצֵא. בְּתוֹךְ הָעִיר הַנְּצוּרָה הָיוּ אֲנָשִׁים עֲשִׁירִים מְאֹד, שֶׁהֵכִינוּ מַחְסָנִים גְּדוֹלִים מְלֵאִים בְּחִטָּה, יַיִן, מֶלַח וְעֵצִים לְהַסָּקָה, שֶׁהִסְפִּיקוּ לְעֶשְׂרִים וְאַחַת שָׁנִים! יְרוּשָׁלַיִם יָכְלָה לְהַחְזִיק מַעֲמָד זְמַן רַב.

אֲבָל בְּתוֹךְ הָעִיר, הָיוּ קְבוּצוֹת שֶׁל יְהוּדִים שֶׁנִּקְרְאוּ "בִּרְיוֹנִים". הֵם רָצוּ לְהִלָּחֵם בָּרוֹמָאִים בְּכֹחַ הַזְּרוֹעַ, וְלֹא רָצוּ לִשְׁמֹעַ עַל שׁוּם הֶסְכֵּם שָׁלוֹם אוֹ פְּשָׁרָה. כְּדֵי לְאַלֵּץ אֶת כָּל תּוֹשָׁבֵי הָעִיר לָצֵאת לְהִלָּחֵם, קָמוּ הַבִּרְיוֹנִים וְעָשׂוּ מַעֲשֶׂה נוֹרָא: הֵם... שָׂרְפוּ אֶת כָּל מַחְסְנֵי הַמָּזוֹן! הֵם שָׂרְפוּ אֶת הַחִטָּה וְאֶת הָעֵצִים! לְפֶתַע, רָעָב אָיוֹם וְנוֹרָא הִכָּה בִּירוּשָׁלַיִם.

הָרְאִיָּה שֶׁל מַנְהִיג הַדּוֹר גְּדוֹל הַדּוֹר בְּאוֹתָם יָמִים הָיָה הַנָּשִׂיא, הַתַּנָּא הַקָּדוֹשׁ רַבָּן יוֹחָנָן בֶּן זַכַּאי. עוֹד אַרְבָּעִים שָׁנָה לִפְנֵי שֶׁהִתְחִילָה הַמִּלְחָמָה, רַבָּן יוֹחָנָן כְּבָר יָדַע שֶׁאֲסוֹן מִתְקָרֵב. הוּא רָאָה נִסִּים מְשֻׁנִּים בְּבֵית הַמִּקְדָּשׁ: לְדֻגְמָה, דַּלְתוֹת הַהֵיכָל הַכְּבֵדוֹת הָיוּ נִפְתָּחוֹת מֵעַצְמָן בַּלַּיְלָה! רַבָּן יוֹחָנָן גָּעַר בָּהֶן וְאָמַר: "הֵיכָל, הֵיכָל! לָמָּה אַתָּה מַפְחִיד אֶת עַצְמְךָ? יוֹדֵעַ אֲנִי בְּךָ שֶׁסּוֹפְךָ לֵחָרֵב!". כָּךְ נִבְּאוּ הַנְּבִיאִים.

עַכְשָׁו, כְּשֶׁרָאָה רַבָּן יוֹחָנָן אֶת הָרָעָב הַקָּשֶׁה וְאֶת הַשִּׂנְאָה שֶׁבְּתוֹךְ הָעִיר, וְאֶת צְבָא רוֹמִי בַּחוּץ, לִבּוֹ נֶחְרַד. הוּא אָמַר לְעַצְמוֹ: "הָעִיר הַזּוֹ הולֶכֶת לִפֹּל! וְאִם יְרוּשָׁלַיִם תֵּחָרֵב, הָרוֹמָאִים יַהַרְגוּ אֶת כָּל חַכְמֵי יִשְׂרָאֵל. מִי יַלְמִיד תּוֹרָה לַיְּלָדִים? מִי יִשְׁמֹר אֶת הַתּוֹרָה שֶׁבְּעַל-פֶּה לַדּוֹרוֹת הַבָּאִים?! אֲנִי חַיָּב לְהַצִּיל אֶת הַתּוֹרָה!".

רַבָּן יוֹחָנָן בֶּן זַכַּאי קָרָא אֵלָיו בַּסֵּתֶר לְאַבָּא סִיקְרָא. מִי זֶה הָיָה? הוּא הָיָה הַמְפַקֵּד שֶׁל הַבִּרְיוֹנִים שֶׁשָּׁמְרוּ עַל הַחוֹמוֹת... וְהוּא גַּם הָיָה הָאַחְיָן שֶׁל רַבָּן יוֹחָנָן! "עַד מָתַי אַתֶּם הוֹרְגִים אֶת כֻּלָּם בָּרָעָב?!" שָׁאַל רַבָּן יוֹחָנָן בִּכְאֵב. "מָה אֶעֱשֶׂה, דּוֹדִי?" הֵשִׁיב אַבָּא סִיקְרָא בְּפַחַד, "אִם אוֹמַר לַבִּרְיוֹנִים מִלָּה עַל שָׁלוֹם, הֵם יַהַרְגוּ אוֹתִי מִיָּד!". "אִם כָּךְ," לָחַשׁ רַבָּן יוֹחָנָן, "אַתָּה חַיָּב לַעֲזֹר לִי לָצֵאת מֵהָעִיר, כְּדֵי שֶׁאוּכַל לְהַצִּיל מְעַט מֵעַם יִשְׂרָאֵל". "אֲבָל אִישׁ אֵינוֹ יוֹצֵא מִשַּׁעֲרֵי יְרוּשָׁלַיִם," אָמַר אַבָּא סִיקְרָא, "רַק... מֵתִים מוֹצִיאִים מִחוּץ לָעִיר לִקְבוּרָה". "אָז תּוֹצִיאוּ אוֹתִי כְּאִלּוּ אֲנִי מֵת!" הֶחְלִיט גְּדוֹל הַדּוֹר בִּמְסִירוּת נֶפֶשׁ.

הַהַבְרָחָה הַמּוֹתַחַת אַבָּא סִיקְרָא נָתַן אֶת הַתָּכְנִית: "תַּעֲשֶׂה אֶת עַצְמְךָ חוֹלֶה מְאֹד. אֲנָשִׁים יָבוֹאוּ לְבַקֵּר. אַחַר כָּךְ יָפִיצוּ שְׁמוּעָה עֲצוּבָה שֶׁחָלִילָה נִפְטַרְתָּ! תִּכָּנֵס לְתוֹךְ אֲרוֹן מֵתִים מֵעֵץ. וּכְדֵי שֶׁלֹּא יַחְשְׁדוּ (כִּי מֵת כָּבֵד יוֹתֵר מֵאָדָם חַי וְהַשּׁוֹמְרִים יַרְגִּישׁוּ), כַּנִּיסוּ אִתְּךָ חֲתִיכוֹת בָּשָׂר שֶׁל נְבֵלָה כְּדֵי שֶׁיִּהְיֶה כָּבֵד וְגַם יַעֲלֶה רֵיחַ רַע, כְּדֵי שֶׁלֹּא יִרְצוּ לִבְדֹּק מִקָּרוֹב".

כָּךְ הָיָה. אֲרוֹן הָעֵץ הַסָּגוּר נִשָּׂא בִּזְהִירוּת עַל כִּתְפֵיהֶם שֶׁל שְׁנֵי תַּלְמִידָיו הַקְּדוֹשִׁים: רַבִּי אֱלִיעֶזֶר בֶּן הוֹרְקְנוֹס וְרַבִּי יְהוֹשֻׁעַ בֶּן חֲנַנְיָה. אִישׁ לֹא סָחַב אֶת הָאָרוֹן מִלְּבַדָּם! הֵם הִגִּיעוּ לְשַׁעֲרֵי יְרוּשָׁלַיִם. אוֹי, אֵיזֶה פַּחַד! הַבִּרְיוֹנִים עָמְדוּ בַּשַּׁעַר, אֲחוּזֵי חֲרָבוֹת שְׁלוּפוֹת. הֵם רָאוּ אֶת הָאָרוֹן וְצָעֲקוּ: "אֲנַחְנוּ רוֹצִים לִדְקֹר בְּתוֹךְ הָאָרוֹן לִרְאוֹת שֶׁזֶּה בֶּאֱמֶת אָדָם מֵת!". רַבִּי אֱלִיעֶזֶר הֶחְוִיר. אִם יִדְקְרוּ בַּחֶרֶב – יַהַרְגוּ אֶת רַבָּן יוֹחָנָן! אֲבָל אַבָּא סִיקְרָא צָעַק בְּחָכְמָה לַשּׁוֹמְרִים: "מָה פִּתְאוֹם! הָרוֹמָאִים יַגִּידוּ שֶׁאֲנַחְנוּ כָּל כָּךְ אַכְזָרִיִּים שֶׁהָרַגְנוּ וְדָקַרְנוּ אֶת רַבֵּנוּ בִּכְבוֹדוֹ וּבְעַצְמוֹ! פִּתְחוּ אֶת הַשַּׁעַר!". הַשּׁוֹמְרִים פָּחֲדוּ, וּפָתְחוּ אֶת הַשַּׁעַר. רַבָּן יוֹחָנָן בֶּן זַכַּאי הוּבָא בְּהַצְלָחָה אֶל מִחוּץ לַחוֹמוֹת!

הַנַּעַל שֶׁהִתְכַּוְּצָה הַתַּלְמִידִים פָּתְחוּ אֶת הָאָרוֹן, וְרַבָּן יוֹחָנָן צָעַד יָשָׁר אֶל אֹהֶל הַפִּקּוּד שֶׁל שַׂר הַצָּבָא הָרוֹמָאִי – אַסְפַּסְיָאנוּס. "שָׁלוֹם עָלֶיךָ, הַמֶּלֶךְ! שָׁלוֹם עָלֶיךָ, הַמֶּלֶךְ!" בֵּרֵךְ אוֹתוֹ רַבָּן יוֹחָנָן בְּכָבוֹד. כָּעַס אַסְפַּסְיָאנוּס וְשָׁאַף עֵינַיִם אֲיֻמּוֹת: "אַתָּה מְדַבֵּר שֶׁקֶר! אֲנִי לֹא מֶלֶךְ הָרוֹמָאִים, אֲנִי רַק שַׂר צָבָא. הַקֵּיסָר יוֹשֵׁב בְּרוֹמָא! עַל הַשֶּׁקֶר הַזֶּה מַגִּיעַ לְךָ מָוֶת!". חִיֵּךְ רַבָּן יוֹחָנָן וְאָמַר: "אִם אַתָּה עֲדַיִן לֹא מֶלֶךְ, אַתָּה עוֹמֵד לִהְיוֹת מֶלֶךְ! כִּי לְבָנוֹן הָאַדִּיר, שֶׁזֶּה בֵּית הַמִּקְדָּשׁ, נוֹפֵל רַק בְּיָדָיו שֶׁל מֶלֶךְ אַדִּיר, כְּמוֹ שֶׁנִּבְּאָה הַתּוֹרָה".

עוֹדָם מְדַבְּרִים, וְהִנֵּה – סוּס מְהִיר-מֵרוּץ הִגִּיעַ מֵרוֹמָא. הַשָּׁלִיחַ קָפַץ מֵהַסּוּס וְצָעַק: "אַסְפַּסְיָאנוּס! הַקֵּיסָר שֶׁלָּנוּ מֵת, וְשָׂרֵי הַמְּדִינָה בָּחֲרוּ בְּךָ לִהְיוֹת הַקֵּיסָר הֶחָדָשׁ! מֵהַיּוֹם אַתָּה הַמֶּלֶךְ שֶׁלָּנוּ!". אַסְפַּסְיָאנוּס קָם עַל רַגְלָיו נִדְהָם! הָרַב הַיְּהוּדִי הַזָּקֵן נִבֵּא אֶת הָאֱמֶת! מֵרֹב שִׂמְחָה וַהֲתְרַגְּשׁוּת טוֹבָה... קָרָה נֵס. אַסְפַּסְיָאנוּס נִסָּה לִנְעֹל אֶת נַעֲלוֹ הַצְּבָאִית עַל רַגְלוֹ, וּפִתְאוֹם... הַנַּעַל לֹא נִכְנֶסֶת! הָרֶגֶל גָּדְלָה! "מָה קוֹרֶה לִי?!" נִבְהַל הַקֵּיסָר. אָמַר לוֹ רַבָּן יוֹחָנָן בְּרֹגַע: "אַל תִּפְחַד. סֵפֶר מִשְׁלֵי קוֹבֵעַ 'שְׁמוּעָה טוֹבָה תְּדַשֶּׁן עֶצֶם'. שָׁמַעְתָּ בְּשׂוֹרָה כָּל כָּךְ טוֹבָה, הָעֲצָמוֹת שֶׁלְּךָ הִתְרַחֲבוּ מֵרֹב שִׂמְחָה!". "וּמָה אֶעֱשֶׂה כְּדֵי שֶׁהַנַּעַל תִּכָּנֵס?" "פָּשׁוּט מְאֹד," עָנָה רַבָּן יוֹחָנָן, "תַּכְנִיס לַחֶדֶר אָדָם שֶׁאַתָּה שׂוֹנֵא. הַכַּעַס כּוֹוֵץ אֶת הָעֲצָמוֹת בַּחֲזָרָה!". וְאָכֵן כָּךְ הָיָה. הַנַּעַל נִכְנְסָה!

הוּא הָיָה הַמְפַקֵּד שֶׁל הַבִּרְיוֹנִים שֶׁשָּׁמְרוּ עַל הַחוֹמוֹת...
הַקֵּיסָר שֶׁלָּנוּ מֵת, וְשָׂרֵי הַמְּדִינָה בָּחֲרוּ בְּךָ לִהְיוֹת הַקֵּיסָר הֶחָדָשׁ!

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “סוד הארון והצלת התורה” על סוד הארון והצלת התורה? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “אֲבָל אַבָּא סִיקְרָא צָעַק בְּחָכְמָה לַשּׁוֹמְרִים: "מָה פִּתְאוֹם!”.
ג. אילו סוד הארון והצלת התורה לא היה פועל כפי שמתואר ב־“אֲבָל בִּגְלַל שֶׁהָיִיתָ חָכָם כָּל כָּךְ, אֲנִי נוֹתֵן לְךָ רְשׁוּת לְבַקֵּשׁ בַּקָּשָׁה!” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “🔹 שָׁלָב 4: מַדְרִיךְ לַמּוֹרֶה / לַמְסַפֵּר (לְשִׁלּוּב בַּמַּצֶּגֶת וּבַשִּׁעוּר)” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. 🔹 שָׁלָב 2: הַסִּפּוּר הַמָּלֵא – "סוֹד הָאָרוֹן וְהַצָּלַת הַתּוֹרָה
2. עַכְשָׁו, כְּשֶׁרָאָה רַבָּן יוֹחָנָן אֶת הָרָעָב הַקָּשֶׁה וְאֶת…
3. אֲרוֹן הָעֵץ הַסָּגוּר נִשָּׂא בִּזְהִירוּת עַל כִּתְפֵיהֶם שֶׁל שְׁנֵי…
4. כִּי לְבָנוֹן הָאַדִּיר, שֶׁזֶּה בֵּית הַמִּקְדָּשׁ, נוֹפֵל רַק…
5. לָכֵן, רַבָּן יוֹחָנָן בֶּן זַכַּאי בִּקֵּשׁ שָׁלוֹשׁ בַּקָּשׁוֹת…
6. אִם הַיּוֹם מֶלֶךְ מֵדוּמְיָן אוֹ נָשִׂיא אָמֶרִיקָאִי אוֹמֵר לָכֶם,…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “הַהַבְרָחָה הַמּוֹתַחַת אַבָּא סִיקְרָא נָתַן אֶת הַתָּכְנִית: "תַּעֲשֶׂה אֶת עַצְמְךָ חוֹלֶה מְאֹד.”.
מכתב: כתבו לסוד הארון והצלת התורה בעקבות “הַקֵּיסָר שֶׁלָּנוּ מֵת, וְשָׂרֵי הַמְּדִינָה בָּחֲרוּ בְּךָ לִהְיוֹת הַקֵּיסָר הֶחָדָשׁ!”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “שׁוֹשֶׁלֶת הַנְּשִׂיאוּת: הוּא בִּקֵּשׁ לִשְׁמֹר מִמָּוֶת עַל הַמִּשְׁפָּחָה שֶׁל רַבָּן גַּמְלִיאֵל, בְּנֵי מִשְׁפַּחַת דָּוִד…”.
דיווח: כתבו ידיעה קצרה שכותרתה “סוד הארון והצלת התורה”.
