לשון טובה ורעה

דף עבודה בעקבות רבן שמעון בן גמליאל
שם: כיתה:

המעשה — הסיפור המלא

פְּתִיחָה: קוֹל הַתּוֹרָה בְּבֵית הַנָּשִׂיא

יְלָדִים, אִם הֱיִיתֶם עוֹמְדִים בַּפֶּתַח שֶׁל בֵּית רַבָּן שִׁמְעוֹן בֶּן גַּמְלִיאֵל, נְשִׂיא יִשְׂרָאֵל, מָה הֱיִיתֶם שׁוֹמְעִים?

הֱיִיתֶם שׁוֹמְעִים קוֹל אַדִּיר, מָתוֹק וְנָעִים, שֶׁל מֵאוֹת תַּלְמִידֵי חֲכָמִים הַלּוֹמְדִים תּוֹרָה. קוֹלוֹת שֶׁל לִמּוּד הָיוּ מִתְעוֹפְפִים מִבַּעַד לַחַלּוֹנוֹת וּמְמַלְאִים אֶת כָּל הָרְחוֹב בִּקְדֻשָּׁה. רֵיחַ נָעִים שֶׁל קְלָפִים יְשָׁנִים שֶׁל סִפְרֵי תּוֹרָה מִלֵּא אֶת הָאֲוִיר.

רַבָּן שִׁמְעוֹן בֶּן גַּמְלִיאֵל יָשַׁב בְּרֹאשׁ הַשֻּׁלְחָן. פָּנָיו הָיוּ מְאִירוֹת, זְקָנוֹ הַלָּבָן גָּלַשׁ עַל חָזֵהוּ, וְעֵינָיו הַטְּהוֹרוֹת הִבִּיטוּ בְּתַלְמִידָיו בְּאַהֲבָה רַבָּה. הַנָּשִׂיא יָדַע שֶׁתַּלְמִידָיו הֵם עֲנָקִים בַּתּוֹרָה, אַךְ כְּמַנְהִיג שֶׁל עַם יִשְׂרָאֵל, הָיָה לוֹ חָשׁוּב מְאֹד דָּבָר נוֹסָף: בֵּין אָדָם לַחֲבֵרוֹ. אֵיךְ הֵם מְדַבְּרִים אֶחָד לַשֵּׁנִי. הַנָּשִׂיא רָצָה לְלַמֵּד אוֹתָם וְאֶת כָּל בְּנֵי בֵּיתוֹ שִׁעוּר חָשׁוּב – לֹא דֶּרֶךְ סֵפֶר, אֶלָּא דֶּרֶךְ מַעֲשֶׂה שֶׁלֹּא יִשְׁכְּחוּ לְעוֹלָם.

הַבַּקָּשָׁה הַמּוּזָרָה

יוֹם אֶחָד, לְאַחַר סִיּוּם הַשִּׁעוּר, קָרָא אֵלָיו הַנָּשִׂיא אֶת מְשָׁרְתוֹ הַנֶּאֱמָן.

הַמְשָׁרֵת נִכְנַס בַּעֲנָוָה, הִרְכִּין אֶת רֹאשׁוֹ וְאָמַר: "כֵּן, אֲדוֹנִי הַנָּשִׂיא? מָה אוּכַל לַעֲשׂוֹת לְמַעַנְךָ?"

"שְׁמַע נָא, אִישׁ יָקָר," אָמַר רַבָּן שִׁמְעוֹן בְּקוֹלוֹ הָרַךְ, "קַח עִמְּךָ סְכוּם נִכְבָּד שֶׁל מַטְבְּעוֹת זָהָב. לֵךְ אֶל הַשּׁוּק הַגָּדוֹל, וּקְנֵה עֲבוּרִי אֶת דְּבַר הַמַּאֲכָל הֲכִי טוֹב, הֲכִי טָעִים וְהֲכִי מְשֻׁבָּח שֶׁיֵּשׁ שָׁם. אַל תַּחְסֹךְ בְּכֶסֶף. הָבֵא אֶת הַדָּבָר הַטּוֹב בְּיוֹתֵר בָּעוֹלָם!".

הַמְשָׁרֵת הִנְהֵן בְּרֹאשׁוֹ, לָקַח אֶת שַׂקִּית הַמַּטְבְּעוֹת שֶׁהִשְׁמִיעָה קוֹל נְקִישׁוֹת צְלִינְג צְלִינְג , וְיָצָא לַדֶּרֶךְ.

בְּהֲמוּלַת הַשּׁוּק

כְּשֶׁהִגִּיעַ הַמְשָׁרֵת לַשּׁוּק, תָּקְפוּ אוֹתוֹ מִכָּל עֵבֶר רֵיחוֹת חֲזָקִים וּצְעָקוֹת. "לֶחֶם חַם שֶׁיָּצָא מֵהַתַּנּוּר!", קָרָא הָאוֹפֶה; "תַּפּוּחִים אֲדֻמִּים וּמְתוּקִים כִּדְבַשׁ!", קָרָא מוֹכֵר הַפֵּרוֹת; רֵיחַ עַז שֶׁל זַעְפְרָן וְכַמּוֹן עָלָה מִדּוּכְנֵי הַתַּבְלִינִים, וַהֲמוֹן אָדָם דָּחֲפוּ זֶה אֶת זֶה.

אַךְ הַמְשָׁרֵת עָבַר דֶּרֶךְ כֻּלָּם. הוּא לֹא חִפֵּשׂ לֶחֶם, וְלֹא פֵּרוֹת. הוּא יָדַע בְּדִיּוּק מָה הוּא הַמַּאֲכָל הֲכִי נֶחְשָׁב. הוּא נִגַּשׁ אֶל מוֹכֵר הַבָּשָׂר (הַקַּצָּב).

"תֵּן לִי בְּבַקָּשָׁה אֶת הַחֵלֶק הֲכִי טוֹב, הֲכִי עָדִין וַהֲכִי מְשֻׁבָּח שֶׁיֵּשׁ לְךָ," בִּקֵּשׁ הַמְשָׁרֵת.

הַקַּצָּב חִיֵּךְ, לָקַח סַכִּין חַדָּה וַחֲלָקָה, קָצַץ חֲתִיכָה נָאָה מְאֹד, אָרַז אוֹתָהּ בְּעָלִים נְקִיִּים וְנָתַן לַמְשָׁרֵת. "זֶה בִּשְׁבִיל הַנָּשִׂיא? אֵין דָּבָר טוֹב מִזֶּה," אָמַר.

הַמְשָׁרֵת חָזַר לְבֵית הַנָּשִׂיא. הוּא נִכְנַס לַמִּטְבָּח, בִּשֵּׁל אֶת הַמַּאֲכָל עִם תַּבְלִינִים עֲדִינִים שֶׁהֵפִיצוּ רֵיחַ נִפְלָא שֶׁעָלָה מֵהַסִּיר הַמַּבְעְבֵּעַ, וְהִגִּישׁ אֶת הַצַּלַּחַת הַמְּעֻטֶּרֶת לְרַבָּן שִׁמְעוֹן.

הַנָּשִׂיא הֵסִיר אֶת הַמִּכְסֶה מֵהַצַּלַּחַת, הִבִּיט בַּתַּבְשִׁיל, וְחִיֵּךְ. עַל הַצַּלַּחַת הָיְתָה מֻנַּחַת לָשׁוֹן . לָשׁוֹן שֶׁל בְּהֵמָה, מְבֻשֶּׁלֶת וְטָעִימָה.

"יָפֶה מְאֹד," אָמַר הַנָּשִׂיא, וְחָלַק אֶת הַסְּעוּדָה עִם מְקֹרָבָיו.

הַהוֹרָאָה הַשְּׁנִיָּה: הַדָּבָר הָרַע בְּיוֹתֵר

עָבְרוּ כַּמָּה יָמִים. וְשׁוּב, קָרָא הַנָּשִׂיא לִמְשָׁרְתוֹ.

"עַתָּה," אָמַר הַנָּשִׂיא וְקוֹלוֹ נִשְׁמַע רְצִינִי מְאֹד, תּוֹךְ כְּדֵי שֶׁכַּמָּה תַּלְמִידִים בַּחֶדֶר עוֹצְרִים אֶת נְשִׁימָתָם וּמַקְשִׁיבִים. "אֲנִי מְבַקֵּשׁ שֶׁתֵּלֵךְ שׁוּב לַשּׁוּק. אַךְ הַפַּעַם – קְנֵה עֲבוּרִי אֶת הַמַּאֲכָל הֲכִי גָּרוּעַ. הֲכִי רַע, הֲכִי שָׁפֵל וְהֲכִי בִּלְתִּי-רָאוּי שֶׁתִּמְצָא בְּכָל הַשּׁוּק!".

הַמְשָׁרֵת יָצָא אֶל שׁוּק הַבָּשָׂר. שׁוּב הוּא הָלַךְ בֵּין הַדּוּכָנִים. יְלָדִים, אִם מִישֶׁהוּ הָיָה מְבַקֵּשׁ מִכֶּם לִקְנוֹת מַשֶּׁהוּ רַע, בְּוַדַּאי הֱיִיתֶם מְחַפְּשִׂים אֵיזֶה תַּפּוּחַ רָקוּב, אוֹ מַשֶּׁהוּ קָשֶׁה מְאֹד כְּמוֹ עֶצֶם שֶׁאִי אֶפְשָׁר לֶאֱכֹל, נָכוֹן?

אֲבָל הַמְשָׁרֵת הַחָכָם חָשַׁב, וְחָשַׁב... וְאָז פָּנָה לַקַּצָּב וּבִקֵּשׁ: "תֵּן לִי שׁוּב אֶת אוֹתוֹ הַמַּאֲכָל שֶׁנָּתַתָּ לִי בַּפַּעַם הַקּוֹדֶמֶת".

הוּא חָזַר לְבֵית הַנָּשִׂיא.

הַהַפְתָּעָה תַּחַת הַמִּכְסֶה

סְבִיב הַשֻּׁלְחָן כְּבָר יָשְׁבוּ הַתַּלְמִידִים הַגְּדוֹלִים. כֻּלָּם הָיוּ מְסֻקְרָנִים וּמְתוּחִים לָדַעַת – מָה הוּא הַמַּאֲכָל "הֲכִי גָּרוּעַ בָּעוֹלָם" שֶׁיָּבִיא הַמְשָׁרֵת?

הַמְשָׁרֵת נִכְנַס, נָשָׂא בְּיָדָיו אֶת קְעָרַת הַחֶרֶס וְהִנִּיחַ אוֹתָהּ בְּאֶמְצַע הַשֻּׁלְחָן. עֵינֵי כֻּלָּם הָיוּ נְשׂוּאוֹת אֶל הַקְּעָרָה. רַבָּן שִׁמְעוֹן בֶּן גַּמְלִיאֵל קָם, הוֹשִׁיט אֶת יָדוֹ וְהֵרִים בְּאִטִּיּוּת אֶת הַמִּכְסֶה...

קְרִיאוֹת פְּלִיאָה קַלּוֹת נִשְׁמְעוּ בַּחֶדֶר. תַּחַת הַמִּכְסֶה, מֻנַּחַת לָהּ – לָשׁוֹן! בְּדִיּוּק אוֹתוֹ חֵלֶק שֶׁהֵבִיא בַּפַּעַם הַקּוֹדֶמֶת!

רַבָּן שִׁמְעוֹן סוֹבֵב אֶת מַבָּטוֹ, וְהִבִּיט בַּמְשָׁרֵת. קוֹלוֹ הִדְהֵד בַּחֶדֶר הַשָּׁקֵט:

"אֱמֹר לִי... כַּאֲשֶׁר אָמַרְתִּי לְךָ: 'לֵךְ הָבֵא אֶת הַדָּבָר הַטּוֹב בְּיוֹתֵר' – הֵבֵאתָ לָשׁוֹן! וְעַכְשָׁו, כְּשֶׁאָמַרְתִּי לְךָ: 'הָבֵא אֶת הַדָּבָר הָרַע בְּיוֹתֵר' – גַּם כֵּן הֵבֵאתָ לָשׁוֹן?! הָכֵיצַד יִתָּכֵן שֶׁאוֹתוֹ דָּבָר עַצְמוֹ הוּא גַּם הֲכִי טוֹב וְגַם הֲכִי רַע?!"

הַמְשָׁרֵת עָמַד מוּל גְּדוֹלֵי הַתּוֹרָה, חִיֵּךְ בַּעֲדִינוּת, וְעָנָה אֶת הַתְּשׁוּבָה שֶׁנִּכְתְּבָה לָנֶצַח בְּסִפְרֵי חֲזַ"ל:

"כֵּן, אֲדוֹנִי הַנָּשִׂיא. מִן הַלָּשׁוֹן – יוֹצֵא הַטּוֹב, וּמִן הַלָּשׁוֹן – יוֹצֵא הָרַע. מִשֶּׁהִיא טוֹבָה – אֵין טוֹב מִמֶּנָּה בָּעוֹלָם! וּמִשֶּׁהִיא רָעָה – אֵין רַע מִמֶּנָּה בָּעוֹלָם! ".

הַלֶּקַח הַנִּצְחִי

שֶׁקֶט הִשְׂתָּרֵר בְּבֵית הַמִּדְרָשׁ. הַתַּלְמִידִים הֵבִינוּ מִיָּד אֶת עֹמֶק הַמּוּסָר. הַנָּשִׂיא שָׂמַח מְאֹד מִתְּשׁוּבָתוֹ הַחֲכָמָה שֶׁל מְשָׁרְתוֹ.

הוּא הִבִּיט בַּתַּלְמִידִים, וְאָמַר לָהֶם בְּאַהֲבָה: "רְאוּ מַהוּ כֹּחוֹ שֶׁל הַדִּבּוּר. כְּשֶׁלָּשׁוֹן שֶׁל אָדָם מְדַבֶּרֶת טוֹב – כְּשֶׁהִיא לוֹמֶדֶת תּוֹרָה, אֲהוּבָה, אוֹמֶרֶת מִלָּה טוֹבָה לְחָבֵר שֶׁעָצוּב, עוֹדֶדֶת, וּמְפַיֶּסֶת – הִיא הַכְּלִי הֲכִי מְתוֹק, הֲכִי מֵאִיר וַהֲכִי טוֹב שֶׁבָּרָא הַקָּדוֹשׁ בָּרוּךְ הוּא!

אַךְ חָלִילָה וְחַס... כַּאֲשֶׁר הַלָּשׁוֹן מְדַבֶּרֶת רַע – כְּשֶׁהִיא מַעֲלִיבָה, פּוֹגַעַת, לוֹעֶגֶת, אוֹ אוֹמֶרֶת לָשׁוֹן הָרָע עַל יְהוּדִי אַחֵר – אֵין דָּבָר הוֹרֵס מִמֶּנָּה! כִּי דִּבּוּר קָשֶׁה זֶה מַשֶּׁהוּ שֶׁמְּצַעֵר אֲחֵרִים וְאִי אֶפְשָׁר לָקַחַת אוֹתוֹ בַּחֲזָרָה".

1. שאלות ידע מתוך הסיפור

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “לשון טובה ורעה” על רבן שמעון בן גמליאל? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “הַקַּצָּב חִיֵּךְ, לָקַח סַכִּין חַדָּה וַחֲלָקָה, קָצַץ חֲתִיכָה נָאָה מְאֹד, אָרַז אוֹתָהּ בְּעָלִים נְקִיִּים וְנָתַן לַמְשָׁרֵת.”.
ג. אילו רבן שמעון בן גמליאל לא היה פועל כפי שמתואר ב־“רַבָּן שִׁמְעוֹן בֶּן גַּמְלִיאֵל קָם, הוֹשִׁיט אֶת יָדוֹ וְהֵרִים בְּאִטִּיּוּת אֶת הַמִּכְסֶה...” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “כְּשֶׁלָּשׁוֹן שֶׁל אָדָם מְדַבֶּרֶת טוֹב – כְּשֶׁהִיא לוֹמֶדֶת תּוֹרָה, אֲהוּבָה, אוֹמֶרֶת מִלָּה טוֹבָה לְחָבֵר שֶׁעָצוּב,…” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. הֱיִיתֶם שׁוֹמְעִים קוֹל אַדִּיר, מָתוֹק וְנָעִים, שֶׁל מֵאוֹת…
2. יוֹם אֶחָד, לְאַחַר סִיּוּם הַשִּׁעוּר, קָרָא אֵלָיו הַנָּשִׂיא אֶת…
3. הוּא יָדַע בְּדִיּוּק מָה הוּא הַמַּאֲכָל הֲכִי נֶחְשָׁב.
4. אַךְ הַפַּעַם – קְנֵה עֲבוּרִי אֶת הַמַּאֲכָל הֲכִי גָּרוּעַ.
5. הַמְשָׁרֵת עָמַד מוּל גְּדוֹלֵי הַתּוֹרָה, חִיֵּךְ בַּעֲדִינוּת,…
6. כָּךְ יָשְׁבוּ הַתַּלְמִידִים בְּצֵל הַנָּשִׂיא, וְלָמְדוּ שֶׁהַפֶּה…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “לֶחֶם חַם שֶׁיָּצָא מֵהַתַּנּוּר!", קָרָא הָאוֹפֶה; "תַּפּוּחִים אֲדֻמִּים וּמְתוּקִים כִּדְבַשׁ!", קָרָא מוֹכֵר הַפֵּרוֹת; רֵיחַ…”.
מכתב: כתבו לרבן שמעון בן גמליאל בעקבות “הֲכִי רַע, הֲכִי שָׁפֵל וְהֲכִי בִּלְתִּי-רָאוּי שֶׁתִּמְצָא בְּכָל הַשּׁוּק!".”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “הַמְשָׁרֵת עָמַד מוּל גְּדוֹלֵי הַתּוֹרָה, חִיֵּךְ בַּעֲדִינוּת, וְעָנָה אֶת הַתְּשׁוּבָה שֶׁנִּכְתְּבָה לָנֶצַח בְּסִפְרֵי…”.
דיווח: כתבו ידיעה קצרה שכותרתה “לשון טובה ורעה”.
