לידתו ביום שנהרג ר' עקיבא — החלפת תינוקות

דף עבודה בעקבות רבי יהודה הנשיא (רבי)
שם: כיתה:

המעשה — הסיפור המלא

להלן מערך שיעור וסיפור חינוכי, מותח ומרגש עד דמעות, על הולדתו של רבי יהודה הנשיא (רבינו הקדוש) ומסירות הנפש של הוריו . הסיפור נכתב במיוחד לבנים בכיתות ב'-ד' בחינוך הדתי-לאומי, על פי כללי הסטוריטלינג, תוך שימוש בתיאורים חושיים, כבוד התורה, והמחשת השגחת ה' הנסתרת והפלאית.

פְּתִיחָה: שֶׁמֶשׁ שׁוֹקַעַת, שֶׁמֶשׁ זוֹרַחַת

יְלָדִים, זֶה הָיָה אֶחָד הַיָּמִים הַעֲצוּבִים בְּיוֹתֵר בְּתוֹלְדוֹת עַם יִשְׂרָאֵל. שֶׁקֶט מַקְפִּיא שָׂרַר בָּרְחוֹבוֹת. אֲנָשִׁים סָגְרוּ אֶת חַלּוֹנוֹת בָּתֵּיהֶם וּבָכוּ בַּסֵּתֶר. בְּאוֹתוֹ יוֹם נִפְטַר עַל קִדּוּשׁ ה', תַּחַת יְדֵיהֶם הָאַכְזָרִיּוֹת שֶׁל הָרוֹמָאִים, רַבָּם שֶׁל יִשְׂרָאֵל – רַבִּי עֲקִיבָא.

הַחֹשֶׁךְ נִרְאָה כָּבֵד מִתָּמִיד. אֲנָשִׁים לָחֲשׁוּ בִּדְמָעוֹת: "מָה יִהְיֶה עַכְשָׁו? מִי יַלְמֵד תּוֹרָה? הַשֶּׁמֶשׁ שֶׁל עַם יִשְׂרָאֵל שָׁקְעָה!".

אֲבָל יְלָדִים, הַקָּדוֹשׁ בָּרוּךְ הוּא אַף פַּעַם לֹא מַשְׁאִיר אֶת יִשְׂרָאֵל בַּחֹשֶׁךְ.

בְּדִיּוּק בְּאוֹתוֹ יוֹם, רַבָּן שִׁמְעוֹן בֶּן גַּמְלִיאֵל, שֶׁהָיָה הַנָּשִׂיא שֶׁל עַם יִשְׂרָאֵל, עָמַד בְּבֵיתוֹ וְדִמְעַת שִׂמְחָה בָּרְקָה בְּעֵינָיו. קוֹל בֶּכִי חַלָּשׁ שֶׁל תִּינוֹק קְטַנְטַן נִשְׁמַע מִן הַחֶדֶר הַסָּמוּךְ. אִשְׁתּוֹ הַצַּדֶּקֶת יָלְדָה בֵּן! בְּרֶגַע שֶׁנָּפַל עַמּוּד הָאוֹר שֶׁל רַבִּי עֲקִיבָא, הִדְלִיק ה' אֶת עַמּוּד הָאוֹר שֶׁל הַתִּינוֹק הַזֶּה – הַתִּינוֹק שֶׁיִּקָּרֵא לְיָמִים "רַבִּי יְהוּדָה הַנָּשִׂיא".

הַגְּזֵרָה הַנּוֹרָאָה וּמְסִירוּת הַנֶּפֶשׁ

אֲבָל הַשִּׂמְחָה שֶׁל רַבָּן שִׁמְעוֹן וְאִשְׁתּוֹ הָיְתָה מְהוּלָה בְּפַחַד מַחְרִיד.

הַקֵּיסָר הָרוֹמָאִי הוֹצִיא גְּזֵרָה אַכְזָרִית לְכָל רַחֲבֵי הָאָרֶץ: אוֹתוֹת אַזְהָרָה עִם חְתָמוֹת מֶלֶךְ תָּלוּ בְּכָל פִּנָּה, וְחַיָּלִים רוֹמָאִים עִם שִׁרְיוֹנוֹת בַּרְזֶל מַבְרִיקִים סִיְּרוּ בָּרְחוֹבוֹת, כְּשֶׁמַּגָּפֵיהֶם הַכְּבֵדִים רוֹקְעִים בְּרַעַשׁ: טְרָאח... טְרָאח...

הַגְּזֵרָה הָיְתָה: אָסוּר לְשׁוּם יְהוּדִי לַעֲשׂוֹת בְּרִית מִילָה לִבְנוֹ!

הִגִּיעַ הַיּוֹם הַשְּׁמִינִי. הַתִּינוֹק רַךְ וּמָתוֹק. מָה יַעֲשׂוּ כָּעֵת? אִם יַעֲשׂוּ בְּרִית – הַחַיָּלִים יַהַרְגוּ אוֹתָם!

אֲבָל רַבָּן שִׁמְעוֹן הִסְתַּכֵּל עַל אִשְׁתּוֹ הַצַּדֶּקֶת, וְהִיא הִנְהֲנָה בְּרֹאשָׁהּ בֶּאֱמוּנָה גְּדוֹלָה. "אֲנַחְנוּ לֹא נוֹתְרִים עַל בְּרִיתוֹ שֶׁל אַבְרָהָם אָבִינוּ," לָחַשׁ רַבָּן שִׁמְעוֹן. "לֹא נְבַטֵּל אֶת צִוּוּי הַקָּדוֹשׁ בָּרוּךְ הוּא מִפְּנֵי הָרְשָׁעִים!".

בַּסֵּתֶר, בְּחֶדֶר פְּנִימִי וְחָשׁוּךְ, כְּשֶׁהַתְּרִיסִים סְגוּרִים הֵיטֵב, מָל רַבָּן שִׁמְעוֹן אֶת בְּנוֹ הַקָּטָן וְקָרָא אֶת שְׁמוֹ בְּיִשְׂרָאֵל: יְהוּדָה . הֵם קִוּוּ שֶׁאִישׁ לֹא שָׁמַע. אֲבָל אָז... הִדְהֲדוּ דְּפִיקוֹת אֵימָה עַל הַדֶּלֶת!

בּוּם! בּוּם! קַתּוֹת הַחֲרָבוֹת הִכּוּ בָּעֵץ. מִישֶׁהוּ הִלְשִׁין לַשִּׁלְטוֹנוֹת.

הַשַּׂר הָרוֹמָאִי שֶׁל הָעִיר צִוָּה מִיָּד: "רַבָּן שִׁמְעוֹן! עָבַרְתָּ עַל דִּבְרֵי הַמֶּלֶךְ! כֵּיוָן שֶׁאַתָּה נָשִׂיא חָשׁוּב, אֲנִי בְּעַצְמִי פּוֹחֵד לַהֲרֹג אוֹתְךָ. אִשְׁתְּךָ תִּקַּח אֶת הַתִּינוֹק, וְחַיָּלַי יִקְחוּ אוֹתָהּ עַד לְרוֹמָא – אֶל הַקֵּיסָר בִּכְבוֹדוֹ וּבְעַצְמוֹ! שֶׁהוּא יַחְלִיט אֵיךְ לְהַעֲנִישׁ אֶתְכֶם!".

בְּמַלְוֹן הַדְּרָכִים – הַהַשְׁגָּחָה הַפְּרָטִית מִתְגַּלָּה

אִמּוֹ שֶׁל יְהוּדָה לָקְחָה אֶת הַתִּינוֹק אֶל לִבָּהּ. הַמַּסָּע לְרוֹמָא הָיָה קָשֶׁה וּמְעַיֵּף. כִּרְכְּרוֹת הָעֵץ הַכְּבֵדוֹת שִׁקְשְׁקוּ עַל אַבְנֵי הַדֶּרֶךְ. בַּלַּיְלָה הָיָה קַר, וְהָאֵם בָּכְתָה וְהִתְפַּלְּלָה מֵעוֹמֶק לִבָּהּ, כְּשֶׁדִּמְעוֹתֶיהָ נוֹשְׁרוֹת עַל פָּנָיו הַזַּכּוֹת שֶׁל הַתִּינוֹק יְהוּדָה: "רִבּוֹנוֹ שֶׁל עוֹלָם! אַתָּה יוֹדֵעַ שֶׁעָשִׂינוּ זֹאת לְמַעַנְךָ. תִּשְׁמֹר עַל הַיֶּלֶד הַזֶּה!".

לִקְרַאת עֶרֶב, עָצְרוּ הַחַיָּלִים בְּאַכְסַנְיָה (מַלְוֹן דְּרָכִים גָּדוֹל) כְּדֵי לָנוּחַ. לְשָׁם, בְּדִיּוּק בְּאוֹתוֹ הַזְּמַן, הִגִּיעָה גַּם אִשָּׁה חֲשׁוּבָה מְאֹד – רוֹמָאִית בְּכִירָה עֲטוּיָה בִּבְגָדִים שֶׁל מֶשִׁי וְזָהָב, אֵשֶׁת שַׂר גָּדוֹל (אֵם מִשְׁפַּחְתּוֹ שֶׁל אַנְטוֹנִינוּס). רֵיחַ שֶׁל בֹּשֶׂם יָקָר נָדַף מִמֶּנָּה.

הַשְּׁתַּיִם הִזְדַּמְּנוּ לְאוֹתוֹ הַחֶדֶר.

הָרוֹמָאִית הִבִּיטָה בָּאֵם הַיְּהוּדִיָּה שֶׁעֵינֶיהָ הָיוּ אֲדֻמּוֹת מִבֶּכִי, וְשָׁאֲלָה בְּרַכּוּת: "מַדּוּעַ אַתְּ בּוֹכָה, אִשָּׁה אֻמְלָלָה? לְאָן הַחַיָּלִים לוֹקְחִים אוֹתָךְ?".

"יָלַדְתִּי בֵּן מָתוֹק," לָחֲשָׁה אֵשֶׁת הַנָּשִׂיא, "וְקִיַּמְנוּ אֶת צִוּוּי אֱלֹהֵינוּ – לַעֲשׂוֹת לוֹ בְּרִית מִילָה. עַכְשָׁו תָּפְסוּ אוֹתָנוּ, וּמוֹבִילִים אוֹתִי לַקֵּיסָר. הַקֵּיסָר יִרְאֶה שֶׁהוּא נִימוֹל... וַאֲנִי יוֹדַעַת שֶׁהוּא יַהֲרֹג אֶת שְׁנֵינוּ!". אָמְרָה – וּפָרְצָה בְּבֶכִי קוֹרֵעַ לֵב.

רַחֲמֶיהָ שֶׁל הָאֵם הָרוֹמָאִית נִכְמְרוּ (הִתְעוֹרְרוּ מְאֹד). סוֹד גָּדוֹל נָטַע הַקָּדוֹשׁ בָּרוּךְ הוּא בְּלִבָּהּ.

הִיא נִגְּשָׁה קָרוֹב לָאֵם הַיְּהוּדִיָּה וְלָחֲשָׁה לָהּ: "שִׁמְעִי לִי. הַהַשְׁגָּחָה הִפְגִּישָׁה בֵּינֵינוּ. גַּם אֲנִי יָלַדְתִּי עַכְשָׁו תִּינוֹק (הַתִּינוֹק אַנְטוֹנִינוּס). הוּא מֻנָּח כָּאן. שֶׁלִּי עָרֵל (לֹא נִימוֹל), וְשֶׁלָּךְ נִימוֹל. בּוֹאִי נַחְלִיף בֵּין הַתִּינוֹקוֹת! אַתְּ תִּקְּחִי אֶת הַתִּינוֹק שֶׁלִּי שֶׁעוֹד לֹא עָשׂוּ לוֹ בְּרִית, וְתַרְאִי אוֹתוֹ לַקֵּיסָר. הוּא יַחְשֹׁב שֶׁהַחַיָּלִים הִמְצִיאוּ, וְיִפְטֹר אוֹתָךְ לְשָׁלוֹם. בֵּינְתַיִם, אֲנִי אֶשְׁמוֹר כָּאן עַל הַתִּינוֹק הַקָּדוֹשׁ שֶׁלָּךְ וַאֲנִיק אוֹתוֹ, עַד שֶׁתַּחְזְרִי בְּשָׁלוֹם!".

אֵשֶׁת רַבָּן שִׁמְעוֹן רָאֲתָה אֵיךְ ה' שׁוֹלֵחַ לָהּ מַלְאָךְ בִּדְמוּת רוֹמָאִית. בַּחֲשַׁאי, בְּאִישׁוֹן לַיְלָה, כְּשֶׁהַשּׁוֹמְרִים יְשֵׁנִים – הֻחְלְפוּ הַתִּינוֹקוֹת!

הַהֶלֶם בָּאַרְמוֹן וְהַבְטָחַת הֶעָתִיד

לְמָחֳרָת, הִגִּיעָה הַשַּׁיָּרָה אֶל אַרְמוֹנוֹ שֶׁל הַקֵּיסָר. זֶה הָיָה אַרְמוֹן מְפֹאָר מְאֹד. עַמּוּדֵי שַׁיִשׁ גְּבוֹהִים וְרִצְפַּת מוֹזָאִיקָה נוֹצֶצֶת שֶׁהֵאִירָה אֶת הַחֶדֶר. עַל כִּסֵּא מִזָּהָב טָהוֹר יָשַׁב הַקֵּיסָר הָאַכְזָר, וּמַבָּט כּוֹעֵס בְּעֵינָיו.

"הַאִם זוֹ הַיְּהוּדִיָּה שֶׁהֵעֵזָּה לְהָפֵר אֶת פְּקֻדָּתִי?!" ְשָׁאַג הַקֵּיסָר בְּקוֹל שֶׁהִרְעִיד אֶת קִירוֹת הָאַרְמוֹן.

"כֵּן, אֲדוֹנִי הַמֶּלֶךְ," עָנוּ הַשָּׂרִים, "הִיא עָשְׂתָה לִבְנָהּ בְּרִית מִילָה!".

הַקֵּיסָר סִמֵּן בְּאֶצְבָּעוֹ. "הָבִיאוּ לִי אֶת הַתִּינוֹק! נִרְאֶה בְּעֵינֵינוּ!".

הָאֵם הַיְּהוּדִיָּה קִדְּמָה עַד לְרַגְלֵי הַכִּסֵּא, יָדֶיהָ רוֹעֲדוֹת אַךְ לִבָּהּ בּוֹטֵחַ בַּה'. הִיא פָּתְחָה לְאַט לְאַט אֶת שְׂמִיכַת הַצֶּמֶר. הַקֵּיסָר הִתְכּוֹפֵף, הִבִּיט... וְאָז נִפְעֲרוּ עֵינָיו בְּתַדְהֵמָה!

הַתִּינוֹק... לֹא הָיָה נִימוֹל!

הַקֵּיסָר הֵרִים אֶת מַבָּטוֹ וְנָעַץ עֵינַיִם זְעוּמוֹת מֵאֵשׁ בַּשַּׂר הָרוֹמָאִי שֶׁלִּוָּה אוֹתָם. "שַׁקְרָנִים!" זָעַק. "הֲכִי הֵבֵאתֶם אֶת הָאִשָּׁה הַמִּסְכֵּנָה הַזּוֹ סְתָם?! הֲרֵי הַתִּינוֹק עָרֵל לְחַלּוּטִין! אֵין כָּאן שׁוּם בְּרִית!".

בְּכַעֲסוֹ הַגָּדוֹל, צִוָּה הַקֵּיסָר לְהַעֲנִישׁ קָשׁוֹת אֶת הַשַּׂר הָרוֹמָאִי שֶׁל יְרוּשָׁלַיִם שֶׁהִלְשִׁין עֲלֵיהֶם. לָאֵם אָמַר לָשׁוּב לְבֵיתָהּ בְּשָׁלוֹם חָפְשִׁיָּה לְגַמְרֵי – וּבְאוֹתָהּ הִזְדַּמְּנוּת אֲפִלּוּ בִּטֵּל אֶת הַגְּזֵרָה! מֵהַיּוֹם מֻתָּר לַיְּהוּדִים לָמוּל אֶת בְּנֵיהֶם. אֵיזֶה נֵס עָצוּם!

בְּדַרְכָּהּ חֲזָרָה, עָצְרָה הָאֵם בְּאוֹתוֹ מַלְוֹן. בִּדְמָעוֹת שֶׁל תּוֹדָה הִיא מָסְרָה אֶת אַנְטוֹנִינוּס חֲזָרָה לְאִמּוֹ הָרוֹמָאִית, וְחִבְּקָה אֶת בְּנָהּ יְהוּדָה לְחֵיקָהּ.

הָרוֹמָאִית הַחֲכָמָה לָחֲצָה אֶת יָדָהּ שֶׁל הָאֵם הַיְּהוּדִיָּה וְאָמְרָה: "רְאִי אֵיזֶה פֶלֶא קָרָה כָּאן. הַתִּינוֹק שֶׁלִּי יָנַק מֵחֲלָבֵךְ הַיְּהוּדִי, וְהַתִּינוֹק שֶׁלָּךְ הִצִּיל אוֹתָנוּ. הַקָּדוֹשׁ בָּרוּךְ הוּא עוֹשֶׂה נִסִּים. מֵהַיּוֹם – שְׁנֵי הַתִּינוֹקוֹת הָאֵלֶּה יִהְיוּ חֲבֵרִים בְּלֵב וָנֶפֶשׁ, הֵם וּבְנֵיהֶם אַחֲרֵיהֶם!".

וְאָכֵן יְלָדִים, שָׁנִים עָבְרוּ. הַתִּינוֹק רַבִּי יְהוּדָה גָּדַל לִהְיוֹת "רַבִּי יְהוּדָה הַנָּשִׂיא", מְאַסֵּף וְכוֹתֵב הַמִּשְׁנָה הַקְּדוֹשָׁה שֶׁאָנוּ לוֹמְדִים עַד הַיּוֹם. וְאוֹתוֹ תִּינוֹק רוֹמָאִי – אַנְטוֹנִינוּס, עָלָה לַגְּדֻלָּה וְהָפַךְ לְקֵיסָר רוֹמָא אֲבָל... קֵיסָר אֹהֵב יִשְׂרָאֵל! הוּא לָמַד תּוֹרָה בַּסֵּתֶר מֵרַבִּי יְהוּדָה, כִּבֵּד אוֹתוֹ יוֹתֵר מִכָּל אָדָם, וְהָפַךְ לַחֲבֵרוֹ הַטּוֹב בְּיוֹתֵר שֶׁל מַנְהִיג יִשְׂרָאֵל.

א. שְׁאֵלַת פְּתִיחָה (לְעוֹרֵר סַקְרָנוּת):

"יְלָדִים, אִם אֲנַחְנוּ נִמְצָאִים בְּחֶדֶר מָלֵא בַּחֹשֶׁךְ, וְיֵשׁ לָנוּ רַק נֵר אֶחָד קָטָן שֶׁמֵּאִיר... וּפִתְאוֹם רוּחַ חֲזָקָה מְכַבָּה אוֹתוֹ. מָה נַעֲשֶׂה? (לְתַת לַיְּלָדִים לַעֲנוֹת: נַדְלִיק מֵחָדָשׁ / נָבִיא פָּנָס כּוּ'). נָכוֹן! הַיּוֹם נִשְׁמַע עַל יוֹם שֶׁבּוֹ נֵר גָּדוֹל מְאֹד בְּעַם יִשְׂרָאֵל נִכְבָּה, אַךְ הַקָּדוֹשׁ בָּרוּךְ הוּא, בְּדִיּוּק בְּאוֹתוֹ רֶגַע, הִדְלִיק עֲבוּרֵנוּ פָּנָס חָדָשׁ שֶׁיֵּאִָיר אֶת הָעוֹלָם כֻּלּוֹ".

ב. שְׁאֵלַת סִיּוּם:

"לָמָּה לְדַעְתְּכֶם ה' עָשָׂה שֶׁדַּוְקָא תִּינוֹק גּוֹי רוֹמָאִי, שֶׁשַּׁיָּךְ לָאוֹיְבִים שֶׁלָּנוּ, יִהְיֶה זֶה שֶׁיַּצִּיל אֶת הַנָּשִׂיא שֶׁל יִשְׂרָאֵל?

(תְּשׁוּבָה: כְּדֵי לְהַרְאוֹת שֶׁה' שׁוֹלֵט בְּהַכֹּל, וַאֲפִלּוּ מִתּוֹךְ הָאוֹיְבִים עַצְמָם, הוּא יָכוֹל לְהַצְמִיחַ תּוֹרָה וִישׁוּעָה!)."

ג. הֶעָרוֹת לְבִצּוּעַ וְסְטוֹרִיטֵלִינְג לַמּוֹרֶה:

ה. פְּעִילוּת מְסַכֶּמֶת מֻמְלֶצֶת:

"פָּנָס שֶׁל מְסִירוּת"

הַמּוֹרֶה יְחַלֵּק לְכָל יֶלֶד צִיּוּר שֶׁל פָּנָס רֵיק מִנְּיָר.

הוֹרָאָה לַתַּלְמִידִים: "בְּתוֹךְ הַפָּנָס שֶׁלָּכֶם, צַיְּרוּ אוֹ כִּתְבוּ עַל אֵיזֶה 'מִצְוָה' אוֹ 'הִתְאַמְּצוּת' אַתֶּם מְבַקְשִׁים שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא יַעֲזֹר לָכֶם, בְּדִיּוּק כְּפִי שֶׁעָזַר לְהוֹרָיו שֶׁל רַבִּי יְהוּדָה. אֵיזֶה חֹשֶׁךְ אַתֶּם רוֹצִים לְהָאִיר?".

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “להלן מערך שיעור וסיפור חינוכי, מותח ומרגש עד דמעות, על הולדתו של רבי יהודה הנשיא (רבינו הקדוש) ומסירות הנפש של הוריו .” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “אֲבָל הַשִּׂמְחָה שֶׁל רַבָּן שִׁמְעוֹן וְאִשְׁתּוֹ הָיְתָה מְהוּלָה בְּפַחַד מַחְרִיד.”?
ג. כיצד הגיב רבי יהודה הנשיא (רבי) למצב המתואר: “אִמּוֹ שֶׁל יְהוּדָה לָקְחָה אֶת הַתִּינוֹק אֶל לִבָּהּ.”?
ד. מה השתנה בין “אֵשֶׁת רַבָּן שִׁמְעוֹן רָאֲתָה אֵיךְ ה' שׁוֹלֵחַ לָהּ מַלְאָךְ בִּדְמוּת רוֹמָאִית.” לבין “לָאֵם אָמַר לָשׁוּב לְבֵיתָהּ בְּשָׁלוֹם חָפְשִׁיָּה לְגַמְרֵי – וּבְאוֹתָהּ הִזְדַּמְּנוּת אֲפִלּוּ בִּטֵּל אֶת…”?
ה. כיצד הגיע הסיפור אל האירוע: “(לְתַת לַיְּלָדִים לַעֲנוֹת: נַדְלִיק מֵחָדָשׁ / נָבִיא פָּנָס כּוּ').”?

לפני ואחרי

מה קרה לפני: “אַתְּ תִּקְּחִי אֶת הַתִּינוֹק שֶׁלִּי שֶׁעוֹד לֹא עָשׂוּ לוֹ בְּרִית, וְתַרְאִי אוֹתוֹ לַקֵּיסָר.”?
מה קרה אחרי: “בִּדְמָעוֹת שֶׁל תּוֹדָה הִיא מָסְרָה אֶת אַנְטוֹנִינוּס חֲזָרָה לְאִמּוֹ הָרוֹמָאִית, וְחִבְּקָה אֶת בְּנָהּ יְהוּדָה לְחֵיקָהּ.”?

סיבה ותוצאה מתוך המעשה

האירוע הנתוןמה גרם לו?מה הייתה תוצאתו?
בַּסֵּתֶר, בְּחֶדֶר פְּנִימִי וְחָשׁוּךְ, כְּשֶׁהַתְּרִיסִים סְגוּרִים הֵיטֵב, מָל רַבָּן שִׁמְעוֹן אֶת בְּנוֹ הַקָּטָן וְקָרָא אֶת שְׁמוֹ בְּיִשְׂרָאֵל:…
כֵּן, אֲדוֹנִי הַמֶּלֶךְ," עָנוּ הַשָּׂרִים, "הִיא עָשְׂתָה לִבְנָהּ בְּרִית מִילָה!".

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “לידתו ביום שנהרג ר' עקיבא — החלפת תינוקות” על רבי יהודה הנשיא (רבי)? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “יָלַדְתִּי בֵּן מָתוֹק," לָחֲשָׁה אֵשֶׁת הַנָּשִׂיא, "וְקִיַּמְנוּ אֶת צִוּוּי אֱלֹהֵינוּ – לַעֲשׂוֹת לוֹ בְּרִית מִילָה.”.
ג. אילו רבי יהודה הנשיא (רבי) לא היה פועל כפי שמתואר ב־“לָאֵם אָמַר לָשׁוּב לְבֵיתָהּ בְּשָׁלוֹם חָפְשִׁיָּה לְגַמְרֵי – וּבְאוֹתָהּ הִזְדַּמְּנוּת אֲפִלּוּ בִּטֵּל אֶת הַגְּזֵרָה!” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “(לְתַת לַיְּלָדִים לַעֲנוֹת: נַדְלִיק מֵחָדָשׁ / נָבִיא פָּנָס כּוּ').” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. להלן מערך שיעור וסיפור חינוכי, מותח ומרגש עד דמעות, על הולדתו של רבי…
2. הַגְּזֵרָה הָיְתָה: אָסוּר לְשׁוּם יְהוּדִי לַעֲשׂוֹת בְּרִית מִילָה…
3. בַּלַּיְלָה הָיָה קַר, וְהָאֵם בָּכְתָה וְהִתְפַּלְּלָה מֵעוֹמֶק…
4. עַל כִּסֵּא מִזָּהָב טָהוֹר יָשַׁב הַקֵּיסָר הָאַכְזָר, וּמַבָּט…
5. מֵהַיּוֹם – שְׁנֵי הַתִּינוֹקוֹת הָאֵלֶּה יִהְיוּ חֲבֵרִים בְּלֵב…
6. הוֹרָאָה לַתַּלְמִידִים: "בְּתוֹךְ הַפָּנָס שֶׁלָּכֶם, צַיְּרוּ אוֹ…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “אִמּוֹ שֶׁל יְהוּדָה לָקְחָה אֶת הַתִּינוֹק אֶל לִבָּהּ.”.
מכתב: כתבו לרבי יהודה הנשיא (רבי) בעקבות “כֵּן, אֲדוֹנִי הַמֶּלֶךְ," עָנוּ הַשָּׂרִים, "הִיא עָשְׂתָה לִבְנָהּ בְּרִית מִילָה!".”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “הַתִּינוֹק רַבִּי יְהוּדָה גָּדַל לִהְיוֹת "רַבִּי יְהוּדָה הַנָּשִׂיא", מְאַסֵּף וְכוֹתֵב הַמִּשְׁנָה הַקְּדוֹשָׁה שֶׁאָנוּ…”.
דיווח: כתבו ידיעה קצרה שכותרתה “לידתו ביום שנהרג ר' עקיבא — החלפת תינוקות”.

המחזה

רגע מן הסיפורהדמויותמה יאמרו או יעשו?
בַּסֵּתֶר, בְּחֶדֶר פְּנִימִי וְחָשׁוּךְ, כְּשֶׁהַתְּרִיסִים סְגוּרִים הֵיטֵב, מָל רַבָּן שִׁמְעוֹן…
לָאֵם אָמַר לָשׁוּב לְבֵיתָהּ בְּשָׁלוֹם חָפְשִׁיָּה לְגַמְרֵי – וּבְאוֹתָהּ הִזְדַּמְּנוּת…

7. חיבור לחיים מתוך המעשה

א. מה אפשר ללמוד מן הדרך שבה רבי יהודה הנשיא (רבי) פעל באירוע “אֵשֶׁת רַבָּן שִׁמְעוֹן רָאֲתָה אֵיךְ ה' שׁוֹלֵחַ לָהּ מַלְאָךְ בִּדְמוּת רוֹמָאִית.”?
ב. תארו מצב מחיי הכיתה שבו אפשר ליישם את המסר של “לידתו ביום שנהרג ר' עקיבא — החלפת תינוקות”.
