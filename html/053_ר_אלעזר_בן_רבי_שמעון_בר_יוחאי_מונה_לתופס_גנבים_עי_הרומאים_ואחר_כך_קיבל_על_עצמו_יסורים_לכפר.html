מונה לתופס גנבים ע"י הרומאים, ואחר כך קיבל על עצמו יסורים לכפר

דף עבודה בעקבות ר' אלעזר בן רבי שמעון (בר יוחאי)
שם: כיתה:

המעשה — הסיפור המלא

פְּתִיחָה: כּוֹחוֹ שֶׁל בֶּן הַמְּעָרָה

יְלָדִים, אַתֶּם וַדַּאי זוֹכְרִים אֶת רַבִּי שִׁמְעוֹן בַּר יוֹחַאי וּבְנוֹ רַבִּי אֶלְעָזָר, שֶׁיָּשְׁבוּ שְׁתֵּים עֶשְׂרֵה שָׁנִים בַּמְּעָרָה, אָכְלוּ חֲרוּבִים, שָׁתוּ מַיִם מֵהַמַּעְיָן וְלָמְדוּ אֶת כָּל סוֹדוֹת הַתּוֹרָה.

אַחֲרֵי שֶׁיָּצְאוּ מֵהַמְּעָרָה, גָּדַל רַבִּי אֶלְעָזָר וְהָפַךְ לְתַלְמִיד חָכָם גָּדוֹל עַד מְאֹד. אֲבָל חוּץ מֵהַחָכְמָה הָעֲצוּמָה שֶׁלּוֹ, הַקָּדוֹשׁ בָּרוּךְ הוּא נָתַן לְרַבִּי אֶלְעָזָר מַתָּנָה נוֹסֶפֶת: כּוֹחַ גּוּפָנִי אַדִּיר. הוּא הָיָה חָזָק וְחָסוֹן מִכָּל הָאֲנָשִׁים בַּסְּבִיבָה!

בְּאוֹתָם יָמִים, הָרוֹמָאִים הָרְשָׁעִים שָׁלְטוּ בְּאֶרֶץ יִשְׂרָאֵל. הֵם לֹא בֶּאֱמֶת שָׁמְרוּ עַל הַסֵּדֶר, וּבַלֵּילוֹת חֲשׁוּכִים הָיוּ מִסְתּוֹבְבוֹת כְּנֻפְיּוֹת שֶׁל שׁוֹדְדִים וְגַנָּבִים, עַמֵּי הָאָרֶץ וַאֲנָשִׁים רָעִים, שֶׁהָיוּ גּוֹנְבִים רְכוּשׁ, מַפְחִידִים יְלָדִים, וּבוֹרְחִים לִפְנֵי שֶׁמִּישֶׁהוּ הִסְפִּיק לִרְאוֹת אוֹתָם.

הַצַּיָּד שֶׁל הַמֶּלֶךְ

לַיְלָה אֶחָד, הָלַךְ רַבִּי אֶלְעָזָר בַּסִּמְטָאוֹת וְשָׁמַע קוֹל פְּסִיעוֹת שֶׁל גַּנָּבִים. רַבִּי אֶלְעָזָר לֹא הִסֵּס לְרֶגַע. בִּקְפִיצָה אַחַת שֶׁל גְּבוּרָה, הוּא תָּפַס אֶת הַשּׁוֹדְדִים, תָּפַס אוֹתָם בְּיָדָיו הַחֲזָקוֹת וְלֹא נָתַן לָהֶם לִבְרֹחַ!

קְצִין מִשְׁטָרָה רוֹמָאִי שֶׁעָבַר שָׁם, רָאָה מֵרָחוֹק אֶת הָאִישׁ שֶׁתּוֹפֵס אֶת כָּל הַגַּנָּבִים לְבַדּוֹ, בְּלִי חֶרֶב וּבְלִי מָגֵן. הַקָּצִין נִדְהַם. "מִי זֶה הָאִישׁ הַחָזָק הַזֶּה?!" שָׁאַל.

הַשְּׁמוּעָה הִגִּיעָה מַהֵר מְאֹד לַשִּׁלְטוֹנוֹת הָרוֹמָאִיִּים. הֵם קָרְאוּ לְרַבִּי אֶלְעָזָר וְאָמְרוּ לוֹ: "מֵהַיּוֹם, אַתָּה תִּהְיֶה מְמוּנֶה מִטַּעַם הַמֶּלֶךְ עַל תְּפִיסַת הַגַּנָּבִים! אֲנַחְנוּ דּוֹרְשִׁים שֶׁאַתָּה תַּסְגִּיר אוֹתָם אֵלֵינוּ!".

רַבִּי אֶלְעָזָר הִסְכִּים. הוּא חָשַׁב בְּלִבּוֹ: "הֲרֵי זוֹ מִצְוָה! אֲנִי מְבַעֵר אֶת הָרְשָׁעִים וְאֶת הַגַּנָּבִים מֵעַם יִשְׂרָאֵל, כְּדֵי שֶׁלַּיְּהוּדִים הַטּוֹבִים יִהְיֶה שֶׁקֶט וְשָׁלוֹם".

וְאָכֵן, רַבִּי אֶלְעָזָר הָיָה מְסַיֵּר וְתוֹפֵס גַּנָּבִים וּמוֹסֵר אוֹתָם. כֻּלָּם פָּחֲדוּ מִפָּנָיו!

הַשְּׁאֵלָה שֶׁחָדְרָה כְּחֵץ לַלֵּב

אַךְ לֹא כָּל חַכְמֵי יִשְׂרָאֵל אָהֲבוּ אֶת הַתַּפְקִיד הֶחָדָשׁ שֶׁל רַבִּי אֶלְעָזָר.

יוֹם אֶחָד, הָלַךְ רַבִּי אֶלְעָזָר בָּרְחוֹב, וְלִקְרָאתוֹ יָצָא חָכָם זָקֵן וְקָדוֹשׁ – רַבִּי יְהוֹשֻׁעַ בֶּן קָרְחָה. עֵינָיו שֶׁל רַבִּי יְהוֹשֻׁעַ הָיוּ מְלֵאוֹת עֶצֶב וְתוֹכֵחָה.

הוּא עָמַד מוּל רַבִּי אֶלְעָזָר, וְצָעַק בְּקוֹל שֶׁרָעַד מִכְּאֵב: "חֹמֶץ בֶּן יַיִן! חֹמֶץ בֶּן יַיִן! עַד מָתַי אַתָּה מוֹסֵר אֶת עַמּוֹ שֶׁל אֱלֹקֵינוּ לַהֲרִיגָה?!"

רַבִּי אֶלְעָזָר נִדְהַם. "אֲבָל רַבִּי," הוּא עָנָה, "אֲנִי מְנַקֶּה אֶת הַקּוֹצִים מֵהַכֶּרֶם! אֲנִי מְנַקֶּה אֶת הָרְשָׁעִים כְּדֵי שֶׁעַם יִשְׂרָאֵל יִהְיֶה טָהוֹר כְּמוֹ כֶּרֶם יָפֶה!".

אַךְ רַבִּי יְהוֹשֻׁעַ הִסְתַּכֵּל יָשָׁר אֶל תּוֹךְ עֵינָיו וְאָמַר לוֹ: "יָבֹא בַּעַל הַכֶּרֶם – וִינַקֶּשׁ אֶת קוֹצָיו" .

הַמַּשְׁמָעוּת הָיְתָה: הַקָּדוֹשׁ בָּרוּךְ הוּא (בַּעַל הַכֶּרֶם) יוֹדֵעַ לְהַעֲנִישׁ אֶת הַגַּנָּבִים (הַקּוֹצִים). אַתָּה, רַבִּי אֶלְעָזָר, תַּלְמִיד חָכָם! אַל תַּעֲשֶׂה תַּפְקִידִים בִּשְׁבִיל הַשִּׁלְטוֹן הָרוֹמָאִי הָאַכְזָרִי שֶׁהוֹרֵג אֲנָשִׁים!

דְּמָעוֹת שֶׁל חֲרָטָה עֲמֻקָּה

יְלָדִים, מָה אָדָם רָגִיל הָיָה עוֹשֶׂה אִם הָיוּ אֹמְרִים לוֹ מִשְׁפָּט כָּזֶה? הָיָה נֶעֱלָב! הָיָה כּוֹעֵס! הָיָה אוֹמֵר "מִי אַתָּה שֶׁתַּגִּיד לִי מָה לַעֲשׂוֹת?! הֲרֵי גַּם אֲנִי רַב עָצוּם!".

אֲבָל רַבִּי אֶלְעָזָר הָיָה צַדִּיק אֲמִתִּי. מִי שֶׁצַּדִּיק – אִכְפַּת לוֹ מֵהָאֱמֶת.

הַמִּלִּים שֶׁל רַבִּי יְהוֹשֻׁעַ נִכְנְסוּ אֶל תּוֹךְ לִבּוֹ שֶׁל רַבִּי אֶלְעָזָר. הוּא הֵבִין שֶׁטָּעָה. הוּא הֵבִין שֶׁאָסוּר הָיָה לוֹ לִמְסֹר יְהוּדִים (אֲפִלּוּ אִם הֵם חָטְאוּ וְגָנְבוּ) לַיָּדַיִם שֶׁל שִׁלְטוֹן גּוֹי וְאַכְזָר.

הוּא עָזַב אֶת מִשְׂרָתוֹ מִיָּד! הוּא פָּרַשׁ מֵהַמִּשְׁטָרָה הָרוֹמָאִית.

אַךְ זֶה לֹא הִסְפִּיק לוֹ. בַּלַּיְלָה, כְּשֶׁשָׁכַב עַל מִטָּתוֹ, דְּמָעוֹת כְּבֵדוֹת זָלְגוּ עַל פָּנָיו הַקְּדוֹשׁוֹת וְהִרְטִיבוּ אֶת הַכַּר עָלָיו יָשֵׁן. אָמְנָם הוּא עָזַב, אֲבָל מָה עִם אֵלּוּ שֶׁכְּבָר נָתַן לָרוֹמָאִים? מָה עִם הַטָּעוּת שֶׁכְּבָר עָשָׂה?

"רִבּוֹנוֹ שֶׁל עוֹלָם", לָחַשׁ רַבִּי אֶלְעָזָר בַּחֲשֵׁכָה, "אֲנִי רוֹצֶה לְתַקֵּן אֶת מָה שֶׁעָשִׂיתִי. אֲנִי מְבַקֵּשׁ שֶׁיָּבוֹאוּ עָלַי יִסּוּרִים, שֶׁיִּכְאַב לִי הַגּוּף עַד מְאֹד – וּבִזְכוּת הַכְּאֵב הַזֶּה תִּתְנַקֶּה וְתִטָּהֵר הַנְּשָׁמָה שֶׁלִּי מֵהַטָּעוּת שֶׁעָשִׂיתִי!".

רַבִּי אֶלְעָזָר הִזְמִין עַל עַצְמוֹ יִסּוּרִים וְכְאֵבִים נוראיים, רַק כְּדֵי לִהְיוֹת נָקִי לְגַמְרֵי מֵחֵטְא מוּל בּוֹרֵא עוֹלָם! וְהוּא הָיָה מְדַבֵּר לַיִּסּוּרִים כְּאִלּוּ הָיוּ חֲבֵרִים שֶׁלּוֹ וְאוֹמֵר בְּלַחַשׁ: "בּוֹאוּ אַחַי וְרֵעַי, בּוֹאוּ לְנַקּוֹת אֶת נִשְׁמָתִי".

הַהֶסְכֵּם עִם הַיִּסּוּרִים

כָּל הַלַּיְלָה הָיָה הַצַּדִּיק הַגָּדוֹל סוֹבֵל מִכְּאֵבִים חֲזָקִים בְּגוּפוֹ. זֵעָה כִּסְּתָה אֶת מִצְחוֹ. אִשְׁתּוֹ שֶׁל רַבִּי אֶלְעָזָר יָשְׁבָה לְצִדּוֹ, בָּכְתָה וְטִפְּלָה בּוֹ הֵיטֵב, רָחֲצָה אֶת פָּנָיו בְּמַיִם קָרִים כְּדֵי לְהָקֵל עָלָיו.

אַךְ רְאוּ זֶה פֶּלֶא – מַה יִקְרֶה בַּבֹּקֶר? אֵיךְ יוּכַל אָדָם שֶׁכּוֹאֵב לוֹ כָּל כָּךְ בַּלַּיְלָה לָקוּם בַּבֹּקֶר לִלְמֹד תּוֹרָה כְּדַרְכָּם שֶׁל רַבָּנִים קְדוֹשִׁים?

לְרַבִּי אֶלְעָזָר הָיָה הֶסְכֵּם עִם הַקָּדוֹשׁ בָּרוּךְ הוּא!

כַּאֲשֶׁר עָלָה אוֹר הַבֹּקֶר, וְקַרְנֵי שֶׁמֶשׁ רִאשׁוֹנוֹת עָלְו מֵעַל הָרֵי הַגָּלִיל, הָיָה רַבִּי אֶלְעָזָר פּוֹקֵחַ אֶת עֵינָיו, מֵרִים אֶת קוֹלוֹ וְאוֹמֵר: "לְכוּ לָכֶם כְּאֵבִים וְיִסּוּרִים! שֶׁלֹּא תְּבַטְּלוּ אוֹתִי מִלִּמּוּד תּוֹרָה!"

וּבְאוֹתוֹ הָרֶגַע – בְּנֵס שֶׁאִי אֶפְשָׁר לְתָאֵר כְּלָל – הַכְּאֵבִים הָיוּ מִסְתַּלְּקִים וְנֶעֱלָמִים לְגַמְרֵי! רַבִּי אֶלְעָזָר הָיָה מִזְדַּקֵּף, קָם מִמִּטָּתוֹ גִּבּוֹר וּבָרִיא כְּמוֹ אֲרִי, לוֹבֵשׁ אֶת בְּגָדָיו, הוֹלֵךְ לְבֵית הַמִּדְרָשׁ – וְלוֹמֵד תּוֹרָה מִתּוֹךְ שִׂמְחָה וְאוֹר שֶׁאֵין כְּמוֹתוֹ! כָּל אוֹתוֹ הַיּוֹם, וְלֹא הִרְגִּישׁ שׁוּם כְּאֵב! תַּלְמִידָיו בִּכְלָל לֹא יָדְעוּ מָה עוֹבֵר עָלָיו בַּלַּיְלָה. מִתְקוּת הַתּוֹרָה מִלְּאָה אֶת לִבּוֹ הַטָּהוֹר.

וּבַלַּיְלָה? שׁוּב. "בּוֹאוּ יִסּוּרַי," כְּדֵי לְהִזְדַּכֵּךְ. וּבַבֹּקֶר "לְכוּ לָכֶם, שֶׁלֹּא אֶבָּטֵל מִתּוֹרָה."

וְכָךְ יָמִים רַבִּים, עַד שֶׁנִּזְדַּכְּכָה נִשְׁמָתוֹ לְגַמְרֵי וְהוּא הָפַךְ לְטָהוֹר וְקָדוֹשׁ מֵאֵין כָּמוֹהוּ, כְּמַלְאַךְ ה' צְבָאוֹת מַמָּשׁ. כָּךְ הֶרְאָה לָנוּ הַצַּדִּיק – לָקַחַת אַחֲרָיוּת עַל טָעוּת, וְלֹא לוְותֵּר עַל לִמּוּד הַתּוֹרָה אֲפִלּוּ מִתּוֹךְ הַקֹּשִׁי הַגָּדוֹל בְּיוֹתֵר.

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “יְלָדִים, אַתֶּם וַדַּאי זוֹכְרִים אֶת רַבִּי שִׁמְעוֹן בַּר יוֹחַאי וּבְנוֹ רַבִּי אֶלְעָזָר, שֶׁיָּשְׁבוּ שְׁתֵּים עֶשְׂרֵה שָׁנִים בַּמְּעָרָה, אָכְלוּ…” — מה קרה מיד לאחר מכן?
