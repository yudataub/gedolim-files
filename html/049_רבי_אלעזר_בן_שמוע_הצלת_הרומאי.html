סיפור 49 • רבי אלעזר בן שמוע • הצלת הרומאי

הצלת הרומאי

דף עבודה בעקבות רבי אלעזר בן שמוע
שם: כיתה:

המעשה — הסיפור המלא

פְּתִיחָה: סְעָרָה עַל שְׂפַת הַיָּם

יְלָדִים, אֲנִי רוֹצֶה שֶׁתַּעַצְמוּ רֶגַע אֶת הָעֵינַיִם. תַּקְשִׁיבוּ הֵיטֵב. שׁוֹמְעִים אֶת זֶה? שְׁשְּׁשְׁשְׁ... ווּווּווּ... רוּחַ חֲזָקָה וְקָרָה נוֹשֶׁבֶת, מְעִיפָה אֶת הַחוֹל לְכָל עֵבֶר. גַּלִּים עֲנָקִיִּים, גְּבוֹהִים כְּמוֹ הָרִים מִמַּיִם, מִתְנַפְּצִים בְּכֹחַ עַל הַסְּלָעִים: בּוּם! קְרָאשׁ! רֵיחַ חָזָק וּמָלוּחַ שֶׁל מֵי יָם מְמַלֵּא אֶת הָאֲוִיר, וְטִפּוֹת קָרוֹת נִתָּזוֹת עַל הַפָּנִים.

בְּיוֹם סוֹעֵר וְזֹעֵף כָּזֶה, לִפְנֵי שָׁנִים רַבּוֹת בְּאֶרֶץ יִשְׂרָאֵל, טִיֵּל הַתַּנָּא הַקָּדוֹשׁ רַבִּי אֶלְעָזָר בֶּן שַׁמּוּעַ עַל שְׂפַת הַיָּם בְּעִיר מוֹשָׁבוֹ. הוּא הָיָה עָטוּף בִּמְעִילוֹ הַחַם שֶׁלְּגּוּפוֹ, כְּשֶׁלְּפֶתַע שָׁמַע צְעָקוֹת. הֲמוֹן קוֹלוֹת, שִׁבְרֵי עֵצִים שֶׁמִּתְרַסְּקִים, וַאֲנָשִׁים רָצִים אֶל הַחוֹף.

סְפִינָה גְּדוֹלָה מֵרוֹמָא נִקְלְעָה לַסְּעָרָה, הִתְרַסְּקָה אֶל סַלְעֵי הַחוֹף וְטָבְעָה בִּמְצוּלוֹת.

הַנִּצּוֹל הָעֵירֹם בַּחוֹף

הַגַּלִּים הַזּוֹעֲמִים פָּלְטוּ אֶל הַחוֹף אָדָם אֶחָד שֶׁשָּׂרַד. הוּא שָׁכַב עַל הַחוֹל הָרָטֹב, רוֹעֵד כֻּלּוֹ מִקֹּר, שִׁנָּיו נוֹקְשׁוֹת זוֹ בָּזוֹ: טַק-טַק-טַק . הַבְּגָדִים שֶׁלּוֹ נִקְרְעוּ לְגַמְרֵי בַּיָּם, וְהוּא הָיָה עָיֵף, קָפוּא, מָלֵא בְּחוֹל, וְכִמְעַט לְלֹא נְשִׁימָה.

קְבוּצָה שֶׁל יְהוּדִים עָבְרָה בַּמָּקוֹם. הֵם הִסְתַּכְּלוּ עַל הָאִישׁ הַשָּׂרוּעַ, וְאֶחָד מֵהֶם אָמַר בְּכַעַס:

"זֶה רוֹמָאִי! שִׁמְעוּ אֶת הַשָּׂפָה שֶׁבָּהּ הוּא בּוֹכֶה! הַרוֹמָאִים הָאֵלֶּה הִחְרִיבוּ לָנוּ אֶת בֵּית הַמִּקְדָּשׁ, הֵם גּוֹבִים מֵאִתָּנוּ מִסִּים כְּבֵדִים, הֵם כּוֹבְשִׁים אַכְזָרִים! אֲנִי לֹא הוֹלֵךְ לַעֲזֹר לוֹ. מַגִּיעַ לוֹ שֶׁהַיָּם טָבַע אוֹתוֹ!"

וְהֵם עָזְבוּ אוֹתוֹ שָׁם, קוֹפֵא עַל הַחוֹל.

אֲבָל יְלָדִים, אָנוּ חַיָּבִים לְחַדֵּד מַשֶּׁהוּ חָשׁוּב מְאֹד: אִם מְחַבֵּל בָּא עִם נֶשֶׁק כְּדֵי לִפְגֹּעַ בָּנוּ – מִצְוָה לְהִלָּחֵם בּוֹ. "הַבָּא לְהָרְגְּךָ הַשְׁכֵּם לְהָרְגוֹ". אַךְ הָאִישׁ הַזֶּה לֹא בָּא לְהִלָּחֵם. הוּא הָיָה סְתָם יורד ים רומאי. הוּא הָיָה פָּצוּעַ, עֵירֹם וְחֲסַר אוֹנִים לְגַמְרֵי. רַבִּי אֶלְעָזָר רָאָה מִוּלוֹ אָדָם שֶׁנִּבְרָא בְּצֶלֶם אֱלוֹקִים וְנִמְצָא בְּסַכָּנַת מָוֶת. לֵב הַחֶסֶד הַיְּהוּדִי שֶׁל הַתַּנָּא הַקָּדוֹשׁ לֹא יָכוֹל הָיָה לַעֲמֹד מִנֶּגֶד.

מְעִיל, מָרָק וּמָאתַיִם מַטְבְּעוֹת

רַבִּי אֶלְעָזָר פָּשַׁט אֶת הַמְּעִיל הֶחָם שֶׁלּוֹ – מְעִיל הַכָּבוֹד שֶׁלּוֹ כְּרַב גָּדוֹל – וְעָטַף בּוֹ אֶת הַגּוֹי הָרוֹעֵד. הוּא הֵקִים אוֹתוֹ בַּעֲדִינוּת עַל רַגְלָיו, עָזַר לוֹ לִצְעֹד וְהִכְנִיס אוֹתוֹ אֶל תּוֹךְ בֵּיתוֹ.

בַּבַּיִת, הוֹשִׁיב רַבִּי אֶלְעָזָר אֶת הָרוֹמָאִי לְיַד הַתַּנּוּר הַחַם. הָרַבָּנִית מִהֲרָה לְהָבִיא קְעָרָה גְּדוֹלָה שֶׁל מָרָק מַהְבִּיל. הָרֵיחַ הַמָּתוֹק וְהַטָּעִים שֶׁל הַמָּרָק הֵשִׁיב לָאִישׁ אֶת רוּחוֹ. הוּא אָכַל וְשָׁתָה עַד שֶׁגּוּפוֹ הִפְשִׁיר.

לְמָחֳרָת, כַּאֲשֶׁר הָאִישׁ אָגַר מַסְפִּיק כֹּחַ לַחְזֹר לְאַרְצוֹ, רַבִּי אֶלְעָזָר לֹא שִׁחְרֵר אוֹתוֹ בְּיָדַיִם רֵיקוֹת. הוּא לָקַח שַׂק קָטָן, וְהִכְנִיס לְתוֹכוֹ מָאתַיִם מַטְבְּעוֹת דִּינָר. הַמַּטְבְּעוֹת צִלְצְלוּ בְּתוֹךְ הַשַּׂק: קְלִינְג, קְלִינְג, קְלִינְג . סְכוּם עָצוּם שֶׁל כֶּסֶף!

"קַח אֶת זֶה, אִישׁ זָר," אָמַר רַבִּי אֶלְעָזָר, "זֶה יַעֲזֹר לְךָ בַּדֶּרֶךְ חֲזָרָה לְרוֹמָא. לֵךְ לְשָׁלוֹם."

הָרוֹמָאִי עָמַד נִדְהָם. הַלֵּב שֶׁלּוֹ הָיָה הָמוּם. הוּא קַד קִדָּה עֲמֻקָּה וְלָחַשׁ: "לְעוֹלָם לֹא אֶשְׁכַּח אוֹתְךָ, הָאִישׁ הַיְּהוּדִי."

הַגְּזֵרָה הַקָּשָׁה שֶׁל הַקֵּיסָר הֶחָדָשׁ

עָבְרוּ הַרְבֵּה מְאֹד שָׁנִים. הַיָּמִים חָלְפוּ, הַקֵּיסָר הָרוֹמָאִי שֶׁבְּרוֹמָא מֵת, וּבִמְקוֹמוֹ הֻמְלַךְ קֵיסָר חָדָשׁ אֶחָד שֶׁהָיָה שַׂר כְּלָל יָדוּעַ. (יֵשׁ מְפָרְשִׁים שֶׁהָיָה זֶה מוֹשֵׁל חָדָשׁ שֶׁקִּבֵּל סַמְכוּת נַרְחֶבֶת).

לְפֶתַע, יוֹם אֶחָד, נָפְלָה צָרָה צְרוּרָה עַל הָעִיר שֶׁבָּהּ גָּר רַבִּי אֶלְעָזָר. אִגֶּרֶת נוֹרָאָה הִגִּיעָה מֵרוֹמָא: עַל הַיְּהוּדִים בָּעִיר לְשַׁלֵּם בְּתוֹךְ יָמִים סְפוּרִים סְכוּם עָתֵק שֶׁל אַרְבַּעַת אֲלָפִים דִּינְרֵי כֶּסֶף הַיָּשָׁר לְקוּפַּת הַקֵּיסָר! "וְאִם לֹא יְשַׁלְּמוּ," גָּזַר הַקֵּיסָר, "כָּל תּוֹשָׁבֵי הָעִיר יֵהָרְגוּ וְאוֹתָם יִלְקְחוּ לַעֲבָדִים בַּשְּׁבִי!"

הַפַּחַד הָיָה אָיוֹם וְנוֹרָא. אֲנָשִׁים בָּכוּ בָּרְחוֹבוֹת. "מֵאַיִן נָבִיא כָּל כָּךְ הַרְבֵּה כֶּסֶף?!" זָעֲקוּ הַתּוֹשָׁבִים.

כֻּלָּם פָּנוּ לְרַבִּי אֶלְעָזָר. "רַבֵּנוּ," בָּכוּ פַּרְנְסֵי הַקְּהִלָּה, "אַתָּה גְּדוֹל הַדּוֹר! תְּנַסֶּה אַתָּה לִנְסֹעַ לְרוֹמָא. אוּלַי, בִּזְכוּת תּוֹרָתְךָ, הַקֵּיסָר יְרַחֵם עָלֵינוּ."

רַבִּי אֶלְעָזָר, שֶׁהָיָה כְּבָר אָדָם מְבֻגָּר, שָׂם אֶת הַבִּטָּחוֹן בַּקָּדוֹשׁ בָּרוּךְ הוּא. "אֲנִי אֶלֵּךְ," אָמַר. הוּא יָצָא לַמַּסָּע הָאָרֹךְ וְהַמְּיַגֵּעַ לְרוֹמָא, מַסָּע בְּסְפִינוֹת וּבַדְּרָכִים, כְּשֶׁלִּבּוֹ מְמַלְמֵל תְּפִלָּה חֲרִישִׁית לְרִבּוֹנוֹ שֶׁל עוֹלָם שֶׁיַּעֲשֶׂה עִמָּהֶם נֵס.

בְּאַרְמוֹן הַזָּהָב

רַבִּי אֶלְעָזָר הִגִּיעַ לְרוֹמָא. הָאַרְמוֹן הָיָה מְפֹאָר מֵאֵין כָּמוֹהוּ. עַמּוּדֵי שַׁיִשׁ גְּבוֹהִים, רִצְפָּה נוֹצֶצֶת שֶׁאֶפְשָׁר לִרְאוֹת בָּהּ אֶת הַבָּבוּאָה (הַהִשְׁתַּקְפוּת), וְעִטּוּרֵי זָהָב סִנְוְרוּ אֶת הָעֵינַיִם.

הַדְּלָתוֹת הֶעָבוֹת שֶׁל אֶרֶץ כֵּס הַמַּלְכוּת נִפְתְּחוּ בַּחֲרִיקָה עָבָה מְאַיֶּמֶת: קְרִיקְקְק... טְרָאח!

רַבִּי אֶלְעָזָר פָּסַע פְּנִימָה. בְּסוֹף הָאוּלָם הָעֲנָקִי, עַל כִּסֵּא רָם וְנִשָּׂא מְצֻפֶּה אַבְנֵי חֵן, יָשַׁב הַקֵּיסָר הָרוֹמָאִי בְּבִגְדֵי אַרְגָּמָן. מִשְּׁנֵי צִדָּיו עָמְדוּ שׁוֹמְרִים חֲמוּשִׁים בְּחֲרָבוֹת נְחוֹשֶׁת אֲרֻכּוֹת שֶׁבָּרְקוּ בָּאוֹר. הַשֶּׁקֶט בָּאוּלָם הָיָה כָּבֵד, אֶפְשָׁר הָיָה לִשְׁמֹעַ סִכָּה נוֹפֶלֶת.

רַבִּי אֶלְעָזָר הִתְקָרֵב בְּעָנְוָה, כְּדֵי לְבַקֵּשׁ עַל עַמּוֹ.

לְפֶתַע, הַקֵּיסָר זָקַף אֶת מַבָּטוֹ. עֵינָיו נִפְעֲרוּ לִרְוָחָה. הוּא הִבִּיט בַּזָּקֵן הַיְּהוּדִי בַּעַל הַזָּקָן הַלָּבָן.

הַקֵּיסָר קָפַץ מִכִּסֵּא הַמַּלְכוּת שֶׁלּוֹ! הַשּׁוֹמְרִים נִדְרְכוּ, מִי הֵעֵז לְהַפְרִיעַ לַמֶּלֶךְ? אַךְ הַקֵּיסָר לֹא הִתְעַנְיֵן בָּהֶם. הוּא רָץ בְּמַדְרֵגוֹת הַכֵּס, יָרַד אֶל רַצְפַּת הַשַּׁיִשׁ, קָרָא: "הָרַב הַיְּהוּדִי?!" – וְנָפַל מַמָּשׁ עַל פָּנָיו בְּקִדָּה לְפָנָיו!

רַבִּי אֶלְעָזָר נִדְהַם. "מָה כְּבוֹד הַמֶּלֶךְ עוֹשֶׂה?" שָׁאַל.

הַקֵּיסָר קָם, דְּמָעוֹת הִבְרִיקוּ בְּעֵינָיו. "הַאִם אֵינְךָ מַכִּיר אוֹתִי, רַבִּי?" שָׁאַל בְּקוֹל רוֹעֵד מֵהִתְרַגְּשׁוּת. "לִפְנֵי שָׁנִים רַבּוֹת... סְעָרָה עַל שְׂפַת הַיָּם. סְפִינָה טוֹבַעַת. אָדָם עֵירֹם וְקָפוּא... אָנֹכִי הוּא אוֹתוֹ רוֹמָאִי שֶׁהִצַּלְתָּ אֶת חַיָּיו! כֻּלָּם זָרְקוּ אוֹתִי לָמוּת, אַךְ אַתָּה פָשַׁטְתָּ אֶת מְעִילְךָ וְעָטַפְתָּ אוֹתִי בְּאַהֲבָה! בִּזְכוּת הַכֶּסֶף שֶׁנָּתַתָּ לִי, חָזַרְתִּי לְכָאן, וּלְאַט לְאַט הָפַכְתִּי לַקֵּיסָר!"

הַהַצָּלָה וְהַחֲסָדִים

הַקֵּיסָר הִמְשִׁיךְ וְאָמַר: "מָה שְׁאֵלָתְךָ, יְדִידִי הַמַּצִּיל, וּמָה בַּקָּשָׁתְךָ? עַד חֲצִי הַמַּלְכוּת וְיִנָּתֵן לָךְ!"

"כְּבוֹד הַקֵּיסָר," עָנָה רַבִּי אֶלְעָזָר, "גְּזֵרָה נוֹרָאָה הוֹצֵאתָ עַל עִירִי, לְשַׁלֵּם אַרְבַּעַת אֲלָפִים דִּינָר. אֲנִי מְבַקֵּשׁ מִמְּךָ לְבַטֵּל אֶת הַגְּזֵרָה כְּדֵי שֶׁעַמִּי יִנָּצֵל."

"מִיָּד!" זָעַק הַקֵּיסָר. "הַגְּזֵרָה מְבֻטֶּלֶת כָּלִיל!"

וְלֹא רַק שֶׁהַקֵּיסָר בִּטֵּל אֶת הַגְּזֵרָה, הוּא מִלֵּא תֵּבוֹת עֲנָקִיּוֹת בְּזָהָב, כֶּסֶף וְאַבְנֵי חֵן, וְנָתַן לְרַבִּי אֶלְעָזָר מַתָּנָה כְּפוּלָה וּמְכֻפֶּלֶת בִּשְׁבִיל שֶׁיְּחַלֵּק לַעֲנִיֵּי עִירוֹ, כְּדֵי לְהַרְאוֹת אֶת הַכָּרַת הַטּוֹב הָעֲצוּמָה שֶׁלּוֹ.

כְּשֶׁחָזַר רַבִּי אֶלְעָזָר לְעִירוֹ, יָצְאוּ כָּל הַיְהוּדִים לִקְרָאתוֹ בְּתֻפִּים וּבִמְחוֹלוֹת, וְהֵבִינוּ אֵיךְ חֶסֶד שֶׁל אֱמֶת, שֶׁנַּעֲשָׂה עִם אָדָם חֲסַר אוֹנִים – חוֹזֵר בַּסּוֹף וּמַצִּיל אֶת כָּל עַם יִשְׂרָאֵל! מַמָּשׁ "שַׁלַּח לַחְמְךָ עַל פְּנֵי הַמַּיִם, כִּי בְּרוֹב הַיָּמִים תִּמְצָאֶנּוּ".

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “רוּחַ חֲזָקָה וְקָרָה נוֹשֶׁבֶת, מְעִיפָה אֶת הַחוֹל לְכָל עֵבֶר.” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “הוּא שָׁכַב עַל הַחוֹל הָרָטֹב, רוֹעֵד כֻּלּוֹ מִקֹּר, שִׁנָּיו נוֹקְשׁוֹת זוֹ בָּזוֹ: טַק-טַק-טַק .”?
ג. כיצד הגיב רבי אלעזר בן שמוע למצב המתואר: “הוּא הֵקִים אוֹתוֹ בַּעֲדִינוּת עַל רַגְלָיו, עָזַר לוֹ לִצְעֹד וְהִכְנִיס אוֹתוֹ אֶל תּוֹךְ בֵּיתוֹ.”?
ד. מה השתנה בין “קַח אֶת זֶה, אִישׁ זָר," אָמַר רַבִּי אֶלְעָזָר, "זֶה יַעֲזֹר לְךָ בַּדֶּרֶךְ חֲזָרָה לְרוֹמָא.” לבין “הוּא יָצָא לַמַּסָּע הָאָרֹךְ וְהַמְּיַגֵּעַ לְרוֹמָא, מַסָּע בְּסְפִינוֹת וּבַדְּרָכִים, כְּשֶׁלִּבּוֹ מְמַלְמֵל…”?
ה. כיצד הגיע הסיפור אל האירוע: “בִּזְכוּת הַכֶּסֶף שֶׁנָּתַתָּ לִי, חָזַרְתִּי לְכָאן, וּלְאַט לְאַט הָפַכְתִּי לַקֵּיסָר!”?

3. מה היה ראשון?

כתבו ליד כל אירוע מספר מ־1 עד 5 לפי סדר התרחשותו בסיפור.

□ הַמַּטְבְּעוֹת צִלְצְלוּ בְּתוֹךְ הַשַּׂק: קְלִינְג, קְלִינְג, קְלִינְג .
□ רוּחַ חֲזָקָה וְקָרָה נוֹשֶׁבֶת, מְעִיפָה אֶת הַחוֹל לְכָל עֵבֶר.
□ מַמָּשׁ "שַׁלַּח לַחְמְךָ עַל פְּנֵי הַמַּיִם, כִּי בְּרוֹב הַיָּמִים תִּמְצָאֶנּוּ".
□ הַרוֹמָאִים הָאֵלֶּה הִחְרִיבוּ לָנוּ אֶת בֵּית הַמִּקְדָּשׁ, הֵם גּוֹבִים מֵאִתָּנוּ מִסִּים כְּבֵדִים, הֵם כּוֹבְשִׁים אַכְזָרִים!
□ עַמּוּדֵי שַׁיִשׁ גְּבוֹהִים, רִצְפָּה נוֹצֶצֶת שֶׁאֶפְשָׁר לִרְאוֹת בָּהּ אֶת הַבָּבוּאָה (הַהִשְׁתַּקְפוּת), וְעִטּוּרֵי זָהָב סִנְוְרוּ אֶת הָעֵינַיִם.

לפני ואחרי

מה קרה לפני: “הַמַּטְבְּעוֹת צִלְצְלוּ בְּתוֹךְ הַשַּׂק: קְלִינְג, קְלִינְג, קְלִינְג .”?
מה קרה אחרי: “עַמּוּדֵי שַׁיִשׁ גְּבוֹהִים, רִצְפָּה נוֹצֶצֶת שֶׁאֶפְשָׁר לִרְאוֹת בָּהּ אֶת הַבָּבוּאָה (הַהִשְׁתַּקְפוּת), וְעִטּוּרֵי זָהָב סִנְוְרוּ אֶת הָעֵינַיִם.”?

סיבה ותוצאה מתוך המעשה

האירוע הנתוןמה גרם לו?מה הייתה תוצאתו?
אֲבָל יְלָדִים, אָנוּ חַיָּבִים לְחַדֵּד מַשֶּׁהוּ חָשׁוּב מְאֹד: אִם מְחַבֵּל בָּא עִם נֶשֶׁק כְּדֵי לִפְגֹּעַ בָּנוּ – מִצְוָה לְהִלָּחֵם בּוֹ.
לְפֶתַע, יוֹם אֶחָד, נָפְלָה צָרָה צְרוּרָה עַל הָעִיר שֶׁבָּהּ גָּר רַבִּי אֶלְעָזָר.

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “הצלת הרומאי” על רבי אלעזר בן שמוע? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “לְמָחֳרָת, כַּאֲשֶׁר הָאִישׁ אָגַר מַסְפִּיק כֹּחַ לַחְזֹר לְאַרְצוֹ, רַבִּי אֶלְעָזָר לֹא שִׁחְרֵר אוֹתוֹ בְּיָדַיִם רֵיקוֹת.”.
ג. אילו רבי אלעזר בן שמוע לא היה פועל כפי שמתואר ב־“הוּא יָצָא לַמַּסָּע הָאָרֹךְ וְהַמְּיַגֵּעַ לְרוֹמָא, מַסָּע בְּסְפִינוֹת וּבַדְּרָכִים, כְּשֶׁלִּבּוֹ מְמַלְמֵל תְּפִלָּה…” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “בִּזְכוּת הַכֶּסֶף שֶׁנָּתַתָּ לִי, חָזַרְתִּי לְכָאן, וּלְאַט לְאַט הָפַכְתִּי לַקֵּיסָר!” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “הוּא הֵקִים אוֹתוֹ בַּעֲדִינוּת עַל רַגְלָיו, עָזַר לוֹ לִצְעֹד וְהִכְנִיס אוֹתוֹ אֶל תּוֹךְ בֵּיתוֹ.”.
מכתב: כתבו לרבי אלעזר בן שמוע בעקבות “לְפֶתַע, יוֹם אֶחָד, נָפְלָה צָרָה צְרוּרָה עַל הָעִיר שֶׁבָּהּ גָּר רַבִּי אֶלְעָזָר.”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “מִשְּׁנֵי צִדָּיו עָמְדוּ שׁוֹמְרִים חֲמוּשִׁים בְּחֲרָבוֹת נְחוֹשֶׁת אֲרֻכּוֹת שֶׁבָּרְקוּ בָּאוֹר.”.
דיווח: כתבו ידיעה קצרה שכותרתה “הצלת הרומאי”.
