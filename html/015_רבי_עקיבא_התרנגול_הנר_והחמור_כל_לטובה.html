סיפור 15 • רבי עקיבא • התרנגול, הנר והחמור (כל לטובה)

התרנגול, הנר והחמור (כל לטובה)

דף עבודה בעקבות רבי עקיבא
שם: כיתה:

המעשה — הסיפור המלא

פְּתִיחָה: דֶּרֶךְ אֲרֻכָּה מְלֵאַת אָבָק

יְלָדִים, נַסּוּ לְדַמְיֵן שֶׁאַתֶּם הוֹלְכִים בְּדֶרֶךְ עָפָר אֲרֻכָּה מְאֹד. אָבָק מְכַסֶּה אֶת הַנַּעֲלַיִם שֶׁלָּכֶם, הַשֶּׁמֶשׁ שׁוֹקַעַת לְאַט לְאַט צְבוּעָה בְּצֶבַע כָּתֹם-אָדֹם, וְרוּחַ קָרָה שֶׁל עֶרֶב מַתְחִילָה לְנַשֵּׁב וּלְצַמְרֵר אֶת הָעוֹר. הָרַגְלַיִם כּוֹאֲבוֹת וַעֲיֵפוֹת, וְהַבֶּטֶן קְצָת מְקַרְקֶרֶת.

כָּךְ בְּדִיּוּק צָעַד הַתַּנָּא הַקָּדוֹשׁ וְהַגָּדוֹל, רַבִּי עֲקִיבָא .

הוּא הָיָה בְּדַרְכּוֹ מֵעִיר אַחַת לְעִיר אַחֶרֶת, כְּדֵי לְלַמֵּד תּוֹרָה לְעַם יִשְׂרָאֵל. לְרַבִּי עֲקִיבָא לֹא הָיְתָה מְכוֹנִית, לֹא מִזְוָדָה גְּדוֹלָה שֶׁעוֹלָה לַמָּטוֹס, וַאֲפִלּוּ לֹא מַחֲנֵה אֹהָלִים נוֹחַ.

הָיוּ לוֹ רַק שְׁלוֹשָׁה "חֲבֵרִים" שֶׁלִּוּוּ אוֹתוֹ בַּמַּסָּע הֶעָיֵף:

הָרִאשׁוֹן הָיָה חֲמוֹר חָזָק וְאָפֹר, שֶׁעָזַר לָרַב לִרְכֹּב כְּשֶׁהָרַגְלַיִם כָּאֲבוּ וְנָשָׂא אֶת מְעַט הַחֲפָצִים.

הַשֵּׁנִי הָיָה תַּרְנְגוֹל אָדוֹם-כַּרְבֹּלֶת, שֶׁתַּפְקִידוֹ הָיָה מְאֹד חָשׁוּב: לְשַׁמֵּשׁ שָׁעוֹן מְעוֹרֵר! מִדֵּי בֹּקֶר, מֻקְדָּם מְאֹד, כְּשֶׁעוֹד חָשׁוּךְ, הָיָה הַתַּרְנְגוֹל קוֹרֵא 'קוּ-קוּ-רִיקוּ' חָזָק, כְּדֵי שֶׁרַבִּי עֲקִיבָא יָקוּם בִּזְרִיזוּת לְלִמּוּד הַתּוֹרָה וְלִתְפִלַּת וָתִיקִין.

וְהַשְּׁלִישִׁי? הַשְּׁלִישִׁי הָיָה נֵר קָטָן עָשׂוּי מִשַּׁעֲוָה וּפְתִילָה. בִּשְׁבִיל מָה צָרִיךְ נֵר? כְּדֵי שֶׁכַּאֲשֶׁר יוֹרֵד הַלַּיְלָה הַחֲשֵׁכָה, רַבִּי עֲקִיבָא לֹא יְבַטֵּל אֲפִלּוּ דַּקָּה אַחַת שֶׁל נַחַת מִלִּמּוּד הַתּוֹרָה מִתּוֹךְ סְפָרָיו הַקְּדוֹשִׁים.

הַדְּלָתוֹת הַנִּטְרָקוֹת

הַחֹשֶׁךְ יָרַד, וְהַשָּׁמַיִם נִצְבְּעוּ בְּשָׁחֹר עָמֹק. קוֹלוֹת שֶׁל תַּנִּים מְיַלְּלִים נִשְׁמְעוּ מִן הַיַּעַר, וְהַקֹּר עָשָׂה צְמַרְמֹרֶת בַּגּוּף. מֵרָחוֹק דָּלְקוּ אוֹרוֹת צְהֻבִּים שֶׁל בָּתִּים קְטַנִּים. זֶה הָיָה כְּפָר עַתִּיק שֶׁעָמַד עַל אֵם הַדֶּרֶךְ.

רַבִּי עֲקִיבָא שָׂמַח: "הִנֵּה, מָקוֹם שֶׁבּוֹ אוּכַל לָנוּחַ הַלַּיְלָה בְּחֹם וּבְבִטָּחוֹן!". הוּא הִתְקָרֵב לַדֶּלֶת הָרִאשׁוֹנָה שֶׁמָּצָא, וְדָפַק בַּעֲדִינוּת: טּוּק, טּוּק, טּוּק...

הַדֶּלֶת נִפְתְּחָה לְאִטָּהּ. רֹאשׁוֹ שֶׁל אִישׁ עֲרָבִי הֵצִיץ מִבִּפְנִים.

"שָׁלוֹם עָלֶיךָ," אָמַר הַתַּנָּא מֵאִיר הַפָּנִים, "הַאִם תַּסְכִּים לְאָרֵחַ אוֹתִי רַק לְלַיְלָה אֶחָד? יֵשׁ לִי עִמִּי תַּרְנְגוֹל וַחֲמוֹר...".

הָאִישׁ אֲפִלּוּ לֹא עָנָה. הוּא הִבִּיט בַּזָּקֵן, עִקֵּם אֶת פַּרְצוּפוֹ וְ... טְרָאח! סָגַר אֶת הַדֶּלֶת בְּחָזְקָה!

רַבִּי עֲקִיבָא לֹא אִבֵּד תִּקְוָה. הוּא פָּסַע לַבַּיִת הַבָּא. וְדָפַק שׁוּב. "אֶפְשָׁר בְּבַקָּשָׁה מָקוֹם לִישׁוֹן?" מֵהַחַלּוֹן הֵצִיצוּ אֲנָשִׁים, אַךְ אִישׁ לֹא פָּתַח לוֹ אֶת הַדֶּלֶת. בְּאוֹתָם יָמִים, הָרוֹמָאִים חִפְּשׂוּ יְהוּדִים שֶׁלּוֹמְדִים תּוֹרָה, וְכַנִּרְאֶה תּוֹשָׁבֵי הַכְּפָר פָּחֲדוּ لְהַכְנִיס זָרִים לְבֵיתָם, וְהָיוּ שֶׁפָּשׁוּט הָיוּ רָעִים וְקַמְצָנִים וְלֹא רָצוּ לַעֲשׂוֹת חֶסֶד שֶׁל הַכְנָסַת אוֹרְחִים.

אִישׁ רָגִיל, אִם הָיוּ מְגָרְשִׁים אוֹתוֹ בַּלַּיְלָה הַקַּר כָּךְ לָרְחוֹב, הָיָה מַתְחִיל לִכְעֹס, לִבְכּוֹת אוֹ לְהִתְלוֹנֵן: "לָמָּה זֶה מַגִּיעַ לִי? אֲנִי רַב צַדִּיק! כָּכָה מִתְנַהֲגִים לְאוֹרְחִים?!".

אֲבָל רַבִּי עֲקִיבָא? הַלֵּב שֶׁלּוֹ הָיָה מָלֵא בְּאַהֲבַת ה'. הוּא יָצָא מֵהַכְּפָר, הִבִּיט לַשָּׁמַיִם הַשְּׁחוֹרִים, עָצַם עֵינַיִם, וְאָמַר בְּקוֹל שָׁקֵט וּבוֹטֵחַ אֶת מִשְׁפַּט הַקֶּסֶם: "כָּל מָה שֶׁעוֹשֶׂה הַקָּדוֹשׁ בָּרוּךְ הוּא – הַכֹּל לְטוֹבָה הוּא עוֹשֶׂה!" . מָה שֶׁה' גָּזַר, הֲכִי מֻשְׁלָם עֲבוּרִי! וְעִם הָאֱמוּנָה הַזּוֹ בַּלֵּב, הוּא חִיֵּךְ וְהִמְשִׁיךְ לָלֶכֶת לְכִוּוּן הַיַּעַר הֶעָבֹת, שֶׁעֵצָיו גְּבוֹהִים וּמַפְחִידִים בַּחֲשֵׁכָה.

אֹבְדַן הָאֹר, הַשָּׁעוֹן וְהָרֶכֶב

רַבִּי עֲקִיבָא מָצָא עֵץ זַיִת גָּדוֹל עִם רֵיחַ נִפְלָא שֶׁל אֲדָמָה לַחָה, קָשַׁר אֶת חֲמוֹרוֹ, הוֹשִׁיב לְצִדּוֹ אֶת הַתַּרְנְגוֹל הַמְכֻרְבָּל בְּנוֹצוֹתָיו, וְהִתְיַשֵּׁב עַל הָאֲדָמָה הַקָּרָה.

זֶהוּ. עַכְשָׁו אֶפְשָׁר לִלְמֹד תּוֹרָה!

הוּא הִדְלִיק בְּעֶזְרַת אֲבָנִים אֶת הַנֵּר הַקָּטָן. הָאוֹר הַצְּהַבְהַב רִצֵּד עַל עֲלֵי הָעֵץ, וְחִמֵּם אֶת הָאֲוִיר בְּאוֹר שֶׁל קְדֻשָּׁה. רַבִּי עֲקִיבָא פָּתַח מְגִלָּה וְשָׁקַע בְּדִבְרֵי אֱלֹקִים חַיִּים.

אַךְ אָז... פִישששש! רוּחַ חֲזָקָה וּקְרִירָה נָשְׁבָה מִבֵּין הֶהָרִים. הַנֵּר זָז שְׂמֹאלָה וְיָמִינָה, שׁוּב נָשְׁבָה רוּחַ – וְהַנֵּר כָּבָה!

חשֶׁךְ מִצְרַיִם בַּיַּעַר. אִי אֶפְשָׁר לִקְרֹא מִלָּה אַחַת בַּסֵּפֶר!

יְלָדִים, כַּמָּה כּוֹעֲסִים כְּשֶׁבָּאֶמְצַע לִמּוּד נִכְבֵּית הַמְּנוֹרָה וְהַכֹּל נֶהֱרָס? אֲבָל רַבִּי עֲקִיבָא רַק נָשַׁם נְשִׁימָה עֲמֻקָּה, וְאָמַר לְתוֹךְ הַלַּיְלָה הַשָּׁחוֹר: "כָּל מָה שֶׁעוֹשֶׂה הַקָּדוֹשׁ בָּרוּךְ הוּא – לְטוֹבָה וְלִבְרָכָה!" . "אִם ה' רוֹצֶה שֶׁאִישַׁן וְלֹא אֶלְמַד, כַּנִּרְאֶה זֶה הֲכִי נָכוֹן לִי".

הַתַּנָּא שָׁכַב עַל הָאֲדָמָה כְּדֵי לִישֹׁן. פִּתְאוֹם, לִפְנֵי שֶׁהִסְפִּיק לַעֲצֹם עֵינַיִם, נִשְׁמְעָה יְלָלָה חַדָּה וְנִמְשֶׁכֶת! חָתוּל-בַּר חָזָק וְאַכְזָרִי זִנֵּק מִתּוֹךְ הַשִּׂיחִים, הִתְנַפֵּל עַל הַתַּרְנְגוֹל – וְטָרַף אוֹתוֹ בִּשְׁנִיָּה מֵהִירָה. הַתַּרְנְגוֹל נֶעֱלַם בַּחֲשֵׁכָה!

אָבַד הַשָּׁעוֹן. מִי יָעִיר כָּעֵת אֶת רַבִּי עֲקִיבָא בַּבֹּקֶר לִתְפִלַּת הַשַּׁחַר? מִי יִקְרָא בַּחֲשֵׁכָה כְּדֵי שֶׁלֹּא יִשְׁכַּב הָרַב מְאֻחָר בַּמִּטָּה?

אַךְ שׁוּב, אוֹתָהּ אֱמוּנָה מֻפְלָאָה הִדְהֲדָה בַּחֲשֵׁכָה: "כָּל דְּעָבִיד רַחֲמָנָא – לְטָב עָבִיד! כָּל מָה שֶׁה' רַחְמָן עוֹשֶׂה – רַק לְטוֹבָה הִיא!" .

עָבְרוּ רַק כַּמָּה דַּקּוֹת. הַשֶּׁקֶט חָזַר. וְהִנֵּה... קוֹל פְּסִיעוֹת אִטִּיּוֹת וּכְבֵדוֹת שֶׁל כַּפּוֹת רַגְלַיִם גְּדוֹלוֹת מָעֲכוּ אֶת עֲלֵי הַשַּׁלֶּכֶת... ררראארררוווו! – שַׁאֲגַת אֵימָה הִרְעִידָה אֶת גִּזְעֵי הָעֵצִים! הָאֲדָמָה זָזָה! מֵהַיַּעַר יָצָא לְפֶתַע מֶלֶךְ הַחַיּוֹת הַמַּפְחִיד – אַרְיֵה רָעֵב וְעָצוּם. הָאַרְיֵה זִנֵּק יְשִׁירוֹת עַל הַחֲמוֹר הָאֻמְלָל, נָגַס בּוֹ, הֵטִיל אוֹתוֹ עַל אֲדָמָה, וְטָרַף אֶת רִכְבּוֹ הַיָּקָר שֶׁל הָרַב בְּבַת אַחַת. כָּךְ נֶעֱלַם כְּלִי הַתַּחְבּוּרָה הַיָּחִיד שֶׁל הָרַב.

יְלָדִים! זֶה רֶגַע קָשֶׁה שֶׁלֹּא יֵאָמֵן! רַבִּי עֲקִיבָא זָקֵן, עָיֵף, קוֹפֵא מִקֹּר. דָּחוּ אוֹתוֹ מֵהַבָּתִּים. הוּא נִמְצָא בְּסַכָּנַת מָוֶת מִיְּדֵי שׁוֹדְדִים אוֹ אֲרָיוֹת, אִבֵּד אֶת הָאֹר שֶׁלּוֹ, אֶת הַשָּׁעוֹן שֶׁלּוֹ וְאֶת הַחֲמוֹר שֶׁלּוֹ. מִי יָכוֹל לְשַׂמֵּחַ אָדָם בְּמַצָּב כָּזֶה?! רַק אָדָם אֶחָד יָכוֹל – מִי שֶׁיּוֹדֵעַ שֶׁ"אַבָּא שֶׁבַּשָּׁמַיִם מַחְזִיק לוֹ אֶת הַיָּד".

רַבִּי עֲקִיבָא, מִתּוֹךְ הַסַּכָּנָה וְהַחֹשֶׁךְ, לָחַשׁ עִם חִיּוּךְ זָעִיר שֶׁל תִּקְוָה וֶאֱמוּנָה: "הַכֹּל הַכֹּל מֵאֵת ה' – וְהַכֹּל לְטוֹבָה הַמֻּחְלֶטֶת!" . הוּא הִתְעַטֵּף בִּמְעִילוֹ, וְנִרְדַּם בְּשֶׁקֶט, שׁוֹמֵר עַל דְּמָמָה מוּחְלֶטֶת שֶׁל אֵין-אוֹר וְאֵין-צְלִיל.

הַסּוֹד מִתְגַּלֶּה בַּבֹּקֶר

בְּאֶמְצַע הַלַּיְלָה... שָׁבַר אֶת הַשֶּׁקֶט קוֹל עָצוּם, נָהוֹר, שֶׁהִגִּיעַ לֹא מֵהַיַּעַר – אֶלָּא שֶׁל אֲנָשִׁים הַצּוֹעֲקִים. הַצְּעָקוֹת בָּאוּ מִכִּוּוּן הַכְּפָר שֶׁמִּמֶּנּוּ גֵּרְשׁוּ אֶת הָרַב בָּעֶרֶב!

רַבִּי עֲקִיבָא קָפַץ. הוּא הִרִיחַ לְפֶתַע רֵיחַ חָרִיף שֶׁל עָשָׁן אָפֹר וְאֵשׁ בּוֹעֶרֶת, וְרָאָה עֲנָנִים אֲדֻמִּים מֵעַל גַּגּוֹת הַכְּפָר. "גְּיָסוֹת!" לָחֲשׁוּ הָעֵצִים, רוֹמָאִים אוֹ שׁוֹדְדִים רְשָׁעִים וְאַכְזָרִיִּים פָּרְצוּ לְתוֹךְ הַכְּפָר בְּאִישׁוֹן לַיְלָה!

הַשּׁוֹדְדִים שָׁבְרוּ דְּלָתוֹת, צָעֲקוּ עַל הַתּוֹשָׁבִים, קָשְׁרוּ אוֹתָם בַּחֲבָלִים, וְלָקְחוּ אֶת כֻּלָּם לְשֶׁבִי קָשֶׁה וּכְאֵב נוֹרָא כְּדֵי לִמְכֹּר אוֹתָם כַּעֲבָדִים מִסְכֵּנִים. הַכְּפָר עָלָה בְּאֵשׁ וְנֶחְרַב כָּל אוֹתוֹ הַלַּיְלָה, עַד שֶׁהָרְשָׁעִים הִסְתַּלְּקוּ מִשָּׁם עִם הַשְּׁבוּיִים לְעֵבֶר הֶהָרִים.

כְּשֶׁעָלָה עַמּוּד הַשַּׁחַר וּקְרַנֵּי הַשֶּׁמֶשׁ הָרִאשׁוֹנוֹת הֵאִירוּ אֶת הַיַּעַר, נָעֲמַד הַתַּנָּא רַבִּי עֲקִיבָא לְהִתְפַּלֵּל תְּפִלַּת שַׁחֲרִית מִתּוֹךְ הוֹדָיָה עֲצוּמָה וּדְמָעוֹת. לְפֶתַע הֵבִין אֶת הַחִידָה.

עַכְשָׁו הַכֹּל נִרְאָה כְּמוֹ פָּאזֶל שֶׁהִתְחַבֵּר שָׁלֵם!

רַבִּי עֲקִיבָא פָּנָה לְתַלְמִידָיו בַּדּוֹרוֹת הַבָּאִים וְאָמַר: "וְלֹא אָמַרְתִּי לָכֶם כִּי כָּל מָה שֶׁעוֹשֶׂה הַקָּדוֹשׁ בָּרוּךְ הוּא, הַכֹּל לְטוֹבָה הִיא?

חִשְׁבוּ מָה הָיָה קוֹרֶה! אִלּוּ אַנְשֵׁי הַכְּפָר הָיוּ מַכְנִיסִים אוֹתִי לָלוּן בְּבֵיתָם – הָיוּ הַשּׁוֹדְדִים וְהָרוֹמָאִים תּוֹפְסִים אוֹתִי יַחַד עִמָּם, וְהָיִיתִי גַּם אֲנִי בַּשֶּׁבִי!

וְאִלּוּ הַנֵּר שֶׁלִּי הָיָה מַמְשִׁיךְ לִדְלֹק וְלֹא כָּבֶה מֵהָרוּחַ – מֵרָחוֹק הָיוּ הַשּׁוֹדְדִים רוֹאִים מְעַט אוֹר בַּיַּעַר, הָיוּ רָצִים וְלוֹכְדִים אוֹתִי!

וְאִם הַתַּרְנְגוֹל הָיָה מְקַרְקֵר 'קוּ-קוּ-רִיקוּ', אוֹ הַחֲמוֹר הָיָה נוֹעֵר 'הִי-אָה' בְּקוֹל רָעַשׁ מַחֲרִישׁ אָזְנַיִם בַּלַּיְלָה – כָּל צְבָא הַשּׁוֹדְדִים הָיָה בָּא לִתְפֹּס אוֹתִי!

הַדָּבָר הֲכִי טוֹב שֶׁיָּכוֹל הָיָה לִקְרוֹת לִי – זֶה שֶׁהָרוּחַ תְּכַבֶּה, שֶׁהַחָתוּל יִטְרֹף, וְשֶׁהָאַרְיֵה יֹאכַל. כָּכָה ה' הִסְתִּיר אוֹתִי מְיֻחָד בַּחֹשֶׁךְ מֵעֵינֵי הָרוֹצְחִים!

גַּם כְּשֶׁנִּרְאֶה לָנוּ שֶׁזֶּה הַדָּבָר הֲכִי נוֹרָא בָּעוֹלָם, תָּמִיד נִזְכֹּר – הַכֹּל, רַק אֶלָּא לְטוֹבָה! "

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “יְלָדִים, נַסּוּ לְדַמְיֵן שֶׁאַתֶּם הוֹלְכִים בְּדֶרֶךְ עָפָר אֲרֻכָּה מְאֹד.” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “מִדֵּי בֹּקֶר, מֻקְדָּם מְאֹד, כְּשֶׁעוֹד חָשׁוּךְ, הָיָה הַתַּרְנְגוֹל קוֹרֵא 'קוּ-קוּ-רִיקוּ' חָזָק, כְּדֵי שֶׁרַבִּי עֲקִיבָא יָקוּם בִּזְרִיזוּת…”?
ג. כיצד הגיב רבי עקיבא למצב המתואר: “הָאוֹר הַצְּהַבְהַב רִצֵּד עַל עֲלֵי הָעֵץ, וְחִמֵּם אֶת הָאֲוִיר בְּאוֹר שֶׁל קְדֻשָּׁה.”?
ד. מה השתנה בין “אַךְ שׁוּב, אוֹתָהּ אֱמוּנָה מֻפְלָאָה הִדְהֲדָה בַּחֲשֵׁכָה: "כָּל דְּעָבִיד רַחֲמָנָא – לְטָב עָבִיד!” לבין “שָׁבַר אֶת הַשֶּׁקֶט קוֹל עָצוּם, נָהוֹר, שֶׁהִגִּיעַ לֹא מֵהַיַּעַר – אֶלָּא שֶׁל אֲנָשִׁים הַצּוֹעֲקִים.”?
ה. כיצד הגיע הסיפור אל האירוע: “וְאִלּוּ הַנֵּר שֶׁלִּי הָיָה מַמְשִׁיךְ לִדְלֹק וְלֹא כָּבֶה מֵהָרוּחַ – מֵרָחוֹק הָיוּ הַשּׁוֹדְדִים רוֹאִים מְעַט אוֹר בַּיַּעַר, הָיוּ רָצִים…”?

□ גַּם כְּשֶׁנִּרְאֶה לָנוּ שֶׁזֶּה הַדָּבָר הֲכִי נוֹרָא בָּעוֹלָם, תָּמִיד נִזְכֹּר – הַכֹּל, רַק אֶלָּא לְטוֹבָה!
□ רַבִּי עֲקִיבָא שָׂמַח: "הִנֵּה, מָקוֹם שֶׁבּוֹ אוּכַל לָנוּחַ הַלַּיְלָה בְּחֹם וּבְבִטָּחוֹן!".
□ הַצְּעָקוֹת בָּאוּ מִכִּוּוּן הַכְּפָר שֶׁמִּמֶּנּוּ גֵּרְשׁוּ אֶת הָרַב בָּעֶרֶב!

לפני ואחרי

מה קרה לפני: “חָתוּל-בַּר חָזָק וְאַכְזָרִי זִנֵּק מִתּוֹךְ הַשִּׂיחִים, הִתְנַפֵּל עַל הַתַּרְנְגוֹל – וְטָרַף אוֹתוֹ בִּשְׁנִיָּה מֵהִירָה.”?
מה קרה אחרי: “הַצְּעָקוֹת בָּאוּ מִכִּוּוּן הַכְּפָר שֶׁמִּמֶּנּוּ גֵּרְשׁוּ אֶת הָרַב בָּעֶרֶב!”?

סיבה ותוצאה מתוך המעשה

האירוע הנתוןמה גרם לו?מה הייתה תוצאתו?
הוּא הִתְקָרֵב לַדֶּלֶת הָרִאשׁוֹנָה שֶׁמָּצָא, וְדָפַק בַּעֲדִינוּת: טּוּק, טּוּק, טּוּק...
הוּא נִמְצָא בְּסַכָּנַת מָוֶת מִיְּדֵי שׁוֹדְדִים אוֹ אֲרָיוֹת, אִבֵּד אֶת הָאֹר שֶׁלּוֹ, אֶת הַשָּׁעוֹן שֶׁלּוֹ וְאֶת הַחֲמוֹר שֶׁלּוֹ.

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “התרנגול, הנר והחמור (כל לטובה)” על רבי עקיבא? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “אִם ה' רוֹצֶה שֶׁאִישַׁן וְלֹא אֶלְמַד, כַּנִּרְאֶה זֶה הֲכִי נָכוֹן לִי".”.
ג. אילו רבי עקיבא לא היה פועל כפי שמתואר ב־“שָׁבַר אֶת הַשֶּׁקֶט קוֹל עָצוּם, נָהוֹר, שֶׁהִגִּיעַ לֹא מֵהַיַּעַר – אֶלָּא שֶׁל אֲנָשִׁים הַצּוֹעֲקִים.” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “וְאִלּוּ הַנֵּר שֶׁלִּי הָיָה מַמְשִׁיךְ לִדְלֹק וְלֹא כָּבֶה מֵהָרוּחַ – מֵרָחוֹק הָיוּ הַשּׁוֹדְדִים רוֹאִים מְעַט אוֹר…” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. יְלָדִים, נַסּוּ לְדַמְיֵן שֶׁאַתֶּם הוֹלְכִים בְּדֶרֶךְ עָפָר אֲרֻכָּה…
2. כְּדֵי שֶׁכַּאֲשֶׁר יוֹרֵד הַלַּיְלָה הַחֲשֵׁכָה, רַבִּי עֲקִיבָא לֹא…
3. רַבִּי עֲקִיבָא פָּתַח מְגִלָּה וְשָׁקַע בְּדִבְרֵי אֱלֹקִים חַיִּים.
4. הָאַרְיֵה זִנֵּק יְשִׁירוֹת עַל הַחֲמוֹר הָאֻמְלָל, נָגַס בּוֹ, הֵטִיל…
5. גְּיָסוֹת!" לָחֲשׁוּ הָעֵצִים, רוֹמָאִים אוֹ שׁוֹדְדִים רְשָׁעִים…
6. גַּם כְּשֶׁנִּרְאֶה לָנוּ שֶׁזֶּה הַדָּבָר הֲכִי נוֹרָא בָּעוֹלָם,…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “הָאוֹר הַצְּהַבְהַב רִצֵּד עַל עֲלֵי הָעֵץ, וְחִמֵּם אֶת הָאֲוִיר בְּאוֹר שֶׁל קְדֻשָּׁה.”.
מכתב: כתבו לרבי עקיבא בעקבות “הוּא נִמְצָא בְּסַכָּנַת מָוֶת מִיְּדֵי שׁוֹדְדִים אוֹ אֲרָיוֹת, אִבֵּד אֶת הָאֹר שֶׁלּוֹ, אֶת הַשָּׁעוֹן שֶׁלּוֹ וְאֶת הַחֲמוֹר…”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “הַשּׁוֹדְדִים שָׁבְרוּ דְּלָתוֹת, צָעֲקוּ עַל הַתּוֹשָׁבִים, קָשְׁרוּ אוֹתָם בַּחֲבָלִים, וְלָקְחוּ אֶת כֻּלָּם לְשֶׁבִי קָשֶׁה…”.
דיווח: כתבו ידיעה קצרה שכותרתה “התרנגול, הנר והחמור (כל לטובה)”.

המחזה

רגע מן הסיפורהדמויותמה יאמרו או יעשו?
הוּא הִתְקָרֵב לַדֶּלֶת הָרִאשׁוֹנָה שֶׁמָּצָא, וְדָפַק בַּעֲדִינוּת: טּוּק, טּוּק, טּוּק...
שָׁבַר אֶת הַשֶּׁקֶט קוֹל עָצוּם, נָהוֹר, שֶׁהִגִּיעַ לֹא מֵהַיַּעַר – אֶלָּא שֶׁל אֲנָשִׁים…

7. חיבור לחיים מתוך המעשה

א. מה אפשר ללמוד מן הדרך שבה רבי עקיבא פעל באירוע “אַךְ שׁוּב, אוֹתָהּ אֱמוּנָה מֻפְלָאָה הִדְהֲדָה בַּחֲשֵׁכָה: "כָּל דְּעָבִיד רַחֲמָנָא – לְטָב עָבִיד!”?
ב. תארו מצב מחיי הכיתה שבו אפשר ליישם את המסר של “התרנגול, הנר והחמור (כל לטובה)”.
