כיבוד אורח — רק משל המארח

דף עבודה בעקבות ר' חנינא בר חמא
שם: כיתה:

המעשה — הסיפור המלא

פְּתִיחָה: שְׁקִיעָה בְּהָרֵי הַגָּלִיל

יְלָדִים, עִצְמוּ רֶגַע אֶת הָעֵינַיִם. דַּמְיְנוּ שֶׁאַתֶּם נִמְצָאִים בָּעִיר צִפּוֹרִי, לִפְנֵי כְּאַלְפַּיִם שָׁנָה. הַשֶּׁמֶשׁ מַתְחִילָה לָרֶדֶת מֵעַל הָרֵי הַגָּלִיל וְצוֹבַעַת אֶת הַשָּׁמַיִם בִּצְבָעִים מַרְהִיבִים שֶׁל כָּתֹם, אָדֹם וְזָהָב. הָרוּחַ שֶׁל שְׁעַת עֶרֶב מְנַשֶּׁבֶת וּמְבִיאָה אִתָּהּ קֹר נָעִים וְרֵיחַ שֶׁל עֲצֵי זַיִת וּתְאֵנָה.

בִּרְחוֹבָהּ הָרָאשִׁי שֶׁל הָעִיר, עָמַד בַּיִת אֶחָד שֶׁדְּלָתוֹ הָיְתָה פְּתוּחָה לִרְוָחָה. זֶהוּ בֵּיתוֹ שֶׁל גְּדוֹל הַדּוֹר, הָאָמוֹרָא הַקָּדוֹשׁ רַבִּי חֲנִינָא בַּר חָמָא . הַקָּדוֹשׁ בָּרוּךְ הוּא בֵּרֵךְ אֶת רַבִּי חֲנִינָא בְּעֹשֶׁר רַב, וְהוּא הִשְׁתַּמֵּשׁ בְּכָל מָמוֹנוֹ כְּדֵי לְהַרְבּוֹת בְּמִצְווֹת וּמַעֲשִׂים טוֹבִים.

וְהִנֵּה, מִקְּצֵה הָרְחוֹב נִשְׁמַע קוֹל תִּקְתּוּק פַּרְסוֹת חֲמוֹר: טִיק טַק... טִיק טַק... עַל גַּב הַחֲמוֹר יָשַׁב אִישׁ עָיֵף וְרָעֵב. הַבְּגָדִים שֶׁלּוֹ הָיוּ מְלֵאִים בְּאָבָק דָּבִיק וַאֲפַרְפַּר שֶׁל הַדְּרָכִים, הַשְּׂפָתַיִם שֶׁלּוֹ הָיוּ יְבֵשׁוֹת מִצָּמָא, וְעַל הָעֲגָלָה הַקְּטַנָּה שֶׁלּוֹ הָיוּ מֻנָּחִים כַּמָּה שַׂקִּים וְעֵצִים שֶׁאָסַף בַּדֶּרֶךְ.

הוּא חִפֵּשׂ מָקוֹם לָנוּחַ בּוֹ הַלַּיְלָה. כְּשֶׁרָאָה אֶת הַדֶּלֶת הַפְּתוּחָה שֶׁל רַבִּי חֲנִינָא, הָרְגִּישׁ תִּקְוָה גְּדוֹלָה מְמַלֵּאת אֶת לִבּוֹ.

קַבָּלַת הַפָּנִים

טוּק... טוּק... דָּפַק הָאִישׁ עַל עֵץ הַדֶּלֶת הֶעָבֶה.

רַבִּי חֲנִינָא בִּכְבוֹדוֹ וּבְעַצְמוֹ קָם מִשֻּׁלְחַן לִמּוּדוֹ וְנִגַּשׁ אֵלָיו. פָּנָיו שֶׁל הַצַּדִּיק הָיוּ מְאִירוֹת בְּשִׂמְחָה עֲצוּמָה!

"שָׁלוֹם עָלֶיךָ, יְהוּדִי יָקָר! בָּרוּךְ בּוֹאֲךָ לְבֵיתִי!" בֵּרֵךְ הָרַב. הוּא מִיָּד קָרָא לִמְשָׁרְתוֹ: "מַהֵר, קַח מִן הָאוֹרֵחַ אֶת הֶחָמוֹר, תֵּן לוֹ מַיִם קָרִים וָקֶשׁ מֵהָאֲסָם. וְאַתָּה, אָחִי," פָּנָה לָאוֹרֵחַ, "הִכָּנֵס! הִנֵּה קְעָרַת מַיִם חַמִּים לִרְחֹץ אֶת רַגְלֶיךָ הַכּוֹאֲבוֹת. שֵׁב וְנַח כָּאן, וַאֲנַחְנוּ נָכִין לְךָ מִיָּד סְעוּדָה חַמָּה וּמְשִׁיבַת נֶפֶשׁ".

הָאוֹרֵחַ הִתְיַשֵּׁב עַל הַכִּסֵּא הָרַךְ, נָשַׁם נְשִׁימָה אֲרֻכָּה שֶׁל רְוָחָה, וְהִרְגִּישׁ שֶׁכָּל הָעֲיֵפוּת יוֹצֵאת לְאַט לְאַט מֵהַגּוּף שֶׁלּוֹ. הַשֶּׁקֶט שֶׁל בֵּית הַצַּדִּיק עָטַף אוֹתוֹ.

הַבְּעָיָה בַּמִּטְבָּח וְהַפִּתְרוֹן הַשָּׁגוּי

בַּזְּמַן הַזֶּה, בַּמִּטְבָּח בָּאֲחוֹרֵי הַבַּיִת, הַמְּשָׁרֵת מִהֵר לַעֲבוֹדָתוֹ. רַבִּי חֲנִינָא צִוָּה עָלָיו לְהָכִין תַּבְשִׁיל מְשֻׁבָּח לָאוֹרֵחַ. הַמְּשָׁרֵת חָתַךְ יְרָקוֹת, שָׂם נֶתַח דָּג שָׁמֵן בְּתוֹךְ סִיר כֶּרֶס, וְנִגַּשׁ לַתַּנּוּר. אֲבָל... הִנֵּה בְּעָיָה. סַל הָעֵצִים לַתַּנּוּר הָיָה רֵיק! נִשְׁאֲרוּ בּוֹ רַק כַּמָּה עֵצִים לַחִים שֶׁהֶעֱלוּ הַרְבֵּה עָשָׁן וְלֹא נִדְלְקוּ כְּמוֹ שֶׁצָּרִיךְ.

"אוֹי וַאֲבוֹי," חָשַׁב הַמְּשָׁרֵת, "אִם אֵלֵךְ עַכְשָׁו אֶל הַשּׁוּק כְּדֵי לִקְנוֹת עֵצִים יְבֵשִׁים לְהַסָּקָה, הַדָּבָר יִקַּח זְמַן רַב! הָאוֹרֵחַ הַמִּסְכֵּן יִצְטָרֵךְ לְחַכּוֹת שָׁעָה אֲרֻכָּה עִם בֶּטֶן מְקַרְקֶרֶת! הָרַב צִוָּה עָלַי לְמַהֵר... מָה אֶעֱשֶׂה?".

פִּתְאוֹם נָפְלָה מַבָּטוֹ עַל הָעֲגָלָה שֶׁל הָאוֹרֵחַ, שֶׁחָנְתָה בֶּחָצֵר. בְּתוֹךְ הָעֲגָלָה הָיוּ מֻנָּחִים כַּמָּה גִּזְרֵי עֵץ יְבֵשִׁים וּמְצֻיָּנִים.

"זֶה רַעְיוֹן נִפְלָא!" מִלְמֵל הַמְּשָׁרֵת. "אֲנִי אֶקַּח מִשָּׁם עֵץ אֶחָד אוֹ שְׁנַיִם. הֲרֵי זֶה נוֹעַד כְּדֵי לְבַשֵּׁל עֲבוּרוֹ ! הָאוֹרֵחַ בֶּטַח לֹא יִכְעַס, הוּא אֲפִלּוּ יִשְׂמַח שֶׁהָאוֹכֶל שֶׁלּוֹ יִהְיֶה מוּכָן מַהֵר!".

יְלָדִים, הַמְּשָׁרֵת חָשַׁב שֶׁהוּא עוֹשֶׂה מַעֲשֶׂה חָכָם, אֲבָל הוּא לֹא הֵבִין שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא בּוֹחֵן אוֹתָם עַכְשָׁו מִלְּמַעְלָה. מִצְוָה אִי אֶפְשָׁר לַעֲשׂוֹת עַל חֶשְׁבּוֹן שֶׁל מִישֶׁהוּ אַחֵר.

הַמְּשָׁרֵת לָקַח אֶת הָעֵצִים שֶׁל הָאוֹרֵחַ, הִכְנִיס אוֹתָם לַתַּנּוּר וְהִדְלִיק מְדוּרָה. אֵשׁ חַמָּה וּכְתֻמָּה פָּרְצָה כְּלַפֵּי מַעְלָה. פִּיץ פָּאץ' נִשְׁמַע פִּצְפּוּץ הָעֵצִים הַיְבֵשִׁים, וְתוֹךְ זְמַן קָצָר, נִיחוֹחַ מְטֹרָף וּמַהְבִּיל שֶׁל אוֹכֶל טָעִים וַחַם הִתְפַּשֵּׁט בְּכָל הַבַּיִת! הָאוֹרֵחַ כִּמְעַט הִתְעַלֵּף מֵרֹב צִפִּיָּה כְּשֶׁהֵרִיחַ אֶת הָרֵיחַ הַנִּפְלָא.

הַגִּלּוּי בַּחֲדַר הָאֹכֶל

הַמְּשָׁרֵת נִכְנַס לַחֶדֶר כְּשֶׁהוּא נוֹשֵׂא מַגָּשׁ עֲנָק עִם כַּלָּחֵי אוֹכֶל רוֹתְחִים וּמַעֲלֵי אֵדִים.

רַבִּי חֲנִינָא בַּר חָמָא חִיֵּךְ לָאוֹרֵחַ, וְהֵם נָטְלוּ יָדַיִם. אֲבָל לִפְנֵי שֶׁרַבִּי חֲנִינָא הֵחֵל בַּסְּעוּדָה, עֵינָיו הַחֲכָמוֹת שֶׁל הַצַּדִּיק בָּחֲנוּ אֶת הַמַּגָּשׁ.

הוּא קָרָא לַמְּשָׁרֵת הַצִּדָּה וְלָחַשׁ לוֹ: "אֱמֹר לִי בְּבַקָּשָׁה, אֵיךְ הִצְלַחְתָּ לְהָכִין תַּבְשִׁיל שֶׁכָּזֶה כָּל כָּךְ מַהֵר? הֲרֵי בַּבֹּקֶר רָאִיתִי שֶׁעֲצֵי הַהַסָּקָה שֶׁלָּנוּ הָיוּ רְטֻבִּים וְלַחִים!".

הַמְּשָׁרֵת אָמַר אֶת הָאֱמֶת, עִם חִיּוּךְ גֵּאֶה: "רַבִּי, מָצָאתִי פִּתְרוֹן נִפְלָא! לָקַחְתִּי כַּמָּה עֵצִים בּוֹדְדִים מֵהָעֲגָלָה שֶׁל הָאוֹרֵחַ. כָּךְ יָכֹלְתִּי לְבַשֵּׁל לוֹ מִיָּד. אֲנִי בָּטוּחַ שֶׁהוּא יִשְׂמַח לָדַעַת שֶׁזֶּה נַעֲשָׂה עֲבוּרוֹ!".

פִּתְאוֹם, הַחִיּוּךְ נֶעֱלַם מִפָּנָיו שֶׁל רַבִּי חֲנִינָא. עֵינָיו נִפְעֲרוּ בְּתַדְהֵמָה וְזַעֲזוּעַ עָמֹק פָּקַד אוֹתוֹ. אָדָם רָגִיל אוּלַי הָיָה מַסְכִּים, אֲבָל צַדִּיק יְסוֹד עוֹלָם יוֹדֵעַ אֶת הָאֱמֶת הַנְּקִיָּה שֶׁל תּוֹרַת ה'.

"מָה עָשִׂיתָ?!" אָמַר הָרַב בְּקוֹל רוֹעֵד מִיִּרְאַת שָׁמַיִם. " וְכִי מַאֲכִילִים אָנוּ אֶת הָאוֹרֵחַ מִשֶּׁלּוֹ?! הַאִם כָּךְ נִרְאֵית מִצְוַת הַכְנָסַת אוֹרְחִים? לָקַחְתָּ רְכוּשׁ שֶׁל יְהוּדִי בְּלִי שֶׁשָּׁאַלְתָּ אוֹתוֹ, אֲפִלּוּ אִם זֶה בִּשְׁבִילוֹ! הַכְנָסַת אוֹרְחִים אֲמִתִּית מַשְׁמָעוּתָהּ שֶׁאֲנִי, הַמְּאָרֵחַ, אַפְסִיד מִמָּמוֹנִי וְאֶתֵּן מִשֶּׁלִּי עֲבוּר הַשֵּׁנִי. לֹא שֶׁאֶעֱשֶׂה לוֹ טוֹבָה מֵהַכִּיס שֶׁלּוֹ עַצְמוֹ!".

שְׂכַר הַמִּצְוָה

הַמְּשָׁרֵת הוֹרִיד אֶת רֹאשׁוֹ בְּבוּשָׁה וְהֵבִין שֶׁטָּעָה טָעוּת מָרָה. "מָה נַעֲשֶׂה עַכְשָׁו, כְּבוֹד הָרַב?" שָׁאַל בְּלַחַשׁ.

רַבִּי חֲנִינָא נִגַּשׁ לַקֻּפָּה הָאִישִׁית שֶׁלּוֹ. הוּא הוֹצִיא מִשָּׁם מַטְבְּעוֹת רַבִּים. הוּא פָּנָה אֶל הָאוֹרֵחַ, שֶׁכְּבָר הִתְחִיל לֶאֱכֹל וְלֵהָנוֹת מֵהַטַּעַם הֶעָרֵב שֶׁל הַמָּזוֹן.

"אָחִי הַיָּקָר," אָמַר לוֹ רַבִּי חֲנִינָא בַּעֲדִינוּת. "לְצַעֲרִי בַּמִּטְבָּח הִשְׁתַּמְּשׁוּ לְלֹא רְשׁוּת בְּכַמָּה עֵצִים שֶׁלְּךָ כְּדֵי לְהַסִּיק אֶת הַתַּנּוּר. אֲנִי מְבַקֵּשׁ אֶת סְלִיחָתְךָ. קַח נָא אֶת הַכֶּסֶף הַזֶּה," הָרַב הִנִּיחַ בְּיָדוֹ חֹפֶן נָדִיב שֶׁל מַטְבְּעוֹת שֶׁהָיוּ שָׁוִים פִּי כַּמָּה וְכַמָּה מִיֹּפִי הָעֵצִים הַפְּשׁוּטִים. "אֲנִי מְשַׁלֵּם לְךָ עַל הָעֵצִים תַּשְׁלוּם מָלֵא. רַק כָּךְ, הַסְּעוּדָה שֶׁאַתָּה אוֹכֵל עַכְשָׁו, תִּהְיֶה בֶּאֱמֶת מַתָּנָה שֶׁלִּי אֵלֶיךָ מִכָּל הַלֵּב!".

הָאוֹרֵחַ הָיָה הָמוּם. הוּא מֵעוֹלָם לֹא רָאָה אָדָם שֶׁכָּל כָּךְ יָשָׁר, שֶׁמַּקְפִּיד אֲפִלּוּ עַל חֲתִיכַת עֵץ קְטַנָּה בְּשֵׁם מִצְוַת "וְאָהַבְתָּ לְרֵעֲךָ כָּמוֹךָ". אוֹתוֹ לַיְלָה יָשַׁב הָאוֹרֵחַ וְהִקְשִׁיב לְדִבְרֵי הַתּוֹרָה שֶׁל רַבִּי חֲנִינָא, וְהֵבִין שֶׁהוּא שׁוֹהֶה בְּבֵיתוֹ שֶׁל אָדָם גָּדוֹל בֶּאֱמֶת – מָקוֹם בּוֹ כְּבוֹד הַחָבֵר שָׁוֶה יוֹתֵר מִזָּהָב.

מקורות: תלמוד ירושלמי, מסכת בבא בתרא, פרק ב', הלכה י"א (ע"פ חלוקות מסוימות). 1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “דַּמְיְנוּ שֶׁאַתֶּם נִמְצָאִים בָּעִיר צִפּוֹרִי, לִפְנֵי כְּאַלְפַּיִם שָׁנָה.” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “כְּשֶׁרָאָה אֶת הַדֶּלֶת הַפְּתוּחָה שֶׁל רַבִּי חֲנִינָא, הָרְגִּישׁ תִּקְוָה גְּדוֹלָה מְמַלֵּאת אֶת לִבּוֹ.”?
ג. כיצד הגיב ר' חנינא בר חמא למצב המתואר: “הַמְּשָׁרֵת חָתַךְ יְרָקוֹת, שָׂם נֶתַח דָּג שָׁמֵן בְּתוֹךְ סִיר כֶּרֶס, וְנִגַּשׁ לַתַּנּוּר.”?
ד. מה השתנה בין “יְלָדִים, הַמְּשָׁרֵת חָשַׁב שֶׁהוּא עוֹשֶׂה מַעֲשֶׂה חָכָם, אֲבָל הוּא לֹא הֵבִין שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא בּוֹחֵן…” לבין “הֲרֵי בַּבֹּקֶר רָאִיתִי שֶׁעֲצֵי הַהַסָּקָה שֶׁלָּנוּ הָיוּ רְטֻבִּים וְלַחִים!".”?
ה. כיצד הגיע הסיפור אל האירוע: “לְצַעֲרִי בַּמִּטְבָּח הִשְׁתַּמְּשׁוּ לְלֹא רְשׁוּת בְּכַמָּה עֵצִים שֶׁלְּךָ כְּדֵי לְהַסִּיק אֶת הַתַּנּוּר.”?

3. מה היה ראשון?

כתבו ליד כל אירוע מספר מ־1 עד 5 לפי סדר התרחשותו בסיפור.

□ הָאוֹרֵחַ בֶּטַח לֹא יִכְעַס, הוּא אֲפִלּוּ יִשְׂמַח שֶׁהָאוֹכֶל שֶׁלּוֹ יִהְיֶה מוּכָן מַהֵר!".
□ דַּמְיְנוּ שֶׁאַתֶּם נִמְצָאִים בָּעִיר צִפּוֹרִי, לִפְנֵי כְּאַלְפַּיִם שָׁנָה.
□ אוֹתוֹ לַיְלָה יָשַׁב הָאוֹרֵחַ וְהִקְשִׁיב לְדִבְרֵי הַתּוֹרָה שֶׁל רַבִּי חֲנִינָא, וְהֵבִין שֶׁהוּא שׁוֹהֶה בְּבֵיתוֹ שֶׁל אָדָם גָּדוֹל בֶּאֱמֶת –…
□ הִנֵּה קְעָרַת מַיִם חַמִּים לִרְחֹץ אֶת רַגְלֶיךָ הַכּוֹאֲבוֹת.
□ הַמְּשָׁרֵת אָמַר אֶת הָאֱמֶת, עִם חִיּוּךְ גֵּאֶה: "רַבִּי, מָצָאתִי פִּתְרוֹן נִפְלָא!

לפני ואחרי

מה קרה לפני: “הָאוֹרֵחַ בֶּטַח לֹא יִכְעַס, הוּא אֲפִלּוּ יִשְׂמַח שֶׁהָאוֹכֶל שֶׁלּוֹ יִהְיֶה מוּכָן מַהֵר!".”?
מה קרה אחרי: “הַמְּשָׁרֵת אָמַר אֶת הָאֱמֶת, עִם חִיּוּךְ גֵּאֶה: "רַבִּי, מָצָאתִי פִּתְרוֹן נִפְלָא!”?

סיבה ותוצאה מתוך המעשה

האירוע הנתוןמה גרם לו?מה הייתה תוצאתו?
שֵׁב וְנַח כָּאן, וַאֲנַחְנוּ נָכִין לְךָ מִיָּד סְעוּדָה חַמָּה וּמְשִׁיבַת נֶפֶשׁ".
הָאוֹרֵחַ כִּמְעַט הִתְעַלֵּף מֵרֹב צִפִּיָּה כְּשֶׁהֵרִיחַ אֶת הָרֵיחַ הַנִּפְלָא.

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “כיבוד אורח — רק משל המארח” על ר' חנינא בר חמא? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “פִּתְאוֹם נָפְלָה מַבָּטוֹ עַל הָעֲגָלָה שֶׁל הָאוֹרֵחַ, שֶׁחָנְתָה בֶּחָצֵר.”.
ג. אילו ר' חנינא בר חמא לא היה פועל כפי שמתואר ב־“הֲרֵי בַּבֹּקֶר רָאִיתִי שֶׁעֲצֵי הַהַסָּקָה שֶׁלָּנוּ הָיוּ רְטֻבִּים וְלַחִים!".” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “לְצַעֲרִי בַּמִּטְבָּח הִשְׁתַּמְּשׁוּ לְלֹא רְשׁוּת בְּכַמָּה עֵצִים שֶׁלְּךָ כְּדֵי לְהַסִּיק אֶת הַתַּנּוּר.” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. דַּמְיְנוּ שֶׁאַתֶּם נִמְצָאִים בָּעִיר צִפּוֹרִי, לִפְנֵי…
2. רַבִּי חֲנִינָא בִּכְבוֹדוֹ וּבְעַצְמוֹ קָם מִשֻּׁלְחַן לִמּוּדוֹ…
3. נִשְׁאֲרוּ בּוֹ רַק כַּמָּה עֵצִים לַחִים שֶׁהֶעֱלוּ הַרְבֵּה עָשָׁן…
4. פִּיץ פָּאץ' נִשְׁמַע פִּצְפּוּץ הָעֵצִים הַיְבֵשִׁים, וְתוֹךְ זְמַן…
5. אָדָם רָגִיל אוּלַי הָיָה מַסְכִּים, אֲבָל צַדִּיק יְסוֹד עוֹלָם…
6. אוֹתוֹ לַיְלָה יָשַׁב הָאוֹרֵחַ וְהִקְשִׁיב לְדִבְרֵי הַתּוֹרָה שֶׁל…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “הַמְּשָׁרֵת חָתַךְ יְרָקוֹת, שָׂם נֶתַח דָּג שָׁמֵן בְּתוֹךְ סִיר כֶּרֶס, וְנִגַּשׁ לַתַּנּוּר.”.
מכתב: כתבו לר' חנינא בר חמא בעקבות “הָאוֹרֵחַ כִּמְעַט הִתְעַלֵּף מֵרֹב צִפִּיָּה כְּשֶׁהֵרִיחַ אֶת הָרֵיחַ הַנִּפְלָא.”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “לָקַחְתָּ רְכוּשׁ שֶׁל יְהוּדִי בְּלִי שֶׁשָּׁאַלְתָּ אוֹתוֹ, אֲפִלּוּ אִם זֶה בִּשְׁבִילוֹ!”.
דיווח: כתבו ידיעה קצרה שכותרתה “כיבוד אורח — רק משל המארח”.

המחזה

רגע מן הסיפורהדמויותמה יאמרו או יעשו?
שֵׁב וְנַח כָּאן, וַאֲנַחְנוּ נָכִין לְךָ מִיָּד סְעוּדָה חַמָּה וּמְשִׁיבַת נֶפֶשׁ".
הֲרֵי בַּבֹּקֶר רָאִיתִי שֶׁעֲצֵי הַהַסָּקָה שֶׁלָּנוּ הָיוּ רְטֻבִּים וְלַחִים!".

7. חיבור לחיים מתוך המעשה

א. מה אפשר ללמוד מן הדרך שבה ר' חנינא בר חמא פעל באירוע “יְלָדִים, הַמְּשָׁרֵת חָשַׁב שֶׁהוּא עוֹשֶׂה מַעֲשֶׂה חָכָם, אֲבָל הוּא לֹא הֵבִין שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא בּוֹחֵן אוֹתָם…”?
