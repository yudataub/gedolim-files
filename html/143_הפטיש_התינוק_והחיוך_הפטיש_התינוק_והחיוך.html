הפטיש, התינוק, והחיוך

דף עבודה בעקבות הפטיש, התינוק, והחיוך
שם: כיתה:

המעשה — הסיפור המלא

הסיפור מנוקד במלואו כנדרש.

🔹 שָׁלָב 1: בֵּרוּר מַקְדִּים וְנִתּוּחַ (לַמּוֹרֶה/לַמְסַפֵּר)

1. הִדּוּר מִצְוָה וְאַהֲבָתָהּ: לְחַפֵּשׂ תָּמִיד אֵיךְ לְקַיֵּם אֶת הַמִּצְוָה בַּצּוּרָה הֲכִי שְׁלֵמָה וְיָפָה, וְלַחְשֹׁב עַל הַמִּצְווֹת (וְעַל הַשַּׁבָּת) כָּל הַשָּׁבוּעַ.

2. מְאוֹר פָּנִים: הָאָדָם הַקַּפְּדָן בְּיוֹתֵר בַּהֲלָכָה, הוּא זֶה שֶׁטָּבַע אֶת הַכְּלָל: "הֱוֵי מְקַבֵּל אֶת כָּל הָאָדָם בְּסֵבֶר פָּנִים יָפוֹת!" – קַפְּדָנוּת לַה', אַךְ חִיּוּךְ רָחָב לַבְּרִיּוֹת.

🔹 שָׁלָב 2: הַסִּפּוּר הַמָּלֵא – "הַפַּטִּישׁ, הַתִּינוֹק, וְהַחִיּוּךְ"

פְּתִיחָה: הָעֵינַיִם הַמְחַפְּשׂוֹת יְלָדִים, אֲנַחְנוּ מַכִּירִים אֶת הִלֵּל הַזָּקֵן הֶעָנָיו, אֲבָל מִי הָיָה הֶחָבֵר הֲכִי טוֹב שֶׁלּוֹ? הֶחָבֵר שֶׁיָּשַׁב לְצִדּוֹ וְהִנְהִיג יַחַד אִתּוֹ אֶת כָּל עַם יִשְׂרָאֵל? זֶה הָיָה שַׁמַּאי הַזָּקֵן! אַב בֵּית הַדִּין שֶׁל הַסַּנְהֶדְרִין. הַרְבֵּה אֲנָשִׁים חָשְׁבוּ שֶׁשַׁמַּאי הוּא אָדָם קָשֶׁה מְאֹד. לָמָּה? כִּי כְּשֶׁהָיָה מְדֻבָּר בִּכְבוֹדוֹ שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא, בְּדִינֵי הַתּוֹרָה הַקְּדוֹשָׁה – שַׁמַּאי לֹא וִתֵּר עַל מִילִימֶטֶר! הַתּוֹרָה הָיְתָה אֶצְלוֹ מְדֻיֶּקֶת וּקְדוֹשָׁה כְּאֵשׁ.

אֲבָל אִם הֱיִיתֶם רוֹאִים אֶת הָעֵינַיִם שֶׁל שַׁמַּאי, הֱיִיתֶם מְבִינִים שֶׁבְּתוֹךְ לִבּוֹ בָּעֲרָה אֵשׁ שֶׁל אַהֲבָה גְּדוֹלָה! רְאוּ לְדֻגְמָה מָה קָרָה בְּכָל יוֹם רִאשׁוֹן בַּבֹּקֶר. יוֹם רִאשׁוֹן זֶה תְּחִלַּת הַשָּׁבוּעַ, נָכוֹן? הַשַּׁבָּת רַק נִגְמְרָה לִפְנֵי רֶגַע. אַךְ כַּאֲשֶׁר שַׁמַּאי הָיָה יוֹצֵא לַשּׁוּק בְּיוֹם רִאשׁוֹן, וְהָיָה מֵרִיחַ אֶת רֵיחוֹת הַתַּבְלִינִים וְשׁוֹמֵעַ אֶת קוֹלוֹת הַסּוֹחֲרִים, הוּא הָיָה מְחַפֵּשׂ רַק דָּבָר אֶחָד. וְהִנֵּה, הוּא רוֹאֶה עֵגֶל יָפֶה, בָּרִיא וְשָׁמֵן מְאֹד. מִיָּד הָיְתָה פָּנָיו שֶׁל שַׁמַּאי זוֹרְחוֹת. הוּא הָיָה מְשַׁלֵּם אֶת הַכֶּסֶף, קוֹנֶה אֶת הַבְּהֵמָה הַמְשֻׁבַּחַת, מַצְבִּיעַ עָלֶיהָ בְּאֶצְבָּעוֹ וְאוֹמֵר בְּקוֹל שָׂמֵחַ: "זֶה – לִכְבוֹד שַׁבָּת!". כָּךְ הָיָה עוֹשֶׂה בְּיוֹם רִאשׁוֹן! מֵכִין לְשַׁבָּת שֶׁל עוֹד שִׁשָּׁה יָמִים!

וְאִם בְּיוֹם שְׁלִישִׁי, כְּשֶׁעָבַר בַּשּׁוּק, רָאָה פִּתְאוֹם בְּהֵמָה יָפָה עוֹד יוֹתֵר מֵהָרִאשׁוֹנָה? הָיָה שַׁמַּאי קוֹנֶה אֶת הַבְּהֵמָה הַחֲדָשָׁה שֶׁתִּהְיֶה לְשַׁבָּת קֹדֶשׁ, וְאֶת הָרִאשׁוֹנָה שֶׁנִּשְׁאֲרָה הָיָה אוֹכֵל בְּיוֹם חֹל. לָמָּה? כִּי הַלֵּב שֶׁלּוֹ כָּל כָּךְ אָהַב אֶת יְמֵי הַקֹּדֶשׁ שֶׁל ה', עַד שֶׁ"כָּל יָמָיו אָכַל לִכְבוֹד שַׁבָּת"! הַשַּׁבָּת לֹא יָצְאָה לוֹ מֵהָרֹאשׁ אֲפִלּוּ לְדַקָּה.

הַתִּינוֹק הַקָּטָן בֶּחָג הַסֻּכּוֹת הַהִתְלַהֲבוּת שֶׁל שַׁמַּאי הַזָּקֵן הָיְתָה כָּל כָּךְ יָפָה, וְהִיא בָּלְטָה בִּמְיֻחָד בְּחַג הַסֻּכּוֹת. שָׁנָה אַחַת, בְּחַג הַסֻּכּוֹת מַמָּשׁ, קָרָה דָּבָר מְשַׂמֵּחַ בְּמִשְׁפַּחְתּוֹ שֶׁל שַׁמַּאי. כַּלָּתוֹ (אֵשֶׁת בְּנוֹ) יָלְדָה תִּינוֹק! אֵיזוֹ שִׂמְחָה! הַיּוֹלֶדֶת שָׁכְבָה בַּמִּטָּה שֶׁלָּהּ בְּתוֹךְ הַבַּיִת, בַּחֶדֶר שֶׁלָּהּ. וְלִידָהּ, בַּעֲרִיסָה קְטַנְטַנָּה, שָׁכַב הַנֶּכֶד הֶחָדָשׁ שֶׁל שַׁמַּאי הַזָּקֵן, עָטוּף בִּשְׂמִיכָה דַּקָּה. יְלָדִים, אֲנַחְנוּ לוֹמְדִים בַּתּוֹרָה שֶׁתִּינוֹק קָטָן שֶׁרַק נוֹלַד – פָּטוּר מֵהַמִּצְווֹת. הוּא לֹא מֵבִין כְּלוּם, הוּא לֹא חַיָּב לִישֹׁן בַּסֻּכָּה. אַף אֶחָד לֹא יְבַקֵּשׁ לְהוֹצִיא אִמָּא חַלָּשָׁה אוֹ תִּינוֹק קָטָן הַחוּצָה לַסֻּכָּה בַּלַּיְלָה הַקָּר.

אֲבָל שַׁמַּאי הַזָּקֵן נִכְנַס לַחֶדֶר. הוּא הִבִּיט בַּנֶּכֶד הַקָּטָן הַיָּשֵׁן שְׁנַת יְשָׁרִים. וְאָז הוּא הִבִּיט לַמַּעְלָה, אֶל הַתִּקְרָה (הַ"מַּעֲזִיבָה") שֶׁל הַחֶדֶר שֶׁהָיְתָה מְצֻפָּה בְּטִיחַ וְאָבָק. "אֵיךְ אֶפְשָׁר..." לָחַשׁ שַׁמַּאי בְּחִיּוּךְ רָחָב, "אֵיךְ אֶפְשָׁר שֶׁנֶּכֶד יְהוּדִי שֶׁרַק הִגִּיעַ לָעוֹלָם, יִישַׁן בְּיָמִים אֵלּוּ מִתַּחַת לְתִקְרַת אֶבֶן? אֲנִי כָּל כָּךְ אוֹהֵב אֶת הַמִּצְווֹת שֶׁל ה', אֲנִי רוֹצֶה שֶׁהַתִּינוֹק הַזֶּה יִנְשֹׁם אֶת הָאֲוִיר שֶׁל הַמִּצְוָה וְיִישַׁן בְּתוֹךְ קְדֻשָּׁה!".

שׁוֹבְרִים אֶת הַתִּקְרָה! וְאָז – עָשָׂה שַׁמַּאי מַעֲשֶׂה פֶּלֶא! הוּא לֹא הוֹצִיא אֶת הַתִּינוֹק הַחוּצָה לַקֹּר. בִּמְקוֹם זֶה – הוּא עָלָה עַל גַּג הַבַּיִת עִם כֵּלִים כְּבֵדִים, וְהִתְחִיל לִדְפֹּק עַל הַגַּג מַמָּשׁ בַּמָּקוֹם שֶׁמֵּעַל הַמִּטָּה שֶׁל הַיּוֹלֶדֶת וְהַתִּינוֹק! "בּוּם! טְרָאח!" אַבְנֵי הַתִּקְרָה נִשְׁבְּרוּ. אֲבָק קָטָן הִתְפַּזֵּר בָּאֲוִיר. הַשְּׁכֵנִים בַּחוּץ רָאוּ אֶת אַב בֵּית הַדִּין הַזָּקֵן שׁוֹבֵר לְעַצְמוֹ אֶת הַבַּיִת! שַׁמַּאי הוֹצִיא לְגַמְרֵי אֶת הַכִּסּוּי שֶׁל הַגַּג מֵעַל מִטַּת הַתִּינוֹק, עַד שֶׁנִּהְיָה שָׁם חֹר עֲנָק שֶׁדַּרְכּוֹ רָאוּ אֶת הַשָּׁמַיִם. וּמָה הוּא עָשָׂה עַכְשָׁו? הוּא הֵבִיא עֲנָפִים יְרֻקִּים, עֲלֵי אֵיקָלִיפְּטוּס וְתָמָר שֶׁרֵיחָם נָעִים וּמָתוֹק, וְהִנִּיחַ אוֹתָם בִּזְהִירוּת מֵעַל הַחֹר. הוּא בָּנָה סֻכָּה פְּרָטִית קְטַנָּה מַמָּשׁ בְּתוֹךְ תִּקְרַת הַחֶדֶר, רַק מֵעַל הַמִּטָּה שֶׁל הַתִּינוֹק הֶחָדָשׁ! וְכָךְ, הַתִּינוֹק יָשַׁן בְּמָקוֹם חַם, בְּחֶדֶר סָגוּר, אֲבָל מֵעַל רֹאשׁוֹ הָיָה סְכָךְ יָרֹק וְכָשֵׁר לִמְהַדְּרִין. רַבִּי שַׁמַּאי הַזָּקֵן הִבִּיט בַּסְּכָךְ וְחִיֵּךְ לְעַצְמוֹ בְּשִׂמְחָה עֲצוּמָה. זֶה מָה שֶׁנִּקְרָא "קַנָּאוּת לַמִּצְווֹת" – אִכְפַּת לוֹ מֵהַמִּצְוָה כָּל כָּךְ מַהֵר וְכָל כָּךְ הַרְבֵּה, אֲפִלּוּ כְּשֶׁלֹּא חַיָּבִים!

(הֶעָרָה לַמּוֹרֶה/מְסַפֵּר: מַמָּשׁ כְּמוֹ שֶׁבְּיוֹם הַכִּפּוּרִים לֹא רָצָה שַׁמַּאי לְהַאֲכִיל אֶת בְּנוֹ הַקָּטָן, מֵרֹב שֶׁהָיְתָה לוֹ יִרְאַת שָׁמַיִם רְצִינִית עַד שֶׁגָּזְרוּ עָלָיו חֲכָמִים לְהַאֲכִילוֹ. הַכֹּל מִתּוֹךְ רְצִינוּת עֲצוּמָה לִדְבַר ה'!).

הַסּוֹד שֶׁל שַׁמַּאי וְעַכְשָׁו יְלָדִים, אַתֶּם בֶּטַח שׁוֹאֲלִים אֶת עַצְמְכֶם: אָדָם שֶׁהוּא כָּל כָּךְ דַּקְדְּקָן, שֶׁכָּל דָּבָר שֶׁל הַתּוֹרָה אֶצְלוֹ זֶה הֲכִי חָשׁוּב וַהֲכִי מְדֻיָּק... אֵיךְ הוּא הִתְנַהֵג אֶל אֲנָשִׁים בָּרְחוֹב? הַאִם הוּא עָשָׂה לָהֶם פַּרְצוּף חוֹמֵץ וְרָצִינִי כָּל הַזְּמַן? אֲזַי תִּשְׁמְעוּ דָּבָר נִפְלָא. דַּוְקָא שַׁמַּאי הַזָּקֵן, הָרַב הֲכִי קַפְּדָן שֶׁלֹּא וִתֵּר עַל כְּבוֹד הַתּוֹרָה, הוּא זֶה שֶׁלִּמֵּד אֶת הַכְּלָל הֲכִי נָעִים וְשָׂמֵחַ בְּמַסֶּכֶת אָבוֹת. כָּךְ הָיָה שַׁמַּאי מוֹרֶה לְתַלְמִידָיו: "עֲשֵׂה תּוֹרָתְךָ קֶבַע, אֱמֹר מְעַט וַעֲשֵׂה הַרְבֵּה... וֶהֱוֵי מְקַבֵּל אֶת כָּל הָאָדָם בְּסֵבֶר פָּנִים יָפוֹת!". "סֵבֶר פָּנִים יָפוֹת" זֶה אוֹמֵר: חִיּוּךְ רָחָב! פָּנִים מְאִירוֹת! לֹא מְשַׁנֶּה מִי נִגָּשׁ אֵלֶיךָ בָּרְחוֹב – תַּקְבִּיל אֶת פָּנָיו בְּחִיּוּךְ מָתוֹק וּבְעַיִן טוֹבָה.

2. וְהִנֵּה, הוּא רוֹאֶה עֵגֶל יָפֶה, בָּרִיא וְשָׁמֵן מְאֹד.
3. וְלִידָהּ, בַּעֲרִיסָה קְטַנְטַנָּה, שָׁכַב הַנֶּכֶד הֶחָדָשׁ שֶׁל…
4. שַׁמַּאי הוֹצִיא לְגַמְרֵי אֶת הַכִּסּוּי שֶׁל הַגַּג מֵעַל מִטַּת…
5. הַסּוֹד שֶׁל שַׁמַּאי וְעַכְשָׁו יְלָדִים, אַתֶּם בֶּטַח שׁוֹאֲלִים אֶת…
6. הַיּוֹם נִפְגֹּשׁ תַּנָּא כָּל כָּךְ גָּדוֹל, שֶׁמֵּרֹב שֶׁאָהַב אֶת…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “הַיּוֹלֶדֶת שָׁכְבָה בַּמִּטָּה שֶׁלָּהּ בְּתוֹךְ הַבַּיִת, בַּחֶדֶר שֶׁלָּהּ.”.
מכתב: כתבו להפטיש, התינוק, והחיוך בעקבות “הוּא הֵבִיא עֲנָפִים יְרֻקִּים, עֲלֵי אֵיקָלִיפְּטוּס וְתָמָר שֶׁרֵיחָם נָעִים וּמָתוֹק, וְהִנִּיחַ אוֹתָם בִּזְהִירוּת מֵעַל…”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “דַּוְקָא שַׁמַּאי הַזָּקֵן, הָרַב הֲכִי קַפְּדָן שֶׁלֹּא וִתֵּר עַל כְּבוֹד הַתּוֹרָה, הוּא זֶה שֶׁלִּמֵּד אֶת הַכְּלָל הֲכִי…”.
דיווח: כתבו ידיעה קצרה שכותרתה “הפטיש, התינוק, והחיוך”.

המחזה

רגע מן הסיפורהדמויותמה יאמרו או יעשו?
הָיָה שַׁמַּאי קוֹנֶה אֶת הַבְּהֵמָה הַחֲדָשָׁה שֶׁתִּהְיֶה לְשַׁבָּת קֹדֶשׁ, וְאֶת הָרִאשׁוֹנָה…
זֶה מָה שֶׁנִּקְרָא "קַנָּאוּת לַמִּצְווֹת" – אִכְפַּת לוֹ מֵהַמִּצְוָה כָּל כָּךְ מַהֵר וְכָל…

7. חיבור לחיים מתוך המעשה

א. מה אפשר ללמוד מן הדרך שבה הפטיש, התינוק, והחיוך פעל באירוע “בִּמְקוֹם זֶה – הוּא עָלָה עַל גַּג הַבַּיִת עִם כֵּלִים כְּבֵדִים, וְהִתְחִיל לִדְפֹּק עַל הַגַּג מַמָּשׁ בַּמָּקוֹם שֶׁמֵּעַל…”?
ב. תארו מצב מחיי הכיתה שבו אפשר ליישם את המסר של “הפטיש, התינוק, והחיוך”.
