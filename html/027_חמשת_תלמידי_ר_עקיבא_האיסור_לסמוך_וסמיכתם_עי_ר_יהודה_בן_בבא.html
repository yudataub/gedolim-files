האיסור לסמוך וסמיכתם ע"י ר' יהודה בן בבא

דף עבודה בעקבות חמשת תלמידי ר' עקיבא
שם: כיתה:

המעשה — הסיפור המלא

להלן עיבוד חינוכי, רגשי ומרתק לסיפור ההיסטורי והמכונן על רבי יהודה בן בבא והסמכת חמשת תלמידי רבי עקיבא . הסיפור נכתב בקפידה לפי הנחיותיך, מותאם לבנים בכיתות ב'-ד' בחינוך הדתי-לאומי, ומדגיש את מסירות הנפש למען נצח ישראל והעברת שרשרת התורה.

הסיפור מנוקד במלואו כנדרש.

פְּתִיחָה: דְּמָמָה כּוֹאֶבֶת בְּבֵית הַמִּדְרָשׁ

יְלָדִים, נַסּוּ לַעֲצֹם עֵינַיִם וּלְדַמְיֵן אֶת קוֹל הַתּוֹרָה בְּבֵית מִדְרָשׁ עֲנָקִי. אֲלָפִים שֶׁל תַּלְמִידִים יּוֹשְׁבִים וְלוֹמְדִים, מִתְוַכְּחִים, שָׁרִים אֶת הַהֲלָכוֹת. רַעַשׁ מָתוֹק מְמַלֵּא אֶת הָאֲוִיר.

כָּךְ בְּדִיּוּק הָיָה בִּישִׁיבָתוֹ הַגְּדוֹלָה שֶׁל מָרָן רַבִּי עֲקִיבָא. אֲבָל פִּתְאוֹם... שֶׁקֶט.

דְּמָמָה מַפְחִידָה יָרְדָה עַל אֶרֶץ יִשְׂרָאֵל. "שְׁנֵים עָשָׂר אֶלֶף זוּגוֹת תַּלְמִידִים" הָיוּ לוֹ לְרַבִּי עֲקִיבָא, וְכֻלָּם נִפְטְרוּ לְעוֹלָמָם בְּמַגֵּפָה נוֹרָאָה עַד לַ"ג בָּעֹמֶר. וְאִם זֶה לֹא מַסְפִּיק, הָרוֹמָאִים הָאַכְזָרִיִּים כָּלְאוּ אֶת רַבִּי עֲקִיבָא בְּבֵית הָאֲסוּרִים!

רֵיחַ שֶׁל עָשָׁן וְאֵפֶר עָלָה מֵהֶעָרִים הַחֲרֵבוֹת. אֲנָשִׁים פָּחֲדוּ לָצֵאת מֵהַבַּיִת. גְּזֵרָה חֲדָשָׁה וְנוֹרָאָה יָצְאָה מֵאֵת הַקֵּיסָר הָרוֹמָאִי – גְּזֵרָה שֶׁמַּטְּרָתָהּ לְנַתֵּק, חָלִילָה, אֶת קֶשֶׁר הַתּוֹרָה שֶׁל עַם יִשְׂרָאֵל: "כָּל רַב שֶׁיִּסְמֹךְ (יַעֲנִיק תְּעוּדַת רַבָּנוּת) לְתַלְמִיד חָדָשׁ – יֵהָרֵג! וְכָל תַּלְמִיד שֶׁיֻּסְמַךְ – יֵהָרֵג! וְהָעִיר שֶׁבָּהּ זֶה יִקְרֶה – תֵּחָרֵב וְתֵהָרֵס מִיָּד!"

הָרוֹמָאִים הָיוּ חֲכָמִים לְהָרַע. הֵם יָדְעוּ: רַב זָקֵן שֶׁנִּפְטַר וְלֹא הִסְמִיךְ רַב צָעִיר שֶׁיַּמְשִׁיךְ אֶת דַּרְכּוֹ וְיִפְסֹק הֲלָכָה – מְבִיאוֹ לְכָךְ שֶׁהַתּוֹרָה כְּאִלּוּ נִגְמֶרֶת, כִּי אֵין מִי שֶׁיַּעֲבִיר אוֹתָהּ לַדּוֹר הַבָּא. מִי יַגִּיד מָה כָּשֵׁר וּמָה טָרֵף? מִי יַגִּיד מָתַי שַׁבָּת?

הָעוֹלָם הָיָה שָׁמֵם וַעֲצוּב. נִדְמֶה הָיָה שֶׁהַחֹשֶׁךְ נִצֵּחַ אֶת הָאוֹר.

הַהַחְלָטָה שֶׁל חֲסִיד עֶלְיוֹן

אֲבָל הָרוֹמָאִים שָׁכְחוּ עִם מִי הֵם מִתְעַסְּקִים. הֵם שָׁכְחוּ שֶׁהַתּוֹרָה הִיא חַיֵּינוּ.

בְּאוֹתָם יָמִים, קָם סָבָא קָדוֹשׁ וְטָהוֹר, שֶׁשְּׁמוֹ הָיָה רַבִּי יְהוּדָה בֶּן בָּבָא . הוּא הָיָה צַדִּיק כָּל כָּךְ גָּדוֹל, שֶׁמֵּעוֹלָם לֹא עָשָׂה עֲבֵרָה.

רַבִּי יְהוּדָה בֶּן בָּבָא קָרָא אֵלָיו בַּסֵּתֶר חֲמִשָּׁה תַּלְמִידִים צְעִירִים. אֵלּוּ הָיוּ הַתַּלְמִידִים הַיְחִידִים שֶׁשָּׂרְדוּ מִישִׁיבָתוֹ שֶׁל רַבִּי עֲקִיבָא. יְלָדִים, אֲנַחְנוּ לוֹמְדִים אֶת הַשֵּׁמוֹת שֶׁלָּהֶם בְּכָל יוֹם בַּמִּשְׁנָה: רַבִּי מֵאִיר, וְרַבִּי יְהוּדָה, וְרַבִּי שִׁמְעוֹן (בַּר יוֹחַאי), וְרַבִּי יוֹסֵי, וְרַבִּי אֶלְעָזָר בֶּן שַׁמּוּעַ.

בְּאִישׁוֹן לַיְלָה, כְּשֶׁרַק אוֹר חִוֵּר שֶׁל יָרֵחַ בּוֹקֵעַ מִבַּעַד לָעֲנָנִים, נֶאֶסְפוּ הַחֲמִשָּׁה סְבִיב הַזָּקֵן הַקָּדוֹשׁ. רַעַד עָבַר בְּלִבָּם.

"בָּנַי," לָחַשׁ רַבִּי יְהוּדָה בֶּן בָּבָא בְּקוֹל שֶׁקֶט, "הָרְשָׁעִים מְנַסִּים לְכַבּוֹת אֶת אֵשׁ הַתּוֹרָה. אֲבָל לֹא נִתֵּן לָהֶם! אֲנִי הוֹלֵךְ לִסְמֹךְ אֶתְכֶם (לַעֲשׂוֹת אֶתְכֶם לְרַבָּנִים שֶׁפּוֹסְקִים הֲלָכָה). אַתֶּם תִּהְיוּ אֵלּוּ שֶׁתַּעֲבִירוּ אֶת מָסֹרֶת הַתּוֹרָה לַדּוֹרוֹת הַבָּאִים!"

"אֲבָל רַבִּי..." גִּמְגֵּם אֶחָד הַתַּלְמִידִים, "הַגְּזֵרָה הַנּוֹרָאָה... הָרוֹמָאִים יַחֲרִיבוּ אֶת הָעִיר שֶׁנִּהְיֶה בָּהּ! הֵם יַהַרְגוּ אוֹתְךָ וְאוֹתָנוּ!"

חִיֵּךְ הַצַּדִּיק הַזָּקֵן. הָיָה לוֹ רַעְיוֹן גְּאוֹנִי וּמָלֵא בִּמְסִירוּת נֶפֶשׁ. "אֲנַחְנוּ לֹא נַעֲשֶׂה זֹאת בְּתוֹךְ עִיר," הוּא אָמַר, "בּוֹאוּ אַחֲרַי."

הַהֲלִיכָה בֶּהָרִים לְעֵבֶר הַסְּמִיכָה הַחֲשָׁאִית

הַחֲבוּרָה הַקְּטַנָּה, רַב זָקֵן וַחֲמִשָּׁה תַּלְמִידִים עֲנָקִים, יָצְאוּ אֶל הַחֹשֶׁךְ. הֵם הָלְכוּ בְּלִי לַעֲשׂוֹת שׁוּם רַעַשׁ. כַּפּוֹת רַגְלֵיהֶם דָּרְכוּ עַל אֲדָמָה סֶלַעִית וְעַל עֲשָׂבִים קוֹצְנִיִּים. קֹר חָדַר עַד לָעֲצָמוֹת, וְרוּחַ קְרִירָה נָשְׁבָה בְּפָנֵיהֶם. שֶׁמָּא יֵשׁ מְרַגְּלִים רוֹמָאִים בַּסְּבִיבָה? שֶׁמָּא מִישֶׁהוּ יִרְאֶה אוֹתָם?

רַבִּי יְהוּדָה בֶּן בָּבָא הוֹבִיל אוֹתָם לְמָקוֹם נִדָּח וְרָחוֹק מְאֹד. זֶה הָיָה בְּאֶמְצַע שׁוּם מָקוֹם – "בֵּין שְׁנֵי הָרִים גְּדוֹלִים, בֵּין שְׁתֵּי עֲיָרוֹת (אוּשָׁא וּשְׁפַרְעָם), בֵּין שְׁנֵי תְּחוּמֵי שַׁבָּת".

לָמָּה דַּוְקָא שָׁם? כִּי שָׁם זֶה לֹא שַׁיָּךְ לְאַף עִיר! אִם הָרוֹמָאִים יְגַלּוּ אוֹתָם, הֵם לֹא יוּכְלוּ לְהַעֲנִישׁ וּלְהַחֲרִיב שׁוּם יִישּׁוּב שֶׁל יְהוּדִים עַל לֹא עָוֶל בְּכַפָּם. כָּל כָּךְ אָהַב הַצַּדִּיק הַזֶּה אֶת עַם יִשְׂרָאֵל, שֶׁהוּא דָּאַג שֶׁאַף יְהוּדִי חַף מִפֶּשַׁע לֹא יִפָּגַע בִּגְלָלָם!

הַשַּׁחַר הֵחֵל לַעֲלוֹת בַּמִּזְרָח, צוֹבֵעַ אֶת הַשָּׁמַיִם בְּצֶבַע כָּתֹם דַּק. הַצִּפּוֹרִים עֲדַיִן לֹא צִיְּצוּ.

רַבִּי יְהוּדָה בֶּן בָּבָא עָמַד מוּל חֲמֵשֶׁת הַתַּלְמִידִים. הוּא הֵנִיחַ אֶת יָדָיו הַקְּדוֹשׁוֹת עַל רֹאשׁוֹ שֶׁל כָּל אֶחָד וְאֶחָד מֵהֶם. זֶה רֶגַע מַרְטִיט, יְלָדִים. מַשָּׁב הָרוּחַ זִעֲזֵעַ אֶת זְקָנוֹ הַלָּבָן שֶׁל הָרַב. בְּקוֹל רוֹעֵד מִיִּרְאַת שָׁמַיִם וּמִקְּדֻשָּׁה, הוּא אָמַר לְכָל אֶחָד אֶת מִלּוֹת הַקֶּסֶם הָעַתִּיקוֹת שֶׁמַּעֲבִירוֹת אֶת סַמְכוּת הַתּוֹרָה מִדּוֹר לְדוֹר:

"יוֹרֶה יוֹרֶה, יָדִין יָדִין!"

(כְּלוֹמַר – אַתָּה מֻסְמָךְ לְהוֹרוֹת הֲלָכָה לְעַם יִשְׂרָאֵל).

כָּךְ עָבַר תַּלְמִיד אַחַר תַּלְמִיד: רַבִּי מֵאִיר... רַבִּי שִׁמְעוֹן בַּר יוֹחַאי... שׁוֹשֶׁלֶת נֵצַח שֶׁשּׁוּם קֵיסָר לֹא יוּכַל לִשְׁבֹּר בְּחֶרֶב!

קוֹלוֹת שֶׁל בַּרְזֶל וְסוּסִים עֲרָבִים

לְפֶתַע...

טְרָאח! קְלִיק... קְלִיק...

רַעַשׁ שֶׁל עֲשָׂרוֹת פַּרְסוֹת סוּסִים שֶׁשּׁוֹעֲטִים עַל הַסְּלָעִים הֵפֵר אֶת הַדְּמָמָה. רֵיחַ שֶׁל אָבָק מִתְעוֹפֵף מִלֵּא אֶת הַוָּאדִי.

"הָרוֹמָאִים!" זָעַק אֶחָד הַתַּלְמִידִים. "הֵם מָצְאוּ אוֹתָנוּ!"

עֲנַן אָבָק עָנָק סָגַר עֲלֵיהֶם מִכָּל הַכִּוּוּנִים. הַחַיָּלִים הָרוֹמָאִים עִם שִׁרְיוֹנוֹת קַשְׂקַשִּׂים שֶׁהִבְרִיקוּ בָּאוֹר, וּרְמָחִים חַדִּים וַאֲרֻכִּים בִּידֵיהֶם, מִהֲרוּ אֲלֵיהֶם. אֶלֶף קִלְשׁוֹנוֹת וַחֲנִיתוֹת הִסְתַּעֲרוּ אֶל עֵבֶר הַצַּדִּיק וְתַלְמִידָיו. מִישֶׁהוּ מַלְשִׁין הוֹדִיעַ לָהֶם!

רַבִּי יְהוּדָה בֶּן בָּבָא לֹא נִבְהַל. עֵינָיו בָּרְקוּ. הוּא יָדַע שֶׁהַהַשְׁגָּחָה הַפְּרָטִית שֶׁל ה' שָׁמְרָה שֶׁהָרוֹמָאִים יַגִּיעוּ רַק אַחֲרֵי שֶׁהוּא סִיֵּם לִסְמֹךְ אֶת עַמּוּדֵי הַתּוֹרָה! הַמִּשְׂמָה הֻשְׁלְמָה! הַתּוֹרָה הוּצְלָה!

הוּא פָּנָה לַחֲמֵשֶׁת הַתַּלְמִידִים הַצְּעִירִים וְזָעַק מִנַּהֲמַת לִבּוֹ:

"בָּנַי, רוּצוּ! בִּרְחוּ עַל נַפְשְׁכֶם מִיָּד!"

הַתַּלְמִידִים קָפְאוּ בִּמְקוֹמָם. לַעֲזֹב אֶת הָרַב שֶׁלָּהֶם?! הַצַּדִּיק הַזָּקֵן שֶׁלֹּא חָטָא מֵיָמָיו?

"רַבִּי!" זָעֲקוּ, וּדְמָעוֹת זָלְגוּ עַל הַחֲלוֹתֵיהֶם (הַבְּגָדִים שֶׁלָּהֶם). "מָה תְּהֵא עָלֶיךָ?! אֲנַחְנוּ נִשָּׁאֵר אִתְּךָ!"

הַצַּדִּיק הִבִּיט בָּהֶם בְּמַבָּט נֶחְרָץ, אוֹהֵב וּתקיף כְּאֶחָד: "הִנְנִי מֻטָּל לִפְנֵיהֶם כְּאֶבֶן שֶׁאֵין לָהּ הוֹפְכִין! אֲנִי זָקֵן, עָשִׂיתִי אֶת שֶׁלִּי. אַתֶּם – אַתֶּם הֶעָתִיד שֶׁל עַם יִשְׂרָאֵל! לְכוּ, וְהָפִיצוּ אֶת הַתּוֹרָה!"

הַהַקְרָבָה מִלּוֹת קְדֻשָּׁה וָנֶצַח

בְּלֵב נִשְׁבָּר וְדֶמַע צוֹרֵב, סָבוּ הַתַּלְמִידִים וּפָתְחוּ בְּרִיצָה מְהִירָה מְאֹד אֶל מַעֲלֵה הָהָר, מִסְתַּתְּרִים בֵּין הַכֵּפִים.

הָרוֹמָאִים שֶׁהִגִּיעוּ בִּדְהָרָה כְּבָר לֹא הִצְלִיחוּ לְהַשִּׂיג אוֹתָם. הֵם הִשִּׂיגוּ רַק צַדִּיק אֶחָד, זָקֵן, שֶׁעָמַד שָׁלוּ וּמֵאִיר מוּלָם, "כְּאֶבֶן שֶׁאֵין לָהּ הוֹפְכִין".

הַחַיָּלִים שֶׁהָיוּ צְמֵאֵי דָּם זָעֲמוּ עַל כָּךְ שֶׁהַתַּלְמִידִים בָּרְחוּ וְהַגְּזֵרָה הוּפְרָה, עַד שֶׁנָּעֲצוּ בְּגּוּפוֹ הַקָּדוֹשׁ שֶׁל רַבִּי יְהוּדָה בֶּן בָּבָא שְׁלוֹשׁ מֵאוֹת חֲנִיתוֹת שֶׁל בַּרְזֶל וַעֲשָׂאוּהוּ נְקָבִים כִּכְבָרָה (כְּמוֹ רֶשֶׁת). נִשְׁמָתוֹ עָלְתָה בִּסְעָרָה הַשָּׁמַיְמָה עַל קִדּוּשׁ ה'. הוּא אֶחָד מֵ"עֲשָׂרָה הֲרוּגֵי מַלְכוּת".

אֲבָל – הָרוֹמָאִים הִפְסִידוּ! אִמְפֶּרְיַת רוֹמִי הַגְּדוֹלָה וְהַחֲזָקָה כְּבָר מֵתָה וְנִשְׁכְּחָה מִן הָעוֹלָם. אֲבָל חֲמֵשֶׁת הַתַּלְמִידִים שֶׁרַבִּי יְהוּדָה בֶּן בָּבָא הִצִּיל בְּגוּפוֹ – מִלְּאוּ אֶת כָּל אֶרֶץ יִשְׂרָאֵל בְּתוֹרָה! הֵם הֵקִימוּ אֶת יְשִׁיבוֹת הַגָּלִיל, כָּתְבוּ מִשְׁנָיוֹת, וְדַרְכָּם הִגִּיעָה הַתּוֹרָה יָשָׁר אֵלֵינוּ – לַיְּלָדִים שֶׁיּוֹשְׁבִים כָּאן הַיּוֹם בַּכִּתָּה, אַלְפַּיִם שָׁנָה אַחֲרֵי!

א. שְׁאֵלַת פְּתִיחָה (לְעוֹרֵר סַקְרָנוּת):

"יְלָדִים, דַּמְיְנוּ מֵרוֹץ שְׁלִיחִים מָהִיר! מָה מַעֲבִירִים בְּמֵרוֹץ? (מַקֵּל!). נָכוֹן מְאֹד. אֲבָל מָה יִקְרֶה אִם מִישֶׁהוּ יִגְנֹב לָרָץ אֶת הַמַּקֵּל הַזֶּה, וְאִי אֶפְשָׁר יוֹתֵר לְהַעֲבִיר אוֹתוֹ הָלְאָה לַשַּׂחְקָן הַבָּא? אֵיךְ נְנַצֵּחַ אֶת הַמֵּרוֹץ?

"הַיּוֹם אֲסַפֵּר לָכֶם עַל רֶגַע אֶחָד, שֶׁבּוֹ אֹיְבִים רְשָׁעִים נִסּוּ לִגְנֹב לָנוּ אֶת 'הַמַּקֵּל' הֲכִי קָדוֹשׁ בָּעוֹלָם שֶׁעוֹבֵר סָבָא-לְאַבָּא-לְבֵן... הַתּוֹרָה כִּמְעַט נֶעֶצְרָה חָלִילָה. עַד שֶׁקָּם אִישׁ אֶחָד אָמִיץ."

ב. הֶעָרוֹת בִּצּוּעַ וּסְטוֹרִיטֶלִינְג (Storytelling):

ג. הַמְלָצוֹת לַמַּצֶּגֶת (4 שְׁקוּפִיּוֹת מוּמְלָצוֹת - צְנִיעוּת וּקְדֻשָּׁה):

ד. פְּעִילוּת מְסַכֶּמֶת ("שַׁרְשֶׁרֶת הַדּוֹרוֹת"):

מקורות: תלמוד בבלי, מסכת סנהדרין דף י"ד, עמוד א': "פעם אחת גזרה מלכות הרשעה גזירה על ישראל... מה עשה יהודה בן בבא? הלך וישב לו בין שני הרים גדולים ובין שתי עיירות גדולות ובין שני תחומי שבת... וסמך שם חמישה זקנים... אמרו לו: רבי מה תהא עליך? אמר להן: הריני מוטל לפניהם כאבן שאין לה הופכין... נעצו בו שלוש מאות לונביאות של ברזל." תלמוד בבלי, מסכת יבמות דף ס"ב, עמוד ב': אזכור פטירתם של 24,000 תלמידי ר"ע עד שבא לרבותינו שבדרום ולימד את חמשת התלמידים והם העמידו את התורה. (וכן הוזכרו תלמידי ר"ע במסמך [1-14]).

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “להלן עיבוד חינוכי, רגשי ומרתק לסיפור ההיסטורי והמכונן על רבי יהודה בן בבא והסמכת חמשת תלמידי רבי עקיבא .” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “הֵם יָדְעוּ: רַב זָקֵן שֶׁנִּפְטַר וְלֹא הִסְמִיךְ רַב צָעִיר שֶׁיַּמְשִׁיךְ אֶת דַּרְכּוֹ וְיִפְסֹק הֲלָכָה – מְבִיאוֹ לְכָךְ שֶׁהַתּוֹרָה כְּאִלּוּ…”?
ג. כיצד הגיב חמשת תלמידי ר' עקיבא למצב המתואר: “אֲנִי הוֹלֵךְ לִסְמֹךְ אֶתְכֶם (לַעֲשׂוֹת אֶתְכֶם לְרַבָּנִים שֶׁפּוֹסְקִים הֲלָכָה).”?
ד. מה השתנה בין “כָּל כָּךְ אָהַב הַצַּדִּיק הַזֶּה אֶת עַם יִשְׂרָאֵל, שֶׁהוּא דָּאַג שֶׁאַף יְהוּדִי חַף מִפֶּשַׁע לֹא יִפָּגַע…” לבין “הַחַיָּלִים הָרוֹמָאִים עִם שִׁרְיוֹנוֹת קַשְׂקַשִּׂים שֶׁהִבְרִיקוּ בָּאוֹר, וּרְמָחִים חַדִּים וַאֲרֻכִּים…”?
ה. כיצד הגיע הסיפור אל האירוע: “אִמְפֶּרְיַת רוֹמִי הַגְּדוֹלָה וְהַחֲזָקָה כְּבָר מֵתָה וְנִשְׁכְּחָה מִן הָעוֹלָם.”?

3. מה היה ראשון?

לפני ואחרי

מה קרה לפני: “זֶה הָיָה בְּאֶמְצַע שׁוּם מָקוֹם – "בֵּין שְׁנֵי הָרִים גְּדוֹלִים, בֵּין שְׁתֵּי עֲיָרוֹת (אוּשָׁא וּשְׁפַרְעָם), בֵּין שְׁנֵי תְּחוּמֵי שַׁבָּת".”?
מה קרה אחרי: “אֶלֶף קִלְשׁוֹנוֹת וַחֲנִיתוֹת הִסְתַּעֲרוּ אֶל עֵבֶר הַצַּדִּיק וְתַלְמִידָיו.”?

האירוע הנתוןמה גרם לו?מה הייתה תוצאתו?
אֵלּוּ הָיוּ הַתַּלְמִידִים הַיְחִידִים שֶׁשָּׂרְדוּ מִישִׁיבָתוֹ שֶׁל רַבִּי עֲקִיבָא.

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “האיסור לסמוך וסמיכתם ע"י ר' יהודה בן בבא” על חמשת תלמידי ר' עקיבא? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “קֹר חָדַר עַד לָעֲצָמוֹת, וְרוּחַ קְרִירָה נָשְׁבָה בְּפָנֵיהֶם.”.
ג. אילו חמשת תלמידי ר' עקיבא לא היה פועל כפי שמתואר ב־“הַחַיָּלִים הָרוֹמָאִים עִם שִׁרְיוֹנוֹת קַשְׂקַשִּׂים שֶׁהִבְרִיקוּ בָּאוֹר, וּרְמָחִים חַדִּים וַאֲרֻכִּים בִּידֵיהֶם, מִהֲרוּ…” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “אִמְפֶּרְיַת רוֹמִי הַגְּדוֹלָה וְהַחֲזָקָה כְּבָר מֵתָה וְנִשְׁכְּחָה מִן הָעוֹלָם.” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. להלן עיבוד חינוכי, רגשי ומרתק לסיפור ההיסטורי והמכונן על רבי יהודה בן…
2. בְּאוֹתָם יָמִים, קָם סָבָא קָדוֹשׁ וְטָהוֹר, שֶׁשְּׁמוֹ הָיָה רַבִּי…
3. אֲנַחְנוּ לֹא נַעֲשֶׂה זֹאת בְּתוֹךְ עִיר," הוּא אָמַר, "בּוֹאוּ…
4. הוּא הֵנִיחַ אֶת יָדָיו הַקְּדוֹשׁוֹת עַל רֹאשׁוֹ שֶׁל כָּל אֶחָד…
5. הוּא פָּנָה לַחֲמֵשֶׁת הַתַּלְמִידִים הַצְּעִירִים וְזָעַק מִנַּהֲמַת…
6. הַמְלָצוֹת לַמַּצֶּגֶת (4 שְׁקוּפִיּוֹת מוּמְלָצוֹת - צְנִיעוּת…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “אֲנִי הוֹלֵךְ לִסְמֹךְ אֶתְכֶם (לַעֲשׂוֹת אֶתְכֶם לְרַבָּנִים שֶׁפּוֹסְקִים הֲלָכָה).”.
מכתב: כתבו לחמשת תלמידי ר' עקיבא בעקבות “בְּקוֹל רוֹעֵד מִיִּרְאַת שָׁמַיִם וּמִקְּדֻשָּׁה, הוּא אָמַר לְכָל אֶחָד אֶת מִלּוֹת הַקֶּסֶם הָעַתִּיקוֹת שֶׁמַּעֲבִירוֹת אֶת…”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “רַבִּי!" זָעֲקוּ, וּדְמָעוֹת זָלְגוּ עַל הַחֲלוֹתֵיהֶם (הַבְּגָדִים שֶׁלָּהֶם).”.
דיווח: כתבו ידיעה קצרה שכותרתה “האיסור לסמוך וסמיכתם ע"י ר' יהודה בן בבא”.

המחזה

רגע מן הסיפורהדמויותמה יאמרו או יעשו?
