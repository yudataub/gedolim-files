החזרת חמש הסלעים לאבי הבכור

דף עבודה בעקבות רבי טרפון
שם: כיתה:

המעשה — הסיפור המלא

פְּתִיחָה: הַיָּדַיִם שֶׁל אֶלְעָזָר

יְלָדִים, אִם הֱיִיתֶם בּוֹחֲנִים אֶת הַיָּדַיִם שֶׁל אֶלְעָזָר, הֱיִיתֶם מְבִינִים מִיָּד כִּי מְדֻבָּר בְּאָדָם שֶׁעוֹבֵד קָשֶׁה. כַּפּוֹת יָדָיו הָיוּ מְחֻסְפָּסוֹת, מְלֵאוֹת בְּיַבָּלוֹת קָשׁוֹת, וַחֲרִיצִים עֲמֻקִּים עִטְּרוּ אֶת אֶצְבְּעוֹתָיו. אֶלְעָזָר הָיָה אָדָם עָנִי מְאֹד. בְּכָל בֹּקֶר הָיָה קָם עִם צִיּוּץ הַצִּפֳּרִים הָרִאשׁוֹן, עוֹטֵף עַצְמוֹ בִּמְעִילוֹ הַדַּק שֶׁלֹּא מַמָּשׁ חִמֵּם מִפְּנֵי הָרוּחַ הַקָּרָה שֶׁל הַבֹּקֶר, וְיוֹצֵא לַחְטֹב עֵצִים וְלִסְחֹב אֲבָנִים כְּבֵדוֹת.

אַךְ בַּיָּמִים הָאַחֲרוֹנִים, הָרוּחַ שֶׁל אֶלְעָזָר לֹא הָיְתָה כְּבֵדָה כְּמוֹ הָאֲבָנִים שֶׁסָּחַב. לְהֵפֶךְ! לִבּוֹ רָקַד מִשִּׂמְחָה שֶׁמִּלְּאָה אֶת חָזֵהוּ. בַּבַּיִת הַקָּטָן וְהַדַּל שֶׁלּוֹ, בְּתוֹךְ עֲרִיסַת קַשׁ פְּשׁוּטָה, יָשַׁן לוֹ תִּינוֹק קָטָן, רַךְ וּמָתוֹק. בֵּן בְּכוֹר! הַקָּדוֹשׁ בָּרוּךְ הוּא זִכָּה אֶת אֶלְעָזָר וְאִשְׁתּוֹ בְּבֵן בְּכוֹר.

חִסָּכוֹן שֶׁל מְסִירוּת נֶפֶשׁ

אֶלְעָזָר יָדַע אֶת צִוּוּי הַתּוֹרָה. בֵּן בְּכוֹר שַׁיָּךְ לַה', וְכַאֲשֶׁר הַתִּינוֹק מַגִּיעַ לְגִיל שְׁלוֹשִׁים יוֹם, חַיָּב הָאָב לִפְדּוֹת אוֹתוֹ (לָתֵת דְּמֵי פִּדְיוֹן) מִידֵי הַכֹּהֵן, כְּדֵי שֶׁהַתִּינוֹק יִהְיֶה שֶׁלּוֹ. כַּמָּה צָרִיךְ לְשַׁלֵּם לַכֹּהֵן? הַתּוֹרָה קָבְעָה מְחִיר: חֲמִשָּׁה סְלָעִים . סֶלַע הִיא מַטְבֵּעַ כֶּסֶף טָהוֹר, נוֹצֵץ וְכָבֵד. חֲמִשָּׁה סְלָעִים הָיוּ סְכוּם עָצוּם שֶׁל כֶּסֶף עֲבוּר אָדָם שֶׁאֵין לוֹ אֲפִלּוּ לֶחֶם טָרִי!

אֵיךְ יַשִּׂיג אֶלְעָזָר חֲמִשָּׁה סְלָעִים מִכֶּסֶף טָהוֹר?

בְּמֶשֶׁךְ חֹדֶשׁ שָׁלֵם, אֶלְעָזָר כִּמְעַט וְלֹא אָכַל. כְּשֶׁהַפּוֹעֲלִים הָאֲחֵרִים יָשְׁבוּ בַּצָּהֳרַיִם וְאָכְלוּ אֲרוּחָה חַמָּה, אֶלְעָזָר הוֹצִיא מִכִּיסוֹ רַק קְצָת קֶמַח חָרֵב, עִרְבֵּב עִם מַיִם מֵהַבְּאֵר, לָעַס וּבָלַע בְּקֹשִׁי כְּדֵי שֶׁיִּהְיֶה לוֹ מְעַט כֹּחַ לְהַמְשִׁיךְ לַעֲבֹד. רֵיחַ תַּבְשִׁילֵי הַבָּשָׂר שֶׁל הָאֲנָשִׁים בַּשּׁוּק עָלָה בְּאַפּוֹ, וְהַבֶּטֶן שֶׁלּוֹ קִרְקְרָה מֵרָעָב, אַךְ הוּא קָפַץ אֶת יָדָיו, חִיֵּךְ לַשָּׁמַיִם וְלָחַשׁ: "רִבּוֹנוֹ שֶׁל עוֹלָם, הַכֹּל כְּדֵי לְקַיֵּם אֶת הַמִּצְוָה שֶׁלְּךָ הֲכִי יָפֶה שֶׁאֶפְשָׁר!". בְּכָל יוֹם שָׂם עוֹד פְּרוּטָה בְּכִיסוֹ, הֵמִיר אוֹתָן לְכֶסֶף צָרוּף, עַד שֶׁבַּיּוֹם הַשְּׁלוֹשִׁים לְהֻלֶּדֶת בְּנוֹ... קִשְקְשׁוּ בְּתוֹךְ שַׂקִּית בַּד קְטַנָּה חֲמִשָּׁה סְלָעִים. אֲמִתִּיִּים. טְהוֹרִים.

הַהֲלִיכָה לְבֵיתוֹ שֶׁל עֲנַק הַתּוֹרָה

אֶלְעָזָר לָבַשׁ אֶת בִּגְדֵי הַשַּׁבָּת שֶׁלּוֹ שֶׁהָיוּ נְקִיִּים אַךְ דַּהֲוּיִים, נָטַל אֶת הַתִּינוֹק עַל כַּפָּיו בַּעֲדִינוּת – רֵיחַ שֶׁל אֲבָקַת בֹּשֶׂם שֶׁל תִּינוֹקוֹת מִלֵּא אֶת הָאֲוִיר – וְהָלַךְ לְכִוּוּן בֵּיתוֹ שֶׁל גְּדוֹל הַדּוֹר. לְמִי הָלַךְ? לַתַּנָּא הַקָּדוֹשׁ רַבִּי טַרְפוֹן .

רַבִּי טַרְפוֹן לֹא הָיָה רַק אֶחָד מִגְּדוֹלֵי חַכְמֵי יִשְׂרָאֵל, הוּא גַּם הָיָה כֹּהֵן מְיֻחָס מִזֶּרַע אַהֲרֹן. יוֹתֵר מִכָּךְ – רַבִּי טַרְפוֹן הָיָה אָדָם עָשִׁיר מְאֹד. הָיוּ לוֹ שָׂדוֹת, כְּרָמִים וֶהָמוֹן נְכָסִים.

כַּאֲשֶׁר הִתְקָרֵב אֶלְעָזָר לַבַּיִת הַגָּדוֹל, שָׁמַע מִבַּחוּץ קוֹל שֶׁל מְתִיקוּת אֵין סוֹפִית – קוֹלוֹת שֶׁל תּוֹרָה, שֶׁל רַבִּי טַרְפוֹן וְתַלְמִידָיו הָעוֹסְקִים בְּסֻגְיוֹת הַתַּלְמוּד. הָרֵיחַ בַּחֶדֶר הָיָה שֶׁל גְּוִילֵי קְלָף יְשָׁנִים וּנְרוֹת שֶׁמֶן-זַיִת זַךְ.

אֶלְעָזָר נִכְנַס בְּרַעַד, עוֹמֵד בְּבֹשֶׁת פָּנִים. רַבִּי טַרְפוֹן שֶׁלִּבּוֹ הָיָה רָחָב וְאוֹהֵב אֶת כָּל אֶחָד מִיִּשְׂרָאֵל, עָצַר אֶת הַלִּמּוּד, חִיֵּךְ חִיּוּךְ שֶׁמִּלֵּא אֶת הַחֶדֶר בְּאוֹר וְקָם לִקְרַאת אֶלְעָזָר. "בּוֹא, בְּנִי, בּוֹא. מָה אַתָּה מְבַקֵּשׁ?".

"כָּבוֹד הָרַב", גִּמְגֵּם אֶלְעָזָר בְּקוֹל רוֹעֵד, מַחְזִיק אֶת תִּינוֹקוֹ הֶעָטוּף. "זֶהוּ בְּנִי בְּכוֹרִי. הֵבֵאתִי אֶת הַמַּטְבְּעוֹת. אֲנִי מְבַקֵּשׁ לִפְדּוֹת אוֹתוֹ מִידֵי הַכֹּהֵן כְּדָת מֹשֶׁה וְיִשְׂרָאֵל".

הַמִּצְוָה וְהַמַּטְבְּעוֹת עַל הַשֻּׁלְחָן

רַבִּי טַרְפוֹן לָבַשׁ אֶת בִּגְדֵי הַקֹּדֶשׁ שֶׁלּוֹ. הוּא נָטַל אֶת הַתִּינוֹק לִזְרוֹעוֹתָיו הַחֲזָקוֹת וְהַקְּדוֹשׁוֹת. תַּלְמִידֵי הַיְשִׁיבָה קָמוּ וְהִסְתַּכְּלוּ בְּהִתְרַגְּשׁוּת בַּמַּעֲמָד.

הַשֶּׁקֶט הָיָה מֻחְלָט. רַבִּי טַרְפוֹן הֵישִׁיר מַבָּט עָמֹק וְחַם אֶל הָאָב וְשָׁאַל אוֹתוֹ אֶת הַשְּׁאֵלָה שֶׁשּׁוֹאֲלִים בְּמַעֲמָד זֶה:

"אֶלְעָזָר בְּנִי, בְּמָה אַתָּה חוֹפֵץ יוֹתֵר? לִתֵּן לִי אֶת בִּנְךָ בְּכוֹרְךָ שֶׁלְּךָ, שֶׁהוּא בְּכוֹר לְאִמּוֹ, אוֹ אַתָּה רוֹצֶה לִפְדּוֹתוֹ בַּחֲמִשָּׁה סְלָעִים?".

לִבּוֹ שֶׁל אֶלְעָזָר דָּפַק. הוּא הִכְנִיס אֶת יָדוֹ הַמְיֻבֶּלֶת לְכִיסוֹ הֶעָמֹק, וְהוֹצִיא אֶת הַשַּׂקִּית. הוּא פָּתַח אוֹתָהּ, וְאֶחָד לְאֶחָד, הִנִּיחַ אֶת הַמַּטְבְּעוֹת עַל שֻׁלְחָן הָעֵץ הֶעָבֶה.

קְלִינְג... קְלִינְג... צְלִיל עָמוּם שֶׁל כֶּסֶף זַךְ. הַמַּטְבְּעוֹת הִבְרִיקוּ בְּאוֹר הַנֵּר. זֶה הָיָה הַלֶּחֶם שֶׁלּוֹ. אֵלּוּ הָיוּ חֹדֶשׁ שֶׁל רָעָב וְזֵעָה שֶׁמֻּנָּחִים עַכְשָׁו אֵצֶל אָדָם עָשִׁיר! אַךְ אֶלְעָזָר לֹא הֵסֵס, הוּא אָמַר בְּקוֹל צָלוּל:

"כָּבוֹד הַכֹּהֵן, אֲנִי חָפֵץ לִפְדּוֹת אֶת בְּנִי! וְהֵילָךְ (קַח לְךָ) דְּמֵי פִּדְיוֹנוֹ!".

רַבִּי טַרְפוֹן בֵּרֵךְ, לָקַח אֶת חֲמֵשֶׁת הַסְּלָעִים בְּיָדוֹ, בֵּרֵךְ אֶת הַתִּינוֹק שֶׁיִּגְדַּל לְתוֹרָה וּלְחֻפָּה וּלְמַעֲשִׂים טוֹבִים, וְהֶחְזִירוֹ לִזְרוֹעוֹת אָבִיו.

הַמִּצְוָה הִסְתַּיְּמָה!

מַתְּנַת הַצַּדִּיק שֶׁלֹּא מְבַיֶּשֶׁת

אֶלְעָזָר נָשַׁם לִרְוָחָה. הַמִּצְוָה קֻיְּמָה. אַךְ עַכְשָׁו, בְּדַרְכּוֹ הַבַּיְתָה, הַכִּיס רֵיק. אֵין דְּבַר, קוֹל קָטָן אָמַר לוֹ, הַשֵּׁם יַעֲזֹר.

"רֶגַע אֶחָד, רַבִּי אֶלְעָזָר!" נִשְׁמַע פִּתְאוֹם קוֹלוֹ שֶׁל רַבִּי טַרְפוֹן כְּשֶׁאֶלְעָזָר עָמַד לָצֵאת מִן הַדֶּלֶת.

אֶלְעָזָר סָבַב עַל עֲקֵבוֹ וְחָזַר.

רַבִּי טַרְפוֹן קָם מִמְּקוֹמוֹ, נִגַּשׁ לְאֶלְעָזָר וְעָמַד מַמָּשׁ קָרוֹב אֵלָיו. הוּא בָּחַן אֶת פָּנָיו הָעֲיֵפוֹת שֶׁל הָאָב וְאֶת הַיָּדַיִם הַקָּשׁוֹת. רַבִּי טַרְפוֹן יָדַע הֵיטֵב כִּי מַטְבְּעוֹת אֵלּוּ לָקְחוּ מֵהָאִישׁ הַזֶּה מַאֲמָץ שֶׁל שָׁבוּעוֹת.

רַבִּי טַרְפוֹן לָקַח אֶת שַׂקִּית הַכֶּסֶף, וּבִתְנוּעָה רַכָּה וְעֲדִינָה, פָּתַח אֶת חָפְנוֹ שֶׁל אֶלְעָזָר, וְהִנִּיחַ שָׁם בַּחֲזָרָה אֶת חֲמֵשֶׁת הַסְּלָעִים הַנּוֹצְצִים.

אֶלְעָזָר נִבְהַל! עֵינָיו נִפְעֲרוּ לִרְוָחָה! "כְּבוֹד הָרַב! הַמִּצְוָה פְּסוּלָה? שָׁגִיתִי בְּמַשֶּׁהוּ? לָמָּה הָרַב מַחְזִיר לִי אֶת הַכֶּסֶף? זֶהוּ קֹדֶשׁ לַכֹּהֵן!".

לִטֵּף רַבִּי טַרְפוֹן אֶת כְּתֵפוֹ שֶׁל הָאָב וְאָמַר לוֹ בְּחִיּוּךְ אוֹהֵב, וְעֵינָיו בּוֹרְקוֹת בְּדִמְעָה שֶׁל צַדִּיק עָנָיו:

"לֹא, יַקִּירִי. הַמִּצְוָה נַעֲשְׂתָה בְּהִדּוּר וּבִשְׁלֵמוּת. הַכֶּסֶף הַזֶּה הָפַךְ לִהְיוֹת שֶׁלִּי, לְגַמְרֵי שֶׁלִּי עַל פִּי דִּין תּוֹרָה. אַךְ מִכֵּיוָן שֶׁהַכֶּסֶף עַכְשָׁו שֶׁלִּי, יֵשׁ לִי הָרְשׁוּת לַעֲשׂוֹת בּוֹ מָה שֶׁאֲנִי חָפֵץ... וְאֵין דָּבָר שֶׁאֲנִי חָפֵץ בּוֹ יוֹתֵר, מֵאֲשֶׁר לָתֵת אֶת חֲמֵשֶׁת הַסְּלָעִים הָאֵלּוּ – כְּמַתָּנָה אִישִׁית מִמֶּנִּי – עֲבוּר הַתִּינוֹק הַמָּתוֹק וַעֲבוּר מִשְׁפַּחְתְּךָ. קַח זֹאת בְּאַהֲבָה. עֲשֵׂה בָּזֶה צְרָכִים לַבַּיִת וּקְנֵה מָזוֹן וּבְגָדִים".

אֶלְעָזָר הֵבִיט בַּכֶּסֶף וְאָז בְּרַבִּי טַרְפוֹן. הוּא הֵבִין אֶת הַגְּדֻלָּה הָעֲצוּמָה: רַבִּי טַרְפוֹן עָשָׂה אֶת הַמִּצְוָה כַּהֲלָכָה – הוּא לָקַח אֶת הַכֶּסֶף בִּשְׁבִיל שֶׁתִּהְיֶה פְּדוּת תְּקִינָה לַבֵּן – אַךְ מִיָּד לְאַחַר מִכֵּן נָהַג בְּחֶסֶד לִפְנִים מִשּׁוּרַת הַדִּין וְהֶעֱנִיק אֶת דְּמֵי הַפִּדְיוֹן לָאָב הֶעָנִי כְּדֵי שֶׁלֹּא יִרְעַב!

בְּאוֹתוֹ לַיְלָה הָיְתָה סְעוּדָה גְּדוֹלָה בְּבֵית אֶלְעָזָר. וְיוֹתֵר מִשֶּׁהַמָּזוֹן הִשְׂבִּיעַ אֶת בִּטְנוֹ, הָאַהֲבָה שֶׁל גְּדוֹלֵי יִשְׂרָאֵל הִשְׂבִּיעָה וְהֵאִירָה אֶת נִשְׁמָתוֹ לָנֶצַח.

מקורות: תלמוד בבלי, מסכת בכורות, דף נא עמוד ב: "אמר רב חנינא: היה עומד רבי טרפון ונוטל חמש סלעים ומחזירן לבעלים..." (הגמרא דנה בכך שרבי טרפון היה מחזיר את הכסף כדי לעזור לעניים לאחר קיום המצווה מן התורה). ערוך השולחן, יורה דעה, הלכות פדיון הבן, סימן שה, סעיף ל': המציין את התנהגותו של רבי טרפון כדרך חסד בעיקר כלפי הורים עניים. מדרש ספרי (במדבר, עה) ומסכת קידושין: להמחשת מעמדו הרם של רבי טרפון, חסדו ועושרו ("אביהם של כל ישראל").

1. שאלות ידע מתוך הסיפור

א. בתחילת הסיפור נאמר: “יְלָדִים, אִם הֱיִיתֶם בּוֹחֲנִים אֶת הַיָּדַיִם שֶׁל אֶלְעָזָר, הֱיִיתֶם מְבִינִים מִיָּד כִּי מְדֻבָּר בְּאָדָם שֶׁעוֹבֵד קָשֶׁה.” — מה קרה מיד לאחר מכן?
ב. מדוע התרחש הדבר המתואר במשפט: “הַקָּדוֹשׁ בָּרוּךְ הוּא זִכָּה אֶת אֶלְעָזָר וְאִשְׁתּוֹ בְּבֵן בְּכוֹר.”?
ג. כיצד הגיב רבי טרפון למצב המתואר: “כַּאֲשֶׁר הִתְקָרֵב אֶלְעָזָר לַבַּיִת הַגָּדוֹל, שָׁמַע מִבַּחוּץ קוֹל שֶׁל מְתִיקוּת אֵין סוֹפִית – קוֹלוֹת שֶׁל תּוֹרָה, שֶׁל רַבִּי טַרְפוֹן…”?
ד. מה השתנה בין “רַבִּי טַרְפוֹן הֵישִׁיר מַבָּט עָמֹק וְחַם אֶל הָאָב וְשָׁאַל אוֹתוֹ אֶת הַשְּׁאֵלָה שֶׁשּׁוֹאֲלִים בְּמַעֲמָד זֶה:” לבין “רַבִּי טַרְפוֹן קָם מִמְּקוֹמוֹ, נִגַּשׁ לְאֶלְעָזָר וְעָמַד מַמָּשׁ קָרוֹב אֵלָיו.”?
ה. כיצד הגיע הסיפור אל האירוע: “הַכֶּסֶף הַזֶּה הָפַךְ לִהְיוֹת שֶׁלִּי, לְגַמְרֵי שֶׁלִּי עַל פִּי דִּין תּוֹרָה.”?

לפני ואחרי

מה קרה לפני: “אֲנִי מְבַקֵּשׁ לִפְדּוֹת אוֹתוֹ מִידֵי הַכֹּהֵן כְּדָת מֹשֶׁה וְיִשְׂרָאֵל".”?
מה קרה אחרי: “רַבִּי טַרְפוֹן קָם מִמְּקוֹמוֹ, נִגַּשׁ לְאֶלְעָזָר וְעָמַד מַמָּשׁ קָרוֹב אֵלָיו.”?

סיבה ותוצאה מתוך המעשה

האירוע הנתוןמה גרם לו?מה הייתה תוצאתו?
בְּכָל יוֹם שָׂם עוֹד פְּרוּטָה בְּכִיסוֹ, הֵמִיר אוֹתָן לְכֶסֶף צָרוּף, עַד שֶׁבַּיּוֹם הַשְּׁלוֹשִׁים לְהֻלֶּדֶת בְּנוֹ...
אֵלּוּ הָיוּ חֹדֶשׁ שֶׁל רָעָב וְזֵעָה שֶׁמֻּנָּחִים עַכְשָׁו אֵצֶל אָדָם עָשִׁיר!

4. הבנה והעמקה

המסר העולה מן הסיפור: המידה הטובה והדרך לעבודת השם העולות מן המעשה
א. מה מלמד המעשה “החזרת חמש הסלעים לאבי הבכור” על רבי טרפון? הביאו שתי ראיות מדויקות מן הסיפור.
ב. הסבירו במילים שלכם את המשפט: “כָּבוֹד הָרַב", גִּמְגֵּם אֶלְעָזָר בְּקוֹל רוֹעֵד, מַחְזִיק אֶת תִּינוֹקוֹ הֶעָטוּף.”.
ג. אילו רבי טרפון לא היה פועל כפי שמתואר ב־“רַבִּי טַרְפוֹן קָם מִמְּקוֹמוֹ, נִגַּשׁ לְאֶלְעָזָר וְעָמַד מַמָּשׁ קָרוֹב אֵלָיו.” — כיצד היה הסיפור עשוי להסתיים?
ד. כיצד האירוע “הַכֶּסֶף הַזֶּה הָפַךְ לִהְיוֹת שֶׁלִּי, לְגַמְרֵי שֶׁלִּי עַל פִּי דִּין תּוֹרָה.” מחזק את המסר של הסיפור?

5. קומיקס — אירועים ממשיים מן הסיפור

ציירו כל אירוע והוסיפו מתחתיו משפט מתאים.

1. יְלָדִים, אִם הֱיִיתֶם בּוֹחֲנִים אֶת הַיָּדַיִם שֶׁל אֶלְעָזָר,…
2. בֵּן בְּכוֹר שַׁיָּךְ לַה', וְכַאֲשֶׁר הַתִּינוֹק מַגִּיעַ לְגִיל…
3. הָרֵיחַ בַּחֶדֶר הָיָה שֶׁל גְּוִילֵי קְלָף יְשָׁנִים וּנְרוֹת…
4. הוּא פָּתַח אוֹתָהּ, וְאֶחָד לְאֶחָד, הִנִּיחַ אֶת הַמַּטְבְּעוֹת עַל…
5. רַבִּי טַרְפוֹן יָדַע הֵיטֵב כִּי מַטְבְּעוֹת אֵלּוּ לָקְחוּ מֵהָאִישׁ…
6. וְיוֹתֵר מִשֶּׁהַמָּזוֹן הִשְׂבִּיעַ אֶת בִּטְנוֹ, הָאַהֲבָה שֶׁל…

6. כתיבה ויצירה בעקבות הסיפור

יומן: כתבו מנקודת מבטו של מי שראה את האירוע “כַּאֲשֶׁר הִתְקָרֵב אֶלְעָזָר לַבַּיִת הַגָּדוֹל, שָׁמַע מִבַּחוּץ קוֹל שֶׁל מְתִיקוּת אֵין סוֹפִית – קוֹלוֹת שֶׁל תּוֹרָה, שֶׁל…”.
מכתב: כתבו לרבי טרפון בעקבות “אֵלּוּ הָיוּ חֹדֶשׁ שֶׁל רָעָב וְזֵעָה שֶׁמֻּנָּחִים עַכְשָׁו אֵצֶל אָדָם עָשִׁיר!”.
דו־שיח: חברו שיחה שהייתה יכולה להתרחש רגע לפני “רַבִּי טַרְפוֹן לָקַח אֶת שַׂקִּית הַכֶּסֶף, וּבִתְנוּעָה רַכָּה וְעֲדִינָה, פָּתַח אֶת חָפְנוֹ שֶׁל אֶלְעָזָר, וְהִנִּיחַ שָׁם…”.
דיווח: כתבו ידיעה קצרה שכותרתה “החזרת חמש הסלעים לאבי הבכור”.

המחזה

רגע מן הסיפורהדמויותמה יאמרו או יעשו?
בְּכָל יוֹם שָׂם עוֹד פְּרוּטָה בְּכִיסוֹ, הֵמִיר אוֹתָן לְכֶסֶף צָרוּף, עַד שֶׁבַּיּוֹם…
רַבִּי טַרְפוֹן קָם מִמְּקוֹמוֹ, נִגַּשׁ לְאֶלְעָזָר וְעָמַד מַמָּשׁ קָרוֹב אֵלָיו.

7. חיבור לחיים מתוך המעשה

א. מה אפשר ללמוד מן הדרך שבה רבי טרפון פעל באירוע “רַבִּי טַרְפוֹן הֵישִׁיר מַבָּט עָמֹק וְחַם אֶל הָאָב וְשָׁאַל אוֹתוֹ אֶת הַשְּׁאֵלָה שֶׁשּׁוֹאֲלִים בְּמַעֲמָד זֶה:”?
